"""
Propose a low‑objective step‑function.

The method consists of three phases:

1. **Extensive parametric sweep** – we search a two‑parameter family
   a_i ∝ (x_i·(1−x_i)+ε)^{−α} possibly mixed with a uniform component.
   The sweep covers a wide range of α (0.2–0.6) and several ε/λ values.
   The best candidate is kept as a starting point.

2. **Sub‑gradient Adam refinement** – the exact objective
   2·n·max(conv)/sum² is non‑smooth because of the max.  We use the
   sub‑gradient given by the active convolution index (or indices) and
   update the vector with Adam, projecting onto the probability simplex
   after each step.

3. **Light simulated‑annealing polish** – random mass‑transfer moves
   (source ↔ destination) are performed while the time budget allows;
   any improvement is kept.

All operations are based on FFTs for O(n log n) speed, and the whole
search respects the 1000 s wall‑time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the harness:
       2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) with a very low
    evaluation score under the given time budget.  Returns a plain Python
    list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # Problem size – a power of two works best with the FFTs.
    # 8192 gives a finer discretisation than the previous 4096 while staying cheap.
    n = 8192

    # ------------------------------------------------------------------
    # 1) Wide parametric sweep (Beta‑like family + uniform mixture)
    # ------------------------------------------------------------------
    x = (np.arange(n) + 0.5) / n                # mid‑bin positions in (0,1)
    prod = x * (1.0 - x)                         # x·(1−x)

    best_seq = None
    best_score = float("inf")

    alphas = np.linspace(0.20, 0.60, 81)         # step ≈0.005
    epsilons = [0.0, 1e-12, 1e-9, 1e-6, 1e-4, 1e-2]
    mix_lambdas = [0.0, 0.05, 0.10, 0.15, 0.20] # weight of the uniform component

    # pre‑compute a uniform vector for mixing
    uniform_vec = np.full(n, 1.0 / n)

    for alpha in alphas:
        # early exit if we run out of time
        if time.time() >= deadline:
            break
        for eps in epsilons:
            # base shape (no mixture)
            w = (prod + eps) ** (-alpha)
            # normalise once – the mixture will be re‑normalised automatically
            w_norm = w / w.sum()
            for lam in mix_lambdas:
                # convex combination with the uniform distribution
                a = (1.0 - lam) * w_norm + lam * uniform_vec
                # the mixture already sums to 1, but guard against tiny drift
                a = a / a.sum()
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep α={alpha:.3f}, eps={eps:e}, λ={lam:.2f} => {score:.6f}")

    # ------------------------------------------------------------------
    # 2) If a global starter exists (e.g. from a previous run) use it
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
                        print(f"[{time.time() - start_time:.2f}s] starter improved to {cur_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Sub‑gradient Adam refinement (exact max‑convolution)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)   # fallback – uniform

    a = best_seq.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.04                     # a bit larger than the previous code

    # FFT length for the correlation (w * reversed a)
    L_corr = _next_pow2(3 * n)

    iteration = 0
    while time.time() < deadline - 2.0:   # leave a couple of seconds for the polish
        iteration += 1

        # ---- self‑convolution (FFT) ------------------------------------------------
        L_conv = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(a, n=L_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L_conv)
        conv = conv_full[: 2 * n - 1]                       # length 2n‑1
        max_conv = conv.max()

        # ---- identify active indices (the argmax of the convolution) ---------------
        # use a tiny tolerance to collect all practically equal maxima
        tol = max_conv * 1e-12 + 1e-16
        active = np.where(conv >= max_conv - tol)[0]

        # ---- build weight vector w (indicator of active indices, normalised) --------
        w_vec = np.zeros(2 * n - 1, dtype=np.float64)
        w_vec[active] = 1.0 / len(active)

        # ---- gradient of the exact (unsmoothed) objective --------------------------
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w_vec, n=L_corr)
        a_rev_fft = np.fft.rfft(a_rev, n=L_corr)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L_corr)
        # extract the part that corresponds to ∂(w·conv)/∂a, length n
        Ba = corr_full[n - 1: n - 1 + n]
        gradient = 4.0 * n * Ba   # ∇F ≈ 4·n·B·a (sum constraint handled by projection)

        # ---- Adam update -----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * gradient
        v = beta2 * v + (1.0 - beta2) * (gradient ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)

        # ---- keep the best exact objective seen so far ----------------------------
        if iteration % 40 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={iteration:,}  best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Very light simulated‑annealing polish (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()

        total_moves = 35_000
        for t in range(1, total_moves + 1):
            if time.time() >= deadline:
                break
            # source chosen proportionally to current mass (ensures we move something)
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue
            # transfer up to 15 % of the source mass
            delta = a[src] * rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta

            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={t:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
