"""
Candidate generator for the step‑function minimisation problem.
--------------------------------------------------------------
The algorithm consists of three stages.

1️⃣  **Multiple random restarts with smooth‑max mirror descent**  
    – a is kept on the probability simplex (via multiplicative updates).  
    – the non‑smooth max is approximated by a soft‑max with a schedule of
       increasing smoothing parameters α.  
    – a simple momentum / Adam‑style step‑size schedule is used.
    – several short restarts (including the provided best‑known sequence
       and random perturbations) are run; the best result is kept.

2️⃣  **Projected sub‑gradient refinement** – classical sub‑gradient descent
    on the exact objective with a 1/√t step‑size and Euclidean projection
    onto the simplex.

3️⃣  **Greedy hill‑climbing polish** – tiny mass‑transfer moves are tried,
    always keeping the move if it improves the true objective.

All phases stop promptly when the wall‑clock deadline (≈ 1000 s) is reached.
The final sequence is normalised to unit sum, guaranteeing that the
external ``evaluate_sequence`` routine receives a valid candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear convolution of two real vectors u and v using FFT.
    Returns the full linear convolution (len(u)+len(v)-1 elements).
    """
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum(a)≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max mirror descent (exponentiated gradient with Adam‑style steps)
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray,
                    n: int,
                    L_self: int,
                    L_grad: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 200_000,
                    alpha_start: float = 0.5,
                    alpha_end: float = 800.0,
                    n_alpha: int = 30) -> tuple[np.ndarray, float]:
    """
    Perform a few hundred thousand exponentiated‑gradient steps while
    gradually sharpening the soft‑max approximation of the max‑convolution.
    The update rule uses Adam‑style moment estimates on the log‑domain.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)               # stay in log‑space to avoid underflow
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    # schedule for the smoothing parameter alpha (geometric)
    alphas = np.geomspace(alpha_start, alpha_end, num=n_alpha)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    mt = np.zeros_like(log_a)
    vt = np.zeros_like(log_a)

    it = 0
    for alpha in alphas:
        # inner loop – stop when time or iteration budget exhausted
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # ---- 1) current distribution on the simplex
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum

            # ---- 2) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # ---- 3) soft‑max weights w_k ∝ exp(alpha * conv_k)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stability shift
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # ---- 4) gradient of the smooth max (w.r.t. a)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

            # ---- 5) chain rule to log‑domain (∂J/∂log_a = a·(grad_a - <a,grad_a>))
            dot = np.dot(a, grad_a)
            grad_log = a * (grad_a - dot)

            # ---- 6) Adam update on log_a
            mt = beta1 * mt + (1.0 - beta1) * grad_log
            vt = beta2 * vt + (1.0 - beta2) * (grad_log ** 2)
            m_hat = mt / (1.0 - beta1 ** it)
            v_hat = vt / (1.0 - beta2 ** it)
            step = 0.05 / (np.sqrt(v_hat) + eps)   # base learning rate 0.05
            log_a -= step * m_hat

            # occasional true‑objective check
            if it % 1000 == 0:
                cur_score = _eval_fast(np.exp(log_a) / np.exp(log_a).sum(), L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = np.exp(log_a) / np.exp(log_a).sum()
                    print(f"[{time.time() - start_time:.2f}s] α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    # final normalisation
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(start_a: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient method with Euclidean projection onto the simplex.
    Uses a 1/√t step‑size and quits early if no improvement for many steps.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = init_best_score
    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        if stagnation >= 20_000:
            break

        # active indices (those attaining the maximum, tolerance needed)
        tol = 1e-14
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of max‑convolution (sum of contributions from active sums)
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = 0.3 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_seq: np.ndarray,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves: pick a source proportional to its mass,
    a destination uniformly, move a random fraction of the source mass,
    keep the move only if it strictly improves the exact objective.
    """
    a = start_a.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.choice(n, p=a)               # source weighted by current mass
        dst = rng.integers(0, n)               # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.

    The search consists of several short random restarts of a smooth‑max
    mirror‑descent phase, a sub‑gradient refinement, and a final hill‑climb
    polishing step.  The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # length needed for smooth‑max grad

    # ------------------------------------------------------------------
    # Load an initial vector (global best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adapt length if necessary
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑0 : several short smooth‑max mirror‑descent restarts
    # ------------------------------------------------------------------
    n_restarts = 5
    # allocate roughly a third of the remaining time to these restarts
    time_for_restarts = (deadline - time.time()) * 0.35
    restart_deadline = start_time + time_for_restarts
    per_restart_deadline = restart_deadline / n_restarts

    for r in range(n_restarts):
        # pick a start point: first restart uses the current best,
        # others use random perturbations of the best
        if r == 0:
            a0 = best_seq.copy()
        else:
            # random Dirichlet‑like perturbation around the best sequence
            noise = np.random.lognormal(mean=0.0, sigma=0.5, size=n)
            a0 = best_seq * noise
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            if a0_sum == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0_sum

        # run a short smooth descent for this restart
        this_deadline = min(deadline - 1.0, time.time() + per_restart_deadline)
        seq, sc = _smooth_descent(a0, n, L_self, L_grad,
                                  this_deadline, start_time,
                                  max_iters=150_000,
                                  alpha_start=0.5,
                                  alpha_end=800.0,
                                  n_alpha=25)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {r+1}/{n_restarts} improved score = {sc:.6f}")

        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Phase‑1 : projected sub‑gradient refinement (fast polish)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             L_self, sub_deadline,
                                             start_time, best_score,
                                             max_iters=400_000)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : greedy hill‑climbing polish (tiny mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _hill_climb(best_seq, n,
                                        L_self, deadline,
                                        start_time,
                                        best_seq, best_score)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
