"""
A fast two‑stage optimiser for the step‑function objective.

Stage 1 – a dense parametric sweep of Beta‑type families
          a_i ∝ (x_i·(1−x_i)+ε)^{−α}  (α∈[0.20,0.70], ε∈{0,1e‑8,…}),
          evaluated with an FFT‑based exact objective.

Stage 2 – a smooth Adam optimiser that minimises a log‑sum‑exp
          surrogate of  max(conv).  The smoothing β is increased
          from 5 to 1000 so that the surrogate converges to the true
          max.  The iterate is kept on the probability simplex.

Stage 3 – a cheap sub‑gradient descent on the true max‑convolution.
          The lag attaining the maximum is identified, the
          corresponding gradient (a shifted copy of the vector) is
          taken with a small step and the point is projected back onto
          the simplex.  This “peak‑shaving’’ step frequently drops the
          objective a few thousandths.

The whole routine respects the 1000 s wall‑time limit, works with a
single 8192‑point vector (large enough to capture the continuous
optimum) and finally returns a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed to be already on the simplex, i.e.
    sum(seq)≈1, therefore the denominator is ≈1.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _smooth_gradient(a: np.ndarray, beta: float, L: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        F_β(a) = (2·n/β)·log( Σ_k exp(β·conv_k) )
    Returns ∇F_β(a) (shape n).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    # soft‑max weights
    b = beta * conv
    b_max = b.max()
    w = np.exp(b - b_max)
    w_sum = w.sum()
    w /= w_sum                       # shape (2n‑1,)

    # B·a = correlation of w with reversed a
    a_rev = a[::-1]
    w_fft = np.fft.rfft(w, n=L)
    a_rev_fft = np.fft.rfft(a_rev, n=L)
    corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
    Ba = corr_full[n - 1 : n - 1 + n]     # size n
    grad = 4.0 * n * Ba                  # ∇F_β = 4·n·B·a
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with minimal evaluation score.
    Returns a plain list of floats (the sequence is automatically normalised).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a few seconds as a safety margin for the harness.
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Problem size – a power of two works well with the FFTs.
    # ------------------------------------------------------------------
    n = 8192                           # larger than the previous 4096

    # ------------------------------------------------------------------
    # 2) Fine sweep over a richer Beta‑family (α,ε).
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions
    best_seq = None
    best_score = float("inf")

    alphas = np.linspace(0.20, 0.70, 101)           # wide sweep, fine grid
    epsilons = [0.0, 1e-8, 1e-6, 1e-4, 1e-2]       # a few edge‑smoothings

    for alpha in alphas:
        for eps in epsilons:
            # avoid division by zero when eps=0 and x=0 or 1
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        # early exit if we have run out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3) Accept an external starter if present.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                total = starter.sum()
                if total > 0:
                    starter = starter / total
                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4) Adam optimisation on the smooth surrogate.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)          # safe fallback

    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.02

    # Length for the two FFTs needed in the smoothed gradient.
    L = _next_pow2(3 * n)

    iteration = 0
    beta_start, beta_end = 5.0, 1000.0          # stronger final smoothing

    while time.time() < deadline - 5.0:        # leave ~5 s for the final polishing
        iteration += 1

        # ----------- smoothing parameter (linear schedule) ----------
        elapsed = time.time() - start_time
        prop = min(1.0, elapsed / (budget_s - 5.0))
        beta = beta_start + (beta_end - beta_start) * prop

        # ----------- compute smoothed gradient ----------------------
        grad = _smooth_gradient(a, beta, L)

        # ----------- Adam update ------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)

        # ---------- keep the best exact objective seen so far -----
        if iteration % 100 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{elapsed:.2f}s] Adam iter={iteration:,} best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Sub‑gradient descent on the true max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        sub_iter = 0
        # initial learning rate – will decay with √t
        base_lr_sub = 0.04

        while time.time() < deadline - 0.5:
            sub_iter += 1

            # compute the (linear) convolution via a single FFT
            a_fft = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            k_max = int(conv.argmax())
            max_val = conv[k_max]

            # map k to the smallest offset (0…n‑1) using symmetry
            if k_max > n - 1:
                k = 2 * n - 2 - k_max
            else:
                k = k_max

            # gradient of that particular lag (sub‑gradient of the max)
            g = np.zeros_like(a)
            if k > 0:
                # contributions a_i * a_{i+k}
                g[: n - k] += a[k:]
                g[k:] += a[: n - k]
            else:                # k == 0  → derivative of sum of squares
                g = 2.0 * a

            # simple decaying step size
            lr = base_lr_sub / math.sqrt(sub_iter)
            a = a - lr * g
            a = _project_simplex(a)

            # occasional exact evaluation
            if sub_iter % 200 == 0:
                cur_score = _objective(a)
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:.2f}s] Sub‑grad step={sub_iter:,} best={cur_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] After sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Light random mass‑transfer polish (optional, very cheap).
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.1:
        a = best_seq.copy()
        cur_score = _objective(a)
        rng = np.random.default_rng()
        moves = 15_000
        for t in range(moves):
            if time.time() >= deadline:
                break
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps = a[src] * rng.random() * 0.08   # move ≤8 %
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] polish step={t:,} best={new_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Normalise and return.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
