    u = np.sort(v)[::-1]

    ind = np.arange(len(v)) + 1

def propose_candidate(seed=0, budget_s=1000):

    deadline = start + budget_s - 0.2

    # load previous best seque if present
    init_seq = None
    try:
        init_seq_np = np.asarray(height_sequence_1, dtype=float)
        if init_seq_np.ndim==1 and init_seq_np.size>0:
            init_seq = init_seq_np
    except NameError:
        pass

    # function to get normalized vector length n.
    def normalize(v):
        v = np.maximum(v,0)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.pad(v, (0,n-v.size), mode='constant')
        s = v.sum()
        if s>0:
            v = v/s
        else:
            v = np.full(n,1.0/n)
        return v

    # create list of initial vectors
    inits = []

    if init_seq is not None:

    for _ in range(3):
        dir_alpha = np.random.rand(n) * 0.5 + 0.01
        rnd = np.random.gamma(dir_alpha, 1.0)
        inits.append(normalize(rnd))

    best_seq = None

    # allocate time for each init based on total budget fraction
    # use 70% for smoothing descent for each init equally
    total_alloc = deadline - start
    # allocate across each init: time_per = total_alloc * 0.65 / len(inits)
    time_per = total_alloc * 0.65 / len(inits)
    # in case of less than needed time for each, we adapt.
    for idx, init_a in enumerate(inits):
        sub_deadline = min(deadline-0.05, start + (idx+1)*time_per)
        a, sc = _smooth_descent(init_a, n, sub_deadline, start)
        if sc < best_score:

"""
Improved search for the minimum of

    2 * n * max(conv(a, a)) / (sum(a)**2)

for a fixed length n = 4096.

The algorithm consists of three stages

1️⃣  A quick parametric sweep (beta‑shaped, sinusoidal) that often
    lands very close to the optimum.

2️⃣  An extended exponentiated‑gradient (smooth‑max mirror descent)
    with a finer schedule of smoothing parameters and a longer
    budget (≈ 70 % of the time).  The updates stay on the probability
    simplex and use a per‑iteration stepsize that decays with √t
    and with the smoothing strength.

3️⃣  A projected sub‑gradient descent (with momentum) for a precise
    polish until the time runs out.

All stages keep track of the best vector seen so far and finally
return a normalised non‑negative list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a·a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _simplex_project(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation of the algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweep – beta‑shaped & sinusoidal families
# ----------------------------------------------------------------------
def _parametric_sweep(a0: np.ndarray, n: int, L_self: int) -> tuple[np.ndarray, float]:
    """Search a few low‑dimensional families (beta‑shape, sin‑shape)."""
    best_seq = a0.copy()
    best_score = _eval_fast(best_seq, L_self)

    # beta‑shape: a_i ∝ (i+1)^α * (n-i)^β
    alphas = np.linspace(-0.5, 2.0, 13)   # 13 values → reasonable coverage
    betas = np.linspace(-0.5, 2.0, 13)
    idx = np.arange(n, dtype=float)

    for α in alphas:
        for β in betas:
            seq = ((idx + 1) ** α) * ((n - idx) ** β)
            seq = np.maximum(seq, 0.0)
            seq_sum = seq.sum()
            if seq_sum == 0.0:
                continue
            seq /= seq_sum
            sc = _eval_fast(seq, L_self)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                # occasional print
                print(f"[sweep] beta(α={α:.2f},β={β:.2f}) score={sc:.6f}")

    # sinusoidal family: a_i ∝ sin(π·(i+0.5)/n)^p
    powers = np.linspace(0.5, 4.0, 15)
    base = np.sin(np.pi * (idx + 0.5) / n)
    for p in powers:
        seq = base ** p
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            continue
        seq /= total
        sc = _eval_fast(seq, L_self)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[sweep] sin^p(p={p:.2f}) score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth‑max approximation
    of the max‑convolution.  Uses a geometric schedule of smoothing
    parameters α that ends at a very large value (≈ 4000) to tighten
    the approximation.
    """
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # schedule – more aggressive than the original code
    alphas = np.geomspace(0.2, 4000.0, num=40)

    it = 0
    max_total_iters = 2_000_000               # absolute cap
    for alpha in alphas:
        # inner loop runs as long as we have time left and haven’t hit the cap
        while time.time() < deadline - 0.05 and it < max_total_iters:
            it += 1

            # 1) self‑convolution (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights (numerically stable)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            #    step size slowly decays both with iteration count and with α
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            # avoid exact zeros (helps stability of the next log‑step)
            a = np.maximum(a, 1e-20)
            a /= a.sum()

            # occasional objective evaluation
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"α={alpha:.1f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.05:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Projected sub‑gradient descent (with momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the exact max‑convolution.
    Uses a simple momentum term and a stepsize ηₜ = η₀ / √t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    it = 0
    L_fft = _next_pow2(3 * n - 2)

    while time.time() < deadline - 0.05:
        it += 1

        # full self‑convolution (FFT, un‑padded)
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

        # active shift (first occurrence of the max)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑convolution (see comment in the starter code)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / math.sqrt(it)
        a = a - eta * momentum

        # simplex projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score
