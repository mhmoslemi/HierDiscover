def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:           # numerically impossible, but keep safe guard
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT‑based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) with a very low
    evaluation score under the given time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a short safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two works well with the FFTs
    # ------------------------------------------------------------------
    n = 8192                     # larger than the previous 4096

    # ------------------------------------------------------------------
    # 1) Fine sweep over the Beta‑family (α, ε)
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions
    best_seq = None
    best_score = float("inf")

    alphas = np.linspace(0.40, 0.60, 81)            # step ≈0.0025
    epsilons = [0.0, 1e-8, 1e-6, 1e-4, 1e-3, 1e-2]

    for alpha in alphas:
        for eps in epsilons:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Optional global starter (if the harness provided one)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    a = best_seq.copy()
    best_seq = a.copy()               # ensure we keep the best seen
    best_score = _objective(a)

    # parameters for the descent
    lr0 = 0.05                        # initial step‑size
    decay = 0.5                       # sqrt‑decay factor
    max_iter = 20000                  # absolute cap – will be cut by time

    L = _next_pow2(2 * n - 1)         # FFT length (used every iteration)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline - 2.0:
            break

        # ----- convolution and max‑index -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        idx_max = int(conv.argmax())

        # ----- sub‑gradient of max(conv) w.r.t. a -----
        # g_i = 2 * a[idx_max - i] for indices where the shift stays inside [0,n)
        j = np.arange(n)
        shift_idx = idx_max - j
        mask = (shift_idx >= 0) & (shift_idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[shift_idx[mask]]

        # ----- step‑size (decaying) -----
        lr = lr0 / (np.sqrt(it) * decay + 1.0)

        # ----- gradient step & projection -----
        a_next = a - lr * grad
        a_next = _project_simplex(a_next)

        # ----- keep the best exact objective -----
        cur_score = _objective(a_next)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_next.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] iter={it:,}  best={cur_score:.6f}")

        a = a_next

    # ------------------------------------------------------------------
    # 4) Very light simulated‑annealing polish (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_moves = 50000

        for t in range(1, total_moves + 1):
            if time.time() >= deadline:
                break
            # source chosen proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-16:
                continue
            # move a modest fraction (≤ 5 %)
            eps = a[src] * rng.random() * 0.05
            a_candidate = a.copy()
            a_candidate[src] -= eps
            a_candidate[dst] += eps

            new_score = _objective(a_candidate)
            if new_score < cur_score:
                a = a_candidate
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:.2f}s] anneal step={t:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
