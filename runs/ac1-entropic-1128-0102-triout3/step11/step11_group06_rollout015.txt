        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        C = conv.max()
        kmax = conv.argmax()
        S = a.sum()
        # compute gradient of C:
        grad_C = np.zeros_like(a)
        i_start = max(0, kmax - (n-1))

"""
Propose a low‑objective non‑negative step sequence.

Algorithm
---------

1. **Broad Beta‑family sweep** –  
   Scan many (α,ε) pairs for the family  
   a_i ∝ (x_i·(1−x_i)+ε)^{−α} (x_i = (i+½)/n).  
   The best candidate is kept as a warm start.

2. **Fast sub‑gradient descent on the exact objective** –  
   The objective is   F(a)=2·n·max(a∗a)/(∑a)^2.
   For a normalised vector (∑a=1) the gradient of the
   active term C = max(a∗a) is
       ∂C/∂a_i = 2·a_{k*−i}   (k* = argmax(a∗a), 0≤k*<2n−1)
   and therefore
       ∇F = 2·n·(∂C/∂a − 2C·1).

   The gradient is cheap to compute (one FFT for the convolution,
   O(n) work for the sub‑gradient) and we apply an Adam style
   update followed by Euclidean projection onto the simplex.
   Hundreds of thousands of iterations fit easily in the 1000 s budget.

3. **Light greedy mass‑transfer polish** –  
   After the Adam phase we perform many tiny “move ε from i to j”
   proposals and keep any improvement.  This finishes the search
   with a deterministic hill‑climb that often squeezes a few
   extra‑digits out of the bound.

All operations stay on the probability simplex (non‑negative,
sum = 1) so the evaluation function is scale‑invariant and never
rejects the returned sequence.  The code respects the 1000 s wall‑time
limit and prints occasional progress reports.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w≥0, Σw = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The implementation uses an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _max_conv_and_gradient(a: np.ndarray):
    """
    Return (C, k, gradC) where
        C   = max(a∗a)
        k   = index of the maximum (first one if several)
        gradC = ∂C/∂a   (sub‑gradient of the max term)
    The function also returns the full convolution array for occasional use.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    C = conv.max()
    k = int(conv.argmax())                     # first argmax

    # gradient of C:  ∂C/∂a_i = 2·a_{k-i} if 0≤k-i<n, else 0
    gradC = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    gradC[idx] = 2.0 * a[k - idx]
    return C, k, gradC, conv

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    the evaluation score under the given time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size (a power of two works best with the FFTs)
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Broad sweep over the Beta‑family (α, ε)
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions
    best_seq = None
    best_score = float("inf")

    alphas = np.linspace(0.10, 0.90, 81)            # finer coverage
    epsilons = [0.0, 1e-9, 1e-6, 1e-4, 1e-2]

    for alpha in alphas:
        for eps in epsilons:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()                         # normalised to sum=1
            score = _objective(a)                  # cheap: one FFT
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) If a previous construction exists, consider it as well
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                s = starter.sum()
                if s > 0:
                    starter = starter / s
                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Fast sub‑gradient descent (Adam) on the exact objective
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)

    a = best_seq.copy()
    # Adam optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 5e-5                     # small step (scaled inside Adam)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --- convolution and sub‑gradient of the active max term ----------
        C, k_max, gradC, conv = _max_conv_and_gradient(a)

        # (the vector is always kept normalised → sum = 1)
        # Gradient of the full objective:
        #   ∇F = 2·n·(gradC - 2·C·1)
        scale = 2.0 * n                    # sum(a)=1 ⇒ 1/(sum^2)=1
        coeff = 2.0 * C
        grad = scale * (gradC - coeff * np.ones_like(a))

        # ---- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = _project_simplex(a_new)       # stay on the simplex

        # keep the best exact objective seen so far
        # we can obtain it directly from C (since sum=1)
        cur_score = 2.0 * n * C
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] iter={iteration:,}  best={best_score:.6f}")

        a = a_new

        # leave a couple of seconds for the final polish
        if time.time() >= deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # 4) Greedy mass‑transfer polishing (hill climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = 2.0 * n * _max_conv_and_gradient(a)[0]   # exact
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        moves = 40_000

        for t in range(1, moves + 1):
            if time.time() >= deadline:
                break

            # choose a source proportional to its mass (more impact)
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            # move a modest fraction (≤5 %) of the source mass
            delta = a[src] * rng.random() * 0.05
            a_candidate = a.copy()
            a_candidate[src] -= delta
            a_candidate[dst] += delta

            # evaluate exact objective (fast – reuses the FFT routine)
            cand_score = _objective(a_candidate)
            if cand_score < cur_score - 1e-9:       # accept only true improvements
                a = a_candidate
                cur_score = cand_score
                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_local = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:.2f}s] anneal step={t:,}  best={cand_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
