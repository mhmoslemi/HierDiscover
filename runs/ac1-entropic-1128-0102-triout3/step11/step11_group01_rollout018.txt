def _adam_opt(a0, deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps=1e-8
    step0 = 0.5
    t=0
    best_a = a.copy()
    best_score = _objective(a)
    while time.time()<deadline:
        t+=1
        # compute conv and max
        conv = _conv_self(a)
        cur_score = 2.0 * a.shape[0] * conv.max() / (a.sum()**2)   # but _objective already
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if (t%10000)==0: print progress

        max_val = conv.max()
        # active set indices within some tolerance, relative and absolute
        tol = 1e-12 * max_val
        active = np.where(conv >= max_val - tol)[0]  # might be 1 element
        # accumulate gradient
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (a.shape[0] - 1))
            hi = min(a.shape[0] - 1, k)
            idx = np.arange(lo, hi+1)
            grad[idx] += 2.0 * a[k - idx]
        if len(active) > 0:
            grad /= len(active)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / np.sqrt(t)   # we can incorporate but Adam already adapts
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_simplex(a)

    return best_a, best_score

"""
Propose a step‑function (non‑negative heights) with a low value of
evaluate_sequence.  The algorithm works as follows:

1.  Choose a moderate support size `n` (here 16384).  The evaluation
    function is scale‑invariant, so we normalise every candidate to
    sum = 1; the objective then equals ``2·n·max(conv)``.

2.  Generate a small pool of promising initial shapes:
    * a grid of symmetric “tent‑with‑plateau’’ shapes
      (_make_shape) parametrised by an exponent *p* and a plateau width *α*,
    * a handful of random Dirichlet draws,
    * the optional global starter ``height_sequence_1`` if it exists.

3.  Keep the best ``top_k`` candidates (default 5) and refine each of them
    independently with a projected **Adam** optimiser.  The gradient of the
    non‑smooth objective ``max(conv)`` is taken as the average of the
    sub‑gradients belonging to all *active* indices where the convolution
    is within a tiny tolerance of its maximum.  After each step the vector
    is projected onto the probability simplex (Euclidean projection).

4.  After the Adam phase, a light simulated‑annealing polish moves small
    packets of mass between random positions, accepting Metropolis steps.
    This finalises the solution.

5.  Trailing/leading zeros are trimmed (they would otherwise increase the
    factor ``2·n`` in the objective) and the vector is renormalised.

The whole search respects the 1000 s budget; a small safety margin is
kept to avoid exceeding the hard limit.  Progress is printed to STDOUT.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex { w ≥ 0, Σ w = 1 }.
    O(n log n) implementation (Wang & Carreira‑Perpiñán).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generator: power‑ramps with an optional flat plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length n.
    - p     : exponent of the side ramps (p > 0).
    - alpha : relative plateau width ∈ [0, 0.5]; alpha = 0 → pure tent,
              alpha = 0.5 → uniform distribution.
    The shape is normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    # keep plateau even (or zero) to preserve exact symmetry
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected Adam optimiser for the non‑smooth objective (max‑convolution)
# ----------------------------------------------------------------------
def _adam_subgradient(a0: np.ndarray,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Run a projected Adam optimiser on the exact objective.
    Returns the best iterate seen (and its score).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.5
    it = 0

    # tolerance for considering an index “active’’
    rel_tol = 1e-12

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                      f"best={cur_score:.6f}")

        max_val = conv.max()
        # active set – indices whose value is within a tiny tolerance of the maximum
        tol = rel_tol * max_val
        active = np.where(conv >= max_val - tol)[0]

        # accumulate sub‑gradient of all active constraints
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        if active.size > 0:
            grad /= float(active.size)

        # Adam update (with a slowly decaying step size)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions (Metropolis)."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of annealing attempts – limited by remaining time
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # choose a source proportionally to its mass (more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 25 % of the source mass
        eps_mass = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps_mass
        a_prop[dst] += eps_mass

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                if it % 10000 == 0:
                    print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                          f"best={new_score:.6f}")

    return best_a, best_score
