def grad_smoothmax(a, beta):
    # a shape (n,)
    conv = _conv_self(a)   # length 2n-1
    max_c = conv.max()

    # exponentiate with beta; careful for overflow: diff * beta <=0 so exponent <= 1.

    exp_vals = np.exp(beta * diff)
    w = exp_vals / exp_vals.sum()  # shape L

    # compute gradient = 2 * cross-correlation of w and a
    # cross-correlation (w ⋆ a)[i] = sum_{j} w[i+j] a_j.

                best_seq = a.copy()

"""
Propose a low‑score step function.

Algorithm
---------
1. **Fine‑grained Beta‑family sweep** –  Treat the heights as a discrete
   Beta(θ,θ)‑density: a_i ∝ (x_i·(1−x_i)+ε)^{θ‑1}.  Scan a dense grid of the
   exponent θ (≈0.2 … 0.8) and several tiny regularisers ε.  Keep the best
   candidate.

2. **Smoothed max‑gradient descent (Adam)** –  Replace the nonsmooth
   max‑operator by a log‑sum‑exp smoothing with a temperature β that is
   annealed from a small value to a large one.  The gradient of the smooth
   objective is a cross‑correlation of the soft‑max weights with the current
   vector.  Adam updates are performed, each iterate being projected onto the
   probability simplex.

3. **Aggressive simulated‑annealing polish** –  Random mass‑transfer moves
   (source chosen proportionally to its current mass) are attempted; a move
   is accepted if it improves the exact objective or with a tiny Metropolis
   probability that decays linearly with the iteration count.

4. **Normalization** –  Clip tiny negatives, renormalise to unit sum (the
   score is scale‑invariant) and return the plain Python list.

The whole routine respects the 1000 s wall‑time limit, prints progress
messages and works for any 1‑CPU environment.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the verifier:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed to be a 1‑D NumPy array.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed max and its gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    The smoothing uses the log‑sum‑exp with temperature beta.
    """
    # convolution a*a
    conv = _conv_self(a)                 # length 2n‑1
    max_c = conv.max()
    diff = conv - max_c                  # ≤ 0

    # soft‑max weights (sum to 1)
    exp_vals = np.exp(beta * diff)       # safe: diff ≤ 0
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp               # shape (2n‑1,)

    # gradient = 2 * (w ⋆ a)  (cross‑correlation)
    n = a.shape[0]
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, L)
    A_rev = np.fft.rfft(a[::-1], L)
    corr_full = np.fft.irfft(W * A_rev, L)
    corr_full = corr_full[: len(w) + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]   # shape (n,)

    smooth_val = max_c + np.log(sum_exp) / beta
    return smooth_val, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    beta_start: float = 1.0,
    beta_end: float = 150.0,
    lr: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the smoothed max.  Each iterate is projected onto the simplex.
    Returns the best vector (in terms of the exact objective) seen before deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    max_iters = 300000                # safety bound; real stopping is time‑driven

    while time.time() < deadline and it < max_iters:
        it += 1
        # linear schedule for beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 20000.0)

        _, grad = _smooth_max_and_grad(a, beta)

        # Adam update
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a = _project_simplex(a_new)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                f"score={cur_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves with a linear cooling schedule.
    Returns the best vector found before the deadline.
    """
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    T0 = 0.025                      # initial temperature
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        T = T0 * (1.0 - it / max_steps)   # linear cooling

        # pick a source proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        # move up to 30 % of the source mass
        eps_move = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or (T > 0 and math.exp(-delta / T) > rng.random()):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] anneal step={it:,} "
                    f"best={new_score:.6f}"
                )

    return best_a, best_score
