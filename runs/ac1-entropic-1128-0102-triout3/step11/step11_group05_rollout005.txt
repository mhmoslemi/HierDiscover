def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self(a, L_self=None):
    n = a.shape[0]
    if L_self is None:
        L_self = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af*Af, n=L_self)[:2*n-1]
    return conv

def _smoothmax_obj_and_grad(z, alpha, n, L_self):
    # returns objective (smooth approx of max) and gradient wrt z.
    a = np.exp(z)
    a_sum = a.sum()
    a = a / a_sum
    # linear conv
    conv = _conv_self(a, L_self)
    # compute softmax over alpha*conv
    exp_arg = alpha * conv
    max_exp_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_exp_arg)
    w_sum = w.sum()
    w = w / w_sum
    # smooth max value
    smooth_max = (np.log(w_sum) + max_exp_arg) / alpha  # because log-sum-exp = max + log(sum(exp(alpha*(conv-max))))
    # objective = 2 * n * smooth_max (since sum a=1)
    obj = 2.0 * n * smooth_max

    # grad wrt a:
    # compute correlation of w and a
    L_corr = _next_pow2(len(w) + n - 1)
    Wf = np.fft.rfft(w, n=L_corr)
    Af = np.fft.rfft(a, n=L_corr)
    corr_full = np.fft.irfft(Wf * np.conj(Af), n=L_corr)[:len(w) + n - 1]  # length 3n-2

    # grad wrt z:

"""
propose_candidate – smooth‑max Adam + fast simulated‑annealing

The algorithm minimises the objective

    2·n·max(conv(a,a)) / (sum(a))²          (n = 4096)

for a non‑negative step‑function a.  The key ideas are:

* Work on the probability simplex by representing a as a soft‑max of free
  logits z (a_i = exp(z_i)/∑exp(z)).
* Approximate the max‑convolution with a smooth log‑sum‑exp (temperature α).
  For a fixed α the objective is smooth and its gradient can be written
  analytically using FFT‑based convolutions.
* Run a long Adam optimisation while increasing α geometrically (continuation
  from a very smooth landscape to the true max).  The best vector found
  during this phase is remembered.
* Finish with a cheap incremental simulated‑annealing polish (identical to the
  one used in the reference solution) to exploit the remaining time budget.

All heavy work is done with FFTs (O(n log n)) so millions of iterations fit
comfortably into the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolutions (FFT, linear, full length)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation for sum(a)=1 (used for progress reports)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    conv = _conv_self(a, L_self)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient w.r.t. logits z
# ----------------------------------------------------------------------
def _smooth_obj_grad(z: np.ndarray, alpha: float,
                     n: int, L_self: int):
    """
    Returns:
        obj   – 2·n·smooth_max(conv)   (smooth_max ≈ max)
        grad  – ∇_z obj
        a     – current probability vector (sum=1)
        conv  – a*a  (full length 2n‑1)
        smooth_max – the smooth‑max value (not multiplied by 2·n)
    """
    # ----- a from logits -----
    a = np.exp(z)
    a_sum = a.sum()
    a = a / a_sum                     # a ∈ ℝ₊ⁿ, ∑a = 1

    # ----- self‑convolution -----
    conv = _conv_self(a, L_self)      # length 2n‑1

    # ----- smooth max (log‑sum‑exp) -----
    exp_arg = alpha * conv
    max_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_arg)     # w_i ∝ exp(alpha·conv_i)
    w_sum = w.sum()
    w = w / w_sum                     # softmax weights, ∑w = 1
    smooth_max = (max_arg + np.log(w_sum)) / alpha

    obj = 2.0 * n * smooth_max

    # ----- gradient w.r.t. a (sub‑gradient of smooth_max) -----
    #  grad_a_i = 2 * Σ_k w_k * a_{k-i}
    # use linear correlation (FFT) to obtain it efficiently
    L_corr = _next_pow2(len(w) + n - 1)          # length 3n‑2
    Wf = np.fft.rfft(w, n=L_corr)
    Af = np.fft.rfft(a, n=L_corr)
    # correlation = ifft( W * conj(A) )
    corr_full = np.fft.irfft(Wf * np.conj(Af), n=L_corr)[: len(w) + n - 1]
    grad_a = 2.0 * corr_full[n - 1:n - 1 + n]    # slice length n

    # ----- chain rule: gradient w.r.t. logits z (soft‑max Jacobian) -----
    dot = np.dot(a, grad_a)                     # scalar = Σ a_i·grad_a_i
    grad_z = a * (grad_a - dot)                 # ∂obj/∂z_i = a_i·(∂obj/∂a_i – Σ a·∂obj/∂a)

    return obj, grad_z, a, conv, smooth_max


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max family
# ----------------------------------------------------------------------
def _adam_optimize(z0: np.ndarray, n: int, L_self: int,
                   deadline: float, start_time: float):
    """
    Adam minimisation on a continuation of smooth‑maxes.
    Returns the best probability vector found and its true score.
    """
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_seq = None
    best_score = float("inf")
    it = 0

    # geometric schedule of smoothing parameters (α → ∞ = true max)
    alphas = np.geomspace(5.0, 4000.0, num=40)

    for alpha in alphas:
        # max number of inner steps for this α (will be cut short by the timer)
        max_inner = 3000
        for _ in range(max_inner):
            if time.time() >= deadline:
                break
            it += 1

            obj, grad_z, a, conv, _ = _smooth_obj_grad(z, alpha, n, L_self)

            # Adam update (descent)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # learning‑rate schedule – larger α needs a smaller step
            lr = 0.12 / (np.sqrt(alpha) + 1.0) / (math.sqrt(it) + 1.0)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation (fast version)
            if it % 250 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.1f} "
                          f"it={it:,} score={cur_score:.6f}")

    # fall back to current a if nothing was stored
    if best_seq is None:
        best_seq = a.copy()
        best_score = _eval_fast(best_seq, L_self)

    return best_seq, best_score

# Incremental simulated‑annealing (identical to the reference implementation)
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-10,
                        max_moves: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that updates the convolution incrementally."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # sum(a)=1 ⇒ denominator = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        # linear temperature schedule (exponential decay)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.35
        if eps <= 0.0:
            continue

        # incremental update of the convolution
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / T) > rng.random():
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
