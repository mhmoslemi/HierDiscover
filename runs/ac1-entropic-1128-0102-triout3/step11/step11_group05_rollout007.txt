    a = a0.copy()
    m = np.zeros_like(a) # first moment
    v = np.zeros_like(a) # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        # compute conv
        conv = _conv_self(a, L_self) # maybe using L_self
        cur_max = conv.max()

        score = 2.0 * n * cur_max / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # find k_max
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # ADAM update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.01 / (np.sqrt(v_hat) + eps) # adapt step
        a = a - step * m_hat
        # project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)
    return best_a, best_score

def _improved_anneal(a0, n, deadline, start_time, best_score, best_seq):

    conv = _conv_self(a, L_self)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_a = best_seq.copy()
    best_cur_score = best_score

    rng = np.random.default_rng()
    max_moves = 5_000_000
    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        # compute temperature linearly dec from start_temp to end_temp
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * ((end_temp / start_temp) ** (elapsed / total))

        # choose source index bias
        # compute max index of convolution
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n-1))
        hi = min(n-1, k_max)
        # compute weight for all indices: weight_i = a[i] * a[k_max - i] if i in [lo,hi] else 0
        w = np.zeros(n)
        idx_range = np.arange(lo, hi+1, dtype=int)
        w[idx_range] = a[idx_range] * a[k_max - idx_range]
        # sample src weighted by w^0.5 (more emphasis)
        if w.sum() > 0:
            prob = w / w.sum()
            src = rng.choice(n, p=prob)
        else:
            src = rng.integers(n)
        # choose destination far from range
        # we could sample uniformly but reject if too close
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # propose epsilon proportion of src mass

"""
Optimisation of the max‑self‑convolution functional

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

for n = 4096.  The code follows a multi‑stage strategy:

0️⃣  Initialise from the global best or a small parametric family.
1️⃣  Sweep a few simple analytic families (symmetric beta‑like and
    power‑law shapes) and keep the best candidate.
2️⃣  A long smooth‑max mirror‑descent (exponentiated‑gradient) on the
    probability simplex with a geometric schedule of smoothing parameters,
    extended to α up to 2000 and up to 6·10⁵ iterations.
3️⃣  ADAM‑style projected sub‑gradient descent that follows the true
    max‑convolution (no smoothing) – this quickly polishes the solution.
4️⃣  An annealing phase that moves mass from indices that contribute most
    to the current maximal convolution entry to far‑away indices.  The
    temperature schedule starts much hotter (0.5) than in the reference
    implementation and the allowed mass move size is increased early on.
    The convolution is updated incrementally, allowing a few‑million moves
    within the remaining time budget.

All phases stop as soon as the global time limit (≈ 1000 s) is reached.
The final vector is projected back onto the probability simplex,
clipped to the allowed range and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (full length)."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast objective (sum ≈ 1) – used during the search
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a non‑negative vector (sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:                       # safety – should never happen
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 0️⃣  Simple parametric families (beta‑like, power‑law)
# ----------------------------------------------------------------------
def _beta_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α)‑like density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    # shape ∝ (i+1)^{α-1}·(n-i)^{α-1}
    a = (i + 1) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _power_law_family(n: int, beta: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+0.5)(n‑0.5‑i))^{‑β}."""
    i = np.arange(n, dtype=float)
    x = (i + 0.5) * (n - 0.5 - i)
    # prevent division by zero at the very ends
    eps = 1e-12
    a = np.where(x > eps, x ** (-beta), 0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 1️⃣  Smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    L_self: int,
                    L_grad: int) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the soft‑max of the convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    # use a more aggressive α‑schedule (up to 2000)
    alphas = np.geomspace(0.5, 2000.0, num=40)

    it = 0
    max_iterations = 600_000
    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # 1) self‑convolution via FFT (re‑use transform)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights (log‑sum‑exp trick)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)          # avoid exact zeros
            a /= a.sum()

            # occasional exact score check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  ADAM‑style projected sub‑gradient descent (true max)
# ----------------------------------------------------------------------
def _adam_subgradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      L_self: int,
                      init_score: float) -> tuple[np.ndarray, float]:
    """ADAM on the exact max‑convolution (projected onto the simplex)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score

    m = np.zeros_like(a)       # first moment
    v = np.zeros_like(a)       # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        # exact self‑convolution
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ADAM sub‑grad it={t:,} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # ADAM update (step size tuned for this problem)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.02 / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat

        # projection onto the simplex (non‑negative + sum = 1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Fast incremental simulated‑annealing (biased mass transfer)
# ----------------------------------------------------------------------
def _improved_anneal(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     cur_best_score: float,
                     cur_best_seq: np.ndarray,
                     L_self: int,
                     start_temp: float = 5e-1,
                     end_temp: float = 1e-10,
                     max_moves: int = 8_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer with a bias towards indices that drive the max."""
    a = a0.copy()
    conv = _conv_self(a, L_self)                 # full length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max               # sum(a)=1 already
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # linear‑in‑log temperature schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # ------------------------------------------------------------------
        # choose a source index – bias towards contributors to the current max
        # ------------------------------------------------------------------
        kmax = int(conv.argmax())
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        # contribution weight = a[i] * a[kmax‑i]
        contrib = np.zeros(n, dtype=float)
        contrib[idx_range] = a[idx_range] * a[kmax - idx_range]
        if contrib.sum() > 0:
            prob = contrib / contrib.sum()
            src = rng.choice(n, p=prob)
        else:
            src = rng.integers(n)

        if a[src] < 1e-14:
            continue

        # destination – try to stay away from the max‑support region
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # epsilon – size of the mass move (larger while temperature is high)
        max_frac = min(0.6, 5.0 * T)                       # up to 60 % early on
        eps = a[src] * rng.random() * max_frac
        if eps <= 0.0:
            continue

        # ----- incremental convolution update (copy‑on‑write) ----------------
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            # accept
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = new_score
            cur_max = new_max

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} "
                      f"best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (length 4096) that attempts to
    minimise `evaluate_sequence`.  The routine follows the stages described
    above and respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0️⃣  Load a starter vector (global best if present, else uniform)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')

    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 0️⃣‑a  Small parametric sweep (beta & power‑law families)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # symmetric beta family (α from 0.2 to 5)
        for alpha in np.geomspace(0.2, 5.0, num=15):
            cand = _beta_family(n, alpha)
            sc = _eval_fast(cand, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] beta α={alpha:.3f} score={sc:.6f}")

        # power‑law family (β from 0.05 to 0.65)
        for beta in np.linspace(0.05, 0.65, num=13):
            cand = _power_law_family(n, beta)
            sc = _eval_fast(cand, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] power β={beta:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Smooth‑max mirror descent (extended)
    # --------------------------------------------------------------
    if time.time() < deadline - 15.0:
        smooth_deadline = deadline - 12.0
        best_seq, best_score = _smooth_descent(best_seq, n,
                                               smooth_deadline,
                                               start_time,
                                               L_self, L_grad)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best "
              f"score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  ADAM‑style sub‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 3.0
        a_sub, sc_sub = _adam_subgradient(best_seq, n,
                                           sub_deadline, start_time,
                                           L_self, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] ADAM sub‑grad improved "
                  f"score = {sc_sub:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Incremental simulated‑annealing polish (bias & hotter start)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_final, sc_final = _improved_anneal(best_seq, n,
                                             anneal_deadline,
                                             start_time,
                                             best_score,
                                             best_seq,
                                             L_self,
                                             start_temp=5e-1,
                                             end_temp=1e-10,
                                             max_moves=8_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved "
                  f"score = {sc_final:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{best_score:.6f}")
    return [float(v) for v in best_seq]
