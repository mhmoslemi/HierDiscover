def propose_candidate(seed=0, budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    np.random.seed(seed)

    # 1. Grid search for arcsine exponents
    n = 4096
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    best_seq, best_score = None, float('inf')
    # Additional param: α grid fine
    alphas = np.linspace(0.33, 0.48, 100)  # step .0015
    epsilons = [0.0, 1e-8, 1e-6, 1e-4]
    for eps in epsilons:
        base = x*(1.0-x)+eps
        for alpha in alphas:
            w = base ** (-alpha)
            a = w / w.sum()
            score = _objective(a)  # using local fast objective (our own)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        # early break if below target+margin
        if best_score < 1.504:
            break

    # optionally try mixing uniform
    if best_score > 1.5035:
        mix_lambdas = np.linspace(0.0, 1.0, 21)

            w_base = (x*(1-x)+1e-6) ** (-alpha)
            for lam in mix_lambdas:
                w = lam * w_base + (1 - lam) * np.ones_like(x)
                a = w / w.sum()
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()

    # incorporate global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim ==1 and starter.size>0:
            starter = np.maximum(starter,0.0)
            if starter.sum()>0:

                cur_score = _objective(starter)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
    except Exception:
        pass

    # 2. Refine via mask subgradient descent
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining * 0.75
        refined_seq, refined_score = _subgradient_mask_descent(best_seq, sub_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] mask‑subgrad refined score={refined_score:.6f}")

    # 3. Light annealing polish
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_steps = 50000
        for it in range(1, total_steps+1):
            if time.time() >= deadline:
                break
            # source selection weighted by gradient contributions to max conv (approx)
            # Compute conv and max
            if it % 1000 == 0:

                if prob_src.sum() > 0:
                    prob_src = prob_src / prob_src.sum()
                else:
                    prob_src = a  # fallback
                src = rng.choice(n, p=prob_src)
                # choose dest to lower gradient values
                prob_dst = (grad.max() - grad) + 1e-12
                prob_dst = prob_dst / prob_dst.sum()
                dst = rng.choice(n, p=prob_dst)
            else:
                src = rng.choice(n, p=a)  # fallback for speed
                dst = rng.integers(n)

            if src == dst or a[src] < 1e-14:
                continue
            delta = a[src] * rng.random() * 0.15  # up to 15% of source
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta
            new_score = _objective(a_prop)
            if new_score < cur_score or rng.random() < 0.001:

    if total > 0:

    else:

"""
Propose a low‑score step function.

Algorithm
---------
1. **Fine‑grained search of the arcsine family** –  a_i ∝ (x·(1−x)+ε)^{−α}.
   A dense grid of exponents (α≈0.33…0.48) and a few tiny ε is swept.
   The best candidate is kept.

2. **Mask‑subgradient descent** –  Instead of using the single
   maximal convolution entry, all entries that are within a small
   relative tolerance of the maximum are treated as active.
   The sub‑gradient of the (soft) max is the cross‑correlation of the
   current vector with that mask.  A projected (momentum) sub‑gradient
   method is run for the rest of the budget.

3. **Light annealing polish** –  A few‑ten‑thousand random mass‑transfer
   moves are attempted; moves are biased by the current mask‑gradient
   (source = large gradient, destination = small gradient) and kept
   if they improve the objective.

The whole procedure respects the 1000 s wall‑time limit, works on a
single CPU, and returns a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (full length) using FFT."""
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Objective and projection
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the verifier."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Mask‑subgradient descent (many active constraints)
# ----------------------------------------------------------------------
def _mask_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using a mask of near‑max entries."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    momentum = np.zeros_like(a)
    beta = 0.9          # momentum coefficient
    step0 = 0.20        # initial step size factor
    mask_rel = 0.001   # relative tolerance for the active set

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] mask‑subgrad iter={it:,} "
                f"score={cur_score:.6f}"
            )

        # --- active mask -------------------------------------------------
        tol = maxc * mask_rel
        mask = (conv >= maxc - tol).astype(np.float64)          # length 2n‑1

        # gradient = 2 * (a ⋆ mask)  (cross‑correlation)
        mask_rev = mask[::-1]
        grad_full = _conv_full_fft(a, mask_rev)                # length 3n‑2
        idx_start = len(mask) - 1                               # 2n‑2
        grad = 2.0 * grad_full[idx_start - np.arange(n)]

        # --- momentum & step --------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep on the simplex
        a = _project_simplex(a)

        # stop if steps become negligible
        if step < 1e-14:
            break

    return best_a, best_score

    return [float(v) for v in best_seq]
