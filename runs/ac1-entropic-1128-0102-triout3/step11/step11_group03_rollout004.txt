"""
Propose a low‑score step function.

Algorithm
---------
1. **Low‑dimensional family** – Construct probability vectors as a mixture of
   three elementary components:
        * a Beta‑type shape    w_i ∝ (x_i+ε)^{a‑1} (1‑x_i+ε)^{b‑1}
        * the uniform distribution
        * a pair of spikes at the two ends (mass ½ at each extreme)

   The mixing weights, the Beta exponents (a,b) and the regulariser ε are
   searched globally with `scipy.optimize.differential_evolution`, a cheap
   derivative‑free optimiser.  The search is performed on a reasonably large
   discretisation (n = 8192) so that the evaluation of the true objective
   is accurate.

2. **Refinement** – The best candidate from the global search is refined by
   a projected sub‑gradient descent on the exact non‑smooth objective
   `2·n·max(conv(a,a))/‖a‖₁²`.  The method keeps the iterate on the
   probability simplex, uses momentum and a decreasing step size.

3. **Light annealing polish** – Finally a few‑ten‑thousand random mass‑transfer
   moves are attempted; a move is kept if it improves the objective (or
   with a tiny Metropolis probability).  This removes last‑minute irregularities.

All stages respect the supplied wall‑time budget (default 1000 s) and the
function always returns a non‑negative list (scaling does not affect the
score).  If an older construction `height_sequence_1` is available it is
taken into account as an additional candidate.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Optional SciPy import (used for the global optimiser)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import differential_evolution
    _SCIPY_AVAILABLE = True
except Exception:                         # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# FFT‑based self‑convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the verifier:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed to be a 1‑D NumPy array with non‑negative entries.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the simplex { w ≥ 0, Σ w = 1 }
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (nonsmooth objective)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float, start_time: float):
    """
    Perform a simple projected sub‑gradient descent on the exact
    (non‑smooth) objective.  Returns the best iterate seen before
    `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum coefficient
    step0 = 0.25        # initial step size factor
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} score={cur_score:.6f}")

        # indices that attain the maximum (all of them to get a smoother
        # sub‑gradient, not only one)
        max_idxs = np.where(conv == cur_max)[0]

        grad = np.zeros_like(a)
        for k in max_idxs:              # usually only a few indices
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            # derivative of conv_k w.r.t a_i is a_{k-i}+a_{k-i} = 2*a_{k-i}
            grad[idx] += 2.0 * a[k - idx]

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: build a probability vector from a low‑dimensional parameter set
# ----------------------------------------------------------------------
def _build_vector(params: np.ndarray, x: np.ndarray,
                  spike: np.ndarray, uniform: np.ndarray) -> np.ndarray:
    """
    Decode a parameter vector:
        params = [log10(a), log10(b), log10(eps), w1, w2, w3]
    into a probability vector of length len(x).  The three weights w1,w2,w3 are
    turned into mixing coefficients via a soft‑max.
    """
    a = 10.0 ** params[0]
    b = 10.0 ** params[1]
    eps = 10.0 ** params[2]

    # mixing weights – simple soft‑max to guarantee non‑negativity and sum=1
    raw = np.maximum(params[3:6], 0.0)
    if raw.sum() == 0:
        mix = np.array([1.0, 0.0, 0.0])   # fall back to spikes only
    else:
        mix = raw / raw.sum()
    lam_spike, lam_uniform, lam_beta = mix

    # beta‑type shape (Beta(a,b) with a small regulariser)
    beta_shape = ((x + eps) ** (a - 1.0)) * ((1.0 - x + eps) ** (b - 1.0))
    beta_shape = beta_shape / beta_shape.sum()

    # final mixture (each component already sums to 1)
    w = lam_spike * spike + lam_uniform * uniform + lam_beta * beta_shape

    # very small numerical drift – re‑normalise to be safe
    w = np.maximum(w, 0.0)
    w = w / w.sum()
    return w

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Choose a discretisation size
    # ------------------------------------------------------------------
    # 8192 points gives a good compromise between accuracy and speed
    n = 8192
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑point of each bin

    # elementary components
    spike = np.zeros(n, dtype=np.float64)
    spike[0] = spike[-1] = 0.5                     # each spike carries half the mass
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = None
    best_score = float("inf")
    best_params = None

    # ------------------------------------------------------------------
    # 2) Global search in the low‑dimensional family
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] start global search (n={n})")

    def _eval_wrap(params):
        """Wrapper that returns the objective value for differential evolution."""
        vec = _build_vector(params, x, spike, uniform)
        return _objective(vec)

    if _SCIPY_AVAILABLE:
        # bounds for the six optimisation variables
        bounds = [
            (-1.0, 0.3),          # log10(a)  → a ∈ [0.1, 2]
            (-1.0, 0.3),          # log10(b)  → b ∈ [0.1, 2]
            (-13.0, -4.0),        # log10(eps) → eps ∈ [1e‑13, 1e‑4]
            (0.0, 5.0),           # raw weight w1 (spikes)
            (0.0, 5.0),           # raw weight w2 (uniform)
            (0.0, 5.0),           # raw weight w3 (beta)
        ]

        # a relatively small budget – 400‑500 function evaluations are enough
        max_iter = 30
        popsize = 15
        result = differential_evolution(
            func=_eval_wrap,
            bounds=bounds,
            strategy='best1bin',
            maxiter=max_iter,
            popsize=popsize,
            polish=False,
            seed=seed,
            updating='deferred',
            workers=1,
            tol=1e-7,
        )
        candidate = _build_vector(result.x, x, spike, uniform)
        cand_score = _objective(candidate)
        print(f"[{time.time() - start_time:.2f}s] DE best score={cand_score:.6f}")

        best_seq = candidate
        best_score = cand_score
        best_params = result.x.copy()
    else:
        # –‑fallback‑‑: a dense random sweep (still very cheap)
        n_samples = 3000
        for i in range(n_samples):
            if time.time() >= deadline - 5.0:
                break
            a_log = np.random.uniform(-1.0, 0.3)
            b_log = np.random.uniform(-1.0, 0.3)
            eps_log = np.random.uniform(-13.0, -4.0)
            w_raw = np.random.rand(3) * 5.0
            params = np.hstack([a_log, b_log, eps_log, w_raw])
            vec = _build_vector(params, x, spike, uniform)
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_params = params.copy()
                print(f"[{time.time() - start_time:.2f}s] random sweep new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Compare against a possible global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur_score = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] starter score={cur_score:.6f}")
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
                    best_params = None
    except Exception:
        pass   # no global starter present

    # ------------------------------------------------------------------
    # 4) Sub‑gradient refinement (takes the lion’s share of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and best_seq is not None:
        sub_deadline = time.time() + 0.85 * remaining
        refined_seq, refined_score = _subgradient_descent(best_seq, sub_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] refined by sub‑grad score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_steps = 30000

        for it in range(1, total_steps + 1):
            if time.time() >= deadline:
                break

            # linear cooling from 0.02 down to 2e‑5
            T = 0.02 * (2e-5 / 0.02) ** (it / total_steps)

            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            # move up to 15 % of the source mass
            move = a[src] * rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= move
            a_prop[dst] += move

            new_score = _objective(a_prop)
            delta = new_score - cur_score

            if delta < 0.0 or math.exp(-delta / (T + 1e-12)) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improves to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # The verifier only cares about the numerical values; scaling does not matter.
    return [float(v) for v in best_seq]
