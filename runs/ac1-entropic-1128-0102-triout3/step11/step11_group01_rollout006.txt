def _gradient_softmax(a, T):
    conv = _conv_self(a) # length 2N-1
    max_conv = conv.max()
    w = np.exp((conv - max_conv) / T)
    w_sum = w.sum()
    w = w / w_sum

def propose_candidate(seed=0, budget_s=1000, **kwargs):
   np.random.seed(seed)
   start_time = time.time()
   deadline = start_time + budget_s - 0.5

   # Choose N based on remaining time: use 65536 if we have >400 seconds, else 32768.
   remaining = deadline - start_time
   if remaining > 400:
      N = 65536
   else:
      N = 32768

   # Prepare initial sequences
   candidates = []
   # basic shape
   p_vals = [0.5, 0.8, 1.0, 1.2, 1.5]
   alpha_vals = [0.0, 0.1, 0.2, 0.3, 0.4]
   for p in p_vals:
       for alpha in alpha_vals:
           a = _make_shape(N, p, alpha)
           candidates.append(a)

   # if global starter exists
   try:
       starter = np.asarray(height_sequence_1, dtype=float)
       if starter.ndim == 1 and starter.size > 0:
           # upscale
           if starter.size != N:
               factor = N // starter.size

               starter_up = starter_up / starter_up.sum()

def _smooth_gradient_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # hyperparameters
    step0 = 0.2
    T0 = 0.05  # starting temperature
    T_final = 1e-4
    max_iter = 2000000  # just as limit
  
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # conv and objective
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        # track best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it} score={cur_score:.6f}")

        # update temperature: linear schedule
        # fraction of total time left for descent:
        # compute proportion = it / max_iter
        fraction = it / max_iter

        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_wr = np.fft.irfft(W * A_rev, n=L)[: 3*n - 2]
        grad = 2.0 * conv_wr[n-1 : n-1 + n]

        # momentum optional: we'll use simple momentum
        # Let's maintain momentum vector across iterations
        if it == 1:
            momentum = np.zeros_like(a)
        beta = 0.9
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / np.sqrt(it)
        a = a - step * momentum

        # project onto simplex
        a = _project_simplex(a)

    return best_a, best_score

"""
Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Initialisation** – generate a few analytic shapes (power‑ramp
   “tent‑with‑plateau”), a few random vectors and, if available,
   the global starter `height_sequence_1`.  The best few candidates are kept.

2. **Smooth gradient descent** – minimise a smoothed version of the
   objective using a soft‑max over the convolution entries.  The gradient
   is obtained efficiently by FFT (correlation of the soft‑max weights
   with the current vector).  Adam‑style adaptive steps are used and the
   iterate is projected onto the probability simplex after each step.

3. **Polishing** – a short sub‑gradient descent (as in the reference code)
   followed by a light simulated‑annealing move pack to escape possible
   stagnation.

All steps respect the 1000 s time budget; the function returns a plain
Python list of non‑negative numbers (the harness will feed it to
`evaluate_sequence`).  The implementation is self‑contained and uses only
NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generator (power‑ramps with optional plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric shape of length n, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a smooth surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.05          # base learning rate
    T0 = 0.02             # initial soft‑max temperature
    T_end = 1e-4          # final temperature

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"score={cur_score:.6f}")

        # ----- temperature schedule (linear decay) -----
        frac = it / (it + 1000.0)        # smooth freeze
        T = max(T_end, T0 * (1.0 - frac))

        # ----- soft‑max weights -----
        # shift by max_conv for numerical stability
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # ----- gradient via correlation (w * a_rev) -----
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)        # length for convolution w * a_rev
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]  # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]    # slice of length n

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (same as reference implementation)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑conv term
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
