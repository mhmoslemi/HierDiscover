"""
Propose a low‑score step function.

Algorithm
---------
1. **Explore a few simple families** (arcsine‑like, exponential, linear,
   random, the global starter if available) and keep the best candidate.

2. **Adam optimisation on a smoothed “max‑convolution’’**.
   The non‑smooth objective   F(a)=2·n·max(a*a)/‖a‖₁²   is approximated by a
   soft‑max with a temperature β that is increased linearly from a small
   value to a large one.  The gradient of the weighted sum of all
   convolution entries is obtained by two FFT‑based convolutions,
   yielding a fast O(n log n) iteration.  After each step the iterate is
   projected back onto the probability simplex.

3. **Light mass‑transfer annealing** – a few‑ten‑thousand random moves
   (transfer a fraction of the mass from one index to another) are tried;
   a move is kept if it improves the true objective (or is accepted
   according to a tiny Metropolis rule).

All steps respect the 1000 s wall‑time limit, use only the public
`evaluate_sequence` function for scoring and return a plain list of
non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (exact) – used for logging and final evaluation
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact verifier objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, Σ x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolution entries
# ----------------------------------------------------------------------
def _gradient_weighted_conv(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a ( Σ_k w_k (a*a)_k ) efficiently via FFT.
    a : shape (n,)
    w : shape (2n-1,) – weights for each convolution index
    Returns gradient of shape (n,).
    """
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)          # length needed for linear conv of (2n‑1) and n
    A = np.fft.rfft(a, n=L)

    # first convolution: w * a
    W = np.fft.rfft(w, n=L)
    conv_full = np.fft.irfft(A * W, n=L)[: 3 * n - 2]

    # second convolution: reversed w * a
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=L)
    conv_rev_full = np.fft.irfft(A * W_rev, n=L)[: 3 * n - 2]

    start = n - 1                      # slice that corresponds to indices 0 … n‑1
    grad = conv_full[start:start + n] + conv_rev_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    beta0: float = 0.5,
    beta_max: float = 50.0,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the exact objective using a soft‑max approximation
    of the max‑convolution.  Returns the best iterate and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution and soft‑max weights -----
        conv = _conv_self(a)                     # length 2n‑1
        max_conv = conv.max()
        # linear schedule for the temperature β
        ratio = (time.time() - start_time) / (deadline - start_time)
        beta = beta0 + (beta_max - beta0) * ratio

        # numerically stable soft‑max
        exp_term = np.exp(beta * (conv - max_conv))
        w = exp_term / exp_term.sum()            # weights, sum to 1
        M = (w * conv).sum()                     # soft max (expectation)

        sum_a = a.sum()
        # ----- gradient of the smoothed objective -----
        grad_M = _gradient_weighted_conv(a, w)    # ∇_a Σ w_k conv_k
        # F = 2·n·M / sum_a**2
        grad_F = 2.0 * n * (grad_M / (sum_a ** 2) - 2.0 * M * np.ones_like(a) / (sum_a ** 3))

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_F
        v = beta2 * v + (1.0 - beta2) * (grad_F ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- back to the feasible set -----
        a = _project_simplex(a)

        # ----- exact score for logging -----
        cur_score = 2.0 * n * conv.max() / (sum_a ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                f"β={beta:.2f} score={cur_score:.6f}"
            )
    return best_a, best_score
