"""
Propose a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm – “LP‑refinement + multi‑restart hill‑climb”
-----------------------------------------------------------------
1. **Multi‑start** – several different random/simple initialisations
   (uniform, Dirichlet‑spikes, the historic best) are tried in parallel.
2. **Iterated linear‑programming refinement** – for a current vector `f`
   we compute the peak `M = max(f*f)`.  The active set consists of the
   indices where the convolution is close to `M`.  We then solve the LP

        maximise   Σ g_i
        subject to 0 ≤ g_i,
                    (f * g)_k ≤ M   for every active k.

   The LP is tiny because only the active constraints are kept (typically
   a few dozen).  The solution `g` points towards a direction that can
   lower the peak.  We blend the current vector with the normalised `g`
   (convex combination) and keep the best of a few step lengths.
3. **Polish – greedy hill‑climb** – after the LP phase we run a fast
   local‑search that moves a small amount of mass from one index to
   another, accepting only improving moves.  The full convolution is
   recomputed only when a move is considered, which is still cheap for
   n = 4096 thanks to FFT.

All phases respect a wall‑clock deadline (default 1000 s).  The returned
list is normalised to sum = 1, guaranteeing that ``evaluate_sequence`` sees
a valid candidate.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Active‑set handling
# ----------------------------------------------------------------------
def _active_set(conv: np.ndarray, max_val: float,
                rel_tol: float = 0.02, max_active: int = 200) -> np.ndarray:
    """
    Return indices where conv is within ``rel_tol`` of the maximum.
    At most ``max_active`` indices are kept (the largest values).
    """
    # indices with value >= (1 - rel_tol) * max_val
    thresh = max_val * (1.0 - rel_tol)
    cand = np.nonzero(conv >= thresh)[0]
    if cand.size <= max_active:
        return cand
    # keep the top‑max_active entries
    top = np.argpartition(conv[cand], -max_active)[-max_active:]
    return cand[top]


def _build_A_ub(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape len(active_idx) × n) such that
    (f * g)[k] = A[k] · g   for each active k.
    """
    n = f.shape[0]
    m = active_idx.size
    A = np.zeros((m, n), dtype=np.float64)

    for row, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices of f that contribute
        f_idx = np.arange(lo, hi + 1, dtype=np.int32)
        # corresponding positions in g
        g_idx = k - f_idx
        # fill the row
        A[row, g_idx] = f[f_idx]

    return A


# ----------------------------------------------------------------------
# One LP refinement step
# ----------------------------------------------------------------------
def _lp_refine_step(f: np.ndarray, L_self: int,
                    rel_tol: float = 0.02,
                    max_active: int = 200,
                    blend_factors: tuple = (0.1, 0.3, 0.5, 0.8)) -> tuple[np.ndarray, float]:
    """
    Perform one LP‑based refinement of ``f``.
    Returns (new_f, new_score).  If the LP fails or yields no improvement,
    the original ``f`` is returned unchanged.
    """
    n = f.shape[0]
    conv = _conv_self(f, L_self)
    M = conv.max()
    active = _active_set(conv, M, rel_tol, max_active)

    # Build the linear constraints (f * g)_k ≤ M for active k
    A = _build_A_ub(f, active)
    b = np.full(active.shape[0], M, dtype=np.float64)

    # Objective: maximise Σ g_i  → minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Solve the LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x.sum() == 0.0:
        # LP failed – fall back to identity step
        return f, _eval_fast(f, L_self)

    g = res.x
    # Normalise g to have sum = 1 (so we stay on the simplex)
    g_norm = g / g.sum()

    # Try a few convex‑combination factors and keep the best
    best_f = f
    best_score = _eval_fast(f, L_self)

    for t in blend_factors:
        candidate = (1.0 - t) * f + t * g_norm
        candidate = np.maximum(candidate, 0.0)
        candidate /= candidate.sum()
        score = _eval_fast(candidate, L_self)
        if score < best_score:
            best_score = score
            best_f = candidate

    return best_f, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (same idea as the previous benchmark but with a
# simple early‑stop on stagnation)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                L_self: int,
                deadline: float,
                start_time: float,
                init_score: float,
                max_no_improve: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb.  Moves up to 30 % of the source mass.
    Stops when ``max_no_improve`` iterations have passed without improvement.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = init_score
    cur_score = init_score
    n = a.shape[0]
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        src = rng.choice(n, p=a)          # source weighted by current mass
        dst = rng.integers(0, n)          # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * 0.30   # up to 30 % of source mass
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # Evaluate the new candidate
        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved = {best_score:.6f}")
        else:
            no_improve += 1

    return best_seq, best_score

        hist = np.asarray(height_sequence_1, dtype=float)
        if hist.ndim == 1 and hist.size > 0:
            hist = np.maximum(hist, 0.0)
            if hist.sum() > 0.0:
                hist = hist / hist.sum()
                if hist.size != n:
                    if hist.size > n:
                        hist = hist[:n]
                    else:
                        hist = np.pad(hist, (0, n - hist.size))
                inits.append(hist)
    except Exception:
        pass

    # 3) few random Dirichlet draws (concentration <1 makes spiky vectors)
    for _ in range(3):
        a = np.random.gamma(0.3, 1.0, size=n)
        a = a / a.sum()
        inits.append(a)

    # 4) deterministic “single‑spike” at centre and at a random location
    spike = np.zeros(n)
    spike[n // 2] = 1.0
    inits.append(spike)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # For each start, run an LP‑refinement loop followed by hill‑climb
    # --------------------------------------------------------------
    for start_id, a0 in enumerate(inits):
        if time.time() >= deadline - 5.0:
            break   # keep enough time for final polishing

        print(f"[{time.time() - start_time:.2f}s] Restart {start_id + 1}/{len(inits)}")
        a = a0.copy()
        cur_score = _eval_fast(a, L_self)

        # --- LP refinement loop ---
        lp_iters = 0
        while time.time() < deadline - 2.0 and lp_iters < 30:
            a_new, new_score = _lp_refine_step(a, L_self)
            lp_iters += 1
            if new_score < cur_score - 1e-12:
                a = a_new
                cur_score = new_score
                print(f"[{time.time() - start_time:.2f}s]  LP iter {lp_iters} score = {cur_score:.6f}")
            else:
                # no improvement – stop LP phase for this start
                break

        # --- hill‑climb polishing ---
        if time.time() < deadline - 0.5:
            a, cur_score = _hill_climb(a, L_self, deadline, start_time,
                                       init_score=cur_score)

        # keep the best overall
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] *** New global best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Safety: if nothing succeeded, fall back to uniform
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _eval_fast(best_seq, L_self)

    # Final normalisation (should already be sum≈1)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Return plain Python list as required by the evaluation harness
    return [float(v) for v in best_seq]
