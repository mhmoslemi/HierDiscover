"""
Propose a step sequence with a very low evaluation score.

The algorithm works in four stages:

1. **Parametric sweep** – explore a family a_i ∝ (x_i(1−x_i)+ε)^{−α}
   (the “Beta‑family”) on a moderately large grid and keep the best
   candidate.

2. **Smoothed Adam optimisation** – minimise a log‑sum‑exp surrogate
   of the original non‑smooth objective.  The smoothing parameter β is
   increased linearly so that the surrogate becomes close to the true
   max‑convolution.  Adam with simplex projection drives the vector to a
   good basin.

3. **Direct sub‑gradient descent** – after the smooth phase we switch
   to the exact non‑smooth objective.  The lag(s) where the convolution
   attains its maximum are identified; the gradient of the max‑lag is
   exactly 2·a shifted.  A tiny amount of mass is transferred from the
   index with the largest gradient to the one with the smallest,
   accepting only improving moves.  This often pushes the bound a few
   percent lower.

4. **Light annealing polish** – a short simulated‑annealing style
   random‑mass‑transfer phase finishes the search.

All steps respect the simplex constraint (non‑negative, Σa = 1) and the
overall wall‑time limit (≈1000 s).  The final sequence is returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT‑based linear convolution (fast for large n).
    The sequence is assumed to be already normalised (sum≈1); the function
    does not enforce the simplex – it works for any non‑negative vector.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a with zero‑padding to length L (FFT)."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    return conv_full

                    best_score = score
                    best_seq = h_up.copy()
        except Exception as e:
            print("Failed to reuse height_sequence_1:", e)

    # ------------------------------------------------------------------
    # 3) Smoothed Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n)
    else:
        a = best_seq.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.02

    # FFT length sufficient for both self‑convolution and the extra
    # correlation needed for the gradient of the smooth surrogate.
    L = _next_pow2(3 * n)

    iteration = 0
    beta_start, beta_end = 5.0, 500.0   # final β is fairly large → close to true max

    # Run Adam until we have about 150 s left for the exact phase
    while time.time() < deadline - 150:
        iteration += 1

        # ---- self‑convolution (linear) ---------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # ---- smoothing parameter (increasing) -------------------------
        elapsed = time.time() - start_time
        prop = min(1.0, elapsed / (budget_s - 150))
        beta = beta_start + (beta_end - beta_start) * prop

        # ---- log‑sum‑exp weights (soft‑max) ---------------------------
        beta_conv = beta * conv
        max_beta = beta_conv.max()
        exp_shift = np.exp(beta_conv - max_beta)          # shift for stability
        sum_exp = exp_shift.sum()
        w = exp_shift / sum_exp                           # shape (2n‑1,)

        # ---- gradient of the smooth surrogate -------------------------
        # gradient = 4·n·B·a, where B·a = cross‑correlation of w with a
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        Ba = corr_full[n - 1:n - 1 + n]                    # length n
        gradient = 4.0 * n * Ba

        # ---- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * gradient
        v = beta2 * v + (1.0 - beta2) * (gradient ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)          # keep on the probability simplex

        # ---- keep the best exact objective seen so far -----------------
        if iteration % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{elapsed:.2f}s] Adam iter={iteration:,}  best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact sub‑gradient descent (mass‑transfer moves)
    # ------------------------------------------------------------------
    while time.time() < deadline - 20:
        # compute exact convolution and the current max value
        conv_full = _conv_full(a, L)
        conv = conv_full[: 2 * n - 1]
        M = conv.max()
        cur_score = 2.0 * n * M                   # sum(a) == 1 ⇒ denominator = 1

        # identify all lags that are essentially maximal
        tol = max(1e-12, 1e-8 * M)
        max_lags = np.where(conv >= M - tol)[0]

        # gradient of max‑lag(s): G = Σ_l 2·a shifted by lag l
        G = np.zeros_like(a)
        for lag in max_lags:
            start = max(0, lag - (n - 1))
            end = min(lag, n - 1)
            if start <= end:
                idx = np.arange(start, end + 1)
                G[idx] += 2.0 * a[lag - idx]

        # source = index with large gradient and non‑negligible mass
        mask_pos = a > 1e-14
        if not np.any(mask_pos):
            break
        src_candidates = np.where(mask_pos)[0]
        src = src_candidates[np.argmax(G[src_candidates])]

        # destination = index with smallest gradient
        dst = np.argmin(G)
        if src == dst:
            # pick the second‑best destination if needed
            order = np.argsort(G)
            for d in order:
                if d != src:
                    dst = d
                    break

        # step size – we move a modest fraction of the source mass
        remaining = deadline - time.time()
        frac = 0.2 * (remaining / (deadline - start_time)) + 0.01   # shrinks linearly
        delta = frac * a[src]
        if delta < 1e-15:
            break

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] subgrad src={src} dst={dst} "
                      f"δ={delta:.2e}  best={new_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Very light simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        rng = np.random.default_rng()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        total_moves = 30_000

        for t in range(1, total_moves + 1):
            if time.time() >= deadline:
                break
            src = rng.choice(n, p=a)               # pick source proportional to mass
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps = a[src] * rng.random() * 0.10      # move ≤10 % of source
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={t:,} "
                          f"best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
