def _anneal(a0, deadline, start_time, L_fft):
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a)
    cur_score = best_score

    rng = np.random.default_rng()
    T0 = 0.1 * best_score  # relative scale
    T = T0
    decay = 0.9995  # per iteration
    iter = 0
    while time.time() < deadline:
        iter += 1
        # choose src weighted by a
        src = rng.choice(len(a), p=a/ a.sum())
        dst = rng.integers(len(a))
        if src == dst:
            continue
        # epsilon magnitude: random fraction up to max_move
        max_move = 0.3 * a[src]
        if max_move <= 0:
            continue
        eps = rng.random() * max_move
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # project to simplex? Since we moved mass within the simplex, sum unchanged, non-neg ensured as src had enough mass; dst non-negative automatically.
        # However a_new may have small negative due to rounding if eps > a[src]; but we limited.
        # Evaluate
        new_score = _objective(a_new)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                print(f"[{time.time()-start_time:.2f}s] Anneal iteration {iter} best={best_score:.6f} T={T:.2e}")
        T = max(T * decay, 1e-8)
    return best, best_score
```

def _anneal(a0, deadline, start_time, L_fft):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a, L_fft)
    cur_score = best_score
    rng = np.random.default_rng()
    n = a.shape[0]
    # parameters
    T = best_score * 0.05  # initial temperature relative to score
    T_min = 1e-8
    decay = 0.9995
    move_factor = 0.4
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute max conv index
        conv = _conv_self(a, L_fft)
        k_max = int(conv.argmax())
        # gradient contributions
        j = k_max - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # source and destination
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        # If src and dst same or src mass negligible, fallback to random move
        if src == dst or a[src] < 1e-12:
            src = rng.choice(n, p=a/ a.sum())
            dst = rng.integers(n)
            # continue
        max_move = move_factor * a[src]
        if max_move <= 0:
            continue
        eps = rng.random() * max_move
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # compute objective
        new_score = _objective(a_new, L_fft)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={iteration} best={best_score:.6f} T={T:.2e}")
        # temperature decay
        T = max(T * decay, T_min)
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of:
1. A fast exploration of several parametric families (geometric,
   power, beta, cosine‑window, triangular) with mixing with the uniform
   distribution.
2. A simulated‑annealing phase that moves mass from indices that
   contribute most to the current maximal convolution value towards
   indices that contribute least.  Both downhill and occasional uphill
   moves are accepted.
3. A cheap hill‑climb polish that performs tiny random mass transfers
   while only keeping improvements.

The whole routine respects the given time budget, prints progress and
returns a list suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self(a, L_fft)
    maxb = conv.max()
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Parametric families exploration
# ----------------------------------------------------------------------
def _search_param_families(n: int, L_fft: int,
                          deadline: float, start_time: float,
                          best_seq: np.ndarray, best_score: float):
    """Quick sweep over several analytic families, keeping the best."""
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    def _maybe_update(seq: np.ndarray, descr: str):
        nonlocal best_seq, best_score
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0:
            return
        seq = seq / seq.sum()
        sc = _objective(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] {descr} score={sc:.6f}")

    # geometric families
    eps_grid = np.logspace(-4, -0.1, 50)
    for eps in eps_grid:
        if time.time() > deadline - 30:
            break
        r = 1.0 - eps
        geo = r ** idx
        for lam in (0.0, 0.5, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            _maybe_update(seq, f"Geo eps={eps:.2e} lam={lam}")

    # power families
    for p in np.linspace(-1.0, 2.0, 30):
        if time.time() > deadline - 30:
            break
        pow_seq = (idx + 1) ** p
        _maybe_update(pow_seq, f"Power p={p:.2f}")

    # beta‑type families
    beta_ps = [0.0, 0.5, 1.0, 1.5, 2.0, 3.0]
    for p in beta_ps:
        for q in beta_ps:
            if time.time() > deadline - 30:
                break
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            _maybe_update(beta_seq, f"Beta p={p:.1f} q={q:.1f}")

    # cosine‑window families (Hann‑like)
    cos_shape = (1.0 - np.cos(np.pi * idx / (n - 1))) / 2.0
    for p in np.linspace(0.5, 4.0, 30):
        if time.time() > deadline - 30:
            break
        seq = cos_shape ** p
        _maybe_update(seq, f"Cosine p={p:.2f}")

    # triangular families
    tri_shape = np.maximum(
        0.0,
        1.0 - np.abs(idx - (n - 1) / 2.0) / ((n - 1) / 2.0 + 1.0),
    )
    for p in np.linspace(0.5, 4.0, 30):
        if time.time() > deadline - 30:
            break
        seq = tri_shape ** p
        _maybe_update(seq, f"Triangular p={p:.2f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated annealing with gradient‑guided moves
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray,
            deadline: float,
            start_time: float,
            L_fft: int) -> tuple[np.ndarray, float]:
    """Gradient‑guided simulated annealing."""
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a, L_fft)
    cur_score = best_score
    n = a.shape[0]
    rng = np.random.default_rng()

    # temperature schedule
    T = best_score * 0.05  # start with a modest fraction of the score
    T_min = 1e-8
    decay = 0.9995
    move_factor = 0.4  # max fraction of source mass moved in one step
    it = 0

    while time.time() < deadline:
        it += 1

        # convolution and location of the current maximum
        conv = _conv_self(a, L_fft)
        k_max = int(conv.argmax())

        # sub‑gradient of the max‑term w.r.t. a
        j = k_max - np.arange(n)
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # source = index with largest contribution, destination = smallest
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # fallback to a random move if the gradient gives a degenerate pair
        if src == dst or a[src] < 1e-14:
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(n)

        # propose a transfer of a fraction of src's mass
        max_move = move_factor * a[src]
        if max_move <= 0:
            continue
        eps = rng.random() * max_move
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        new_score = _objective(a_prop, L_fft)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it} best={best_score:.6f} T={T:.2e}")

        # temperature decay
        T = max(T * decay, T_min)

    return best, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (only improvements)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int,
                max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Local search performing tiny random transfers, keeping improvements."""
    a = a0.copy()
    best_score = _objective(a, L_fft)
    best = a.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.05  # move up to 5 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best = a_prop.copy()
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    metric.  The routine mixes a fast parametric sweep, a gradient‑guided
    simulated annealing, and a final hill‑climb polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    n = 4096                               # length of the sequence
    L_fft = _next_pow2(2 * n - 1)          # FFT length for convolutions

    # --------------------------------------------------
    # Baseline (uniform) and optional global starter
    # --------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # use a previously discovered construction if it exists
    starter = globals().get("height_sequence_1")
    if starter is not None:
        try:
            starter = np.asarray(starter, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() == 0:
                    starter = uniform.copy()
                else:
                    starter = starter / starter.sum()
                cur = _objective(starter, L_fft)
                if cur < best_score:
                    best_seq = starter.copy()
                    best_score = cur
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
        except Exception:
            pass

    # --------------------------------------------------
    # 1️⃣  Fast parametric sweep
    # --------------------------------------------------
    best_seq, best_score = _search_param_families(
        n, L_fft, deadline, start_time, best_seq, best_score
    )

    # --------------------------------------------------
    # 2️⃣  Simulated annealing (gradient‑guided)
    # --------------------------------------------------
    if time.time() < deadline - 20.0:
        anneal_deadline = deadline - 10.0
        best_seq, best_score = _anneal(best_seq, anneal_deadline, start_time, L_fft)

    # --------------------------------------------------
    # 3️⃣  Tiny hill‑climb polishing
    # --------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline, start_time, L_fft)

    # --------------------------------------------------
    # Normalisation and final output
    # --------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")

    # return a plain python list as required by the evaluator
    return [float(x) for x in best_seq]
