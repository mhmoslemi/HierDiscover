"""
Propose a step function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where conv(a,a) is the linear self‑convolution.  The algorithm uses
a smoothed “log‑sum‑exp’’ approximation of the max‐operator and
gradient descent with Euclidean projection onto the probability simplex.
A simple simulated‑annealing polish is performed at the end.
The whole routine respects a 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Identical to evaluate_sequence (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """
    Compute weights w_i = exp((conv_i - max(conv))/eps) / Σ_j exp(...).
    For very small eps the distribution concentrates on the maximal entry.
    """
    if eps <= 0:
        eps = 1e-12
    max_c = conv.max()
    # stability: subtract max_c
    exp_vals = np.exp((conv - max_c) / eps)
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        # fallback to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / sum_exp


def _grad_from_weights(weights: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_i w_i·conv_i(a) w.r.t. a.
    This equals 2 * (weights ⋆ a)[0:n] where ⋆ is the (linear) cross‑correlation.
    Implemented efficiently by FFT as a convolution with the reversed a.
    """
    n = a.shape[0]
    # reverse a
    a_rev = a[::-1]
    # length of full convolution of weights (len 2n-1) and a_rev (len n)
    L = _next_pow2(len(weights) + len(a_rev) - 1)
    Wf = np.fft.rfft(weights, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(Wf * Af, n=L)[: len(weights) + len(a_rev) - 1]
    # the required slice corresponds to indices n-1 … n-1+n-1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad
