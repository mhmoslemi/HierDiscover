"""
Propose a low‑score step function.

Algorithm
---------
1.  Choose a relatively large length (default 16384).  Build a family of
    U‑shaped “arcsine‑type” vectors
        a_i ∝ (x_i·(1‑x_i)+ε)^{‑α}
    with a fine grid of exponents α∈[0.45,0.55] and a few tiny regularisers
    ε.  The best candidate according to the exact objective is kept.

2.  Refine the best candidate with a smooth approximation of the
    non‑smooth objective:
        f(a)=2·n·max(conv(a,a))
    The max is replaced by a log‑sum‑exp with an increasing temperature
    β.  The gradient is obtained analytically (via FFT‑based correlation)
    and an Adam optimiser with projection onto the probability simplex
    is applied for the remaining budget.

3.  A short projected sub‑gradient descent (momentum) is run as a final
    polish, followed by a tiny simulated‑annealing sweep.

All steps respect the 1000 s wall‑time limit, never produce negative or
overly large values, and finally return a normalised list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation t_i = Σ_k w_k * a_{k‑i}.
    a : length n
    w : length 2n‑1
    Returns vector t of length n.
    """
    n = a.shape[0]
    # correlation = convolution of a with reversed w
    w_rev = w[::-1]
    L = _next_pow2(len(a) + len(w_rev) - 1)
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(A * W_rev, n=L)[: len(a) + len(w_rev) - 1]
    # the required indices are 2n‑2‑i   (i=0…n‑1)
    start = len(w_rev) - 1          # = 2n‑2
    t = conv[start - np.arange(n)]
    return t


# ----------------------------------------------------------------------
# Exact objective (the verifier’s metric)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the verifier: 2·n·max(conv(seq,seq))/‖seq‖₁²."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _grad_lse(a: np.ndarray, beta: float):
    """
    Gradient of the smoothed max (log‑sum‑exp) of a*a.
    Returns (grad, max_conv) where grad is ∂LSE/∂a.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_c = conv.max()
    # stable soft‑max weights
    exp_shift = np.exp(beta * (conv - max_c))
    w = exp_shift / exp_shift.sum()               # shape (2n‑1,)

    # t_i = Σ_k w_k * a_{k‑i}
    t = _correlation(a, w)

    # correction for the diagonal terms where 2i = k
    diag = w[2 * np.arange(n)] * a

    grad = 2.0 * t - diag
    return grad, max_c


def _adam_optimize(a0: np.ndarray,
                   beta_start: float,
                   beta_max: float,
                   step0: float,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed problem.
    Keeps the iterate on the simplex and records the best exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # schedule for β (inverse temperature)
    beta_inc_iters = 2000          # after this many steps we have reached β_max

    while time.time() < deadline:
        it += 1
        # linear increase of β for the first beta_inc_iters iterations
        if it <= beta_inc_iters:
            beta = beta_start + (beta_max - beta_start) * it / beta_inc_iters
        else:
            beta = beta_max

        grad_lse, cur_max = _grad_lse(a, beta)

        # gradient of the (smoothed) objective = 2·n·grad_lse
        g = 2.0 * n * grad_lse

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # keep on the simplex
        a = _project_simplex(a)

        # evaluate exact objective occasionally
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam iter={it:5d} "
                f"β={beta:.1f} score={cur_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (as in the reference implementation)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Simple momentum sub‑gradient descent on the exact objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum coefficient
    step0 = 0.25        # initial step size factor
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] "
                f"sub‑grad iter={it:,} score={cur_score:.6f}"
            )

        # index where the maximum is attained
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score

def _anneal_polish(a0: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """Few thousand random mass‑transfer moves with Metropolis acceptance."""
    a = a0.copy()
    cur_score = _objective(a)
    best_local = a.copy()
    best_local_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]
    total_steps = 20000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 20 % of the source mass
        eps_move = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}"
                )

    if best_local_score < cur_score:
        return best_local
    return a

                    best_score = cur_score
                    best_seq = starter.copy()
        # If the starter is longer/shorter, pad or truncate to current n
        if best_seq.shape[0] != n:
            # simple projection onto length‑n simplex (keep first n entries)
            tmp = np.zeros(n, dtype=float)
            L = min(n, best_seq.shape[0])
            tmp[:L] = best_seq[:L]
            best_seq = _project_simplex(tmp)
    except Exception:
        pass  # no global starter – ignore

    # ------------------------------------------------------------------
    # 5) Adam‑based smooth optimisation (majority of remaining budget)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    remaining = deadline - time.time()
    if remaining > 15.0:
        # allocate ~60 % of the remainder to Adam
        adam_deadline = time.time() + 0.6 * remaining
        refined_seq, refined_score = _adam_optimize(
            best_seq,
            beta_start=5.0,
            beta_max=200.0,
            step0=0.08,
            deadline=adam_deadline,
            start_time=start_time,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam refined score={refined_score:.6f}"
            )
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 6) Projected sub‑gradient descent polish
    # ------------------------------------------------------------------
    if remaining > 10.0:
        sub_deadline = time.time() + 0.5 * remaining
        sub_seq, sub_score = _subgradient_descent(
            best_seq, sub_deadline, start_time
        )
        if sub_score < best_score:
            best_score = sub_score
            best_seq = sub_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad polish score={sub_score:.6f}"
            )
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 7) Light simulated‑annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polished = _anneal_polish(best_seq, deadline, start_time)
        polished_score = _objective(polished)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished
            print(
                f"[{time.time() - start_time:.2f}s] final anneal improve score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 8) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # The verifier only cares about non‑negative numbers; scaling does not matter.
    return [float(v) for v in best_seq]
