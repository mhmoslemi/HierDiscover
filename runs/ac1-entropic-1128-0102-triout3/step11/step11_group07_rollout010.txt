"""
Optimisation of a non‑negative step‑function for the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a combination of:
* a fast random parametric search (beta, Gaussian, triangular, cosine windows)
* a long projected exponential‑gradient (mirror‑descent) optimiser with a
  log‑sum‑exp smooth approximation of the max,
* a refined guided‑move sub‑gradient routine,
* a tiny smoothing / hill‑climbing polish.

The code respects a 1000 s budget and works with the supplied
``evaluate_sequence`` function (which is not imported here).
All helper routines are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# optional SciPy – used only for the LP phase (fallback if unavailable)
try:
    from scipy.optimize import linprog
    _has_scipy = True
except Exception:  # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum ≈ 1). Returns the full objective.
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _beta_seq(idx: np.ndarray, p: float, q: float) -> np.ndarray:
    seq = (np.power(idx + 1, p) * np.power(idx[-1] - idx, q))
    return seq


def _gaussian_seq(idx: np.ndarray, mu: float, sigma: float) -> np.ndarray:
    seq = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    return seq


def _triangular_seq(idx: np.ndarray, centre: float, halfwidth: float) -> np.ndarray:
    dist = np.abs(idx - centre)
    seq = np.maximum(0.0, (halfwidth - dist) / halfwidth)
    return seq


def _raised_cosine_seq(idx: np.ndarray, centre: float, width: float) -> np.ndarray:
    # Cosine bump of total width `width`; zero outside.
    dist = np.abs(idx - centre)
    inside = dist <= width / 2.0
    seq = np.zeros_like(idx, dtype=np.float64)
    seq[inside] = 0.5 * (1.0 + np.cos(2.0 * np.pi * (dist[inside]) / width))
    return seq


# ----------------------------------------------------------------------
# Exponential‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _expgrad_optimize(start_a: np.ndarray,
                      deadline: float,
                      start_time: float,
                      L_fft: int,
                      L_grad: int,
                      tau0: float = 0.1,
                      tau_decay: float = 0.9999,
                      eta0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient (mirror descent) on the simplex.
    Uses a log‑sum‑exp smoothing of the max with temperature `tau`.
    """
    n = start_a.shape[0]
    # work in log‑space to stay in the simplex after exponentiation + normalisation
    log_a = np.log(start_a + 1e-16)
    best_log = log_a.copy()
    best_score = _eval_fast(start_a, L_fft)

    it = 0
    tau = tau0
    eta = eta0

    while time.time() < deadline:
        it += 1
        # reconstruct a from log_a
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # convolution and soft‑max weights
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        exp_vals = np.exp((conv - M) / max(tau, 1e-12))
        w = exp_vals / exp_vals.sum()          # soft‑max over all shifts

        # gradient of the smoothed max w.r.t. a:
        # grad = 2 * (a *reversed w)  (as in the original code)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # exponentiated (mirror) update
        log_a = log_a - eta * grad

        # normalise back onto simplex (using log‑sum‑exp trick)
        # we simply exponentiate and renormalise – cheap and works.
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        log_a = np.log(a_new + 1e-16)

        # evaluate
        cur_score = _eval_fast(a_new, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_log = log_a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGrad it={it:,} τ={tau:.2e} η={eta:.3f} score={cur_score:.6f}")

        # decay temperature & step size slowly
        tau = max(1e-8, tau * tau_decay)
        eta = max(1e-4, eta * 0.9995)

    best_a = np.exp(best_log)
    best_a = best_a / best_a.sum()
    return best_a, best_score


# ----------------------------------------------------------------------
# Guided sub‑gradient move (same as in the reference code)
# ----------------------------------------------------------------------
def _guided_move(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 L_fft: int,
                 max_moves: int = 5000) -> tuple[np.ndarray, float]:
    """Move mass from the highest‑gradient index to the lowest‑gradient one."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        conv = _conv_self(a, L=L_fft)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of the max‑term

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * 0.15                # move up to 15 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] GuidedMove it={it:,} score={new_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (random local search)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Very cheap local search: random tiny transfers accepted only if they improve."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (kept from the reference – activates when SciPy is present)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 6,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Iteratively improve a by solving a small LP on the active constraints."""
    if not _has_scipy:                     # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])
        K = active.size

        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)   # maximise sum(g) -> minimise -sum(g)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})

        if not res.success:                 # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()

        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)
        if new_score < best_score:
            best_score = new_score
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP refine it={it} score={new_score:.6f}")
        else:
            break

    return a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    n = 4096

    L_grad = _next_pow2(3 * n)                     # for gradient convolutions

    # ------------------------------------------------------------------
    # Baseline (uniform)

    best_seq = uniform.copy()

    # ------------------------------------------------------------------

    try:

            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:                               # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (beta, Gaussian, triangular, cosine)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(seed + 12345)

        elif family == 'tri':

            halfwidth = rng.uniform(n / 200.0, n / 2.0)
            seq = _triangular_seq(idx, centre, halfwidth)
        else:  # cosine bump

            width = rng.uniform(n / 50.0, n / 3.0)
            seq = _raised_cosine_seq(idx, centre, width)

        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0:
            continue
        seq /= total
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] ParamSweep {family} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Exponential‑gradient optimiser (long run)

    while time.time() < deadline - 200.0:

        start = np.maximum(start, 0.0)

        sub_deadline = min(deadline - 5.0, time.time() + 30.0)  # up to 30 s per run
        cand_seq, cand_score = _expgrad_optimize(start, sub_deadline,
                                                 start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGrad new global best score={best_score:.6f}")

    # ------------------------------------------------------------------

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
