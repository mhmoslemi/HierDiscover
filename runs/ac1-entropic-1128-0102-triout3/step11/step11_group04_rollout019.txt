    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2*n-1]

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm – max‑convolution aware hill‑climb
------------------------------------------------
1. Load the best known sequence (``height_sequence_1``) if available,
   otherwise start from the uniform distribution.
2. Re‑normalise to sum = 1 (the objective is scale‑invariant).
3. Repeatedly:
   * Compute the full linear self‑convolution using an FFT.
   * Locate the index ``k_max`` where the convolution attains its
     current maximum.
   * For this lag compute the *pair contributions* ``p_i = a[i] *
     a[k_max‑i]`` (only for indices that actually participate).
   * Pick the source index ``src`` with the largest contribution –
     moving mass away from it reduces the dominant peak most.
   * Choose a destination index ``dst`` whose partner ``k_max‑dst`` is
     (almost) zero, i.e. adding mass there does not increase the peak.
   * Transfer a small random fraction (up to 20 %) of the source mass to
     the destination, re‑normalise and keep the move only if the true
     objective improves.
   * Keep the best sequence seen so far.
4. The loop runs until the wall‑clock deadline (≈1000 s) is reached.
5. Return the best sequence found (as a plain Python list of floats).

The method is deliberately simple, cheap per iteration and focuses the
search on the actual bottleneck of the problem – the maximal
convolution entry – which makes it able to push the bound a little
further than a generic hill‑climb.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, power‑of‑two)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the full linear convolution of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Exact evaluation of the objective (sum is assumed ≈ 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self_fft(a, L)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Max‑convolution aware hill‑climb
# ----------------------------------------------------------------------
def _maxconv_hill_climb(a0: np.ndarray,
                        n: int,
                        L: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Targeted hill‑climb:
    - always move mass from the index that contributes most to the current
      maximal convolution value,
    - to a destination that (almost) does not increase that value.
    Returns the best sequence encountered and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a, n, L)
    cur_score = best_score

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # full convolution and location of the current peak
        conv = _conv_self_fft(a, L)
        k = int(np.argmax(conv))

        # indices that actually participate in this lag
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx                      # partner = k‑i, always valid here

        # contributions p_i = a_i * a_{k-i}
        contrib = a[idx] * a[partner]
        src = idx[np.argmax(contrib)]          # source = index with biggest product

        # Choose a destination that minimally increases the peak.
        # Sample a modest set of candidates and pick the best.
        cand = rng.choice(n, size=20, replace=False)
        cand = cand[cand != src]               # avoid self‑move
        if cand.size == 0:
            continue

        # partner value for each candidate (zero if out of range)
        partner_vals = np.empty_like(cand, dtype=float)
        # indicate whether k‑dest lies inside [0, n‑1]
        inside = (k - cand >= 0) & (k - cand < n)
        partner_vals[inside] = a[k - cand[inside]]
        partner_vals[~inside] = 0.0
        dst = cand[np.argmin(partner_vals)]

        # Random transfer amount (up to 20 % of the source mass)
        max_transfer = 0.20
        delta = a[src] * rng.random() * max_transfer
        if delta <= 1e-15:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # Guard against negative numbers caused by rounding
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total <= 0.0:
            continue
        a_new /= total   # keep the sum exactly 1

        new_score = _objective(a_new, n, L)

        # Keep only improving moves (a strict descent)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"New global best = {best_score:.6f}")

        it += 1
        # occasional break‑check to avoid overshooting the deadline
        if it % 5000 == 0 and time.time() > deadline - 0.5:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the external harness
    deadline = start_time + budget_s - 0.5

    n = 4096
    L = _next_pow2(2 * n - 1)          # FFT length for self‑convolution

    # ------------------------------------------------------------------
    # Load a starter vector (global best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adapt length if necessary
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _objective(best_seq, n, L)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Targeted hill‑climb – the core optimisation phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        # run the climb for the remaining wall‑clock time
        best_seq, best_score = _maxconv_hill_climb(best_seq, n, L,
                                                  deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] Finished climbing – best "
              f"score ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
