"""
Propose a low‑score step function.

Algorithm
---------
1. **Initial candidate** – sweep a simple arcsine‑type family
   a_i ∝ (x_i·(1−x_i)+ε)^{−α} on a dense grid of exponents α∈[0.30,0.49]
   (plus a few random starts and the optional global starter) and keep the
   best among them.

2. **Smooth‑max gradient descent** – minimise a smoothed version of the
   objective  2·n·max(conv(a,a))/‖a‖₁².
   The smoothed max is  T·log∑exp(conv/T).  Its gradient is computed
   analytically and the update is performed on log‑probabilities
   (exponentiated‑gradient / mirror descent) so the iterate always stays on
   the probability simplex.  The temperature T is annealed from 0.2 down to
   1e‑4.

3. **Light simulated‑annealing polish** – a few‑thousand cheap mass‑transfer
   moves are attempted; a move is kept if it improves the exact objective
   (or is accepted with a tiny Metropolis probability).

All steps respect the given wall‑time budget; the best sequence seen
throughout the run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the verifier."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _softmax(x: np.ndarray, T: float) -> np.ndarray:
    """Numerically stable softmax of x/T."""
    # shift by max to avoid overflow
    y = (x - np.max(x)) / T
    e = np.exp(y)
    return e / e.sum()


def _grad_smooth(a: np.ndarray, T: float) -> (np.ndarray, float):
    """
    Gradient of the smoothed objective (smooth max with temperature T)
    w.r.t. the probability vector a.
    Returns (gradient, max_convolution_value_before_smoothing).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n-1
    w = _softmax(conv, T)                    # softmax weights, sum to 1
    a_rev = a[::-1]                          # reversed a
    L = _next_pow2(len(w) + len(a_rev) - 1)  # length of convolution
    # convolution w * a_rev (full length = 3n-2)
    conv_full = np.fft.irfft(np.fft.rfft(w, L) * np.fft.rfft(a_rev, L), n=L)
    # gradient = 4·n·(w * a_rev) evaluated at positions n‑1 … n‑1+n‑1
    start = n - 1
    grad = 4.0 * n * conv_full[start:start + n]
    return grad, conv.max()
