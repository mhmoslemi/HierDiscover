        elapsed = time.time() - start_time

"""
Search for a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds in several stages:

1. Initialise from a global starter (if present) or from the uniform vector.
2. Run a very fast parametric sweep over geometric, power and beta families
   – this already gives a noticeable improvement over the uniform baseline.
3. A long simulated‑annealing phase that repeatedly transfers a random
   amount of mass from one index to another.  The Metropolis criterion
   allows occasional uphill moves so that the search can escape local
   minima.
4. A greedy sub‑gradient descent that continuously moves mass from the
   index that contributes most to the current maximal convolution to the
   index that contributes least.  The step size is slowly reduced.
5. An active‑set linear‑program refinement (the same LP used in the
   reference implementation) is run until it stops improving the bound.
6. A short Adam‑style smooth‑max optimisation is used as a final polish.

All stages respect the overall time budget (default 1000 s).  The returned
list is compatible with ``evaluate_sequence`` and satisfies the required
non‑negative/normalisation constraints.
"""

import time
import math
import numpy as np

# optional SciPy imports – if unavailable the LP phase is skipped
try:
    from scipy.optimize import linprog
    _has_scipy = True
except Exception:                     # pragma: no cover

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear (full) self‑convolution a*a using FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑style smoothed gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max smoothed objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    tau = 0.05                # initial temperature for soft‑max
    tau_min = 1e-8
    tau_decay = 0.9995
    step0 = 0.15

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient via FFT: grad_j = 2 * Σ_i a_i w_{i+j}
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update & projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Active‑set linear‑program refinement (LP)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 40,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Iteratively improve a by solving a small LP on the active constraints.
    The routine stops when no improvement is observed or the time budget
    expires.
    """
    if not _has_scipy:                     # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])
        K = active.size

        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1, dtype=int)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(g)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(1, int(deadline - time.time()))})

        if not res.success:                     # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()                 # re‑normalise to sum‑1

        # convex combination (t = 0.5)
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)
        if new_score < best_score:
            best_score = new_score
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP refine it={it} score={new_score:.6f}")
        else:
            # no progress – stop early
            break

    return a, best_score

# Greedy sub‑gradient move (many more iterations than the original code)
# ----------------------------------------------------------------------
def _guided_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move mass from the index that contributes most to the
    currently maximal convolution to the one that contributes least.
    The step size (fraction of source mass) is gradually reduced.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    step_frac = 0.20          # start with a fairly aggressive step
    min_frac = 0.001
    decay = 0.9995

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L=L_fft)
        k = int(conv.argmax())                     # shift attaining the max
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the maximal term
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] < 1e-14:
            # nothing to move – reduce step size a little and continue
            step_frac = max(min_frac, step_frac * decay)
            continue

        eps = a[src] * step_frac * np.random.rand()
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] Guided it={it:,} score={new_score:.6f}")
            # when improvement is found we keep the current step size
        else:
            # no improvement – shrink the step
            step_frac = max(min_frac, step_frac * decay)

    return a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer search
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         init_temp: float = 0.02,
                         max_eps_frac: float = 0.20,
                         min_eps_frac: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Classic Metropolis‑style annealing.  Mass is transferred from one index
    to another; uphill moves are accepted with probability exp(-Δ/T).  The
    temperature is linearly cooled to zero.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)
    best_seq = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total_time = deadline - start_time
        frac = min(1.0, elapsed / total_time)
        T = init_temp * (1.0 - frac)                 # linear cooling

        # pick source proportionally to its mass (more likely to move large chunks)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # step size follows the temperature: large at high T, tiny at low T
        eps_frac = min_eps_frac + (max_eps_frac - min_eps_frac) * (T / init_temp)
        eps = a[src] * eps_frac * rng.random()
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # projection unnecessary because mass is conserved and eps≤a[src]
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score

        if delta < 0 or (T > 1e-12 and rng.random() < math.exp(-delta / T)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[{elapsed:.2f}s] SA it={it:,} new best {best_score:.6f}")

        # occasional random perturbation if we have been stuck for a while
        if it % 25000 == 0 and time.time() - last_improve > 30.0:
            noise = rng.normal(scale=0.02, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            cur_score = _eval_fast(a, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[{time.time() - start_time:.2f}s] SA perturb improved to {best_score:.6f}")

    return best_seq, best_score

            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type family – richer grid
    beta_vals = np.linspace(0.0, 3.0, 31)
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 40:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            s = beta_seq.sum()
            if s == 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Simulated‑annealing phase (large portion of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    deadline,
                                                    start_time,
                                                    L_fft,
                                                    init_temp=0.02)

    # ------------------------------------------------------------------
    # 3️⃣ Greedy sub‑gradient descent (many more moves than the original)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print(f"[{time.time() - start_time:.2f}s] Starting guided descent")
        best_seq, best_score = _guided_descent(best_seq,
                                               deadline,
                                               start_time,
                                               L_fft,
                                               max_iters=500_000)

    # ------------------------------------------------------------------
    # 4️⃣ Active‑set LP refinement (more iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print(f"[{time.time() - start_time:.2f}s] Starting LP refinement")
        best_seq, best_score = _lp_refine(best_seq,
                                          deadline,
                                          start_time,
                                          L_fft,
                                          max_iters=30)

    # ------------------------------------------------------------------
    # 5️⃣ Final Adam‑style polish (short)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(10.0, remaining)
        print(f"[{time.time() - start_time:.2f}s] Final Adam smoothing")
        cand_seq, cand_score = _adam_smooth_descent(best_seq,
                                                    sub_deadline,
                                                    start_time,
                                                    L_fft,
                                                    L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam polish improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalisation and final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
