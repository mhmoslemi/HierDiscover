def _next_pow2(x):

def _conv(seq, L=None):
    n = seq.shape[0]
    if L is None:
        L = _next_pow2(2*n-1)
    a_fft = np.fft.rfft(seq, n=L)

def _objective(seq):
    # ensure 1D numpy array
    conv = _conv(seq)  # shape 2n-1
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float('inf')
    n = seq.shape[0]
    return float(2.0 * n * max_conv / (sum_seq*sum_seq))

def _smooth_grad(seq, beta, L):
    n = seq.shape[0]
    # compute conv
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]

    return [float(v) for v in best_seq]

"""
A multi‑stage optimizer for the step‑function problem.

The algorithm consists of:

1. **Broad parametric sweep** – several analytically‑motivated families
   (Beta‑type, sin‑type, symmetric power‑law) are sampled on a fine grid.
   The best candidate (lowest exact objective) is kept.

2. **Adam on a smooth surrogate** – the non‑smooth max‑convolution term
   is replaced by a log‑sum‑exp (soft‑max) with an annealing parameter β.
   Adam with simplex projection drives the sequence to a low‑objective
   region.  β is increased from 5 to 500 during the run.

3. **Hard‑max sub‑gradient refinement** – once β is large,
   the exact sub‑gradient of the true objective (based on the
   index of the maximum convolution) is used with a decaying step size.

4. **Light simulated‑annealing polish** – random mass‑transfer moves
   are accepted only if they improve the exact objective.

All phases respect the 1000 s wall‑time limit and keep the vector on the
probability simplex (non‑negative, sum = 1).  The final sequence is
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of seq with itself (full length 2·n‑1)."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    return conv_full[: 2 * n - 1]


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    """
    n = seq.shape[0]
    conv = _conv(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _smooth_gradient(seq: np.ndarray, beta: float, L: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        2·n/β·log Σ exp(β·conv)  (denominator Σ a = 1)
    Returns the (unnormalised) gradient 4·n·B·a where
    B·a = correlation of the soft‑max weights with the reversed sequence.
    """
    n = seq.shape[0]

    # self‑convolution
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    # soft‑max over the convolution values
    beta_conv = beta * conv
    max_beta = beta_conv.max()
    exp_shift = np.exp(beta_conv - max_beta)
    w = exp_shift / exp_shift.sum()          # shape (2n‑1,)

    # correlation w * a_rev  →  B·a
    a_rev = seq[::-1]
    w_fft = np.fft.rfft(w, n=L)
    a_rev_fft = np.fft.rfft(a_rev, n=L)
    corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)

    Ba = corr_full[n - 1:n - 1 + n]          # length n
    grad = 4.0 * n * Ba
    return grad


def _maxgrad_and_maxconv(seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient of the original objective (hard max).
    Returns (gradient, max_convolution_value).
    """
    n = seq.shape[0]
    conv = _conv(seq)
    K = int(np.argmax(conv))                # index of the maximal entry
    max_val = float(conv[K])

    # gradient:  d maxconv / d a_j = 2·a_{K-j} (if within bounds)
    idx = np.arange(n)
    mask = (K - idx >= 0) & (K - idx < n)
    grad = np.zeros_like(seq)
    grad[mask] = 4.0 * n * seq[K - idx[mask]]   # factor 4·n (see derivation)
    return grad, max_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    `evaluate_sequence`.  The search proceeds through a parametric sweep,
    an Adam optimisation on a smoothed objective, a hard‑max sub‑gradient
    refinement and a short simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # problem size – a power of two works well with the FFTs
    n = 4096

    # --------------------------------------------------------------
    # 1) Broad parametric sweep over several families
    # --------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions

    best_seq = None
    best_score = float("inf")

    # ----- families -------------------------------------------------
    families = [
        ("beta", {"alphas": np.linspace(0.30, 0.70, 41),
                  "epsilons": [1e-12, 1e-8, 1e-4]}),
        ("sin", {"alphas": np.linspace(0.20, 0.80, 31),
                 "epsilons": [1e-12, 1e-8, 1e-4]}),
        ("pow", {"alphas": np.linspace(0.20, 0.80, 31),
                 "epsilons": [1e-12, 1e-8, 1e-4]})
    ]

    for name, params in families:
        alphas = params["alphas"]
        epsilons = params["epsilons"]
        for alpha in alphas:
            for eps in epsilons:
                if name == "beta":
                    w = (x * (1.0 - x) + eps) ** (-alpha)
                elif name == "sin":
                    w = (np.sin(np.pi * x) + eps) ** (-alpha)
                else:   # "pow"
                    w = (x + eps) ** (-alpha) + ((1.0 - x) + eps) ** (-alpha)

                a = w / w.sum()
                # safety – replace NaNs / Infs (possible for eps≈0)
                a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
                total = a.sum()
                if total == 0:
                    continue
                a = a / total

                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
            # early‑exit if time is gone
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2) Use a global starter if supplied
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    cur = _objective(starter)
                    if cur < best_score:
                        best_score = cur
                        best_seq = starter.copy()
        except Exception:
            pass

    # Fallback – uniform distribution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 3) Smoothed Adam optimisation (log‑sum‑exp surrogate)
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.02

    L = _next_pow2(3 * n)            # length sufficient for both convolutions
    beta_start, beta_end = 5.0, 500.0
    it = 0

    while time.time() < deadline - 2.0:    # leave a couple of seconds for later phases
        it += 1

        # ----- gradient of the smooth surrogate --------------------
        elapsed = time.time() - start_time
        prop = min(1.0, elapsed / (budget_s - 0.5))
        beta = beta_start + (beta_end - beta_start) * prop

        grad = _smooth_gradient(a, beta, L)

        # ----- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)

        # ----- occasional exact evaluation -------------------------
        if it % 100 == 0:
            cur = _objective(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{elapsed:.2f}s] Adam iter={it:,}  best={cur:.6f}")

    # --------------------------------------------------------------
    # 4) Hard‑max sub‑gradient refinement (fast, many tiny steps)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a = best_seq.copy()
        lr0 = 0.01
        for t in range(1, 50001):
            if time.time() >= deadline - 2.0:
                break
            grad, _ = _maxgrad_and_maxconv(a)
            lr = lr0 / math.sqrt(t)
            a = a - lr * grad
            a = _project_simplex(a)

            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Sub‑grad step={t:,}  best={cur:.6f}")

    # --------------------------------------------------------------
    # 5) Very light simulated‑annealing polish (mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_moves = 40000

        for mv in range(1, total_moves + 1):
            if time.time() >= deadline:
                break
            # source proportional to current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # move up to 8 % of the source mass
            eps = a[src] * rng.random() * 0.08
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Anneal move={mv:,}  best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalisation and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
