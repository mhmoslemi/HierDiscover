def _optimize(a0, deadline, start_time, alpha0=0.5, alpha_mul=1.02, alpha_max=200.0, step0=0.4, beta=0.9):
    n = len(a0)
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    # compute initial objective
    conv = _conv_self(a)
    best_score = float(2 * n * conv.max())
    it = 0
    alpha = alpha0
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Opt iter={it:,} score={cur_score:.6f}")

        # compute softmax weights
        shift = conv - cur_max  # max <=0
        # alpha may be large; use np.exp(alpha*shift)
        exp_terms = np.exp(alpha * shift)  # values <=1
        weight_sum = exp_terms.sum()
        if weight_sum == 0:
            # numeric underflow; fallback to uniform over max indices?
            # Use top-k approach
            top_idxs = np.argpartition(conv, -5)[-5:]
            grad = np.zeros_like(a)
            for k in top_idxs:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                grad[idx] += 2.0 * a[k - idx]
            grad /= len(top_idxs)
            # momentum etc...
        else:
            w = exp_terms / weight_sum
            # gradient via convolution
            a_rev = a[::-1]
            L = _next_pow2(len(w) + len(a_rev) - 1)
            wf = np.fft.rfft(w, n=L)
            af = np.fft.rfft(a_rev, n=L)
            grad_full = np.fft.irfft(wf * af, n=L)[: len(w) + len(a_rev) - 1]
            grad = 2.0 * grad_full[len(w)-1 : len(w)-1 + n]
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # project onto simplex
        a = _project_simplex(a)
        # update alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)
    return best_a, best_score

"""
Propose a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a fast smooth‑max gradient descent (Adam) together with a short
simulated‑annealing polish.  The smooth‑max replaces the max in the
objective by a log‑sum‑exp on a *scaled* convolution, which yields a
well‑behaved gradient.  Adam gives an adaptive step‑size and the
iterate is projected onto the probability simplex after each update.
Several simple warm‑starts (uniform, triangular, random Dirichlet) are
tried; the best result is kept.  Finally a cheap annealing phase moves
tiny random packets of mass between coordinates (Metropolis rule) to
exploit any remaining local improvement.

The whole search respects the given time budget (default 1000 s) and
returns a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Identical to evaluate_sequence (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smooth‑max of the self‑convolution.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.02                # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smooth‑max hyper‑parameters
    scale = float(2 * n - 1)          # scaling to bring conv into O(1) range
    alpha = 1.0                       # temperature (inverse smoothing)
    alpha_mul = 1.001                 # slow exponential increase
    alpha_max = 200.0

    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- current objective ------------------------------------------------
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # ----- smooth max weights ------------------------------------------------
        conv_scaled = conv * scale
        max_scaled = conv_scaled.max()
        shift = conv_scaled - max_scaled            # ≤ 0
        exp_terms = np.exp(alpha * shift)           # safe because shift ≤ 0
        w = exp_terms / exp_terms.sum()             # sum‑to‑1 weights

        # ----- gradient of the smooth max ---------------------------------------
        # We need   g_j = 2·scale·∑_i w_i·a_{i‑j}
        a_rev = a[::-1]
        L = _next_pow2(len(w) + len(a_rev) - 1)
        wf = np.fft.rfft(w, n=L)
        af = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(wf * af, n=L)[:len(w) + len(a_rev) - 1]
        grad = 2.0 * scale * grad_full[len(w) - 1: len(w) - 1 + n]

        # ----- Adam update -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # optional decay of the learning‑rate with √t
        step = lr0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep a feasible probability vector --------------------------------
        a = _project_simplex(a)

        # ----- increase the temperature (sharpen the max) -----------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap annealing that repeatedly moves a small random packet of mass.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # The number of proposals is deliberately high; the loop respects the deadline.
    it = 0
    while time.time() < deadline:
        it += 1
        # exponential cooling schedule (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / 40000.0)

        # draw a source proportionally to the current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 25 %) of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – a power of two yields fast FFTs
    # ------------------------------------------------------------------
    n = 16384

    # ------------------------------------------------------------------
    # generate a small pool of diverse warm‑starts
    # ------------------------------------------------------------------
    inits = []

    # uniform
    inits.append(np.full(n, 1.0 / n))

    # symmetric triangular (tent)
    x = np.arange(n, dtype=float) / (n - 1)
    tent = np.minimum(x, 1.0 - x)
    if tent.sum() > 0:
        inits.append(tent / tent.sum())
    else:
        inits.append(np.full(n, 1.0 / n))

    # random Dirichlet – moderately sparse
    rng = np.random.default_rng()
    inits.append(rng.dirichlet(np.full(n, 0.5)))

    # random Dirichlet – smoother
    inits.append(rng.dirichlet(np.full(n, 2.0)))

    # previous global best if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                inits.append(starter)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # optimise each warm‑start (time‑slicing)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for idx, a0 in enumerate(inits, 1):
        remaining = deadline - time.time()
        if remaining < 30.0:        # not enough time for a meaningful run
            break
        # allocate a fraction of the remaining budget to this start
        sub_deadline = time.time() + min(remaining * 0.6, 250.0)

        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation #{idx} "
              f"with budget {sub_deadline - time.time():.1f}s")
        seq, val = _adam_optimize(a0, sub_deadline, start_time)

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after Adam #{idx}: {val:.6f}")

    # ------------------------------------------------------------------
    # Light annealing polish (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing polish "
              f"with {deadline - time.time():.1f}s left")
        seq, val = _anneal(best_seq, deadline, start_time)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Global best after anneal: {val:.6f}")

    # ------------------------------------------------------------------
    # final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    if best_seq is None:
        # as a last resort fall back to uniform
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq = np.maximum(best_seq, 0.0)

    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
