"""
Propose a low‑score step function.

Algorithm
---------
1. **Broad parametric sweep** – generate many candidate shapes:
   * singular arcsine‑type: (x·(1‑x)+ε)^(-α)   with α∈[0.30,0.70]
   * interior‑peaked β‑polynomials: (x·(1‑x))**β   with β∈[0,5]
   * truncated Gaussians:  exp(-γ·(x‑0.5)²)    with γ∈[0,30]
   * sine‑powers:            sin(πx)**p          with p∈[0,10]
   * mixtures of arcsine & uniform.
   All candidates are normalised to unit ℓ₁‑norm and the best
   according to the exact objective is retained.

2. **Soft‑max Adam optimisation** – starting from the best
   candidate we minimise the smooth approximation
       M_τ = τ·log Σ_i exp(conv_i/τ)
   of the maximal self‑convolution.  The gradient of M_τ is the
   weighted correlation 2·a⋆w, computed by FFT.  Adam updates are
   projected onto the probability simplex and the vector is kept
   symmetric (a = (a+a[::-1])/2).  τ is annealed exponentially.

3. **Light annealing polish** – a large number of cheap
   mass‑transfer moves are attempted; a move is kept if it lowers
   the true objective (or with a tiny Metropolis probability).

4. If a global starter `height_sequence_1` is available it is
   evaluated and used if better.

The whole search respects the supplied time budget (default 1000 s)
and finally returns a list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _correlate_linear(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation (a ⋆ w)[k] = Σ_j a_j·w_{j+k} for k=0..len(a)-1.
    Implemented via FFT with zero‑padding to length L ≥ len(a)+len(w)-1.
    """
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w[::-1], n=L)      # reverse w for correlation
    corr = np.fft.irfft(A * W_rev, n=L)[: a.shape[0]]
    return corr


# ----------------------------------------------------------------------
# Projection onto the probability simplex {v ≥ 0, Σ v = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (the verifier's score)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²).  Assumes a is 1‑D."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score
    within a wall‑time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # ------------------------------------------------------------------
    # 1) Parameter sweep over several analytic families
    # ------------------------------------------------------------------
    n = 16384                                 # a fairly large length
    x = (np.arange(n, dtype=np.float64) + 0.5) / n    # mid‑point grid
    base = x * (1.0 - x)

    best_seq = None
    best_score = float("inf")

    # helper to evaluate and possibly store a candidate
    def try_candidate(arr: np.ndarray, label: str):
        nonlocal best_seq, best_score
        arr = np.maximum(arr, 0.0)
        if arr.sum() == 0:
            return
        arr = arr / arr.sum()
        sc = _objective(arr)
        if sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"[{time.time() - start_time:.2f}s] {label} score={sc:.6f}")

    # arcsine‑type (singular at the ends)
    epsilons = [0.0, 1e-8, 1e-6, 1e-4]
    alphas = np.linspace(0.30, 0.70, 41)          # extend a little beyond the old range
    for eps in epsilons:
        base_eps = base + eps
        for alpha in alphas:
            w = base_eps ** (-alpha)
            try_candidate(w, f"arc α={alpha:.3f} eps={eps:g}")

    # interior‑peaked polynomial family (β ≥ 0)
    betas = np.linspace(0.0, 5.0, 26)
    for beta in betas:
        w = base ** beta
        try_candidate(w, f"poly β={beta:.2f}")

    # truncated Gaussian family
    gammas = np.linspace(0.0, 30.0, 31)
    for gamma in gammas:
        w = np.exp(-gamma * (x - 0.5) ** 2)
        try_candidate(w, f"gauss γ={gamma:.1f}")

    # sine‑power family
    ps = np.linspace(0.0, 10.0, 21)
    for p in ps:
        w = np.sin(np.pi * x) ** p
        try_candidate(w, f"sine p={p:.1f}")

    # mixture of arcsine and uniform
    mix_lambdas = np.linspace(0.0, 1.0, 11)
    for alpha in alphas:
        w_arc = (base + 1e-6) ** (-alpha)
        for lam in mix_lambdas:
            w = lam * w_arc + (1.0 - lam) * np.ones_like(x)
            try_candidate(w, f"mix α={alpha:.3f} λ={lam:.2f}")

    # safety fallback – uniform if nothing was found (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 2) Soft‑max Adam optimisation (smooth approximation of max‑conv)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12
    lr0 = 0.2                     # base learning rate

    # FFT length large enough for both self‑convolution (2n‑1) and
    # correlation with a weight vector of length 2n‑1.
    L_fft = _next_pow2(3 * n)

    it = 0
    best_local = a.copy()
    best_local_score = _objective(a)

    while time.time() < deadline:
        it += 1

        # FFT of the current a (re‑used for both convolutions)
        A_fft = np.fft.rfft(a, n=L_fft)

        # self‑convolution → conv_self (length 2n‑1)
        conv_full = np.fft.irfft(A_fft * A_fft, n=L_fft)[: 2 * n - 1]

        max_conv = conv_full.max()
        # soft‑max temperature (exponential decay)
        tau = max(max_conv * 0.5 * (0.98 ** (it // 250)), 1e-12)

        # soft‑max weights (approximate arg‑max of the convolution)
        exp_w = np.exp((conv_full - max_conv) / tau)
        w = exp_w / exp_w.sum()                     # shape (2n‑1,)

        # gradient = 2 * (a ⋆ w)  (linear correlation)
        grad = 2.0 * _correlate_linear(a, w, L_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step

        # enforce symmetry (helps keep the convolution balanced)
        a = (a + a[::-1]) * 0.5

        # project back onto the simplex {non‑negative, sum=1}
        a = _project_simplex(a)

        # occasional exact evaluation – keep the best exact iterate seen
        if (it % 200) == 0:
            cur_score = _objective(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")
                if cur_score < 1.503:
                    # good enough – we can stop early
                    break

        # stop a little early if we are close to the hard deadline
        if time.time() > deadline - 5.0:
            # final check
            cur_score = _objective(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                print(f"[{time.time() - start_time:.2f}s] near deadline score={cur_score:.6f}")
            break

    # ------------------------------------------------------------------
    # 3) Light simulated‑annealing polish (many cheap mass‑transfer moves)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a = best_local.copy()
        cur_score = best_local_score
        best_score = cur_score
        best_seq = a.copy()
        rng = np.random.default_rng()
        # number of moves is limited by remaining wall‑time
        move_counter = 0
        while time.time() < deadline:
            move_counter += 1
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # transfer up to 15 % of the source mass
            delta = a[src] * rng.random() * 0.15
            a2 = a.copy()
            a2[src] -= delta
            a2[dst] += delta
            new_score = _objective(a2)
            if new_score < cur_score:
                a = a2
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal move={move_counter:,} best={new_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Compare with the global starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                starter_score = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] global starter score={starter_score:.6f}")
                if starter_score < best_score:
                    best_seq = starter
                    best_score = starter_score
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Final score: {_objective(best_seq):.6f}")
    return [float(v) for v in best_seq]
