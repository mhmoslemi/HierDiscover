"""
Propose candidate for the step‑function minimisation problem.

We use a three‑phase optimisation pipeline (similar to the previous
submission) but with several aggressive refinements:

1️⃣  **Extended smooth‑max mirror descent** – the smoothing α is taken
    up to 2000 and we run many more internal iterations.  The update
    stays on the probability simplex via an exponentiated‑gradient step.

2️⃣  **Longer projected sub‑gradient descent** – we keep the momentum
    scheme, but run it until only a few seconds are left before the
    annealing phase.

3️⃣  **High‑volume incremental simulated‑annealing** – we update the
    self‑convolution in‑place (no full copy on each trial) and run many
    more moves (≈15 M).  The temperature decays exponentially with the
    elapsed fraction of the budget.

4️⃣  **Local L∞‑gradient polishing** – after the annealer we run a few
    hundred iterations of a sub‑gradient step that targets only the
    current maximal convolution entries, which often squeezes a few
    extra digits out of the objective.

All phases respect the total time budget (default 1000 s) and print
progress reports.  The final vector is normalised to sum = 1 and
returned as a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear_fft(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast objective for a probability vector (sum ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self_fft(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Extended smooth‑max mirror descent."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # α goes from modest smoothing to a very sharp approximation of max
    alphas = np.geomspace(0.5, 2000.0, num=48)

    it = 0
    max_iterations = 1_200_000
    for alpha in alphas:
        while time.time() < deadline - 0.3 and it < max_iterations:
            it += 1

            # 1) self‑convolution (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights (log‑sum‑exp trick)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)   # keep positivity
            a /= a.sum()

            # occasional check of the true objective
            if it % 800 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.3:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Longer momentum sub‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.35
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # indices achieving the current max (could be a few)
        max_idx = np.flatnonzero(conv == cur_max)
        # pick one of them (deterministic for reproducibility)
        k = max_idx[0]

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        # sub‑gradient of the max convolution entry
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # simplex projection (exact)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : high‑volume incremental simulated annealing
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-12,
                        max_moves: int = 15_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer with in‑place convolution updates."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # sum(a)=1 ⇒ denominator = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # exponential temperature schedule based on elapsed fraction
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.33    # up to 33 % of src mass
        if eps <= 0.0:
            continue

        # ----- apply incremental delta to convolution (in‑place) -----
        # store the changes so we can revert if needed
        # slice updates
        delta_src = -2.0 * eps * a          # contribution removed from src
        delta_dst =  2.0 * eps * a          # contribution added to dst

        # apply slice updates
        conv[src:src + n] += delta_src
        conv[dst:dst + n] += delta_dst

        # delta‑delta isolated indices
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            # Metropolis criterion
            if math.exp(-delta_score / T) > rng.random():
                accept = True

        if accept:
            # keep the move
            a[src] -= eps
            a[dst] += eps
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")
        else:
            # revert the updates
            conv[src:src + n] -= delta_src
            conv[dst:dst + n] -= delta_dst
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps
            # a unchanged

    return best_seq, best_score

# Phase‑4 : local L∞‑gradient polishing
# ----------------------------------------------------------------------
def _local_linf_polish(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       init_score: float,
                       max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """A few iterations of sub‑gradient descent that only targets the
    currently maximal convolution entries."""
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = np.convolve(a, a)                 # O(n²) but n=4096, cheap for few iters
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # indices attaining the max (could be several)
        maxi = np.flatnonzero(conv == cur_max)
        # Build sub‑gradient as average over all active maxima
        grad = np.zeros_like(a)
        for k in maxi:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        if len(maxi) > 0:
            grad /= len(maxi)

        # simple diminishing step size
        step = 0.2 / math.sqrt(it + 1)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    The routine uses several optimisation stages (smooth‑max descent,
    sub‑gradient descent, massive annealing, and a final L∞‑polish).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    n = 4096

    # --------------------------------------------------------------
    # 0️⃣  Load an initial vector (global best if present, else uniform)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')

    total = starter.sum()
    if total > 0.0:
        starter = starter / total
    else:
        starter = np.full(n, 1.0 / n)

    L_self = _next_pow2(2 * n - 1)
    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Extended smooth‑max mirror descent
    # --------------------------------------------------------------
    if time.time() < deadline - 140.0:
        smooth_deadline = deadline - 130.0
        best_seq, best_score = _smooth_descent(best_seq, n,
                                               smooth_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best "
              f"score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Longer projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 50.0:
        sub_deadline = deadline - 40.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad improved "
                  f"score = {sc_sub:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Massive incremental annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        anneal_deadline = deadline - 4.0
        a_anneal, sc_anneal = _incremental_anneal(best_seq, n,
                                                  anneal_deadline, start_time,
                                                  best_score, best_seq,
                                                  start_temp=1e-2,
                                                  end_temp=1e-12,
                                                  max_moves=15_000_000)
        if sc_anneal < best_score:
            best_score = sc_anneal
            best_seq = a_anneal.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved "
                  f"score = {sc_anneal:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final L∞‑gradient polish (few hundred cheap iterations)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.2:
        polish_deadline = deadline - 0.1
        a_polish, sc_polish = _local_linf_polish(best_seq, n,
                                                 polish_deadline, start_time,
                                                 best_score,
                                                 max_iters=2000)
        if sc_polish < best_score:
            best_score = sc_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:.2f}s] Polishing yielded "
                  f"score = {sc_polish:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{best_score:.6f}")

    # Return as a Python list of floats
    return [float(v) for v in best_seq]
