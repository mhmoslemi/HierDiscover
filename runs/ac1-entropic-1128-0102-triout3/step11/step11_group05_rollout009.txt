"""
A multi‑phase optimizer for the step‑function objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

The routine works on a vector a of length n (here n = 4096) with a ≥ 0,
∑a = 1, so the objective reduces to 2·n·max(conv(a,a)).  The algorithm
consists of four stages

1️⃣  **Linear‑programming improvement** –  given the current a,
    solve   maximise   Σb   subject to   a∗b ≤ M   (M = max(a∗a)),
    b ≥ 0.  Scaling the solution to sum 1 never increases the maximal
    convolution, so we obtain a monotone improvement.

2️⃣  **Smooth‑max mirror descent** – an exponentiated‑gradient (mirror)
    descent on the log‑sum‑exp smooth approximation of the max‑convolution.
    A geometric schedule of smoothing parameters drives the iterates quickly
    toward a low‑max region.

3️⃣  **Projected sub‑gradient refinement** – a short momentum sub‑gradient
    descent on the exact non‑smooth objective, keeping the iterate on the
    probability simplex.

4️⃣  **Incremental simulated annealing** – a fast “mass‑transfer’’ annealer
    that updates the self‑convolution in‑place and performs Metropolis
    moves to polish the solution.

All phases respect a global time budget (default 1000 s) and print
intermediate scores.  The final vector is returned as a plain Python list.
"""

import time
import math
import numpy as np

# optional imports – if SciPy is unavailable the LP phase is skipped
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Full self‑convolution a*a via FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation (sum(a) is assumed ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a non‑negative vector with sum≈1."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : linear‑programming improvement
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, n: int, L_self: int) -> tuple[np.ndarray, float]:
    """
    Solve   maximise Σb   subject to   a∗b ≤ M, b ≥ 0,
    where M = max(conv(a,a)).  Return the normalised solution and its score.
    If SciPy is unavailable or the LP fails, the original vector is returned.
    """
    if not _SCIPY_AVAILABLE:
        return a, _eval_fast(a, L_self)

    # current maximal convolution value
    conv = _conv_self(a, L_self)
    M = conv.max()

    # Build the sparse constraint matrix A (shape (2n‑1) × n) such that
    # (A @ b)[k] = (a∗b)[k].
    # A has constant diagonals – the i‑th diagonal (offset –i) consists of a[i].
    offsets = -np.arange(n)                     # -0, -1, …, -(n‑1)
    data = [np.full(n, a_i) for a_i in a]      # each diagonal repeats the same coeff
    A = _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')

    # Linear program: minimise -Σb  (i.e. maximise Σb)
    c = -np.ones(n)
    ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = _opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                       method='highs', options={'presolve': True})
    # Free the huge matrix as early as possible
    A = None

    if not res.success:
        # LP failed – return the unchanged vector
        return a, _eval_fast(a, L_self)

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return a, _eval_fast(a, L_self)

    a_new = b / s                # re‑normalise to ∑a = 1
    score = _eval_fast(a_new, L_self)
    return a_new, score


# ----------------------------------------------------------------------
# Phase‑2 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # a more aggressive schedule of the smoothing parameter α
    alphas = np.geomspace(0.5, 800.0, num=40)

    it = 0
    max_iterations = 400_000
    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # 1) self‑convolution (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stabilised exponentials
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)   # avoid exact zeros
            a /= a.sum()

            # occasional check of the real objective
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # index of a (first) maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]               # sub‑gradient of max‑convolution

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : fast incremental simulated‑annealing polish
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-10,
                        max_moves: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that updates the convolution incrementally."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1, cheap for n=4096
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # denominator = 1 because Σa = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # linear temperature schedule (exponential decay)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.35   # move up to 35 % of src mass
        if eps <= 0.0:
            continue

        # update the convolution incrementally
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / T) > rng.random():
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
