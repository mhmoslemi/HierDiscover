"""
Propose a step‑function (non‑negative heights) with a low value of the
evaluation function.

Algorithm
---------
1. **Parameterised families** – generate candidates from several analytic
   families:
      * Power‑ramp (central peak) and its inverse (edge‑peak).
      * Symmetric Beta(α,α) densities (U‑ or ∧‑shaped).
      * Random Dirichlet draws for extra diversity.

   All candidates are normalised to sum = 1 (the objective is scale‑invariant).

2. **Maintain a small pool** of the best `top_k` candidates (score,
   vector).  The pool is updated on the fly while respecting the time
   budget.

3. **Local refinement** – for every vector in the pool run a fast
   projected Adam optimiser (sub‑gradient of the max‑convolution) until
   the allocated sub‑budget expires.  The best iterate seen so far is
   retained.

4. **Optional polish** – a lightweight simulated‑annealing move‑packet
   phase that swaps tiny mass amounts between random positions.

5. Return the best vector (projected onto the probability simplex) as a
   plain list.  All steps respect the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – identical to evaluate_sequence with sum=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    cur_max = conv.max()
    return float(2.0 * n * cur_max)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σ w_i = 1, w_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “tent‑with‑plateau’’ shape.
    * p    – exponent of side ramps (>0).
    * α    – relative plateau width ∈ [0, 0.5].
    """
    alpha = np.clip(alpha, 0.0, 0.5)
    plateau_len = int(round(alpha * n))
    # keep symmetry – make plateau length even
    if plateau_len % 2 == 1:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_inverse_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Inverse of the power shape – high at the edges, low in the centre.
    The construction mirrors the forward shape and then flips it.
    """
    # start from a forward shape and invert the values (keeping non‑negativity)
    base = _make_power_shape(n, p, alpha)
    inv = base.max() - base  # flip vertically
    total = inv.sum()
    if total > 0:
        inv /= total
    else:
        inv = np.full(n, 1.0 / n)
    return inv


def _make_beta_shape(n: int, a: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density sampled on a regular grid.
    a > 0 (α parameter).  For a < 1 the density is U‑shaped,
    for a > 1 it is ∧‑shaped.
    """
    a = max(a, 0.05)      # avoid pathological extremes
    x = (np.arange(n) + 0.5) / n  # points in (0,1)
    # β‑function denominator is omitted – we renormalise anyway.
    shape = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    # Clip tiny negatives that may arise from rounding
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the max‑convolution objective.
    Returns the best iterate seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max       # sum(a) == 1 after every projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the maximum convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]   # 2·a_{k‑i}

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny mass‑packet swaps with Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # a modest budget of attempts – respects remaining time
    max_steps = 120_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # simple exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # more likely to move from heavy sites
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (≤ 25 %) of the source mass
        eps_move = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

    """
    Search for a step‑function (non‑negative heights) with a low evaluation
    score.  Returns a plain Python list suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid crossing the hard limit
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power‑of‑two for fast FFTs; 32768 offers a good trade‑off
    # --------------------------------------------------------------
    n = 32768

    best_seq = None
    best_score = float("inf")
    top_k = 6
    top_candidates: list[tuple[float, np.ndarray]] = []

    # --------------------------------------------------------------
    # 1) Analytic families
    # --------------------------------------------------------------
    p_grid = np.linspace(0.2, 5.0, 25)           # more aggressive exponent range
    alpha_grid = np.linspace(0.0, 0.5, 11)       # plateau fractions

    for p in p_grid:
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            # forward (central peak) shape
            a_fwd = _make_power_shape(n, p, alpha)
            score_fwd = _objective(a_fwd)
            if score_fwd < best_score:
                best_score, best_seq = score_fwd, a_fwd.copy()
            # keep pool
            if len(top_candidates) < top_k:
                top_candidates.append((score_fwd, a_fwd.copy()))
                top_candidates.sort(key=lambda x: x[0])
            elif score_fwd < top_candidates[-1][0]:
                top_candidates[-1] = (score_fwd, a_fwd.copy())
                top_candidates.sort(key=lambda x: x[0])

            # inverse (edge‑peak) shape
            a_inv = _make_inverse_power_shape(n, p, alpha)
            score_inv = _objective(a_inv)
            if score_inv < best_score:
                best_score, best_seq = score_inv, a_inv.copy()
            if len(top_candidates) < top_k:
                top_candidates.append((score_inv, a_inv.copy()))
                top_candidates.sort(key=lambda x: x[0])
            elif score_inv < top_candidates[-1][0]:
                top_candidates[-1] = (score_inv, a_inv.copy())
                top_candidates.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2) Beta families + random mixes
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(200):
        if time.time() >= deadline:
            break
        a_param = rng.uniform(0.1, 5.0)          # Beta α = β
        beta_shape = _make_beta_shape(n, a_param)

        # optional mixing with a tiny uniform component to stabilise numerics
        mix = rng.random()
        a_mix = mix * beta_shape + (1.0 - mix) * np.full(n, 1.0 / n)
        a_mix /= a_mix.sum()

        score_mix = _objective(a_mix)
        if score_mix < best_score:
            best_score, best_seq = score_mix, a_mix.copy()
        if len(top_candidates) < top_k:
            top_candidates.append((score_mix, a_mix.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif score_mix < top_candidates[-1][0]:
            top_candidates[-1] = (score_mix, a_mix.copy())
            top_candidates.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 3) Random Dirichlet seeds (extra diversity)
    # --------------------------------------------------------------
    for _ in range(30):
        if time.time() >= deadline:
            break
        a_rand = rng.gamma(shape=1.0, size=n)
        a_rand /= a_rand.sum()
        score_rand = _objective(a_rand)
        if score_rand < best_score:
            best_score, best_seq = score_rand, a_rand.copy()
        if len(top_candidates) < top_k:
            top_candidates.append((score_rand, a_rand.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif score_rand < top_candidates[-1][0]:
            top_candidates[-1] = (score_rand, a_rand.copy())
            top_candidates.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 4) Incorporate the global starter if it exists
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {cur:.6f}")
                if cur < best_score:
                    best_score, best_seq = cur, starter.copy()
                if len(top_candidates) < top_k:
                    top_candidates.append((cur, starter.copy()))
                    top_candidates.sort(key=lambda x: x[0])
                elif cur < top_candidates[-1][0]:
                    top_candidates[-1] = (cur, starter.copy())
                    top_candidates.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 5) Local refinement – Adam descent on the pool
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # allocate roughly 70 % of the remaining budget to refinements
        refine_budget = remaining * 0.70
        per_candidate = refine_budget / len(top_candidates)
        print(f"[{time.time() - start_time:.2f}s] Starting refinement "
              f"({len(top_candidates)} candidates, ~{per_candidate:.1f}s each)")

        for init_score, init_seq in top_candidates:
            sub_deadline = time.time() + per_candidate
            refined_seq, refined_score = _adam_descent(init_seq, sub_deadline, start_time)
            if refined_score < best_score:
                best_score, best_seq = refined_score, refined_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] New best after Adam = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 6) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 7) Final normalisation & conversion
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
