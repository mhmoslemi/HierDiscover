"""
Gradient‑plus‑incremental‑annealing optimizer for the step‑function
minimisation problem.
The search starts from the globally supplied best candidate (if any)
or from a symmetric triangular warm‑start.  A short projected
sub‑gradient phase (the same implementation as the reference) gives
a good starting point.  Then a cheap incremental simulated‑annealing
routine moves tiny packets of mass between entries while updating the
self‑convolution in O(n) time, allowing many more iterations than a
full FFT‑based evaluation.  The best sequence seen is returned.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgrad_topk(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  top_k: int = 7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent (same as reference)."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(conv, -top_k)[-top_k:]
            top_idx = part[np.argsort(-conv[part])]

        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(top_idx))

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score

def _incremental_anneal(a: np.ndarray,
                        conv: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing that moves a small packet of mass.
    Updates the convolution in O(n) time, allowing many more iterations.
    Returns the best sequence observed.
    """
    n = a.shape[0]
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    total_len = conv.shape[0]  # 2*n-1

    # Temperature schedule parameters
    T0 = 0.05
    T_end = 1e-6
    # We'll simply compute fraction of elapsed budget
    while time.time() < deadline:
        # progress fraction (0..1)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (T_end / T0) ** frac

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst:
            continue
        a_src = a[src]
        if a_src <= 0.0:
            continue

        # epsilon: up to 30% of source mass, scaled by remaining temperature
        eps_factor = 0.3 * (1.0 - frac) + 0.01  # shrink towards end
        eps = a_src * rng.random() * eps_factor
        if eps <= 0.0:
            continue

        # --- apply tentative update to conv ---------------------------------
        # Update conv in place using the old a (which has not been modified yet)
        # slice updates: conv[src:src+n] -= 2*eps*a
        #                 conv[dst:dst+n] += 2*eps*a
        # v⊛v adjustments
        # Store a reference for vectorized operations
        # Because we will need to possibly revert, keep a copy of conv changes?
        # We'll apply, compute new_max, then decide to keep or revert.
        # Apply src contribution
        lo_src = src
        hi_src = src + n
        conv[lo_src:hi_src] -= 2.0 * eps * a
        # Apply dst contribution
        lo_dst = dst
        hi_dst = dst + n
        conv[lo_dst:hi_dst] += 2.0 * eps * a
        # v⊛v terms
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        cross = src + dst
        if cross < total_len:
            conv[cross] -= 2.0 * eps * eps

        new_max = conv.max()
        delta = new_max - cur_max

        # Metropolis acceptance
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if T > 0.0 and math.exp(-delta / T) > rng.random():
                accept = True

        if accept:
            # keep move, update a and cur_max
            a[src] -= eps
            a[dst] += eps
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal best max={new_max:.8f}")
        else:
            # revert conv changes (use the same a, which is still old)
            conv[lo_src:hi_src] += 2.0 * eps * a
            conv[lo_dst:hi_dst] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            if cross < total_len:
                conv[cross] += 2.0 * eps * eps
            # a unchanged

    return best_a, best_max

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Problem size – power of two helps FFT speed.
    n = 16384

    # Warm‑start: use the global best if available,
    # otherwise a symmetric linear tent normalised to sum = 1.
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size == n:
            a0 = starter.copy()
        else:
            raise ValueError
    except Exception:
        # triangular (tent) shape
        x = np.arange(n, dtype=float) / (n - 1)
        tent = np.minimum(x, 1.0 - x)
        a0 = tent / tent.sum()

    # ------------------------------------------------------------------
    # Optionally improve with a short projected sub‑gradient phase.
    # ------------------------------------------------------------------
    sub_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.45)
    if sub_deadline > time.time() + 5.0:   # need at least a few seconds
        a_sub, sub_score = _subgrad_topk(a0, sub_deadline, start_time, top_k=9)
        # accept if better
        if sub_score < _objective(a0):
            a0 = a_sub
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad improved score={sub_score:.6f}")

    # --------------------------------------------------------------
    # Compute full convolution once – used by the annealer.
    # --------------------------------------------------------------
    conv = _conv_self(a0)

    # --------------------------------------------------------------
    # Incremental simulated‑annealing polish.
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_best, best_max = _incremental_anneal(a0, conv, deadline, start_time)
    else:
        a_best = a0
        best_max = conv.max()

    # --------------------------------------------------------------
    # Final normalisation (ensures sum = 1 and non‑negative).
    # --------------------------------------------------------------
    a_best = np.maximum(a_best, 0.0)
    total = a_best.sum()
    if total > 0:
        a_best /= total
    else:
        a_best = np.full(n, 1.0 / n)

    # Return as plain Python list.
    return [float(v) for v in a_best]

"""
Gradient‑plus‑incremental‑annealing optimizer for the step‑function
minimisation problem.

The search proceeds in three stages:

1. **Warm‑start** – use the globally supplied best candidate if it has
   the expected length; otherwise fall back to a symmetric “tent”
   distribution normalised to the probability simplex.

2. **Projected sub‑gradient** – a short run of the reference sub‑gradient
   method (averaging the top‑k convolution entries, momentum and a
   diminishing step size) yields a solid baseline.

3. **Incremental simulated annealing** – mass is moved between two
   positions while the self‑convolution is updated **in‑place** in
   O(n) time (no FFT is required).  This cheap operation allows many
   thousands of Metropolis steps within the remaining budget, polishing
   the solution further.

The best sequence seen is returned as a plain list of non‑negative
floats.  The routine respects the 1000 s time limit and works with a
single CPU only.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent – same idea as the reference code
# ----------------------------------------------------------------------
def _subgrad_topk(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  top_k: int = 9) -> tuple[np.ndarray, float]:
    """
    Runs a projected sub‑gradient descent where the sub‑gradient is the
    average of the `top_k` largest convolution entries.
    Returns the best iterate and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices of the top‑k convolution entries
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(conv, -top_k)[-top_k:]
            top_idx = part[np.argsort(-conv[part])]

        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(top_idx))

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Incremental simulated‑annealing polish (O(n) per move)
# ----------------------------------------------------------------------
def _incremental_anneal(a: np.ndarray,
                        conv: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Moves tiny packets of mass between coordinates while updating the
    convolution in O(n) time.  The Metropolis rule with a decreasing
    temperature accepts occasional uphill moves.
    Returns the best sequence seen and its maximal convolution value.
    """
    n = a.shape[0]
    total_len = conv.shape[0]          # = 2·n‑1
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng()
    T0 = 0.05
    T_end = 1e-6

    while time.time() < deadline:
        # linearised fraction of the remaining budget (0…1)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (T_end / T0) ** frac

        src = rng.choice(n, p=a)          # bias towards heavy piles
        dst = rng.integers(n)
        if src == dst:
            continue
        a_src = a[src]
        if a_src <= 0.0:
            continue

        # ε up to 30 % of the source mass, shrinks as the schedule progresses
        eps_factor = 0.3 * (1.0 - frac) + 0.01
        eps = a_src * rng.random() * eps_factor
        if eps <= 0.0:
            continue

        # ----- apply the *tentative* update to the convolution -------------
        lo_src = src
        hi_src = src + n
        conv[lo_src:hi_src] -= 2.0 * eps * a          # src contribution

        lo_dst = dst
        hi_dst = dst + n
        conv[lo_dst:hi_dst] += 2.0 * eps * a          # dst contribution

        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        cross = src + dst
        conv[cross] -= 2.0 * eps * eps                # cross term

        new_max = conv.max()
        delta = new_max - cur_max

        # ----- Metropolis acceptance ---------------------------------------
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if T > 0.0 and math.exp(-delta / T) > rng.random():
                accept = True

        if accept:
            # keep the move
            a[src] -= eps
            a[dst] += eps
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal best max={new_max:.8f}")
        else:
            # revert the convolution to its previous state
            conv[lo_src:hi_src] += 2.0 * eps * a
            conv[lo_dst:hi_dst] -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[cross] += 2.0 * eps * eps
            # a stays unchanged

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overshooting the limit
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two yields fast FFTs.
    # --------------------------------------------------------------
    n = 16384

    # --------------------------------------------------------------
    # 1) Warm start – use globally supplied best if possible.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size == n:
            a0 = starter.copy()
        else:
            raise ValueError
    except Exception:
        # symmetric linear tent, normalised to the simplex
        x = np.arange(n, dtype=float) / (n - 1)
        tent = np.minimum(x, 1.0 - x)
        a0 = tent / tent.sum()

    # --------------------------------------------------------------
    # 2) Short projected sub‑gradient phase (≈45 % of remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    sub_deadline = time.time() + max(0.0, remaining * 0.45)
    if sub_deadline - time.time() > 5.0:       # need at least a few seconds
        a_sub, sub_score = _subgrad_topk(a0, sub_deadline, start_time, top_k=9)
        if sub_score < _objective(a0):
            a0 = a_sub
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad improved score={sub_score:.6f}")

    # --------------------------------------------------------------
    # 3) Compute the full convolution once – needed by the annealer.
    # --------------------------------------------------------------
    conv = _conv_self(a0)

    # --------------------------------------------------------------
    # 4) Incremental simulated‑annealing polish (uses the remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_best, best_max = _incremental_anneal(a0, conv, deadline, start_time)
    else:
        a_best = a0
        best_max = conv.max()

    # --------------------------------------------------------------
    # 5) Final normalisation & conversion to a plain list.
    # --------------------------------------------------------------
    a_best = np.maximum(a_best, 0.0)
    total = a_best.sum()
    if total > 0.0:
        a_best /= total
    else:
        a_best = np.full(n, 1.0 / n)

    return [float(v) for v in a_best]
