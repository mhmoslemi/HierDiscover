"""
Fast simulated‑annealing / hill‑climbing optimizer for the
step‑function objective

    2·n·max(conv(a,a)) / (sum(a)²)

The programme works in three phases

1.  A cheap grid‑search over a few parametrised families
    (edge‑heavy power‑law, edge‑heavy exponential, the global
    starter if available and the uniform vector).  The best
    member of this pool becomes the warm‑start.

2.  A long simulated‑annealing run that repeatedly moves a
    random (small) packet of mass from a source index to a
    destination index.  The convolution is recomputed with a
    real FFT, the Metropolis rule is applied and the best
    sequence seen is remembered.  The temperature is linearly
    decreased to zero.

3.  Final re‑normalisation and conversion to a plain Python list.

All heavy work (FFT‑based convolution, objective evaluation) is
performed with NumPy in compiled code, so millions of proposals
fit easily into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a (real FFT implementation).
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)²).
    The scaling cancels, but we keep the formula for clarity.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Simple parametrised families (used for the warm‑start)
# ----------------------------------------------------------------------
def _edge_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n-i)^{-p}  (symmetric, edge‑heavy)."""
    idx = np.arange(n, dtype=float) + 1.0               # 1 … n
    a = idx ** (-p) + (n + 1 - idx) ** (-p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a


def _edge_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) + exp(-lam·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a

        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = init_T * (1.0 - frac)

        # pick a source index proportionally to its current weight
        prob = a / a.sum()
        src = rng.choice(n, p=prob)
        dst = rng.integers(n)
        if src == dst or a[src] <= 0.0:
            continue

        # amount of mass to move (random fraction of the source)
        eps = a[src] * move_factor * rng.random()
        if eps <= 1e-16:
            continue

        # --- perform the move (in‑place) ---------------------------------
        a_src_old = a[src]
        a_dst_old = a[dst]
        a[src] = a_src_old - eps
        a[dst] = a_dst_old + eps

        new_score = _objective(a)

        # --- acceptance test ---------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        elif T > 0.0:
            # Metropolis rule
            if math.exp(-(new_score - cur_score) / T) > rng.random():
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_seq_score:
                best_seq_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] new best score={new_score:.6f}")
        else:
            # revert the move
            a[src] = a_src_old
            a[dst] = a_dst_old

        # occasional re‑normalisation to fight numerical drift
        if int(elapsed) % 30 == 0:
            total_mass = a.sum()
            if total_mass > 0.0:
                a /= total_mass

    # ------------------------------------------------------------------
    # 3) Final normalisation & conversion.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
