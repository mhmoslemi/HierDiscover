def _subgradient_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.8
    it = 0
    last_print = start_time
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # print
            t = time.time()
            if t - last_print > 30:
                print(f"[{t - start_time:.2f}s] Subgrad it {it:,} score {cur_score:.6f}")
                last_print = t
        # compute gradient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        # apply momentum update
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

def _make_gaussian(n, sigma):
    x = np.arange(n) - (n-1)/2
    shape = np.exp(-0.5*(x/sigma)**2)
    shape = np.maximum(shape, 0)
    shape/= shape.sum()

"""
Propose a step‑function with a low value of `evaluate_sequence`.

Algorithm
---------
1. Choose a fairly large canvas (`n = 65536`, a power of two) – the
   discretisation error of the continuous optimum 1.5 decays roughly as
   O(1/n), so doubling the previous size gives a theoretical bound
   around 1.504.

2. Build a small library of analytic shapes:
      – symmetric “tent‑with‑plateau’’ families parameterised by exponent `p`
        (curvature) and plateau width `α`;
      – a Gaussian‑like bell (controlled by a width σ).
   All shapes are normalised to sum 1 (the objective is scale‑invariant).

3. Perform a cheap grid / random search over the shape parameters,
   keep the best few candidates (top‑k) and the overall best.

4. Refine the current best shape by **projected sub‑gradient descent** on the
   exact (non‑smooth) objective.  The sub‑gradient is taken only from the
   convolution entry attaining the current maximum; after each update the
   iterate is projected onto the probability simplex (Euclidean projection).
   Momentum (β = 0.9) and a 1/√t step‑size are used.  The descent runs until
   the allocated time budget is exhausted.

5. A light **simulated‑annealing** polish moves tiny mass packets between
   random positions, accepting worsening moves with the Metropolis rule.
   This helps to escape very shallow local minima.

6. The final vector is clipped to non‑negative values, normalised to sum 1
   and returned as a plain Python list.

The code respects the 1000 s time limit, prints occasional progress
messages, and contains all helper functions at top level as required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric tent–with–plateau shape.
    p     – exponent of the side ramps (>0).
    alpha – relative plateau width ∈ [0, 0.5] (0 → pure tent, 0.5 → uniform).
    Returns a vector of length n normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    # enforce even plateau length to keep perfect symmetry
    if plateau_len % 2 == 1:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                      # rises from 0 to 1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # correct rounding artefacts
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """
    Discrete Gaussian bell centred at the middle of the array.
    sigma – standard deviation in index units (must be >0).
    """
    if sigma <= 0:
        sigma = 1.0
    x = np.arange(n) - (n - 1) / 2.0
    shape = np.exp(-0.5 * (x / sigma) ** 2)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (non‑smooth objective)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent until `deadline`.
    Returns the best iterate seen (and its score).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.8
    it = 0
    last_print = start_time

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            now = time.time()
            if now - last_print > 30.0:
                print(f"[{now - start_time:.2f}s] Sub‑grad it={it:,} "
                      f"score={cur_score:.6f}")
                last_print = now

        # sub‑gradient at the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + adaptive step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project back onto the simplex (Σ a = 1, a ≥ 0)
        a = _project_simplex(a)

    rng = np.random.default_rng()

        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # pick a source proportional to its mass (more mass -> more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps

        delta = new_score - cur_score

                best_score = new_score
