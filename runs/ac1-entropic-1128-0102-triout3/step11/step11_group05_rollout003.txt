"""
Optimisation of a non‑negative step‑function (length 4096) for the
objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a combination of exponentiated‑gradient (smooth‑max mirror
descent) and a **targeted simulated‑annealing** that moves mass from the
coordinates that contribute most to the current peak of the
self‑convolution towards the periphery.  Several short random restarts
are performed inside the 1000 s budget; the best sequence found is
returned.

The implementation follows the same helper structure as the reference
code (FFT based convolution, fast evaluation on the probability simplex)
but replaces the generic annealer with a move selection that directly
targets the dominant convolution terms.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation on the simplex (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the probability simplex with a
    smoothed max (log‑sum‑exp) surrogate."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # a geometric schedule of smoothing parameters (more aggressive)
    alphas = np.geomspace(0.5, 400.0, num=48)

    it = 0
    max_iterations = 400_000
    for alpha in alphas:
        # break early if the global deadline is near
        while time.time() < deadline - 0.1 and it < max_iterations:
            it += 1

            # 1) self‑convolution via FFT (only the part we need)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weighting
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stabilise
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)   # keep positivity
            a /= a.sum()

            # occasional real objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")
        if time.time() >= deadline - 0.1:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : targeted incremental simulated‑annealing
# ----------------------------------------------------------------------
def _targeted_anneal(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     cur_best_seq: np.ndarray,
                     cur_best_score: float,
                     start_temp: float = 0.08,
                     end_temp: float = 1e-9,
                     max_moves: int = 4_000_000) -> tuple[np.ndarray, float]:
    """

    peak of the self‑convolution towards indices that are outside the
    overlapping window.  Acceptance follows a standard Metropolis rule
    with a geometric temperature schedule.
    """
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # denominator = 1 (sum a = 1)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_moves:
        it += 1

        # --------------------------------------------------------------
        # temperature (geometric decay)
        # --------------------------------------------------------------
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # --------------------------------------------------------------
        # locate the current peak and compute contributing indices
        # --------------------------------------------------------------
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)

        i_range = np.arange(lo, hi + 1)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob = contrib / total_contrib
            src = rng.choice(i_range, p=prob)
        else:
            src = rng.integers(lo, hi + 1)

        # --------------------------------------------------------------
        # pick a destination outside the overlap window (if possible)
        # --------------------------------------------------------------
        left_available = lo > 0
        right_available = hi < n - 1
        if left_available and right_available:
            if rng.random() < 0.5:
                dst = rng.integers(0, lo)            # left side
            else:
                dst = rng.integers(hi + 1, n)        # right side
        elif left_available:
            dst = rng.integers(0, lo)
        elif right_available:
            dst = rng.integers(hi + 1, n)
        else:
            # fall‑back: any different index
            while True:
                dst = rng.integers(0, n)
                if dst != src:
                    break

        # --------------------------------------------------------------
        # magnitude of the transferred mass (large early, small later)
        # --------------------------------------------------------------
        eps_factor = (1.0 - progress) * 0.5 + progress * 0.05
        eps = a[src] * rng.random() * eps_factor
        if eps <= 0.0:
            continue

        # --------------------------------------------------------------
        # incremental convolution update (exact)
        # --------------------------------------------------------------
        conv_new = conv.copy()
        # remove contribution of the transferred mass from `src`
        conv_new[src:src + n] -= 2.0 * eps * a
        # add contribution of the transferred mass to `dst`
        conv_new[dst:dst + n] += 2.0 * eps * a
        # pure quadratic term adjustments
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if T > 0.0 and math.exp(-delta / T) > rng.random():
                accept = True

        if accept:
            # apply the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            # keep the best ever seen
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                if it % 1000 == 0:
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"targeted anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (length 4096) that minimises the
    supplied `evaluate_sequence`.  The routine performs a few short
    random restarts, each consisting of a smooth‑max mirror descent
    followed by the targeted simulated annealing described above.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.6   # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣ initialise a starter vector (global best if present, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')

    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Several short restarts (max 3)
    # ------------------------------------------------------------------
    max_restarts = 3
    for r in range(max_restarts):
        if time.time() >= deadline:
            break

        # allocate half of the remaining time to smooth descent,
        # the rest to annealing
        remaining = deadline - time.time()
        smooth_deadline = time.time() + 0.45 * remaining
        anneal_deadline = deadline

        # initialise a0 -------------------------------------------------
        if r == 0:
            a0 = best_seq.copy()            # start from the global best
        else:
            # random Dirichlet start (sum = 1)
            a0 = np.random.dirichlet(np.ones(n))

        # --------------------------------------------------------------
        # Phase‑1 : smooth‑max mirror descent
        # --------------------------------------------------------------
        a_smooth, score_smooth = _smooth_descent(a0, n, smooth_deadline,
                                                start_time)
        # --------------------------------------------------------------
        # Phase‑2 : targeted annealing
        # --------------------------------------------------------------
        a_final, score_final = _targeted_anneal(
            a_smooth, n, anneal_deadline, start_time,
            cur_best_seq=a_smooth,
            cur_best_score=score_smooth,
            start_temp=0.08,
            end_temp=1e-9,
            max_moves=5_000_000
        )

        # keep the best across restarts
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {r} "
                  f"new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{best_score:.6f}")
    return [float(v) for v in best_seq]
