def _smooth_gradient_descent(a0, n, deadline, start_time, best_score, best_seq):

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2
    it = 0
    L_self = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n - 2)
    while time.time() < deadline:
        it += 1
        # compute conv self
        conv = _conv_self(a, L=L_self)
        # compute weights:
        max_conv = conv.max()
        beta_smooth = min(10.0, 0.5 + it * 5e-5) # Example schedule
        exps = np.exp(beta_smooth * (conv - max_conv))
        sum_exp = exps.sum()
        w = exps / sum_exp
        # gradient via convolution
        a_rev = a[::-1]
        # conv_w_arev via FFT
        W = np.fft.rfft(w, n=L_grad)
        A_rev = np.fft.rfft(a_rev, n=L_grad)
        conv_w_arev = np.fft.irfft(W * A_rev, n=L_grad)
        grad = 2.0 * conv_w_arev[:n][::-1]  # shape n
        
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / (1 + it * 1e-4)  # decay slowly
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        a = np.maximum(a, 0.0)
        s = a.sum()

"""
Multi‑restart smoothed gradient descent (Adam) + final microscopic hill‑climb.

* Uses a smooth approximation of the “max‑convolution’’ (log‑sum‑exp) so the
  gradient incorporates information from many shifts, not only the single
  active one used in a plain sub‑gradient step.
* The gradient of the smoothed objective is obtained analytically and
  computed with a single FFT‑based convolution per iteration.
* Several random restarts (including the supplied ``height_sequence_1`` and a
  uniform start) are performed; the best sequence over the whole budget is
  kept.
* After the main optimisation a tiny random‐mass‑exchange hill‑climb is run
  to polish the solution.
* All updates keep the vector on the probability simplex (non‑negative,
  sum‑to‑one) by a simple renormalisation, which is fast enough for the
  thousands of iterations possible within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_general(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    m = x.shape[0] + y.shape[0] - 1
    if L is None:
        L = _next_pow2(m)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    res = np.fft.irfft(X * Y, n=L)[:m]
    return res


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum is assumed ≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smoothed_max_and_grad(a: np.ndarray,
                           beta: float,
                           L_self: int,
                           L_grad: int) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad) where smooth_max ≈ max(conv(a,a))
    and grad is the gradient of that smooth max w.r.t. a.
    """
    # self‑convolution
    conv = _conv_self(a, L_self)
    m = conv.max()
    # log‑sum‑exp (stable)
    exp_shift = np.exp(beta * (conv - m))
    sum_exp = exp_shift.sum()
    smooth_max = m + np.log(sum_exp) / beta

    # weights for the gradient (soft arg‑max)
    w = exp_shift / sum_exp                     # shape (2n‑1,)

    # gradient = 2 * cross‑correlation(w, a)
    a_rev = a[::-1]
    corr = _conv_general(w, a_rev, L_grad)     # length 3n‑2
    grad = 2.0 * corr[:a.shape[0]][::-1]       # take first n and reverse
    return smooth_max, grad


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _adam_descent(start_a: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_seq: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the smoothed objective until ``deadline``.
    Returns the best sequence seen during the run.
    """
    a = start_a.copy()
    rng = np.random.default_rng()
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0

    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        # ----- smoothed max and its gradient -----
        beta = min(8.0, 0.5 + it * 5e-5)          # schedule: slowly increase
        smooth_max, grad = _smoothed_max_and_grad(a, beta, L_self, L_grad)

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / (1 + it * 1e-4)               # slow decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- projection onto the simplex (fast renorm) -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # ----- evaluate exact objective -----
        cur_score = _eval_fast(a, L_self)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cur_score:.6f}")

        # ----- escape plateaus -----
        if it - last_improve > 5000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = it

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(start_a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int) -> tuple[np.ndarray, float]:
    """
    Extremely small mass transfers (ε≈10⁻⁴·a_src) that are kept only if the
    true objective improves.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the supplied
    evaluation function.  The search consists of

        * a quick parametric sweep (geometric / power / beta families),
        * several Adam‑based gradient‑descent runs with a smoothed max,
        * a final microscopic hill‑climb.

    The routine respects the 1000 s budget and returns the best sequence
    found as a plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)   # for all exact evaluations

    # ------------------------------------------------------------------
    # baseline: uniform
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # try to use the globally provided starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (geometric / power / beta)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric family
    eps_grid = np.logspace(-4, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # power family
    p_grid = np.linspace(-1.0, 2.0, 51)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p)
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type family
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Multi‑restart Adam‑based gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        restarts = 4
        # allocate time roughly equally among the restarts
        remaining = deadline - time.time()
        for r in range(restarts):
            if time.time() >= deadline - 5.0:
                break
            sub_deadline = time.time() + remaining / (restarts - r) - 2.0
            # choose a starting point: uniform, random Dirichlet, or best so far
            if r == 0:
                start_a = best_seq.copy()                 # refine the current best
            elif r == 1:
                start_a = uniform.copy()
            else:
                # random Dirichlet (alpha=0.5 gives a bit more sparsity)
                start_a = np.random.dirichlet(np.full(n, 0.5))
            start_a = np.maximum(start_a, 0.0)
            start_a /= start_a.sum()
            best_seq, best_score = _adam_descent(start_a, n,
                                                 sub_deadline,
                                                 start_time,
                                                 best_seq,
                                                 best_score)
            remaining = deadline - time.time()
            print(f"[{time.time() - start_time:.2f}s] Restart {r+1}/{restarts} best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final microscopic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _tiny_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb final score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise (just in case) and return as a Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
