"""
Optimise the L∞ norm of the self‑convolution of a non‑negative
vector (the “step‑function’’) by a smooth soft‑max gradient descent.
The method starts from a small library of classical window functions
(Hann, Blackman, Kaiser, sinusoidal powers,…), picks the best
candidate, then refines it using a differentiable approximation of the
max‐operator (soft‑max with a gradually increasing temperature).  Each
iteration performs a momentum update and a Euclidean projection onto
the probability simplex.  A light simulated‑annealing polishing phase
finishes the run.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _softmax_grad_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          step0: float = 0.2,
                          beta: float = 0.9,
                          alpha_min: float = 2.0,
                          alpha_max: float = 30.0,
                          alpha_ramp_steps: int = 2000) -> tuple[np.ndarray, float]:
    """
    Smooth gradient descent using a soft‑max approximation of the max‑convolution.
    Returns best vector and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)

    it = 0
    L = _next_pow2(3 * n)   # size for the gradient convolution (w * a_rev)

    while time.time() < deadline:
        it += 1
        # ---------- convolution ----------
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv  # sum(a)=1 because we keep it on the simplex
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SD iter={it:,} score={cur_score:.6f}")

        # ---------- soft‑max weights ----------
        # linearly increase alpha from alpha_min to alpha_max over alpha_ramp_steps
        if it <= alpha_ramp_steps:
            alpha = alpha_min + (alpha_max - alpha_min) * (it / alpha_ramp_steps)
        else:
            alpha = alpha_max

        # avoid overflow: shift by max_conv
        w_unnorm = np.exp(alpha * (conv - max_conv))
        w = w_unnorm / w_unnorm.sum()   # shape (2n‑1,)

        # ---------- gradient ----------
        # grad_i = 2 * sum_k w_k * a_{k-i}
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_grad = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 1]   # length 3n‑1 (more than needed)
        # we need the slice of length n starting at index n‑1
        grad = 2.0 * conv_grad[n - 1: n - 1 + n]

        # ---------- momentum + step ----------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # ---------- projection ----------
        a = _project_simplex(a)

    return best_a, best_score

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Simple annealing that moves a small random packet of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 200_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # linear cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)   # same as previous code

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

def _generate_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a list of candidate probability vectors based on classical
    window functions and simple parametric families.
    """
    x = np.arange(n, dtype=float) / (n - 1)         # in [0,1]
    candidates = []

    # 1) Uniform (for reference)
    candidates.append(np.full(n, 1.0 / n))

    # 2) Bartlett (triangular) – already the warm start of the original code
    tent = np.minimum(x, 1.0 - x)
    if tent.sum() > 0:
        candidates.append(tent / tent.sum())

    # 3) Hann window
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    candidates.append(hann / hann.sum())

    # 4) Blackman window
    blackman = (0.42
                - 0.5 * np.cos(2.0 * np.pi * x)
                + 0.08 * np.cos(4.0 * np.pi * x))
    candidates.append(blackman / blackman.sum())

    # 5) Kaiser windows with different betas
    for beta in (2.0, 4.0, 8.0, 12.0):
        kaiser = np.kaiser(n, beta)
        candidates.append(kaiser / kaiser.sum())

    # 6) Sinⁿ family  a_i = sin(π·x)ⁿ
    sinx = np.sin(np.pi * x)
    for p in np.linspace(0.5, 4.0, 8):
        shape = sinx ** p
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    # 7) Power‑Beta family a_i = (x·(1‑x))^p
    xb = x * (1.0 - x)
    for p in np.linspace(0.5, 3.0, 8):
        shape = xb ** p
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    return candidates

        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_grad = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]   # length 3n‑2
        grad = 2.0 * conv_grad[n - 1: n - 1 + n]

        # momentum + step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # projection onto the simplex (keeps sum = 1 and non‑negative)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap annealing that moves a small random packet of mass.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 200_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)   # exponential cooling

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Generation of a small library of classical windows and simple families
# ----------------------------------------------------------------------
def _generate_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Return a list of probability vectors built from standard window functions
    (Bartlett, Hann, Blackman, Kaiser) and a few parametric families.
    """
    x = np.arange(n, dtype=float) / (n - 1)   # values in [0,1]
    candidates: list[np.ndarray] = []

    # Uniform (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # Bartlett (triangular / tent)
    tent = np.minimum(x, 1.0 - x)
    if tent.sum() > 0:
        candidates.append(tent / tent.sum())

    # Hann window
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    candidates.append(hann / hann.sum())

    # Blackman window
    blackman = (0.42
                - 0.5 * np.cos(2.0 * np.pi * x)
                + 0.08 * np.cos(4.0 * np.pi * x))
    candidates.append(blackman / blackman.sum())

    # Kaiser windows with several betas
    for beta in (2.0, 4.0, 8.0, 12.0):
        kaiser = np.kaiser(n, beta)
        candidates.append(kaiser / kaiser.sum())

    # Sin^p family: a_i ∝ sin(π·x)^p
    sinx = np.sin(np.pi * x)
    for p in np.linspace(0.5, 4.0, 8):
        shape = sinx ** p
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    # Power‑beta family: a_i ∝ (x·(1‑x))^p
    xb = x * (1.0 - x)
    for p in np.linspace(0.5, 3.0, 8):
        shape = xb ** p
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    return candidates


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function under a 1000 s budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------
    # Problem size – power of two for fast FFTs
    # --------------------------------------
    n = 16384

    # --------------------------------------
    # 0) Build a modest library of initial candidates
    # --------------------------------------
    candidates = _generate_initial_candidates(n)

    # If a global starter exists, add it (resampled if needed)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # linear resampling to the target length
                idx = np.linspace(0, starter.size - 1, n)
                starter = np.interp(idx, np.arange(starter.size), starter)
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0:
                starter = starter / total
                candidates.append(starter)
    except Exception:
        pass

    # --------------------------------------
    # 1) Quick scan – keep the best candidate
    # --------------------------------------
    best_a = None
    best_score = np.inf
    for i, cand in enumerate(candidates):
        score = _objective(cand)
        print(f"[{time.time() - start_time:.2f}s] Candidate {i} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = cand.copy()
    print(f"[{time.time() - start_time:.2f}s] Best initial score={best_score:.6f}")

    # --------------------------------------
    # 2) Smooth soft‑max gradient descent (the heavy part)
    # --------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        grad_deadline = time.time() + remaining * 0.85
        refined_a, refined_score = _softmax_grad_descent(
            best_a,
            grad_deadline,
            start_time,
            step0=0.3,
            beta=0.9,
            alpha_min=2.0,
            alpha_max=30.0,
            alpha_ramp_steps=3000
        )
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a.copy()
            print(f"[{time.time() - start_time:.2f}s] After smooth‑descent score={best_score:.6f}")

    # --------------------------------------
    # 3) Light annealing polish (if time permits)
    # --------------------------------------
    if time.time() < deadline:
        annealed_a, annealed_score = _anneal(best_a, deadline, start_time)
        if annealed_score < best_score:
            best_score = annealed_score
            best_a = annealed_a.copy()
            print(f"[{time.time() - start_time:.2f}s] After anneal best score={best_score:.6f}")

    # --------------------------------------
    # 4) Final normalisation & conversion to a plain list
    # --------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    final_score = _objective(best_a)
    print(f"[{time.time() - start_time:.2f}s] Final reported score={final_score:.6f}")

    return [float(v) for v in best_a]
