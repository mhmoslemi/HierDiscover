def _subgradient_descent(start_a, deadline, start_time, L_fft):

    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    eta0 = 0.5 maybe; schedule: η = η0 / sqrt(it+1).

    while time.time() < deadline and it < max_iter:

        it += 1
        conv = _conv_self(a, L_fft)
        k = int(conv.argmax())
        # compute gradient of max term: g_i = 2*a_{k - i} for i ranging.
        grad = np.zeros_like(a)
        lo = max(0, k - (a.shape[0] - 1))
        hi = min(k, a.shape[0] - 1)
        idx = np.arange(lo, hi+1, dtype=int)
        grad[idx] = 2 * a[k - idx]  # vectorized
        # gradient descent step handling positivity + sum=1 via exponentiated gradient
        # use exponentiated update: a_i = a_i * exp(-η * grad_i )
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # projection onto simplex
        a = _project_simplex(a)
        # evaluate
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} score={cur_score:.6f}")

    return best_a, best_score

    from scipy.optimize import differential_evolution, linprog
    _has_scipy = True
except Exception:                                      # pragma: no cover
    _has_scipy = False


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length L."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # small negative imag part may appear due to round‑off – force real
    return np.real(conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (w≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum≈1) – used repeatedly during the search
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return the exact value of evaluate_sequence for a (assumes sum≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:                     # defensive – should never happen
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Parametric families – quick sweep
# ----------------------------------------------------------------------
def _sweep_parametric(n: int, L_fft: int, deadline: float, start_time: float,
                      best_seq: np.ndarray, best_score: float):
    """Explore a variety of low‑dimensional families and keep the best."""
    idx = np.arange(n, dtype=np.float64)

    # ----- pure geometric (decay) ------------------------------------
    for eps in np.logspace(-4, -1, 80):
        if time.time() >= deadline - 30:
            return best_seq, best_score
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        sc = _eval_fast(geo, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.2e} score={sc:.6f}")

        # reverse geometric (mass at the right side)
        rev = geo[::-1]
        sc = _eval_fast(rev, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, rev.copy()
            print(f"[{time.time() - start_time:.2f}s] RevGeo eps={eps:.2e} score={sc:.6f}")

    # ----- power law -------------------------------------------------
    for p in np.linspace(-1.0, 2.0, 31):
        if time.time() >= deadline - 30:
            break
        pow_seq = (idx + 1) ** p
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # ----- beta‑type (U‑shaped possible) -----------------------------
    for p in np.linspace(0.1, 0.9, 5):
        for q in np.linspace(0.1, 0.9, 5):
            if time.time() >= deadline - 30:
                break
            beta = (idx + 1) ** p * (n - idx) ** q
            beta /= beta.sum()
            sc = _eval_fast(beta, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, beta.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")

    # ----- cosine‑taper (high on edges, low in centre) -------------
    #   w_i = 0.5 * (1 + cos(pi * i/(n-1)))   → 1 at the ends, 0 in the middle
    cos_taper = 0.5 * (1.0 + np.cos(np.pi * idx / (n - 1)))
    cos_taper /= cos_taper.sum()
    sc = _eval_fast(cos_taper, L_fft)
    if sc < best_score:
        best_score, best_seq = sc, cos_taper.copy()
        print(f"[{time.time() - start_time:.2f}s] Cosine‑taper score={sc:.6f}")

    # ----- two‑sided exponential (U‑shaped) -------------------------
    for decay in np.logspace(-3, -0.5, 12):
        if time.time() >= deadline - 30:
            break
        left = np.exp(-decay * idx)
        right = np.exp(-decay * (n - 1 - idx))
        uexp = left + right
        uexp /= uexp.sum()
        sc = _eval_fast(uexp, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, uexp.copy()
            print(f"[{time.time() - start_time:.2f}s] U‑exp decay={decay:.2e} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Differential evolution on a Gaussian‑mixture parameterisation
# ----------------------------------------------------------------------
def _gaussian_mixture(params: np.ndarray, n: int) -> np.ndarray:
    """Build a non‑negative, normalised sequence from K isotropic Gaussians."""
    seq = np.zeros(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    K = len(params) // 3
    for i in range(K):
        amp = max(0.0, params[3 * i])          # non‑negative amplitude
        cen = params[3 * i + 1]
        wid = max(1e-3, params[3 * i + 2])
        seq += amp * np.exp(-((x - cen) ** 2) / (2.0 * wid * wid))
    seq = np.clip(seq, 0.0, None)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)
    return seq


def _de_phase(n: int, L_fft: int, deadline: float,
              start_time: float,
              best_seq: np.ndarray,
              best_score: float) -> tuple[np.ndarray, float]:
    """Run a fairly long DE optimisation (if SciPy is available)."""
    if not _has_scipy:                                         # pragma: no cover
        return best_seq, best_score

    K = 8                # number of Gaussian components – a bit richer
    dim = 3 * K
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 10.0))       # amplitude
        bounds.append((0.0, n - 1))      # centre
        bounds.append((0.5, n / 2.0))    # width – keep them reasonably wide

    def objective(p):
        seq = _gaussian_mixture(p, n)
        return _eval_fast(seq, L_fft)

    remaining = deadline - time.time()
    # allocate roughly half the remaining budget to DE (but cap to 200 s)
    max_generations = int(min(200, max(1, remaining // 5)))
    popsize = 15

    print(f"[{time.time() - start_time:.2f}s] DE start – max_gen={max_generations}")
    result = differential_evolution(objective,
                                    bounds,
                                    maxiter=max_generations,
                                    popsize=popsize,
                                    tol=1e-7,
                                    polish=False,
                                    seed=42,
                                    workers=1,
                                    updating='deferred')
    seq = _gaussian_mixture(result.x, n)
    sc = _eval_fast(seq, L_fft)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] DE improved to {sc:.6f}")
    else:
        print(f"[{time.time() - start_time:.2f}s] DE finished – no improvement")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Sub‑gradient / Exponentiated gradient descent (non‑smooth)
# ----------------------------------------------------------------------
def _subgradient_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         max_iter: int = 3000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the exact max‑convolution objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_sc = _eval_fast(a, L_fft)

    n = a.shape[0]
    it = 0
    eta0 = 0.7                       # initial step size (for exponentiated update)

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a, L=L_fft)
        k = int(conv.argmax())       # active index – the one attaining the max

        # sub‑gradient of max at k:   g_i = 2 * a_{k-i}  (if i contributes)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # exponentiated‑gradient step  a_i ← a_i * exp(-η * g_i)
        eta = eta0 / math.sqrt(it)        # simple decay
        a = a * np.exp(-eta * grad)

        # enforce the simplex constraint
        a = _project_simplex(a)

        sc = _eval_fast(a, L_fft)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} score={sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4️⃣  Tiny random hill‑climb (accepted only if improvement)
# ----------------------------------------------------------------------
def _random_hill_climb(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       L_fft: int,
                       max_moves: int = 20000) -> tuple[np.ndarray, float]:
    """Very cheap local search – move a random epsilon between two indices."""
    a = a0.copy()
    best_sc = _eval_fast(a, L_fft)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_moves:
        it += 1
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.07   # up‑to 7 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)
        sc = _eval_fast(a_prop, L_fft)
        if sc < best_sc:
            a, best_sc = a_prop, sc
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} score={sc:.6f}")

    return a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The pipeline consists of:
        * a quick parametric sweep (geometric, power, beta, cosine, U‑exp)
        * a longer differential‑evolution on a Gaussian‑mixture model
        * an exponentiated‑gradient descent on the exact max‑convolution
        * a final random hill‑climb polish
    All candidates are projected onto the simplex after every modification.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.6   # small safety margin

    # ------------------------------------------------------------------
    # Choose a modest length – 4096 works well with FFT‑based ops.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Uniform sequence – provides the guaranteed upper bound 2.0
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter sequence is present (from previous runs) use it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = _project_simplex(starter)
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_seq, best_score = starter.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:                                         # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric families sweep
    # ------------------------------------------------------------------
    best_seq, best_score = _sweep_parametric(
        n, L_fft, deadline, start_time, best_seq, best_score)

    # ------------------------------------------------------------------
    # 2️⃣  Differential Evolution on Gaussian mixture (more time dedicated)
    # ------------------------------------------------------------------
    if time.time() < deadline - 250 and best_score > 1.8:   # only if we haven’t already beaten 2
        best_seq, best_score = _de_phase(
            n, L_fft, deadline, start_time, best_seq, best_score)

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient (sub‑gradient) descent – the core optimiser
    # ------------------------------------------------------------------
    while time.time() < deadline - 120:
        # start from current best + a tiny random jitter
        noise = 0.03 * np.random.randn(n) * best_seq
        start = best_seq + noise
        start = np.clip(start, 0.0, None)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 15.0)
        cand_seq, cand_score = _subgradient_descent(start, sub_deadline,
                                                    start_time, L_fft)
        if cand_score < best_score:
            best_seq, best_score = cand_seq.copy(), cand_score
            print(f"[{time.time() - start_time:.2f}s] SubGrad new global best={cand_score:.6f}")

        # early exit if we already reached a plausible bound (<1.55)
        if best_score < 1.55:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb polish (guaranteed not to worsen)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _random_hill_climb(
            best_seq, deadline, start_time, L_fft)

    # ------------------------------------------------------------------
    # Normalisation (the sequence must sum to 1 before returning)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Return a plain Python list – evaluate_sequence expects a list of floats.
    return [float(x) for x in best_seq]
