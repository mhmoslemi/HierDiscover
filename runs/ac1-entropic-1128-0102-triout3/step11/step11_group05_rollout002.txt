```
def propose_candidate(...):
    start = time.time()
    deadline = start + budget_s - 0.1
    n = 4096
    # start vector

    a = np.array(starter, dtype=float)
    # symmetrize

    # Helper: evaluate objective (fast)
    L_self = _next_pow2(2*n-1) etc.

    def eval_obj(a):
         conv = _conv_self(a, L_self)
         maxb = conv.max()
         return 2.0 * n * maxb/(a.sum()**2)

    best_a = a.copy()
    best_score = eval_obj(a)

    # Phase A: smooth max descent with Adam
    a_sm = a.copy()
    m = np.zeros_like(a_sm)
    v = np.zeros_like(a_sm)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    alphas = np.geomspace(1.0, 2000.0, num=20)
    L_self = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n-2)
    for alpha in alphas:
        while time.time() < deadline - 10.0: # reserve time for later phases
            # compute conv
            Af = np.fft.rfft(a_sm, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]

            # softmax weights
            exparg = alpha * conv
            max_exp = exparg.max()
            w = np.exp(exparg - max_exp)
            w /= w.sum()

            # compute gradient: 2 * (w * a_rev) conv
            a_rev = a_sm[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)

            grad = 2.0 * grad_full[n-1:n-1+n]

            it += 1
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2n\;\max (a*a)}{\bigl(\sum_i a_i\bigr)^2},
\qquad n=4096,
\]

where `a*a` is the linear self‑convolution of the non‑negative vector `a`.  
Because the objective is homogeneous, the vector can be normalised to a probability
distribution (`\sum a_i = 1`).  After normalisation the problem becomes

\[
\min_{a\ge 0,\;\sum a=1}\; 2n\;\max (a*a).
\]

The optimum is a *flat* convolution – the value of each entry of `a*a` should be as
close as possible to the average `1/(2n-1)`.  The programme below pursues this
goal by three complementary stochastic‑optimisation phases:

1. **Smooth‑max mirror descent (exponentiated‑gradient)** –  
   we minimise a smooth approximation of the `max` function using
   the soft‑max weight  
   \[
   w_i\propto\exp(\alpha\,(a*a)_i)
   \]
   and an Adam‑type adaptive step size.  The smoothing parameter `α` is
   increased geometrically up to 2000, which drives the algorithm to the true
   `max`‑subgradient.  After each update we enforce the simplex constraint
   and *symmetrise* the vector (`a←(a+rev(a))/2`) – the optimal solution is
   symmetric, and this projection dramatically reduces the search space.

2. **Projected sub‑gradient descent with momentum** –  
   a classic sub‑gradient method for the exact `max` objective, again with a
   simplex projection and occasional symmetrisation.  This phase cleans up the
   remaining irregularities left by the first phase.

3. **Incremental simulated annealing** –  
   a fast mass‑transfer annealer that updates the convolution *in‑place*,
   allowing a few‑million cheap moves inside the remaining time budget.
   The temperature schedule is exponential, and after the annealer we perform
   a short deterministic “local‑improve” sweep that moves mass from indices
   contributing to the current peak to indices where it lowers the peak.

All phases share a small collection of fast FFT‑based helpers.  The routine
stops well before the 1000 s limit, prints intermediate scores and finally
returns the best sequence found as a plain Python list.

```python
"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

The optimiser consists of three stages:
1. Smooth‑max (soft‑max) mirror descent with an Adam‑style adaptive
   learning rate and forced symmetry.
2. Projected sub‑gradient descent with momentum, again keeping the vector
   symmetric.
3. Fast incremental simulated‑annealing (mass transfers) followed by a short
   deterministic local improvement pass.

All phases respect the simplex constraint (sum = 1) and the non‑negativity
requirement.  The code never exceeds the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent (Adam, symmetry)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style exponentiated‑gradient on the soft‑max of the convolution."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    # smoothing parameters – more aggressive than the reference code
    alphas = np.geomspace(0.5, 2000.0, num=40)

    for alpha in alphas:
        # stop early if we run out of time
        while time.time() < deadline - 5.0:
            it += 1

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.shape[0]
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update (multiplicative – stays on the simplex)
            eta = 0.12 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

            # 5) simplex projection and symmetrisation
            a = np.maximum(a, 1e-30)
            a /= a.sum()
            a = (a + a[::-1]) / 2.0          # enforce symmetry
            a /= a.sum()                     # renormalise after symmetrisation

            # occasional real objective evaluation
            if it % 20000 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

            if time.time() >= deadline - 5.0:
                break
        if time.time() >= deadline - 5.0:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent (momentum, symmetry)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution (first max index)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # enforce symmetry (helps the objective)
        a = (a + a[::-1]) / 2.0
        a /= a.sum()

    return best_a, best_score

# Phase 3 – fast incremental simulated annealing
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 1e-2,
                        end_temp: float = 1e-10,
                        max_moves: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer with incremental convolution updates."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1, cheap for n=4096
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # denominator = 1 because sum(a)=1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        # exponential cooling schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.35   # move up to 35 % of src mass
        if eps <= 0.0:
            continue

        # incremental update of the convolution (copy‑on‑write)
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / T) > rng.random():
            # accept the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

# Deterministic local improvement (tiny moves that always lower the peak)
# ----------------------------------------------------------------------
def _local_improve(a: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iter: int = 20_000) -> np.ndarray:
    """Greedy mass transfers from indices contributing to the current peak."""
    conv = np.convolve(a, a)
    cur_max = conv.max()

    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # source = index inside the peak interval with the largest a
        src = idx[np.argmax(a[idx])]
        # destination = any index not in the interval (uniform pick)
        outside = np.setdiff1d(np.arange(n), idx)
        if outside.size == 0:
            break
        dst = outside[np.random.randint(outside.shape[0])]

        eps = a[src] * 0.2        # move 20 % of the source mass
        if eps < 1e-14:
            continue

        # incremental convolution update (same formula as the annealer)
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        if new_max < cur_max - 1e-12:          # accept only improving moves
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max

    return a

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣  Load an initial vector (global best if present, else uniform)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)

    # normalise to the simplex and enforce symmetry immediately
    starter /= starter.sum()
    starter = (starter + starter[::-1]) / 2.0
    starter /= starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Smooth‑max mirror descent (Adam, symmetry)
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:          # keep enough budget for later phases
        smooth_deadline = deadline - 130.0
        best_seq, best_score = _smooth_max_descent(
            best_seq, n, smooth_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best "
              f"score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient refinement (symmetry)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 20.0
        a_sub, sc_sub = _subgradient_descent(
            best_seq, n, sub_deadline, start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved "
                  f"score = {sc_sub:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Incremental simulated‑annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        anneal_deadline = deadline - 5.0
        a_anneal, sc_anneal = _incremental_anneal(
            best_seq, n, anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2,
            end_temp=1e-12,
            max_moves=5_000_000)
        if sc_anneal < best_score:
            best_score = sc_anneal
            best_seq = a_anneal.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved score = {sc_anneal:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final deterministic local improvement (tiny guaranteed moves)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq = _local_improve(best_seq, n, deadline, start_time)

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # one final evaluation for reporting
    final_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{final_score:.6f}")

    # return as plain Python list
    return [float(v) for v in best_seq]
```
