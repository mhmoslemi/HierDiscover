```
    best_a = a.copy()
    best_score = _eval_fast(a, _next_pow2(2*n-1))
    L_self = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n-2)
    it = 0
    alphas = np.geomspace(0.1, 2000.0, num=40) # more points
    max_iters_total = 800_000
    max_iters_per_alpha = max_iters_total // len(alphas) + 1
    while time.time() < deadline - 0.5:
        for alpha in alphas:
            for _ in range(max_iters_per_alpha):
                if time.time() >= deadline - 0.5:
                    break
                it +=1
                # compute conv via FFT (one FFT)
                Af = np.fft.rfft(a, n=L_self)
                conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]

                # compute weights
                exp_arg = alpha * conv
                max_arg = exp_arg.max()
                w = np.exp(exp_arg - max_arg)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0/w.shape[0])
                else:
                    w /= w_sum

                # gradient
                a_rev = a[::-1]
                Wf = np.fft.rfft(w, n=L_grad)
                Revf = np.fft.rfft(a_rev, n=L_grad)
                grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[:3*n-2]
                grad = 2.0 * grad_full[n-1:n-1 + n]

                # exponentiated gradient step
                eta = 0.08 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
                a = a * np.exp(-eta * grad)
                # enforce positivity and renorm
                a = np.maximum(a, 1e-30)
                a /= a.sum()

                if it % 500 == 0:
                    cur_score = _eval_fast(a, L_self)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                        print(f"[{time.time()-start_time:.2f}s] MD it {it} α={alpha:.2f} score={cur_score:.6f}")

            if time.time() >= deadline - 0.5:
                break
    return best_a, best_score
```

```
def _anneal(a0, n, deadline, start_time, cur_best_score, cur_best_seq):
    a = a0.copy()
    conv = np.convolve(a, a)  # full length 2n-1
    cur_score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    moves = 0
    while time.time() < deadline and moves < 200_000:
        moves += 1

        # Determine temperature linearly from start to end
        t_frac = (time.time() - start_time) / (deadline - start_time)
        T = 0.5 * (1.0 - t_frac) + 1e-12 # start 0.5 gradually to 0

        if T < 1e-12:
            T = 0.0

        # pick src with probability proportional to a (avoid zero)
        src = rng.choice(n, p=a/ a.sum())
        if a[src] < 1e-12:  # skip
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.3  # move up to 30%
        if eps <= 0.0:
            continue

        # new conv using incremental update (no copy)
        # Compute delta contributions
        # conv_new = conv + delta
        conv_new = conv.copy()
        # subtract contributions from src
        conv_new[src:src+n] -= 2.0 * eps * a
        # add contributions from dst
        conv_new[dst:dst+n] += 2.0 * eps * a
        # add self contributions
        conv_new[2*src] += eps*eps
        conv_new[2*dst] += eps*eps
        conv_new[src+dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        if delta_score <= 0.0 or (T > 0.0 and math.exp(-delta_score / T) > rng.random()):
            # accept
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time()-start_time:.2f}s] Anneal move {moves} best={best_score:.6f}")

    return best_seq, best_score
```

```
def _subgradient_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_self=_next_pow2(3*n-2)) # compute conv
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2) # sum=1 so denominator 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] SubGrad it={it} score={cur_score:.6f}")
        # find index of max conv
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        # subgradient
        grad[idx] = 2.0 * a[k - idx]
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full_like(a, 1.0 / n)
    return best_a, best_score
```

assistantfinal**Solution Overview**

This implementation searches for a non‑negative step‑function `a` (length 4096) that minimizes  

\[
\mathrm{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad n=4096 .
\]

The optimizer is built from three independent phases that run in the 1000 s time budget:

1. **Multi‑restart smoothed mirror‑descent** – a fast exponentiated‑gradient method
   that minimises the *log‑sum‑exp* approximation of the maximum convolution.
   Several random initialisations (uniform, Dirichlet, sparse) are tried and the
   best intermediate result is kept.

2. **Projected sub‑gradient refinement** – a short momentum‑based sub‑gradient
   step that works directly on the true (non‑smooth) maximum.

3. **Targeted simulated‑annealing polish** – mass‑transfer moves that update the
   convolution incrementally.  The temperature schedule is generous at the start,
   allowing the search to escape shallow basins, and decays to near‑zero for a
   fine‑grained greedy finish.

All steps keep the vector on the probability simplex (`sum = 1`) and respect the
non‑negativity requirement.  Helper functions use FFT‑based convolution for speed
and a fast in‑place convolution update for the annealer.

The final sequence is normalised and returned as a plain Python list; it can be
directly fed to `evaluate_sequence`.

```python
"""
Multi‑restart smoothed mirror descent + sub‑gradient refinement + targeted annealing.

Goal: minimise 2·n·max(conv(a,a)) / (sum(a)**2) for n = 4096 under the
strict non‑negative, sum‑normalised constraints.  The algorithm
  * launches several random starts,
  * runs a fast exponentiated‑gradient (mirror) descent on the log‑sum‑exp
    smoothed objective,
  * refines the best candidate with a projected sub‑gradient step,
  * finally performs an incremental simulated‑annealing polish that updates the
    convolution in‑place.
All phases respect the overall 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (full length 2·len(a)-1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smoothed mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Runs exponentiated‑gradient on the log‑sum‑exp of the self‑convolution.
    Returns the best vector seen and its (fast) score.
    """
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # geometric schedule of smoothing parameters (more aggressive than the original)
    alphas = np.geomspace(0.2, 2000.0, num=40)

    it = 0
    max_iterations = 340_000                     # overall budget for this phase
    it_per_alpha = max_iterations // len(alphas) + 1

    for alpha in alphas:
        for _ in range(it_per_alpha):
            if time.time() >= deadline - 0.5:
                break
            it += 1

            # 1) a*a (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights w = softmax(α·conv)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-30)       # avoid exact zeros
            a /= a.sum()

            # occasional real‑objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] MD α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.5:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained sub‑gradient descent on the true max‑convolution.
    Uses a simple momentum scheme and projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.35
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        # full self‑convolution (FFT for speed)
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # index of the active maximal entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]        # sub‑gradient of max‑convolution

        # momentum + simplex‑projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a.fill(1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : incremental simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray,
            n: int,
            deadline: float,
            start_time: float,
            cur_best_score: float,
            cur_best_seq: np.ndarray,
            start_temp: float = 1e-1,
            end_temp: float = 1e-10,
            max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealer that updates the convolution in‑place.
    Uses a linear temperature schedule (fast decay) and accepts
    improving moves greedily.
    """
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2·n‑1 (cheap for n=4096)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max           # sum(a)=1 ⇒ denominator = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_moves:
        it += 1

        # temperature decreasing linearly from start_temp to end_temp
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # pick a source index proportional to its mass (avoid empty src)
        total = a.sum()
        if total == 0.0:
            break
        src = rng.choice(n, p=a / total)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.30   # move up to 30 % of src mass
        if eps <= 0.0:
            continue

        # incremental update of convolution (copy‑on‑write)
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta <= 0.0 or (T > 0.0 and math.exp(-delta / T) > rng.random()):
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Build a collection of diverse starting points
    # ------------------------------------------------------------------
    starters = []

    # uniform
    starters.append(np.full(n, 1.0 / n))

    # previous best if available
    try:
        h = np.asarray(height_sequence_1, dtype=float)
        if h.ndim == 1 and h.size > 0:
            if h.size > n:

            elif h.size < n:

            h = np.maximum(h, 0.0)

            else:

            starters.append(h)

        a[idx] = vals
        a /= a.sum()
        starters.append(a)

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1️⃣  Multi‑restart smoothed mirror descent
    # ------------------------------------------------------------------
    for i, init in enumerate(starters):

        if sc_md < best_score:

            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Improved after sub‑grad: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final annealing polish (bulk of the remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        a_fin, sc_fin = _anneal(best_seq, n, deadline,

    # ------------------------------------------------------------------

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
