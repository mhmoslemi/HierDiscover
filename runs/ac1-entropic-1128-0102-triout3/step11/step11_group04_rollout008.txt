"""
Implementation of a multi‑restart sub‑gradient optimizer with a
low‑dimensional random search (differential evolution) for the
non‑negative sequence that minimises ``evaluate_sequence`` for n=4096.
The algorithm respects a wall‑clock budget and returns the best
sequence found.
"""

import time
import math
import numpy as np

# Global constants
N = 4096
L_SELF = 1 << ((2 * N - 2).bit_length())   # power of two length for FFT convolution

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self(a, L=None):
    """Linear self‑convolution a*a using FFT. Returns length 2·len(a)‑1."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a, L_self=L_SELF):
    """Exact objective for a sequence that already sums to 1."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    # sum is 1 by construction – guard against tiny values
    if a.sum() < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b)

def _subgradient_descent(a0, deadline, start_time,
                         max_iters=200_000, step0=0.4,
                         momentum_beta=0.9):
    """
    Projected sub‑gradient descent on the exact objective.
    Returns the best sequence visited and the best score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(best_a)
    n = a.shape[0]
    momentum = np.zeros_like(a)
    it = 0
    stagnation = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max   # sum(a)=1
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} best={best_score:.6f}")
        else:
            stagnation += 1

        if stagnation >= 20_000:
            break

        # active indices (where the max is attained)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # compute sub‑gradient of max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + projection
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

def _random_sparse_initial(m, rng):
    """Return a probability vector with exactly m non‑zero entries."""
    indices = rng.choice(N, size=m, replace=False)
    values = rng.random(m)
    seq = np.zeros(N, dtype=float)
    seq[indices] = values
    if seq.sum() == 0.0:
        seq[indices[0]] = 1.0
    seq /= seq.sum()
    return seq

def _run_restart(seed, sparsity, deadline, start_time):
    """Run a short sub‑gradient run from a random start."""
    rng = np.random.default_rng(seed)
    a0 = _random_sparse_initial(sparsity, rng)
    # limited number of iterations – we rely on outer loop to allocate time
    sub_deadline = deadline
    a_opt, sc_opt = _subgradient_descent(a0, sub_deadline, start_time,
                                         max_iters=5000, step0=0.5)
    return a_opt, sc_opt

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Multi‑restart optimiser for the max‑convolution objective.

    The algorithm consists of three stages:
    1. A quick low‑dimensional differential‑evolution search on a
       block‑wise parametrisation (B=64).  The block weights are
       transformed with a softmax so the simplex constraint is satisfied.
    2. A large number of random‑restart projected sub‑gradient runs,
       each starting from a sparse random vector.  The best result of
       this stage becomes the start for a final long sub‑gradient polish.
    3. A short hill‑climb that moves a tiny fraction of mass between
       two indices and keeps improvements.

    The returned list is normalised to have sum = 1, guaranteeing that
    ``evaluate_sequence`` receives a valid candidate.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = N
    L_self = L_SELF

    # ------------------------------------------------------------------
    # Stage 0 – baseline – try to use the global best if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:

    except Exception:
        starter = np.full(n, 1.0 / n)

        else:

```python
   random starts (the bulk of the time budget);
3. a long sub‑gradient polish from the best point found;
4. a tiny hill‑climb that only accepts improving mass‑transfer moves.

All stages obey the wall‑clock deadline (default 1000 s) and the
returned sequence is normalised to sum = 1, guaranteeing a valid call to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global constants (problem size)
# ----------------------------------------------------------------------
N = 4096                     # length of the step function
L_SELF = 1 << ((2 * N - 2).bit_length())   # FFT length for self‑convolution


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_self: int = L_SELF) -> float:
    """
    Exact objective for a sequence that already satisfies sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    if a.sum() < 1e-12:
        return float("inf")
    return float(2.0 * a.shape[0] * max_b)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 200_000,
                         step0: float = 0.4,
                         momentum_beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent with Euclidean projection onto the simplex.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(best_a)
    n = a.shape[0]
    momentum = np.zeros_like(a)
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max   # sum(a) is 1 by construction

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} best={best_score:.6f}")
        else:
            stagnation += 1

        # early stop if we are stuck for a long time
        if stagnation >= 20_000:
            break

        # indices where the maximum is attained (active constraints)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + projection
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random sparse initialiser
# ----------------------------------------------------------------------
def _random_sparse_initial(m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Create a probability vector with exactly `m` non‑zero entries.
    The non‑zero values are drawn uniformly at random and then normalised.
    """
    indices = rng.choice(N, size=m, replace=False)
    values = rng.random(m)
    a = np.zeros(N, dtype=float)
    a[indices] = values
    if a.sum() == 0.0:
        a[indices[0]] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Single short restart (used in the massive‑restart loop)
# ----------------------------------------------------------------------
def _run_restart(seed: int, sparsity: int,
                 deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Run a short sub‑gradient optimisation from a random sparse start.
    Returns the best sequence found in that run and its score.
    """
    rng = np.random.default_rng(seed)
    a0 = _random_sparse_initial(sparsity, rng)
    # a modest number of iterations – the outer loop will launch many restarts
    a_opt, sc_opt = _subgradient_descent(a0, deadline, start_time,
                                         max_iters=5_000,
                                         step0=0.5,
                                         momentum_beta=0.9)
    return a_opt, sc_opt


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` for n = 4096.

    The procedure follows the four stages described in the module header.
    The final list is normalised to have sum = 1, guaranteeing that
    `evaluate_sequence` receives a valid candidate.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the external harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Stage 0 – initialise from the global best if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(N, 1.0 / N)

    if starter.size != N:
        if starter.size > N:
            starter = starter[:N]
        else:
            starter = np.pad(starter, (0, N - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(N, 1.0 / N)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – cheap block‑wise differential evolution
    # ------------------------------------------------------------------
    stage1_deadline = start_time + 0.15 * budget_s
    if time.time() < stage1_deadline:
        try:
            from scipy.optimize import differential_evolution
        except Exception:
            print("[Stage 1] SciPy not available – skipping DE.")
        else:
            B = 64                         # number of blocks
            block_len = N // B
            if B * block_len != N:
                raise RuntimeError("Block length does not divide N")
            # objective for a block‑vector y (pre‑softmax)
            def block_obj(y):
                # soft‑max enforces the simplex constraint
                max_y = np.max(y)
                exp_y = np.exp(y - max_y)
                block_weights = exp_y / exp_y.sum()           # sum = 1
                a = np.repeat(block_weights / block_len, block_len)
                return _eval_fast(a)

            bounds = [(-10.0, 10.0)] * B
            result = differential_evolution(
                block_obj,
                bounds,
                maxiter=30,
                popsize=12,
                polish=False,
                init='latinhypercube',
                updating='deferred',
                disp=False,
                tol=1e-6,
                workers=1,
                seed=seed,
            )
            y_best = result.x
            max_y = np.max(y_best)
            exp_y = np.exp(y_best - max_y)
            block_weights = exp_y / exp_y.sum()
            a_de = np.repeat(block_weights / block_len, block_len)
            de_score = _eval_fast(a_de)
            print(f"[{time.time() - start_time:.2f}s] Stage 1 DE score = {de_score:.6f}")
            if de_score < best_score:
                best_score = de_score
                best_seq = a_de.copy()

    # ------------------------------------------------------------------
    # Stage 2 – many random‑restart short sub‑gradient runs
    # ------------------------------------------------------------------
    stage2_deadline = deadline - 5.0   # keep a few seconds for polishing & hill‑climb
    rng = np.random.default_rng(seed + 12345)
    restart_counter = 0

    while time.time() < stage2_deadline:
        # choose a sparsity level (between 8 % and 30 % of the entries)
        sparsity = int(rng.integers(low=int(0.08 * N), high=int(0.30 * N)))
        a_opt, sc_opt = _run_restart(seed + restart_counter,
                                      sparsity,
                                      stage2_deadline,
                                      start_time)
        restart_counter += 1
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {restart_counter} improved score = {best_score:.6f}")

        # stop as soon as we beat the target bound
        if best_score <= 1.503:
            print("[Stage 2] Target bound reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Stage 3 – long sub‑gradient polishing from the current best point
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_polished, sc_polished = _subgradient_descent(
            best_seq,
            deadline,
            start_time,
            max_iters=200_000,
            step0=0.3,
            momentum_beta=0.9,
        )
        if sc_polished < best_score:
            best_score = sc_polished
            best_seq = a_polished.copy()
            print(f"[{time.time() - start_time:.2f}s] Polishing improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – tiny hill‑climb for the very last tweaks
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = best_score
        rng = np.random.default_rng(seed + 99999)
        while time.time() < deadline:
            src = rng.choice(N, p=a)         # source chosen proportionally to its mass
            dst = rng.integers(0, N)         # uniform destination
            if src == dst or a[src] < 1e-15:
                continue
            # move up to 25 % of the source mass
            eps = a[src] * rng.random() * 0.25
            if eps <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_score = _eval_fast(a_prop)
            if new_score < cur_score - 1e-12:
                a = a_prop
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```

The optimiser respects the wall‑clock limit, works entirely with NumPy (and optionally SciPy for the cheap DE stage), and returns a valid list of non‑negative heights that is very likely to improve on the previously reported bound `1.508668`.
