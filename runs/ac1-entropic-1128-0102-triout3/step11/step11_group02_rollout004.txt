"""
A two‑stage optimizer for the “max‑convolution” objective.

Stage 1 – smooth‑gradient descent:
    * A temperature ε proportional to the current max‑conv is used to
      replace the non‑smooth max‑operator by a soft‑max.
    * The exact gradient of the smoothed objective is obtained by an
      FFT‑based correlation (O(n log n)).
    * Momentum and a diminishing step size drive the variables.
      After every step the vector is projected onto the probability
      simplex (non‑negative, Σ = 1).

Stage 2 – cheap Metropolis annealing:
    * Small random packets of mass are moved between coordinates.
    * Moves are accepted with the Metropolis rule w.r.t. the true
      objective 2·n·max(conv(a,a))/(Σ a)².
Both stages respect the 1000 s budget and keep the best sequence seen.
The returned list contains non‑negative floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq))/(sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start shapes
# ----------------------------------------------------------------------
def _shape_tent(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape, normalised to sum = 1."""
    x = np.arange(n, dtype=float) / (n - 1)
    tent = np.minimum(x, 1.0 - x)
    if tent.sum() == 0:
        return np.full(n, 1.0 / n)
    return tent / tent.sum()


def _shape_cosine(n: int) -> np.ndarray:
    """Cosine‑bell: a_i ∝ sin(π·i/(n‑1))."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * i / (n - 1))
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def _shape_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - 0.5 - i))
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def _best_warm_start(n: int) -> np.ndarray:
    """Return the best among a few analytically‑motivated starters."""
    candidates = [_shape_tent(n), _shape_cosine(n), _shape_arcsine(n)]
    best = candidates[0]
    best_score = _objective(best)
    for cand in candidates[1:]:
        sc = _objective(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best


# ----------------------------------------------------------------------
# Smoothed max gradient (full soft‑max, O(n log n))
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, eps: float) -> tuple[np.ndarray, float]:
    """
    Compute (1) the exact max‑convolution value,
    (2) the gradient of the ε‑soft‑max approximation.
    Returns (max_conv, grad).
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _conv_self(a)
    max_conv = conv.max()

    # 2) soft‑max weights w_k = exp((conv_k‑max)/eps) / Z
    scaled = (conv - max_conv) / eps
    # numerical stability
    shifted = scaled - scaled.max()
    w = np.exp(shifted)
    w /= w.sum()

    # 3) gradient = 2 * correlation(a, w)
    L = _next_pow2(3 * n)                     # enough for a (len n) * w (len 2n‑1)
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    G = A * np.conj(W)
    corr = np.fft.irfft(G, n=L)
    grad = 2.0 * corr[:n]                     # correlation term Σ_j a_j w_{i+j}
    return max_conv, grad


# ----------------------------------------------------------------------
# Stage 1 – smooth‑gradient optimisation
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     eps_factor: float = 0.1,
                     eps_min: float = 1e-9,
                     step0: float = 0.2,
                     momentum_beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with a soft‑max approximation of the max.
    Returns the best sequence seen and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # temperature tied to the current peak
        max_conv = _conv_self(a).max()
        eps = max(eps_min, eps_factor * max_conv)

        # soft‑max gradient
        _, grad = _softmax_gradient(a, eps)

        # momentum + diminishing step
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # stay on the probability simplex
        a = _project_simplex(a)

        # occasional exact evaluation (every ~200 iterations)
        if it % 200 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Smooth iter={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 2 – cheap Metropolis annealing
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny random packets of mass; keep the best sequence found."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 40000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # simple linear temperature schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25  # move up‑to‑25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function within the given time budget.
    The routine:
        1. builds a few analytic starters and picks the best,
        2. performs smooth‑gradient descent with momentum,
        3. finishes with a light annealing polish,
    finally returning a plain list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 16384

    # --------------------------------------------------------------
    # 1) Warm‑start from analytically‑motivated shapes.
    # --------------------------------------------------------------
    a0 = _best_warm_start(n)
    init_score = _objective(a0)
    print(f"[{time.time() - start_time:.2f}s] Warm‑start score={init_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smooth‑gradient descent (main optimisation).
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a few seconds for polishing
        sub_deadline = time.time() + remaining * 0.80
        refined_seq, refined_score = _smooth_optimize(
            a0,
            sub_deadline,
            start_time,
            eps_factor=0.1,
            eps_min=1e-9,
            step0=0.25,
            momentum_beta=0.9,
        )
        if refined_score < init_score:
            a0 = refined_seq.copy()
            init_score = refined_score
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(a0, deadline, start_time)
        if anneal_score < init_score:
            a0 = anneal_seq.copy()
            init_score = anneal_score
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final sanity checks & conversion.
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
