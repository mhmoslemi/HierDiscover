    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of real vectors x and y using a real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    m = x.shape[0] + y.shape[0] - 1
    L = _next_pow2(m)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:m]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) / (sum(seq)**2).  Identical to
    evaluate_sequence but without the defensive checks.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, Σ w = 1} (O(n log n) algorithm)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generator (trapezoidal / tent‑with‑plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length n.
    p       – exponent of the side ramps (p>0).
    alpha   – relative plateau width ∈ [0, 0.5];
              alpha = 0 → pure tent, alpha = 0.5 → uniform.
    The vector is normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    # keep the plateau even so that the shape stays perfectly symmetric
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                      # rises from ≈0 to 1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # safety for rounding artefacts
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)

    return shape


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of the max‑convolution.
    Uses a temperature β that is increased geometrically; for a given β
    the objective is  (1/β) log Σ exp(β·conv_i).
    The gradient can be written as a convolution of the soft‑max weights
    with the reversed sequence, which we compute via FFT.
    Returns the best iterate seen and its (non‑smooth) score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum for faster convergence
    mom = np.zeros_like(a)
    beta_mom = 0.9

    # temperature schedule
    beta_start = 0.2
    beta_end = 2000.0
    levels = 30               # number of temperature increases
    it = 0

    # total number of inner steps – we keep it modest to stay within budget
    max_iters = 5000

    # pre‑compute the number of steps per level (roughly equal)
    steps_per_level = max_iters // levels

    for lvl in range(levels):
        # geometric increase of β
        beta = beta_start * (beta_end / beta_start) ** (lvl / (levels - 1))
        for _ in range(steps_per_level):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            # ----- smooth max -----
            conv = _conv_self(a)                     # length 2n‑1
            cmax = conv.max()
            # stable soft‑max weights
            w = np.exp(beta * (conv - cmax))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient w.r.t. a -----
            a_rev = a[::-1]
            grad_full = _fft_convolve(w, a_rev)     # length 3n‑2
            grad = 2.0 * grad_full[n - 1:n - 1 + n]  # slice the relevant part

            # ----- update -----
            mom = beta_mom * mom + (1.0 - beta_mom) * grad
            step = 0.6 / math.sqrt(it)              # decaying step size
            a -= step * mom
            a = _project_simplex(a)

            # occasional evaluation of the true (non‑smooth) objective
            if it % 200 == 0:
                cur_score = _objective(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smooth‑GD it={it:,} "
                          f"β={beta:.1f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (identical to the original version)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

        # index where the max is attained
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Randomly moves a tiny fraction of mass; Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) achieving a low
    value of evaluate_sequence.  Returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – larger canvas reduces discretisation error
    n = 65536  # power of two → fast FFTs

    # --------------------------------------------------------------
    # 1) A quick grid / random search over the analytic family
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    top_k = 5
    top_candidates: list[tuple[float, np.ndarray]] = []

    p_grid = np.linspace(0.2, 2.2, 21)      # 0.2 … 2.2
    alpha_grid = np.linspace(0.0, 0.5, 13)  # 0 … 0.5

    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_shape(n, p, alpha)
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
            if len(top_candidates) < top_k:
                top_candidates.append((score, a.copy()))
                top_candidates.sort(key=lambda x: x[0])
            else:
                if score < top_candidates[-1][0]:
                    top_candidates[-1] = (score, a.copy())
                    top_candidates.sort(key=lambda x: x[0])

            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # a few random draws for diversity
    rng = np.random.default_rng()
    for _ in range(300):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.1, 2.5)
        alpha = rng.uniform(0.0, 0.5)
        a = _make_shape(n, p, alpha)
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        if len(top_candidates) < top_k:
            top_candidates.append((score, a.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if score < top_candidates[-1][0]:
                top_candidates[-1] = (score, a.copy())
                top_candidates.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] Grid/random best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) If a global starter sequence exists, consider it
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    if len(top_candidates) < top_k:
                        top_candidates.append((cur, starter.copy()))
                        top_candidates.sort(key=lambda x: x[0])
                    else:
                        if cur < top_candidates[-1][0]:
                            top_candidates[-1] = (cur, starter.copy())
                            top_candidates.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) Smooth‑max gradient descent on the best candidate
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    remaining = deadline - time.time()
    if remaining > 30.0:  # leave a few seconds for polishing steps
        gd_deadline = time.time() + 0.7 * remaining
        refined_seq, refined_score = _smooth_gradient_descent(best_seq,
                                                             gd_deadline,
                                                             start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth‑GD improved score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light sub‑gradient refinement (helps after the temperature jump)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        sub_deadline = time.time() + 0.25 * remaining
        refined_seq2, refined_score2 = _subgradient_descent(best_seq,
                                                            sub_deadline,
                                                            start_time)
        if refined_score2 < best_score:
            best_score = refined_score2
            best_seq = refined_seq2.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish score = {refined_score2:.6f}")

    # --------------------------------------------------------------
    # 5) Optional annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
