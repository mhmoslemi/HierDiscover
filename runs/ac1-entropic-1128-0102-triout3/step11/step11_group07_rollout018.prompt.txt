def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)

    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _gaussian_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """Build a non‑negative sequence of length n from a mixture of Gaussians."""
    seq = np.zeros(n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    K = len(params) // 3
    for i in range(K):
        amp = max(0.0, params[3 * i])          # non‑negative amplitude
        cen = params[3 * i + 1]
        wid = max(1e-3, params[3 * i + 2])    # avoid zero width

    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)
    return seq

def _lp_refine(a0: np.ndarray,

            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        # active set: indices whose value is within tol_frac of the maximum
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])
        K = active.size

        # build dense A_ub (K x n)
        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1)
            i = k - j

        c = -np.ones(n, dtype=np.float64)

                      A_ub=A_ub,
                      b_ub=b_ub,

            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()                 # re‑normalise to sum‑1
