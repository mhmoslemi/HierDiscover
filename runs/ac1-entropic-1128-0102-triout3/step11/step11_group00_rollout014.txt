def solve_lp(f, M):
    n = len(f)
    rows = 2*n - 1

    # Build data for sparse matrix
    # We will use `scipy.sparse.csr_matrix` directly
    # Use list of row indices, col indices, data.
    row_idx = []
    col_idx = []
    data = []

    for k in range(rows):
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # j = i indices
        # store f[k - i] for each i in [i_min,i_max]
        # number of entries m = i_max - i_min + 1
        # for efficiency use slice view
        # iterate using range
        for i in range(i_min, i_max+1):
            row_idx.append(k)
            col_idx.append(i)
            data.append(f[k - i])
    A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))

    # A_ub * g <= M, but we need each row bound individually: b = M * np.ones(rows)
    b = np.full(rows, M)

    # objective: maximize sum(g) => minimize -ones dot g
    c = -np.ones(n)

    # bounds: (0, None)
    bounds = [(0, None)] * n

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        print('LP solve failed', res.message)
        # fallback: return uniform
        return np.full(n, 1.0/n)
    g = res.x
    return g

"""
Propose a step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
1️⃣  Quick parametric sweep (geometric / power / β‐families) to obtain a
    decent starting vector (normalised to sum = 1).

2️⃣  A few iterations of the *LP‑mix* scheme described in the statement:
    – fix the current f, compute M = max(f∗f);
    – solve the linear program   max 1ᵀg   s.t.   f∗g ≤ M , g≥0 ;
    – normalise g to sum = 1 and replace f ← (1‑t)·f + t·g (t≈0.5).
    The LP is built with a sparse Toeplitz matrix (size (2n‑1)×n) and
    solved by ``scipy.optimize.linprog`` (HiGHS).  This step usually
    yields a noticeable drop in the objective.

3️⃣  A long projected sub‑gradient descent on the simplex (momentum,
    step ∝1/√it) – the same machinery used previously – to polish the
    vector further.

4️⃣  A final micro‑hill‑climb that moves tiny amounts of mass between
    random coordinates and keeps only improving moves.

All phases respect the overall 1000 s budget.  The routine returns a
plain list of Python floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweep (geometric, power, beta families)
# ----------------------------------------------------------------------
def _parametric_sweep(n: int,
                      best_seq: np.ndarray,
                      best_score: float,
                      start_time: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Very fast family sweep – updates best_seq / best_score in‑place."""
    L_fft = _next_pow2(2 * n - 1)
    uniform = np.full(n, 1.0 / n)
    idx = np.arange(n, dtype=np.float64)

    # geometric families
    for eps in np.logspace(-4, -1, 70):
        if time.time() >= deadline - 5.0:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # power families
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline - 5.0:
            break
        pow_seq = np.power(idx + 1, p)
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type families
    betas = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in betas:
        for q in betas:
            if time.time() >= deadline - 5.0:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  LP‑mix improvement (solve max‑mass g under f∗g ≤ M)
# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max 1ᵀg   s.t.   f∗g ≤ M , g≥0.
    Returns the optimal g (may be dense).
    """
    import scipy.optimize as opt
    import scipy.sparse as sp

    n = f.shape[0]
    # Build sparse Toeplitz matrix A where A[k,j] = f[k‑j] (zero when out of range)
    diagonals = [np.full(n, f[d], dtype=float) for d in range(n)]
    offsets = -np.arange(n)                     # sub‑diagonals
    A = sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')

    b = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)               # minimise -sum(g)  → maximise sum(g)
    bounds = [(0.0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f'LP failed: {res.message}')
    return res.x


def _lp_improve(start_seq: np.ndarray,
                start_score: float,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a handful of LP‑mix iterations (t≈0.5, then decay).
    Returns the best sequence found.
    """
    n = start_seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_score = start_score
    t = 0.5                # mixing weight

    it = 0
    while time.time() < deadline - 5.0 and it < 20:
        it += 1
        conv = _conv_self(seq, L=L_fft)
        M = conv.max()
        try:
            g = _solve_lp_for_g(seq, M)
        except Exception as e:
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} failed: {e}")
            break

        if g.sum() < 1e-12:
            print(f"[{time.time() - start_time:.2f}s] LP returned almost‑zero g, stopping.")
            break

        g_norm = g / g.sum()          # enforce sum = 1
        seq_new = (1.0 - t) * seq + t * g_norm
        seq_new = np.maximum(seq_new, 0.0)
        seq_new /= seq_new.sum()

        sc = _eval_fast(seq_new, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq_new.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {it} improved score={sc:.6f}")

        seq = seq_new
        t = max(0.1, t * 0.9)          # slowly shrink mixing weight

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Long projected sub‑gradient descent (same as the previous code)
# ----------------------------------------------------------------------
def _subgradient_descent_long(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Heavy‑weight sub‑gradient descent on    f(a)=2·n·max(conv(a,a))
    (sum of a kept = 1 by projection onto the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    beta = 0.9
    alpha0 = 0.8
    it = 0
    last_improve = 0
    L_fft = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the active convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = alpha0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

        # escape if stagnating
        if it - last_improve > 5000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score

def _tiny_hill_climb(start_a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int) -> tuple[np.ndarray, float]:
    """
    Move ε≈10⁻⁴·a_src from a random source to a random destination,
    accepting only improving moves.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        sc = _eval_fast(a_prop, L_fft)
        if sc < best_score:
            a = a_prop
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={best_score:.6f}")

    return best_a, best_score

            best_seq, n, deadline - 5.0, start_time, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Sub‑grad finished score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _tiny_hill_climb(best_seq, deadline, start_time, L_fft)

    # ------------------------------------------------------------------
    # Normalise finally and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
