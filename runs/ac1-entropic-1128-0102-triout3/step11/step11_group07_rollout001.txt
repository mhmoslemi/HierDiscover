"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

by combining:
* a dense parametric sweep (geometric, power, beta families),
* several restarts of Adam‑style smoothed gradient descent,
* a higher‑dimensional Gaussian‑mixture differential‑evolution,
* an active‑set LP refinement (if scipy is present),
* a dedicated exact sub‑gradient descent on the true max‑convolution,
* a final cheap hill‑climb.

The routine respects the 1000 s budget, prints progress and returns a list
compatible with ``evaluate_sequence``.  The key new ingredient compared to
the previous baseline is the exact sub‑gradient phase, which uses the true
max‑convolution (no soft‑max) and a simple decreasing step‑size rule.
"""

import time
import math
import numpy as np

# optional SciPy imports – if unavailable the corresponding phases are skipped
try:
    from scipy.optimize import linprog, differential_evolution
    _has_scipy = True
except Exception:  # pragma: no cover
    _has_scipy = False

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2*len(a)-1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast version of evaluate_sequence (assumes sum≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

def _search_parametric(n, L_fft, deadline, start_time, best_seq, best_score):
    idx = np.arange(n)

    # geometric family
    eps_grid = np.logspace(-4, -1, 140)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.1f} score={sc:.6f}")

    # power family
    p_grid = np.linspace(-1.0, 2.0, 51)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type family (including edge cases)
    beta_vals = np.linspace(0.0, 2.0, 13)
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            s = beta_seq.sum()
            if s == 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    return best_seq, best_score

# 2️⃣ Adam‑style smoothed gradient descent (soft‑max)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         sub_deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """Adam optimizer on the soft‑max smoothed objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    tau = 0.05                # initial temperature
    tau_min = 1e-8
    tau_decay = 0.9995
    step0 = 0.15
    it = 0

    while time.time() < sub_deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        # soft‑max weights (stable)
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient via FFT (grad_j = 2 * Σ_i a_i w_{i+j})
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update & simplex projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

# 3️⃣ Gaussian‑mixture DE search (low‑dimensional)
# ----------------------------------------------------------------------
def _gaussian_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """Build a non‑negative sequence of length n from a mixture of Gaussians."""
    seq = np.zeros(n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    K = len(params) // 3
    for i in range(K):
        amp = max(0.0, params[3 * i])
        cen = params[3 * i + 1]
        wid = max(1e-3, params[3 * i + 2])
        seq += amp * np.exp(- ((idx - cen) ** 2) / (2.0 * wid * wid))
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)
    return seq


def _de_optimise(n: int,
                 L_fft: int,
                 deadline: float,
                 start_time: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """Run differential evolution on a Gaussian‑mixture parameterisation."""
    if not _has_scipy:                     # pragma: no cover
        return best_seq, best_score

    K = 8                                 # more components than before
    dim = 3 * K
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 10.0))        # amplitude
        bounds.append((0.0, n - 1))       # centre
        bounds.append((1.0, n / 2.0))     # width

    def de_obj(params):
        seq = _gaussian_sequence(params, n)
        return _eval_fast(seq, L_fft)

    remaining = deadline - time.time()
    # allocate ~30 % of remaining time, but cap iterations
    maxiter = max(1, int(remaining // 15))
    maxiter = min(maxiter, 30)

    print(f"[{time.time() - start_time:.2f}s] Starting DE (maxiter={maxiter})")
    result = differential_evolution(de_obj,
                                    bounds,
                                    maxiter=maxiter,
                                    popsize=14,
                                    tol=1e-7,
                                    polishing=False,
                                    seed=0,
                                    workers=1,
                                    updating='deferred')
    seq_de = _gaussian_sequence(result.x, n)
    score_de = _eval_fast(seq_de, L_fft)
    if score_de < best_score:
        best_score = score_de
        best_seq = seq_de.copy()
        print(f"[{time.time() - start_time:.2f}s] DE improved score={score_de:.6f}")

    return best_seq, best_score

# 4️⃣ Active‑set LP refinement (if SciPy available)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 8,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Iteratively improve a by solving a small LP on the active constraints."""
    if not _has_scipy:                     # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])
        K = active.size

        # Build A_ub (K x n) with convolution structure
        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)   # maximise sum(g) ↔ minimise -sum(g)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(1, int(deadline - time.time()))})

        if not res.success:                     # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()

        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)
        if new_score < best_score:
            best_score = new_score
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP refine it={it} score={new_score:.6f}")
        else:
            break

    return a, best_score

# 5️⃣ Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int,
                     max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    step0 = 0.25          # initial step size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L=L_fft)
        max_val = conv.max()
        # indices that are (almost) maximal – tolerance helps stability
        idxs = np.where(conv >= max_val * 0.9999999)[0]
        # compute averaged gradient from all active indices
        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # j runs from lo..hi, corresponding a[k-j]
            j = np.arange(lo, hi + 1)
            grad[j] += 2.0 * a[k - j]
        grad /= max(1, len(idxs))

        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} score={cur_score:.6f}")

        # optional early stop when improvement stalls
        if it % 2000 == 0 and time.time() >= deadline:
            break

    return best_a, best_score

# 6️⃣ Tiny random hill‑climb (final polish)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Random tiny transfers accepted only if they improve."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # bias towards larger entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.12
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")
    return a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The search mixes a parametric sweep, Adam smoothing, DE over Gaussian
    mixtures, LP refinement, an exact sub‑gradient phase and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # Baseline uniform
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (if provided)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:                               # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Fast parametric sweep
    # ------------------------------------------------------------------
    best_seq, best_score = _search_parametric(n, L_fft, deadline,
                                              start_time,
                                              best_seq, best_score)

    # ------------------------------------------------------------------
    # 2️⃣ Adam smooth descent (multiple restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        base = best_seq if np.random.rand() < 0.65 else uniform
        noise = 0.04 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 12.0)
        cand_seq, cand_score = _adam_smooth_descent(start, sub_deadline,
                                                    start_time,
                                                    L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth new global best={best_score:.6f}")

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣ Gaussian‑mixture DE (low‑dimensional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0 and best_score > 1.503:
        best_seq, best_score = _de_optimise(n, L_fft, deadline,
                                            start_time,
                                            best_seq, best_score)

    # ------------------------------------------------------------------
    # 4️⃣ Active‑set LP refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0 and best_score > 1.503:
        best_seq, best_score = _lp_refine(best_seq, deadline,
                                          start_time, L_fft)

    # ------------------------------------------------------------------
    # 5️⃣ Exact sub‑gradient descent on max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0 and best_score > 1.503:
        sub_deadline = deadline - 7.0
        best_seq, best_score = _subgrad_descent(best_seq,
                                                sub_deadline,
                                                start_time,
                                                L_fft,
                                                max_iters=30000)

    # ------------------------------------------------------------------
    # 6️⃣ Final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq,
                                           deadline,
                                           start_time,
                                           L_fft)

    # ------------------------------------------------------------------
    # Normalisation to satisfy evaluate_sequence requirements
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
