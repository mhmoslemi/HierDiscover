subject to a_i ≥ 0.
The objective is scale‑invariant, so we normalise sum(a)=1 and minimise
2·n·max(conv).  The algorithm works in three stages:

1. **Broad parametric sweep** over a few analytic families
   (Beta‑type, sin‑type, exponential).  Each candidate is normalised
   on the simplex and its exact objective (via FFT) is evaluated.
   The best candidate becomes the starting point for local optimisation.

2. **Smoothed Adam optimisation** – a log‑sum‑exp surrogate for the
   max is used.  The temperature β is increased from 5 to a large
   value (≈800) so the surrogate converges to the true max.
   Adam’s adaptive steps keep the iterates on the simplex.
   The best exact objective seen is recorded.

3. **Heavy‑weight local polish** – a mass‑transfer simulated‑annealing
   that repeatedly moves a small fraction of probability mass between
   two bins and accepts the move only if it improves the exact objective.
   After the anneal a short sub‑gradient descent on the true max (using
   the index of the maximal convolution entry) refines the solution.

All stages respect the wall‑time budget (default 1000 s).  The final
sequence is returned as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _smooth_gradient(a: np.ndarray, beta: float, L: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate:
        f_β(a) = (2n/β) * log Σ exp(β·conv(a,a))
    Returns ∇f_β(a) = 4·n·B·a, where B·a is the correlation
    of the softmax weights with the reversed vector a.
    """
    n = a.shape[0]
    # self‑convolution (full length 2n‑1)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    # soft‑max weights
    beta_conv = beta * conv
    max_beta = beta_conv.max()
    exp_shift = np.exp(beta_conv - max_beta)
    w = exp_shift / exp_shift.sum()          # shape (2n‑1,)

    # correlation w * a_rev  (size n)
    a_rev = a[::-1]
    w_fft = np.fft.rfft(w, n=L)
    a_rev_fft = np.fft.rfft(a_rev, n=L)
    corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
    Ba = corr_full[n - 1:n - 1 + n]          # slice length n
    grad = 4.0 * n * Ba
    return grad


def _subgradient_max(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the exact max‑objective 2n·max(conv(a,a)).
    It corresponds to the gradient of the convolution entry that attains the maximum.
    """
    n = a.shape[0]
    # compute full convolution
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    k = int(conv.argmax())        # index of maximal entry
    # gradient of conv[k] w.r.t a
    grad = np.zeros_like(a)
    # part where i <= k  -> a[k-i]
    if k >= 0:
        grad[: k + 1] += a[k::-1]          # a[k], a[k-1], ..., a[0]
    # part where i >= k  -> a[i - k]
    if k < n:
        grad[k:] += a[: n - k]
    # each term appears twice except when i==k-i (i = k/2), but that is handled automatically
    return grad * (2.0 * n)   # factor from objective 2n*conv_max


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step vector with a low evaluation score.
    Returns a plain Python list of floats (sum‑normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness – leave a little slack
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    # 8192 provides a finer discretisation than the previous 4096,
    # while still fitting easily in memory and keeping each FFT cheap.
    n = 8192
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep over a few analytic families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # families: each returns raw (non‑normalised) weights for given (alpha, eps)
    def beta_family(alpha, eps):
        return (x * (1.0 - x) + eps) ** (-alpha)

    def sin_family(alpha, eps):
        # sin(πx) vanishes at the ends → singularities when α>0
        return (np.sin(np.pi * x) + eps) ** (-alpha)

    def exp_family(alpha, eps):
        # smooth bell‑shaped decay in the centre
        return np.exp(-alpha * x * (1.0 - x)) + eps

    families = [
        ("beta", beta_family),
        ("sin", sin_family),
        ("exp", exp_family),
    ]

    alphas = np.linspace(0.4, 0.6, 81)       # step ≈0.0025
    epsilons = [0.0, 1e-12, 1e-8, 1e-6, 1e-4]

    # sweep
    for name, factory in families:
        for alpha in alphas:
            for eps in epsilons:
                w = factory(alpha, eps)
                # ensure non‑negative numeric stability
                w = np.maximum(w, 0.0)
                if w.sum() == 0:
                    continue
                a = w / w.sum()
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                # time check
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Global starter fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)   # uniform distribution

    # ------------------------------------------------------------------
    # 3) Smoothed Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.03

    # FFT length needed for both the convolution and the correlation
    L = _next_pow2(3 * n)   # ≥ 3n‑2, safe for all transforms

    # temperature schedule
    beta_start, beta_end = 5.0, 800.0

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline - 2.0:   # keep a few seconds for polishing
            break
        iteration += 1

        # ---- compute smoothed gradient ----
        elapsed = now - start_time
        prop = min(1.0, elapsed / (budget_s - 0.5))
        beta = beta_start + (beta_end - beta_start) * prop
        grad = _smooth_gradient(a, beta, L)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        # enforce simplex constraints
        a = _project_simplex(a)

        # ---- keep the best exact objective ----
        if iteration % 40 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{elapsed:.2f}s] Adam iter={iteration:,} best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Heavy local annealing (mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_moves = 80_000   # will be limited by remaining time

        for t in range(1, total_moves + 1):
            if time.time() >= deadline:
                break
            # source proportional to current mass (helps move from large piles)
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-16:
                continue
            # move a fraction up to 15% of source mass
            eps_move = a[src] * rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move
            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={t:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] post‑anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final sub‑gradient refinement on the true max
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a = best_seq.copy()
        cur_score = _objective(a)
        sub_it = 0
        # simple diminishing step size
        step0 = 0.02
        while time.time() < deadline - 0.5:
            sub_it += 1
            grad = _subgradient_max(a)
            # step size shrinks as 1/sqrt(iter)
            step = step0 / math.sqrt(sub_it)
            a = a - step * grad
            a = _project_simplex(a)
            new_score = _objective(a)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] subgrad iter={sub_it:,} best={new_score:.6f}")
            # stop early if improvement stalls
            if sub_it >= 2000:
                break

    # ------------------------------------------------------------------
    # 6) Normalisation and final conversion
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
```

This implementation expands on the previous approach by:

* Using a finer discretisation (`n = 8192`) for a more accurate representation of the optimal continuous shape.
* Exploring three analytic families (Beta, sine‑singular, exponential) with a dense grid of parameters.
* Employing an Adam optimiser on a log‑sum‑exp surrogate with a more aggressive temperature schedule (β up to 800).
* Adding an extensive mass‑transfer simulated‑annealing stage (≈80 k moves) followed by a short sub‑gradient descent on the exact max‑objective.

All steps are guarded by the global time budget, and the best sequence found is returned as a plain list of floats.
