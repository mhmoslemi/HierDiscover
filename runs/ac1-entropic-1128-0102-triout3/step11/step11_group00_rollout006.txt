"""
Propose a very low‑score step function.

The objective 2·n·max(conv(a,a))/(sum(a)**2) is minimised when the
sequence is strongly decreasing.  For a shape a_i ∝ (n‑i)^q the
continuous analysis shows that the maximal convolution occurs at the
centre and its value is ≈C(q)/n with

    C(q) = (q+1)²·Beta(q+1,q+1) .

C(q) drops quickly as q grows, while the contribution of the first
entry a₀² grows only like (q+1)²/n.  For a moderate exponent
(q≈5–6) the two contributions are balanced and the resulting bound is
≈ 2·C(q) ≈ 0.02 – far better than the previous 1.5‑level results.

The code simply tries a short list of exponents, builds the decreasing
power‑law vector, normalises it to sum = 1 and evaluates it with the
provided ``evaluate_sequence`` function.  The best vector is returned.
"""

import time
import numpy as np

def _evaluate(arr):
    """Wrapper that calls the global evaluate_sequence safely."""
    # The evaluation function expects a plain list of Python floats.
    return evaluate_sequence(arr.tolist())

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step function (list of floats) with a very small
    evaluation score.

    The routine:
        1. Chooses a problem size n (uses the length of the globally
           supplied ``height_sequence_1`` if present, otherwise 4096).
        2. Scans a small set of exponents q ∈ {0,0.5,1,2,3,4,5,6,7,8},
           builds a_i = (n‑i)ᵠ, normalises to sum = 1 and evaluates.
        3. Returns the sequence that achieved the lowest score.
    """
    # ------------------------------------------------------------------
    # Determine the length of the vector to output.
    # If a global starter is available we reuse its length; otherwise we
    # fall back to the size used in the reference implementation.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        if n <= 1:
            n = 4096
    except Exception:
        n = 4096

    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Exponents to test – a short but sufficiently rich list.
    # ------------------------------------------------------------------
    exponents = [0, 0.5, 1, 2, 3, 4, 5, 6, 7, 8]

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Main search loop (trivial – runs in well under a second).
    # ------------------------------------------------------------------
    for q in exponents:
        if time.time() >= deadline:
            break

        # decreasing power‑law: a_i ∝ (n‑i)ᵠ   (i = 0 … n‑1)
        # we use np.arange with dtype=float to avoid integer overflow.
        raw = np.arange(n, 0, -1, dtype=float) ** q

        # Normalise to L1‑norm 1 (the objective is homogeneous).
        total = raw.sum()
        if total == 0:
            continue
        seq = raw / total

        # Evaluate (the provided function also clamps negatives, etc.).
        score = _evaluate(seq)

        # Keep the best result.
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] exponent q={q} → score {score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen) – uniform distribution.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _evaluate(best_seq)
        print("Fallback to uniform sequence, score",
              f"{best_score:.6f}")

    # Final normalisation safety (already normalised, but we repeat to be sure).
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full(n, 1.0 / n)

    total_time = time.time() - start_time
    print(f"[{total_time:.2f}s] Finished – best score ≈ {best_score:.6f}")

    # Return a plain Python list as required by the evaluator.
    return [float(v) for v in best_seq]
