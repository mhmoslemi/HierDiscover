"""
Propose a low‑score step‐function.

Algorithm
---------
1. **Quick arcsine sweep** – try a dense grid of the family  
      a_i ∝ (x_i·(1−x_i)+ε)^{−α}  (α≈0.30‑0.55, ε≈10^{−12}‑10^{−4}) and keep the best.

2. **Continuous global optimisation** – use SciPy's ``dual_annealing`` to
   fine‑tune the three parameters (α, log10 ε, mixture λ with a uniform
   component).  The objective is the exact score
   ``2·n·max(conv(a,a)) / (sum(a)**2)``.

3. **Projected sub‑gradient descent** – starting from the best vector
   obtained above, run a momentum based sub‑gradient method on the exact
   non‑smooth objective (the same as the original code) for the bulk of
   the remaining time budget.

4. **Light hill‑climbing** – perform many cheap mass‑transfer moves;
   keep a move if it improves the true score.

All steps respect the overall time budget (default 1000 s).  The final
vector is projected onto the probability simplex; scaling does not affect
the score, so any non‑negative list is accepted.
"""

import time
import math
import numpy as np
from scipy.optimize import dual_annealing

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the verifier:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed to be a 1‑D NumPy array.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (non‑smooth objective)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Simple momentum‑based projected sub‑gradient descent on the exact
    objective.  Returns the best iterate seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum coefficient
    step0 = 0.25        # initial step‑size factor
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution & score
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] "
                f"sub‑grad iter={it:,} score={cur_score:.6f}"
            )

        # sub‑gradient of the max‑convolution
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score

        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 10.0:
        sub_deadline = time.time() + 0.85 * remaining
        refined_seq, refined_score = _subgradient_descent(
            best_seq, sub_deadline, start_time
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad refinement score={refined_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 4) Light hill‑climbing (random mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng(seed + 12345)

        # a few hundred thousand cheap moves usually fit into the remaining time
        total_moves = 300_000
        for it in range(1, total_moves + 1):
            if time.time() >= deadline:
                break

            # pick a source index proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            # move a random fraction (up to 15 %) of the source mass
            eps_move = a[src] * rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] hill‑climb iter={it:,} best={new_score:.6f}"
                    )

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb improved score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # The verifier accepts any non‑negative list; scaling does not affect the score.
    return [float(v) for v in best_seq]
