   * move a small fraction of mass from the index with the *largest*
     gradient (most responsible for the peak) to the index with the
     *smallest* gradient, accepting the move only if the objective
     improves.  The step size decays geometrically; the loop stops
     when the wall‑clock budget is almost exhausted.

   This phase is a deterministic analogue of the “hill‑climb” used in
   the reference code, but the direction of the move is computed from
   the exact sub‑gradient instead of being random, which yields a
   tighter descent.

3. **Randomised simulated‑annealing polishing** – after the
   deterministic phase we perform a classic SA walk:
   * pick a source index proportionally to its current mass,
   * pick a destination uniformly,
   * transfer a random fraction (up to 50 %) of the source mass,
   * accept the move with the Metropolis rule `exp(-(Δ‑score)/T)`,
     where the temperature `T` is decreased linearly to a tiny final
     value.
   This stage helps escape shallow local minima and exploits the
   remaining time budget.

4. **Final normalisation** – clamp tiny negatives to zero,
   renormalise to sum = 1 and return the plain Python list.

All heavy work (convolution, gradient) is performed with NumPy’s FFT,
which runs in ≈ 10⁻⁴ s for the required sizes, allowing several‑hundred
thousand iterations inside the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum of a is assumed to be 1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Phase‑1 : deterministic active‑set gradient descent
# ----------------------------------------------------------------------
def _active_gradient_descent(a0: np.ndarray,
                             n: int,
                             L_self: int,
                             L_grad: int,
                             deadline: float,
                             start_time: float,
                             init_best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move mass from the index with the largest sub‑gradient
    contribution to the index with the smallest contribution.
    Only moves that strictly improve the objective are kept.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    # start with a relatively aggressive step; it will shrink automatically
    step_factor = 0.5           # fraction of source mass moved each try
    min_step = 1e-8
    it = 0
    stagnation = 0

    while time.time() < deadline - 5.0:
        it += 1
        # --- 1) current convolution and active set ---
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # if we already beat the best achievable bound, break early
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Active‑GD it={it:,} new best={best_score:.6f}")

        # active indices (exact max – numerical tolerance)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # --- 2) sub‑gradient of max‑convolution (FFT version) ---
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / float(active.size)   # uniform convex combination
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        if src == dst:
            # nothing to do – maybe we are at a stationary point
            break

        # --- 3) propose a mass transfer ---
        eps = a[src] * step_factor
        if eps <= 0.0:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # --- 4) evaluate the proposal ---
        new_score = _score(a_prop, n, L_self)

        if new_score < cur_score - 1e-14:          # accept only if improvement
            a = a_prop
            step_factor = min(0.9, step_factor * 1.02)   # slowly enlarge step when we succeed
            stagnation = 0
        else:
            # reject – shrink step to become more cautious
            step_factor = max(min_step, step_factor * 0.5)
            stagnation += 1
            if stagnation >= 5000:                 # no progress for a while → give up
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : randomised simulated‑annealing polish
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         cur_best_a: np.ndarray,
                         cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Classic SA walk: move a random fraction of mass from a source
    (chosen with probability proportional to its height) to a random
    destination and accept with Metropolis probability.
    """
    a = cur_best_a.copy()
    best_seq = cur_best_a.copy()
    best_score = cur_best_score
    cur_score = cur_best_score

    init_T = 0.015          # initial temperature
    final_T = 1e-7
    rng = np.random.default_rng()

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        progress = (elapsed) / (deadline - start_time)
        T = max(init_T * (1.0 - progress), final_T)

        # propose a transfer
        src = rng.choice(n, p=a)               # weighted by current mass
        dst = rng.integers(0, n)               # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 50 % of source mass
        eps = a[src] * rng.random() * 0.5
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score(a_prop, n, L_self)

        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA best={best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(new_score - cur_score) / T):
                a = a_prop
                cur_score = new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget (≈ 1000 s).
    The method combines a deterministic active‑set gradient descent with
    a final simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin for the external harness
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)          # 8192 – length for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # length needed for active‑set gradient

    # ------------------------------------------------------------------
    # Load starter (global best if present, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adapt length if necessary
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    total = starter.sum()
    if total <= 1e-12:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / total

    # ------------------------------------------------------------------
    # Phase‑1 : deterministic active‑set gradient descent
    # ------------------------------------------------------------------
    cur_score = _score(starter, n, L_self)
    best_seq = starter.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    if time.time() < deadline - 300.0:   # keep a few minutes for polishing
        gd_deadline = deadline - 200.0
        best_seq, best_score = _active_gradient_descent(
            best_seq, n, L_self, L_grad,
            gd_deadline, start_time, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After active‑GD best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : simulated‑annealing polishing (uses remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, n, L_self,
            deadline, start_time,
            best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After SA best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
