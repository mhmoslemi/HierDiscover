    # p is shape (n,)
    # Compute a via softmax
    maxp = np.max(p)  # for numerical stability maybe subtract
    exp_p = np.exp(p - maxp)
    a = exp_p / exp_p.sum()
    # Compute conv via FFT
    conv = _conv_self(a, L=L_fft) # shape length 2n-1
    max_c = conv.max()
    # compute M_tau
    # shift conv by max_c for stability
    exp_vals = np.exp((conv - max_c) / tau)
    sum_exp = exp_vals.sum()
    M = max_c + tau * np.log(sum_exp)  # smoothed max

    # objective = 2*n * M  (since sum a =1)
    obj = 2.0 * n * M

    # gradient of M wrt a:
    w = exp_vals / sum_exp   # shape same as conv.
    # gradient_a = 2 * conv_of_a_and_w_rev

    # compute convolution gradient via FFT
    # Use L_fft length maybe bigger for safety: L_grad = ... predetermined.

    Wf = np.fft.rfft(w_rev, n=L_fft)

    offset = 2 * n - 2

    # da/dp = diag(a) - a a^T
    # So grad_p = a * ( grad_a - np.dot(a, grad_a) )
    dot = np.dot(a, grad_a)
    grad_p = a * (grad_a - dot)

    return obj, grad_p

def _lbfgs_refine(start_a, tau_list, deadline, start_time, L_fft, L_grad):
    """Refine sequence via L‑BFGS‑B on smooth max‑convolution."""
    n = start_a.shape[0]
    # Initial logits
    p = np.log(start_a + 1e-12)

    best_a = start_a.copy()
    best_score = _eval_fast(best_a, L_fft)

    for tau in tau_list:
        if time.time() >= deadline:
            break

        # objective closure
        def func(p_vec):
            # softmax to a
            maxp = np.max(p_vec)
            e = np.exp(p_vec - maxp)
            a = e / e.sum()
            # convolution a*a
            conv = _conv_self(a, L=L_fft)
            max_c = conv.max()
            exp_vals = np.exp((conv - max_c) / tau)
            sum_exp = exp_vals.sum()
            M = max_c + tau * np.log(sum_exp)
            obj = 2.0 * n * M

            # gradient
            w = exp_vals / sum_exp
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=L_grad)
            Wf = np.fft.rfft(w_rev, n=L_grad)
            grad_full = np.fft.irfft(Af * Wf, n=L_grad)[: 2 * n - 1]
            offset = 2 * n - 2
            grad_a = 2.0 * grad_full[offset - np.arange(n)]
            dot = np.dot(a, grad_a)
            grad_p = a * (grad_a - dot)
            return obj, grad_p.astype(np.float64)

        # minimize
        try:
            res = minimize(func, p, method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 40, 'disp': False})
        except Exception as e:
            # fallback: break
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS‑B failed: {e}")
            break

        p = res.x
        # get final a
        maxp = np.max(p)
        e = np.exp(p - maxp)
        a = e / e.sum()
        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LBFGS tau={tau:.4f} score={sc:.6f}")

    return best_a, best_score

All sequences are kept on the probability simplex (non‑negative,
sum = 1) so the scaling in the objective is irrelevant.
The code respects the 1000 s budget, prints progress and finally returns
the best sequence it has found.
"""

import time
import numpy as np

# optional imports – if unavailable the corresponding phase is skipped
try:
    from scipy.optimize import minimize
    _has_scipy_opt = True
except Exception:                     # pragma: no cover
    _has_scipy_opt = False

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple hill‑climb (tiny random moves, accept only improvements)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Very cheap local search: random tiny transfers accepted only if they improve."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# L‑BFGS‑B refinement on the smooth max‑convolution
# ----------------------------------------------------------------------
def _lbfgs_refine(start_a: np.ndarray,
                  tau_list: list[float],
                  deadline: float,
                  start_time: float,
                  L_fft: int,
                  L_grad: int) -> tuple[np.ndarray, float]:
    """
    Refine a sequence via L‑BFGS‑B on a soft‑max approximation of max(conv).
    The temperature τ is decreased according to ``tau_list``.
    """
    if not _has_scipy_opt:                     # pragma: no cover
        return start_a, _eval_fast(start_a, L_fft)

    n = start_a.shape[0]
    # initialise logits for the soft‑max representation
    p = np.log(start_a + 1e-12)

    best_a = start_a.copy()
    best_score = _eval_fast(best_a, L_fft)

    for tau in tau_list:
        if time.time() >= deadline:
            break

        def fun(p_vec):
            """Objective + gradient with respect to the logits p."""
            # soft‑max transform
            maxp = np.max(p_vec)
            e = np.exp(p_vec - maxp)
            a = e / e.sum()

            # convolution a*a
            conv = _conv_self(a, L=L_fft)
            max_c = conv.max()

            # soft‑max of the convolution values
            exp_vals = np.exp((conv - max_c) / tau)
            sum_exp = exp_vals.sum()
            M = max_c + tau * np.log(sum_exp)          # smooth max

            obj = 2.0 * n * M                           # exact objective (sum(a)=1)

            # gradient of M w.r.t. a
            w = exp_vals / sum_exp                     # weights summing to 1
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=L_grad)
            Wf = np.fft.rfft(w_rev, n=L_grad)
            grad_full = np.fft.irfft(Af * Wf, n=L_grad)[: 2 * n - 1]
            offset = 2 * n - 2
            grad_a = 2.0 * grad_full[offset - np.arange(n)]

            # chain rule: ∂obj/∂p = (∂obj/∂a)·∂a/∂p,
            # and ∂a/∂p = diag(a) – a·aᵀ
            dot = np.dot(a, grad_a)
            grad_p = a * (grad_a - dot)

            return obj, grad_p.astype(np.float64)

        # run the optimiser – a rather small number of iterations is enough
        try:
            res = minimize(fun, p, method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 40, 'disp': False})
        except Exception as exc:                    # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS‑B failed: {exc}")
            break

        p = res.x
        # obtain the sequence from the final logits
        maxp = np.max(p)
        e = np.exp(p - maxp)
        a = e / e.sum()
        sc = _eval_fast(a, L_fft)

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LBFGS τ={tau:.4f} score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine mixes parametric sweeps, random Dirichlet sampling,
    L‑BFGS‑B refinement on a smooth max‑convolution and a tiny hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # A moderate size gives a good speed/quality trade‑off
    n = 2048
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # Baseline (uniform)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try to use a global starter if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.size != n:
                # linear interpolation to our working size
                idx_old = np.linspace(0, starter.size - 1, starter.size)
                idx_new = np.linspace(0, starter.size - 1, n)
                starter = np.interp(idx_new, idx_old, starter)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:                               # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric families
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric family (r from 0.1 to 0.999)
    for r in np.linspace(0.1, 0.999, 120):
        if time.time() >= deadline - 120:
            break
        geo = r ** idx
        s = geo.sum()
        if s == 0:
            continue
        seq = geo / s
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo r={r:.3f} score={sc:.6f}")

    # power family (p from -2 to 3)
    for p in np.linspace(-2.0, 3.0, 101):
        if time.time() >= deadline - 120:
            break
        pow_seq = (idx + 1) ** p
        s = pow_seq.sum()
        if s == 0:
            continue
        seq = pow_seq / s
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type family (p,q in [-1,5] step 0.5)
    for p in np.arange(-1.0, 5.1, 0.5):
        for q in np.arange(-1.0, 5.1, 0.5):
            if time.time() >= deadline - 120:
                break
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            s = beta_seq.sum()
            if s == 0:
                continue
            seq = beta_seq / s
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet sampling
    # ------------------------------------------------------------------
    while time.time() < deadline - 80:
        conc = np.random.choice([0.01, 0.05, 0.1, 0.5, 1.0, 2.0])
        a = np.random.dirichlet(np.full(n, conc))
        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet α={conc:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  L‑BFGS‑B refinement (smooth max‑convolution)
    # ------------------------------------------------------------------
    if _has_scipy_opt and time.time() < deadline - 20:
        # gather a few promising seeds (the current best plus a couple of fresh Dirichlet draws)
        seeds = [best_seq]
        for _ in range(3):
            if time.time() >= deadline - 20:
                break
            a = np.random.dirichlet(np.full(n, 0.5))
            seeds.append(a)

        tau_schedule = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
        for seed_seq in seeds:
            if time.time() >= deadline - 5:
                break
            cand_seq, cand_score = _lbfgs_refine(seed_seq, tau_schedule,
                                                deadline, start_time,
                                                L_fft, L_grad)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] LBFGS global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline, start_time, L_fft)

    # ------------------------------------------------------------------
    # Normalisation and final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
