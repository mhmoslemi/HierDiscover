def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose n
    n = 8192

    # parametric sweep over Beta shapes (Alpha, epsilon)
    x = (np.arange(n) + 0.5) / n
    best_seq = None
    best_score = float('inf')
    alphas = np.linspace(0.3, 0.8, 51) # guess
    eps = [0.0, 1e-8, 1e-6, 1e-4]
    for alpha in alphas:
        for e in eps:
            w = (x * (1 - x) + e) ** (-alpha)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        if time.time() >= deadline:
            break

    # consider existing global as starter
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                score = _objective(starter)
                if score < best_score:
                    best_score = score
                    best_seq = starter.copy()
        except Exception:
            pass

    # If no best yet, initialize with uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)

    a = best_seq.copy()

    # Phase 1: subgradient descent on max conv
    # Determine number of iterations based on remaining time
    # Estimate time per iteration ~ some microseconds; we can just loop until deadline.
    iteration = 0
    # Learning rate schedule: start high and decay.
    eta0 = 0.05
    eta_min = 0.0005
    total_iters_target = 5000  # approximate
    while time.time() < deadline - 5.0:  # keep time for final polishing
        iteration += 1
        # compute convolution via FFT
        L = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2 * n - 1]
        k_max = int(conv.argmax())
        # gradient
        g = np.zeros_like(a)
        # compute j = k_max - i for i in range
        # vectorized:
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * a[j[mask]]
        # Step size schedule
        # simple decay: eta = eta0 * (1 - iteration/total_iters_target)
        eta = eta0 * max(0.0, 1.0 - iteration / total_iters_target)
        eta = max(eta, eta_min)
        # gradient step (move opposite to gradient)
        a = a - eta * g
        # project onto simplex
        a = _project_simplex(a)
        # occasional exact evaluation
        if iteration % 100 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:.2f}s] iter={iteration:,}  best={cur_score:.6f}")

    # Phase 2: random mass transfer (simulated annealing) for polishing
    # Use remaining budget
    remaining_time = deadline - time.time()
    if remaining_time > 2.0:
        # start from best_seq
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng(seed + 12345)
        moves = int(remaining_time * 2000)  # estimate moves per second
        for m in range(moves):
            if time.time() >= deadline:
                break
            # select source proportional to mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue
            # transfer fraction up to 20% of src and also may add noise
            eps_transfer = a[src] * rng.random() * 0.2
            a_new = a.copy()
            a_new[src] -= eps_transfer
            a_new[dst] += eps_transfer
            new_score = _objective(a_new)
            if new_score < cur_score or rng.random() < 0.001:  # small chance to accept worse
                a = a_new
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    elapsed = time.time() - start
                    print(f"[{elapsed:.2f}s] anneal step={m:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:.2f}s] final anneal best={best_score:.6f}")

    # Final normalisation (should already be mostly normalized)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Propose a low‑objective step sequence.

The new search differs from the previous baseline in three main ways:

1. **Larger problem size** – we work with n = 8192 (instead of 4096) which
   gives the optimiser more degrees of freedom and a finer discretisation of
   the underlying continuous problem.

2. **Exact sub‑gradient descent** – instead of the smooth log‑sum‑exp
   surrogate we use the true (sub‑)gradient of the nonsmooth objective:
   for the current index k* where the convolution reaches its maximum,
   ∂max c/∂aᵢ = 2·a_{k*–i} (zero when the index is out of range).  A
   simple decaying step‑size together with Euclidean projection onto the
   probability simplex drives the iterates towards a local minimum.

3. **Richer parametric seeding** – before the descent we sweep a few
   families of analytically easy shapes (Beta‑type, symmetric power‑law,
   mixed uniform/power‑law).  The best seed found is handed to the gradient
   phase.  A final light simulated‑annealing polish is kept for escaping
   shallow local minima.

All helpers are top‑level; the algorithm respects the 1000 s wall‑time
budget, prints progress, and finally returns a plain list of floats that
meets the harness’s non‑negative, normalised requirements.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w >= 0 , sum w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT‑based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Core optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) with a very low
    evaluation score under the given time budget.  Returns a plain list of
    floats whose sum is 1 (the harness normalises internally anyway).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ----------- 1) problem size ------------------------------------
    # a power of two works well with the FFTs
    n = 8192

    # ----------- 2) parametric sweep for good seeds -----------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n           # bin centres

    best_seq = None
    best_score = float("inf")

    # ---- a) Beta‑family (already used in the reference code) -----
    alphas = np.linspace(0.30, 1.20, 41)                    # finer range
    epsilons = [0.0, 1e-9, 1e-6, 1e-4]

    for alpha in alphas:
        for eps in epsilons:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        if time.time() >= deadline:
            break

    # ---- b) symmetric power‑law ----------------------------------
    # a_i ∝ (i+1)^{-p} + (n-i)^{-p}
    powers = np.linspace(0.3, 2.0, 35)
    for p in powers:
        i = np.arange(n, dtype=np.float64) + 1.0
        w = i ** (-p) + (n - i + 1) ** (-p)
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        if time.time() >= deadline:
            break

    # ---- c) mixture of uniform and power‑law ---------------------
    for p in [0.5, 0.8, 1.0]:
        mix = 0.5
        i = np.arange(n, dtype=np.float64) + 1.0
        w = mix * np.ones_like(i) + (1 - mix) * (i ** (-p) + (n - i + 1) ** (-p))
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        if time.time() >= deadline:
            break

    # ---- d) possibly reuse a global starter if available ----------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
        except Exception:
            pass

    # fallback – uniform distribution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 3) Exact sub‑gradient descent on max(conv)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    iteration = 0
    # simple linear step‑size schedule
    eta_start = 0.07
    eta_end = 0.0005
    max_iters = 8000                     # generous upper bound

    L = _next_pow2(2 * n - 1)            # FFT length (reuse across iters)

    while time.time() < deadline - 3.0 and iteration < max_iters:
        iteration += 1

        # ----- convolution (FFT) -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        k_max = int(conv.argmax())

        # ----- sub‑gradient of the active entry -----
        # g_i = 2 * a_{k_max - i}  (zero when the index is out of range)
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j_idx[mask]]

        # ----- step‑size (linear decay) -----
        eta = eta_start - (eta_start - eta_end) * (iteration / max_iters)
        eta = max(eta, eta_end)

        # ----- gradient step and projection onto simplex -----
        a = a - eta * g
        a = _project_simplex(a)

        # ----- occasional exact evaluation -----
        if iteration % 100 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:.2f}s] iter={iteration:,}  best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light simulated‑annealing polish (random mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng(seed + 9999)

        # remaining budget roughly determines how many moves we can afford
        remaining = deadline - time.time()
        # very rough estimate: ~2000 moves per second are cheap
        total_moves = int(remaining * 2000)

        for mv in range(total_moves):
            if time.time() >= deadline:
                break
            # source selected proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # move a random fraction (≤15 %) of the source mass
            transfer = a[src] * rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= transfer
            a_prop[dst] += transfer

            new_score = _objective(a_prop)
            # accept if improvement, otherwise accept with tiny probability
            if new_score < cur_score or rng.random() < 0.001:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:.2f}s] anneal step={mv:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
