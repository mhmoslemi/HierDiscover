"""
Propose a step function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a smooth approximation of the max‑convolution and an
exponentiated‑gradient (mirror‑descent) scheme on the probability
simplex.  The algorithm proceeds as follows:

1. Warm‑start from the best global construction (``height_sequence_1``)
   if it exists, otherwise from a symmetric triangular “tent”.
   The warm‑start is up‑sampled to a power‑of‑two length (default 16384)
   to obtain fast FFTs.

2. Repeatedly
   * compute the linear self‑convolution ``c = a*a`` (FFT based);
   * build a soft‑max weighting ``p_k ∝ exp(t·c_k)`` that approximates
     the max‑operator (the temperature ``t`` grows during the run);
   * compute the gradient of the smoothed objective:
        g_i = 2·∑_j p_{i+j}·a_j   (cross‑correlation of ``p`` and ``a``);
   * update the probabilities with an exponentiated‑gradient step
        a ← a·exp(−η·g)   and renormalise to the simplex.

   The learning rate η decays as 1/√iteration, the temperature t is
   increased geometrically every few hundred iterations, and the best
   iterate (exact objective) is tracked.

3. A cheap annealing polish moves a small random packet of mass between
   two coordinates, accepting improvements (or Metropolis moves) to
   escape tiny local minima.

The whole procedure stays well under the 1000 s budget and returns the
best sequence found as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)²).
    The vector a is assumed to be non‑negative; the function works for any
    scaling (the objective is scale‑invariant).  Here we normalise to
    sum(a)=1, therefore the denominator is 1.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _cross_corr(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient:
        g_i = 2·∑_j p_{i+j}·a_j   (0 ≤ i < n).

    It is computed via FFT using the identity
        g = 2·corr(p, a) = 2·ifft(fft(p_rev)·conj(fft(a))).
    """
    n = a.shape[0]
    m = p.shape[0]                     # = 2·n‑1
    L = _next_pow2(n + m - 1)

    # reverse p for correlation
    p_rev = p[::-1]

    Fa = np.fft.rfft(a, n=L)
    Fp = np.fft.rfft(p_rev, n=L)

    corr_full = np.fft.irfft(Fp * np.conj(Fa), n=L)[: n + m - 1]

    # gradient entries: g_i = 2·corr_full[m‑1‑i]
    g = 2.0 * corr_full[m - 1 - np.arange(n)]
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1) initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    # try to use the globally supplied best construction
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        if init.sum() > 0:
            init /= init.sum()
        else:
            raise ValueError
        print(f"[0.00s] Using global starter (len={init.size})")
    except Exception:
        # fallback: symmetric triangular tent
        n0 = 4096
        x = np.arange(n0, dtype=float) / (n0 - 1)
        tent = np.minimum(x, 1.0 - x)
        if tent.sum() == 0:
            init = np.full(n0, 1.0 / n0)
        else:
            init = tent / tent.sum()
        print(f"[0.00s] Using triangular warm‑start (len={n0})")

    # target length – a power of two for fast FFTs
    target_n = 16384
    if init.size != target_n:
        # linear interpolation to the new grid
        old_x = np.linspace(0.0, 1.0, init.size)
        new_x = np.linspace(0.0, 1.0, target_n)
        init = np.interp(new_x, old_x, init)
        init = np.maximum(init, 0.0)
        if init.sum() > 0:
            init /= init.sum()
        else:
            init = np.full(target_n, 1.0 / target_n)

    a = init.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) exponential‑gradient descent on a smoothed max ---------------
    # ------------------------------------------------------------------
    it = 0
    step0 = 0.5                      # initial learning‑rate factor
    eta0 = 0.5                       # same scale as step0
    t = 10.0                         # initial temperature (smoothness)
    t_max = 5000.0
    t_inc_every = 2000              # increase t every this many iters
    t_factor = 1.4

    while time.time() < deadline:
        it += 1

        # self‑convolution of the current iterate
        conv = _conv_self(a)                     # length 2·n‑1
        cur_score = 2.0 * n * conv.max()         # exact objective (sum(a)=1)

        # keep the best solution encountered
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} "
                  f"new best = {cur_score:.6f}")

        # ----- smoothed max (soft‑max) weights -----------------------
        conv_max = conv.max()
        # stable exponentiation
        exp_arg = t * (conv - conv_max)
        w = np.exp(exp_arg)
        p = w / w.sum()          # probability vector of length 2·n‑1

        # ----- gradient of the smooth objective ----------------------
        g = _cross_corr(p, a)    # shape (n,)

        # ----- exponentiated‑gradient step ---------------------------
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * g)

        # renormalise onto the probability simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ----- schedule for the temperature -------------------------
        if it % t_inc_every == 0 and t < t_max:
            t = min(t * t_factor, t_max)
            print(f"[{time.time() - start_time:.2f}s] increase temp t={t:.1f}")

        # occasional stop‑check
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3) Light annealing polish ---------------------------------------
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    anneal_steps = 20000
    for i in range(anneal_steps):
        if time.time() >= deadline:
            break

        # choose source index weighted by current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move a modest fraction of the source mass
        eps = a[src] * rng.random() * 0.15
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={i:,} "
                      f"improved = {new_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final normalisation & return ---------------------------------
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    tot = best_a.sum()
    if tot > 0:
        best_a /= tot
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    print(f"[{time.time() - start_time:.2f}s] final best score = {best_score:.6f}")
    return [float(v) for v in best_a]
