def _softmax_gradient_step(a, beta, step_size):
    conv = _conv_self(a)
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    weights = exp_terms / exp_terms.sum()
    # compute grad via FFT
    w_rev = weights[::-1]
    L = _next_pow2(len(w_rev) + len(a) - 1)
    Aw = np.fft.rfft(w_rev, n=L)
    Aa = np.fft.rfft(a, n=L)
    grad_full = np.fft.irfft(Aw * Aa, n=L)[:len(w_rev) + len(a) - 1]
    grad = 2.0 * grad_full[2*len(a)-2 - np.arange(len(a))]
    # update a
    a_new = a - step_size * grad
    # project onto simplex
    a_proj = _project_simplex(a_new)
    return a_proj, conv.max()

"""
Propose a low‑score step‑function.

Algorithm
---------
1. Choose a moderate length (n≈8000) – a finer discretisation
   reduces the discretisation error of the continuous problem.

2. Sweep a rich family of “U‑shaped” densities:
   a_i ∝ (x·(1−x)+ε)^{−α} (β‑distribution with equal parameters).
   α is explored on a fine grid and also by random sampling,
   ε varies over several decades.  A few simple mixtures
   with the uniform and triangular shapes are also examined.
   The best candidate is retained.

3. Refine the best candidate with a **soft‑max gradient descent**.
   Instead of the non‑smooth max we minimise a smooth approximation
   𝑓_β(a)=2·n·log‑sum‑exp(β·conv(a,a))/(β·‖a‖₁²) .
   The gradient of the soft‑max is computed by a second FFT‑based
   convolution, and the iterate is projected onto the probability
   simplex (Euclidean projection).  β is increased slowly from 5
   to ≈200 so that the objective approaches the true max.
   An Adam‑style adaptive step‑size is used for fast, stable
   convergence.

4. If time permits, a short simulated‑annealing polish is performed,
   moving small random fractions of mass between bins.

The whole search respects the 1000 s wall‑time limit and always returns
a non‑negative list (scaling does not influence the score)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Convolution of a and b using FFT of length L (full length)."""
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    C = np.fft.irfft(Af * Bf, n=L)
    return C[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
# Objective (exact, as used by the verifier)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Soft‑max gradient of the smoothed objective
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, beta: float,
                      L_self: int, L_grad: int) -> tuple[np.ndarray, float]:
    """
    Compute gradient of the smoothed max (soft‑max) of conv(a,a) and the
    current (non‑smoothed) max value.
    Returns (grad, max_conv).
    """
    # 1) self‑convolution (used for max)
    conv = _conv_self(a, L_self)
    max_conv = conv.max()

    # 2) soft‑max weights (stable)
    shifted = conv - max_conv
    exp_terms = np.exp(beta * shifted)
    w = exp_terms / exp_terms.sum()                     # shape 2·n‑1

    # 3) gradient = 2 * cross‑correlation of w with a
    w_rev = w[::-1]
    grad_full = _conv_fft(w_rev, a, L_grad)            # length 3·n‑2
    n = a.shape[0]
    # extract the needed slice: grad[i] = 2 * grad_full[2·n‑2‑i]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_conv

        beta2 = 0.999
        eps_adam = 1e-8
        # smoothing parameter schedule
        beta = 5.0
        beta_max = 200.0
        beta_factor = 1.04
        step0 = 0.20
        it = 0
        while time.time() < deadline - 5.0:
            it += 1
            grad, cur_max = _softmax_gradient(a, beta, L_self, L_grad)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = step0 / (np.sqrt(v_hat) + eps_adam)

            a = a - step * m_hat
            a = _project_simplex(a)

            # exact objective for monitoring
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] grad descent it={it} β={beta:.1f} score={cur_score:.6f}")

            # increase the soft‑max sharpness slowly
            if beta < beta_max:
                beta = min(beta_max, beta * beta_factor)

            # occasional break check
            if it % 500 == 0 and (time.time() + 2.0) > deadline:
                break

    # ------------------------------------------------------------------
    # 7) Final simulated‑annealing polish (if time remains)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng(seed + 5555)
        total_steps = 30000
        for it in range(1, total_steps + 1):
            if time.time() >= deadline:
                break
            # linear cooling schedule
            T = 0.02 * (0.001 / 0.02) ** (it / total_steps)

            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps_move = a[src] * rng.random() * 0.20
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            new_score = _objective(a_prop)
            delta = new_score - cur_score
            if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal it={it} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8) Normalise and return as plain Python list
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        else:
            best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
