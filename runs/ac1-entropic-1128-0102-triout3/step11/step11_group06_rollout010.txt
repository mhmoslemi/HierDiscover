def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8192

    # Predefine target uniform conv value
    target_val = 1.0 / (2 * n - 1)

    # Define FFT length L = next_pow2(3*n)
    L = _next_pow2(3 * n)

    # 1) initial Beta sweep
    alphas = np.linspace(0.48, 0.52, 9)
    epsilons = [1e-8]

    best_seq = None
    best_score = float('inf')
    for α in alphas:
        for eps in epsilons:
            if time.time() >= deadline:
                break
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            w = (x * (1 - x) + eps) ** (-α)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        if time.time() >= deadline:
            break

    # If nothing, fallback uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Ensure normalized sum = 1
    best_seq = best_seq / best_seq.sum()

    # 2) Flattening stage (L2 to uniform)
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.05   # may need adjust
    # get initial error
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    init_err = np.mean((conv - target_val) ** 2)
    err0 = init_err if init_err > 0 else 1.0
    iteration = 0

    while time.time() < deadline - 10.0:  # leave 10 seconds for later stage
        iteration += 1
        # compute conv
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

        corr_full = np.fft.irfft(d_fft * a_rev_fft, n=L)

            if score < best_score:

                best_seq = a.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:.2f}s] flatten iter={iteration} obj={score:.6f}")

        # maybe break early if error tiny or no improvement?

"""
Propose a step‑function with a very low evaluation score.

The algorithm works in three phases on a reasonably large vector (default n=8192).

1️⃣  **Good initial guess** – sweep a Beta‑family  a_i ∝ (x·(1−x)+ε)^{−α}
    with α≈0.5 (the “arcsine” shape) on the probability simplex.

2️⃣  **Flattening stage** – minimise the L² distance between the
    self‑convolution a∗a and the perfectly flat target
    u_k = 1/(2 n‑1).  The gradient of ½‖a∗a−u‖² is
    4·corr(d, a) where d = (a∗a−u) and corr denotes a
    (linear) cross‑correlation.  An Adam optimiser with a
    decreasing learning‑rate (scaled by the current error) drives the
    distribution towards a nearly uniform sum‑distribution.

3️⃣  **Max‑smoothing stage** – use the same Adam scheme but now on the
    smooth surrogate of the true objective:
        F_β(a)=2 n·(1/β)·log Σ_k exp(β·(a∗a)_k)
    with a linearly increasing β (up to 500).  This polishes the
    solution with respect to the original max‑convolution objective.

A very light simulated‑annealing polish is performed if time permits.
All vectors are projected onto the probability simplex after every
update.  The whole routine respects the 1000 s wall‑time limit and
returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) with a very low
    evaluation score under the given time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 8192                         # length of the candidate
    L = _next_pow2(3 * n)           # FFT length used for all convolutions
    target_val = 1.0 / (2 * n - 1)   # uniform target for a*a

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Good initial guess – sweep a Beta‑family
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    alphas = np.linspace(0.48, 0.52, 9)   # around the arcsine exponent
    epsilons = [0.0, 1e-8]

    for alpha in alphas:
        for eps in epsilons:
            if time.time() >= deadline:
                break
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Use a global starter if the harness supplied one
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing else was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Helper to keep the current best solution
    # ------------------------------------------------------------------
    def _maybe_update_best(cur_a):
        nonlocal best_seq, best_score
        sc = _objective(cur_a)
        if sc < best_score:
            best_score = sc
            best_seq = cur_a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] new best objective = {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Flattening stage – minimise ‖a∗a – uniform‖²
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.05

    # compute initial error for a simple schedule
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    init_err = np.mean((conv - target_val) ** 2)
    err0 = max(init_err, 1e-12)

    it = 0
    # reserve ~20 s for the final polishing stage
    while time.time() < deadline - 20.0:
        it += 1

        # a∗a
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # error vector
        d = conv - target_val

        # gradient = 4 * corr(d, a)
        d_fft = np.fft.rfft(d, n=L)
        a_rev_fft = np.fft.rfft(a[::-1], n=L)
        corr_full = np.fft.irfft(d_fft * a_rev_fft, n=L)
        grad = 4.0 * corr_full[n - 1 : n - 1 + n]

        # learning‑rate tuned by current error
        err = np.mean(d * d)
        lr = base_lr * np.sqrt(err / err0 + 1e-12)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)

        # occasional exact evaluation
        if it % 200 == 0:
            _maybe_update_best(a)

    # ------------------------------------------------------------------
    # 4) Max‑smoothing stage – Adam on a smooth approximation of max
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.02
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    beta_start, beta_end = 5.0, 500.0
    it2 = 0

    while time.time() < deadline - 2.0:
        it2 += 1

        # self‑convolution
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # smoothing parameter (slowly increase)
        elapsed = time.time() - start_time
        prop = min(1.0, elapsed / (budget_s - 0.5))
        beta = beta_start + (beta_end - beta_start) * prop

        # soft‑max weights over the convolution entries
        beta_conv = beta * conv
        max_beta = beta_conv.max()
        exp_shift = np.exp(beta_conv - max_beta)
        w = exp_shift / exp_shift.sum()        # shape (2n‑1,)

        # gradient of the smooth max: 4·n·corr(w, a)
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a[::-1], n=L)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        Ba = corr_full[n - 1 : n - 1 + n]
        gradient = 4.0 * n * Ba

        # Adam step
        m = beta1 * m + (1.0 - beta1) * gradient
        v = beta2 * v + (1.0 - beta2) * (gradient ** 2)
        m_hat = m / (1.0 - beta1 ** it2)
        v_hat = v / (1.0 - beta2 ** it2)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)

        if it2 % 50 == 0:
            _maybe_update_best(a)

    # ------------------------------------------------------------------
    # 5) Very light simulated‑annealing polish (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_moves = 20000

        for t in range(1, total_moves + 1):
            if time.time() >= deadline:
                break
            src = rng.choice(n, p=a)          # source proportional to mass
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps = a[src] * rng.random() * 0.10   # move up to 10 % of src
            a_proposed = a.copy()
            a_proposed[src] -= eps
            a_proposed[dst] += eps

            new_score = _objective(a_proposed)
            if new_score < cur_score:
                a = a_proposed
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:.2f}s] anneal step={t:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_seq = best_local.copy()
            best_score = best_local_score
            print(f"[{time.time() - start_time:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
