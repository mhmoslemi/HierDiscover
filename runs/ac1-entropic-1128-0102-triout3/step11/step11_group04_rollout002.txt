"""
Hybrid optimizer for the step‑function minimisation problem.

The objective is   

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

with n = 4096 and a_i ≥ 0.  Since the ratio is scale invariant we keep
the vector normalised to sum = 1 during the search.

Algorithm
---------
1. **Initialisation** – load the best known sequence (global variable
   `height_sequence_1`) or start from the uniform distribution.

2. **Smooth‑max mirror descent** – exponentiated‑gradient updates on a
   smooth approximation of the max‑convolution (the same scheme used in
   the reference solution, but with a slightly larger smoothing range).

3. **Projected sub‑gradient descent** – classic sub‑gradient method on the
   exact objective with a 1/√t step‑size and Euclidean projection onto the
   probability simplex.

4. **Simulated annealing** – a mass‑transfer Monte‑Carlo search that
   accepts worsening moves with a temperature schedule.  This stage can
   escape the local minima that the monotone hill‑climb gets stuck in.

All phases respect the wall‑clock budget (default 1000 s).  The final
vector is re‑normalised to sum = 1 and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (full length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum of a is assumed to be 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective assuming a ≥ 0 and sum(a)=1."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b)  # denominator = 1²


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    L_self: int,
                    L_grad: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror descent on a smooth approximation of the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    # smoothing schedule – a bit wider than in the reference code
    alphas = np.geomspace(0.5, 2000.0, num=45)

    m = np.zeros_like(a)          # momentum buffer in log‑domain
    beta = 0.9
    it = 0

    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # self‑convolution (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # smoothed max via soft‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)       # stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # gradient of the smooth max (convolution with reversed a)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # momentum + exponentiated‑gradient update (mirror descent)
            m = beta * m + (1.0 - beta) * grad
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * m)
            a = np.maximum(a, 1e-30)   # keep strictly positive
            a /= a.sum()

            # occasional exact check
            if it % 1000 == 0:
                cur = _eval_fast(a, L_self)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.1f} it={it:,} score={cur:.6f}")

        if time.time() >= deadline - 0.2:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (exact objective)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # sum(a)=1

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        if stagnation >= 20_000:
            break

        # active indices (those achieving the maximum)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         cur_score: float) -> tuple[np.ndarray, float]:
    """
    Simple SA: transfer a random fraction of mass from a source index to a
    destination index.  Moves that worsen the objective are occasionally
    accepted depending on a temperature that decays to near zero.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    cur_score = cur_score
    rng = np.random.default_rng()

    # initial temperature – a small fraction of the current excess over 1.0
    T0 = max(1e-4, (cur_score - 1.0) * 0.5)
    T_min = 1e-7
    # exponential decay factor (chosen so that after ~2·10⁵ iters temperature ≈ T_min)
    decay = 0.99995

    it = 0
    while time.time() < deadline:
        it += 1
        # geometric temperature decay
        T = max(T_min, T0 * (decay ** it))

        # choose a source weighted by current mass (helps move mass from large entries)
        src = rng.choice(n, p=a)
        dst = rng.integers(0, n)
        if src == dst:
            continue

        # fraction of source mass to move – larger at high temperature
        max_frac = min(0.5, T * 20.0)          # at start max ≈ 0.5, later tiny
        move_frac = rng.random() * max_frac
        eps = a[src] * move_frac
        if eps < 1e-15:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_self)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            # avoid overflow in exp
            if delta / T < 700.0:
                if rng.random() < math.exp(-delta / T):
                    accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")

    return best_a, best_score

    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
