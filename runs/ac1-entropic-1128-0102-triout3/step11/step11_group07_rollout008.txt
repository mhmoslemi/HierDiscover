    """smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """self‑convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numeric fallback
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast surrogate for ``evaluate_sequence`` – assumes sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweeps (geometric, power, beta) + fine geometric ref.
# ----------------------------------------------------------------------
def _sweep_geometric(best_seq, best_score, n, L_fft, start_time, deadline):
    idx = np.arange(n, dtype=np.float64)

    # coarse grid
    eps_grid = np.logspace(-6, -1, 140)          # r = 1-eps
    lam_grid = np.linspace(0.0, 1.0, 21)        # convex blend with uniform

    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        s = geo.sum()
        if s == 0.0:
            continue
        geo /= s
        for lam in lam_grid:
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            cur = _eval_fast(seq, L_fft)
            if cur < best_score:
                best_score, best_seq = cur, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} → {cur:.6f}")

    # -- local refinement around the best geometric parameters ----------------
    # retrieve the parameters that produced the current best_seq (approx)
    # we keep a simple hill‑climb over (eps, lam) space.
    if best_seq is not None:
        # infer approximate eps, lam by fitting the current best_seq to the form
        # However for simplicity we just start from the best known eps,lam.
        # Use the last eps,lam that gave improvement as seeds.
        # If not available we just start from a modest guess.
        eps0 = 0.01
        lam0 = 0.5
        for _ in range(300):
            if time.time() >= deadline - 20:
                break
            # propose a small random move in log‑eps and lam
            log_eps = math.log(eps0) + np.random.randn() * 0.15
            eps1 = math.exp(log_eps)
            lam1 = np.clip(lam0 + np.random.randn() * 0.07, 0.0, 1.0)
            r = 1.0 - eps1
            geo = np.power(r, idx, dtype=np.float64)
            geo /= geo.sum()
            seq = lam1 * np.full(n, 1.0 / n) + (1.0 - lam1) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            cur = _eval_fast(seq, L_fft)
            if cur < best_score:
                best_score, best_seq = cur, seq.copy()
                eps0, lam0 = eps1, lam1
                print(f"[{time.time() - start_time:.2f}s] Geo‑refine eps={eps0:.1e} lam={lam0:.2f} → {cur:.6f}")

    return best_seq, best_score


def _sweep_power(best_seq, best_score, n, L_fft, start_time, deadline):
    idx = np.arange(n, dtype=np.float64)
    p_grid = np.linspace(-4.0, 3.0, 141)   # finer than previous
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        seq = np.power(idx + 1.0, p, dtype=np.float64)
        s = seq.sum()
        if s == 0.0:
            continue
        seq /= s
        cur = _eval_fast(seq, L_fft)
        if cur < best_score:
            best_score, best_seq = cur, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} → {cur:.6f}")
    return best_seq, best_score


def _sweep_beta(best_seq, best_score, n, L_fft, start_time, deadline):
    idx = np.arange(n, dtype=np.float64)
    param_grid = np.linspace(0.0, 5.0, 11)          # more possibilities
    for p in param_grid:
        for q in param_grid:
            if time.time() >= deadline - 30:
                break
            seq = np.power(idx + 1.0, p) * np.power(n - idx, q)
            s = seq.sum()
            if s == 0.0:
                continue
            seq /= s
            cur = _eval_fast(seq, L_fft)
            if cur < best_score:
                best_score, best_seq = cur, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} → {cur:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Gaussian‑mixture Differential Evolution (low‑dimensional)
# ----------------------------------------------------------------------
def _gaussian_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """Build a non‑negative sequence of length n from a mixture of Gaussians."""
    seq = np.zeros(n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    K = len(params) // 3
    for i in range(K):
        amp = max(0.0, params[3 * i])          # non‑negative
        cen = params[3 * i + 1]
        wid = max(1e-3, params[3 * i + 2])
        seq += amp * np.exp(-((idx - cen) ** 2) / (2.0 * wid * wid))
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0.0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)
    return seq


def _de_optimise(n: int, L_fft: int, deadline: float,
                 start_time: float, best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """Run differential evolution on the Gaussian‑mixture parametrisation."""
    if not _has_scipy:                     # pragma: no cover
        return best_seq, best_score

    K = 8                                   # a few more components
    dim = 3 * K
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 10.0))          # amplitude
        bounds.append((0.0, n - 1))         # centre
        bounds.append((1.0, n / 2.0))       # width

    def obj(params):
        seq = _gaussian_sequence(params, n)
        return _eval_fast(seq, L_fft)

    remaining = deadline - time.time()
    if remaining < 5.0:
        return best_seq, best_score
    maxiter = max(1, int(remaining // 8))    # ~8 s per generation
    maxiter = min(maxiter, 70)               # keep it reasonable

    print(f"[{time.time() - start_time:.2f}s] DE start (maxiter={maxiter})")
    result = differential_evolution(obj,
                                    bounds,
                                    maxiter=maxiter,
                                    popsize=15,
                                    tol=1e-7,
                                    polishing=False,
                                    seed=0,
                                    workers=1,
                                    updating='deferred')
    seq_de = _gaussian_sequence(result.x, n)
    score_de = _eval_fast(seq_de, L_fft)
    if score_de < best_score:
        best_score, best_seq = score_de, seq_de.copy()
        print(f"[{time.time() - start_time:.2f}s] DE improved → {score_de:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam‑style smoothed gradient descent (soft‑max max)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max smoothed objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    tau = 0.08                     # start a bit warmer than before
    tau_min = 1e-9
    tau_decay = 0.9997
    step0 = 0.18

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and soft‑max weights
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient of soft‑max approx (via convolution of a with reversed w)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        cur = _eval_fast(a, L_fft)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} → {cur:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Active‑set Linear‑Program refinement (more iterations)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 20,
               tol_frac: float = 0.015) -> tuple[np.ndarray, float]:
    """Iteratively improve `a` by solving LPs built from the active set."""
    if not _has_scipy:                     # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])
        K = active.size

        # Build the constraint matrix A_ub (K × n)
        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1, dtype=int)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)

        # maximise sum(g)  ⇔ minimise −sum(g)
        c = -np.ones(n, dtype=np.float64)

        # time limit for each LP ≈ remaining time / (remaining iters)
        lp_time = max(1, int((deadline - time.time()) / (max_iters - it)))
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': lp_time})
        if not res.success:                     # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP it={it} failure: {res.message}")
            break

        g = res.x
        if g.sum() == 0.0:
            break
        g = g / g.sum()                         # normalise to sum‑1

        # convex combination ‑‑‑ decrease step if no improvement
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)

        if new_score < best_score:
            best_score, a = new_score, a_new
            print(f"[{time.time() - start_time:.2f}s] LP it={it} → {new_score:.6f}")
        else:
            # early stop if stagnating
            print(f"[{time.time() - start_time:.2f}s] LP it={it} no improvement")
            break

    return a, best_score


# ----------------------------------------------------------------------
# 5️⃣  Simulated‑Annealing polishing (accept uphill moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """Tiny SA: random mass transfers, occasional uphill acceptance."""
    n = a0.shape[0]
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)
    best_seq, best_score = a.copy(), cur_score

    rng = np.random.default_rng()
    it = 0
    T0 = 2e-4                     # start temperature (roughly 1 % of score)
    while time.time() < deadline:
        it += 1
        # pick source proportionally to its mass – biases toward larger entries
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction of src's mass
        eps = a[src] * rng.random() * 0.12
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # no projection needed – sum conserved, non‑negativity guaranteed
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score

        # temperature schedule – exponential cooling
        T = T0 * (0.9996 ** it)
        accept = (delta < 0) or (rng.random() < math.exp(-delta / max(T, 1e-15)))
        if accept:
            a, cur_score = a_prop, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The workflow:
        * start from uniform / a possible global starter,
        * extensive parametric sweeping (geometric, power, beta) + fine refinement,
        * low‑dimensional Gaussian‑mixture DE,
        * long Adam‑style smoothed descent (several restarts),
        * aggressive active‑set LP refinement,
        * final simulated‑annealing polish.
    The best sequence (normalised to sum = 1) is returned as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # baseline: uniform
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # optional global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_seq, best_score = starter.copy(), cur
                print(f"[{time.time() - start_time:.2f}s] Global starter → {cur:.6f}")
    except Exception:                     # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Parametric sweeps + local geometric refinement
    best_seq, best_score = _sweep_geometric(best_seq, best_score,
                                            n, L_fft, start_time, deadline)
    best_seq, best_score = _sweep_power(best_seq, best_score,
                                        n, L_fft, start_time, deadline)
    best_seq, best_score = _sweep_beta(best_seq, best_score,
                                       n, L_fft, start_time, deadline)

    # ------------------------------------------------------------------
    # 2️⃣ Gaussian‑mixture DE (low‑dimensional)
    if time.time() < deadline - 30.0 and best_score > 1.503:
        best_seq, best_score = _de_optimise(n, L_fft, deadline,
                                            start_time, best_seq, best_score)

    # ------------------------------------------------------------------
    # 3️⃣ Adam‑style smoothed gradient descent – several restarts
    while time.time() < deadline - 40.0 and best_score > 1.503:
        # perturb current best (or uniform) moderately
        base = best_seq if np.random.rand() < 0.8 else uniform
        noise = 0.05 * np.random.randn(n) * base   # 5 % Gaussian noise
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 18.0)
        cand_seq, cand_score = _adam_smooth_descent(start, sub_deadline,
                                                    start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_seq, best_score = cand_seq.copy(), cand_score
            print(f"[{time.time() - start_time:.2f}s] Adam new global best → {best_score:.6f}")

        # optional early exit
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4️⃣ Active‑set LP refinement (more aggressive)
    if time.time() < deadline - 20.0 and best_score > 1.503:
        best_seq, best_score = _lp_refine(best_seq, deadline,
                                          start_time, L_fft,
                                          max_iters=20, tol_frac=0.012)

    # ------------------------------------------------------------------
    # 5️⃣ Simulated‑Annealing polish (accept uphill moves)
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    deadline,
                                                    start_time,
                                                    L_fft)

    # ------------------------------------------------------------------
    # final normalisation & output
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
