"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a blend of:
* quick parametric families (geometric, power, beta, symmetric‑geometric)
* many random Dirichlet starts
* long Adam‑style smoothed gradient descent with a log‑sum‑exp surrogate
* a final simulated‑annealing polish that can accept uphill moves

The implementation is deliberately lightweight (no SciPy / LP) and works
under the 1000 s budget.  It returns a list of floats compatible with
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Small linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return the exact evaluation of the objective for a (sum≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑style descent on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         sub_deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector found during the run and its exact score.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # temperature for the soft‑max (log‑sum‑exp)
    tau = 0.05
    tau_min = 1e-9
    tau_decay = 0.9997      # slower decay → smoother gradients for longer
    step0 = 0.12            # base step size

    while time.time() < sub_deadline:
        it += 1

        # ---- smooth max -------------------------------------------------
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        # numerically stable soft‑max (log‑sum‑exp) weights
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # ---- gradient of the surrogate ----------------------------------
        # grad_j = 2 * Σ_i a_i * w_{i+j}
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        conv_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * conv_full[offset - np.arange(n)]

        # ---- Adam update + projection ------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # ---- exact score -------------------------------------------------
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"τ={tau:.2e} score={cur_score:.6f}")

        # ---- temperature decay -------------------------------------------
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Tiny annealer that moves a random fraction of mass from one index to another.
    Accepts uphill moves with a decreasing temperature.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    best_score = cur_score

    T = 0.05               # initial temperature
    Tmin = 1e-7
    it = 0

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # choose source proportional to its mass (more likely to move where there is mass)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst:
            continue
        max_move = a[src] * 0.2   # move at most 20 % of the source mass
        if max_move <= 1e-14:
            continue
        delta = rng.random() * max_move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta   # sum is preserved, non‑negativity guaranteed

        new_score = _eval_fast(a_prop, L_fft)
        delta_score = new_score - cur_score

        if delta_score < 0 or np.exp(-delta_score / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} best={best_score:.6f}")

        # geometrically cool the temperature
        T = max(Tmin, T * 0.99995)

    return best_a, best_score

                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Very fast parametric families
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric families (including symmetric two‑sided)
    for eps in np.logspace(-4, -1, 90):
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        # plain geometric
        for lam in (0.0, 0.3, 0.6, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.1f} score={sc:.6f}")
        # symmetric (both ends)
        sym_geo = geo + geo[::-1]
        sym_geo /= sym_geo.sum()
        for lam in (0.0, 0.5, 1.0):
            seq = lam * uniform + (1.0 - lam) * sym_geo
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SymGeo eps={eps:.1e} lam={lam:.1f} score={sc:.6f}")

    # power families
    for p in np.linspace(-1.0, 2.0, 45):
        if time.time() >= deadline - 30:
            break
        pow_seq = (idx + 1) ** p
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type families
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet starts – cheap way to inject diversity
    # ------------------------------------------------------------------
    while time.time() < deadline - 120:            # keep a few seconds for the final SA
        alpha = np.random.choice([0.2, 0.5, 1.0, 2.0])
        cand = np.random.default_rng().gamma(alpha, 1.0, size=n)
        cand = _project_simplex(cand)
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet α={alpha:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam‑style smoothed gradient descent with many restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # initialise from the current best (with a small perturbation) or uniform
        if np.random.rand() < 0.8:
            base = best_seq
        else:
            base = uniform
        noise = 0.03 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 28.0)   # ≈28 s per run
        cand_seq, cand_score = _adam_smooth_descent(start,
                                                    sub_deadline,
                                                    start_time,
                                                    L_fft,
                                                    L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new global best score={best_score:.6f}")

        # early exit if we already beat the (soft) target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing polish (tiny uphill moves allowed)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    start_time,
                                                    deadline,
                                                    L_fft)
        print(f"[{time.time() - start_time:.2f}s] After SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against accidental zero‑sum) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
