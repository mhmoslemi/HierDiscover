        elapsed = time.time() - start_time
        total_budget = deadline - start_time
        progress = elapsed / total_budget
        temp = max(t_final, t0 * (1.0 - progress)) # linear cooling
        # choose move

        src = np.random.choice(n, p=a)

        if eps <= 0: continue
        a_new = a.copy()
        a_new[src] -= eps

        # keep sum same
        # compute new score

"""
Propose a step‑function (non‑negative simplex vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a multi‑stage strategy:

1.  Quick parametric sweep (geometric, mixed‑geometric, symmetric,
    cosine, Gaussian) to obtain a decent seed.
2.  Several restarts of an Adam‑style smoothed gradient descent
    (soft‑max approximation of the max) with a very slow temperature
    decay – this explores the neighbourhood of the seed more thoroughly.
3.  A global Simulated‑Annealing phase that moves mass between coordinates
    and accepts occasional uphill moves, allowing escape from local minima.
4.  Optional active‑set linear‑program refinement (if SciPy is available).
5.  Final tiny random hill‑climb for polishing.

All vectors are projected onto the probability simplex after every
update, guaranteeing non‑negativity and a unit sum (the objective is
scale‑invariant).  The routine respects the 1000 s budget and prints
progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum of entries is assumed ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return the objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self(a, L=L_fft)
    maxb = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * a.shape[0] * maxb / (s * s))


# ----------------------------------------------------------------------
# Adam‑style smoothed descent (soft‑max approximation)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         sub_deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max smoothed objective with a very slow temperature decay."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    # start temperature relatively high; final temperature essentially 0
    tau = 0.2
    tau_min = 1e-8
    tau_decay = 0.9999  # slower decay than original code
    step0 = 0.13

    while time.time() < sub_deadline:
        it += 1

        # convolution and soft‑max weights
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()               # w_i ≈ soft‑max over conv

        # gradient of the smoothed max w.r.t. a:
        # ∂/∂a_j Σ_i w_i * (a * a)[i] = 2 * Σ_i w_i * a_{i-j}
        # This is a convolution of a with reversed w.
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update + projection onto simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # exact score
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑Annealing global search
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA on the simplex. Accepts uphill moves with temperature."""
    n = a0.shape[0]
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)
    best_seq = a.copy()
    best_score = cur_score

    # SA schedule
    t0 = 0.08               # start temperature (controls step size)
    t_final = 1e-6
    max_step_frac = 0.5     # max proportion of source mass to move at hot start

    while time.time() < deadline:
        # linear cooling based on elapsed proportion
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        temp = max(t_final, t0 * (1.0 - progress))

        # pick a source (probability proportional to current mass)
        src = np.random.choice(n, p=a)
        dst = np.random.randint(0, n)
        if src == dst or a[src] <= 1e-14:
            continue

        # step fraction shrinks with temperature
        step_frac = max_step_frac * (temp / t0)
        eps = a[src] * step_frac * np.random.rand()
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        new_score = _eval_fast(a_new, L_fft)
        delta = new_score - cur_score

        if delta <= 0 or np.exp(-delta / max(temp, 1e-12)) > np.random.rand():
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")

        # occasional random perturbation to keep diversity
        if np.random.rand() < 0.001:
            noise = 0.02 * np.random.randn(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_score = _eval_fast(a, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (requires SciPy)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 6,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Iteratively improve a via a small LP on the active constraints."""
    try:
        from scipy.optimize import linprog
    except Exception:
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        act = np.where(conv >= M * (1.0 - tol_frac))[0]
        if act.size == 0:
            act = np.array([int(conv.argmax())])
        K = act.size

        # Build A_ub (K x n) s.t. (a * g)[k] <= M
        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(act):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)

        # maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        try:
            res = linprog(c,
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'presolve': True,
                                   'time_limit': max(1,
                                                     int(deadline - time.time()))})
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()

        # convex combination (fixed t=0.5)
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)

        if new_score < best_score:
            best_score = new_score
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP refine it={it} score={new_score:.6f}")
        else:
            break

    return a, best_score

# Tiny random hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Very cheap local search: random tiny mass transfers accepted only if they improve."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Parametric families sweep
# ----------------------------------------------------------------------
def _sweep_parametric(n, L_fft, start_time, deadline, best_seq, best_score):
    """Explore a wide range of analytic families and keep the best."""
    idx = np.arange(n)

    # ---- geometric (decreasing) ----
    for eps in np.logspace(-5, -0.5, 30):
        if time.time() >= deadline - 5:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        sc = _eval_fast(geo, L_fft)
        if sc < best_score:
            best_seq, best_score = geo.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

    # ---- reversed geometric (increasing) ----
    for eps in np.logspace(-5, -0.5, 30):
        if time.time() >= deadline - 5:
            break
        r = 1.0 - eps
        rev = r ** (n - 1 - idx)
        rev /= rev.sum()
        sc = _eval_fast(rev, L_fft)
        if sc < best_score:
            best_seq, best_score = rev.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] RevGeo eps={eps:.1e} score={sc:.6f}")

    # ---- mixture of two geometries ----
    r_vals = np.linspace(0.85, 0.999, 8)
    alphas = np.linspace(0.0, 1.0, 5)
    for r1 in r_vals:
        for r2 in r_vals:
            for a in alphas:
                if time.time() >= deadline - 5:
                    break
                mix = a * (r1 ** idx) + (1.0 - a) * (r2 ** idx)
                mix /= mix.sum()
                sc = _eval_fast(mix, L_fft)
                if sc < best_score:
                    best_seq, best_score = mix.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] MixGeo r1={r1:.3f} r2={r2:.3f} a={a:.2f} score={sc:.6f}")

    # ---- symmetric (mirror) shape ----
    for eps in np.logspace(-5, -0.5, 20):
        if time.time() >= deadline - 5:
            break
        r = 1.0 - eps
        sym = (r ** idx) + (r ** (n - 1 - idx))
        sym /= sym.sum()
        sc = _eval_fast(sym, L_fft)
        if sc < best_score:
            best_seq, best_score = sym.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Sym eps={eps:.1e} score={sc:.6f}")

    # ---- raised cosine (single bump) ----
    cos_seq = (1.0 - np.cos(2.0 * np.pi * idx / (n - 1))) / 2.0
    cos_seq /= cos_seq.sum()
    sc = _eval_fast(cos_seq, L_fft)
    if sc < best_score:
        best_seq, best_score = cos_seq.copy(), sc
        print(f"[{time.time() - start_time:.2f}s] Cosine bump score={sc:.6f}")

    # ---- discrete Gaussian centred in the middle ----
    centres = [n // 2]
    sigmas = [n / 6.0, n / 4.0, n / 3.0]
    for sigma in sigmas:
        if time.time() >= deadline - 5:
            break
        gauss = np.exp(- ((idx - (n - 1) / 2) ** 2) / (2.0 * sigma * sigma))
        gauss /= gauss.sum()
        sc = _eval_fast(gauss, L_fft)
        if sc < best_score:
            best_seq, best_score = gauss.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma:.1f} score={sc:.6f}")

    # ---- power family (including negative exponents) ----
    for p in np.linspace(-1.5, 2.5, 35):
        if time.time() >= deadline - 5:
            break
        pow_seq = (idx + 1) ** p
        pow_seq = np.maximum(pow_seq, 0.0)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_seq, best_score = pow_seq.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (simplex vector) that minimises
    ``evaluate_sequence``.  Returns a list of floats compatible with the
    evaluator.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose length – 4096 works well for the benchmark.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # Uniform baseline (objective = 2)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter is available, use it.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_seq, best_score = starter.copy(), cur
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric families sweep (fast)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        best_seq, best_score = _sweep_parametric(
            n, L_fft, start_time, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # 2️⃣  Repeated Adam‑style smoothed descent with random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 40:
        # generate a noisy start around current best (or uniform occasionally)
        base = best_seq if np.random.rand() < 0.75 else uniform
        noise = 0.03 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 12.0)  # ≤12 s per Adam run
        cand_seq, cand_score = _adam_smooth_descent(start, sub_deadline,
                                                    start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_seq, best_score = cand_seq.copy(), cand_score
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth global best score={best_score:.6f}")

        # Early exit if we already meet the target bound
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Global simulated‑annealing phase (escape local minima)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30 and best_score > 1.503:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, L_fft)
        print(f"[{time.time() - start_time:.2f}s] After SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Optional active‑set LP refinement (if SciPy is present)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20 and best_score > 1.503:
        best_seq, best_score = _lp_refine(best_seq, deadline,
                                          start_time, L_fft)
        print(f"[{time.time() - start_time:.2f}s] After LP best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final tiny hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline,
                                           start_time, L_fft)
        print(f"[{time.time() - start_time:.2f}s] Final polished best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and output as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
