"""
Improved search for a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096 (≈ 1 000 s budget).

Algorithm
---------
1. Load a starter vector (the global `height_sequence_1` if present,
   otherwise uniform).  Normalise to the probability simplex.

2. **Smooth‑max mirror descent** – the same exponentiated‑gradient scheme
   as before, but with a slightly larger final smoothing horizon.

3. **Projected sub‑gradient descent** – classic Euclidean projection onto
   the simplex with momentum.

4. **Greedy hill‑climb** – random mass transfers (biased towards the
   periphery) that are kept only when they improve the exact objective.

5. **Exponentiated sub‑gradient refinement** – a dedicated mirror‑descent
   on the *exact* max‑convolution (no smoothing).  The active set of
   indices where the convolution attains its maximum provides a sub‑gradient;
   we use a multiplicative update followed by a simplex projection.
   This step often squeezes a few extra percent out of the objective.

The phases respect the global wall‑clock deadline.  The best sequence
found is returned as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum of a is assumed to be 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    L_self: int,
                    L_grad: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation of the maximum convolution.
    Uses exponentiated‑gradient updates with a momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    # schedule for the smoothing parameter alpha – extended a bit
    alphas = np.geomspace(0.5, 1200.0, num=48)

    # momentum buffer on the (log‑domain) gradient
    m = np.zeros_like(a)

    it = 0
    beta = 0.9

    for alpha in alphas:
        # inner loop – stop when time or iteration budget exhausted
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # 1) self‑convolution (a * a)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights w_k ∝ exp(alpha * conv_k)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) momentum + exponentiated‑gradient (mirror‑descent on simplex)
            m = beta * m + (1.0 - beta) * grad
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * m)
            a = np.maximum(a, 1e-30)
            a /= a.sum()

            # occasional true‑objective check
            if it % 700 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (classic)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Uses a 1/√t step‑size and Euclidean projection onto the simplex.
    Stops early if no improvement is seen for a while.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    stagnation = 0          # counts iterations without improvement

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        # stop early if stagnating for many iterations
        if stagnation >= 20_000:
            break

        # active indices (within a tolerance)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : greedy hill‑climbing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_seq: np.ndarray,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing: move a fraction of mass from a source index
    (chosen proportionally to height) to a random destination (biased
    towards the outer half of the domain).  Keep the move only if it
    improves the exact objective.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()

    # pre‑compute a mask that favours outer indices
    outer_weight = np.linspace(0, 1, n)
    outer_weight = np.maximum(outer_weight, 1 - outer_weight)  # 0 at centre, 1 at ends
    outer_weight /= outer_weight.sum()

    while time.time() < deadline:
        # source proportional to current mass
        src = rng.choice(n, p=a)

        # destination biased to the periphery
        dst = rng.choice(n, p=outer_weight)

        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase‑4 : exponentiated sub‑gradient refinement (exact max)
# ----------------------------------------------------------------------
def _exp_subgrad_refine(a0: np.ndarray,
                        n: int,
                        L_self: int,
                        L_grad: int,
                        deadline: float,
                        start_time: float,
                        max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution.
    The active set (where conv equals its maximum) yields a sub‑gradient.
    We perform a multiplicative update (exponentiated‑gradient) and
    project back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a, L_self)
    best_score = 2.0 * n * conv.max()          # sum(a) == 1
    m = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        M = conv.max()
        cur_score = 2.0 * n * M

        # activity mask (average over all active indices)
        tol = 1e-12
        active_idx = np.where(conv >= M - tol)[0]
        num_active = len(active_idx)
        if num_active == 0:
            # should not happen, but safeguard
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active_idx] = 1.0 / num_active

        # gradient = 2 * (w ⋆ a_rev), extracted from the central block
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # momentum
        m = beta * m + (1.0 - beta) * grad

        # step‑size schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update (clip exponent for safety)
        delta = -eta * m
        delta = np.clip(delta, -20.0, 20.0)
        a = a * np.exp(delta)

        # project back onto the simplex
        a = _project_simplex(a)

        # keep track of best
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] ExpSubgrad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        # early‑stop if no progress for a long while
        if stagnation >= 30_000:
            break

    return best_a, best_score

# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    The search runs several optimisation phases (smooth‑max mirror descent,
    projected sub‑gradient, hill‑climb and a final exponentiated
    sub‑gradient refinement).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the external harness
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # length needed for the refined gradient

    # ------------------------------------------------------------------
    # Load a starter vector (global best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adapt length if necessary
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑1 : smooth‑max mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        smooth_deadline = deadline - 20.0
        best_seq, best_score = _smooth_descent(best_seq, n,
                                                L_self, L_grad,
                                                smooth_deadline,
                                                start_time,
                                                max_iters=300_000)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : projected sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sub_deadline = deadline - 5.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             L_self, sub_deadline,
                                             start_time, best_score,
                                             max_iters=500_000)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 : greedy hill‑climbing polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_hc, sc_hc = _hill_climb(best_seq, n,
                                  L_self, deadline - 2.0,
                                  start_time,
                                  best_seq, best_score)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved score = {sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Phase‑4 : exponentiated sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        refine_deadline = deadline - 0.5
        a_ref, sc_ref = _exp_subgrad_refine(best_seq, n,
                                            L_self, L_grad,
                                            refine_deadline,
                                            start_time,
                                            max_iters=400_000)
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp‑subgrad improved score = {sc_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
