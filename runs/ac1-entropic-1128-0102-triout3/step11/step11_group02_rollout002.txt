"""
Deterministic‑annealing optimizer for the self‑convolution L∞‑ratio.

The algorithm treats the objective

    F(a) = 2·n·max(conv(a,a)) / (∑a)² ,   a_i ≥ 0 ,  ∑a = 1

as the limit of a smooth approximation

    F_β(a) = 2·n· ( 1/β·log∑_k exp(β·(conv(a,a))_k) ) / (∑a)² .

For a fixed temperature β the gradient of the inner log‑sum‑exp is
the soft‑max weight vector w_k ∝ exp(β·conv_k).  Using the identity

    ∂(conv_k)/∂a_i = 2·a_{k‑i}   (zero when the index is out of range),

the gradient of the smooth max is

    ∇F_β(a) = 2·corr(w , a) ,

where corr is the (linear) cross‑correlation.  Correlation and convolution are
computed with real FFTs, giving O(n log n) per iteration.

We start from a few sensible warm‑starts (tent, sine, a random Dirichlet
and the optional global `height_sequence_1`), run a schedule of increasing
β values with projected momentum gradient descent, and finish with a cheap
simulated‑annealing polish.  The whole search respects the 1000 s budget
and returns the best sequence found as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Identical to evaluate_sequence (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a)[i] = Σ_j w[i+j]·a[j], for i=0..len(a)-1.
    Implemented via real FFTs.
    """
    L = _next_pow2(len(w) + len(a) - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(np.conj(W) * A, n=L)
    return corr_full[:len(a)]


# ----------------------------------------------------------------------
# Smooth‑max gradient descent with deterministic annealing
# ----------------------------------------------------------------------
def _smoothmax_opt(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   beta_schedule: list[float] | None = None) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of the max‑convolution objective.
    Returns the best sequence seen and its hard (non‑smoothed) objective.
    """
    if beta_schedule is None:
        # a moderately long schedule – later stages focus on the true max
        beta_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0]

    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta_mom = 0.9
    step0 = 1.0   # base step size (will be divided by sqrt(iter))

    it = 0
    for beta in beta_schedule:
        # allocate a modest number of iterations per temperature;
        # we will stop early if the global deadline is reached.
        max_iter_this_beta = 8000
        for _ in range(max_iter_this_beta):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a)                     # length 2n‑1
            # soft‑max weights
            scaled = beta * conv
            max_scaled = np.max(scaled)
            exp_vals = np.exp(scaled - max_scaled)   # stable
            w = exp_vals / np.sum(exp_vals)

            # gradient of the smooth max (2·corr(w,a))
            grad = 2.0 * _correlate(w, a)

            # momentum + diminishing step
            momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
            step = step0 / math.sqrt(it)
            a -= step * momentum

            # projection onto the simplex (keeps sum = 1, non‑negative)
            a = _project_simplex(a)

            # evaluate the true objective occasionally
            if it % 200 == 0 or it == 1:
                cur_score = _objective(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"beta={beta:5.1f} iter={it:,} best={best_score:.6f}")

        # reset momentum when temperature increases – it helps to
        # keep the search stable across scales.
        momentum.fill(0.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as the previous baseline)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap annealing that repeatedly moves a small random packet of mass.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of proposals – we break as soon as time runs out
    total_steps = 120_000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling; the final temperature is ~2e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        # pick a source according to current mass distribution
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a small fraction (up to 25 %) of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={best_score:.6f}")

    return best_a, best_score

# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function using deterministic annealing
    followed by a tiny simulated‑annealing polish.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid overshooting the hard limit
    deadline = start_time + budget_s - 0.5

    # Problem size – a power of two gives fast FFTs.
    # 8192 points provide a fine discretisation while staying cheap.
    n = 8192

    # ------------------------------------------------------------------
    # Build a list of warm‑start candidates.
    # ------------------------------------------------------------------
    candidates = []

    # (1) symmetric tent shape
    x = np.linspace(0.0, 1.0, n, dtype=float)
    tent = np.minimum(x, 1.0 - x)
    if tent.sum() > 0:
        candidates.append(tent / tent.sum())

    # (2) half‑sine shape (zero at both ends)
    sine = np.sin(np.pi * x)
    if sine.sum() > 0:
        candidates.append(sine / sine.sum())

    # (3) a few random Dirichlet draws
    for _ in range(2):
        rand = np.random.dirichlet(np.ones(n))
        candidates.append(rand)

    # (4) optional global starter provided by the platform
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                candidates.append(starter)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Run optimisation for each candidate, keep the best overall.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for idx, init in enumerate(candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.2f}s] Starting candidate {idx + 1}/{len(candidates)}")
        # deterministic annealing (smooth max)
        seq_opt, score_opt = _smoothmax_opt(init, deadline, start_time)

        # final light annealing polish (if time permits)
        if time.time() < deadline:
            seq_opt, score_opt = _anneal(seq_opt, deadline, start_time)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (should not happen), use uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final normalisation & conversion to plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(v) for v in best_seq]
