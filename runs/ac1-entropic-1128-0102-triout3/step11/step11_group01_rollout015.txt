def _smooth_obj_and_grad(a, beta):
    # a shape (n,)
    conv = np.convolve(a, a)  # length 2*n-1
    max_conv = conv.max()
    # compute softmax
    # subtract max for stability
    z = conv - max_conv
    expz = np.exp(beta * z)
    sum_exp = expz.sum()
    w = expz / sum_exp
    # gradient = 2 * (w * a) convolution truncated
    grad_full = np.convolve(w, a, mode='full')
    grad = 2.0 * grad_full[:a.shape[0]]
    # smoothed objective approximated
    smooth_val = (1.0 / beta) * (max_conv + np.log(sum_exp))
    return smooth_val, grad, max_conv

def _adam_opt(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    t=0
    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a,a).max()
    while time.time() < deadline:
        t+=1
        # compute gradient using softmax with beta schedule
        beta = 10.0  # or gradually increase
        smooth_val, grad, max_conv = _smooth_obj_and_grad(a, beta)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = 0.5 / (np.sqrt(v_hat) + epsilon)  # maybe scale
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)

            best_score = cur_score
            best_a = a.copy()

    if seq.shape[0] == n_target:

    x_old = np.linspace(0, 1, seq.shape[0])
    x_new = np.linspace(0, 1, n_target)
    seq_resized = np.interp(x_new, x_old, seq)

"""
Propose a step‑function of non‑negative heights minimizing the
evaluation function.

Algorithm
---------
1. Build a tiny pool of promising candidates:
   * the global starter `height_sequence_1` (if present);
   * a uniform distribution;
   * a few random sparse vectors;
   * a handful of analytic “tent‑with‑plateau’’ shapes.

2. For each elite candidate run a **smooth projected Adam optimiser**.
   The non‑smooth L∞‑objective is approximated by a soft‑max:
       F(a) ≈ max(conv(a,a)) + log‑sum‑exp(β·(conv−max))/β .
   Its gradient is obtained analytically via FFT correlation.
   After each Adam step the iterate is projected onto the probability
   simplex (non‑negative, sum = 1).

3. A short simulated‑annealing polish moves tiny mass packets between
   random positions, accepting moves that lower the true objective
   (or with Metropolis probability).

4. Return the best sequence found (as a plain Python list).  All helper
   functions are defined at module level so they can be pickled.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation (w ⋆ a) for all lags.
    Returns an array of length len(w)+len(a)-1.
    The lag‑0 element is at index len(a)-1.
    """
    L = _next_pow2(len(w) + len(a) - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(np.conj(W) * A, n=L)
    Lcorr = len(w) + len(a) - 1
    return corr_full[:Lcorr]


# ----------------------------------------------------------------------
# Exact objective (sum is already normalised to 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (sum(seq)=1 ⇒ denominator = 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {w ≥ 0, Σ w = 1} (Wang‑Carreira‑Perpiñán O(n log n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generator (tent with optional plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length n.
    p – exponent of the side ramps (p > 0).
    alpha – relative plateau width ∈ [0, 0.5]; 0 ⇒ pure tent,
            0.5 ⇒ uniform distribution.
    The shape is normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    # keep plateau even so that symmetry is exact
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Smooth approximation of the max‑convolution and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, beta: float):
    """
    Return (smooth objective, gradient, exact max conv) for a.
    Gradient of the smooth max is:
        g = 2 * (w ⋆ a)_lag0..n-1,
    where w are the soft‑max weights on conv(a,a).
    """
    n = a.shape[0]

    # --- self‑convolution -------------------------------------------------
    L1 = _next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, n=L1)
    conv = np.fft.irfft(Fa * Fa, n=L1)[: 2 * n - 1]
    max_conv = conv.max()

    # --- soft‑max weights -------------------------------------------------
    # stability: subtract max before exponentiation
    exp_arg = beta * (conv - max_conv)
    exp_arg = np.clip(exp_arg, -700, 700)      # avoid overflow
    exp_vals = np.exp(exp_arg)
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp                     # w_i ≥ 0 , Σ w_i = 1

    # smooth approximation value (not used for optimisation, only logging)
    smooth_val = max_conv + np.log(sum_exp) / beta

    # --- gradient via correlation (w ⋆ a) ----------------------------------
    L2 = _next_pow2(3 * n - 2)
    grad_full = _correlation_fft(w, a)          # length 3n‑2
    # lag‑0 is at index n‑1; take lags 0 … n‑1
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return smooth_val, grad, max_conv


# ----------------------------------------------------------------------
# Adam optimiser on the probability simplex
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray,
              deadline: float,
              start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimisation of the true objective.
    Returns the best iterate and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # schedule for the smoothness parameter β
        beta = 5.0 + (it / 2000.0) * 45.0   # linearly climb to ~50

        _, grad, _ = _smooth_obj_and_grad(a, beta)

        # Adam momentum updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple decreasing learning rate
        lr = 0.2 / math.sqrt(it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions, Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)  # fast exponential cooling

        src = rng.choice(n, p=a)                      # more likely to move heavy buckets
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25            # move ≤25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to resize a vector (linear interpolation)
# ----------------------------------------------------------------------
def _resize(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Resize a 1‑D array to ``new_len`` points using linear interpolation."""
    if seq.shape[0] == new_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.shape[0])
    x_new = np.linspace(0.0, 1.0, new_len)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size – larger n yields a tighter discretisation bound
    # --------------------------------------------------------------
    n = 32768                     # power of two → cheap FFTs

    # --------------------------------------------------------------
    # 1) Build a small pool of promising seeds
    # --------------------------------------------------------------
    pool = []                     # list of (score, vector)

    # a) global starter, if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = _resize(starter, n)
            if starter.sum() > 0.0:
                starter /= starter.sum()
            else:
                starter = np.full(n, 1.0 / n)
            pool.append((_objective(starter), starter))
    except Exception:
        pass

    # b) uniform distribution (baseline)
    uniform = np.full(n, 1.0 / n)
    pool.append((_objective(uniform), uniform.copy()))

    # c) a few random sparse vectors (K non‑zero entries)
    rng = np.random.default_rng()
    for _ in range(5):
        K = rng.integers(5, 30)                # 5‑30 spikes
        idx = rng.choice(n, size=K, replace=False)
        weights = rng.random(K)
        weights /= weights.sum()
        a = np.zeros(n)
        a[idx] = weights
        pool.append((_objective(a), a))

    # d) analytic shapes – a tiny grid (fast because we reuse FFT)
    p_grid = np.linspace(0.4, 4.0, 9)           # 0.4 … 4.0
    alpha_grid = np.linspace(0.0, 0.5, 7)       # 0 … 0.5
    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_shape(n, p, alpha)
            pool.append((_objective(a), a))
    # keep only the best few seeds
    pool.sort(key=lambda x: x[0])
    elite = [vec for _, vec in pool[:3]]
    best_global_score = pool[0][0]
    best_global_vec = pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_global_score:.6f}")

    # --------------------------------------------------------------
    # 2) Run the Adam optimiser on each elite seed
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and elite:
        per_candidate = remaining / len(elite)
        for idx, a0 in enumerate(elite):
            sub_deadline = time.time() + per_candidate * 0.9
            a_opt, score_opt = _adam_opt(a0, sub_deadline, start_time)
            if score_opt < best_global_score:
                best_global_score = score_opt
                best_global_vec = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] New best after Adam "
                      f"({idx}) = {score_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_fin, score_fin = _anneal(best_global_vec, deadline, start_time)
        if score_fin < best_global_score:
            best_global_score = score_fin
            best_global_vec = a_fin.copy()
            print(f"[{time.time() - start_time:.2f}s] Improved after anneal = "
                  f"{score_fin:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_global_vec = np.maximum(best_global_vec, 0.0)
    total = best_global_vec.sum()
    if total > 0.0:
        best_global_vec /= total
    else:
        best_global_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ "
          f"{best_global_score:.6f}")
    return [float(v) for v in best_global_vec]
