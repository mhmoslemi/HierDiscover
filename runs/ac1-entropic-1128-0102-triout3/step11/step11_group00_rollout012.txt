"""
Propose a non‑negative step function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages:

1️⃣  Quick initialise – uniform, optional global starter and a few Dirichlet
    random restarts.  The best of these is kept.

2️⃣  Smoothed gradient descent on the *soft‑max* approximation of the
    max‑convolution.  The temperature (β) is slowly increased (continuation
    method) so that early iterations follow a well‑behaved gradient while
    later ones focus on the true max.  After each step we project back onto
    the probability simplex.  Tiny Gaussian perturbations are added when the
    run stalls for many iterations.

3️⃣  A short projected sub‑gradient descent (as in the reference code) and a
    final microscopic hill‑climb that moves ε‑size packets of mass between
    random indices.  The best sequence seen throughout the whole budget is
    returned as a plain list of Python floats.

All heavy convolutions are performed with FFT (O(n log n)).  The routine
automatically respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the first len(x)+len(y)-1 entries.
    """
    n, m = len(x), len(y)
    if L is None:
        L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: n + m - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes the sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector whose sum≈1."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    # a is (almost) normalised, but we keep the denominator for safety
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Gradient of the soft‑max relaxed objective
# ----------------------------------------------------------------------
def _grad_from_weights(a: np.ndarray, w: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Given the current distribution a (length n) and the soft‑max weights w
    over the self‑convolution (length 2·n‑1), compute the gradient of
        Σ_k w_k * conv_k(a,a)
    which equals 2·Σ_k w_k·a_{k‑i} for each i.
    """
    # reverse w to turn the weighted correlation into a convolution
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev, L=L_grad)          # length = 3·n‑2
    # gradient entry i corresponds to index (2·n‑2‑i) in conv_rev
    idx = 2 * len(a) - 2 - np.arange(len(a))
    grad = 2.0 * conv_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (used as a final polish)
# ----------------------------------------------------------------------
def _subgradient_descent_long(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              best_score_initial: float) -> tuple[np.ndarray, float]:
    """Heavy‑weight sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    beta = 0.9
    alpha0 = 0.8
    it = 0
    last_improve = 0
    L_self = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = alpha0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

        # escape plateau
        if it - last_improve > 5000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(start_a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_self: int) -> tuple[np.ndarray, float]:
    """Micro‑mass transfers accepted only if they strictly improve the true objective."""
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_self)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={best_score:.6f}")

    return best_a, best_score

    return [float(x) for x in best_seq]
