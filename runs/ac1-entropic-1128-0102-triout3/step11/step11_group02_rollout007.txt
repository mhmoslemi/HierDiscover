"""
Optimization of a step‑function (non‑negative heights) for the objective

        2·n·max(conv(a,a)) / (sum(a)²)

The optimizer works in three phases:

1. **Warm‑start** – use the previously found sequence (`height_sequence_1`)
   if it exists and has the right size, otherwise start from a uniform
   distribution on a power‑of‑two length (default 2¹⁵ = 32768).

2. **Gradient descent**
   • First 30 % of the time minimise the L₂‑norm of the self‑convolution
     (∑ (a∗a)²).  This pushes the convolution towards a flat shape,
     which is known to lower the ‑∞‑norm.
   • The remaining 60 % of the time minimise a smooth approximation of the
     maximum using a soft‑max (temperature τ decreasing as 1/√t).  The
     gradient of the soft‑max is a weighted average of the sub‑gradients
     of the convolution entries; it is computed by a single FFT‑based
     convolution.

   Updates use momentum (β = 0.9) and a diminishing step‑size
   `step = 0.05 / sqrt(iter)`.  After each update the iterate is projected
   onto the probability simplex (non‑negative, sum = 1).

3. **Light annealing** – if time permits, perform cheap random mass‑transfer
   moves (Metropolis acceptance) to polish the best found vector.

The best vector seen during the run (lowest value of `evaluate_sequence`)
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient of the L2‑flattening loss
# ----------------------------------------------------------------------
def _grad_L2_with_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of   L(a) = Σ_k ( (a*a)[k] )²   with respect to a,
    given the pre‑computed self‑convolution `conv`.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(conv) + n - 1)    # length = (2n‑1) + n – 1 = 3n‑2
    C = np.fft.rfft(conv, n=L)
    Ar = np.fft.rfft(a_rev, n=L)
    conv2 = np.fft.irfft(C * Ar, n=L)
    start = n - 1
    grad = 4.0 * conv2[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the soft‑max (smooth max) of the convolution
# ----------------------------------------------------------------------
def _grad_softmax_with_conv(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of   f(a) = τ·log Σ_k exp( (a*a)[k] / τ )
    with respect to a, using the pre‑computed self‑convolution `conv`.
    """
    n = a.shape[0]
    maxc = conv.max()
    # stability: shift by the maximum before exponentiation
    exp_w = np.exp((conv - maxc) / tau)
    w = exp_w / exp_w.sum()

    a_rev = a[::-1]
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, n=L)
    Ar = np.fft.rfft(a_rev, n=L)
    conv_w = np.fft.irfft(W * Ar, n=L)
    start = n - 1
    grad = 2.0 * conv_w[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence but without checks)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(a,a)) / (sum(a)²).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """
    Very cheap annealing that repeatedly moves a small random packet of mass.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 200_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.02 * (1e-5 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)               # pick source proportionally to its mass
        dst = rng.integers(n)                   # uniform destination
        if src == dst or a[src] < 1e-16:
            continue

        eps = a[src] * rng.random() * 0.2       # move up to 20 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a
