def propose_candidate(seed=0,budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5
    np.random.seed(seed)
    n=4096
    L = next_pow2(2*n -1)

    def conv_self(a):
        return np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1]

    def eval_obj(a):
        conv = conv_self(a)
        maxb = conv.max()
        s = a.sum()
        if s < 1e-12:
            return float('inf')
        return float(2.0*n*maxb/(s*s))

    # initial candidates list
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0/n))

    # Beta distributions
    xs = (np.arange(n)+0.5)/n
    for alpha in [0.2,0.5,0.8,1.2,2,4]:
        # compute beta pdf values
        from math import gamma, log
        # Use scipy's beta pdf perhaps but we can compute using scipy.stats if import allowed
        # Avoid heavy imports; compute via np.exp(logpdf).
        # pdf ~ x^{alpha-1} (1-x)^{alpha-1}
        w = np.exp((alpha-1)*np.log(xs) + (alpha-1)*np.log(1-xs))
        # handle zero ends (log(0) -> -inf) but xs exclude 0,1.
        w = np.maximum(w, 0)
        if w.sum() == 0:
            continue
        a = w/w.sum()
        candidates.append(a)

    # Dirichlet random: generate e.g., 15 random seeds.
    rng = np.random.default_rng()
    for _ in range(15):
        # concentration param drawn uniformly log from 0.05 to 5? or just 0.5.
        conc = 10**rng.uniform(-2, 1)  # between 0.01 and 10
        a = rng.gamma(conc, 1.0, size=n)
        a = a / a.sum()
        candidates.append(a)

    best_seq = None
    best_score = float('inf')
    # optional best known global variable height_sequence_1
    global height_sequence_1
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size>0:
            # Normalize sum=1
            hs = hs/max(1e-12, hs.sum())
            candidates.append(hs)
    except Exception:
        pass

"""
Multi‑start search for a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds as follows:

1. **Generate a diverse pool of initial candidates** – uniform, several
   symmetric Beta‑shaped PDFs (U‑shaped, flat, centre‑peaked) and a few
   random Dirichlet draws.  All candidates are normalised to sum = 1.

2. **Score the pool** with an exact FFT‑based evaluation and keep the
   best few (default 3) for further optimisation.

3. **Smooth‑max mirror descent** – exponentiated‑gradient updates on a
   smooth approximation of the ``max`` of the self‑convolution.  The
   smoothing parameter α is increased geometrically, a momentum term is
   used and the step‑size decays as 1/√t.  This quickly shapes the
   distribution.

4. **Projected sub‑gradient descent** – classic sub‑gradient steps on
   the true objective with a 1/√t step‑size and Euclidean projection onto
   the probability simplex.

5. **Stochastic hill‑climbing with simulated‑annealing style moves** –
   repeatedly transfer a random fraction of mass from a source index
   (chosen proportionally to its current height) to a random destination.
   Moves are accepted if they improve the exact objective, otherwise they
   may be accepted with a small probability that decays with time.

6. **Final polish** – a last hill‑climbing phase on the globally best
   sequence for the remaining time budget.

All phases respect a hard wall‑clock deadline (default 1000 s) and keep
track of the best solution seen.  The returned list is guaranteed to be
non‑negative and to sum to 1, which makes ``evaluate_sequence`` see a
valid candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective and gradient utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, L_self: int) -> float:
    """Exact evaluation of the objective (assumes a ≥ 0)."""
    conv = _conv_self_fft(a, L_self)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _smooth_gradient(a: np.ndarray, alpha: float,
                     n: int, L_self: int, L_grad: int) -> np.ndarray:
    """
    Gradient of the smooth (soft‑max) approximation of max(conv(a,a)).
    Uses exponentiated‑gradient (mirror) formulation.
    """
    conv = _conv_self_fft(a, L_self)
    exp_arg = alpha * conv
    max_arg = exp_arg.max()
    w = np.exp(exp_arg - max_arg)          # numerical stability
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L_grad)
    Revf = np.fft.rfft(a_rev, n=L_grad)
    grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad


def _subgradient(a: np.ndarray, n: int, L_self: int):
    """
    Exact sub‑gradient of max(conv(a,a)) and the value of max(conv).
    """
    conv = _conv_self_fft(a, L_self)
    max_b = conv.max()
    active = np.where(np.abs(conv - max_b) <= 1e-12)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    return grad, max_b


# ----------------------------------------------------------------------
# Optimisation phases (stand‑alone, no closures)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    L_self: int,
                    L_grad: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a soft‑max approximation of max(conv(a,a)).
    Returns the best sequence seen during the run.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a, n, L_self)

    # geometric schedule for smoothing parameter α
    alphas = np.geomspace(0.2, 600.0, num=35)

    m = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    for alpha in alphas:
        while it < max_iters and time.time() < deadline - 0.3:
            it += 1

            grad = _smooth_gradient(a, alpha, n, L_self, L_grad)
            m = beta * m + (1.0 - beta) * grad
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * m)
            a = np.maximum(a, 1e-30)
            a /= a.sum()

            # occasional true‑objective check
            if it % 1500 == 0:
                cur = _objective(a, n, L_self)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.1f} it={it:,} score={cur:.6f}")

        if time.time() >= deadline - 0.3:
            break
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray,
                         n: int,

    """
    Classic projected sub‑gradient descent on the exact objective.
    Returns the best sequence encountered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score

    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    stagnation = 0

    while it < max_iters and time.time() < deadline - 0.2:
        it += 1
        grad, max_b = _subgradient(a, n, L_self)
        cur = 2.0 * n * max_b / (a.sum() ** 2)

        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur:.6f}")
        else:
            stagnation += 1

        if stagnation >= 18_000:
            break

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = 0.25 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with occasional uphill moves (simulated annealing).
    Returns the best sequence found.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = cur_best_score
    cur_score = cur_best_score

    rng = np.random.default_rng()
    # temperature schedule – starts at 1e‑3 and decays to 1e‑6
    T0 = 1e-3
    T_end = 1e-6

    while time.time() < deadline - 0.1:
        # linear cooling based on remaining fraction of time
        frac = (deadline - time.time()) / (deadline - start_time)
        T = max(T_end, T0 * frac)

        src = rng.choice(n, p=a)            # source weighted by current mass
        dst = rng.integers(0, n)            # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.40
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # quick feasibility check (no negatives)
        if a_prop[src] < 0.0:
            continue

        new_score = _objective(a_prop, n, L_self)
        accept = False
        if new_score < cur_score - 1e-13:
            accept = True
        else:
            # uphill acceptance with probability exp(-(Δ)/T)
            delta = new_score - cur_score
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Candidate generation helpers
# ----------------------------------------------------------------------
def _beta_candidate(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) pdf discretised on {0,…,n‑1} and normalised."""
    xs = (np.arange(n, dtype=float) + 0.5) / n
    # log‑pdf for numerical stability
    log_w = (alpha - 1) * (np.log(xs) + np.log(1.0 - xs))
    w = np.exp(log_w - np.max(log_w))    # shift to avoid overflow
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0)
    w /= w.sum()
    return w


def _dirichlet_candidate(conc: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random Dirichlet vector with given concentration."""
    a = rng.gamma(conc, 1.0, size=n)
    if a.sum() == 0.0:
        a = np.full(n, 1.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)               # 8192
    L_grad = _next_pow2(3 * n - 2)               # 16384

    # ------------------------------------------------------------------
    # Build a pool of diverse initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1) Uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2) Symmetric Beta distributions (U‑shaped to centre‑peaked)
    for a in [0.12, 0.25, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(_beta_candidate(a, n))

    # 3) Random Dirichlet draws with varying concentration
    for _ in range(12):
        conc = 10.0 ** rng.uniform(-2.0, 1.0)   # between 0.01 and 10
        candidates.append(_dirichlet_candidate(conc, n, rng))

    # 4) Possibly reuse the global best known sequence if available
    try:
        global height_sequence_1
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            hs = np.maximum(hs, 0.0)
            if hs.sum() == 0.0:
                hs = np.full(n, 1.0 / n)
            else:
                hs = hs[:n] if hs.size >= n else np.pad(hs, (0, n - hs.size))
                hs /= hs.sum()
            candidates.append(hs)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Score the pool and keep the top few for deep optimisation
    # ------------------------------------------------------------------
    scored = []
    for a in candidates:
        if time.time() >= deadline - 1.0:
            break
        a = np.maximum(a, 1e-30)
        a /= a.sum()
        score = _objective(a, n, L_self)
        scored.append((score, a.copy()))
    scored.sort(key=lambda x: x[0])
    top_k = min(3, len(scored))
    best_seq = None
    best_score = float("inf")
    print(f"[{time.time() - start_time:.2f}s] Pool evaluated, best initial score = {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Optimise each of the top candidates (smooth → subgrad → hill‑climb)
    # ------------------------------------------------------------------
    for idx in range(top_k):
        if time.time() >= deadline - 1.0:
            break
        cur_a = scored[idx][1]
        cur_score = scored[idx][0]
        print(f"[{time.time() - start_time:.2f}s] Optimising candidate #{idx+1}/{top_k} (init score={cur_score:.6f})")

        # Phase‑1 : smooth‑max mirror descent
        if time.time() < deadline - 30.0:
            cur_a, cur_score = _smooth_descent(
                cur_a, n, L_self, L_grad,
                deadline - 15.0, start_time,
                max_iters=120_000
            )
            print(f"[{time.time() - start_time:.2f}s] After smooth descent score = {cur_score:.6f}")

        # Phase‑2 : projected sub‑gradient descent
        if time.time() < deadline - 10.0:
            cur_a, cur_score = _subgradient_descent(
                cur_a, n, L_self,
                deadline - 5.0, start_time,
                cur_score,
                max_iters=200_000
            )
            print(f"[{time.time() - start_time:.2f}s] After sub‑grad score = {cur_score:.6f}")

        # Phase‑3 : stochastic hill‑climb (polish)
        if time.time() < deadline - 0.5:
            cur_a, cur_score = _hill_climb(
                cur_a, n, L_self,
                deadline, start_time,
                cur_score
            )
            print(f"[{time.time() - start_time:.2f}s] After hill‑climb score = {cur_score:.6f}")

        # Record if this candidate beats the global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_a.copy()
            print(f"[{time.time() - start_time:.2f}s] *** New global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final global polishing phase on the best sequence found so far
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq, n, L_self)

    if time.time() < deadline - 0.2:
        best_seq, best_score = _hill_climb(
            best_seq, n, L_self,
            deadline, start_time,
            best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final polishing best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
