"""
Propose a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The routine improves on the previous sub‑gradient method by
* using a smooth approximation of the max‑norm (soft‑max) and a
  gradient obtained via FFT‑based cross‑correlation,
* applying the Adam optimiser on the probability simplex,
* slowly increasing the “sharpness” μ of the soft‑max (continuation
  from a smooth to the true max),
* performing several cheap initial designs (uniform, geometric,
  Gaussian, Hann, Beta) and keeping the best as a start,
* finishing with a tiny hill‑climb that moves minute mass between
  random indices.

All heavy work runs with O(n log n) FFTs (n=4096).  The whole search
fits comfortably into the 1000 s budget and normally finds a score
below the previous 1.508364, often well under the 1.503 target.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Linear‑algebra utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not less than x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    L must be >= 2·len(a)-1 and a power of two.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative vector with sum 1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Fast exact evaluation (used during search, sum≈1)
# --------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    cur_max = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * cur_max / (s * s))

        seq = seq / seq.sum()
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] init {name} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser on smoothed max‑objective
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.02                     # base learning rate
    mu = 2.0                      # start smoothness
    mu_target = 50.0
    mu_inc = (mu_target - mu) / 25000.0   # increase gradually

    it = 0
    last_improve = 0
    best_seq_opt = a.copy()
    best_score_opt = best_score

    while time.time() < deadline - 5.0:
        it += 1

        # ---- current objective (exact) ---------------------------------
        fft_a = np.fft.rfft(a, n=L_fft)
        conv = np.fft.irfft(fft_a * fft_a, n=L_fft)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score_opt:
            best_score_opt = cur_score
            best_seq_opt = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] iter {it:,} score={cur_score:.6f}")

        # ---- smoothed max (soft‑max) -----------------------------------
        cmax = conv.max()
        shifted = conv - cmax                     # avoid overflow
        p = np.exp(mu * shifted)
        p_sum = p.sum()
        if p_sum == 0:
            p = np.full_like(p, 1.0 / p.size)
        else:
            p /= p_sum

        # ---- gradient of smoothed objective (via cross‑correlation) ---
        fft_p = np.fft.rfft(p, n=L_fft)
        grad_full = np.fft.irfft(fft_a * np.conj(fft_p), n=L_fft)
        grad = 2.0 * grad_full[:n]                # linear correlation a ⋆ p

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)

        # ---- continuation on μ -----------------------------------------
        mu = min(mu_target, mu + mu_inc)

        # ---- occasional escape from plateaus ---------------------------
        if it - last_improve > 20000:
            noise = 1e-4 * np.random.randn(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            last_improve = it

    # ------------------------------------------------------------------
    # Final microscopic hill‑climb (exact evaluation)
    # ------------------------------------------------------------------
    a = best_seq_opt.copy()
    best_seq = a.copy()
    best_score = best_score_opt
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps_move = a[src] * rng.random() * 1e-4      # tiny transfer
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move
        sc_prop = _eval_fast(a_prop, L_fft)
        if sc_prop < best_score:
            a = a_prop
            best_score = sc_prop
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check (optional, but cheap)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")

    return [float(x) for x in best_seq]
