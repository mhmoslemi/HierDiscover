"""
Low‑score step‑function generator.

The algorithm works in three phases

1️⃣  A dense sweep of the (x·(1‑x)+ε)^{‑α} family (the classic “arcsine”
    family) to obtain a good starting distribution.

2️⃣  A smooth‑approximation optimisation:
    – minimise 2·n·‖a∗a‖_p / ‖a‖₁²  (p‑norm of the self‑convolution,
      p slowly increased from 2 → 8) using Adam with Euclidean projection
      onto the probability simplex.
    – the gradient of the p‑norm is obtained analytically via a
      correlation of a weighted convolution, all computed with FFTs.

3️⃣  A biased simulated‑annealing polish that transfers a small
    fraction of mass from indices that contribute most to the current
    maximal convolution entry to random destinations.

The whole search respects the 1000 s wall‑time limit and always returns
a non‑negative list (scaling is irrelevant for the verifier).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # numerical noise may produce tiny negatives – clip to zero
    return np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Exact objective (the verifier uses this)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed p‑norm objective and its gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_obj_grad(a: np.ndarray, p: float) -> tuple[float, np.ndarray]:
    """
    Returns a tuple (obj, grad) where
        obj  = 2·n·‖a∗a‖_p               (p‑norm of the self‑convolution)
        grad = ∇_a obj  (exact gradient)
    The gradient is computed via a correlation:
        grad = 4·n·corr(w, a)   with   w_i ∝ (conv_i)^{p‑1}.
    """
    n = a.shape[0]

    # ----- convolution -----
    conv = _conv_self(a)                     # length 2n‑1, non‑negative
    # avoid exact zeros when p>2
    conv_safe = np.maximum(conv, 1e-30)

    # ----- p‑norm of conv -----
    pow_p = np.power(conv_safe, p)
    S = np.sum(pow_p)                        # Σ conv_i^p
    if S <= 0.0:
        return float("inf"), np.zeros_like(a)

    Lp = S ** (1.0 / p)                      # (Σ conv_i^p)^{1/p}
    obj = 2.0 * n * Lp

    # ----- weights w_i = (conv_i / Lp)^{p‑1} -----
    # To keep numerical stability we use the identity
    #   w_i = conv_i^{p‑1} / Lp^{p‑1} = conv_i^{p‑1} / S^{(p‑1)/p}
    scale = S ** ((p - 1.0) / p)             # = Lp^{p‑1}
    w = np.power(conv_safe, p - 1.0) / scale

    # ----- correlation w ⋆ a  (gives a vector of length 3n‑2) -----
    a_rev = a[::-1]                          # reversed a
    Lfft = _next_pow2(len(w) + n - 1)        # length for FFT
    W = np.fft.rfft(w, n=Lfft)
    A_rev = np.fft.rfft(a_rev, n=Lfft)
    corr_full = np.fft.irfft(W * A_rev, n=Lfft)   # length = len(w)+n‑1 = 3n‑2
    # extract the slice that corresponds to indices 0 … n‑1
    start = n - 1
    corr = corr_full[start:start + n]

    # gradient of the exact objective
    grad = 4.0 * n * corr
    return obj, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 8000,
    p_start: float = 2.0,
    p_end: float = 8.0,
) -> tuple[np.ndarray, float]:

    Adam optimisation of the smoothed objective.
    Returns the best iterate seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for p
        p = p_start + (p_end - p_start) * (it / max_iters)

        obj_smooth, grad = _smooth_obj_grad(a, p)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # a modest decaying learning rate
        base_lr = 0.12
        lr = base_lr / (1.0 + it * 1e-3)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the simplex (keeps non‑negativity and sum=1)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} p={p:.2f} "
                f"score={cur_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Biased simulated‑annealing polish
# ----------------------------------------------------------------------
def _biased_anneal(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    A very light annealing that preferentially moves mass from indices
    that contribute most to the current maximal convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    for it in range(1, moves + 1):
        if time.time() >= deadline:
            break

        # update temperature (log‑linear cooling)
        T = 0.02 * (1e-6 / 0.02) ** (it / moves)

        # compute the current convolution and its peak location
        conv = _conv_self(a)
        k = int(conv.argmax())

        # contributions to the peak (a[i] * a[k-i]) – only defined when both indices exist
        contrib = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib[idx] = a[idx] * a[k - idx]

        # pick a source index proportional to its contribution
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            src = rng.integers(n)
        else:
            prob = contrib / total_contrib
            src = rng.choice(n, p=prob)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 15 % of the source mass
        eps_move = a[src] * rng.random() * 0.15
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={new_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) Parameters
    # --------------------------------------------------------------
    n = 4096                     # length of the final sequence (fits comfortably)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 2) Dense sweep of the (x·(1‑x)+ε)^{‑α} family (arcsine‑like)
    # --------------------------------------------------------------
    alphas = np.linspace(0.30, 0.49, 55)   # finer grid than before
    epsilons = [0.0, 1e-9, 1e-7, 1e-5]

    for eps in epsilons:
        base = x * (1.0 - x) + eps
        for alpha in alphas:
            w = base ** (-alpha)
            a = w / w.sum()                 # normalise to sum = 1
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep α={alpha:.4f} eps={eps:g} "
                    f"score={score:.6f}"
                )
        # early exit if we already cross the target
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 3) Compare against the optional global starter (if provided)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur_score = _objective(starter)
                print(
                    f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}"
                )
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
    except Exception:
        pass  # no starter available – ignore

    # --------------------------------------------------------------
    # 4) Adam optimisation on a smoothed objective (major part of budget)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # we keep the best sequence seen across several restarts
    global_best_seq = best_seq.copy()
    global_best_score = best_score

    # allocate roughly 80 % of the remaining time to the Adam phase
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.80 * remaining

    # first run on the pure arcsine seed
    refined_seq, refined_score = _adam_optimize(
        best_seq,
        adam_deadline,
        start_time,
        max_iters=8000,
        p_start=2.0,
        p_end=8.0,
    )
    if refined_score < global_best_score:
        global_best_score = refined_score
        global_best_seq = refined_seq.copy()

    # a few random restarts (pick a Dirichlet sample and mix with the current best)
    num_random_restarts = 3
    for r in range(num_random_restarts):
        if time.time() >= adam_deadline:
            break
        # random Dirichlet with concentration 0.3 (spiky) + a small mix of the current best
        rand_dir = np.random.dirichlet(np.full(n, 0.3))
        mix = 0.6 * global_best_seq + 0.4 * rand_dir
        mix = _project_simplex(mix)

        refined_seq, refined_score = _adam_optimize(
            mix,
            adam_deadline,
            start_time,
            max_iters=6000,
            p_start=2.5,
            p_end=8.0,
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_seq = refined_seq.copy()

    print(
        f"[{time.time() - start_time:.2f}s] after Adam best score={global_best_score:.6f}"
    )

    # --------------------------------------------------------------
    # 5) Light biased simulated‑annealing polish (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        polished_seq, polished_score = _biased_anneal(
            global_best_seq, deadline, start_time, moves=120_000
        )
        if polished_score < global_best_score:
            global_best_score = polished_score
            global_best_seq = polished_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final anneal improved score={global_best_score:.6f}"
            )

    # --------------------------------------------------------------
    # 6) Final normalisation & conversion to plain list
    # --------------------------------------------------------------
    final = np.maximum(global_best_seq, 0.0)
    total = final.sum()
    if total > 0:
        final /= total
    else:
        final = np.full(n, 1.0 / n)

    # The verifier only cares about non‑negative numbers; scaling is irrelevant.
    return [float(v) for v in final]
