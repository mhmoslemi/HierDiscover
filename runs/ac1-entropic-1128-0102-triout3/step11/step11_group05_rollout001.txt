def _incremental_anneal(a0, n, deadline, start_time, cur_best_score, cur_best_seq):
    a = a0.copy()
    conv = np.convolve(a, a)
    cur_max = conv.max()
    cur_score = 2.0*n*cur_max/(a.sum()**2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    init_temp = 1e-1
    final_temp = 1e-12
    while time.time() < deadline:
        progress = (time.time() - start_time) / (deadline - start_time)
        T = init_temp * (final_temp / init_temp) ** progress
        # sample src with prob proportional to a (higher mass)
        src = rng.choice(n, p=a / a.sum())
        # choose dst randomly but not src
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1
        # epsilon scaled
        max_eps = min(a[src], 0.2*a[src]) # up to 20% of src mass
        eps = max_eps * rng.random() * (T / init_temp)  # shrink with temp
        if eps <= 0:

"""
Search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where n = len(a).  The algorithm works in three phases
and performs several short random restarts inside the 1000 s time
budget.

Phase 1 – Smooth‑max mirror descent (exponentiated‑gradient) on the
probability simplex.  A geometric schedule of smoothing parameters
α∈[0.5, 2000] and an Adam‑style step size are used; the true max
convolution is approached progressively.

Phase 2 – Projected sub‑gradient descent with momentum.  A single
active constraint (the current maximal autocorrelation) is used to
compute a sub‑gradient; the step size decays as 1/√t.

Phase 3 – Targeted incremental simulated‑annealing.  In each move a
fraction of the mass of a “heavy” index (sampled proportionally to a)
is transferred to a random destination.  The size of the transferred
mass shrinks together with the temperature, and the convolution is
updated incrementally (O(n) per move).  Only improving moves are kept
once the temperature is very low.

Multiple short restarts (different random initialisations) are tried
while time permits; the best sequence seen overall is returned as a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast objective for a probability vector (sum ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the simplex using a smooth‑max
    approximation of the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # geometric schedule – more aggressive than the baseline
    alphas = np.geomspace(0.5, 2000.0, num=48)
    # Adam‑style accumulator
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    it = 0
    max_iterations = 800_000
    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # 1) self‑convolution (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)           # log‑sum‑exp trick
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam‑style step on the simplex (exponentiated‑gradient)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            eta = 0.07 / (np.sqrt(v_hat) + eps_adam)   # element‑wise
            a = a * np.exp(-eta * m_hat)
            a = np.maximum(a, 1e-30)   # avoid exact zeros
            a /= a.sum()

            # occasional real‑objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Fine‑grained projected sub‑gradient descent with momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

        # active (first) index where the max is attained
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        # sub‑gradient of the max‑convolution
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : fast incremental simulated‑annealing (targeted moves)
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-12,
                        max_moves: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that focuses on indices that contribute most
    to the current maximal autocorrelation."""
    a = a0.copy()
    conv = np.convolve(a, a)                     # full length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (a.sum() ** 2)   # sum(a)≈1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # temperature schedule (exponential decay)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # draw a source index proportional to its mass (heavier indices more likely)
        if a.sum() == 0.0:
            src = rng.integers(n)
        else:
            src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-14:
            continue

        # destination index – uniform over the remaining positions
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # magnitude of the move – shrink with temperature for fine polishing
        max_eps = a[src] * 0.30                     # up to 30 % of src mass
        eps = max_eps * rng.random() * max(T / start_temp, 0.02)

        if eps <= 0.0:
            continue

        # incremental update of the convolution (copy‑on‑write)
        conv_new = conv.copy()
        # slice contributions from the two delta positions
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        # delta*delta contributions (three isolated indices)
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max / (a.sum() ** 2)   # denominator stays ≈1
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / max(T, 1e-15)) > rng.random():
            # accept the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

            a3, sc3 = _incremental_anneal(best_seq, n,
                                          deadline, start_time,
                                          best_score, best_seq,
                                          start_temp=5e-2,
                                          end_temp=1e-12,
                                          max_moves=5_000_000)
            if sc3 < best_score:
                best_score, best_seq = sc3, a3.copy()
                print(f"[{time.time() - start_time:.2f}s] after anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full(n, 1.0 / n)

    # Log the score as seen by the official evaluator (optional)
    try:
        external_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence reports {external_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – internal best ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
