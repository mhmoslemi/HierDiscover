def _gradient_hill_climb(a0, n, L_self, deadline, start_time, init_score):
    a = a0.copy()
    best_seq = a.copy()
    best_score = init_score
    cur_score = init_score
    stagnation = 0
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_self)
        max_val = conv.max()
        # tolerance relative
        tol = max_val * 1e-12 + 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # compute subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1)
            grad[idx] += 2.0 * a[k - idx]  # a[k-idx] safe since k-idx within bounds
        # Identify source and dest
        # source must have a > some epsilon
        eps_mass = 1e-15
        candidates = np.where(a > eps_mass)[0]
        if candidates.size == 0:
            break
        source = candidates[np.argmax(grad[candidates])]
        dest = np.argmin(grad)
        if source == dest:
            # find next smallest grad index not equal to source
            sorted_idxs = np.argsort(grad)
            for d in sorted_idxs:
                if d != source:
                    dest = d
                    break
        # compute transfer amount
        # step size schedule
        frac = min(0.3, max(0.001, 1.0 / math.sqrt(it)))  # bigger early
        delta = a[source] * frac
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[source] -= delta
        a_new[dest] += delta
        # evaluate new:
        new_score = _eval_fast(a_new, L_self)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            stagnation = 0
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time()-start_time:.2f}s] Grad Hill it={it:,} best={best_score:.6f}")
        else:
            stagnation += 1

        if stagnation > 20000:
            # add small random perturbation and reset stagnation
            rng = np.random.default_rng()
            src = rng.choice(n, p=a)
            dst = rng.integers(0,n)
            if src != dst and a[src] > 0:
                eps = a[src] * rng.random() * 0.05
                a[src] -= eps
                a[dst] += eps
                a = a / a.sum()  # just to keep sum=1 (should be unchanged)
                stagnation = 0

    return best_seq, best_score

            if new_score < cur_best:
                cur_best = new_score
                print(f"[{time.time()-start_time:.2f}s] Anneal it={it} best={cur_best:.6f}")

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------
1. Load the previously best known sequence (``height_sequence_1``) if it exists;
   otherwise start from the uniform distribution.
2. **Smooth‑max mirror descent** – exponentiated‑gradient updates on a smooth
   approximation of the max‑convolution (large α range, momentum, 1/√t
   step‑size).  This quickly shapes the distribution.
3. **Projected sub‑gradient descent** – classic sub‑gradient method on the
   exact objective, with Euclidean projection onto the simplex.
4. **Gradient‑informed hill‑climbing** – at each step we
   * compute the active set of indices attaining the current maximal
     convolution value,
   * obtain the exact sub‑gradient of the max‑convolution via a single FFT,
   * move a fraction of mass from the index with the largest gradient
     (most responsible for the peak) to the index with the smallest
     gradient, accepting the move only if the objective improves.
   This “directed” hill‑climb consistently beats the blind random moves
   used previously.
5. All phases respect the 1000 s wall‑clock budget; the best sequence found
   is finally normalised to sum = 1 and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum of a is assumed ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    L_self: int,
                    L_grad: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on a smooth approximation of the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    # a wide geometric schedule for the smoothing parameter α
    alphas = np.geomspace(0.5, 2000.0, num=50)

    # momentum buffer on the (log‑domain) gradient
    m = np.zeros_like(a)

    it = 0
    beta = 0.9

    for alpha in alphas:
        # inner loop – stop when time or iteration budget exhausted
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # 1) self‑convolution (a * a)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights w ∝ exp(alpha * conv)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) momentum + exponentiated‑gradient (mirror‑descent on simplex)
            m = beta * m + (1.0 - beta) * grad
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * m)          # stay strictly positive
            a = np.maximum(a, 1e-30)
            a /= a.sum()

            # occasional true‑objective check
            if it % 800 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} "
                          f"it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (classic)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    stagnation = 0          # counts iterations without improvement

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        # stop early if stagnating for many successive iterations
        if stagnation >= 20_000:
            break

        # active indices (within a tolerance)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : gradient‑guided hill‑climb
# ----------------------------------------------------------------------
def _gradient_hill_climb(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         L_grad: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from high‑gradient to low‑gradient coordinates while objective improves."""
    a = a0.copy()
    best_seq = a.copy()
    best_score = init_best_score
    cur_score = init_best_score

    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # 1) full self‑convolution and active set
        conv = _conv_self(a, L_self)
        max_val = conv.max()
        # tiny tolerance – enough to capture the exact arg‑max positions
        tol = max_val * 1e-12 + 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # 2) sub‑gradient of the max‑convolution via a single FFT
        w = np.zeros(2 * n - 1, dtype=float)
        w[active] = 1.0
        grad_full = _conv_linear(w, a[::-1], L_grad)   # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]       # size n

        # 3) choose source (largest gradient with non‑zero mass) and dest (smallest gradient)
        eps_mass = 1e-15
        candidates = np.where(a > eps_mass)[0]
        if candidates.size == 0:
            break
        source = candidates[np.argmax(grad[candidates])]
        dest = np.argmin(grad)
        if source == dest:
            # pick the next smallest gradient that is not the source
            sorted_idx = np.argsort(grad)
            for d in sorted_idx:
                if d != source:
                    dest = d
                    break

        # 4) transfer a fraction of the source mass
        frac = min(0.3, max(0.001, 1.0 / math.sqrt(it)))   # larger at the beginning
        delta = a[source] * frac
        if delta <= 0.0:
            continue
        a_prop = a.copy()
        a_prop[source] -= delta
        a_prop[dest] += delta

        # 5) evaluate new objective
        new_score = _eval_fast(a_prop, L_self)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            stagnation = 0
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Gradient hill‑climb it={it:,} "
                      f"best={best_score:.6f}")
        else:
            stagnation += 1

        # occasional random jitter to escape tiny plateaus
        if stagnation >= 15_000:
            rng = np.random.default_rng()
            src = rng.choice(n, p=a)
            dst = rng.integers(0, n)
            if src != dst and a[src] > 0:
                eps = a[src] * rng.random() * 0.05
                a[src] -= eps
                a[dst] += eps
                a = a / a.sum()
            stagnation = 0

    return best_seq, best_score

# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the external harness
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # length needed for gradient convolution

    # ------------------------------------------------------------------
    # Load a starter vector (global best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = None

    if starter is None or starter.size != n:
        starter = np.full(n, 1.0 / n)
    else:
        starter = np.maximum(starter, 0.0)
        if starter.sum() == 0.0:
            starter = np.full(n, 1.0 / n)
        else:
            starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑1 : smooth‑max mirror descent (core shaping phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:          # reserve time for later phases
        smooth_deadline = deadline - 200.0
        best_seq, best_score = _smooth_descent(best_seq, n,
                                                L_self, L_grad,
                                                smooth_deadline,
                                                start_time,
                                                max_iters=250_000)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : projected sub‑gradient refinement (fast descent)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sub_deadline = deadline - 80.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             L_self,
                                             sub_deadline,
                                             start_time,
                                             best_score,
                                             max_iters=400_000)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 : gradient‑guided hill‑climb (final polish)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        hill_deadline = deadline - 2.0
        a_final, sc_final = _gradient_hill_climb(best_seq, n,
                                                  L_self, L_grad,
                                                  hill_deadline,
                                                  start_time,
                                                  best_score)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Gradient hill‑climb improved score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
