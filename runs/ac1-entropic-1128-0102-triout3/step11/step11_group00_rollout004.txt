"""
Gradient‑based optimizer for the “max‑convolution” objective.

The objective for a non‑negative step function a (length n) is

    J(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

Because J is homogeneous we may fix sum(a)=1 and minimise the maximal
auto‑convolution value.  The code uses a soft‑max approximation of the
maximum, computes the exact gradient of this smooth surrogate and
optimises the vector in the probability simplex via a soft‑max
parameterisation together with Adam.  The smoothing parameter (β) is
increased gradually so that the optimiser first sees a smooth landscape
and later focuses on the true peak.

A few random perturbations are injected from time to time to escape flat
plateaus, and a cheap final “tiny hill‑climb” is performed.  All heavy
operations are implemented with FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s budget and returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.special import logsumexp  # stable soft‑max

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full convolution a * b using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    if L is None:
        L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: len(a) + len(b) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    return _conv_fft(a, a, L)


def _objective(a: np.ndarray, L_conv: int) -> float:
    """Exact objective J(a) for a vector with sum≈1."""
    conv = _conv_self(a, L_conv)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * a.shape[0] * max_conv / (s * s))


# ----------------------------------------------------------------------
# Optimiser – smooth‑max + Adam on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _optimise_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on the smooth surrogate of the max‑convolution.
    Returns the best vector (sum≈1) and its true objective value.
    """
    # ---------- parameters ----------
    beta1, beta2 = 0.9, 0.999    # Adam moments
    lr = 0.1                    # base learning rate
    eps_adam = 1e-8
    beta_smooth = 10.0          # start smoothness for the max‑approximation
    beta_max = 200.0            # never exceed this
    smooth_increase_every = 5000   # iterations
    perturb_every = 10000          # iterations

    # ---------- pre‑compute FFT lengths ----------
    L_conv = _next_pow2(2 * n - 1)           # for a*a
    L_grad = _next_pow2(3 * n - 2)           # for w * a_rev

    # ---------- initialise Adam state ----------
    # we optimise over the logits y with a = softmax(y)
    # initialise from a0 (which already sums to 1)
    y = np.log(np.maximum(a0, 1e-30))       # avoid log(0)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = a0.copy()
    best_score = _objective(best_a, L_conv)

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1

        # ----- obtain a on the simplex -----
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)          # stabilise
        a = exp_y / np.sum(exp_y)          # = softmax(y)

        # ----- real objective (for bookkeeping) -----
        cur_score = _objective(a, L_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(
                f"[{time.time() - start_time:.2f}s] iter={it:,} "
                f"J={cur_score:.6f}  beta={beta_smooth:.1f}"
            )

        # ----- smooth max (soft‑max) over the convolution -----
        conv_a = _conv_self(a, L_conv)                     # length 2n‑1
        # soft‑max weights w_k = exp(beta*conv_k)/sum exp(...)
        scaled = conv_a * beta_smooth
        w = np.exp(scaled - np.max(scaled))
        w /= w.sum()                                        # shape (2n‑1,)

        # ----- gradient of the smooth objective w.r.t. a -----
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev, L_grad)           # length 3n‑2
        grad_a = 4.0 * n * conv_w_rev[n - 1 : n - 1 + n]   # length n

        # ----- move to gradient w.r.t. logits y (projected onto simplex) -----
        dot = np.dot(grad_a, a)                            # scalar
        grad_y = a * (grad_a - dot)                        # = ∂J/∂y

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- increase smoothness slowly (focus on the true max) -----
        if it % smooth_increase_every == 0 and beta_smooth < beta_max:
            beta_smooth = min(beta_max, beta_smooth * 1.1)

        # ----- occasional tiny random perturbation to jump out of plateaus -----
        if it % perturb_every == 0:
            rng = np.random.default_rng()
            noise = 1e-3 * rng.standard_normal(n) * a
            y += noise

        # ----- termination shortcut if we have not improved for a long while -----
        if it - last_improve > 50000:
            # re‑initialise a little bit – keep the best found so far as a warm start
            y = np.log(np.maximum(best_a, 1e-30)) + 1e-2 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(
    a_start: np.ndarray,
    deadline: float,
    start_time: float,
    L_conv: int,
) -> tuple[np.ndarray, float]:
    """
    Very small mass swaps (ε≈10⁻⁶·a_i) that are kept only if the true
    objective improves.  Works on the already‑good candidate returned by
    the gradient phase.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = _objective(a, L_conv)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4   # tiny proportion of the source mass
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps

        sc = _objective(prop, L_conv)
        if sc < best_score:
            a = prop
            best_score = sc
            best_a = prop.copy()
            print(f"[{time.time() - start_time:.2f}s] tiny‑hill‑climb J={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The algorithm consists of:
        1. initialise from the global starter or from uniform,
        2. Adam optimiser on a smooth‑max surrogate (soft‑max of the convolution),
        3. a final microscopic hill‑climb.
    The whole search respects the 1000 s timeout.
    Returns the best sequence as a plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.2

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4096
    L_conv = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # initialise – try to use the globally supplied sequence if possible
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    init_a = uniform.copy()
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # noqa: F821
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                init_a = starter
                print(f"[{time.time() - start_time:.2f}s] Using global starter.")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Gradient‑based optimisation (bulk of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max Adam optimisation")
        a_opt, score_opt = _optimise_gradient(
            init_a, n, deadline - 2.0, start_time
        )
    else:
        a_opt, score_opt = init_a, _objective(init_a, L_conv)

    # --------------------------------------------------------------
    # 2️⃣  Final microscopic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb polishing")
        a_opt, score_opt = _tiny_hill_climb(
            a_opt, deadline, start_time, L_conv
        )

    # --------------------------------------------------------------
    # Normalise final answer and return
    # --------------------------------------------------------------
    a_opt = np.maximum(a_opt, 0.0)
    total = a_opt.sum()
    if total > 0:
        a_opt /= total
    else:
        a_opt = np.full(n, 1.0 / n)

    final_score = _objective(a_opt, L_conv)
    print(f"[{time.time() - start_time:.2f}s] Finished – best J ≈ {final_score:.6f}")

    # return plain Python floats (the evaluator will re‑normalise anyway)
    return [float(v) for v in a_opt]
