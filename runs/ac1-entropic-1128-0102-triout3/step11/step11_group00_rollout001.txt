assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (length ≈ 4000) with  

\[
\text{score}(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{(\sum a)^2}
\]

as small as possible.  
Because the score is invariant under scaling we can restrict ourselves to the
simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge 0 .
\]

The program works in three phases

1. **Exploratory sweep** – a quick scan of several parametric families
   (geometric, power, Beta‑type, mixtures) plus a few random Dirichlet
   draws.  
   The best vector from this stage becomes the starting point for the
   optimisation.

2. **Smooth‑max Adam optimisation** – we minimise the true objective, but
   the non‑smooth “max’’ is replaced by a **soft‑max**
   \[
   M_\varepsilon(a)=\varepsilon\,
   \log\!\Bigl(\sum_k\exp\bigl(\mathrm{conv}_k(a)/\varepsilon\bigr)\Bigr)
   \]
   with a decreasing temperature `ε`.  
   The gradient of `Mε` can be written as a convolution and is computed
   with FFTs (O(n log n)).  
   An Adam‑style adaptive step is applied on the probability simplex
   (projection after each step).  The temperature is linearly annealed from
   `ε_start=0.1` down to `ε_end=0.001`.

3. **Tiny hill‑climb** – after the Adam phase a very cheap local search moves
   tiny amounts of mass between two random indices; only improving moves are
   kept.  This polishes the final answer.

The whole routine respects the 1000 s budget, prints progress, and finally
returns the best sequence it discovered as a plain list of floats.

```python
"""
Fast optimisation of a step‑function for the objective

    2 * n * max(conv(a,a)) / (sum(a)**2)

using
* a broad parametric/random sweep,
* Adam optimisation on the probability simplex with a smoothly
  approximated max (soft‑max annealing),
* a final tiny hill‑climb.

All constraints (non‑negativity, sum = 1) are enforced by a Euclidean
projection onto the simplex.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT of length *L*.
    Returns the first (2·n‑1) entries.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _conv(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Linear (non‑circular) convolution of a and b using FFT length *L*."""
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)
    return conv[: a.shape[0] + b.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective and gradient (smooth‑max version)
# ----------------------------------------------------------------------
def _eval_score(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector that already satisfies
    sum(a)=1.
    """
    conv = _conv_self(a, L_fft)
    return float(2.0 * a.shape[0] * conv.max())


def _smooth_gradient(a: np.ndarray, eps: float, L_fft: int, L2: int) -> np.ndarray:
    """
    Gradient of the smoothed max   M_ε(a) = ε·log Σ exp(conv/ε) .
    Returned gradient equals ∂M_ε/∂a   (size = len(a)).
    """
    n = a.shape[0]

    # 1️⃣  convolution a*a (needed both for the score and for the weights)
    conv = _conv_self(a, L_fft)               # length 2n‑1

    # 2️⃣  soft‑max weights, stabilised by subtracting the maximum
    max_c = conv.max()
    w = np.exp((conv - max_c) / eps)
    w /= w.sum()                               # Σ w = 1

    # 3️⃣  gradient = 2 * (w  ⊛  a_rev)  (⊛ = linear convolution)
    a_rev = a[::-1]
    grad_full = _conv(w, a_rev, L2)            # length 3n‑2
    # Extract the slice that corresponds to the n components of a
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]    # shape (n,)

    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth‑max version)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_score: float,
                   best_vec: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the true objective using a smooth‑max surrogate.
    *a0* must already be normalised (sum = 1).  Returns the best vector found.
    """
    # FFT lengths (fixed for the whole run)
    L_fft = _next_pow2(2 * n - 1)          # for a*a
    L2 = _next_pow2(3 * n - 2)             # for w * a_rev

    a = a0.copy()
    m = np.zeros_like(a)                  # first moment
    v = np.zeros_like(a)                  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.02                             # base learning rate

    it = 0
    last_improve = 0
    # temperature schedule (linear annealing)
    eps_start = 0.1
    eps_end = 0.001

    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        # current temperature
        fraction = (time.time() - start_time) / (deadline - start_time)
        eps_temp = eps_end + (eps_start - eps_end) * (1.0 - fraction)

        # smooth gradient
        grad = _smooth_gradient(a, eps_temp, L_fft, L2)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # bias‑corrected step
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        # gradient descent + simplex projection
        a = a - step
        a = _project_simplex(a)

        # evaluate true (non‑smoothed) score
        cur_score = _eval_score(a, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} eps={eps_temp:.4g} score={cur_score:.6f}")

        # occasional escape from plateaus
        if it - last_improve > 20000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = _project_simplex(a)
            last_improve = it

    return best_vec, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(a_start: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very small redistributions of mass between two random indices.
    Only strictly improving moves are kept.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = _eval_score(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4   # tiny amount
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        sc = _eval_score(a_prop, L_fft)
        if sc < best_score:
            a = a_prop
            best = a.copy()
            best_score = sc
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‐climb score={best_score:.6f}")

    return best, best_score

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine combines a parametric/random sweep, an Adam optimisation
    with a smooth‑max surrogate, and a final microscopic hill‑climb.
    It respects the given time budget and returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # --------------------------------------------------------------
    # problem size (the same as the reference implementation)
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # start from uniform + optional global starter
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_vec = uniform.copy()
    best_score = _eval_score(best_vec, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _eval_score(starter, L_fft)
                if sc < best_score:
                    best_score, best_vec = sc, starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  fast parametric sweep (geometric, power, beta families)
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric families (r = 1‑ε)
    for eps in np.logspace(-4, -1, 120):
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        # blend with uniform
        for lam in np.linspace(0.0, 1.0, 9):
            cand = lam * uniform + (1.0 - lam) * geo
            cand /= cand.sum()
            sc = _eval_score(cand, L_fft)
            if sc < best_score:
                best_score, best_vec = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # power families
    for p in np.linspace(-1.0, 2.0, 51):
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p)
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _eval_score(pow_seq, L_fft)
        if sc < best_score:
            best_score, best_vec = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type families
    for p in [0.0, 0.5, 1.0, 1.5, 2.0]:
        for q in [0.0, 0.5, 1.0, 1.5, 2.0]:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            sc = _eval_score(beta_seq, L_fft)
            if sc < best_score:
                best_score, best_vec = sc, beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 1️⃣‑bis   random Dirichlet draws (quick diversification)
    # --------------------------------------------------------------
    max_dirichlet = 200   # keep it cheap; each eval is O(n log n)
    for _ in range(max_dirichlet):
        if time.time() >= deadline - 30:
            break
        # concentration parameter sampled log‑uniformly
        conc = 10 ** rng.uniform(-1, 1)   # between 0.1 and 10
        a = rng.gamma(conc, 1.0, size=n)
        a /= a.sum()
        sc = _eval_score(a, L_fft)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={conc:.2g} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam optimisation with smooth‑max surrogate
    # --------------------------------------------------------------
    if time.time() < deadline - 10:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_vec, best_score = _adam_optimize(best_vec, n,
                                              deadline - 5,
                                              start_time,
                                              best_score,
                                              best_vec)

    # --------------------------------------------------------------
    # 3️⃣  Final microscopic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_vec, best_score = _tiny_hill_climb(best_vec,
                                                deadline,
                                                start_time,
                                                L_fft)

    # --------------------------------------------------------------
    # Normalisation and return
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_vec]
```
