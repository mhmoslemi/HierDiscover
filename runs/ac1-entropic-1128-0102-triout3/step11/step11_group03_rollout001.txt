    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * maxc / (s**2)

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
    else:
        w = np.full_like(v, 1.0/n)
    return w

def _subgradient_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # print progress
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        momentum = beta * momentum + (1-beta) * grad
        step = step0 / np.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)
    return best_a, best_score

"""
Propose a low‑score step function.

Algorithm
---------
1. **Coarse analytic start** – a fine sweep of the family
   a_i ∝ (x_i·(1‑x_i)+ε)^(-α) on a modest grid (n₀=512) gives a good
   initial shape.

2. **Linear‑program “mass‑relocation” step** – with the current shape f,
   compute M = max(f∗f) and solve the LP
        maximise   Σ g_j
        subject to f∗g ≤ M      (point‑wise)
                   g_j ≥ 0 .
   The solution concentrates the mass where f is small, thus lowering the
   future max‑convolution.  A convex combination f←(1‑t)f+t·g (t≈0.5) is
   performed and the process is repeated a handful of times.

3. **Upsampling** – the refined coarse shape is interpolated to the final
   length (n=4096) and normalised to unit sum.

4. **Projected sub‑gradient descent** – a momentum based sub‑gradient method
   (as used in the previous baseline) finishes the optimisation on the
   high‑resolution vector.

5. **Optional light annealing** – a few thousand cheap mass‑transfer moves
   are attempted to squeeze out any remaining improvement.

The whole pipeline respects the 1000 s wall‑time limit; progress is printed
so that the user can see intermediate scores.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Convolution / objective utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the verifier."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max() / (s ** 2))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (non‑smooth)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Momentum based projected sub‑gradient descent on the exact objective.
    Returns the best iterate seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum coefficient
    step0 = 0.25        # initial step size factor
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + decreasing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Coarse analytic sweep (n₀ = 512)
    # ------------------------------------------------------------------
    n0 = 512
    x0 = (np.arange(n0, dtype=np.float64) + 0.5) / n0
    best_coarse = None
    best_score = float("inf")

    # dense grid of exponents around the known good region
    alphas = np.linspace(0.35, 0.49, 30)      # step ≈ 0.0047
    eps = 1e-12                               # tiny regulariser

    for alpha in alphas:
        w = (x0 * (1.0 - x0) + eps) ** (-alpha)
        a = w / w.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_coarse = a.copy()
            print(f"[{time.time() - start_time:.2f}s] α={alpha:.4f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑program refinement (few aggressive iterations)
    # ------------------------------------------------------------------
    # we keep the shape on the cheap grid, solve a small LP each round,
    # and move partway towards the LP solution.
    lp_iters = 12
    while time.time() < deadline - 30.0 and lp_iters > 0:
        lp_iters -= 1
        f = best_coarse
        n = f.shape[0]

        # compute current maximal self‑convolution M
        conv_f = _conv_self(f)
        M = conv_f.max()

        # Build the sparse matrix A where (A @ g)[k] = (f ∗ g)[k].
        # For every i,j we have contribution f[i] * g[j] to position i+j.
        i = np.arange(n, dtype=np.int32)
        j = np.arange(n, dtype=np.int32)
        rows = (i[:, None] + j[None, :]).reshape(-1)
        cols = np.tile(j, n)
        data = np.repeat(f, n)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Linear program: maximise Σ g_j  ⇔  minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30},
        )
        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed, exiting LP loop")
            break

        g = res.x
        g_sum = g.sum()
        if g_sum == 0.0:
            break
        g = g / g_sum                     # keep total mass = 1

        # convex combination (t controls aggressiveness)
        t = 0.5
        f_new = (1.0 - t) * f + t * g
        f_new = _project_simplex(f_new)   # stay on the simplex

        sc_new = _objective(f_new)
        if sc_new < best_score:
            best_score = sc_new
            best_coarse = f_new.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter score={sc_new:.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to the final resolution (n = 4096)
    # ------------------------------------------------------------------
    n_final = 4096
    x_final = (np.arange(n_final, dtype=np.float64) + 0.5) / n_final
    a_high = np.interp(x_final, x0, best_coarse)
    a_high = np.maximum(a_high, 0.0)
    a_high = a_high / a_high.sum()          # normalise to unit sum

    # ------------------------------------------------------------------
    # 4) Projected sub‑gradient descent on the full resolution
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + 0.85 * remaining
        refined_seq, refined_score = _subgradient_descent(
            a_high, sub_deadline, start_time
        )
        if refined_score < best_score:
            best_score = refined_score
            a_high = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] refined score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Very light simulated‑annealing polish (few ten‑thousands moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = a_high.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_steps = 20000
        for it in range(1, total_steps + 1):
            if time.time() >= deadline:
                break

            # simple exponential cooling
            T = 0.02 * (1e-5 / 0.02) ** (it / total_steps)

            # pick a source index weighted by current mass
            src = rng.choice(n_final, p=a)
            dst = rng.integers(n_final)
            if src == dst or a[src] < 1e-14:
                continue

            # transfer up to 20 % of the source mass
            move = a[src] * rng.random() * 0.20
            a_prop = a.copy()
            a_prop[src] -= move
            a_prop[dst] += move

            new_score = _objective(a_prop)
            delta = new_score - cur_score
            if delta < 0.0 or math.exp(-delta / T) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            a_high = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    a_high = np.maximum(a_high, 0.0)
    total = a_high.sum()
    if total > 0:
        a_high = a_high / total
    else:
        a_high = np.full(n_final, 1.0 / n_final)

    return [float(v) for v in a_high]
