"""
Propose a step sequence with a low evaluation score.

The algorithm consists of three stages:

1. **Broad parametric search** –  Enumerate a variety of Beta‑family and
   exponential‑Beta families (including asymmetric exponents and a mixing
   coefficient with the uniform distribution).  The best candidate from this
   cheap sweep is kept.

2. **Projected exponentiated‑gradient descent** –  The exact objective
   `2·n·max(conv)/sum²` is non‑smooth because of the max.  Its sub‑gradient
   can be written in closed form:
        ∂f/∂a = (2 n/Σ²)·B – (4 n·M/Σ³)·1,
   where `M` is the current maximal convolution value,
   `Σ` the L1‑norm (kept equal to 1) and `B_j = 2·a_{k−j}` for the
   index `k` where the maximum is attained.  An exponentiated‑gradient
   (mirror‑descent on the simplex) update
        a ← a·exp(−η·∂f/∂a) ,   a ← a/Σa
   keeps all entries non‑negative and automatically respects the simplex.
   The step size η decays as 1/√t.  The loop runs as long as time permits,
   constantly remembering the best exact objective seen.

3. **Heavy‑weight random mass‑transfer polish** –  A simple simulated‑annealing
   style local search that repeatedly moves a small random fraction of mass
   from one position to another and accepts the move only if the exact
   objective improves.  Because the move size is tiny, the number of
   iterations can be large while still respecting the time budget.

All stages respect the 1000 s wall‑time limit, never produce values larger
than 1000 (the evaluation routine would clamp them anyway), and finally
return a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT‑based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


def _rev_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Compute the vector B where B_j = 2 * a_{k-j} for all j where the
    index k-j lies inside [0, len(a)-1]; otherwise B_j = 0.
    This is the sub‑gradient of `max(conv)` at the index `k`.
    """
    n = a.shape[0]
    B = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    if start <= end:
        idx = np.arange(start, end + 1, dtype=int)
        B[idx] = 2.0 * a[k - idx]
    return B


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (vector) that minimizes the
    evaluation score under the given wall‑time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – power‑of‑two works well with the FFTs
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep (Beta‑family + mixtures)
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # mid‑bin positions
    best_seq = None
    best_score = float("inf")

    # --- symmetric Beta‑family (a_i ∝ (x·(1‑x)+ε)^{‑α}) ---
    alphas = np.linspace(0.30, 0.70, 81)          # fine grid
    epsilons = [0.0, 1e-8, 1e-6, 1e-4]

    for alpha in alphas:
        for eps in epsilons:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            sc = _objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
        if time.time() >= deadline:
            break

    # --- asymmetric Beta‑family with optional exponential damping ---
    # random search in this family (limited by remaining time)
    max_random = 3000
    rand_iter = 0
    while rand_iter < max_random and time.time() < deadline:
        p = np.random.uniform(0.30, 0.80)   # exponent for x
        q = np.random.uniform(0.30, 0.80)   # exponent for (1‑x)
        gamma = np.random.uniform(0.0, 4.0)  # exponential damping
        lam = np.random.uniform(0.0, 1.0)    # mixing weight with uniform
        delta = 1e-8

        w = (x + delta) ** (p - 1.0) * (1.0 - x + delta) ** (q - 1.0)
        if gamma > 0.0:
            w *= np.exp(-gamma * np.abs(x - 0.5))
        # mixture with uniform component
        w = lam * w + (1.0 - lam) * np.ones_like(w)
        a = w
        if a.sum() == 0:
            continue
        a = a / a.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        rand_iter += 1

    # ------------------------------------------------------------------
    # 2) Use a provided global starter if it is better
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0.0:
                    starter = starter / starter.sum()
                    cur = _objective(starter)
                    if cur < best_score:
                        best_score = cur
                        best_seq = starter.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Projected exponentiated‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if best_seq is None:
        # safe fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)

    a = best_seq.copy()
    # keep the exact simplex (sum = 1)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    else:
        a = np.full(n, 1.0 / n)

    L_fft = _next_pow2(2 * n - 1)
    iteration = 0
    base_eta = 0.6   # initial step‑size (will be divided by √t)

    while time.time() < deadline - 5.0:
        iteration += 1

        # ----- compute self‑convolution (FFT) -----
        a_fft = np.fft.rfft(a, n=L_fft)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L_fft)
        conv = conv_full[: 2 * n - 1]          # length 2n‑1
        M = conv.max()
        k_max = int(conv.argmax())
        S = a.sum()   # should be 1.0 (kept normalized)

        # ----- sub‑gradient of the exact objective -----
        B = _rev_shift(a, k_max)               # B has the factor 2 already
        grad = (2.0 * n / (S * S)) * B - (4.0 * n * M / (S * S * S)) * np.ones_like(a)

        # ----- exponentiated‑gradient update -----
        eta = base_eta / math.sqrt(iteration)
        # avoid overflow in the exponent
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)

        # Numerical safety – keep positivity and renormalise
        a = np.maximum(a, 1e-20)
        a = a / a.sum()

        # ----- keep the best exact score seen -----
        if iteration % 50 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={iteration:,}  best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Intense random mass‑transfer polish (simulated‑annealing style)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        # decide how many moves we can afford
        moves_allowed = 100_000
        for t in range(1, moves_allowed + 1):
            if time.time() >= deadline:
                break
            # source chosen proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # move a modest fraction (≤5 %) of the source mass
            eps = a[src] * rng.random() * 0.05
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={t:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
