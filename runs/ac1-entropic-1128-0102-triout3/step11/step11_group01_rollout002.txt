def _subgradient_descent_multi(a0, deadline, start_time, top_k=3):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy(); best_score = _objective(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it +=1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score; best_a = a.copy()
            print(f"[{time.time()- start_time:.2f}s] Sub‑grad iter={it:,} score={cur_score:.6f}")
        # find top_k indices
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            # argpartition to get top_k largest
            ks = np.argpartition(conv, -top_k)[-top_k:]
        # compute gradient
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1, dtype=int)
            # indices into a: k - idx
            grad[idx] += 2.0 * a[k - idx]
        # average across ks
        grad = grad / len(ks)
        # momentum & step
        momentum = beta * momentum + (1 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        # projection onto simplex
        a = _project_simplex(a)
    return best_a,best_score

"""
Propose a step‑function with a very low evaluation score.

Algorithm (≈1000 s budget)
---------------------------
1. **Problem size** – use a power‑of‑two length `n = 65536`.  This is
   large enough that the discretisation error of the continuous optimum
   (≈ 1.5) drops below the target 1.503, yet small enough that the final
   `np.convolve` in `evaluate_sequence` finishes in a reasonable time.

2. **Initial pool** – sweep a simple two‑parameter family
   (power‑ramp + flat plateau) on the full grid `p∈[0.2,2.0]`,
   `α∈[0,0.5]` and keep the best few candidates.  Add a handful of
   random draws – this gives a good starting point without expensive
   optimisation.

3. **Projected sub‑gradient descent** – improve the best candidate.
   In each iteration the self‑convolution is obtained by an FFT (O(n log n)).
   The gradient is formed from the *K* largest convolution entries
   (default K = 3) to obtain a smoother direction.
   An exponential‑moving‑average (momentum) is used and the step‑size
   follows `η_t = η₀ / sqrt(t)`.  After each step the vector is projected
   onto the probability simplex (non‑negative, sum = 1).

4. **Light simulated‑annealing polish** – once the time budget for the
   gradient phase is exhausted, move small packets of mass between
   random positions using a Metropolis rule with an exponential cooling
   schedule.

5. **Finalisation** – clip negatives, renormalise to sum = 1 and return
   the pure Python list required by the harness.

All helper functions are defined at top level so they can be pickled if
the harness runs the solution in a child process.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²)  (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # should never happen for a probability vector
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generator – power ramps with optional flat plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape (length n) built from two power ramps that meet a
    plateau of height 1.  The vector is normalised to sum = 1.
    Parameters
    ----------
    p     – exponent of the side ramps (p>0).
    alpha – relative plateau width ∈[0,0.5]; 0 → pure tent,
            0.5 → uniform.
    """
    alpha = max(0.0, min(0.5, alpha))

    # number of points belonging to the central plateau
    plateau_len = int(round(alpha * n))
    # make the plateau length even (symmetry)
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # values rise from 0…1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # adjust possible off‑by‑one caused by rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (using the K largest convolution entries)
# ----------------------------------------------------------------------
def _subgradient_descent_multi(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               top_k: int = 3) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent until ``deadline``.
    Gradient is averaged over the ``top_k`` largest entries of the self‑convolution.
    Returns the best iterate seen (and its score).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum variables
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices of the K largest convolution values
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # construct the sub‑gradient (average over the selected ks)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        grad /= float(len(ks))

        # momentum + diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions; Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # total number of attempts – chosen adaptively to fit the remaining time
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # source is chosen proportionally to its mass (more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 25 %) of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

        pass

    # 4) Local refinement – projected sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 40.0:                     # keep a few seconds for polishing
        # allocate the majority of the left time to descent
        sub_deadline = time.time() + 0.92 * remaining
        refined_seq, refined_score = _subgradient_descent_multi(best_seq,
                                                               sub_deadline,
                                                               start_time,
                                                               top_k=3)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
