def maximize_g(f, M):
    n = len(f)
    # Build sparse A matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i = k - j => j = k - i
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # for each i, coefficient f[i] for variable g_j where j = k - i
        # we can loop i_range
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    c = -np.ones(n)  # maximize sum g, so minimize -sum
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('LP failed')
    g0 = res.x
    # Rescale to sum = sqrt(2n)
    sum_target = math.sqrt(2*n)
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f  # fallback
    g = g0 * (sum_target / sum_g0)
    return g
```

def _adam_descent(a0, n, deadline, start_time, best_score_initial):

            best_a = a.copy()

            noise = 1e-5 * rng.standard_normal(n) * a

"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  A quick parametric sweep over a richer family of monotone decreasing
    shapes (geometric × power, mixed with the uniform distribution) to
    obtain a good starting point.

2️⃣  A smooth‑max descent using a soft‑max approximation of the “max‑conv”
    term and an Adam‑style projected gradient update on the probability
    simplex.  The temperature of the soft‑max (β) is slowly increased so
    that the optimisation first follows a smooth landscape and then
    focuses on the true objective.

3️⃣  A final hill‑climb that transfers a small fraction of mass between
    random coordinates, accepting only improving moves.  This fine‑tunes
    the solution to the exact objective used by ``evaluate_sequence``.

All heavy work is performed with FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s time budget and returns the best
sequence it discovers (normalised to sum 1 so the evaluation is scale‑free).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a                     (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    # 1. convolution a*a
    conv = _conv_self(a, L=L_conv)          # length 2n‑1
    max_c = conv.max()

    # 2. soft‑max weights (stable)
    shifted = conv - max_c                 # max becomes 0
    exp_vals = np.exp(beta * shifted)      # may under‑flow for large negative values
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # 3. correlation w * a (a reversed) → gradient
    a_rev = a[::-1]                        # length n
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)   # linear convolution w * a_rev

    # Extract the slice that equals Σ_k w_k a_{k-i}
    start = a.shape[0] - 1                 # n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max objective
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the smooth‑max surrogate.
    β starts small (smooth landscape) and is slowly increased.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # FFT sizes (fixed throughout the run)
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # soft‑max temperature schedule
    beta = 5.0               # start fairly smooth
    beta_mul = 1.015         # geometric increase per iteration
    beta_max = 200.0

    while time.time() < deadline:
        t += 1

        # gradient of the smooth surrogate
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # exact score (sum of a is 1 → denominator = 1)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth‑descent it={t:,} "
                f"β={beta:.2f} score={cur_score:.6f}"
            )

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.1 / (np.sqrt(v_hat) + eps)         # base step‑size 0.1
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional tiny random perturbation to escape plateaus
        if t % 5000 == 0:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final microscopic hill‑climb (larger ε than the previous code)
# ----------------------------------------------------------------------
def _final_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two coordinates.
    Only improving moves are kept; ε≈10⁻³·a_src.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3       # up to 0.1 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
    return best_a, best_score
