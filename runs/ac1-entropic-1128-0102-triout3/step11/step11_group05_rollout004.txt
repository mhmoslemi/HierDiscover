assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (∑a = 1) that minimises  

\[
F(a)=2\,n\;\max_k (a*a)[k]\; .
\]

`evaluate_sequence` is scaling‑invariant, therefore we can keep the vector normalised
(`∑a = 1`) during the whole search.  
The new search consists of three ideas that improve on the previous
implementation:

1. **Multiple random restarts** – different initial points avoid getting stuck
   in the same local minimum.  
2. **More aggressive smooth‑max mirror descent** – a finer α‑schedule,
   a larger iteration budget and a simple Adam‑style momentum accelerate
   convergence to a lower `max(conv)`.  
3. **Two‑phase annealing** – a quick “local” anneal after every restart
   (≈ 1 M moves) and a long final anneal (≈ 3 M moves) that updates the
   convolution in‑place.  The temperature schedule is exponential and the
   acceptance test is the standard Metropolis rule.

The whole programme respects the 1000 s budget, prints progress
messages and finally returns the best sequence it has found.

```python
"""
Multi‑restart optimisation for the step‑function problem.

The algorithm proceeds in three layers:

* Several random restarts (Dirichlet‑initialised) each run a long
  smooth‑max mirror‑descent (exponentiated‑gradient) with an Adam‑style
  momentum, followed by a short projected sub‑gradient step and a quick
  simulated‑annealing polish.

* The best vector from all restarts is fed to a final, longer annealer
  (few‑million mass‑transfer moves).  The convolution is updated
  in‑place, which makes the annealer fast enough to explore a much larger
  neighbourhood than the previous implementation.

* The routine is time‑aware: it never exceeds the given budget,
  reserves a few seconds for finalisation and prints intermediate
  objective values.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """
    Fast evaluation of the objective on a (assumed non‑negative, sum≈1).
    Returns 2·n·max(conv) because the denominator is 1 after normalisation.
    """
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    if a.sum() < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent with simple Adam momentum
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the smooth‑max
    objective  (log‑sum‑exp) with a tiny Adam‑style momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # a more aggressive α‑schedule: go up to 2000 and use more points
    alphas = np.geomspace(0.5, 2000.0, num=50)

    it = 0
    # Adam / RMSProp style accumulators (same shape as a)
    m = np.zeros_like(a)          # first‑moment estimate
    v = np.zeros_like(a)          # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.03                    # base learning rate for the multiplicative update

    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # 1) self‑convolution (FFT, full length)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights for the smooth max
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam momentum on the (additive) gradient
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # 5) multiplicative update (mirror descent on simplex)
            eta = lr0 / (np.sqrt(v_hat) + eps)    # adaptive step size
            a = a * np.exp(-eta * m_hat)          # element‑wise
            a = np.maximum(a, 1e-20)              # avoid exact zeros
            a /= a.sum()

            # occasional real‑objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

            # stop early if we run out of time
            if it >= max_iters:
                break
        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true max‑convolution.
    Uses a fixed momentum (beta) and a simple 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (first occurrence of the max)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : fast incremental simulated‑annealing
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-12,
                        max_moves: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealer.
    The convolution is updated in‑place; only the affected slices are
    modified, which makes the inner loop cheap enough for several million
    moves.
    """
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1 (≈8191)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # denominator = 1 (already normalised)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # exponential temperature schedule (based on elapsed fraction)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.35   # move up to 35 % of src mass
        if eps <= 0.0:
            continue

        # --- incremental update of the convolution (in‑place) ---
        # store the delta to be able to revert if the move is rejected
        conv_old = conv.copy()

        # contribution of the moved mass (old a) to the two affected blocks
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a

        # delta‑delta terms (three isolated indices)
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / T) > rng.random():
            # accept
            a[src] -= eps
            a[dst] += eps
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")
        else:
            # reject – revert convolution
            conv = conv_old

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Multi‑restart optimiser for the step‑function problem.
    Returns a list of non‑negative heights (length 4096) that gives the
    smallest value seen by `evaluate_sequence` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – either from the supplied global or uniform
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')

    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Allocate time: keep a slice for a final long anneal
    # ------------------------------------------------------------------
    final_anneal_time = 150.0          # seconds reserved for the final polish
    restart_deadline = deadline - final_anneal_time

    restart_id = 0
    while True:
        now = time.time()
        time_left = restart_deadline - now
        if time_left < 30.0:      # not enough time for another full restart
            break

        # ---- random initial vector for this restart -----------------
        # use a Dirichlet with concentration <1 for a spikier start
        conc = 0.4
        a0 = np.random.dirichlet(np.full(n, conc))

        # occasionally blend with the current global best (helps to
        # explore the neighbourhood of a good solution)
        if np.random.rand() < 0.25:
            a0 = 0.5 * a0 + 0.5 * best_seq
            a0 = a0 / a0.sum()

        # ---- Phase‑1 : smooth‑max mirror descent --------------------
        phase1_deadline = now + time_left * 0.55
        a_smooth, scr_smooth = _smooth_descent(
            a0, n, phase1_deadline, start_time, max_iters=800_000)

        # ---- Phase‑2 : short sub‑gradient refinement ----------------
        phase2_deadline = now + time_left * 0.80
        a_sub, scr_sub = _subgradient_descent(
            a_smooth, n, phase2_deadline, start_time, scr_smooth)

        # ---- Phase‑3 : quick anneal to polish the restart result ---
        phase3_deadline = now + time_left * 0.98
        a_anneal, scr_anneal = _incremental_anneal(
            a_sub, n,
            phase3_deadline,
            start_time,
            scr_sub,
            a_sub,
            start_temp=5e-2,
            end_temp=1e-6,
            max_moves=1_000_000)

        # keep the best from this restart
        if scr_anneal < best_score:
            best_score = scr_anneal
            best_seq = a_anneal.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"New global best after restart {restart_id} → {best_score:.6f}")

        restart_id += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final long annealing (the bulk of the remaining budget)
    # ------------------------------------------------------------------
    final_deadline = deadline
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
    a_final, sc_final = _incremental_anneal(
        best_seq, n,
        final_deadline,
        start_time,
        best_score,
        best_seq,
        start_temp=1e-2,
        end_temp=1e-12,
        max_moves=3_000_000)

    if sc_final < best_score:
        best_score = sc_final
        best_seq = a_final.copy()
        print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – global best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
