"""
Improved search for a low‑value step function.

Overall strategy
-----------------
1. **Rich initial pool** – generate a variety of analytic shapes:
   * symmetric power‑ramp “tent‑with‑plateau’’ shapes,
   * asymmetric two‑sided power‑ramps,
   * a handful of completely random simplex vectors.
   The best few (default 5) are kept.

2. **Adam‑style projected sub‑gradient descent** – for each pooled candidate
   we run a non‑smooth optimisation.  The sub‑gradient of the
   max‑convolution entry is known analytically; we feed it to an Adam
   optimiser and project onto the probability simplex after every step.
   Every ~2·10⁵ iterations a tiny Gaussian perturbation is added to
   escape possible plateaus.

3. **Annealing polish** – a fast simulated‑annealing pass moves tiny
   mass packets between random positions, accepting uphill moves with a
   decreasing temperature schedule.

4. **Final normalisation** – clip any negative artefacts, renormalise the
   vector to sum‑one and return a plain Python list.

The code is deliberately written with only top‑level functions so that
the harness can pickle it if necessary.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    Identical to evaluate_sequence but without defensive checks.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex { w ≥ 0, Σ w = 1 }.
    O(n log n) algorithm (Wang & Carreira‑Perpiñán).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Symmetric shape generator (tent with optional plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length n.
    - p     : exponent of the side ramps (p > 0).
    - alpha : relative plateau width ∈ [0, 0.5];
              α = 0 → pure tent, α = 0.5 → uniform.
    The vector is normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    # enforce even plateau length for perfect symmetry
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Asymmetric two‑sided power‑ramp shape
# ----------------------------------------------------------------------
def _make_asymmetric_shape(n: int, pl: float, pr: float, alpha: float) -> np.ndarray:
    """
    Asymmetric shape of length n.
    - pl, pr : exponents for the left and right ramps (>0).
    - alpha  : relative plateau width ∈ [0, 0.5].
    Returns a non‑negative vector normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    rem = n - plateau_len
    left_len = rem // 2
    right_len = rem - left_len   # ensures total length = n

    # left ramp (rising from 0 to 1)
    if left_len > 0:
        idx_l = np.arange(1, left_len + 1, dtype=float) / left_len
        left = idx_l ** pl
    else:
        left = np.empty(0)

    # right ramp (falling from 1 to 0)
    if right_len > 0:
        idx_r = np.arange(right_len, 0, -1, dtype=float) / right_len
        right = idx_r ** pr
    else:
        right = np.empty(0)

    shape = np.concatenate([left, np.ones(plateau_len), right])

    # pad / trim to exact length
    if shape.size < n:
        shape = np.concatenate([shape, np.zeros(n - shape.size)])
    elif shape.size > n:
        shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)

    return shape


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  verbose: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the non‑smooth objective.
    Returns the best vector seen (and its objective value).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12               # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if verbose:
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                      f"best={cur_score:.6f}")

        # sub‑gradient of the max‑convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a -= step * m_hat

        # projection onto the simplex
        a = _project_simplex(a)

        # occasional random “kick” to avoid stagnation
        if it % 250_000 == 0:
            rng = np.random.default_rng()
            a += 1e-6 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            verbose: bool = True) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions; Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of attempts – limited only by remaining wall‑clock time
    it = 0
    while time.time() < deadline:
        it += 1
        # exponential cooling: start at 0.02, finish near 1e‑6
        T = 0.02 * (1e-6 / 0.02) ** (it / 500_000)

        # choose source preferentially proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 25 %) of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                if verbose:
                    print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                          f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size – power of two for fast FFTs
    # ------------------------------------------------------------------
    n = 32768

    # ------------------------------------------------------------------
    # 1) Build a diverse initial pool
    # ------------------------------------------------------------------
    top_k = 5
    top = []                      # list of (score, vector)

    # symmetric power‑ramp grid
    p_grid = np.linspace(0.2, 2.5, 13)          # 0.2 … 2.5
    alpha_grid = np.linspace(0.0, 0.5, 7)       # 0 … 0.5
    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_shape(n, p, alpha)
            score = _objective(a)
            if len(top) < top_k:
                top.append((score, a.copy()))
                top.sort(key=lambda x: x[0])
            elif score < top[-1][0]:
                top[-1] = (score, a.copy())
                top.sort(key=lambda x: x[0])

    # asymmetric two‑sided ramps
    p_asym = np.linspace(0.5, 2.5, 9)
    for pl in p_asym:
        for pr in p_asym:
            for alpha in alpha_grid:
                a = _make_asymmetric_shape(n, pl, pr, alpha)
                score = _objective(a)
                if len(top) < top_k:
                    top.append((score, a.copy()))
                    top.sort(key=lambda x: x[0])
                elif score < top[-1][0]:
                    top[-1] = (score, a.copy())
                    top.sort(key=lambda x: x[0])

    # a few completely random simplex vectors (lightweight smoothing)
    rng = np.random.default_rng()
    for _ in range(100):
        if time.time() >= deadline:
            break
        a = rng.random(n)
        a = _project_simplex(a)
        score = _objective(a)
        if len(top) < top_k:
            top.append((score, a.copy()))
            top.sort(key=lambda x: x[0])
        elif score < top[-1][0]:
            top[-1] = (score, a.copy())
            top.sort(key=lambda x: x[0])

    # occasional report
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {top[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2) Refine best candidates with Adam‑style descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate ~80 % of remaining time to descent
        descent_budget = remaining * 0.80
        per_candidate = descent_budget / len(top)
        best_overall_a = None
        best_overall_score = float("inf")

        for idx, (init_score, init_vec) in enumerate(top):
            sub_deadline = time.time() + per_candidate
            refined_a, refined_score = _adam_descent(init_vec,
                                                      sub_deadline,
                                                      start_time)
            if refined_score < best_overall_score:
                best_overall_score = refined_score
                best_overall_a = refined_a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Candidate {idx} refined to {refined_score:.6f}")

            if time.time() >= deadline:
                break

        # if descent never improved (unlikely) fallback to the best pool entry
        if best_overall_a is None:
            best_overall_a = top[0][1].copy()
            best_overall_score = top[0][0]

    else:
        # not enough time – just pick the best pool entry
        best_overall_a = top[0][1].copy()
        best_overall_score = top[0][0]

    # ------------------------------------------------------------------
    # 3) Light annealing polish with the leftover budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_overall_a, best_overall_score = _anneal(best_overall_a,
                                                    deadline,
                                                    start_time)

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_overall_a = np.maximum(best_overall_a, 0.0)
    total = best_overall_a.sum()
    if total > 0:
        best_overall_a /= total
    else:
        best_overall_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_overall_score:.6f}")
    return [float(v) for v in best_overall_a]
