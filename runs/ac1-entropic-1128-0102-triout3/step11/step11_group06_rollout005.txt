def subgradient_descent(a, time_deadline):
    n = len(a)
    iteration = 0
    best_seq = a.copy()
    best_score = _objective(a)
    while time.time() < time_deadline:
        iteration += 1
        # compute conv via FFT
        L = _next_pow2(2*n-1)
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        M = conv.max()
        # tolerance for near-maximum
        # tolerance relative: 1e-12?
        tol = 1e-12
        K = np.where(conv >= M - tol)[0]
        # compute subgradient g
        g = np.zeros_like(a)
        for k in K:
            if k < n:
                # left part
                # a_rev[:k+1] reversed is a[:k+1][::-1]
                g[:k+1] += a[k::-1]  # a[k],a[k-1]...a[0]
                # right part (if any)
                if k+1 < n:
                    # a[1: n-k] addition to g[k+1:]
                    g[k+1:] += a[1:n-k]
            else:
                start = k - (n - 1)
                # indices start..n-1
                # reversed a[start:n] i.e., a[start:n][::-1]
                g[start:] += a[start:][::-1]
        # gradient of objective = 2*n * g
        grad = 2.0 * n * g
        # step size schedule: maybe eta = 0.5 / sqrt(iteration)
        eta = 0.5 / np.sqrt(iteration)
        a_new = a - eta * grad
        # ensure non-negative? we later project onto simplex anyway.
        a = _project_simplex(a_new)
        # optionally evaluate periodic exact objective
        if iteration % 20 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={iteration} best={cur_score:.6f}")
        # break near deadline
        if time.time() >= time_deadline - 0.5:  # also leave time for final polished step
            break
    return best_seq, best_score

"""
Propose a low‑objective step sequence.

Algorithm
---------

* Choose a moderately large problem size (n = 8192).  A larger n
  reduces the discretisation error of the known near‑optimal “Beta‑shape”.

* Sweep a tiny 2‑parameter family
        a_i ∝ (x_i·(1‑x_i)+ε)^{‑α}
  over α ∈ [0.44,0.56] (step 0.001) and ε ∈ {0,1e‑10,1e‑8,1e‑6,1e‑4}.
  The best member of the family is kept as a warm start.

* If a previous construction (`height_sequence_1`) is supplied, up‑sample it
  to length *n* and compare its exact objective – keep the best one.

* Refine the warm start with a smoothed Adam optimiser.
  The non‑smooth objective `2·n·max(conv)/sum²` is approximated by a
  log‑sum‑exp surrogate; its gradient can be obtained with a single
  extra FFT (correlation with the soft‑max weights).  The iteration runs
  until the time budget is almost exhausted and periodically stores the
  best exact objective seen.

* A very light simulated‑annealing “mass‑transfer” polish finishes the
  search: move a small fraction of mass from a source index (chosen
  proportionally to its current value) to a random destination; accept
  only improvements.

All updates keep the vector on the probability simplex (non‑negative,
sum = 1) via Euclidean projection.  The final sequence is returned as a
plain list of Python floats; each entry is ≤ 1000 so the harness does
not clip them.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a very low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # problem size – 8192 is a good compromise between quality and speed
    n = 8192

    # ------------------------------------------------------------------
    # 1) Fine sweep over the Beta‑family (α, ε)
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # mid‑bin positions
    best_seq = None
    best_score = float("inf")

    alphas = np.linspace(0.44, 0.56, 121)           # step ≈0.001
    epsilons = [0.0, 1e-10, 1e-8, 1e-6, 1e-4]

    for alpha in alphas:
        # early exit if we are out of time
        if time.time() >= deadline:
            break
        for eps in epsilons:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep α={alpha:.4f} ε={eps} ⇒ {score:.6f}")

    # ------------------------------------------------------------------
    # 2) Incorporate a possible external starter (height_sequence_1)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                # upsample / downsample to length n
                if starter.size != n:
                    # simple nearest‑neighbour repeat/truncate
                    rep = n // starter.size
                    rem = n % starter.size
                    tmp = np.repeat(starter, rep)
                    if rem:
                        tmp = np.concatenate([tmp, starter[:rem]])
                    starter = tmp
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
                        print(f"[{time.time() - start_time:.2f}s] using external starter ⇒ {cur_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Smoothed Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.06

    # FFT length that can accommodate both convolutions (self‑conv + weight‑conv)
    L = _next_pow2(3 * n)

    iteration = 0
    beta_start, beta_end = 5.0, 400.0  # more aggressive sharpening

    while time.time() < deadline - 2.0:
        iteration += 1

        # ---- self‑convolution (linear, via FFT) --------------------
        a_fft = np.fft.rfft(a, n=_next_pow2(2 * n - 1))
        conv_full = np.fft.irfft(a_fft * a_fft, n=_next_pow2(2 * n - 1))
        conv = conv_full[: 2 * n - 1]                     # length 2n‑1

        # ---- smoothing parameter (increases with elapsed time) ----
        elapsed = time.time() - start_time
        prop = min(1.0, elapsed / (budget_s - 0.5))
        beta = beta_start + (beta_end - beta_start) * prop

        # ---- soft‑max weights for the log‑sum‑exp surrogate ------
        beta_conv = beta * conv
        max_beta = beta_conv.max()
        exp_shift = np.exp(beta_conv - max_beta)
        sum_exp = exp_shift.sum()
        w = exp_shift / sum_exp               # shape (2n‑1,)

        # ---- gradient of the smooth surrogate --------------------
        # Correlation of w with a (cross‑correlation)
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        Ba = corr_full[n - 1:n - 1 + n]        # slice to size n
        gradient = 4.0 * n * Ba               # ∇F_s

        # ---- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * gradient
        v = beta2 * v + (1.0 - beta2) * (gradient ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)               # keep on the simplex

        # ---- keep the best exact objective seen so far ------------
        if iteration % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{elapsed:.2f}s] Adam iter={iteration:,} best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light simulated‑annealing polish (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        # number of moves is adaptive to remaining time
        remaining = deadline - time.time()
        est_move_time = 0.00012  # estimate 120 µs per move (conservative)
        total_moves = int(max(1e3, remaining / est_move_time))

        for t in range(1, total_moves + 1):
            if time.time() >= deadline:
                break
            # source chosen proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # move a modest fraction (≤10 %) of the source mass
            eps = a[src] * rng.random() * 0.10
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={t:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # clipping to the limits used in `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
