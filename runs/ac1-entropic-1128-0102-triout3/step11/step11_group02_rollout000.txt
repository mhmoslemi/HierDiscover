"""
Step‑function optimizer for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

using a smooth‐max surrogate and Adam updates on the probability simplex.
The scheme:

1. Warm‑start from the previously‑found best sequence (if available) or a
   symmetric triangular “tent”.
2. Minimise a soft‑max approximation of the max‑convolution:
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
   The gradient of f_α is a cross‑correlation that can be computed with FFTs.
   Adam is applied and the iterate is projected onto the simplex after each step.
   α is increased slowly (continuation) so the optimisation first sees a smooth
   landscape and finally focuses on the true max.
3. A cheap Metropolis annealing moves tiny packets of mass between indices
   to polish the best iterate.
4. Return the normalised vector (list of non‑negative floats).

The whole routine runs for at most the given budget (default 1000 s) and
prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same as in the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap annealing that repeatedly moves a small random packet of mass.
    Returns the best sequence seen and its objective value.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 60_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)               # weighted source
        dst = rng.integers(n)                  # uniform destination
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two gives cheap FFTs.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Warm start: try the globally stored best, otherwise a symmetric tent.
    # --------------------------------------------------------------
    a = None
    try:
        # height_sequence_1 may have any length – truncate / pad to n.
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size > n:
                starter = starter[:n]
            elif starter.size < n:

                                          np.zeros(n - starter.size, dtype=float)])
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                a = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Loaded global starter")
    except Exception:
        pass

    if a is None:
        # symmetric triangular “tent” (a_i ∝ min(i, n‑1‑i))
        x = np.arange(n, dtype=float) / (n - 1)
        tent = np.minimum(x, 1.0 - x)
        if tent.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = tent / tent.sum()

    # keep the best seen vector (objective evaluated on the exact max)
    best_a = a.copy()
    best_score = _objective(a)
    print(f"[{time.time() - start_time:.2f}s] Start score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam‑based minimisation of a smooth‑max surrogate.
    # --------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    # continuation schedule for the soft‑max temperature α
    while time.time() < deadline:
        it += 1

        # ---- 1) convolution of the current iterate -----------------
        conv = _conv_self(a)                     # length 2n‑1

        # ---- 2) soft‑max weights ---------------------------------
        # α grows slowly; value 0.5 → ~30 over a few thousand iterations.
        alpha = 0.5 + 0.015 * it
        maxc = conv.max()
        # stable soft‑max
        exp_term = np.exp(alpha * (conv - maxc))
        w = exp_term / exp_term.sum()            # Σ w = 1

        # ---- 3) gradient of the smooth objective -----------------
        # grad_i = 2 * Σ_j w_{i+j} * a_j
        a_rev = a[::-1]
        Lc = _next_pow2(len(w) + len(a_rev) - 1)   # length for correlation
        Wc = np.fft.rfft(w, n=Lc)
        Arev = np.fft.rfft(a_rev, n=Lc)
        corr = np.fft.irfft(Wc * Arev, n=Lc)
        # slice = indices n‑1 … n‑1+n‑1  (inclusive start, exclusive end)
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # ---- 4) Adam update --------------------------------------
        lr = 0.12 / math.sqrt(it)                # diminishing step size
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- 5) projection onto the simplex -----------------------
        a = _project_simplex(a)

        # ---- 6) keep the best true objective ----------------------
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Iter={it:,} "
                  f"score={cur_score:.6f}")

        # occasional early‑stop if we have exhausted the time budget
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_a, deadline, start_time)
        if anneal_score < best_score:
            best_a = anneal_seq.copy()
            best_score = anneal_score
            print(f"[{time.time() - start_time:.2f}s] Anneal final best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain Python list.
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – length {n}, "
          f"final score≈{best_score:.6f}")
    return [float(v) for v in best_a]
