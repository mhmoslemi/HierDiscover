"""
Propose a low‑objective step sequence.

The algorithm works in three stages:

1. **Parametric sweep** – a quick scan of a few one‑parameter families
   (Beta‑type, symmetric exponential, sinusoidal powers).  The best
   candidate from this cheap search is kept as a starting point.

2. **Sub‑gradient descent on the exact max‑convolution** – the objective
   (with the sum normalised to 1) is 2 n · max(conv(a,a)).  The sub‑gradient
   of the max‑convolution is computed analytically (it is just a shifted
   copy of ``a``).  An additive update with a decreasing step size and a
   projection onto the probability simplex drives the sequence to lower
   peaks.  The loop runs as long as the time budget permits.

3. **Light simulated‑annealing polish** – random “mass‑transfer’’ moves
   are tried; a move is kept if it improves the exact objective.

All operations use FFT‑based convolutions (``np.fft``) for speed, and the
code respects the 1000 s wall‑time limit.  The final sequence is returned
as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution via FFT (full length 2·n‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=L)
    full = np.fft.irfft(af * af, n=L)
    return full[: 2 * n - 1]


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv(a,a))`` with respect to ``a``.
    For each index `k` where the maximum is attained we add the
    vector (a_{k-i}+a_{i-k})_i and finally average over all such `k`.
    """
    n = a.shape[0]
    max_val = conv.max()
    max_idxs = np.flatnonzero(conv == max_val)

    g = np.zeros_like(a)
    idx = np.arange(n)
    for k in max_idxs:
        # term from a_{k-i}
        j1 = k - idx
        mask1 = (j1 >= 0) & (j1 < n)
        g[mask1] += a[j1[mask1]]

        # term from a_{i-k}
        j2 = idx - k
        mask2 = (j2 >= 0) & (j2 < n)
        g[mask2] += a[j2[mask2]]

    g /= len(max_idxs)          # average if several maxima
    return g                     # this is ∂ max_conv / ∂ a
