"""
search for a non‑negative step function that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The implementation uses three stages:

1️⃣  A dense parametric sweep (geometric, power, beta, and a few random
    Dirichlet draws) to obtain a solid starting point.

2️⃣  A long Adam‑style projected gradient descent on the probability simplex.
    The gradient is the sub‑gradient of the current maximal convolution entry;
    when several entries tie, their gradients are averaged.  An adaptive learning
    rate  (α₀/√t) together with Adam’s moment estimates gives a smoother and
    faster descent than the plain sub‑gradient method used before.

3️⃣  A final micro‑hill‑climb that moves extremely small amounts of mass between
    random indices; only improving moves are kept.  This fine‑tunes the solution
    after the Adam phase.

The routine respects the 1000 s budget, prints progress information and finally
returns the best sequence found as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two that is ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes the vector is already normalised to sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    conv = _conv_self(a, L=L_fft)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Returns a sub‑gradient of max(conv(a,a)) at the current a.
    If several shifts attain the maximum, their gradients are averaged.
    """
    n = a.shape[0]
    cur_max = conv.max()
    # indices where the maximum is attained (within tiny tolerance)
    active = np.where(conv >= cur_max - 1e-12)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # grad_i = 2 * a_{k-i}
        grad[idx] += 2.0 * a[k - idx]

    grad /= max(1, len(active))  # average if several active shifts
    return grad


# ----------------------------------------------------------------------
# Adam‑style projected descent on the simplex
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  L_fft: int,
                  deadline: float,
                  start_time: float,
                  best_score_initial: float,
                  best_seq_initial: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex.
    Stops when the time budget expires.
    """
    a = a0.copy()
    best_a = best_seq_initial.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.3               # initial learning‑rate multiplier
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max   # sum(a)=1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the maximum convolution entry
        g = _max_conv_subgrad(a, conv)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decaying step size (α₀ / √t)
        step = alpha0 / math.sqrt(it)

        # Adam update + projection
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # escape plateau if stuck for many iterations
        if it - last_improve > 5000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(start_a: np.ndarray,
                     n: int,
                     L_fft: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Extremely small mass transfers that are accepted only if they improve the true
    objective.  Repeated until the remaining time is exhausted.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        # move a tiny fraction of the source mass to the destination
        eps_move = a[src] * rng.random() * 1e-4
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={best_score:.6f}")

    return best_a, best_score

        if pow_sum == 0:

        pow_seq /= pow_sum

        if sc < best_score:
