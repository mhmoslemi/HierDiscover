    start_time = time.time()
    deadline = start_time + budget_s - 0.3

    c_vals = np.linspace(0.2, 3.0, 281) # step 0.01? maybe less to reduce runtime.

    best_seq = None
    best_score = float('inf')

    i_arr = np.arange(n, dtype=np.float64)

    for c in c_vals:

            break
        r = 1.0 - c / n
        if r <= 0:

        w = np.exp(i_arr * math.log(r))
        a = w / w.sum()
        # Compute objective using efficient fft
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # maybe print
            print(f"c={c:.4f} r={r:.6f} score={score:.6f}")

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. **Exponential sweep** –  Sequences of the form
       a_i ∝ r**i ,    r = 1 - c / n
   are generated for many values of the decay constant `c`.
   The exact objective (2·n·max(conv)/sum²) is evaluated via a
   single FFT per candidate and the best one is kept.

2. **Power‑law sweep** –  As a second, cheap family we try
       a_i ∝ (i+1)^(-p)  for a grid of `p`.  This occasionally finds
   a shape that the exponential family misses.

3. **Projected sub‑gradient descent** –  Starting from the best
   candidate, we directly minimise `max(conv(a,a))` on the simplex.
   For the current vector we locate the shift `k` where the convolution
   attains its maximum.  The sub‑gradient of `max(conv)` w.r.t. the
   entries is simply

        g_i = 2·a_{k−i}   (zero if the index is out of range).

   The update `a ← a - η·2·n·g` is followed by a Euclidean projection
   onto the probability simplex (non‑negative, sum = 1).  A modest,
   slowly decreasing learning rate is used and the exact objective is
   re‑evaluated every few iterations; the best vector seen is retained.

All work respects the given wall‑time budget (≈1000 s).  The final
sequence is normalised to sum = 1, clipped to non‑negative values and
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (stand‑alone, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s ** 2))

# ----------------------------------------------------------------------

    # ------------------------------------------------------------------
    # c controls the decay; we probe a wide range.
    c_vals = np.linspace(0.1, 4.0, 391)          # step ≈0.01
    for c in c_vals:
        if time.time() >= deadline:
            break
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        # compute the geometric vector safely (log‑domain for stability)
        log_r = math.log(r)
        w = np.exp(idx * log_r)
        a = w / w.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[exponential] c={c:.3f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2) Power‑law family sweep  a_i ∝ (i+1)^(-p)
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.0, 2.5, 126)   # 0 → uniform, larger → front‑heavy
    for p in p_vals:
        if time.time() >= deadline:
            break
        w = (idx + 1.0) ** (-p)
        a = w / w.sum()
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[power‑law] p={p:.3f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3) If a starter is supplied by the harness, try it as well
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    sc = _objective(starter)
                    if sc < best_score:
                        best_score = sc
                        best_seq = starter.copy()
                        print(f"[starter]   score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4) Projected sub‑gradient descent on the max‑convolution term
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)      # uniform fallback

    a = best_seq.copy()
    lr = 0.05                             # initial learning rate
    it = 0
    # we keep a very small buffer at the end for safety
    while time.time() < deadline - 0.5:
        it += 1
        # --- convolution and location of the maximal entry ----------
        L = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        k_max = int(conv.argmax())

        # --- sub‑gradient of max(conv) w.r.t a -----------------------
        # g_i = 2 * a_{k_max-i}   (zero when the partner index is outside)
        partner = k_max - np.arange(n)
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[partner[valid]]

        # gradient of the full objective (sum is kept equal to 1)
        grad = 2.0 * n * g

        # --- simple SGD step + simplex projection --------------------
        a = a - lr * grad
        a = _project_simplex(a)

        # --- learning‑rate schedule --------------------------------
        lr = max(lr * 0.9995, 5e-5)   # slowly decay

        # --- keep the best exact score seen -------------------------
        if it % 30 == 0:
            sc = _objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.2f}s] iter={it:6d}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Optional: report the final score using the provided evaluate_sequence
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Returning sequence of length {n} with final score {final_score:.6f}")
    except Exception:
        print(f"Returning sequence of length {n} (final internal score {best_score:.6f})")

    return [float(v) for v in best_seq]
