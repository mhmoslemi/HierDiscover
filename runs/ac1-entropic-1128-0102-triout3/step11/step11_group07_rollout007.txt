"""
Propose a non‑negative step‑function (a simplex vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation improves on the previous baseline by

* a much richer parametric sweep (geometric, power, beta, sinusoid,
  exponential families) with fine grids,
* a long projected Adam optimiser on a soft‑max approximation of the
  max‑convolution (many restarts, slower temperature decay, larger step),
* a pure sub‑gradient descent on the exact max‑convolution,
* optional LP refinement / greedy moves when SciPy is available,
* a final tiny hill‑climb polish.

All phases respect a 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# optional SciPy – used only for LP refinement (fallback is safe)
try:
    from scipy.optimize import linprog
    _has_scipy = True
except Exception:                     # pragma: no cover
    _has_scipy = False


# ----------------------------------------------------------------------
# Basic linear–algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑style optimisation with soft‑max smoothing of max(conv)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max smoothed objective.
    Returns the best vector seen and its score.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # start temperature – a bit higher to allow smoother gradients
    tau = 0.08
    tau_min = 1e-9
    tau_decay = 0.9997        # slower decay → smoother gradients longer
    step0 = 0.30

    while time.time() < deadline:
        it += 1
        # convolution
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # soft‑max weights

        # gradient via FFT: grad_j = 2 * Σ_i a_i w_{i+j}
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update & projection onto simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Pure sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(conv(a,a)).
    Uses a simple diminishing step size.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    step0 = 0.20               # initial step
    it = 0
    tol = 1e-9                 # tolerance for active set

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        active = (conv >= M * (1.0 - tol)).astype(np.float64)   # binary mask
        if active.sum() == 0:      # fallback (should not happen)
            active[np.argmax(conv)] = 1.0

        # sub‑gradient = 2 * a * active_rev   (linear conv)
        w_rev = active[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} step={step:.2e} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Active‑set linear‑program refinement (optional, SciPy dependent)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 6,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Iteratively improve a by solving a small LP on the active constraints."""
    if not _has_scipy:                     # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])
        K = active.size

        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(g)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(1, int(deadline - time.time()))})

        if not res.success:                     # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0:
            break
        g /= g.sum()

        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)

        if new_score < best_score:
            best_score = new_score
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP it={it} score={new_score:.6f}")
        else:
            break

    return a, best_score


# ----------------------------------------------------------------------
# Greedy guided move (same as original)
# ----------------------------------------------------------------------
def _guided_move(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 L_fft: int,
                 max_moves: int = 5000) -> tuple[np.ndarray, float]:
    """Transfer mass from the highest‑gradient index to the lowest‑gradient one."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        conv = _conv_self(a, L=L_fft)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]  # sub‑gradient of the active max‑term

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * 0.13
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] GuidedMove it={it:,} score={new_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polish)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Random tiny moves accepted only if they improve."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.075
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimizes ``evaluate_sequence``.
    The routine mixes:
      • an exhaustive parametric sweep (geometric, power, beta,
        sinusoid, exponential families),
      • many restarts of a smooth Adam optimiser,
      • a pure sub‑gradient descent,
      • optional LP refinement, greedy guided moves and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Fixed problem size (chosen to give a good speed / quality trade‑off)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)                     # for all FFT‑based convolutions

    # ------------------------------------------------------------------
    # Baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Very broad parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # ----- geometric families (including mixtures with uniform) -----
    eps_grid = np.logspace(-5, -0.2, 150)   # finer near 1
    lam_grid = (0.0, 0.2, 0.5, 0.8, 1.0)
    for eps in eps_grid:
        if time.time() >= deadline - 120:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq = _project_simplex(seq)   # ensure exact sum=1
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.1f} score={sc:.6f}")

    # ----- power families -----
    p_grid = np.linspace(-2.0, 3.0, 81)   # more extensive
    for p in p_grid:
        if time.time() >= deadline - 120:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # ----- beta‑type families (i.e. i^p * (n‑i)^q) -----
    p_vals = np.linspace(0.1, 5.0, 26)
    q_vals = np.linspace(0.1, 5.0, 26)
    for p in p_vals:
        if time.time() >= deadline - 120:
            break
        for q in q_vals:
            if time.time() >= deadline - 120:
                break
            beta_seq = np.power(idx + 1, p) * np.power(n - idx, q)
            s = beta_seq.sum()
            if s == 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")

    # ----- sinusoidal–power families -----
    pow_sin_vals = np.linspace(0.5, 4.0, 15)
    sin_term = np.sin(np.pi * (idx + 0.5) / n)          # 0 at edges, 1 at centre
    for p in pow_sin_vals:
        if time.time() >= deadline - 120:
            break
        sin_seq = np.power(sin_term, p)
        s = sin_seq.sum()
        if s == 0:
            continue
        sin_seq /= s
        sc = _eval_fast(sin_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = sin_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SinPower p={p:.2f} score={sc:.6f}")

    # ----- exponential decay families -----
    scale_vals = np.linspace(0.2 * n, 2.0 * n, 18)
    for scale in scale_vals:
        if time.time() >= deadline - 120:
            break
        exp_seq = np.exp(-idx / scale)
        s = exp_seq.sum()
        if s == 0:
            continue
        exp_seq /= s
        sc = _eval_fast(exp_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = exp_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp scale={scale:.0f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Long Adam‑style smoothed gradient descent with many restarts
    # ------------------------------------------------------------------
    # Reserve ~300 s for this phase; remaining budget will be kept for refinements.
    while time.time() < deadline - 250.0:
        # create a perturbed start point
        base = best_seq if np.random.rand() < 0.65 else uniform
        noise = 0.06 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 10.0, time.time() + 28.0)  # ~28 s per run
        cand_seq, cand_score = _adam_smooth_descent(start, sub_deadline,
                                                    start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new global best score={best_score:.6f}")

        if best_score <= 1.503:     # early exit if target crossed
            break

    # ------------------------------------------------------------------
    # 3️⃣  Pure sub‑gradient descent (non‑smooth) for fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0 and best_score > 1.503:
        sub_deadline = min(deadline - 10.0, time.time() + 30.0)
        cand_seq, cand_score = _subgradient_descent(best_seq,
                                                    sub_deadline,
                                                    start_time,
                                                    L_fft,
                                                    L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Optional LP refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0 and best_score > 1.503:
        best_seq, best_score = _lp_refine(best_seq,
                                          deadline,
                                          start_time,
                                          L_fft)

    # ------------------------------------------------------------------
    # 5️⃣  Guided mass‑transfer move
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0 and best_score > 1.503:
        best_seq, best_score = _guided_move(best_seq,
                                            deadline,
                                            start_time,
                                            L_fft)

    # ------------------------------------------------------------------
    # 6️⃣  Final tiny hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq,
                                           deadline,
                                           start_time,
                                           L_fft)

    # ------------------------------------------------------------------
    # Normalisation and output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
