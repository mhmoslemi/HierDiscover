def _optimize(a, deadline, start_time):
    a = a.copy()
    a = _project_simplex(a) # ensure sum=1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9

    t=0
    best_a = a.copy()
    best_score = _objective(a)
    while time.time() < deadline:
        t+=1
        conv = _conv_self(a)
        max_val = conv.max()
        # compute weights using softmax with temperature schedule
        # schedule temperature = 1.0 + (t/2000) maybe.
        beta_temp = min(20.0, 0.5 * math.log(t+1) + 1.0)  # gradually increase
        # shift for stability
        shift = max_val
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        w = w / w_sum
        # compute gradient = 2 * n * (dM/da)

def _adam_optimize(a0, deadline, start_time, top_k=3):
    a = a0.copy()
    # ensure on simplex
    a = _project_simplex(a)
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        # top_k indices
        idxs = np.argpartition(conv, -top_k)[-top_k:]

        # compute gradient
        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += 2.0 * a[k - i_range]
        grad /= top_k

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        # projection onto simplex
        a = _project_simplex(a_new)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} score={cur_score:.6f}")

    return best_a, best_score

"""
Search for a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)**2)

The evaluation is scale‑invariant, so we keep the vector on the
probability simplex (sum = 1).  The algorithm consists of three stages:

1. **Analytic families** – generate a handful of low‑dimensional
   shapes (power‑law decays, exponentials and the “tent‑with‑plateau”
   shapes used previously).  They are evaluated with a fast FFT‑based
   convolution and the best few are collected.

2. **Projected sub‑gradient descent** – starting from each of the
   top candidates we run a momentum sub‑gradient method on the exact
   non‑smooth objective.  The sub‑gradient is taken from the index (or
   a few indices) where the self‑convolution attains its maximum.
   After each step the vector is projected onto the probability simplex.

3. **Simulated‑annealing polish** – a light annealing phase swaps tiny
   mass packets between random positions; Metropolis acceptance with a
   fast exponential cooling schedule helps escape shallow local minima.

All heavy work uses FFTs so the search scales to n≈10⁵ and fits easily
into the 1000 s budget.  The returned object is a plain Python list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    Identical to evaluate_sequence but without defensive checks.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex { w ≥ 0, Σ w = 1 }.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “tent‑with‑plateau’’ shape (used in the reference solution).
    - p     : exponent of the side ramps (p > 0).
    - alpha : relative plateau width ∈ [0, 0.5]; alpha = 0 → pure tent,
              alpha = 0.5 → uniform distribution.
    The shape is normalised to sum 1.
    """
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_decay(n: int, p: float) -> np.ndarray:
    """
    Monotone decreasing power‑law shape: a[i] ∝ (1 - i/(n‑1)) ** p,
    normalised to sum 1.  p ≥ 0 (p=0 gives the uniform distribution).
    """
    if p < 0.0:
        p = 0.0
    i = np.arange(n, dtype=float)
    w = (1.0 - i / (n - 1)) ** p
    total = w.sum()
    if total > 0.0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _make_exponential_decay(n: int, lam: float) -> np.ndarray:
    """
    Monotone decreasing exponential shape: a[i] ∝ exp(-lam * i),
    normalised to sum 1.  lam ≥ 0 (lam=0 gives the uniform distribution).
    """
    if lam < 0.0:
        lam = 0.0
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    total = w.sum()
    if total > 0.0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (momentum version)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         top_k: int = 1) -> tuple[np.ndarray, float]:
    """
    Momentum sub‑gradient descent on the exact objective.
    The gradient is built from the `top_k` largest convolution entries.
    Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = _objective(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices of the largest convolution entries
        if top_k == 1:
            ks = (int(conv.argmax()),)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        grad /= max(1, top_k)  # average contribution

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project onto the simplex (keeps sum = 1)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Move tiny mass packets between random positions.
    Metropolis acceptance with a fast exponential cooling schedule.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)   # fast exponential cool

        src = rng.choice(n, p=a)           # pick source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

            _push_candidate(s, a)

    # ---- monotone power‑law decays
    p_grid2 = np.linspace(0.0, 5.0, 26)        # 0 … 5
    for p in p_grid2:
        a = _make_power_decay(n, p)
        s = _objective(a)
        _push_candidate(s, a)

    # ---- monotone exponential decays (log‑spaced λ)
    lam_grid = np.logspace(-4, 1, 30)         # 1e‑4 … 10
    for lam in lam_grid:
        a = _make_exponential_decay(n, lam)
        s = _objective(a)
        _push_candidate(s, a)

    # ---- a few random draws to diversify
    rng = np.random.default_rng()
    for _ in range(200):
        if time.time() >= deadline:
            break
        if rng.random() < 0.5:
            p = rng.uniform(0.0, 5.0)
            a = _make_power_decay(n, p)
        else:
            lam = 10 ** rng.uniform(-4, 1)
            a = _make_exponential_decay(n, lam)
        s = _objective(a)
        _push_candidate(s, a)

    print(f"[{time.time() - start_time:.2f}s] Grid search best score = "
          f"{top_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional: incorporate a globally stored starter if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = "
                      f"{cur:.6f}")
                _push_candidate(cur, starter)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Refine the best candidates with sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # Allocate the remaining budget roughly evenly among the pool
        per_candidate = remaining * 0.80 / len(top_candidates)

        best_overall_seq = None
        best_overall_score = float("inf")

        for idx, (init_score, init_seq) in enumerate(top_candidates, 1):
            sub_deadline = time.time() + per_candidate
            refined_seq, refined_score = _subgradient_descent(
                init_seq, sub_deadline, start_time, top_k=1
            )
            if refined_score < best_overall_score:
                best_overall_score = refined_score
                best_overall_seq = refined_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Candidate {idx} refined "
                      f"score = {refined_score:.6f}")

            if time.time() >= deadline:
                break

        # fall‑back in the unlikely event that no candidate improved
        if best_overall_seq is None:
            best_overall_seq = top_candidates[0][1].copy()
            best_overall_score = top_candidates[0][0]

    else:
        # not enough time – just take the best grid point
        best_overall_seq = top_candidates[0][1].copy()
        best_overall_score = top_candidates[0][0]

    # ------------------------------------------------------------------
    # 4) Light annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_overall_seq, deadline, start_time)
        if anneal_score < best_overall_score:
            best_overall_seq = anneal_seq.copy()
            best_overall_score = anneal_score
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve "
                  f"score = {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_overall_seq = np.maximum(best_overall_seq, 0.0)
    total = best_overall_seq.sum()
    if total > 0.0:
        best_overall_seq = best_overall_seq / total
    else:
        best_overall_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ "
          f"{best_overall_score:.6f}")
    return [float(v) for v in best_overall_seq]
