"""
Propose a step‑function with the smallest possible
evaluate_sequence() value.

Algorithm (≈1000 s)
-------------------
1. Choose a large power‑of‑two length `n = 2**18 = 262 144`.  The
   discretisation error of the continuous optimum (≈ 1.500…) is then
   O(1/n) ≈ 4·10⁻⁶, so the bound is essentially limited by the shape
   of the function rather than by the grid.

2. Initialise a small pool of candidate shapes.
   * A 2‑parameter family “power‑ramp with plateau’’
     (exponent `p` controls side curvature, `α` the relative width of a
     flat centre) is sampled on a coarse grid.
   * A few random draws are added for diversity.
   * If a global starter `height_sequence_1` exists it is interpolated
     to the current length and used as an additional seed.

3. The best candidate from the pool is refined by **projected
   sub‑gradient descent** on the exact non‑smooth objective.
   * At each iteration the index `k` where the self‑convolution attains
     its maximum is located.
   * The sub‑gradient of `max(conv)` is the gradient of the single
     convolution entry `conv[k]`.  A momentum term (β = 0.9) and a
     decreasing step size `ηₜ = η₀ / √t` are used.
   * After the update the vector is projected onto the probability
     simplex (non‑negative, sum = 1).

4. When the remaining time is small a cheap **simulated‑annealing**
   polish moves tiny packets of mass between random positions
   (Metropolis acceptance).  The annealing temperature decays
   exponentially and the move is accepted if it improves the objective
   or with probability `exp(-Δ/T)`.

5. The final vector is clipped, normalised to sum = 1 and returned as a
   plain Python list.  All heavy work uses FFT‑based convolution, which
   is fast for the chosen size.

The whole search respects the 1000 s budget (a safety margin of
0.5 s is kept).  The implementation contains no closures or lambdas,
all helpers are top‑level functions, and only NumPy (and the already
imported `evaluate_sequence`) is used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns length 2·len(a)‑1 array.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2).  No defensive checks."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto { w ≥ 0, Σ w = 1 }.
    O(n log n) implementation (Wang & Carreira‑Perpiñán).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generator: symmetric power‑ramp with optional flat plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Return a symmetric shape of length n.
    * p     – exponent of the side ramps (p > 0).
    * alpha – relative plateau width ∈[0,0.5]; 0 → pure tent,
              0.5 → uniform distribution.
    The shape is normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    # keep plateau length even to preserve perfect symmetry
    if plateau_len % 2 == 1:
        plateau_len = max(0, plateau_len - 1)
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        # side ramp values: increase from 0 at the very edge to 1 at the plateau
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # monotonically increasing 0..1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # fix possible rounding mismatches
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent until `deadline`.
    Returns best iterate seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum factor
    eta0 = 0.5          # initial step size coefficient
    it = 0
    last_improve = start_time

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"sub‑grad iter={it:,} score={cur_score:.6f}")

        # index where convolution attains its maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the maximal entry
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + adaptive step
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / math.sqrt(it)
        a -= eta * momentum

        # project back onto the simplex
        a = _project_simplex(a)

        # occasional random perturbation to escape stagnation
        if it % 50000 == 0:
            # small Gaussian noise (preserves sum after projection)
            noise = np.random.default_rng().normal(scale=1e-4, size=n)
            a = _project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions, Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # Number of annealing attempts – capped by remaining time.
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # choose source proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction (up to 20 %) of src mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

        p = rng.uniform(0.1, 3.0)
        alpha = rng.uniform(0.0, 0.5)
        a = _make_shape(n, p, alpha)
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        if len(top_candidates) < top_k:
            top_candidates.append((score, a.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if score < top_candidates[-1][0]:
                top_candidates[-1] = (score, a.copy())
                top_candidates.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] "
          f"Best grid/random score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Incorporate a globally supplied starter, if it exists.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # interpolate to the current length
            if starter.size != n:
                x_old = np.linspace(0, 1, starter.size)
                x_new = np.linspace(0, 1, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] "
                      f"global starter score = {cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    # push into the candidate pool
                    if len(top_candidates) < top_k:
                        top_candidates.append((cur, starter.copy()))
                        top_candidates.sort(key=lambda x: x[0])
                    else:
                        if cur < top_candidates[-1][0]:
                            top_candidates[-1] = (cur, starter.copy())
                            top_candidates.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) Local refinement – sub‑gradient descent on the best candidate.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a few seconds for polishing
        # allocate the majority of remaining time to descent
        sub_deadline = time.time() + 0.80 * remaining
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = "
                  f"{refined_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light annealing polish (if any time left).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve "
                  f"score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ "
          f"{best_score:.6f}")
    return [float(v) for v in best_seq]
