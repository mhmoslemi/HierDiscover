def anneal_refine(a, conv, deadline):
    n = a.shape[0]
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max
    best_a = cur_a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        src = rng.choice(n, p=cur_a)   # source weighted by mass
        dst = rng.integers(n)
        if src == dst or cur_a[src] < 1e-15:
            continue
        # delta
        delta = cur_a[src] * rng.random() * 0.2   # up to 20%
        if delta <= 0:
            continue
        # compute new conv
        conv_new = cur_conv.copy()
        # update slices
        conv_new[src:src+n] -= 2*delta * cur_a
        conv_new[dst:dst+n] += 2*delta * cur_a
        conv_new[2*src] += delta*delta
        conv_new[2*dst] += delta*delta
        conv_new[src+dst] -= 2*delta*delta
        
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score
        # temperature schedule:
        T = max(0.001 * (0.99**it), 1e-12)  # start at 0.001, decay
        if delta_score < 0 or math.exp(-delta_score / T) > rng.random():
            # accept
            cur_a[src] -= delta
            cur_a[dst] += delta
            cur_conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                # optional print
        it += 1
        if it % 10000 == 0:
            print(f"[{time.time() - start_time:.2f}s] SA it={it:,} best={best_score:.6f}")
    return best_a, best_score

def _anneal(a0, conv0, deadline, start_time):

    conv = conv0.copy()
    n = a.shape[0]
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    max_frac = 0.2
    # temperature schedule parameters
    T0 = 0.001
    while time.time() < deadline:
        # Choose move
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * max_frac
        # compute new conv
        conv_new = conv.copy()
        conv_new[src:src+n] -= 2 * delta * a
        conv_new[dst:dst+n] += 2 * delta * a
        conv_new[2*src] += delta * delta
        conv_new[2*dst] += delta * delta
        conv_new[src+dst] -= 2 * delta * delta
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        # temperature cooling
        T = max(T0 * (0.999 ** it), 1e-12)

            a[dst] += delta

            cur_score = new_score

                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={best_score:.6f}")
        it += 1
        if it % 50000 == 0:
            # optional early break if not improving
            pass
    return best_a, best_score

"""
A fast, multi‑stage search for a low‑score step function.

1.  Generate many cheap candidates from a few parametric families
    (arcsine‑like, beta densities, mixtures with uniform).  The best
    candidate is kept.
2.  Refine the best candidate with a projected sub‑gradient (Adam‑style)
    descent on the exact nonsmooth objective.
3.  Perform a long simulated‑annealing “mass‑transfer” polish where a
    small amount of probability mass is moved between two positions.
    Convolution updates are done in O(n) time by exploiting the analytic
    effect of a transfer on the self‑convolution.
4.  Return a non‑negative list (the sum is normalised to 1 – the verifier
    is scale‑invariant).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a⭑a using FFT (length 2·len(a)‑1).
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the verifier:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)

    The sequence is assumed to be a 1‑D NumPy array with non‑negative entries.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    # after normalising the sum to 1 the denominator disappears:
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent (nonsmooth objective)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on the exact objective.
    Returns the best iterate seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05          # initial learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution and locate a worst shift
        conv = _conv_self(a)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) at shift k
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update on the gradient (projected afterwards)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / math.sqrt(it)  # slowly decreasing base step
        a_tmp = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a_tmp)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam iter={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polish (mass‑transfer moves)
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray,
            conv0: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of the mass from one index to another.
    Convolution is updated in O(n) using the analytical formula.
    """
    a = a0.copy()
    conv = conv0.copy()
    n = a.shape[0]
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    max_frac = 0.20            # at most 20 % of the source mass per move
    T0 = 1e-3                  # initial temperature for Metropolis acceptance

    while time.time() < deadline:
        src = rng.choice(n, p=a)            # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # --- compute the updated convolution ---------------------------------
        conv_new = conv.copy()
        # contribution of moving delta away from src
        conv_new[src:src + n] -= 2.0 * delta * a
        # contribution of adding delta to dst
        conv_new[dst:dst + n] += 2.0 * delta * a
        # quadratic term of the two point masses
        conv_new[2 * src] += delta * delta
        conv_new[2 * dst] += delta * delta
        conv_new[src + dst] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        dE = new_score - cur_score

        # temperature schedule (geometric cooling)
        T = max(T0 * (0.9999 ** it), 1e-12)

        if dE < 0.0 or math.exp(-dE / T) > rng.random():
            # accept the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={best_score:.6f}")

        it += 1
        if it % 100000 == 0:
            # periodic progress report
            print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} cur={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Fixed problem size (the reference solution used n = 4096)
    # --------------------------------------------------------------
    n = 4096
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑point of each bin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1) Broad sweep over several parametric families
    # --------------------------------------------------------------
    # (a) arcsine‑like family   a_i ∝ (x·(1−x)+ε)^‑α
    alphas = np.linspace(0.30, 0.70, 81)          # step ≈0.005
    epsilons = [0.0, 1e-12, 1e-10, 1e-8, 1e-6]
    for eps in epsilons:
        base = x * (1.0 - x) + eps
        for alpha in alphas:
            w = base ** (-alpha)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] arcsine α={alpha:.4f} eps={eps:g} score={score:.6f}")

    # (b) beta‑density family   a_i ∝ x^{p−1} (1−x)^{q−1}
    pq_vals = np.linspace(0.2, 0.8, 13)          # 13×13 = 169 candidates
    for p in pq_vals:
        for q in pq_vals:
            w = (np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0) + 1e-12)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] beta p={p:.3f} q={q:.3f} score={score:.6f}")

    # (c) mixture of arcsine‑type with uniform component
    mix_lambdas = np.linspace(0.0, 1.0, 21)      # 0 = uniform, 1 = pure arcsine
    arcsine_base = ((x * (1.0 - x) + 1e-8) ** (-0.5))
    uniform_base = np.ones_like(x)
    for lam in mix_lambdas:
        w = lam * arcsine_base + (1.0 - lam) * uniform_base
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] mix λ={lam:.2f} score={score:.6f}")

    # (d) the global starter, if it exists and matches the problem size
    try:
        starter = globals().get("height_sequence_1")
        if starter is not None:
            starter = np.asarray(starter, dtype=float)
            if starter.ndim == 1 and starter.size == n:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    cur_score = _objective(starter)
                    print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
    except Exception:
        pass

    # Safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 2) Refine the best candidate – Adam‑style projected descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        sub_deadline = time.time() + 0.5 * remaining
        refined_seq, refined_score = _adam_descent(best_seq, sub_deadline, start_time)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam refined score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated‑annealing polish (most of the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        conv = _conv_self(best_seq)          # current convolution
        anneal_seq, anneal_score = _anneal(
            best_seq, conv, deadline, start_time
        )
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improvement score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
