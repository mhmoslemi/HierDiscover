```python
"""
Implementation of a hybrid optimization for the step‑function problem.
The algorithm consists of:

1. A quick parametric sweep over several families (geometric, power,
   beta) to get a decent starting point.
2. A long projected Adam‑style sub‑gradient descent on the simplex.
   The sub‑gradient is taken at the index where the auto‑convolution attains
   its maximum.  Stagnation triggers a tiny random perturbation.
3. An occasional linear‑program “boost” step: solve a LP that maximises the
   total mass under the current convolution bound, mix the solution with the
   current vector and keep the best mixture.
4. A final microscopic hill‑climb that transfers vanishing amounts of mass
   between random coordinates.

All heavy work uses FFT‑based convolutions (O(n log n)).  The routine respects
the 1000 s time budget and returns a list of Python floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real‐valued)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    # a is already projected onto the simplex → sum≈1 (tiny numerical error)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
# Linear‑programming “boost” step
# ----------------------------------------------------------------------
def _build_conv_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build matrix A (shape (2n‑1,n)) such that (a * g)_k = (A @ g)_k.
    For each column j we place the whole vector a starting at row j.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_boost(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(g)
        subject to A @ g <= M * 1
                   g >= 0
    where A is the convolution constraint matrix built from a.
    Returns a feasible g (scaled to have the same total mass as a) or None.
    """
    n = a.shape[0]
    A = _build_conv_constraint_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -sum(g)  ⇔ maximise sum(g)
    bounds = [(0.0, None) for _ in range(n)]
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    if not res.success:
        return None
    g = res.x
    total_g = g.sum()
    if total_g == 0:
        return None
    # Scale g so that its total mass equals that of a (which is 1 after projection)
    g = g * (a.sum() / total_g)
    return g

# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent (main phase)
# ----------------------------------------------------------------------
def _adam_subgradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a))   (a lives on the simplex).
    An LP‑boost is attempted every 15 000 iterations.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    last_improve = 0
    L_fft = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1
        # current convolution and objective
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient at the active shift
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update on the (unprojected) vector
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # Escape from long plateaus
        if it - last_improve > 8000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

        # Periodic LP‑boost
        if it % 15000 == 0:
            g = _lp_boost(a, cur_max)
            if g is not None:
                # try a few mixing ratios
                for t in (0.2, 0.4, 0.6, 0.8):
                    mixed = (1.0 - t) * a + t * g
                    mixed = _project_simplex(mixed)
                    mixed_score = _eval_fast(mixed, L_fft)
                    if mixed_score < best_score:
                        best_score = mixed_score
                        best_a = mixed.copy()
                        a = mixed.copy()

"""
Hybrid optimisation for the step‑function problem.

The search consists of three stages:

1️⃣  A fast parametric sweep (geometric, power, beta families) provides a
    solid initial point.

2️⃣  A long projected Adam‑style sub‑gradient descent on the simplex.
    The sub‑gradient is taken at the index where the self‑convolution attains
    its maximum.  Stagnation triggers a tiny Gaussian perturbation.  Every
    15 000 iterations a small linear‑program “boost’’ is performed:
        maximise   Σ g_j
        subject to (a * g)_k ≤ M   (M = current max‑convolution)
                   g_j ≥ 0 .
    The resulting vector g is mixed with the current a and the best mixture
    is kept.

3️⃣  A final microscopic hill‑climb transfers ε≈10⁻⁴·a_src between random
    coordinates and accepts only improving moves.

All heavy work uses FFT‑based convolutions (O(n log n)).  The routine respects
the 1000 s budget and returns a plain list of Python floats ready for
`evaluate_sequence`.
"""

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and Σ w = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1.
    """
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# Linear‑programming boost step
# ----------------------------------------------------------------------
def _build_conv_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build matrix A (shape (2n‑1,n)) such that (a * g)_k = (A @ g)_k.
    For each column j we place the whole vector a starting at row j.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A

def _lp_boost(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to   A @ g ≤ M·1
                     g ≥ 0
    where A encodes the convolution constraints built from a.
    Returns a feasible g (scaled to have the same total mass as a) or None.
    """
    n = a.shape[0]
    A = _build_conv_constraint_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -Σg  ⇔ maximise Σg
    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 10})
    except Exception:
        return None

        return None

    g = res.x

    if total_g == 0.0:

    g = g * (a.sum() / total_g)
    return g
