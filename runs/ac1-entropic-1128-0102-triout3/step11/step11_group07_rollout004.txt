"""
Propose‑candidate for the step‑function optimisation.

The search proceeds in three stages:

1. **Fast parametric sweep** – exhaustive grids over geometric,
   power‐law and beta families (the three most useful low‑dimensional
   families).  The best sequence from the sweep seeds later stages.

2. **Long Adam‑style smoothed gradient descent** – the non‑smooth
   max‑convolution is approximated by a soft‑max (temperature τ that
   decays slowly).  Each iteration uses an exact FFT‑based gradient,
   Adam momentum updates and a Euclidean projection onto the simplex.
   Several restarts (random perturbations of the current best) are run
   until only ~200 s of budget are left.

3. **Simulated‑annealing polishing** – tiny random transfers of mass
   between two positions are accepted either if they improve the score
   or with a Metropolis probability that slowly cools.  This phase
   explores the neighbourhood of the Adam solution and usually squeezes
   a few extra points.

All computations use FFTs for O(n log n) convolution, the sequence length
is fixed to 4096 (the original benchmark size) and the final vector is
normalised to sum = 1 before being returned (the evaluator rescales anyway).

The routine is fully deterministic given the ``seed`` argument and respects
the 1000 s time budget.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Basic linear‑algebra utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a * a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Fast objective (assumes the sum is ≈ 1)
# --------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Returns 2·n·max(conv(a,a)), i.e. the objective 2·n·max_b/(sum_a**2)
    because the sum of a is (practically) 1.
    """
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# Adam‑style smoothed gradient descent
# --------------------------------------------------------------

    The temperature τ is annealed to a very small value,
    yielding an increasingly accurate gradient of the true max.
    """

    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    # schedule parameters
    tau = 0.1            # start temperature
    tau_min = 1e-9
    tau_decay = 0.9997   # very slow decay -> smoother gradients for longer
    step0 = 0.12

    it = 0
    while time.time() < deadline:
        it += 1

        # ----------- convolution and soft‑max weights -------------
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        # stabilised soft‑max
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # ----------- gradient w.r.t. a (FFT‑based) ---------------
        # grad_j = 2 * Σ_i a_i * w_{i+j}
        w_rev = w[::-1]                       # because conv is symmetric
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        # the term for index j lives at index (2*n-2 - j) in the full array
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # ------------------ Adam update -------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)                # keep on the simplex

        # ------------------- evaluate ---------------------------
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} "
                  f"score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# --------------------------------------------------------------
# Simulated‑annealing polishing
# --------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap SA that repeatedly transfers a random fraction of mass
    from a source index to a destination index.  Moves are accepted if
    they improve the objective or with a Metropolis probability that
    slowly cools.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # temperature schedule – start relatively high to escape shallow basins
    temp = 0.05
    temp_min = 1e-7
    decay = 0.99995

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        # pick source proportional to its mass (so we usually move something)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # amount to move – up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        proposal = a.copy()
        proposal[src] -= eps
        proposal[dst] += eps
        # sum is preserved, projection only needed for numerical safety
        proposal = _project_simplex(proposal)

        new_score = _eval_fast(proposal, L_fft)
        delta = new_score - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            a = proposal
            if new_score < best_score:
                best_score = new_score
                best_a = proposal.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} temp={temp:.2e} "
                      f"score={new_score:.6f}")

        # anneal temperature
        temp = max(temp_min, temp * decay)

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (length 4096) that minimises
    ``evaluate_sequence``.  The algorithm combines a fast parametric
    sweep, a long Adam‑style smoothed gradient descent with many restarts,
    and a final simulated‑annealing polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Problem size and FFT lengths
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(4 * n)                  # generous padding for gradient

    # ------------------------------------------------------------------
    # Baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric, power‑law, beta)
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # ----- geometric family -----
    eps_grid = np.logspace(-6, -0.2, 200)               # eps = 1‑r
    for eps in eps_grid:
        if time.time() >= deadline - 180:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        seq = geo / geo_sum
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

    # ----- power‑law family -----
    p_grid = np.linspace(-2.0, 2.0, 240)
    for p in p_grid:
        if time.time() >= deadline - 180:
            break
        pow_seq = np.power(idx + 1.0, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0:
            continue
        seq = pow_seq / s
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ----- beta‑type family (i+1)^p (n-i)^q -----
    beta_vals = np.linspace(0.0, 3.0, 30)
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 180:
                break
            beta_seq = (np.power(idx + 1.0, p) *
                        np.power(n - idx, q))
            s = beta_seq.sum()
            if s == 0:
                continue
            seq = beta_seq / s
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")

    # ----- A simple sine‑window (just for variety) -----
    sin_seq = np.sin(np.pi * (idx + 0.5) / n) ** 2
    sin_seq = sin_seq / sin_seq.sum()
    sc = _eval_fast(sin_seq, L_fft)
    if sc < best_score:
        best_score, best_seq = sc, sin_seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Sine window score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Long Adam‑style smoothed gradient descent with restarts
    # ------------------------------------------------------------------
    # Reserve ~200 s for the final SA phase; use the rest for Adam.
    while time.time() < deadline - 210:
        # random perturbation of the current best (occasionally from uniform)
        base = best_seq if np.random.rand() < 0.8 else uniform
        noise = 0.03 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 18.0)  # ~18 s per run
        cand_seq, cand_score = _adam_smooth_descent(start, sub_deadline,
                                                    start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new global best score={best_score:.6f}")

        # early exit if we have already crossed the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polishing (uses the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    deadline,
                                                    start_time,
                                                    L_fft)
        print(f"[{time.time() - start_time:.2f}s] SA finished best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalisation and final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
