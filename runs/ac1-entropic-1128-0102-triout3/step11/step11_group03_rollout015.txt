        n = 4096
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        # parameters chosen from the earlier sweep (good enough to start)
        alpha = 0.38
        eps = 1e-6
        w = (x * (1.0 - x) + eps) ** (-alpha)
        p = w / w.sum()
        print(f"[{time.time() - start_time:.2f}s] No starter – generated arcsine start (n={n})")

    # --------------------------------------------------------------
    # 2) Initial convolution and best‑so‑far bookkeeping
    # --------------------------------------------------------------
    conv = _conv_self(p)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max           # because Σp = 1
    best_p = p.copy()
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    rng = np.random.default_rng()
    iteration = 0
    last_improve_iter = 0

    # --------------------------------------------------------------
    # 3) Hill‑climbing with mass‑transfer moves
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # locate the current worst shift
        k = int(conv.argmax())
        cur_max = conv.max()

        # indices that influence conv[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # contributions p[i] * p[k-i]; pick the biggest contributor as source
        contrib = p[idx] * p[k - idx]
        src = idx[int(np.argmax(contrib))]

        # choose a destination among the currently smallest entries
        # (avoid the case dst == src)
        inv_weights = 1.0 - p
        total_weight = inv_weights.sum()
        if total_weight <= 0:
            inv_weights = np.ones_like(p)
            total_weight = inv_weights.sum()
        prob = inv_weights / total_weight
        dst = rng.choice(n, p=prob)
        if dst == src:
            continue

        # step size – a decreasing fraction of the source mass
        elapsed = time.time() - start_time
        remain = deadline - time.time()
        # gradually shrink the maximal fraction from 0.2 to 0.01
        max_frac = max(0.01, 0.20 * (1.0 - elapsed / budget_s))
        epsilon = p[src] * rng.random() * max_frac
        if epsilon <= 0.0:
            continue

        # --------------------------------------------------
        # backup current state (cheap copies – O(n))
        # --------------------------------------------------
        p_old = p.copy()
        conv_old = conv.copy()

        # --------------------------------------------------
        # exact update of the convolution (derived analytically)
        # --------------------------------------------------
        # subtract the contribution of the removed mass
        conv[src:src + n] -= 2.0 * epsilon * p
        # add the contribution of the newly placed mass
        conv[dst:dst + n] += 2.0 * epsilon * p
        # quadratic correction terms (ε²)
        conv[2 * src] += epsilon * epsilon
        conv[2 * dst] += epsilon * epsilon
        conv[src + dst] -= 2.0 * epsilon * epsilon

        # update the probability vector
        p[src] -= epsilon
        p[dst] += epsilon

        # safeguard against tiny negative values due to rounding
        if p[src] < 0.0:
            p[src] = 0.0

        # --------------------------------------------------
        # evaluate the new maximal convolution value
        # --------------------------------------------------
        new_max = conv.max()

        # accept only if the peak decreased (strict improvement)
        if new_max < cur_max - 1e-16:
            # successful move – keep the new state
            cur_max = new_max
            if 2.0 * n * cur_max < best_score:
                best_score = 2.0 * n * cur_max
                best_p = p.copy()
                last_improve_iter = iteration
                print(
                    f"[{time.time() - start_time:.2f}s] iter={iteration:,} "
                    f"new best score={best_score:.6f}"
                )
        else:
            # revert to the previous state
            p = p_old
            conv = conv_old

        # --------------------------------------------------
        # Periodically re‑normalise and recompute the convolution
        # (guards against drift from many tiny updates)
        # --------------------------------------------------
        if iteration % 5000 == 0:
            s = p.sum()
            if abs(s - 1.0) > 1e-12:
                p /= s
                conv = _conv_self(p)

        # optional early stop if no improvement for a long stretch
        if iteration - last_improve_iter > 200_000:
            # re‑initialise from the current best with a tiny random perturbation
            p = best_p.copy()
            # add a tiny jitter (≤0.5 % of each entry) to escape possible plateau
            jitter = 0.005 * p * (rng.random(p.shape) - 0.5)
            p = np.maximum(p + jitter, 0.0)
            p /= p.sum()
            conv = _conv_self(p)
            last_improve_iter = iteration
            print(
                f"[{time.time() - start_time:.2f}s] plateau – re‑jittered at iter {iteration}"
            )

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_p = np.maximum(best_p, 0.0)
    total = best_p.sum()
    if total > 0:
        best_p = best_p / total
    else:
        best_p = np.full(n, 1.0 / n)

    # The verifier is scale‑invariant; a probability vector is fine.
    return [float(v) for v in best_p]
