```
def _smooth_descent_adam(a0, deadline, start_time):
    a = a0.copy()
    loga = np.log(np.maximum(a, 1e-30))
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    alpha_schedule = np.geomspace(0.5, 2000.0, num=50)  # smoothness parameter
    best_score = _eval_fast(a, L_self)
    best_a = a.copy()
    for alpha in alpha_schedule:
        while time.time() < deadline - 0.2 and it < max_iters_per_alpha:
            # compute convolution via FFT
            A = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(A * A, n=L_self)[:2*n-1]
            # compute softmax weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)
            w /= w.sum()
            # grad = 2 * (w * a_rev) convolution
            a_rev = a[::-1]
            W = np.fft.rfft(w, n=L_grad)
            rev_f = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(W * rev_f, n=L_grad)[:3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]
            # Update loga using Adam
            it += 1
            m = beta1 * m + (1-beta1) * grad
            v = beta2 * v + (1-beta2) * (grad**2)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            eta = 0.05 / (math.sqrt(it) * max(1.0, math.sqrt(alpha))) # adapt step size
            loga = loga - eta * m_hat / (np.sqrt(v_hat) + eps)
            # recompute a via softmax of loga to enforce simplex
            a = np.exp(loga - np.max(loga))
            a = a / a.sum()
            # occasionally evaluate real objective
            if it % 500 == 0:
                score = _eval_fast(a, L_self)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam alpha={alpha:.2f} it={it} score={score:.6f}")
                # maybe early break if no improvement
        if time.time() >= deadline - 0.2:
            break
    return best_a, best_score
```

"""
Proposes a non‑negative step‑function (list of heights) that minimises
2·n·max(conv(a,a)) / (sum(a)**2) for n=4096.

The algorithm is a multi‑stage stochastic‑gradient search:
1. Load the best known sequence (if any) and normalise it.
2. Run several *Adam* mirror‑descent phases on a smooth approximation of the
   max‑convolution (soft‑max with a temperature that is increased geometrically).
   This quickly flattens the convolution profile.
3. Refine the result with a projected sub‑gradient descent that uses the exact
   max‑convolution and a simplex projection.
4. Apply a greedy hill‑climbing polishing step that transfers a small amount of
   mass between random positions and keeps improving moves.
5. Repeat the whole pipeline from random Dirichlet starts while time permits,
   keeping the best sequence seen.
The code respects the external 1000 s wall‑clock budget and prints progress.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L_self: int) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * a.shape[0] - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

def _smooth_descent_adam(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         L_grad: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth (soft‑max) approximation of the max‑convolution.
    Uses Adam on the log‑domain (exponentiated‑gradient) which automatically
    stays on the simplex.
    """
    a = a0.copy()
    loga = np.log(np.maximum(a, 1e-30))
    m = np.zeros_like(a)   # 1st moment
    v = np.zeros_like(a)   # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)
    it = 0
    # schedule for the soft‑max temperature
    alphas = np.geomspace(0.5, 4000.0, num=55)   # more aggressive final smoothing
    for alpha in alphas:
        # inner loop
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1
            # --- 1) self‑convolution ---
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # --- 2) soft‑max weights w_k ∝ exp(alpha * conv_k) ---
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stabilised
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # --- 3) gradient of smooth‑max (2 * (w ⋆ a_rev)) ---
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # --- 4) Adam update on log‑probabilities ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step size – decays with sqrt(it) and with sqrt(alpha)
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            loga -= eta * m_hat / (np.sqrt(v_hat) + eps)

            # projection back onto the simplex (soft‑max)
            a = np.exp(loga - np.max(loga))
            a /= a.sum()

            # occasional exact evaluation
            if it % 1000 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.1f} it={it:,} score={cur_score:.6f}")
        if time.time() >= deadline - 0.2:
            break
    return best_a, best_score

def _subgradient_exact(a0: np.ndarray,
                       n: int,
                       L_self: int,
                       L_grad: int,
                       deadline: float,
                       start_time: float,
                       max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the true objective.
    In each iteration the sub‑gradient is the average of the gradients of all
    active (max‑attaining) convolution entries.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)
    it = 0
    while time.time() < deadline - 0.2 and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        # active set – any index within a small relative tolerance
        rel_tol = 1e-3   # 0.1 % of the maximum
        active = np.where(conv >= cur_max * (1.0 - rel_tol))[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)
        # construct weight vector that averages over the active set
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * ( w ⋆ a_rev )
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # simple step‑size decreasing with √t
        step = 0.25 / math.sqrt(it)
        a -= step * grad
        a = _project_simplex(a)

        if it % 500 == 0:
            cur_score = _eval_fast(a, L_self)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
    return best_a, best_score

def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_seq: np.ndarray,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb: repeatedly move a random fraction of mass
    from a source index (chosen proportionally to its mass) to a random
    destination and keep the move if it improves the exact objective.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()
    # gradually shrink the maximal transfer fraction
    max_frac = 0.35
    while time.time() < deadline:
        src = rng.choice(n, p=a)               # source weighted by current mass
        dst = rng.integers(0, n)               # uniform destination
        if src == dst or a[src] < 1e-15:
            continue
        # transfer a fraction that slowly decays with time
        frac = max_frac * rng.random()
        eps = a[src] * frac
        if eps <= 0.0:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)
        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb new best={best_score:.6f}")
        # slowly shrink max_frac so that final moves become fine‑grained
        max_frac = max(0.01, max_frac * 0.9999)
    return best_seq, best_score

# ----------------------------------------------------------------------
def _subgradient_exact(a0: np.ndarray,
                       n: int,
                       L_self: int,
                       L_grad: int,
                       deadline: float,
                       start_time: float,
                       max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the true objective.
    The sub‑gradient is the average of the gradients of all (approximately)
    active max‑convolution entries.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)
    it = 0

    while time.time() < deadline - 0.2 and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()

        # active set: indices whose value is within 0.1 % of the maximum
        rel_tol = 1e-3
        active = np.where(conv >= cur_max * (1.0 - rel_tol))[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2·(w ⋆ a_rev)
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # step size ∝ 1/√t
        step = 0.25 / math.sqrt(it)
        a -= step * grad
        a = _project_simplex(a)

        if it % 500 == 0:
            cur_score = _eval_fast(a, L_self)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : Greedy hill‑climb polishing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_seq: np.ndarray,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: move a random fraction of mass from a source index
    (chosen proportionally to its height) to a random destination and keep the
    move only if it improves the exact objective.  The maximal transfer
    fraction shrinks slowly so the search becomes finer towards the end.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()
    max_frac = 0.35

    while time.time() < deadline:
        src = rng.choice(n, p=a)               # source weighted by current mass
        dst = rng.integers(0, n)               # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        # transfer a random fraction of the source mass
        eps = a[src] * max_frac * rng.random()
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb new best={best_score:.6f}")

        # slowly decay the allowed transfer size
        max_frac = max(0.01, max_frac * 0.9999)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)          # length for self‑convolution FFT
    L_grad = _next_pow2(3 * n - 2)          # length needed for the smooth‑max gradient

    # ------------------------------------------------------------------
    # Initialise with the global best (if present) or the uniform vector
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    starter /= starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Repeated random‑restart optimisation while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:   # keep a few seconds for final polish
        # random Dirichlet start – alternate sparse / dense
        concentration = 0.3 if np.random.rand() < 0.5 else 5.0
        rand_vec = np.random.gamma(shape=concentration, size=n)
        a0 = rand_vec / rand_vec.sum()

        # Phase‑1 : Adam on smooth max‑convolution
        phase1_deadline = min(deadline - 5.0, time.time() + 30.0)
        a1, sc1 = _smooth_descent_adam(a0, n, L_self, L_grad,
                                       phase1_deadline, start_time)
        if sc1 < best_score:
            best_seq, best_score = a1.copy(), sc1
            print(f"[{time.time() - start_time:.2f}s] Restart phase‑1 improved to {sc1:.6f}")

        # Phase‑2 : exact sub‑gradient refinement
        phase2_deadline = min(deadline - 3.0, time.time() + 15.0)
        a2, sc2 = _subgradient_exact(a1, n, L_self, L_grad,
                                     phase2_deadline, start_time)
        if sc2 < best_score:
            best_seq, best_score = a2.copy(), sc2
            print(f"[{time.time() - start_time:.2f}s] Restart phase‑2 improved to {sc2:.6f}")

        # Phase‑3 : short hill‑climb polish
        phase3_deadline = min(deadline, time.time() + 5.0)
        a3, sc3 = _hill_climb(a2, n, L_self, phase3_deadline,
                               start_time, best_seq, best_score)
        if sc3 < best_score:
            best_seq, best_score = a3.copy(), sc3
            print(f"[{time.time() - start_time:.2f}s] Restart phase‑3 improved to {sc3:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the overall best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        final_deadline = deadline - 0.5
        best_seq, best_score = _smooth_descent_adam(best_seq, n,
                                                    L_self, L_grad,
                                                    final_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] Final smooth run score = {best_score:.6f}")

    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, n, L_self,
                                           deadline, start_time,
                                           best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
