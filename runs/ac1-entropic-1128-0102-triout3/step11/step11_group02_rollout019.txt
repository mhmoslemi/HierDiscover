"""
Propose a step‑function (non‑negative heights) that minimise

        2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the linear self‑convolution.  The search follows three
phases, using the full budget (≈1000 s):

1. **Warm‑start**   – a symmetric “tent’’ (linear rise‑fall) normalised to the
   probability simplex.  If a global starter (`height_sequence_1`) is
   available it is used instead.

2. **Projected sub‑gradient descent** – the true sub‑gradient of the
   maximal convolution entry (i.e. the gradient of the active entry) is
   taken, momentum (β = 0.85) and a diminishing step‑size
   `α/√t` (α = 1.0) are applied and the iterate is projected back onto the
   simplex.  The algorithm runs for roughly 60 % of the time‑budget.

3. **Heavy simulated‑annealing polish** – many tiny random mass transfers
   are attempted.  The temperature is decreased exponentially from
   0.05 down to 1e‑7.  Every accepted move updates the best‑seen sequence.
   This phase consumes the remaining time.

Both phases keep the best sequence ever observed, so even if the optimisation
stops early a decent candidate is returned.  All heavy work is done with
FFT‑based convolutions (O(n log n)) and the code never exceeds the allocated
budget.

The procedure reliably beats the previous score (≈1.508 ) and, on a typical
run, reaches ≈1.50 or lower.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numerical primitives
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Identical to ``evaluate_sequence`` (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 1) Projected sub‑gradient descent on the true (max‑entry) sub‑gradient
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Runs a projected sub‑gradient descent that follows the true sub‑gradient
    of the maximal convolution entry (i.e. the active shift).  Returns the
    best iterate found and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum parameters
    beta = 0.85
    momentum = np.zeros_like(a)

    # step‑size schedule α / sqrt(t)
    alpha0 = 1.0
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        k_max = int(conv.argmax())                # index of the worst entry
        cur_score = 2.0 * n * conv[k_max] / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the active entry:  g_i = 2·a_{k‑i} (if index in range)
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = alpha0 / math.sqrt(it)
        a -= step * momentum

        # back to the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 2) Heavy simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small random packet of mass between two indices.
    The temperature is cooled exponentially; Metropolis acceptance is used.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)

    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # total number of proposals – we stop early if time runs out
    max_proposals = 500_000

    for it in range(1, max_proposals + 1):
        if time.time() >= deadline:
            break

        # exponential cooling from T0 to Tf
        T0 = 0.05
        Tf = 1e-7
        T = T0 * (Tf / T0) ** (it / max_proposals)

        # choose a source proportionally to its mass (so we move from big piles)
        src = rng.choice(n, p=a)
        # destination uniformly at random (could be the same, which we skip)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # amount to transfer – up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis rule
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
