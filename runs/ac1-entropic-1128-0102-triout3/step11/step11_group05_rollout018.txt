"""
Improved search for the minimum of

    2 * n * max(conv(a,a)) / (sum(a)**2)

for n=4096. The algorithm consists of:

0️⃣ Load a good starting point (previous best or uniform).

1️⃣ Smooth‑max mirror descent (exponentiated‑gradient) with a long
   geometrically increasing schedule for the smoothing parameter
   (up to α≈2000).  The iteration count is limited only by the time
   budget.

2️⃣ Projected sub‑gradient descent (momentum) on the true max‑convolution.

3️⃣ Targeted simulated‑annealing:  at each step the source index is
   sampled proportionally to the sub‑gradient of the active convolution
   index, i.e. from positions that contribute most to the current peak.
   This focuses the mass‑transfer moves where they matter most and
   typically reduces the objective faster than a purely random walk.

All phases respect the time limit; the best sequence discovered is
returned as a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution using FFT (full length)."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a non‑negative vector (sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the soft‑max of the self‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # start with a modest α and finish with a very large one
    alphas = np.geomspace(0.1, 2000.0, num=48)

    it = 0
    max_iterations = 800_000
    for alpha in alphas:
        # pre‑compute the denominator for the smooth‑max weights
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # ---- 1) self‑convolution (FFT) ---------------------------------
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # ---- 2) smooth‑max weights ------------------------------------
            exp_arg = alpha * conv
            # numerical stabilisation
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # ---- 3) gradient of the smooth objective ----------------------
            # gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # ---- 4) exponentiated‑gradient step on the simplex -------------
            # learning‑rate schedule loosely based on the theory for mirror descent
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            # guard against under‑flow
            a = np.maximum(a, 1e-30)
            a /= a.sum()

            # occasional exact evaluation
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Standard sub‑gradient (with momentum) on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25   # smaller step than the original version
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (first occurrence)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑convolution (zero elsewhere)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – targeted simulated annealing
# ----------------------------------------------------------------------
def _targeted_anneal(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     cur_best_score: float,
                     cur_best_seq: np.ndarray,
                     start_temp: float = 5e-2,
                     end_temp: float = 1e-12,
                     max_moves: int = 3_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that prefers source indices with high gradient."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1 (cheap)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # sum(a)=1 ⇒ denominator = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # temperature (exponential decay)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # locate the current peak
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1, dtype=int)

        # gradient of the active peak (zero outside the active window)
        grad = np.zeros(n)
        grad[idx_range] = 2.0 * a[k - idx_range]

        grad_sum = grad.sum()
        if grad_sum == 0.0:
            # fall back to uniform sampling
            src = rng.integers(n)
        else:
            # source chosen proportionally to gradient magnitude
            prob = grad / grad_sum
            src = rng.choice(n, p=prob)

        # destination – try to pick a location that currently contributes little
        # to the peak; we simply sample uniformly until a different index is found
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # amount of mass to transfer
        if a[src] < 1e-12:
            continue
        eps = a[src] * rng.random() * 0.35   # up to 35 % of src mass
        if eps <= 0.0:
            continue

        # incremental update of the convolution (copy‑on‑write)
        conv_new = conv.copy()
        # remove the contribution of the source mass
        conv_new[src:src + n] -= 2.0 * eps * a
        # add the contribution of the destination mass
        conv_new[dst:dst + n] += 2.0 * eps * a
        # handle the delta‑delta terms (three isolated indices)
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / T) > rng.random():
            # accept
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} "
                          f"it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Standard sub‑gradient on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.25
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} "
                  f"score={cur_score:.6f}")

        # active index (first occurrence)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the peak
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: targeted simulated annealing
# ----------------------------------------------------------------------
def _targeted_anneal(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     cur_best_score: float,
                     cur_best_seq: np.ndarray,
                     start_temp: float = 5e-2,
                     end_temp: float = 1e-12,
                     max_moves: int = 4_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that samples sources from the active sub‑gradient."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1 (cheap for n=4096)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # sum(a)=1 ⇒ denominator = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # temperature schedule (exponential decay)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # current peak position
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1, dtype=int)

        # gradient of the active peak (zero outside the window)
        grad = np.zeros(n)
        grad[idx_range] = 2.0 * a[k - idx_range]

        grad_sum = grad.sum()
        if grad_sum == 0.0:
            src = rng.integers(n)
        else:
            prob = grad / grad_sum
            src = rng.choice(n, p=prob)

        # choose a destination that is different from src
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        if a[src] < 1e-12:
            continue
        eps = a[src] * rng.random() * 0.35   # up to 35 % of src mass
        if eps <= 0.0:
            continue

        # incremental convolution update
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / T) > rng.random():
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} "
                      f"best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) of length 4096 that
    minimises `evaluate_sequence` under the given time budget.
    The algorithm uses smooth‑max mirror descent, projected sub‑gradient
    refinement and a targeted simulated‑annealing polish.
    The best sequence discovered is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣ Load a starter (previous best if available, else uniform)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')

    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Smooth‑max mirror descent (core optimisation)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        smooth_deadline = deadline - 15.0
        best_seq, best_score = _smooth_descent(best_seq, n,
                                               smooth_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best "
              f"score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Projected sub‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        sub_deadline = deadline - 5.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved "
                  f"score = {sc_sub:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Targeted simulated annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_final, sc_final = _targeted_anneal(best_seq, n,
                                            anneal_deadline, start_time,
                                            best_score, best_seq,
                                            start_temp=2e-2,
                                            end_temp=1e-12,
                                            max_moves=4_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved "
                  f"score = {sc_final:.6f}")

    # --------------------------------------------------------------
    # Normalise and convert to a plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{best_score:.6f}")
    return [float(v) for v in best_seq]
