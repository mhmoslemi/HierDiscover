"""
A multi‑stage heuristic for the step‑function problem.

1. **Broad parameter sweep** over several analytic families
   (Beta‑type, power‑law, exponential, random Dirichlet).  Each
   candidate is normalised to sum = 1 and evaluated with an FFT based
   exact objective.  The best few candidates are kept.

2. **Local refinement** of the top candidates using a smoothed
   surrogate of the max‑convolution (log‑sum‑exp) and Adam updates.
   The smoothing parameter β grows exponentially from a modest value
   (≈5) to a large one (≈500) so the optimiser first sees a smooth
   landscape and finally concentrates on the true maximum.  After each
   step the vector is projected back onto the probability simplex.

3. **Light stochastic polish** – repeated mass‑transfer moves
   (source→destination) are accepted only when they improve the exact
   objective; this finishes the optimisation.

The whole routine respects the 1000 s wall‑time limit, prints
intermediate scores and returns a plain Python list of non‑negative
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def adam_optimize(
    a_init: np.ndarray,
    start_time: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    base_lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Smoothed Adam optimiser.
    Returns the best vector found and its exact objective.
    """
    a = a_init.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)  # room for cross‑correlations
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    iteration = 0

    best_local = a.copy()
    best_local_score = _objective(a)

    # main loop – stop a little before the global deadline to keep time
    while time.time() < deadline - 1.0:
        iteration += 1

        # ---- a * a ----------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]                     # linear conv

        # ---- smoothing schedule (exponential growth) ---------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        prop = min(1.0, elapsed / total)
        beta = beta_start * (beta_end / beta_start) ** prop

        # ---- soft‑max weights for the log‑sum‑exp surrogate ---------
        beta_conv = beta * conv
        max_beta = beta_conv.max()
        w = np.exp(beta_conv - max_beta)
        w /= w.sum()                                      # shape (2n‑1,)

        # ---- gradient of the smooth surrogate -----------------------
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        Ba = corr_full[n - 1 : n - 1 + n]                 # (B·a)
        gradient = 4.0 * n * Ba                           # ∇F_s

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * gradient
        v = beta2 * v + (1.0 - beta2) * (gradient ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_simplex(a)                           # keep on simplex

        # ---- periodic exact evaluation -------------------------------
        if iteration % 20 == 0:
            cur_score = _objective(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] "
                    f"Adam iter {iteration:,} best {cur_score:.6f}"
                )

    return best_local, best_local_score

        if len(top_candidates) < top_k:
            top_candidates.append((score, seq.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif score < top_candidates[-1][0]:
            top_candidates[-1] = (score, seq.copy())
            top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 0) Seed from a possible external construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    s = _objective(starter)
                    consider(starter, s)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Systematic sweep over analytic families
    # ------------------------------------------------------------------
    # 1a) symmetric Beta‑family
    alphas = np.linspace(0.44, 0.58, 71)               # step ≈0.002
    epsilons = [0.0, 1e-12, 1e-9, 1e-6, 1e-4]
    for alpha in alphas:
        for eps in epsilons:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            consider(a, _objective(a))
        if time.time() >= deadline:
            break

    # 1b) decreasing power‑law a_i ∝ (i+shift)^{-p}
    p_vals = np.linspace(0.2, 1.2, 51)
    shifts = [0.0, 0.5, 1.0, 2.0]
    for p in p_vals:
        for sft in shifts:
            idx = np.arange(n, dtype=np.float64) + sft + 1.0
            w = idx ** (-p)
            a = w / w.sum()
            consider(a, _objective(a))
        if time.time() >= deadline:
            break

    # 1c) exponential decay a_i ∝ r^i
    r_vals = np.linspace(0.90, 0.9995, 30)
    for r in r_vals:
        w = r ** np.arange(n, dtype=np.float64)
        a = w / w.sum()
        consider(a, _objective(a))
        if time.time() >= deadline:
            break

    # 1d) a few random Dirichlet‑type vectors (sparse)
    for _ in range(20):
        if time.time() >= deadline:
            break
        a = np.random.gamma(shape=0.3, scale=1.0, size=n)
        a /= a.sum()
        consider(a, _objective(a))

    print(f"[{time.time() - start_time:.2f}s] Sweep finished. Best so far = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Local refinement with Adam on the best few candidates
    # ------------------------------------------------------------------
    for idx, (score, seq) in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        print(f"[{time.time() - start_time:.2f}s] Refining candidate {idx + 1}/{len(top_candidates)} (init score {score:.6f})")
        refined_seq, refined_score = adam_optimize(seq, start_time, deadline)
        consider(refined_seq, refined_score)

    # ------------------------------------------------------------------
    # 3) Light stochastic polishing (mass‑transfer moves)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        moves = 0
        max_moves = 30000
        while time.time() < deadline - 0.5 and moves < max_moves:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # move up to 20 % of the source mass
            eps = a[src] * rng.random() * 0.20
            a_candidate = a.copy()
            a_candidate[src] -= eps
            a_candidate[dst] += eps
            new_score = _objective(a_candidate)
            if new_score < cur_score:
                a = a_candidate
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Anneal step {moves:,} improved to {new_score:.6f}")
            moves += 1
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
