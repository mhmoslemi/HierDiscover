def _simulated_annealing(a_start, deadline, start_time, L_fft, max_iter=None):
    # set SA parameters
    rng = np.random.default_rng()
    a = a_start.copy()
    cur_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    best_score = cur_score

    # Temperature schedule: start at maybe (cur_score - 1.0) or maybe 0.2? We'll calibrate.

    # We'll do while time < deadline:
    it = 0
    while time.time() < deadline:
        it += 1
        # choose src weighted by current a to ensure moving from high mass likely
        src = rng.choice(len(a), p=a)  # picks index with prob a_i (sum=1)
        dst = rng.integers(len(a))
        if src == dst:
            continue
        # step size factor decreasing over time
        # factor = max_step * (1 - (time - start_time)/(deadline-start_time))**alpha
        # set max_step maybe 0.5 (move up to 50% of src mass)
        # compute eps
        max_step = 0.5
        factor = max_step * (1.0 - (time.time() - start_time) / (deadline - start_time))
        if factor < 1e-4:
            factor = 1e-4
        eps = a[src] * factor * rng.random()
        if eps < 1e-12:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # ensure non-negative (should be)
        cur_score_prop = _eval_fast(a_prop, L_fft)
        delta = cur_score_prop - cur_score

        # Temperature schedule: T = T0 * (1 - (elapsed/total)) maybe.
        # calibrate T0 large enough to accept worse moves early.
        # we can compute dynamic T based on best_score - lower_bound? Since we don't know bound.
        # We'll just use T = start_temp * (1 - progress) + end_temp * progress.
        # let's set start_temp = 0.02, end_temp = 0.00001
        elapsed = time.time() - start_time
        total = deadline - start_time
        T0 = 0.02
        T1 = 1e-5
        T = T0 * (1 - elapsed/total) + T1 * (elapsed/total)

        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            a = a_prop
            cur_score = cur_score_prop
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] SA it={it} score={best_score:.6f}")
    return best_a, best_score

"""
Proposes a step function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search mixes several complementary phases that respect the 1000 s
budget:

1. **Initialisation** – use the globally supplied `height_sequence_1`
   if present, otherwise start from the uniform distribution.

2. **Simulated annealing** – random mass transfers between bins with a
   linearly decreasing temperature and step‑size schedule.  This
   allows the search to escape the basin of attraction of the
   deterministic local optimiser used previously.

3. **Active‑set linear‑program refinement** (optional, requires SciPy):
   given the current shape, solve a small LP that maximises the total
   mass under the constraint that the convolution with the current
   shape never exceeds the current maximum.  Repeating this step pushes
   the distribution toward the true optimum.

4. **Adam‑style smoothed gradient descent** – minimise a soft‑max
   approximation of the `max` term.  Works well for fine‑grained
   polishing.

5. **Deterministic hill‑climb** – tiny random mass moves accepted only
   when they improve the exact objective.

The algorithm keeps the sequence normalised to sum = 1 at every step,
so the objective is invariant to scaling.  All heavy numerical work is
performed with FFT‑based convolution (`_conv_self`) which makes each
evaluation fast (≈ 0.5 ms for n = 4096).  The whole procedure fits easily
into the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation of the objective (assumes sum(a)≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Random mass transfers with an annealing schedule.
    Returns the best sequence found and its score.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    # temperature schedule (linear) and maximal transfer fraction
    T_start = 0.05
    T_end = 1e-5
    max_frac_start = 0.3   # up‑to 30 % of the source bin early on

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear temperature and step‑size decay
        T = T_start * (1 - elapsed / total) + T_end * (elapsed / total)
        max_frac = max_frac_start * (1 - elapsed / total)
        if max_frac < 1e-5:
            max_frac = 1e-5

        # choose source proportionally to current mass (helps move heavy mass)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * max_frac * rng.random()
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # numerical safety (should never go negative, but guard)
        if a_prop[src] < 0.0:
            a_prop[src] = 0.0

        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score

        # Metropolis acceptance
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] SA improved score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program refinement (active‑set)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _has_scipy = True
except Exception:  # pragma: no cover
    _has_scipy = False


def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 30,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Iteratively improve a by solving a small LP on the active set
    of (almost) binding convolution constraints.
    """
    if not _has_scipy:                     # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        # active constraints: values within tol_frac of the max
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])

        K = active.size
        # Build the constraint matrix A_ub (K × n)
        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1, dtype=int)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)   # maximise Σ g_j  <=> minimise -Σ g_j

        # the LP is small – solving it with HiGHS is fast
        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(1, int(deadline - time.time()))})

        if not res.success:                 # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0.0:
            break
        g = g / g.sum()                     # re‑normalise to sum‑1

        # convex blend (t gradually reduced if no progress)
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)

        if new_score < best_score:
            best_score = new_score
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP refine it={it} score={new_score:.6f}")
        else:
            # stagnation – stop early
            break

    return a, best_score

# Adam‑style smoothed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max smoothed version of the objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    tau = 0.05                # initial temperature for the soft‑max
    tau_min = 1e-8
    tau_decay = 0.9995        # slower decay → smoother gradients longer
    step0 = 0.15

    while time.time() < deadline:
        it += 1
        # convolution of the current iterate
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        # soft‑max weights (stable)
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. a (via FFT)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update + simplex projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Random tiny mass transfers accepted only if they improve the exact score."""
    rng = np.random.default_rng()
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the provided
    objective.  The routine respects the 1000 s budget and returns a list
    compatible with ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – larger n gives more freedom but also higher cost
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)                     # for the Adam phase

    # ------------------------------------------------------------------
    # Baseline initialisation
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # try to start from a globally supplied sequence (if any)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_seq = starter.copy()
                    best_score = cur
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:                               # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Simulated annealing (exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sa_deadline = min(deadline - 10.0,
                          time.time() + 0.6 * (deadline - start_time))
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    sa_deadline,
                                                    start_time,
                                                    L_fft)
        print(f"[{time.time() - start_time:.2f}s] After SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Active‑set LP refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if _has_scipy and time.time() < deadline - 20.0:
        best_seq, best_score = _lp_refine(best_seq,
                                          deadline,
                                          start_time,
                                          L_fft,
                                          max_iters=30,
                                          tol_frac=0.02)
        print(f"[{time.time() - start_time:.2f}s] After LP best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Adam‑style smoothed gradient descent (polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        adam_deadline = min(deadline - 5.0,
                            time.time() + 0.25 * (deadline - start_time))
        best_seq, best_score = _adam_smooth_descent(best_seq,
                                                    adam_deadline,
                                                    start_time,
                                                    L_fft,
                                                    L_grad)
        print(f"[{time.time() - start_time:.2f}s] After Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final tiny hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq,
                                           deadline,
                                           start_time,
                                           L_fft)
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    return [float(x) for x in best_seq]
