def gradient_descent(a, n, alpha, max_iters, start_time, deadline, best_seq, best_score):
    step0 = 0.5
    beta = 0.9 # for momentum
    momentum = np.zeros_like(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # conv
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD alpha={alpha} it={it} score={cur_score:.6f}")
        # compute softmax probabilities
        exp_vals = np.exp(alpha * (conv - max_conv))
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp

        # gradient (softmax) = 2 * conv(p, a_rev) trimmed
        a_rev = a[::-1]
        Lg = _next_pow2(len(p) + n - 1)
        Pf = np.fft.rfft(p, n=Lg)
        Ar = np.fft.rfft(a_rev, n=Lg)
        conv_full = np.fft.irfft(Pf * Ar, n=Lg)[: len(p) + n - 1]
        offset = n - 1
        grad = 2.0 * conv_full[offset: offset + n]

        # momentum update

"""
Propose a low‑score step function.

Algorithm
---------
1. **Initialization** – start from a good analytic shape (arcsine‑like)
   and optionally from a random Dirichlet vector.

2. **Smooth‑max gradient descent with continuation** – 
   minimise a smoothed version of the objective 
        F(a)=2·n·max(a*a)   (with Σa=1)
   using the log‑sum‑exp approximation to the max.
   The smoothing parameter α is increased gradually
   (α∈{2,8,32,128}) so that early iterations move quickly
   while later iterations refine the true max.

   Gradient of the smoothed objective is
        ∇F ≈ 2·(p ⊛ a_rev) ,
   where p are soft‑max probabilities of the convolution
   and a_rev is a reversed copy of a.  Both convolutions are
   performed by FFTs, and after each step we project onto the
   probability simplex.

3. **Optional random‑restart** – if time permits, repeat step 2
   from a freshly sampled Dirichlet start.

4. **Light simulated‑annealing polish** – a few‑thousand cheap
   mass‑transfer moves are tried; moves that improve the exact
   objective (or are accepted with a tiny Metropolis probability)
   are kept.

All operations respect the non‑negativity and normalisation
constraints; the final list is returned as plain Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (returns full length)."""
    n = x.shape[0] + y.shape[0] - 1
    L = _next_pow2(n)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    Z = np.fft.irfft(X * Y, n=L)[:n]
    return Z


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear, length 2·len(a)‑1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (continuation on α)
# ----------------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    n: int,
    alphas: list[float],
    max_iters_per_alpha: int,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray | None,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Run gradient descent on the log‑sum‑exp smoothed max.
    Returns the best sequence and its exact objective value.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    beta = 0.9  # momentum coefficient
    step0 = 0.5

    # Ensure a is on the simplex
    a = _project_simplex(a)

    for alpha in alphas:
        # number of iterations for this α (cannot exceed remaining budget)
        it = 0
        while it < max_iters_per_alpha and time.time() < deadline:
            it += 1

            # ----- exact convolution and exact objective -----
            conv = _conv_self(a)
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv   # sum(a)=1 ⇒ denominator = 1

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] α={alpha:.1f} it={it:,} score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            # stabilize exponentials
            exp_vals = np.exp(alpha * (conv - max_conv))
            p = exp_vals / exp_vals.sum()   # soft‑max probabilities, sum to 1

            # ----- gradient of the smoothed objective -----
            a_rev = a[::-1]                 # reversed a
            grad_full = _conv_fft(p, a_rev)  # length = (2n‑1)+n‑1 = 3n‑2
            offset = n - 1
            grad = 2.0 * grad_full[offset : offset + n]

            # momentum + diminishing step
            momentum = beta * momentum + (1.0 - beta) * grad
            step = step0 / math.sqrt(it)
            a = a - step * momentum

            # project back to simplex
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal_polish(
    a0: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Small random‑mass‑transfer annealing starting from a0.
    Returns the best sequence seen.
    """
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a).max()
    best_local = a.copy()
    best_local_score = cur_score
    rng = np.random.default_rng()
    total_steps = 25000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule (very low temperature)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        # pick source proportional to its mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (≤20 %) of src's mass to dst
        eps_move = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={new_score:.6f}")

    if best_local_score < best_score:
        best_score = best_local_score
        best_seq = best_local.copy()
        print(f"[{time.time() - start_time:.2f}s] final anneal improve score={best_score:.6f}")

    return best_seq, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 4096                       # length of the candidate
    alphas = [2.0, 8.0, 32.0, 128.0]   # continuation schedule
    max_iters_per_alpha = 120000      # will be capped by remaining time

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Candidate 1 – analytic arcsine‑type start
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # arcsine family corresponds to exponent -0.5; we add a tiny epsilon to avoid div‑by‑0.
    a0 = 1.0 / np.sqrt(x * (1.0 - x) + 1e-12)
    a0 = a0 / a0.sum()   # project onto simplex

    # run smooth‑max descent
    best_seq, best_score = _smooth_max_descent(
        a0, n, alphas, max_iters_per_alpha, start_time, deadline, best_seq, best_score
    )

    # ------------------------------------------------------------------
    # 2) Optional random restart (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:   # keep a few seconds for polishing
        # Dirichlet(1) → uniform random simplex point
        a_rand = np.random.default_rng().gamma(1.0, size=n)
        a_rand = a_rand / a_rand.sum()
        best_seq, best_score = _smooth_max_descent(
            a_rand, n, alphas, max_iters_per_alpha // 2, start_time, deadline, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 3) Light annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal_polish(
            best_seq, n, start_time, deadline, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 4) Final normalisation (the objective is homogeneous, but the verifier
    #    clamps negatives and caps large values; we keep sum = 1)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full(n, 1.0 / n)

    # Convert to plain Python list
    return [float(v) for v in best_seq]
