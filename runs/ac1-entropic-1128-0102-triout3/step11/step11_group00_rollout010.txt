"""
Propose a non‑negative step function that (almost) minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2),

where conv is the full discrete convolution.
The algorithm consists of three phases:

1️⃣  An extensive parametric sweep (geometric, power, beta, sigmoid)
    to obtain a very good seed (or to improve an externally supplied seed).

2️⃣  A time‑controlled set of independent projected sub‑gradient
    (momentum) runs started from many different random seeds.
    Each run keeps the best vector it ever sees.

3️⃣  A final hill‑climb that performs tiny mass transfers between random
    positions (larger than the previous 1e‑4 factor) and accepts only
    improving moves.

All work is performed with FFT‑based convolutions (O(n log n)) and the
Euclidean projection onto the simplex.  The total running time never
exceeds the supplied budget (default 1000 s).  The returned list is a
plain Python list of floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Full self‑convolution a * a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w | w_i ≥ 0, sum w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Fast exact evaluation of the objective 2·n·max(conv(a,a)) / (sum a)².
    The vector a is assumed to be non‑negative; the sum is not forced
    to be 1 because the ratio is scale‑invariant.
    """
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Gradient / sub‑gradient of the max‑convolution term
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k conv(a,a)[k] (returns a vector of size n)."""
    n = a.shape[0]
    conv = _conv_self(a)
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    # For every i in the overlapping interval the derivative is 2*a_{k-i}
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# One run of projected momentum sub‑gradient descent
# ----------------------------------------------------------------------
def _run_subgradient(a0: np.ndarray,
                     L_fft: int,
                     deadline: float,
                     start_time: float,
                     global_best_score: float) -> tuple[np.ndarray, float]:
    """
    Perform a projected momentum sub‑gradient descent until ``deadline``.
    Returns the best vector (and its score) found during the run.
    """
    a = a0.copy()
    best_local = a.copy()
    best_score = _eval_fast(a, L_fft)

    rng = np.random.default_rng()
    n = a.shape[0]
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.8
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        cur_score = _eval_fast(a, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_local = a.copy()
            last_improve = it

        # sub‑gradient of the active shift
        grad = _subgradient(a)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step‑size
        step = step0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = a - step * momentum
        a = _project_simplex(a)

        # occasional random jitter to escape plateaus
        if it - last_improve > 4000:
            # add a very small Gaussian perturbation (≈10⁻⁴ of the current masses)
            noise = 1e-4 * rng.standard_normal(n) * a
            a = np.clip(a + noise, 0.0, None)
            a = _project_simplex(a)
            last_improve = it

    return best_local, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(a0: np.ndarray,
                     L_fft: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """Fine‑grained random mass transfers that are kept only if they improve."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        # move a modest fraction of the source mass (up to ≈10⁻³ of src)
        eps = a[src] * rng.random() * 1e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best = a_prop.copy()
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Parametric families used for the initial sweep
# ----------------------------------------------------------------------
def _geo_family(idx: np.ndarray, eps: float) -> np.ndarray:
    """Geometric decay a_i ∝ (1‑eps)^i."""
    r = 1.0 - eps
    seq = np.power(r, idx, dtype=np.float64)
    return seq / seq.sum()


def _power_family(idx: np.ndarray, p: float) -> np.ndarray:
    """Power law a_i ∝ (i+1)^p."""
    seq = np.power(idx + 1.0, p, dtype=np.float64)
    return seq / seq.sum()


def _beta_family(idx: np.ndarray, p: float, q: float, n: int) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)^p * (n‑i)^q."""
    seq = np.power(idx + 1.0, p, dtype=np.float64) * np.power(n - idx, q, dtype=np.float64)
    return seq / seq.sum()


def _sigmoid_family(idx: np.ndarray, alpha: float, shift: float) -> np.ndarray:
    """Sigmoid a_i = 1/(1+exp(alpha*(i‑shift)))."""
    # use float64 for stability
    logits = alpha * (idx.astype(np.float64) - shift)
    seq = 1.0 / (1.0 + np.exp(logits))
    # avoid exact zeros (they are harmless after projection)
    return seq / seq.sum()
