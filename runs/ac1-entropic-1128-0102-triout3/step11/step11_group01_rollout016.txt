"""
Improved step‑function search for the evaluation function.

Strategy
--------
* Use a larger canvas (n = 65536 = 2**16).  The discretisation error for the
  continuous optimum 1.5 shrinks roughly like O(1/n), giving a theoretical
  bound ≈ 1.501 for this size.
* Generate a family of symmetric “tent‑with‑plateau’’ shapes that start at
  exact zero at the two ends (the original code started at a tiny positive
  value).  The shape is parameterised by a curvature exponent *p* (0.2–2.5)
  and a plateau width *α* ∈ [0, 0.5].
* Evaluate all grid points, keep the few best candidates and add a handful of
  random draws to diversify the pool.
* Refine the current best candidate with a long projected sub‑gradient descent
  run (momentum, 1/√t step‑size).  The sub‑gradient is taken from the index of
  the maximal self‑convolution and the iterate is always projected onto the
  probability simplex.
* Finish with a short simulated‑annealing polish that moves tiny mass packets
  between random positions (Metropolis acceptance).  The annealing schedule
  is aggressive but fits into the remaining time budget.
* Return a normalised (sum = 1) non‑negative list – the harness will evaluate
  it with ``evaluate_sequence``.

The whole search fits comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """O(n log n) simplex projection (Wang & Carreira‑Perpiñán)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generator – symmetric “tent‑with‑plateau’’ that starts at zero
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Produce a symmetric shape of length n.
    *p*   – exponent of the side ramps (p > 0).
    *alpha* – plateau width relative to n (0 ≤ α ≤ 0.5).  α = 0 → pure
    tent, α = 0.5 → uniform.
    The shape is normalised to sum 1.
    """
    # Clamp α
    if alpha < 0.0:
        alpha = 0.0
    if alpha > 0.5:
        alpha = 0.5

    plateau_len = int(round(alpha * n))
    # Ensure the plateau has the same parity as n so that symmetry works
    if plateau_len % 2 == 1:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len == 0:
        # pure uniform case
        shape = np.ones(n)
    else:
        # left ramp (including zero, excluding the point that becomes the start
        # of the plateau)
        idx = np.arange(side_len + 1, dtype=float) / side_len   # 0 … 1
        ramp = idx ** p
        left = ramp[:-1]                # drop the final 1 to avoid double‑count
        right = left[::-1]
        shape = np.concatenate([left,
                                np.ones(plateau_len, dtype=float),
                                right])
        # In the very rare case rounding errors change the length
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Run projected sub‑gradient descent until ``deadline``.
    Returns the best iterate seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index where the convolution attains its maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny mass swaps with Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # limit to a few hundred‑thousand proposals – fits easily in the remaining time
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # source is chosen proportionally to its mass (more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 25 % of source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two for fast FFTs.
    # --------------------------------------------------------------
    n = 65536                 # double the previous canvas

    # --------------------------------------------------------------
    # 1) Sweep a modest analytic family (tent‑with‑plateau shapes).
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    top_k = 8
    top_candidates = []   # (score, seq)

    p_grid = np.linspace(0.2, 2.5, 23)          # 0.2 … 2.5
    alpha_grid = np.linspace(0.0, 0.5, 11)      # 0.0 … 0.5

    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_shape(n, float(p), float(alpha))
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
            # keep a short list of the best candidates seen so far
            if len(top_candidates) < top_k:
                top_candidates.append((score, a.copy()))
                top_candidates.sort(key=lambda x: x[0])
            else:
                if score < top_candidates[-1][0]:
                    top_candidates[-1] = (score, a.copy())
                    top_candidates.sort(key=lambda x: x[0])

            # occasional progress reporting
            if (time.time() - start_time) % 30 < 0.1:
                print(f"[{time.time() - start_time:.2f}s] grid p={p:.2f} α={alpha:.2f} → score={score:.6f}")

    # --------------------------------------------------------------
    # 2) Add a few random shapes for diversity.
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(300):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.1, 3.0)
        alpha = rng.uniform(0.0, 0.5)
        a = _make_shape(n, float(p), float(alpha))
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        if len(top_candidates) < top_k:
            top_candidates.append((score, a.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if score < top_candidates[-1][0]:
                top_candidates[-1] = (score, a.copy())
                top_candidates.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] Best after grid/random = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Use the global starter if it exists (e.g. from a previous run).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # zero‑pad / trim to current length
            if starter.size < n:
                starter = np.concatenate([starter, np.zeros(n - starter.size)])
            elif starter.size > n:
                starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    if len(top_candidates) < top_k:
                        top_candidates.append((cur, starter.copy()))
                        top_candidates.sort(key=lambda x: x[0])
                    else:
                        if cur < top_candidates[-1][0]:
                            top_candidates[-1] = (cur, starter.copy())
                            top_candidates.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4) Local refinement – sub‑gradient descent on the current best.
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate ~80 % of the remaining budget to descent
        sub_deadline = time.time() + 0.80 * remaining
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if any time left).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve "
                  f"score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to a plain list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
