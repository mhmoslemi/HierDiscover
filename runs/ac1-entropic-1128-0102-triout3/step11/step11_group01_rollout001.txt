"""
Propose a step‑function (non‑negative heights) with a low value of the
provided evaluation function.

Algorithm
---------
1. Choose a relatively large canvas (n = 65536, a power of two – fast FFT).
2. Scan a cheap 2‑parameter family of symmetric “tent‑with‑plateau’’ shapes
   a_i = f(i; p, α) (exponent p on the side ramps, plateau width α).
   The shapes are normalised to sum 1, and the best few are kept.
3. From the best shape start a *projected sub‑gradient* optimisation.
   *   The sub‑gradient is taken from *all* indices whose convolution value
       is within a tiny tolerance of the current maximum (instead of a
       single index).  This gives a more reliable descent direction.
   *   Momentum (β = 0.9) and a diminishing step size  step₀/√t  are used.
   *   After each step the vector is projected onto the probability simplex
       (Euclidean projection) to keep the sum equal to 1.
   *   The loop runs until the time budget for this phase is exhausted,
       keeping the best iterate seen.
4. A light simulated‑annealing polish moves tiny packets of mass between
   random positions; Metropolis acceptance with an exponential cooling
   schedule is used.
5. The final vector is clipped to non‑negative values, normalised to sum 1
   and returned as a plain Python list.

The combination of a finer discretisation (n = 65536) and a more effective
sub‑gradient step (multiple active constraints) typically pushes the
objective below 1.503 within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2).  Identical to evaluate_sequence."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σ w_i = 1, w_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generator: symmetric tent with optional flat plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Build a symmetric shape of length n.
    * p     – exponent of the side ramps (p > 0).
    * alpha – relative plateau width ∈ [0, 0.5]; α = 0 → pure tent,
              α = 0.5 → uniform distribution.
    The shape is normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    # keep plateau length even, easier to stay symmetric
    if plateau_len % 2 == 1:
        plateau_len = max(0, plateau_len - 1)

    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # values from 0 … 1 on one side
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # fix possible length off‑by‑one due to rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (multiple active indices)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on the exact objective.
    Returns the best iterate (and its score) seen so far.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 1.0          # initial step magnitude
    it = 0
    eps_rel = 1e-12      # tolerance for “active” convolution entries

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices where convolution is (almost) maximal
        tol = cur_max * eps_rel
        active = np.where(conv >= cur_max - tol)[0]

        # compute sub‑gradient averaged over all active indices
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(active))

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a_candidate = a - step * momentum
        a_candidate = _project_simplex(a_candidate)

        # simple acceptance test – always move (sub‑gradient may not improve
        # immediately but projection keeps feasibility)
        a = a_candidate

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # pick source proportionally to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
