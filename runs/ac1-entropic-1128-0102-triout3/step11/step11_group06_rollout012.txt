"""
Propose a low‑objective step sequence.

Algorithm
---------
1) **Coarse parametric sweep** – evaluate a family
   a_i ∝ (x_i·(1−x_i)+ε)^{−α} (α≈0.5, ε small) on a modest length
   (default n=2048).  Keep the best few candidates.

2) **Exact gradient descent** – for each selected candidate run a
   projected gradient descent on the *exact* objective
   (the gradient is derived analytically from the maximum of the
   linear convolution).  A simple back‑tracking line search keeps the
   step size stable.

3) **Heavy hill‑climbing** – starting from the best sequence perform a
   large number of random “mass‑transfer’’ moves (move a small
   fraction of the mass from one bin to another).  Only improving
   moves are kept; this simple simulated‑annealing style polish
   quickly squeezes out the remaining few‑thousandths.

All steps respect the simplex constraint (non‑negative, Σa = 1) and
stay well within the 1000 s wall‑time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(a,a)) / (sum(a)**2)
    Computed with an FFT based linear convolution.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_a ** 2))


def _gradient_exact(a: np.ndarray) -> np.ndarray:
    """
    Gradient of the exact objective w.r.t. a (before projection).
    Uses the index k* at which the convolution attains its maximum.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    k = int(conv.argmax())          # location of max
    max_val = float(conv[k])
    S = a.sum()

    gradC = np.zeros_like(a)

    if k < n:
        # range j = 0 .. k
        gradC[:k + 1] = 2 * a[k::-1]
    else:
        start = k - (n - 1)          # first j where k‑j is inside [0,n‑1]
        # fill gradC[start:] with 2*a[n‑1 : start‑1 : -1]
        gradC[start:] = 2 * a[n - 1:start - 1:-1]

    # gradient of f = 2·n·C_k / S²
    grad = (2.0 * n / (S ** 3)) * (S * gradC - 2.0 * max_val)
    return grad


def _gd_step(a: np.ndarray, grad: np.ndarray, cur_obj: float, lr: float):
    """
    Perform one projected gradient‑descent step with a simple
    back‑tracking line search.  Returns (new_a, new_obj, new_lr).
    """
    for _ in range(5):
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)
        new_obj = _objective(a_new)
        if new_obj <= cur_obj:
            # modestly increase step size for the next iteration
            return a_new, new_obj, lr * 1.05
        lr *= 0.5
    # No improvement – shrink the step size
    return a, cur_obj, lr * 0.5


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) with a low
    evaluation score under the given time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # problem size – a power of two works well with the FFTs
    n = 2048

    # ------------------------------------------------------------------
    # 1) Generate a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # ---- a) parametric Beta‑family sweep ----
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions
    alphas = np.linspace(0.40, 0.62, 45)            # a bit wider than before
    epsilons = [0.0, 1e-9, 1e-7, 1e-5]

    for alpha in alphas:
        for eps in epsilons:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            sc = _objective(a)
            candidates.append((a, sc))
        if time.time() >= deadline:
            break

    # ---- b) random Dirichlet draws ----
    for _ in range(30):
        a = np.random.exponential(scale=1.0, size=n)
        a /= a.sum()
        sc = _objective(a)
        candidates.append((a, sc))
        if time.time() >= deadline:
            break

    # ---- c) global starter if provided ----
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    # rescale/interpolate to length n if needed
                    if starter.size != n:
                        xs_old = np.linspace(0, 1, starter.size)
                        xs_new = np.linspace(0, 1, n)
                        starter = np.interp(xs_new, xs_old, starter)
                    sc = _objective(starter)
                    candidates.append((starter, sc))
        except Exception:
            pass

    # keep only the most promising candidates
    candidates.sort(key=lambda tup: tup[1])
    top_candidates = [a.copy() for a, _ in candidates[:12]]

    # ------------------------------------------------------------------
    # 2) Exact gradient descent on the top candidates
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for idx, a in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        cur_obj = _objective(a)
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] start #{idx} obj={cur_obj:.6f}")

        lr = 0.2                     # initial learning rate
        no_improve = 0
        # run a few (up to ~200) gradient steps per candidate
        while time.time() < deadline and no_improve < 12:
            grad = _gradient_exact(a)
            a_new, new_obj, lr = _gd_step(a, grad, cur_obj, lr)
            if new_obj < cur_obj - 1e-10:
                a = a_new
                cur_obj = new_obj
                no_improve = 0
                if cur_obj < best_score:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s]  cand {idx} improved to {cur_obj:.6f}")
            else:
                no_improve += 1

    # If nothing was found (should not happen), fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3) Heavy hill‑climbing polish (mass‑transfer moves)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_obj = _objective(a)
    rng = np.random.default_rng()
    move_counter = 0
    max_moves = 250_000   # generous budget – will stop early when out of time

    while time.time() < deadline and move_counter < max_moves:
        # pick source proportionally to its mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        # transfer a modest fraction (≤15 %) of the source mass
        eps = a[src] * rng.random() * 0.15
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_obj = _objective(a_prop)
        if new_obj < cur_obj:
            a = a_prop
            cur_obj = new_obj
            if new_obj < best_score:
                best_score = new_obj
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill step={move_counter:,} best={new_obj:.6f}")
        move_counter += 1

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
