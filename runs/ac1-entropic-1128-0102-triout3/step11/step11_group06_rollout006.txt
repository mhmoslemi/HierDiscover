"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. **Initial sweep** – Brute force over a one‑parameter Beta‑family
   a_i ∝ (x_i·(1−x_i)+ε)^(-α) with a fine grid on α and a few ε.
   The best candidate (including a possible global starter) is kept.

2. **Projected sub‑gradient descent** – The exact (non‑smoothed) objective
   F(a)=2·n·max(a∗a)/ (∑a)^2 is minimised.
   For the current maximal convolution index k* the gradient is
       ∇F = (4·n / S³)·( S·shift(k*) – M·1 ),
   where S=∑a, M=max(a∗a) and shift(k*)_j = a_{k*‑j} (zero outside).
   We perform many iterations with a diminishing learning rate and
   Euclidean projection onto the probability simplex.

3. **Light simulated‑annealing polish** – Random mass‑transfer moves
   (source chosen proportionally to its current mass, destination uniform)
   are accepted only if they improve the exact objective.

All steps respect the 1000 s wall‑time limit and return a plain Python
list of floats (non‑negative, summed to one).  The code works with only
numpy and scipy (for the linear‑algebra utilities).
"""

import time
import numpy as np
from scipy.optimize import linprog  # only imported for potential future extensions

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT‑based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _beta_shape(n: int, alpha: float, eps: float) -> np.ndarray:
    """Return a symmetric Beta‑family shape of length n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x * (1.0 - x) + eps) ** (-alpha)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step vector with a low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two works well with the FFTs
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)          # length used for all FFTs

    # --------------------------------------------------------------
    # 1) Fine sweep over the Beta‑family (α, ε)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # a finer grid than the previous run
    alphas = np.linspace(0.44, 0.56, 241)   # step ≈ 0.0005
    epsilons = [0.0, 1e-12, 1e-9, 1e-6]

    for alpha in alphas:
        for eps in epsilons:
            cand = _beta_shape(n, alpha, eps)
            score = _objective(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2) Global starter, if the harness supplied one
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.sum() > 0:
                    starter = starter / starter.sum()
                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3) Random Dirichlet restarts (cheap, help escape local minima)
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    for _ in range(20):
        if time.time() >= deadline:
            break
        cand = rng.gamma(1.0, size=n)
        cand = cand / cand.sum()
        score = _objective(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # Ensure we have a valid starting vector on the simplex
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = _project_simplex(best_seq)

    # --------------------------------------------------------------
    # 4) Projected sub‑gradient descent on the exact objective
    # --------------------------------------------------------------
    a = best_seq.copy()
    best_local = a.copy()
    best_local_score = best_score

    iteration = 0
    # learning‑rate schedule parameters
    base_lr = 0.12          # a bit larger than typical Adam step
    while time.time() < deadline - 3.0:   # keep a couple of seconds for polishing
        iteration += 1

        # --- linear convolution via FFT (self‑convolution) ---
        a_fft = np.fft.rfft(a, n=L_fft)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L_fft)
        conv = conv_full[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        M = conv[k_max]
        S = a.sum()

        # --- build the "shifted" vector d_j = a_{k*‑j} (zero outside) ---
        shift = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            shift[idx] = a[k_max - idx]

        # --- exact gradient of the (non‑smoothed) objective ---
        grad = (4.0 * n / (S ** 3)) * (S * shift - M)

        # --- diminishing step size (1/√t) ---
        lr = base_lr / np.sqrt(iteration)

        # --- gradient step and projection onto the simplex ---
        a = a - lr * grad
        a = _project_simplex(a)

        # --- keep track of the best exact score seen ---
        if iteration % 250 == 0:
            cur_score = _objective(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                print(f"[{time.time() - start_time:.2f}s] subgrad iter={iteration:,} best={cur_score:.6f}")

    # store the best result from the sub‑gradient phase
    if best_local_score < best_score:
        best_score = best_local_score
        best_seq = best_local.copy()

    # --------------------------------------------------------------
    # 5) Light simulated‑annealing polish (mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng(seed + 99999)

        # number of moves depends on remaining time
        max_moves = 80_000
        for move in range(1, max_moves + 1):
            if time.time() >= deadline:
                break

            # select source proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

            # transfer a modest (up to 15 %) fraction of the source mass
            transfer = a[src] * rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= transfer
            a_prop[dst] += transfer

            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={move:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
