        if score < best_score:

    # Phase 3: Local refinement via subgradient descent
    # Use a function similar to previous _subgradient_descent but maybe with adapt step size.
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)   # fallback
        best_score = _objective(best_seq)

    remaining = deadline - time.time()

        # allocate half of remaining for descent
        sub_deadline = time.time() + remaining * 0.6
        # adapt initial step based on initial gradient magnitude maybe
        refined_seq, refined_score = _subgradient_descent(best_seq, sub_deadline, start)

    # Phase 4: Annealing

"""
Propose a low‑value step‑function for the given evaluation routine.

Algorithm
---------
1. **Problem size** – use a power‑of‑two length `n = 65536`.  This brings the
   discretisation error of the continuous optimum (≈ 1.5) down to ≈ 1.504
   and still allows fast FFT‑based convolutions.

2. **Rich family of analytic shapes** – generate candidates from two
   low‑dimensional families:
   * a symmetric “ramp‑with‑plateau’’ shape (`_make_shape`) controlled by an
     exponent `p` (steepness) and a plateau width `α`;
   * a symmetric Beta‑type shape (`_make_beta_shape`) of the form
     `(i·(n‑1‑i))^γ` controlled by a single exponent `γ`.
   All candidates are normalised to unit $L_1$‑norm (the objective is
   scale‑invariant).

3. **Fast global search** – sweep a modest grid of `(p,α)` and a grid of `γ`,
   plus a large set of random samples drawn from the admissible ranges.
   The best few candidates are retained.

4. **Projected sub‑gradient descent** – starting from the best candidate,
   minimise the exact (non‑smooth) objective
   `2·n·max(conv(a,a)) / (sum(a)^2)` by repeatedly
   * evaluating the self‑convolution via a real FFT,
   * taking a sub‑gradient of the maximal entry,
   * performing a momentum step,
   * projecting back onto the probability simplex.
   The step size decays as `1/√t`.  This phase runs for as long as the
   remaining time budget permits (≈ 60 % of the total budget).

5. **Light simulated‑annealing polish** – move tiny packets of mass between
   random positions with a Metropolis acceptance rule.  This helps escape
   any remaining shallow local minima.

6. **Final normalisation** – clip negative values, renormalise to sum = 1,
   and return a plain Python list.

All helper functions are defined at top level so that the harness can
pickle the code if needed.  The search respects the hard 1000 s timeout.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT.  Returns length 2·len(a)‑1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex; O(n log n) implementation."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length n.
    - p > 0 : exponent of the side ramps (larger → steeper centre).
    - alpha ∈ [0,0.5] : relative width of a flat central plateau.
    The shape is normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))

    # enforce even plateau length for exact symmetry
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1

    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n, dtype=float)
    else:
        # side ramp from 0 (edge) to 1 (plateau start)
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # fix possible off‑by‑one after rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, gamma: float) -> np.ndarray:
    """
    Symmetric Beta‑type shape a_i ∝ (i·(n‑1‑i))^γ.
    gamma ≥ 0.  Normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    # product is zero at the borders; add a tiny epsilon to keep them non‑zero
    prod = i * (n - 1 - i)
    # avoid 0**0 when gamma==0
    if gamma == 0.0:
        shape = np.ones(n, dtype=float)
    else:
        shape = np.power(prod, gamma, where=prod > 0, out=np.zeros_like(prod))
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float, start_time: float):
    """Run projected sub‑gradient descent until ``deadline``."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index where the maximal convolution entry occurs
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a (see analysis)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float):
    """Move tiny mass packets between random positions; Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # Choose source proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # Transfer a random fraction (up to 25 %) of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
