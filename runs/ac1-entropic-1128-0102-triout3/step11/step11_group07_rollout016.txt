"""
Improved search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The routine mixes:
* an extensive parametric sweep (geometric, power, shifted‑power, beta families)
* Adam‑style gradient descent with a very smooth soft‑max approximation
* a cheap simulated‑annealing hill‑climber that can accept uphill moves
* an optional active‑set LP refinement (if SciPy is available)

The search runs for the whole budget (default 1000 s) and always returns a
feasible sequence (all entries ≥0, sum normalised to 1).  The code is
written with only top‑level functions – no closures or lambdas – and uses
only NumPy/SciPy which are available on the execution platform.
"""

import time
import math
import numpy as np

# optional SciPy imports
try:
    from scipy.optimize import linprog
    _has_scipy = True
except Exception:                     # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation of the objective assuming a sums to 1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

def _geom_sequence(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i , normalised."""
    seq = np.power(r, np.arange(n, dtype=np.float64))
    seq_sum = seq.sum()
    if seq_sum == 0:
        return np.full(n, 1.0 / n)
    return seq / seq_sum


def _power_sequence(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """Power law a_i ∝ (i+offset)**p  (offset≥0), normalised."""
    idx = np.arange(n, dtype=np.float64) + offset
    seq = np.power(idx, p, dtype=np.float64)
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return seq / s


def _beta_sequence(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (i+1)**p * (n-i)**q."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    seq = np.power(idx, p) * np.power(n - idx + 1.0, q)
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return seq / s

# Adam‑style gradient descent on the soft‑max smoothed objective
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max smoothed objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    tau = 0.10                 # start temperature a little higher
    tau_min = 1e-9
    tau_decay = 0.9998
    step0 = 0.12

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        # stable soft‑max on the convolution vector
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed max: 2 * (a * w) convolved (reversed w)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         max_iter: int = 200000) -> tuple[np.ndarray, float]:
    """Simple SA that moves a fraction of mass between two positions."""
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    cur_score = _eval_fast(a, L_fft)
    best_score = cur_score

    # initial temperature calibrated on the first 100 trials
    rng = np.random.default_rng()
    temps = []
    for _ in range(100):
        src = rng.integers(n)
        while a[src] < 1e-14:
            src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.2
        a_tmp = a.copy()
        a_tmp[src] -= eps
        a_tmp[dst] += eps
        # projection keeps sum=1; here we stay on simplex automatically
        t = _eval_fast(a_tmp, L_fft)
        temps.append(abs(t - cur_score))
    T = max(temps) if temps else 0.01
    T = max(T, 1e-4)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.2
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # no need for explicit projection – sum stays constant and
        # non‑negativity is guaranteed by eps≤a[src]
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} T={T:.2e} best={best_score:.6f}")
        # geometric cooling
        T *= 0.99995
        if T < 1e-8:
            T = 1e-8

    return best_a, best_score


# ----------------------------------------------------------------------
# Active‑set linear‑program refinement (LP)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 10,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Iterative LP improvement on the active constraints."""
    if not _has_scipy:                     # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        # active set: k whose value is within tol_frac of the maximum
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])
        K = active.size

        # Build constraint matrix A_ub (K × n) for each active shift
        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1, dtype=int)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)

        # maximise sum(g) <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(1, int(deadline - time.time()))})

        if not res.success:                     # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0:
            break
        g /= g.sum()

        # convex combination (use a moderate step size)
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)
        if new_score < best_score:
            best_score = new_score
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP it={it} score={new_score:.6f}")
        else:
            break

    return a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The strategy consists of a dense parametric sweep, Adam‑style
    smooth‑max descent, a simulated‑annealing polish and a final
    optional LP refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a tiny safety margin

    # Use a reasonably large n (the previous best used 4096)
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)                     # for gradient convolutions

    # ------------------------------------------------------------------
    # Baseline uniform
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Use a global starter if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:                               # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Expanded parametric families
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # Geometric family (r close to 1)
    for r in np.logspace(-5, -0.1, 80):
        if time.time() >= deadline - 30:
            break
        seq = _geom_sequence(n, r)
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Geom r={r:.2e} score={sc:.6f}")

    # Power family, include shifted versions
    for p in np.linspace(-1.5, 1.5, 61):
        if time.time() >= deadline - 30:
            break
        seq = _power_sequence(n, p, offset=0.0)
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

        # shifted offset variants (only modest offsets to keep shape smooth)
        for off in (0.5, 1.0, 2.0):
            seq = _power_sequence(n, p, offset=off)
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} off={off} score={sc:.6f}")

    # Beta‑type family with finer grid
    beta_vals = np.linspace(0.0, 3.0, 13)
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            seq = _beta_sequence(n, p, q)
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style smoothed gradient descent with multiple restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # Random perturbation of the current best (or uniform) for diversity
        base = best_seq if np.random.rand() < 0.6 else uniform
        noise = 0.06 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 12.0)  # ~12 s per run
        cand_seq, cand_score = _adam_smooth_descent(start, sub_deadline,
                                                    start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new global best score={best_score:.6f}")

        # early exit if we have already crossed a plausible target
        if best_score <= 1.45:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polish (accepts uphill moves)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0 and best_score > 1.45:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, L_fft,
                                                    max_iter=500000)

    # ------------------------------------------------------------------
    # 4️⃣  Optional active‑set LP refinement (could catch a final edge)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0 and best_score > 1.45:
        best_seq, best_score = _lp_refine(best_seq, deadline,
                                          start_time, L_fft)

    # ------------------------------------------------------------------
    # Normalisation and final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
