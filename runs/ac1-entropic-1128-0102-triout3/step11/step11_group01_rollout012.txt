def _conv_full(a,b):
    L = _next_pow2(len(a)+len(b)-1)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(A * B, n=L)[:len(a)+len(b)-1]
    return conv

    candidates = []

"""
Search for a low‑value step function.

The algorithm works in three stages:

1. **Initial pool** – generate a diverse set of symmetric, non‑negative
   shapes (Hann, Kaiser, power‑ramp, semi‑circular, up‑sampled global
   starter) on a large canvas (n = 65536).  Keep the best few candidates.

2. **Smooth optimisation** – minimise a smooth approximation of the
   original objective
        Jε(a) = 2·n·ε·log( Σ exp( (a*a) / ε ) )
   using Adam on the probability simplex.  The gradient of Jε can be
   written as a correlation between the weight vector
   w = softmax((a*a)/ε) and the current a, which is computed by a
   single FFT per iteration.  The smoothing parameter ε is reduced
   geometrically.

3. **Refinement** – a few iterations of projected sub‑gradient descent
   (as in the reference solution) and a short simulated‑annealing
   polish finish the search.

All vectors are kept on the simplex (non‑negative, Σ = 1) so the
objective is scale‑invariant.  The routine respects the 1000 s budget
and returns the best sequence found as a plain list of floats.
"""

# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (real FFT, length 2·len(a)-1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (real FFT, length len(a)+len(b)-1)."""
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: len(a) + len(b) - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant; sum(a)=1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, Σ w = 1} (O(n log n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _make_power_ramp(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric power‑ramp with optional plateau (same as the reference code).
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_hann(n: int) -> np.ndarray:
    """Standard Hann window, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window (beta > 0) normalised to sum = 1."""
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    r = np.sqrt(1.0 - (2.0 * i / (n - 1)) ** 2)
    # I0 is the zeroth order modified Bessel function; approximate via series
    # For small beta we can use numpy's special, but to avoid extra imports we
    # fallback to numpy.i0 if available; otherwise use a simple approximation.
    try:
        from numpy import i0
        w = i0(beta * r) / i0(beta)
    except Exception:          # very crude fallback
        w = np.exp(beta * (r - 1.0))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _make_semicircle(n: int, q: float = 1.0) -> np.ndarray:
    """
    Symmetric semi‑circular profile: a_i ∝ (1 - ((i-center)/center)^2)^{q}
    clipped to zero outside the unit interval.  Normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    r = np.abs(i - center) / center
    mask = r <= 1.0
    shape = np.zeros_like(i)
    shape[mask] = (1.0 - r[mask] ** 2) ** q
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# Smooth optimisation (log‑sum‑exp surrogate) using Adam
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the original objective.
    Uses an exponentially decreasing ε and Adam with projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    lr = 1e-7                      # base learning rate (scaled by 4·n later)
    eps_factor = 0.5               # initial ε = eps_factor * max_conv
    eps_min_factor = 1e-4          # stop when ε ≤ eps_min_factor * init_max
    max_iters_per_eps = 2000

    best_a = a.copy()
    best_score = _objective(a)

    # initial ε
    conv = _conv_self(a)
    eps = max(conv) * eps_factor
    eps_min = max(conv) * eps_min_factor

    t = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    while time.time() < deadline and eps > eps_min:
        # run a short inner loop at the current ε
        for _ in range(max_iters_per_eps):
            if time.time() >= deadline:
                break
            t += 1

            conv = _conv_self(a)
            # softmax weights w = exp(conv/ε) / Σ …
            max_c = conv.max()
            # prevent overflow/underflow
            w = np.exp((conv - max_c) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed objective:
            # g = 4·n * corr(w, a)   (corr defined in the analysis)
            a_rev = a[::-1]
            corr_full = _conv_full(w, a_rev)          # length 3n‑2
            corr = corr_full[n - 1 : n - 1 + n]       # length n
            grad = 4.0 * n * corr

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)

            a -= step
            a = _project_simplex(a)

            # keep track of the best iterate seen
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] smooth iter={t:,} ε={eps:.2e} score={cur_score:.6f}")

        # decay ε
        eps *= 0.7

    return best_a, best_score

# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on the exact (non‑smooth) objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} score={cur_score:.6f}")

        # index where the maximum is attained
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {refined_best_score:.6f}")
    return [float(v) for v in final_seq]
