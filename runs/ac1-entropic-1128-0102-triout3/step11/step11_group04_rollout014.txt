"""
Propose a step‑function (list of non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------
1. Load the historically best sequence (``height_sequence_1``) if it exists,
   otherwise start from the uniform distribution.
2. **Smooth‑max mirror descent** – a fast exponentiated‑gradient phase that
   minimises a smooth approximation of the “max‑convolution”.  The smoothing
   parameter α is increased geometrically; a momentum term stabilises the
   updates.
3. **Simulated annealing** – random mass‑transfer moves are accepted
   according to a temperature schedule, allowing the search to escape
   local minima that the deterministic gradient steps may miss.
4. **Targeted hill‑climb** – repeatedly reduces the contribution of the
   currently maximal convolution shift by moving mass away from the pair
   of indices that realise this maximum.
5. **Projected sub‑gradient polish** – classic sub‑gradient descent on the
   exact objective with Euclidean projection onto the probability simplex.
6. Normalise the final vector to sum = 1 and return it as a plain Python list.

All phases respect a hard wall‑clock deadline (default 1000 s).  The code
uses only NumPy and the already‑imported ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Full linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a 1‑D array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective (assumes a ≥ 0, sum(a)≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w ≥ 0, sum(w)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    L_self: int,
                    L_grad: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smooth approximation of the maximum convolution.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    # exponential schedule for the smoothing parameter α
    alphas = np.geomspace(0.5, 2000.0, num=50)

    m = np.zeros_like(a)               # momentum buffer (log‑domain)
    beta = 0.9
    it = 0

    for alpha in alphas:
        # inner loop – stop when time is up or iteration budget exhausted
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smoothed max via soft‑max
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stabilise
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient of the smoothed max (2 * (w ⋆ a_rev))
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) momentum + exponentiated‑gradient update on the simplex
            m = beta * m + (1.0 - beta) * grad
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * m)
            a = np.maximum(a, 1e-30)        # stay strictly positive
            a /= a.sum()                    # renormalise onto simplex

            # occasional exact‑objective check
            if it % 1000 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated annealing (random mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         init_temp: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves accepted with a decreasing temperature.
    Returns the best vector seen during the annealing run.
    """
    a = a0.copy()
    conv = _conv_self(a, L_self)
    cur_score = 2.0 * n * conv.max()          # sum(a) ≃ 1, so denominator ≃ 1
    best_seq = a.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear cooling schedule
        T = max(0.0, init_temp * (1.0 - (elapsed - (start_time)) / total_time))

        # pick a source proportional to its mass (helps larger moves)
        src = rng.choice(n, p=a)
        dst = rng.integers(0, n)
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # tiny numerical renorm (occasionally)
        if (rng.integers(0, 1000) == 0):
            a_prop = np.maximum(a_prop, 0.0)
            a_prop /= a_prop.sum()

        conv_new = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv_new.max()

        accept = False
        if new_score < cur_score:
            accept = True
        elif T > 0.0:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 3 – targeted hill climbing (reduce the dominant shift)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         cur_best_seq: np.ndarray,
                         cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass away from the pair of indices that
    realise the currently maximal convolution value.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score

    while time.time() < deadline:
        conv = _conv_self(a, L_self)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # Determine all contributing pairs (i, j=i+max_idx) within bounds
        if max_idx < n:
            i_start = 0
            i_end = max_idx
        else:
            i_start = max_idx - (n - 1)
            i_end = n - 1

        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = max_idx - i_range
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()
        if total_contrib == 0.0:
            # fallback to generic random move
            src = rng.choice(n, p=a)
            dst = rng.integers(0, n)
            if src == dst or a[src] < 1e-15:
                continue
            eps = a[src] * rng.random() * 0.30
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
        else:
            # sample an index from the contributing pair distribution
            probs = contrib / total_contrib
            i_sel = rng.choice(i_range, p=probs)
            j_sel = max_idx - i_sel

            # pick the endpoint of the pair that currently holds more mass
            src = i_sel if a[i_sel] >= a[j_sel] else j_sel
            dst = rng.integers(0, n)
            if src == dst or a[src] < 1e-15:
                continue
            eps = a[src] * rng.random() * 0.30
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

        # renorm to avoid drift (seldom)
        if (rng.integers(0, 1000) == 0):
            a_prop = np.maximum(a_prop, 0.0)
            a_prop /= a_prop.sum()

        new_score = _eval_fast(a_prop, L_self)

        if new_score < cur_score - 1e-13:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Targeted hill‑climb best={best_score:.6f}")

    return best_seq, best_score

# Phase 4 – projected sub‑gradient descent (final polish)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact objective with Euclidean
    projection onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        if stagnation >= 20_000:
            break

        # active set (indices where conv equals the current maximum)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score
