"""
Propose a low‑objective step function.

The algorithm consists of three stages:
1. **Edge‑heavy warm‑start** – a distribution that puts most mass near the
   two ends of the interval (a_i ∝ (i+1)**−½ + (n−i)**−½) and is normalised
   to the simplex (∑a = 1).  This initial shape already yields a value
   substantially below the uniform bound (≈2).

2. **Smoothed gradient descent** – we minimise the smoothed version of the
   objective
        Fβ(a) = 2·n·(1/β)·log∑_k exp(β·(a*a)_k)
   where (a*a) is the linear self‑convolution.  For a given β the gradient
   of Fβ can be written in closed form and evaluated with two FFTs:

        ∇Fβ(a)_i = 2·∑_j a_j·w_{i+j},
        w_k = exp(β·c_k) / ∑_ℓ exp(β·c_ℓ),   c = a*a.

   The parameter β is increased exponentially from 5 to 200 during the run,
   slowly turning the smooth approximation into the true maximum.
   After each gradient step we project back onto the probability simplex,
   thereby keeping the sum fixed and the variables non‑negative.

3. **Light simulated‑annealing polish** – if time permits we perform a cheap
   Metropolis annealing that moves a small random packet of mass between
   indices, accepting moves that improve the exact objective.

All helper functions are top‑level, the code runs on a single core and stays
well within the 1000 s budget.  The returned list contains non‑negative
floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Identical to evaluate_sequence (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Runs a projected gradient descent on the smoothed objective.
    Returns the best iterate seen (in terms of the exact objective)
    and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta_momentum = 0.9
    step0 = 0.5
    it = 0

    # FFT length large enough for both convolutions (3·n is safe)
    L = _next_pow2(3 * n)

    while time.time() < deadline:
        it += 1
        # --------------------------------------------------------------
        # 1) self‑convolution via a single FFT of a
        # --------------------------------------------------------------
        A_fft = np.fft.rfft(a, n=L)
        conv_fft = A_fft * A_fft
        conv_full = np.fft.irfft(conv_fft, n=L)
        conv = conv_full[: 2 * n - 1]          # linear self‑convolution

        # exact objective (sum(a) is 1 by construction)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"it={it:,}  best={best_score:.6f}")

        # --------------------------------------------------------------
        # 2) temperature schedule for the soft‑max smooth approximation
        # --------------------------------------------------------------
        elapsed_frac = (time.time() - start_time) / (deadline - start_time)
        beta_start, beta_end = 5.0, 200.0
        beta = beta_start * ((beta_end / beta_start) ** elapsed_frac)

        # --------------------------------------------------------------
        # 3) soft‑max weights w_k = exp(beta·c_k) / Σ exp(beta·c_k)
        # --------------------------------------------------------------
        # stability: shift by the maximum convolution entry
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()          # length 2·n‑1

        # --------------------------------------------------------------
        # 4) gradient of the smoothed objective:
        #    ∇ = 2·(a * w_rev) taken at appropriate indices
        # --------------------------------------------------------------
        W_fft = np.fft.rfft(w[::-1], n=L)      # w reversed for the required sum
        grad_fft = A_fft * W_fft
        conv_rev_full = np.fft.irfft(grad_fft, n=L)
        conv_rev = conv_rev_full[: len(a) + len(w) - 1]   # length 3·n‑2

        start_idx = 2 * n - 2
        grad = 2.0 * conv_rev[start_idx : n - 2 : -1]    # shape (n,)

        # --------------------------------------------------------------
        # 5) momentum update + diminishing step size
        # --------------------------------------------------------------
        momentum = beta_momentum * momentum + (1.0 - beta_momentum) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # --------------------------------------------------------------
        # 6) projection onto the simplex (keeps a ≥ 0 and Σa = 1)
        # --------------------------------------------------------------
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as the previous code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis annealing that moves a small random packet of mass
    between indices. Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 40_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)          # choose source proportionally to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance rule
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a short safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1) Edge‑heavy warm start (heavy mass at the two ends).
    # --------------------------------------------------------------
    i = np.arange(n, dtype=float)
    # weights ∝ (i+1)^‑½ + (n‑i)^‑½  (large near edges, small in centre)
    init = (i + 1) ** -0.5 + (n - i) ** -0.5
    init = np.maximum(init, 0.0)
    if init.sum() == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = init / init.sum()                # normalise to the simplex

    # --------------------------------------------------------------
    # 2) If a global starter (height_sequence_1) exists and is better, use it.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # adapt length by truncating or padding with zeros
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.concatenate([starter,
                                              np.zeros(n - starter.size)])
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
                if cur < _objective(a0):
                    a0 = starter.copy()
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) Smoothed gradient descent.
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:            # keep a few seconds for polishing
        a_opt, best_score = _smooth_optimize(a0, deadline, start_time)
        a0 = a_opt
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Light annealing polish (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt2, best_score2 = _anneal(a0, deadline, start_time)
        if best_score2 < best_score:
            a0 = a_opt2
            print(f"[{time.time() - start_time:.2f}s] After anneal best={best_score2:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation & conversion to a plain list.
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
