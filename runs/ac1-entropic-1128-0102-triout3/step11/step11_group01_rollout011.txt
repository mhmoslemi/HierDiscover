"""
propose_candidate
=================

A fast, gradient‑based optimiser for the non‑negative step‑function problem.

* Uses a smooth approximation of the *max* in the objective (soft‑max)
  which yields a well‑behaved gradient.
* Gradient is computed completely in the Fourier domain, i.e.  O(n log n)
  per iteration.
* Momentum + a decaying step‑size together with Euclidean projection onto
  the probability simplex keeps the iterates feasible.
* The schedule for the soft‑max temperature β starts small (very smooth)
  and is increased gradually – a deterministic annealing scheme.
* After the gradient phase a very cheap simulated‑annealing polish
  (mass transfers between random positions) is run for the remaining time.
* The routine works with an up‑sampled version of the globally stored
  ``height_sequence_1`` (if present) and falls back to a simple
  “tent‑with‑plateau’’ shape otherwise.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear self‑convolution a*a (full length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {w ≥ 0, Σ w = 1} using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generator (fallback if no global starter is available)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric “tent‑with‑plateau’’ shape, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    # keep the plateau even length for symmetry
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p               # 0 … 1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # Guard against off‑by‑one caused by rounding
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Upsampling helper – linear interpolation to a larger canvas
# ----------------------------------------------------------------------
def _upscale(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``n``."""
    if seq.shape[0] == n:
        return seq.copy()
    x_old = np.arange(seq.shape[0])
    x_new = np.linspace(0, seq.shape[0] - 1, n)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective (Fourier implementation)
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of 2·n·softmax(conv(a,a)) w.r.t. ``a``.
    Returns (grad, current_exact_score).
    """
    n = a.shape[0]
    # FFT length large enough for all convolutions (a*a and w*a)
    L = _next_pow2(3 * n - 2)

    # FFT of a (single transform – reused)
    A_fft = np.fft.rfft(a, n=L)

    # --- exact self‑convolution (needed for the soft‑max) ---
    conv_full = np.fft.irfft(A_fft * A_fft, n=L)
    conv = conv_full[: 2 * n - 1]          # linear convolution a*a

    # exact objective (for monitoring)
    cur_score = 2.0 * n * conv.max()      # sum(a)=1 by construction

    # --- soft‑max weights ---
    maxc = conv.max()
    # stable exponentiation
    exp_vals = np.exp(beta * (conv - maxc))
    w = exp_vals / exp_vals.sum()         # length 2·n‑1, sum to 1

    # --- gradient via FFT (w * a) ---
    W_fft = np.fft.rfft(w, n=L)
    grad1_full = np.fft.irfft(W_fft * A_fft, n=L)
    grad1 = grad1_full[n - 1: n - 1 + n]          # term a_{k‑s}

    # reversed weights give the other term a_{k+s}
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=L)
    grad2_full = np.fft.irfft(W_rev_fft * A_fft, n=L)
    grad2 = grad2_full[2 * n - 2: 2 * n - 2 + n]  # term a_{k+s}

    grad = grad1 + grad2                       # ∂softmax/∂a
    grad *= 2.0 * n                            # chain rule for   2·n·softmax

    return grad, cur_score


# ----------------------------------------------------------------------
# Projected (accelerated) gradient descent with deterministic annealing
# ----------------------------------------------------------------------
def _gradient_descent(a0: np.ndarray,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Runs a momentum based projected gradient descent on the smooth‑max
    surrogate of the objective.  Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)

    # hyper‑parameters (tuned experimentally)
    step0 = 1.2           # base step‑size
    beta_mom = 0.9        # momentum coefficient
    beta_start = 5.0      # soft‑max temperature at the start
    beta_end = 120.0      # temperature at the end of the run
    max_iter = 10 ** 7    # just a huge upper bound; real stop is time‑limit

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # increase β linearly with the iteration count (deterministic annealing)
        beta_cur = beta_start + (beta_end - beta_start) * min(1.0, it / 4000.0)

        grad, cur_score = _smooth_max_gradient(a, beta_cur)

        # momentum + diminishing step‑size
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # keep feasibility
        a = _project_simplex(a)

        # keep track of the best exact score
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} "
                  f"score={cur_score:.6f}")

        # occasional cheap objective check (every few seconds) to catch
        # any regression caused by the smooth approximation
        if it % 250 == 0:
            exact = _objective(a)
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] exact GD iter={it:,} "
                      f"score={exact:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (as in the reference implementation)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers guided by Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # max steps derived from remaining time (≈ 10⁵–2·10⁵ works well)
    max_steps = 250_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # source chosen proportionally to its mass (more mass → more moves)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (≤ 25 %) of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low value of
    ``evaluate_sequence``.  The routine respects the 1000 s budget,
    uses gradient descent with deterministic annealing and a final
    polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger canvas brings us closer to the continuous
    # optimum (the discretisation error is O(1/n)).
    # --------------------------------------------------------------
    n = 65536                         # power of two → very fast FFTs

    # --------------------------------------------------------------
    # 0) initialise from the globally stored starter if possible.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = _upscale(starter, n)
            starter = _project_simplex(starter)          # normalise to sum 1
            cur = _objective(starter)
            print(f"[{time.time() - start_time:.2f}s] Global starter score = {cur:.6f}")
            init_seq = starter
        else:
            raise ValueError
    except Exception:
        # fallback: a simple analytic shape
        init_seq = _make_shape(n, p=1.0, alpha=0.25)
        cur = _objective(init_seq)
        print(f"[{time.time() - start_time:.2f}s] Fallback starter score = {cur:.6f}")

    # --------------------------------------------------------------
    # 1) (optional) quick random sweep to possibly improve the start.
    #    Only a few hundred random shapes are tried – cheap compared to
    #    the heavy gradient phase.
    # --------------------------------------------------------------
    best_seq = init_seq.copy()
    best_score = cur
    rng = np.random.default_rng()
    for _ in range(300):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.3, 2.5)
        alpha = rng.uniform(0.0, 0.5)
        cand = _make_shape(n, p, alpha)
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Random sweep improved to {sc:.6f}")

    # --------------------------------------------------------------
    # 2) Gradient‑descent refinement (deterministic annealing).
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                               # leave a few seconds for polishing
        sub_deadline = time.time() + 0.80 * remaining
        refined_seq, refined_score = _gradient_descent(best_seq,
                                                      sub_deadline,
                                                      start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Gradient refined score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish for the final few seconds.
    # --------------------------------------------------------------

            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
