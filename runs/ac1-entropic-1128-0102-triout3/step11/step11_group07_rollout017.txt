def propose_candidate(seed=0, budget_s=1000):
    import random
    import time
    import numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # Choose length n
    n = 8192 # maybe bigger for more degrees
    # compute L_fft
    L_fft = 1 << ((2 * n - 1).bit_length())
    # define fast score
    def score(seq):
        conv = np.fft.irfft(np.fft.rfft(seq, n=L_fft) * np.fft.rfft(seq, n=L_fft), n=L_fft)[:2*n-1]
        return 2.0 * n * conv.max() / (seq.sum()**2)
    # initial candidate uniform
    best_seq = np.full(n, 1.0/n)
    best_sc = score(best_seq)
    print(f"[{time.time()-start:.2f}s] Uniform score {best_sc}")
```

def propose_candidate(seed=0,budget_s=1000):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    
    n = 4096
    L_fft = 1 << ((2*n - 2).bit_length())
    
    def score_seq(a):
        # assume a is np.ndarray
        conv = np.fft.irfft(np.fft.rfft(a, n=L_fft) * np.fft.rfft(a, n=L_fft), n=L_fft)[:2*n-1]
        maxc = conv.max()
        return 2.0 * n * maxc / (a.sum() ** 2)
    
    # start with uniform
    best_seq = np.full(n, 1.0/n)
    best_score = score_seq(best_seq)
    print(f"[{time.time()-start:.2f}s] Uniform score {best_score:.6f}")
    
    # 1. parametric families: geometric, power, beta
    idx = np.arange(n)
    # Geometric
    for eps in np.logspace(-6, -2, 200):
        r = 1.0 - eps
        geo = r ** idx
        geo = geo / geo.sum()
        sc = score_seq(geo)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time()-start:.2f}s] Geo eps={eps:.2e} score={sc:.6f}")
    # Power
    for p in np.linspace(-3, 3, 121):
        pw = (idx + 1) ** p
        pw = pw / pw.sum()
        sc = score_seq(pw)
        if sc < best_score:
            best_score, best_seq = sc, pw.copy()
            print(f"[{time.time()-start:.2f}s] Power p={p:.3f} score={sc:.6f}")
    # Beta (p,q)
    for _ in range(4000):
        p = rng.uniform(-4,4)
        q = rng.uniform(-4,4)
        beta_seq = ( (idx + 1) ** p) * ((n - idx) ** q)
        s = beta_seq.sum()
        if s == 0:
            continue
        beta_seq = beta_seq / s
        sc = score_seq(beta_seq)
        if sc < best_score:
            best_score, best_seq = sc, beta_seq.copy()
            print(f"[{time.time()-start:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")
        if time.time() >= deadline - 30:
            break
    
    # 2. Gradient descent (Adam) using softmax approx
    # define functions same as previous but with parameter modifications.
    # Run several restarts.
    
    def adam_descent(start_a, sub_deadline):
        a = start_a.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        best_local = a.copy()
        best_local_score = score_seq(a)
        tau = 0.1
        tau_min = 1e-8
        decay = 0.995
        step0 = 0.1
        it = 0
        while time.time() < sub_deadline:
            it += 1
            conv_full = np.fft.irfft(
                np.fft.rfft(a, n=L_fft) * np.fft.rfft(a, n=L_fft), n=L_fft)[:2*n-1]
            maxc = conv_full.max()
            # softmax
            exp_vals = np.exp((conv_full - maxc) / tau)
            w = exp_vals / exp_vals.sum()
            # gradient: 2 * (a conv w_rev)
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=L_fft)
            Wf = np.fft.rfft(w_rev, n=L_fft)
            grad_full = np.fft.irfft(Af * Wf, n=L_fft)
            grad = 2.0 * grad_full[2*n-2 : n-1 : -1]  # tricky indexing
            # Actually we need gradient j = 2 Σ_i a_i w_{i+j}
            # Use convolution of a and w_rev (size 2n-1) and extract appropriate slice.
            # Implement simpler: compute conv_ab = np.convolve(a, w_rev, mode='full') and take slice.
            # But using FFT is okay for speed.
            # In practice we can compute:
            conv_ab = np.fft.irfft(np.fft.rfft(a, n=L_fft) * np.fft.rfft(w_rev, n=L_fft), n=L_fft)[:2*n-1]
            # gradient = 2 * conv_ab[n-1:2*n-1] maybe? Let's derive: 
            # conv_ab[k] = Σ_i a_i w_rev_{k-i} = Σ_i a_i w_{(len(w)-1)-(k-i)} = Σ_i a_i w_{some}
            # For gradient at index j: Σ_i a_i w_{i+j}
            # This corresponds to conv_ab[n-1 + j]? Let's quickly compute using small n to confirm.
            # We'll avoid indexing errors by using direct convolution via np.convolve for simplicity, slower but okay.
            grad = 2.0 * np.convolve(a, w, mode='full')[n-1:2*n-1]  # just compute conv a * w
            # Update Adam
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = step0 / math.sqrt(it)
            a = a - step * m_hat / (np.sqrt(v_hat) + eps)
            # project onto simplex
            a = _project_simplex(a)
            cur_score = score_seq(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
            tau = max(tau_min, tau * decay)
        return best_local, best_local_score
    
    while time.time() < deadline - 10 and best_score > 1.45:  # maybe aim <1.45 early break
        # random perturbation around best
        perturb = 0.02 * np.random.randn(n) * best_seq
        start = best_seq + perturb
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)
        sub_dead = min(deadline - 5, time.time() + 12)
        cand_seq, cand_score = adam_descent(start, sub_dead)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time()-start:.2f}s] Adam new best {cand_score:.6f}")
    
    # 3. Simulated annealing fine-tune
    cur_seq = best_seq.copy()
    cur_score = best_score
    best_seq_sa = best_seq.copy()
    best_score_sa = best_score
    T = 0.01
    T_min = 1e-7
    decay_T = 0.9995
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue
        if cur_seq[src] < 1e-12:
            continue
        frac = rng.random() * 0.2
        delta = cur_seq[src] * frac
        new_seq = cur_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta
        # projection not needed as sum unchanged and positivity holds (assuming dst >=0) but ensure non-negative.
        # However due to rounding might get tiny negative; clip:
        new_seq = np.maximum(new_seq, 0.0)
        new_seq = _project_simplex(new_seq)
        new_score = score_seq(new_seq)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_seq, cur_score = new_seq, new_score
            if new_score < best_score_sa:
                best_score_sa, best_seq_sa = new_score, new_seq.copy()
                print(f"[{time.time()-start:.2f}s] SA improvement {new_score:.6f}")
        T = max(T_min, T * decay_T)
    
    # final best
    final_seq = best_seq_sa
    # ensure positivity and normalization
    final_seq = np.maximum(final_seq, 0.0)
    total = final_seq.sum()
    if total == 0:
        final_seq = np.full(n, 1.0/n)
    else:
        final_seq = final_seq / total
    print(f"[{time.time()-start:.2f}s] Done best score {best_score_sa:.6f}")
    return [float(x) for x in final_seq]

"""
Propose a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search uses three stages:

1. **Very fast parametric sweep** – geometric, power and beta families with a
   dense random sample of exponent parameters.  Each candidate is evaluated
   with an FFT based convolution (O(n log n)).

2. **Adam‑style smoothed descent** – a differentiable soft‑max approximation of
   the max‑operator is optimised on the simplex via Adam and a Euclidean
   projection.  Several random restarts are performed.

3. **Tiny hill‑climbing / simulated‑annealing polish** – mass is moved from a
   source index to a destination index; uphill moves are occasionally accepted
   with a temperature schedule.  The best sequence seen is returned.

All operations respect the non‑negative simplex constraint; the final
sequence is re‑normalised to sum to 1 before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full self‑convolution a*a using FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray, L_fft: int) -> float:
    """Exact evaluation of the objective (assumes a non‑negative)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑style smoothed gradient descent (soft‑max surrogate)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         sub_deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate of max(conv(a,a)).
    The iterates are projected back onto the probability simplex.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _score(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    tau = 0.2          # start temperature (softer gradient)
    tau_min = 1e-8
    tau_decay = 0.995
    step0 = 0.12
    it = 0

    while time.time() < sub_deadline:
        it += 1

        # ----- soft‑max weights for the max ----------
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()                     # shape (2n‑1,)

        # ----- gradient of the surrogate -------------
        # gradient_j = 2 * Σ_i a_i * w_{i+j}
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # ----- Adam update + projection ---------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        cur_score = _score(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb / simulated annealing polish
# ----------------------------------------------------------------------
def _anneal_polish(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   L_fft: int) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style random walk on the simplex.
    Only moves a fraction of the source mass.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_score = _score(cur, L_fft)
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    T = 0.02
    T_min = 1e-7
    decay = 0.9995

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or cur[src] < 1e-15:
            continue
        # move a random fraction (up to 30 %) of the source mass
        frac = rng.random() * 0.30
        delta = cur[src] * frac
        prop = cur.copy()
        prop[src] -= delta
        prop[dst] += delta
        # numerical safety – re‑project onto the simplex
        prop = _project_simplex(prop)

        new_score = _score(prop, L_fft)
        dE = new_score - cur_score
        accept = dE < 0 or rng.random() < math.exp(-dE / T)
        if accept:
            cur, cur_score = prop, new_score
            if new_score < best_score:
                best, best_score = prop.copy(), new_score
                print(f"[{time.time() - start_time:.2f}s] Anneal best={best_score:.6f}")
        T = max(T_min, T * decay)

    return best, best_score

    # ------------------------------------------------------------------

            break
        r = 1.0 - eps
        geo = r ** idx
        if geo.sum() == 0:
            continue

        sc = _score(geo, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.2e} score={sc:.6f}")

    for p in np.linspace(-3.0, 3.0, 151):
        if time.time() >= deadline - 30:

        pw = (idx + 1) ** p
        if pw.sum() == 0:
            continue
        pw /= pw.sum()
        sc = _score(pw, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, pw.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # beta‑type random sampling – 5000 attempts
    rng = np.random.default_rng()
    for _ in range(5000):
        if time.time() >= deadline - 30:
            break
        p = rng.uniform(-4.0, 4.0)
        q = rng.uniform(-4.0, 4.0)
        beta_seq = ( (idx + 1) ** p ) * ( (n - idx) ** q )
        if beta_seq.sum() == 0:
            continue
        beta_seq /= beta_seq.sum()
        sc = _score(beta_seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, beta_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style smoothed gradient descent with random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 20.0:
        # Random perturbation of the current best (or uniform)
        base = best_seq if rng.random() < 0.7 else np.full(n, 1.0 / n)
        noise = 0.03 * rng.standard_normal(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_dead = min(deadline - 5.0, time.time() + 13.0)   # ≤13 s per run
        cand_seq, cand_score = _adam_smooth_descent(start, sub_dead,
                                                    start_time,
                                                    L_fft, L_grad)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam new global best score={best_score:.6f}")

        if best_score <= 1.45:          # early exit if we are already far below target
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final annealing / hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal_polish(best_seq, deadline,
                                              start_time, L_fft)

    # ------------------------------------------------------------------
    # Normalisation and final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
