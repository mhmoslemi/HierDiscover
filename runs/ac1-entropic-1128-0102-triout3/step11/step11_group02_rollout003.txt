def _smoothmax_grad(a, lam):
    n = a.shape[0]
    L = _next_pow2(3*n)   # ensure enough for conv with rev
    # compute conv = a*a
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]   # auto
    # compute w stable softmax
    max_c = conv.max()
    exp = np.exp(lam * (conv - max_c))
    w = exp / exp.sum()
    # compute gradient of smoothmax:
    # grad_M_i = 2 * sum_k w_k * a_{k-i}
    rev = a[::-1]  # length n
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)  # length L
    # slice segment for i=0..n-1:
    grad = 2.0 * conv_wr[n-1: n-1+n]
    return grad, conv, w

"""
Optimise a non‑negative step‑function a∈ℝⁿ (∑a=1) to minimise

        2·n·max(conv(a,a))

where conv(a,a) is the linear self‑convolution.
The algorithm consists of three phases

1. **Warm‑start** – a collection of simple analytic shapes
   (tent, sine, half‑circle, Gaussian) is generated, the best of
   them (according to the exact objective) is taken as the start
   vector.

2. **Projected Adam optimisation** – at every iteration the *k*
   largest convolution entries are identified (k≈2‑3 % of the
   length).  The sub‑gradient of the averaged top‑k entries is
   obtained *via* a single FFT‑based convolution, which makes the
   update O(n log n).  Adam’s adaptive step‑size together with a
   Euclidean projection onto the probability simplex keeps the
   vector non‑negative and normalised.

3. **Light simulated‑annealing polish** – a cheap Metropolis walk
   moves a random packet of mass between two indices; it is run
   on the remaining time budget to escape possible shallow local
   minima.

All heavy work is performed with NumPy’s FFT which is fast for
powers‑of‑two; the chosen length n=16384 gives a good trade‑off
between expressive power and runtime.  The whole search respects
the 1000 s budget and returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a normalised (∑a=1) vector.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform simplex point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient of the averaged top‑k convolution entries
# ----------------------------------------------------------------------
def _topk_gradient(a: np.ndarray,
                   top_k: int,
                   L: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns the gradient of the *average* of the `top_k` largest
    convolution values together with the convolution itself.
    The gradient is computed analytically:
          g_i = 2·average_{k∈top} a_{k‑i}
    and obtained efficiently by an FFT‑based convolution.
    """
    n = a.shape[0]

    # 1) self‑convolution (full length 2·n‑1)
    conv = _conv_self(a, L)

    # 2) build the weight vector w (average over the selected indices)
    if top_k >= conv.size:
        w = np.full(conv.shape, 1.0 / conv.size, dtype=float)
    else:
        # indices of the top‑k entries (unsorted)
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros_like(conv, dtype=float)
        w[top_idx] = 1.0 / top_k   # average → sum_{k∈top} / top_k

    # 3) gradient via convolution w * a_rev
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Revf = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(Wf * Revf, n=L)
    # extract the n entries that correspond to g_i (see analysis)
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam optimiser (averaged top‑k sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   top_k: int = 150,
                   lr: float = 0.025,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using the gradient of the
    averaged top‑k convolution entries.
    Returns the best iterate encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)                # size sufficient for all FFTs

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)                  # 1st‑moment estimate
    v = np.zeros_like(a)                  # 2nd‑moment estimate

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- gradient of the averaged top‑k entries ----
        grad, conv = _topk_gradient(a, top_k, L)

        # objective (max‑convolution) for bookkeeping
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # ---- Adam update (the objective is 2·n·max, the gradient of the
        #      averaged top‑k entry approximates the sub‑gradient of max) ----
        # Scale the gradient to the true objective’s magnitude.
        grad_obj = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ---- projection onto the probability simplex ----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same as in the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk that moves a random packet of mass between two
    indices.  Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 40_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule (very fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)            # pick a source proportionally to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Warm‑starting shapes
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a small collection of analytically simple candidate vectors."""
    x = np.arange(n, dtype=float) / (n - 1)               # normalised coordinate in [0,1]

    # 1) symmetric linear tent
    tent = np.minimum(x, 1.0 - x)

    # 2) half‑cosine (raised cosine) – zero at the ends, smooth interior
    cosine = np.sin(np.pi * x)                # sin(π·x) already zero at the boundaries

    # 3) half‑circle (Archimedes) – √(1‑(2x‑1)²)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

    # 5) power‑tent (flatter than the linear tent)
    p = 0.7
    power_tent = np.minimum(x, 1.0 - x) ** p

    candidates = [tent, cosine, circ, gauss, power_tent]
    # normalise each to sum to 1 (simple probability simplex)
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals
