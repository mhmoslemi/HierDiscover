"""
Propose a low‑objective step sequence.

The algorithm works in three stages:

1️⃣  **Rich parametric sweep** – A fine grid over a three‑parameter
    family  
        a_i ∝ (x_i·(1−x_i)+ε)^{−α}·(1+γ·x_i·(1−x_i))  
    (α≈0.5, ε tiny, |γ|≤0.5) is evaluated with an FFT‑based convolution.
    The best candidate is kept.

2️⃣  **Exponentiated‑gradient refinement** –  The exact objective
    `2·n·max(conv)/sum²` equals `2·n·max(conv)` when the vector is
    normalised to sum = 1.  The sub‑gradient of the maximum convolution
    can be written as a single correlation, therefore each iteration
    needs only two FFTs.  An exponentiated‑gradient (mirror descent on
    the simplex) update is performed with a decreasing step‑size; the
    iterate with the smallest exact objective is stored.

3️⃣  **Light simulated‑annealing polish** –  Random “mass‑transfer”
    moves are tried while time permits; any improvement is kept.

All operations respect the simplex constraint (non‑negative, sum = 1)
and the whole search stops before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _generate_shape(alpha: float, eps: float, gamma: float,
                    x: np.ndarray) -> np.ndarray:
    """
    Produce a normalised vector of length len(x) belonging to the
    three‑parameter family described in the doc‑string.
    """
    base = (x * (1.0 - x) + eps) ** (-alpha)
    mod = 1.0 + gamma * (x * (1.0 - x))
    w = base * mod
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a very low evaluation
    score under the given time budget.  Returns a plain Python list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two works well with the FFTs.
    # Using a larger n than the previous solution gives a finer
    # discretisation of the continuous problem.
    # ------------------------------------------------------------------
    n = 8192
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions

    # ------------------------------------------------------------------
    # 1) Fine sweep over the 3‑parameter Beta‑family
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    alphas = np.linspace(0.48, 0.52, 81)                # step ≈5·10⁻⁴
    epsilons = [0.0, 1e-10, 1e-8, 1e-6, 1e-4]
    gammas = np.linspace(-0.5, 0.5, 21)                # step 0.05

    # leave about 200 s for the remaining phases
    sweep_deadline = deadline - 200.0

    for alpha in alphas:
        for eps in epsilons:
            for gamma in gammas:
                a = _generate_shape(alpha, eps, gamma, x)
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                if time.time() >= sweep_deadline:
                    break
            if time.time() >= sweep_deadline:
                break
        if time.time() >= sweep_deadline:
            break

    # ------------------------------------------------------------------
    # 2) Optionally accept a starter supplied by the harness
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                s = starter.sum()
                if s > 0.0:
                    starter = starter / s
                    cur = _objective(starter)
                    if cur < best_score:
                        best_score = cur
                        best_seq = starter.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # If everything failed, fall back to the uniform distribution.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 3) Exponentiated‑gradient (mirror‑descent) optimisation of max‑conv
    # ------------------------------------------------------------------
    a = best_seq.copy()
    L = _next_pow2(3 * n)            # length sufficient for both convolutions
    iteration = 0

    while time.time() < deadline - 2.0:
        iteration += 1

        # ----- compute self‑convolution (via FFT) --------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        M = conv.max()

        # ----- indices where the maximum is attained -----------------
        tol = max(1e-15, M * 1e-12)
        idx_max = np.where(conv >= M - tol)[0]

        # ----- sub‑gradient of the max‑convolution ------------------
        ind = np.zeros(2 * n - 1, dtype=np.float64)
        ind[idx_max] = 1.0
        a_rev = a[::-1]
        ind_fft = np.fft.rfft(ind, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        corr_full = np.fft.irfft(ind_fft * a_rev_fft, n=L)
        grad = 2.0 * corr_full[n - 1:n - 1 + n]   # sub‑gradient of M

        # ----- step‑size schedule (decreasing) ----------------------
        eta = 0.5 / np.sqrt(iteration)            # can be tuned

        # ----- exponentiated‑gradient update (keeps us on the simplex)
        log_update = -eta * grad
        log_update = np.clip(log_update, -50, 50)   # numerical safety
        a_new = a * np.exp(log_update)
        total = a_new.sum()
        if total == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= total

        # ----- evaluate the new max‑convolution ----------------------
        a_new_fft = np.fft.rfft(a_new, n=L)
        conv_new_full = np.fft.irfft(a_new_fft * a_new_fft, n=L)
        M_new = conv_new_full[: 2 * n - 1].max()
        new_score = float(2.0 * n * M_new)        # sum(a_new) == 1

        # ----- keep the best exact objective ever seen ---------------
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[{time.time() - start:.2f}s] iter={iteration:,} "
                  f"best={best_score:.6f}")

        # ----- accept the move if it improves the max ----------------
        if M_new <= M:
            a = a_new
        else:
            # try a smaller step in case the first one was too aggressive
            for factor in (0.5, 0.25):
                eta2 = eta * factor
                log_update = -eta2 * grad
                log_update = np.clip(log_update, -50, 50)
                a_candidate = a * np.exp(log_update)
                a_candidate /= a_candidate.sum()
                cand_fft = np.fft.rfft(a_candidate, n=L)
                M_cand = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1].max()
                if M_cand <= M:
                    a = a_candidate
                    break

    # ------------------------------------------------------------------
    # 4) Very light simulated‑annealing polish (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        rng = np.random.default_rng()
        cur_score = best_score
        a = best_seq.copy()
        # keep a copy of the best local solution
        best_local = a.copy()
        best_local_score = cur_score

        while time.time() < deadline:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # move at most 5 % of the source mass
            delta = a[src] * rng.random() * 0.05
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta

            a_prop_fft = np.fft.rfft(a_prop, n=L)
            M_prop = np.fft.irfft(a_prop_fft * a_prop_fft, n=L)[: 2 * n - 1].max()
            score_prop = float(2.0 * n * M_prop)

            if score_prop < cur_score:
                a = a_prop
                cur_score = score_prop
                if score_prop < best_local_score:
                    best_local_score = score_prop
                    best_local = a.copy()
                    print(f"[{time.time() - start:.2f}s] anneal best={score_prop:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
