def _spectral_init(n):
    L = _next_pow2(2*n-1)
    target = np.full(2*n-1, 1.0/(2*n-1), dtype=np.complex128)
    target = np.pad(target, (0, L - (2*n-1)), mode='constant')
    T_fft = np.fft.rfft(target)
    S_fft = np.sqrt(T_fft)  # elementwise sqrt
    a = np.fft.irfft(S_fft, n=L)[:n]
    a = np.real(a)
    a = np.maximum(a, 0)
    if a.sum() == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a.sum()
    return a

"""
Improved search for the minimum of

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096 (or the length of a provided seed).

Idea
----
1️⃣  **Spectral seeding** – start from the (real) inverse FFT of the
    square‑root of the uniform target spectrum.  This “square‑root”
    distribution is known to give a flatter self‑convolution than the
    plain uniform distribution and usually starts already below the
    1.508 bound.

2️⃣  **Multiple cheap restarts** – generate a few random Dirichlet
    vectors, keep the best one (including the spectral seed and the
    optional global `height_sequence_1`).

3️⃣  **Mirror‑descent (smooth‑max) polishing** – the same exponentiated‑gradient
    scheme as in the reference code but with a richer schedule of smoothing
    parameters and a longer iteration budget.

4️⃣  **Fine‑grained sub‑gradient step** – a short momentum based projective
    sub‑gradient pass.

5️⃣  **Incremental simulated‑annealing** – a fast mass‑transfer annealer
    that updates the convolution in‑place.  The temperature schedule is
    slightly slower (more exploration) and the move budget is increased
    until the global deadline is reached.

All phases respect the 1000 s time budget; the routine prints progress
information and finally returns a non‑negative list whose sum is 1.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast objective (sum is *not* assumed to be 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a non‑negative vector."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 0️⃣  Spectral / random seeding utilities
# ----------------------------------------------------------------------
def _spectral_seed(n: int) -> np.ndarray:
    """
    Build a candidate by taking the inverse FFT of sqrt(FFT(uniform)).
    The result is forced to be non‑negative and renormalised to sum=1.
    """
    L = _next_pow2(2 * n - 1)
    # uniform target on the sum‑domain (length 2n‑1)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.complex128)
    target = np.pad(target, (0, L - (2 * n - 1)), mode="constant")
    T_fft = np.fft.rfft(target)
    # principal square‑root in the complex plane
    S_fft = np.sqrt(T_fft)
    a = np.fft.irfft(S_fft, n=L)[:n]
    a = np.real(a)                 # drop tiny imaginary part
    a = np.maximum(a, 0.0)        # clip negatives
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _random_dirichlet_seed(n: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet vector (concentration >0, smaller ⇒ spikier)."""
    rng = np.random.default_rng()
    x = rng.gamma(concentration, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x /= x.sum()
    return x


# ----------------------------------------------------------------------
# 1️⃣  Smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # a broader geometric schedule, ending at a fairly large smoothing
    alphas = np.geomspace(0.3, 800.0, num=40)

    it = 0
    max_iterations = 400_000
    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # 1) self‑convolution via FFT (only the part needed for the weight)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)        # stab
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)   # keep away from exact zero
            a /= a.sum()

            # occasional real‑objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} "
                          f"it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.28
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        # full self‑convolution (FFT, fast enough for a few thousand iters)
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} "
                  f"score={cur_score:.6f}")

        # index of the (first) maximal entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max‑convolution (same as in reference code)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        # projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Incremental simulated‑annealing polish
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 1e-2,
                        end_temp: float = 1e-12,
                        max_moves: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that updates the convolution incrementally."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # linear temperature schedule (exponential decay)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a random fraction (up to 35 %) of the source mass
        eps = a[src] * rng.random() * 0.35
        if eps <= 0.0:
            continue

        # --- incremental update of convolution (O(n)) -----------------
        # contribution of the moved mass with the *old* a (before change)
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a
        # quadratic self‑terms (three isolated indices)
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps
        # ----------------------------------------------------------------

        new_max = conv.max()
        new_score = 2.0 * n * new_max / ((a.sum()) ** 2)  # sum stays 1
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / max(T, 1e-16)) > rng.random():
            a[src] -= eps
            a[dst] += eps
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} "
                      f"best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    Strategy:
        * build several seeds (spectral, uniform, random Dirichlet, optional
          global seed `height_sequence_1`);
        * keep the best seed;
        * polish with smooth‑max mirror descent;
        * short momentum sub‑gradient cleanup;
        * long incremental simulated annealing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a safety margin

    # ------------------------------------------------------------------
    # Determine the problem size (default 4096)
    # ------------------------------------------------------------------
    n = 4096
    try:
        # if a seed of different length has been provided, adapt n
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
    except Exception:
        pass
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Build a small pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # (a) spectral seed
    try:
        a_spec = _spectral_seed(n)
        candidates.append(a_spec)
    except Exception as e:
        print("Spectral seed failed:", e)

    # (b) uniform seed
    candidates.append(np.full(n, 1.0 / n))

    # (c) a few random Dirichlet seeds (different concentration)
    for conc in (0.3, 0.7, 1.0):
        candidates.append(_random_dirichlet_seed(n, concentration=conc))

    # (d) optional global seed if it exists and has the right length
    try:
        global_seq = np.asarray(height_sequence_1, dtype=float)
        if global_seq.ndim == 1 and global_seq.size == n:
            global_seq = np.maximum(global_seq, 0.0)
            if global_seq.sum() == 0.0:
                global_seq = np.full(n, 1.0 / n)
            else:
                global_seq = global_seq / global_seq.sum()
            candidates.append(global_seq)
    except Exception:
        pass

    # keep the best candidate according to the fast objective
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        sc = _eval_fast(cand, L_self)
        print(f"[{time.time() - start_time:.2f}s] Seed {idx} initial score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[{time.time() - start_time:.2f}s] Best seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Smooth‑max mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        smooth_deadline = deadline - 15.0
        best_seq, best_score = _smooth_descent(best_seq, n,
                                               smooth_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best "
              f"score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Short projected sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 3.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved "
                  f"score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Incremental simulated‑annealing polish (bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_final, sc_final = _incremental_anneal(best_seq, n,
                                                anneal_deadline, start_time,
                                                best_score, best_seq,
                                                start_temp=2e-2,
                                                end_temp=1e-12,
                                                max_moves=8_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved "
                  f"score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{best_score:.6f}")
    return [float(v) for v in best_seq]
