        exps = np.exp(beta * (conv - tmax))
        p = exps / exps.sum()
        # gradient
        # correlation of p with a
        g = 2 * _correlate(p, a)  # returns length n gradient
        # Adam update

"""
Propose a low‑score step function.

Algorithm
---------
1. **Parameter sweep** –  Build a few simple families of non‑negative
   vectors (symmetric arcsine‑type, left‑hand power‑law, random
   Dirichlet draws).  Each candidate is normalised to sum 1 and its
   exact objective is evaluated with an FFT‑based convolution.  The best
   candidate of this phase is kept.

2. **Smoothed max‑descent** –  Instead of the non‑smooth sub‑gradient
   of ``max(conv)`` we minimise the smooth approximation

           fβ(a) = (1/β)·log Σk exp(β·convk(a))

   for a gradually increasing β.  The gradient of fβ is a weighted
   correlation of the current probability vector with the soft‑max
   weights.  An Adam optimiser (projected onto the probability simplex)
   drives the search.  The schedule starts with a small β (very smooth)
   and multiplies it by 1.02 each iteration until a cap βmax≈30.

3. **Light annealing polish** –  After the gradient phase a few
   thousand cheap “mass‑transfer’’ moves are attempted; a move is kept
   if it improves the objective (or is accepted with a tiny Metropolis
   probability).  The final vector is projected onto the simplex and
   returned as a plain list.

All steps respect a 1000 s wall‑time budget and print progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (exact verifier’s function without the security checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the verifier:
       2·n·max(conv(seq,seq)) / (sum(seq)**2) .
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑descent (Adam optimiser on the soft‑max approximation)
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Minimise a smooth approximation of the max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    beta_soft = 1.0          # start very smooth
    beta_soft_max = 30.0
    lr0 = 0.3                # base learning rate

    it = 0
    # pre‑compute a safe FFT length for all convolutions (a length n,
    # soft‑max weights length 2n‑1, their convolution length 3n‑2)
    L = _next_pow2(3 * n)

    while time.time() < deadline:
        it += 1

        # FFT of the current vector (re‑used for the self‑convolution and
        # for the correlation with the soft‑max weights)
        A = np.fft.rfft(a, n=L)

        # self‑convolution (needed only for the soft‑max weights)
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        # soft‑max over the convolution values
        maxc = conv.max()
        # scale to avoid overflow
        expo = beta_soft * (conv - maxc)
        w = np.exp(expo)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient = 2 * correlation(w, a)
        # correlation can be obtained via a * (reverse w) convolution
        w_rev = w[::-1]
        P = np.fft.rfft(w_rev, n=L)
        corr_full = np.fft.irfft(A * P, n=L)[: 3 * n - 2]  # a * w_rev
        # pick the slice that corresponds to the gradient formula
        idx = (2 * n - 2) - np.arange(n)            # shape (n,)
        grad = 2.0 * corr_full[idx]

        # Adam update (projected afterwards)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)                    # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # stay on the simplex
        a = _project_simplex(a)

        # evaluation & bookkeeping
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth it={it:,} "
                f"β={beta_soft:.2f} score={cur_score:.6f}"
            )

        # increase smoothness (approach the true max)
        if beta_soft < beta_soft_max:
            beta_soft = min(beta_soft_max, beta_soft * 1.02)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polishing (as in the reference code)
# ----------------------------------------------------------------------
def _anneal_polish(a: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """Very cheap mass‑transfer annealing; returns the best vector found."""
    n = a.shape[0]
    cur_seq = a.copy()
    cur_score = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule (tiny temperature)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        # choose source proportional to current mass
        src = rng.choice(n, p=cur_seq)
        dst = rng.integers(n)
        if src == dst or cur_seq[src] < 1e-14:
            continue

        # move a random fraction (up to 20 %) of source mass
        eps_move = cur_seq[src] * rng.random() * 0.20
        prop = cur_seq.copy()
        prop[src] -= eps_move
        prop[dst] += eps_move

        new_score = _objective(prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            cur_seq = prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] anneal it={it:,} "
                    f"best={new_score:.6f}"
                )
    return best_seq

            best_seq = polished_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final polish score={polished_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # The verifier only cares about non‑negative numbers; scaling does not
    # affect the score, but we keep the sum normalised to 1 for stability.
    return [float(v) for v in best_seq]
```
