def _smooth_objective_and_grad(a, lam):
    n = a.shape[0]
    conv = _conv_self(a, L=None) # length 2n-1
    # compute max of exact objective for reporting but not needed.
    # compute weight vector w
    # For stability compute offset = max(lam * conv)
    scaled = lam * conv
    max_scaled = scaled.max()
    exp_vals = np.exp(scaled - max_scaled)
    Z = exp_vals.sum()
    M = (exp_vals * conv).sum() / Z
    coeff = exp_vals * (1.0 + lam * (conv - M)) / Z  # length 2n-1
    # compute gradient via conv of a_rev and coeff
    a_rev = a[::-1]
    # convolution length 3n-2, compute using FFT.
    L = _next_pow2(len(a_rev) + len(coeff) - 1)
    Af = np.fft.rfft(a_rev, n=L)
    Bf = np.fft.rfft(coeff, n=L)
    conv_rev = np.fft.irfft(Af * Bf, n=L)[: len(a_rev) + len(coeff) - 1]
    # gradient slice
    start = n-1
    end = start + n
    grad = 2.0 * conv_rev[start:end]
    # objective is 2*n * M (since sum a is 1)
    obj = 2.0 * n * M
    return obj, grad

def _adam_descent(init_seq, init_score, lam_schedule, steps_per_lam, deadline, start_time):
    a = init_seq.copy()
    best_seq = a.copy()
    best_score = init_score
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for lam in lam_schedule:
        # adapt learning rate maybe: lr = 0.2 / np.sqrt(lam) or similar.
        lr = 0.2 / (1 + 0.5 * lam)
        for i in range(steps_per_lam):
            if time.time() >= deadline:

"""
Propose a low‑score step function.

Algorithm
---------
1. **Fine sweep of a symmetric Beta‑type family** –  Generate candidates
   a_i ∝ (x_i·(1−x_i)+ε)^{−α} with a dense grid of exponents
   α∈[0.45,0.55] and a few tiny regularisers ε.
   The best candidate is kept.

2. **Smoothed Adam descent** –  Replace the non‑smooth max in the objective
   by a log‑sum‑exp (soft‑max) with temperature λ.
   The gradient of the smooth surrogate is computed analytically via
   FFT‑based convolutions; an Adam optimiser on the simplex (with optional
   symmetrisation) is run with a continuation on λ=[4,8,16,32,64,128,256].

3. **Light simulated‑annealing polish** –  Random “mass‑transfer’’ moves
   are attempted; a move is kept if it improves the exact objective (or is
   accepted with a tiny Metropolis probability).

All steps respect the given wall‑time budget, use only NumPy and the
provided `evaluate_sequence` for validation, and return a plain list of
non‑negative floats (the heights of the step function).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (identical to the verifier)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(seq)^2."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry a[i]=a[n‑1‑i]."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) of max(conv) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, lam: float):
    """
    Returns (smooth_obj, grad) where

    smooth_obj = 2·n·M_smooth,   M_smooth = soft‑max_λ(conv)
    grad       = ∇_a (soft‑max_λ(conv))

    Both are computed with FFT‑based convolutions.
    """
    n = a.shape[0]
    conv = _conv_self(a)                 # length 2·n‑1

    # soft‑max weights
    scaled = lam * conv
    max_scaled = scaled.max()
    exp_vals = np.exp(scaled - max_scaled)
    Z = exp_vals.sum()
    M = (exp_vals * conv).sum() / Z

    # coefficient vector for the gradient (see derivation in analysis)
    coeff = exp_vals * (1.0 + lam * (conv - M)) / Z  # length 2·n‑1

    # gradient = 2 * (a_rev convolved with coeff) restricted to centre part
    a_rev = a[::-1]
    L = _next_pow2(len(a_rev) + len(coeff) - 1)
    Af = np.fft.rfft(a_rev, n=L)
    Bf = np.fft.rfft(coeff, n=L)
    conv_rev = np.fft.irfft(Af * Bf, n=L)[: len(a_rev) + len(coeff) - 1]

    start = n - 1
    end = start + n
    grad = 2.0 * conv_rev[start:end]

    smooth_obj = 2.0 * n * M
    return smooth_obj, grad


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
def _adam_descent(init_a, init_score, lam_schedule, steps_per_lam,

    """
    Perform a continuation Adam optimisation on the smoothed surrogate.
    Returns the best exact‑objective sequence seen.
    """

    best_a = a.copy()
    best_score = init_score

    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) Fine‑grained sweep of the (x·(1−x)+ε)^{−α} family
    # --------------------------------------------------------------
    n = 4096
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    best_seq = None
    best_score = float("inf")

    epsilons = [1e-12, 1e-9, 1e-6, 1e-4]
    alphas = np.linspace(0.45, 0.55, 101)           # step ≈0.001

    for eps in epsilons:
        base = x * (1.0 - x) + eps
        for alpha in alphas:
            w = base ** (-alpha)
            a = w / w.sum()
            a = _symmetrize(a)
            a = _project_simplex(a)

            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep α={alpha:.4f} "
                      f"eps={eps:.0e} score={score:.6f}")

            if best_score < 1.503:      # early exit if target already passed
                break
        if best_score < 1.503:
            break

    # --------------------------------------------------------------
    # 2) Global starter (if provided by the harness)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                starter = _symmetrize(starter)
                starter = _project_simplex(starter)
                cur_score = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] global starter "
                      f"score={cur_score:.6f}")
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
    except Exception:
        pass  # no global starter available

    # --------------------------------------------------------------
    # 3) A few random Dirichlet restarts
    # --------------------------------------------------------------
    for attempt in range(3):
        rand = np.random.gamma(0.3, 1.0, size=n)   # sparse Dirichlet
        a = rand / rand.sum()
        a = _symmetrize(a)
        a = _project_simplex(a)
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] random start {attempt} "
                  f"score={score:.6f}")

    # --------------------------------------------------------------
    # 4) Smoothed Adam descent (continuation on λ)
    # --------------------------------------------------------------
    if best_seq is not None:
        lam_schedule = [4, 8, 16, 32, 64, 128, 256]
        steps_per_lam = 2000
        refined_seq, refined_score = _adam_descent(
            best_seq, best_score, lam_schedule, steps_per_lam,
            deadline, start_time
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] refined final "
                  f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light simulated‑annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_steps = 50000

        for it in range(1, total_steps + 1):
            if time.time() >= deadline:
                break

            # exponential cooling from 0.01 down to 1e‑4
            T = 0.01 * (1e-4 / 0.01) ** (it / total_steps)

            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

            eps_move = a[src] * rng.random() * 0.20
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            new_score = _objective(a_prop)
            delta = new_score - cur_score

            if delta < 0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} "
                          f"best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve "
                  f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to a plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
