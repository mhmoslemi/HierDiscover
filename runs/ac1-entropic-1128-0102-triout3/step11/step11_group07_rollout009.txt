"""
Propose‑candidate for the “max‐convolution” challenge.

Main ideas
----------
* Work with a *symmetrised* representation (half the variables, reflected).
  Symmetry is natural for the optimal step‑function and halves the optimisation
  dimension, allowing many more gradient / evolutionary steps in the same time.
* A fast FFT based objective evaluator (identical to the one used by the
  official checker) – this is the bottleneck that we keep as cheap as possible.
* Three complementary optimisation phases
  1. a dense parametric sweep (geometric, power and beta families);
  2. long Adam‑style smoothed‑max descent on the half‑vector;
  3. a low‑dimensional Gaussian‑mixture optimisation performed by
     ``scipy.optimize.differential_evolution``; the DE run is deliberately
     long (popsize≈20, up to a few hundred generations) because each evaluation
     costs only a few milliseconds.
* Afterwards a short LP–active‑set refinement (if ``scipy`` is present) and a
  simulated‑annealing hill‑climb that accepts occasional worsening moves,
  helping to escape shallow local minima.
* The whole search respects the 1000 s budget; progress is printed and the
  best sequence found is returned (normalised to sum = 1, non‑negative).

The code is self‑contained, uses only ``numpy`` and optional ``scipy``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective (2·n·max(conv)/sum²) – assumes a is already normalised."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Symmetric handling (only half of the coefficients are stored)
# ----------------------------------------------------------------------
def _full_from_half(half: np.ndarray) -> np.ndarray:
    """Mirror the half‑vector to a full length‑n non‑negative sequence."""
    n = half.shape[0] * 2
    full = np.empty(n, dtype=half.dtype)
    full[: half.shape[0]] = half
    full[half.shape[0] :] = half[::-1]
    return full


def _half_project_simplex(half: np.ndarray) -> np.ndarray:
    """Project a half‑vector so that the reflected full vector lies on the simplex."""
    # full sum = 2·sum(half) must be 1 → sum(half) = 0.5
    half = np.maximum(half, 0.0)
    total = half.sum()
    if total == 0:
        half = np.full_like(half, 0.5 / half.shape[0])
    else:
        half = half * (0.5 / total)
    return half


# ----------------------------------------------------------------------
# 1️⃣  Fast parametric sweep (geometric / power / beta families)
# ----------------------------------------------------------------------
def _sweep_candidates(n: int, L_fft: int, deadline: float, start: float) -> tuple[np.ndarray, float]:
    best_seq = np.full(n, 1.0 / n)
    best_score = _eval_fast(best_seq, L_fft)
    idx = np.arange(n)

    # geometric
    for eps in np.logspace(-5, -1, 200):
        if time.time() >= deadline - 15:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo = geo / geo.sum()
        sc = _eval_fast(geo, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start:.2f}s] Geo eps={eps:.1e} → {sc:.6f}")

    # power law
    for p in np.linspace(-1.5, 2.5, 80):
        if time.time() >= deadline - 15:
            break
        pw = (idx + 1) ** p
        pw = pw / pw.sum()
        sc = _eval_fast(pw, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, pw.copy()
            print(f"[{time.time() - start:.2f}s] Power p={p:.2f} → {sc:.6f}")

    # beta‑type (symmetric)
    for p in (0.0, 0.5, 1.0, 1.5, 2.0):
        for q in (0.0, 0.5, 1.0, 1.5, 2.0):
            if time.time() >= deadline - 15:
                break
            beta = (idx + 1) ** p * (n - idx) ** q
            beta = beta / beta.sum()
            sc = _eval_fast(beta, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, beta.copy()
                print(f"[{time.time() - start:.2f}s] Beta p={p:.1f} q={q:.1f} → {sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam‑style descent on the smooth max (temperature annealing)
# ----------------------------------------------------------------------
def _adam_smooth_descent_half(start_half: np.ndarray,
                              deadline: float,
                              start: float,
                              L_fft: int,
                              L_grad: int) -> tuple[np.ndarray, float]:
    """Adam on a *half* representation (symmetry enforced)."""
    half = start_half.copy()
    best_half = half.copy()
    # evaluate full sequence for the score
    best_score = _eval_fast(_full_from_half(half), L_fft)

    m = np.zeros_like(half)
    v = np.zeros_like(half)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    tau = 0.06
    tau_min = 1e-8
    tau_decay = 0.9998          # a bit slower than before
    step0 = 0.12

    while time.time() < deadline:
        it += 1
        full = _full_from_half(half)
        conv = _conv_self(full, L=L_fft)
        maxc = conv.max()

        # soft‑max weight over the full convolution (stable)
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. the half‑vector:
        #   ∂/∂a_i  Σ_k w_k (a*a)_k  =  2 * Σ_k w_k Σ_j a_j δ_{i+j=k}
        #   = 2 * (a * w_rev)[i]  where w_rev is w reversed with respect to length‑2n−1
        w_rev = w[::-1]
        Af = np.fft.rfft(half, n=L_grad)
        Wf = np.fft.rfft(w_rev[: 2 * half.shape[0] - 1], n=L_grad)  # same length as half
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        # the convolution result of length (len(half)+len(w_rev)-1) = 3·half‑size‑2
        # the relevant slice for the gradient of the half‑vector:
        offset = 2 * half.shape[0] - 2
        grad = 2.0 * grad_full[offset - np.arange(half.shape[0])]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        half -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the symmetric simplex (sum(half)=0.5)
        half = _half_project_simplex(half)

        cur_score = _eval_fast(_full_from_half(half), L_fft)
        if cur_score < best_score:
            best_score, best_half = cur_score, half.copy()
            print(f"[{time.time() - start:.2f}s] Adam it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_half, best_score


# ----------------------------------------------------------------------
# 3️⃣  Gaussian‑mixture DE on the half‑vector (more components, longer run)
# ----------------------------------------------------------------------
def _gaussian_half(params: np.ndarray, n_half: int) -> np.ndarray:
    """Build a symmetric sequence from K Gaussian components (half‑vector)."""
    seq = np.zeros(2 * n_half, dtype=np.float64)
    idx = np.arange(2 * n_half, dtype=np.float64)
    K = len(params) // 3
    for i in range(K):
        amp = max(0.0, params[3 * i])                # non‑negative
        cen = params[3 * i + 1]                      # centre on the full grid
        wid = max(1e-3, params[3 * i + 2])
        seq += amp * np.exp(- ((idx - cen) ** 2) / (2.0 * wid * wid))
    seq = np.maximum(seq, 0.0)
    # project onto the symmetric simplex (half‑sum = 0.5)
    half = seq[:n_half]
    half = _half_project_simplex(half)
    return _full_from_half(half)


def _de_optimise_half(n: int,
                      L_fft: int,
                      deadline: float,
                      start: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """DE on a Gaussian mixture (full‑vector) using the symmetric half‑parameterisation."""
    try:
        from scipy.optimize import differential_evolution
    except Exception:  # pragma: no cover
        return best_seq, best_score

    n_half = n // 2
    K = 8                     # number of Gaussian components (larger than before)
    dim = 3 * K
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 20.0))          # amplitude
        bounds.append((0.0, n - 1))         # centre
        bounds.append((1.0, n / 2.0))       # width

    def obj(params):
        seq = _gaussian_half(params, n_half)
        return _eval_fast(seq, L_fft)

    remaining = deadline - time.time()
    # allocate the rest of the time (up to ~400 s) to DE; each generation is cheap
    maxiter = max(1, int(remaining // 5))
    maxiter = min(maxiter, 300)               # safety cap
    popsize = 20

    print(f"[{time.time() - start:.2f}s] DE start (maxiter={maxiter}, popsize={popsize})")
    result = differential_evolution(obj,
                                    bounds,
                                    maxiter=maxiter,
                                    popsize=popsize,
                                    tol=1e-7,
                                    polish=False,
                                    seed=1,
                                    workers=1,
                                    updating='deferred')
    seq_de = _gaussian_half(result.x, n_half)
    score_de = _eval_fast(seq_de, L_fft)
    if score_de < best_score:
        best_seq, best_score = seq_de, score_de
        print(f"[{time.time() - start:.2f}s] DE improved → {score_de:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Active‑set LP refinement (more iterations)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start: float,
               L_fft: int,
               max_iters: int = 15,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Iteratively improve a by solving LPs on the active constraints."""
    try:
        from scipy.optimize import linprog
    except Exception:  # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])

        K = active.size
        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1)
            i = k - j
            A_ub[row, j] = a[i]
        b_ub = np.full(K, M, dtype=np.float64)

        # maximise sum(g) ⇔ minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(1, int(deadline - time.time()))})

        if not res.success:
            print(f"[{time.time() - start:.2f}s] LP {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0.0:
            break
        g = g / g.sum()          # re‑normalise to sum = 1

        # replace a with g if it improves; otherwise keep the old a
        new_score = _eval_fast(g, L_fft)
        if new_score < best_score:
            best_score, a = new_score, g
            print(f"[{time.time() - start:.2f}s] LP it={it} improved → {new_score:.6f}")
        else:
            # no gain – stop early
            break

    return a, best_score


# ----------------------------------------------------------------------
# 5️⃣  Simulated‑annealing hill‑climb (tiny stochastic moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start: float,
                         L_fft: int,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Tiny randomized mass transfers with Metropolis acceptance."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    # initial temperature chosen so that ~30 % of uphill moves are accepted
    T = 0.01

    while time.time() < deadline and it < max_iters:
        it += 1
        # pick a source proportional to current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        # amount to move (dynamic with temperature)
        eps = a[src] * (0.05 + 0.15 * rng.random()) * (T / (T + 1.0))
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # maintain simplex (should stay close, but re‑project to avoid drift)
        a_prop = _project_simplex(a_prop)

        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - best_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            a = a_prop
            if new_score < best_score:
                best_score, best_a = new_score, a.copy()
                print(f"[{time.time() - start:.2f}s] SA it={it:,} new best {new_score:.6f}")

        # simple exponential cooling
        T *= 0.99995

    return best_a, best_score

# Main driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine mixes a deterministic sweep,
    a long Adam‑style descent on a symmetry‑enforced representation,
    an extensive Gaussian‑mixture differential‑evolution,
    LP refinement and a final simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # length of the sequence – the original runs used 4096, we keep it.
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)            # for gradient convolutions

    # --------------------------------------------------------------
    # Baseline – uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric / power / beta)
    # --------------------------------------------------------------
    cand_seq, cand_score = _sweep_candidates(n, L_fft, deadline, start_time)
    if cand_score < best_score:
        best_seq, best_score = cand_seq, cand_score
        print(f"[{time.time() - start_time:.2f}s] Sweep improved → {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam‑style smooth descent, many restarts, longer runs
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # random start: perturb the current best or the uniform baseline
        base = best_seq if np.random.rand() < 0.6 else uniform
        half = _half_project_simplex(base[: n // 2] + 0.04 * np.random.randn(n // 2) * base[: n // 2])
        sub_deadline = min(deadline - 5.0, time.time() + 20.0)   # give each run ~20 s
        half_opt, score_opt = _adam_smooth_descent_half(half, sub_deadline, start_time, L_fft, L_grad)
        full_opt = _full_from_half(half_opt)
        if score_opt < best_score:
            best_seq, best_score = full_opt, score_opt
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth global best → {best_score:.6f}")

        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 3️⃣  Gaussian‑mixture DE (low‑dimensional but long)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0 and best_score > 1.503:
        best_seq, best_score = _de_optimise_half(n, L_fft, deadline, start_time,
                                                 best_seq, best_score)

    # --------------------------------------------------------------
    # 4️⃣  Active‑set LP refinement (more iterations than before)
    # --------------------------------------------------------------
    if time.time() < deadline - 15.0 and best_score > 1.503:
        best_seq, best_score = _lp_refine(best_seq, deadline, start_time, L_fft)

    # --------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq, deadline, start_time, L_fft)

    # --------------------------------------------------------------
    # Normalise to a proper probability vector and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
