def smooth_obj_and_grad(z, alpha, n, L_self, L_grad):
    # compute a=softmax(z)
    max_z = np.max(z)
    ez = np.exp(z - max_z)
    a = ez / ez.sum()
    # compute conv self via FFT
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]
    # softmax w with alpha
    prod = alpha * conv
    max_prod = prod.max()
    exp_prod = np.exp(prod - max_prod)
    w = exp_prod / exp_prod.sum()
    # smoothed max value
    smoothed_max = np.dot(w, conv)
    obj = 2.0 * n * smoothed_max

    a_rev = a[::-1]

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------
1. **Smooth‑max L‑BFGS** – minimise a smooth surrogate of the
   maximum convolution:
        fα(a)=2·n·∑k w_k(α)·(a*a)_k,
   where w_k(α) are the soft‑max weights for the current smoothing
   parameter α.  The variables are the log‑probabilities `z` (so that
   `a = softmax(z)`).  For a geometric schedule of α we run a short
   L‑BFGS optimisation (SciPy `fmin_l_bfgs_b`).  Several random restarts
   are performed; the best point is kept.

2. **Projected sub‑gradient descent** – classic sub‑gradient on the
   exact objective with a 1/√t step‑size, Euclidean projection onto the
   probability simplex and a momentum term.  This polishing phase runs
   until the wall‑clock limit is reached.

3. **Greedy hill‑climb** – move a tiny amount of mass from a source
   index (chosen proportionally to its height) to a random destination;
   keep the move only if the exact objective improves.  This final polish
   explores the neighbourhood of the best point found so far.

All phases respect the 1000 s wall‑clock budget and constantly log
progress.  The returned sequence is normalised to sum = 1, guaranteeing
that `evaluate_sequence` receives a valid candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (convolution lengths are powers of two for speed)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes Σa = 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (for a given α)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, alpha: float,
                         n: int, L_self: int, L_grad: int):
    """
    Returns (value, gradient) of the smoothed objective
    fα(z) = 2·n·∑ w_k(α)·conv_k  with a = softmax(z) and
    w_k(α) = softmax_k(α·conv).
    Gradient is w.r.t. the log‑parameter z.
    """
    # softmax → a (always sums to 1)
    maxz = np.max(z)
    e = np.exp(z - maxz)
    a = e / e.sum()

    # self‑convolution a*a
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

    # softmax weights over the convolution
    exp_arg = alpha * conv
    max_arg = np.max(exp_arg)
    w = np.exp(exp_arg - max_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # smoothed value
    smoothed_max = np.dot(w, conv)
    obj = 2.0 * n * smoothed_max

    # gradient w.r.t a: 2·(w ⋆ a_rev)
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L_grad)
    Revf = np.fft.rfft(a_rev, n=L_grad)
    conv_wa = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
    grad_a = 2.0 * conv_wa[n - 1:n - 1 + n]

    # chain rule through softmax (a = softmax(z))
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    return obj, grad_z

# Phase‑1 : smooth‑max L‑BFGS with continuation on α
# ----------------------------------------------------------------------
def _lbfgs_continuation(start_z: np.ndarray,
                        alphas: np.ndarray,

    """
    Runs L‑BFGS for each α in `alphas`, warm‑starting from the previous
    solution.  Returns the best (a, score) pair discovered.
    """
    try:
        from scipy.optimize import fmin_l_bfgs_b
    except Exception as e:
        print("SciPy not available – skipping L‑BFGS phase.")
        return None, float("inf")

    best_a = None
    best_score = float("inf")
    z = start_z.copy()

    for alpha in alphas:
        if time.time() >= deadline - 5.0:
            break

        # objective wrapper for fmin_l_bfgs_b (returns (f, grad))
        def fun_grad(x):
            return _smooth_obj_and_grad(x, alpha, n, L_self, L_grad)

        # L‑BFGS (unconstrained in the z‑space)
        z, fval, _ = fmin_l_bfgs_b(func=fun_grad,
                                   x0=z,
                                   maxiter=max_iter_per_alpha,
                                   disp=False)

        # convert to a and evaluate the *exact* objective
        a = np.exp(z - np.max(z))
        a /= a.sum()
        cur_score = _eval_fast(a, L_self)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS α={alpha:.2f} → score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (same as previous code)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        if stagnation >= 20_000:
            break

        # active indices (within tolerance)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = 0.3 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : greedy hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_seq: np.ndarray,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.choice(n, p=a)               # source weighted by mass
        dst = rng.integers(0, n)               # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * 0.30    # move up to 30 % of source mass
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # length needed for smooth‑max gradient

    # ------------------------------------------------------------------
    # Initialise a starter vector (global best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑1 : smooth‑max L‑BFGS using continuation on α
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        # geometric schedule of smoothing parameters
        alphas = np.geomspace(0.5, 2000.0, num=35)

        # 1) deterministic continuation from the current best point
        z0 = np.log(best_seq + 1e-15)          # log‑softmax parametrisation
        a_cont, sc_cont = _lbfgs_continuation(z0, alphas,
                                              n, L_self, L_grad,
                                              deadline - 10.0,
                                              start_time,
                                              max_iter_per_alpha=30)
        if a_cont is not None and sc_cont < best_score:
            best_seq = a_cont.copy()
            best_score = sc_cont
            print(f"[{time.time() - start_time:.2f}s] After deterministic L‑BFGS best={best_score:.6f}")

        # 2) a few random restarts (increase diversity)
        n_restarts = 2
        for rs in range(n_restarts):
            if time.time() >= deadline - 30.0:
                break
            # draw a random point on the simplex (Dirichlet) and go to log‑space
            a_rand = np.random.dirichlet(np.ones(n))
            z_rand = np.log(a_rand + 1e-15)
            a_rand_opt, sc_rand_opt = _lbfgs_continuation(z_rand, alphas,
                                                          n, L_self, L_grad,
                                                          deadline - 10.0,
                                                          start_time,
                                                          max_iter_per_alpha=30)
            if a_rand_opt is not None and sc_rand_opt < best_score:
                best_seq = a_rand_opt.copy()
                best_score = sc_rand_opt
                print(f"[{time.time() - start_time:.2f}s] Restart {rs} L‑BFGS improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : projected sub‑gradient refinement (fast descent)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             L_self, sub_deadline,
                                             start_time, best_score,
                                             max_iters=500_000)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 : greedy hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _hill_climb(best_seq, n,
                                        L_self, deadline,
                                        start_time,
                                        best_seq, best_score)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
