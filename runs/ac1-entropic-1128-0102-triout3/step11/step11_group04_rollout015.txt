    best_a = a0.copy()
    best_score = _eval_fast(best_a, L_self)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    alphas = np.geomspace(0.5, 2000, num=50) # maybe bigger
    for alpha in alphas:
        while time.time() < deadline - 0.5:
            it += 1
            a = np.exp(log_a)
            a = a / a.sum()
            # compute conv via FFT
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]
            # compute softmax weights
            w = np.exp(alpha * conv - np.max(alpha * conv))
            w /= w.sum()
            # gradient
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)

"""
Propose a step‑function (list of non‑negative heights) that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------
1. Load a starter sequence (the global ``height_sequence_1`` if it exists,
   otherwise a uniform distribution) and normalise it to the probability simplex.
2. **Smooth‑max mirror descent (Adam)** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  The variables are updated in the
   log‑domain with the Adam optimiser; after every update the vector is
   projected back onto the simplex (exponentiate and renormalise).
3. **Projected Adam on the exact sub‑gradient** – compute the true sub‑gradient
   of the max‑convolution, take an Adam step on the log‑variables and project
   onto the simplex.  This refines the solution on the original (non‑smooth)
   objective.
4. **Gradient‑guided hill‑climbing** – repeatedly move a small fraction of the
   mass from the index that contributes the most to the current maximal
   convolution value (largest sub‑gradient entry) to the index that contributes
   the least (smallest sub‑gradient entry).  Accept the move only if the exact
   objective improves.  This final polish explores the neighbourhood of the
   best point found so far.
5. Return the best sequence (as a plain Python list).  All steps respect the
   wall‑clock deadline (default 1000 s) and print progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolutions (self‑convolution and generic convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors u and v using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {w≥0, sum(w)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (the sum of a is assumed to be 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a non‑negative vector with sum≈1."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent with Adam (log‑domain)
# ----------------------------------------------------------------------
def _smooth_descent_adam(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         L_grad: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth log‑sum‑exp approximation of the max‑convolution.
    Variables are kept in the log‑domain and updated with Adam.
    """
    # initialise log‑variables
    log_a = np.log(np.clip(a0, 1e-30, None))
    best_a = a0.copy()
    best_score = _eval_fast(best_a, L_self)

    # Adam moments
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    it = 0
    # smoothing schedule – start very soft and become increasingly sharp
    alphas = np.geomspace(0.3, 1200.0, num=45)

    for alpha in alphas:
        # inner loop – stop when we run out of time
        while time.time() < deadline - 0.2:
            it += 1

            # current probability vector
            a = np.exp(log_a)
            a /= a.sum()

            # 1) self‑convolution a*a
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights w_k ∝ exp(alpha * conv_k)
            max_arg = (alpha * conv).max()
            w = np.exp(alpha * conv - max_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2 * (w ⋆ a_rev)  (restricted to n entries)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update on the log‑variables
            grad_log = a * grad                     # chain rule  dL/dlog_a = a·dL/da
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            bias_correction1 = 1.0 - beta1 ** it
            bias_correction2 = 1.0 - beta2 ** it
            m_hat = m / bias_correction1
            v_hat = v / bias_correction2

            # a simple decaying learning‑rate schedule
            lr = 0.12 / (math.sqrt(it) + 1.0)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            log_a -= step
            log_a = np.maximum(log_a, -30.0)   # avoid underflow

            # occasional exact evaluation
            if it % 750 == 0:
                cur_score = _eval_fast(np.exp(log_a) / np.exp(log_a).sum(), L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = np.exp(log_a) / np.exp(log_a).sum()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    # final normalisation
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() > 0.0:
        best_a /= best_a.sum()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – projected Adam on the exact sub‑gradient
# ----------------------------------------------------------------------
def _subgrad_adam(a0: np.ndarray,
                  n: int,
                  L_self: int,
                  deadline: float,
                  start_time: float,
                  init_best_score: float,
                  max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the *exact* max‑convolution.
    The update is performed on log‑variables; after each step the vector is
    projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    # Adam moments defined on the log‑domain
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    it = 0
    stagnation = 0                     # count iterations without improvement

    while time.time() < deadline - 0.1 and it < max_iters:
        it += 1

        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Subgrad Adam it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        # early stop if we are stuck
        if stagnation >= 30_000:
            break

        # active constraints (indices attaining the maximum)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # Adam update in log‑space
        log_a = np.log(np.clip(a, 1e-30, None))
        grad_log = a * grad
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        bias_correction1 = 1.0 - beta1 ** it
        bias_correction2 = 1.0 - beta2 ** it
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        lr = 0.15 / (math.sqrt(it) + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        log_a -= step
        log_a = np.maximum(log_a, -30.0)

        # back to the simplex
        a = np.exp(log_a)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – gradient‑guided hill climbing (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb_gradient(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         cur_best_seq: np.ndarray,
                         cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb: move a fraction of mass from the index that most
    contributes to the current maximal convolution value to the index that
    contributes the least.  Keep the move only if it improves the exact
    objective.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()
    step_frac = 0.30                     # maximal fraction of source mass to move

    while time.time() < deadline:
        # compute the sub‑gradient (needed to identify source/destination)
        conv = _conv_self(a, L_self)
        max_val = conv.max()
        active = np.where(np.abs(conv - max_val) <= 1e-12)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # if the best source has negligible mass fall back to a random source
        if a[src] < 1e-15 or src == dst:
            src = rng.choice(n, p=a)
            dst = rng.integers(0, n)
            if src == dst or a[src] < 1e-15:
                continue

        eps = a[src] * rng.random() * step_frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_self)
        if new_score < cur_score - 1e-13:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb gradient best={best_score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgrad_adam(best_seq, n,
                                      L_self,
                                      sub_deadline,
                                      start_time,
                                      best_score,
                                      max_iters=400_000)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad Adam improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _hill_climb_gradient(best_seq, n,
                                                 L_self,
                                                 deadline,
                                                 start_time,
                                                 best_seq,
                                                 best_score)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improved score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
