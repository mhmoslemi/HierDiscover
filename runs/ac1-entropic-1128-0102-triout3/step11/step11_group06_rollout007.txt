"""
Propose a low‑objective step‑function.

The algorithm works in three stages:

1. **Broad parametric sweep** –  a 2‑parameter family
       a_i ∝ (x_i+δ)^{p‑1}·(1‑x_i+δ)^{q‑1}
   (the beta‐density) is evaluated on a dense grid of (p,q).  The best
   candidate is kept as a starting point.

2. **Long smoothed Adam optimisation** –  the non‑smooth objective
   2·n·max(conv)/sum² is approximated by a log‑sum‑exp (soft‑max)
   surrogate.  The gradient of the surrogate can be written with a
   single extra convolution, hence each iteration needs two FFTs.
   We run many thousands of iterations, linearly increasing the
   smoothing parameter β up to 10³ and slowly decaying the learning
   rate.  The iterate with the smallest exact objective seen so far is
   retained.

3. **Very light simulated‑annealing polish** –  random “mass‑transfer”
   moves (move a few percent of the mass from one index to another)
   are accepted only when they improve the exact objective.  This
   gives a cheap local‑search refinement.

All vectors are kept on the probability simplex (non‑negative,
Σa_i = 1).  The routine respects the 1000 s wall‑time limit,
printing progress information and finally returning a plain Python
list of floats.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w ≥ 0, Σw = 1}.
    Implements the algorithm of Condat (2016).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT‑based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a very low evaluation
    score under the given time budget.

    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a tiny safety margin for the harness
    deadline = start + budget_s - 0.5

    # problem size – a power of two works nicely with the FFTs
    n = 4096

    # ------------------------------------------------------------------
    # 0) Pull a previous construction if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.maximum(cand, 0.0)
                if cand.sum() > 0:
                    cand = cand / cand.sum()
                    sc = _objective(cand)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Wide beta‑family sweep (beta = (p,q) in Beta density)
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑point positions
    eps = 1e-12                                        # tiny shift to avoid 0⁰

    # grid of (p,q) – we only need p,q in (0,1) for U‑shaped densities,
    # but we also test a few values >1 to see if a more “bell‑shaped”
    # density helps.
    p_vals = np.linspace(0.25, 0.95, 15)    # 15 values → 225 combos
    q_vals = np.linspace(0.25, 0.95, 15)

    for p in p_vals:
        for q in q_vals:
            # beta‑density (up to a normalising constant)
            w = (x + eps) ** (p - 1) * (1.0 - x + eps) ** (q - 1)
            a = w / w.sum()
            sc = _objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
            # time check
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Fallback to a simple uniform vector if everything else failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3) Long smoothed Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    a = best_seq.copy()                  # current iterate (always on simplex)
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    # learning‑rate schedule (initially relatively aggressive, then decays)
    base_lr0 = 0.08
    # FFT length large enough for both convolutions (self‑conv + cross‑conv)
    L = _next_pow2(3 * n)

    iteration = 0
    # smoothing schedule – start gentle, end very sharp (≈1000)
    beta_start, beta_end = 2.0, 1000.0   # “temperature” for soft‑max

    while time.time() < deadline - 2.0:   # keep ~2 s for final polish
        iteration += 1

        # ---- self‑convolution -------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]          # length 2n‑1

        # ---- smoothing parameter (increase with time) -------------------------
        elapsed = time.time() - start
        prop = min(1.0, elapsed / (budget_s - 2.0))
        beta = beta_start + (beta_end - beta_start) * prop

        # ---- soft‑max weights (log‑sum‑exp) ------------------------------------
        beta_conv = beta * conv
        max_beta = beta_conv.max()
        # subtract max for numerical stability
        exp_shift = np.exp(beta_conv - max_beta)
        w = exp_shift / exp_shift.sum()          # shape (2n‑1,)

        # ---- gradient of the smooth surrogate ---------------------------------
        # B·a = correlation of w with a (cross‑correlation)
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        # slice to obtain the n‑length vector B·a
        Ba = corr_full[n - 1:n - 1 + n]
        # gradient of the smoothed objective (see derivation in the comment
        # at the top of the file)
        grad = 4.0 * n * Ba

        # ---- Adam update -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        # simple learning‑rate decay
        lr = base_lr0 * (1.0 - prop) + 0.001 * prop
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        # keep a on the simplex
        a = _project_simplex(a)

        # ---- keep the best exact objective seen so far -------------------------
        if iteration % 200 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{elapsed:.1f}s] Adam itr={iteration:,}  best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Very light simulated‑annealing polish (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        # estimate how many moves we can afford (roughly 1 µs per move)
        moves_allowed = 80_000
        for t in range(1, moves_allowed + 1):
            if time.time() >= deadline:
                break
            # choose source proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue
            # move at most 10 % of src's mass
            delta = a[src] * rng.random() * 0.10
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta
            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start:.1f}s] anneal step={t:,} best={new_score:.6f}")
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start:.1f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
