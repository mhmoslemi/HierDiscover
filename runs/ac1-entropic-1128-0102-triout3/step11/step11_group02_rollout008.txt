def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 32768
    # Generate warm start: symmetric tent (triangular) shape normalized.
    x = np.arange(n, dtype=float)/(n-1)
    tent = np.minimum(x, 1.0 - x)
    a_initial = tent / tent.sum() if tent.sum()>0 else np.full(n, 1.0/n)

    # also maybe generate a few random Dirichlet initial vectors.
    restarts = 5
    best_a = None
    best_obj = np.inf

    for i in range(restarts):
        # choose initial
        if i == 0:
            a = a_initial.copy()
        else:
            # Dirichlet random
            alpha = np.ones(n) * 0.5 # or maybe 0.2 to encourage sparsity.
            a = np.random.dirichlet(alpha) # returns sum=1
        # run optimizer
        a_opt, obj = _optimize(a, deadline, start_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt
            print(f"[{time.time() - start_time:.2f}s] Restart {i} best obj {obj:.6f}")

        # check time; maybe break early
        if time.time() >= deadline - 5:
            break

    # final annealing if time left
    if best_a is not None and time.time() < deadline:
        best_a, best_obj = _anneal(best_a, deadline, start_time)

    # final normalization to sum=1 (ensuring positive)
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0/n)

    return [float(v) for v in best_a]

def _optimize(a0, deadline, start_time, max_iter=50000):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2  # base learning rate
    t = 0
    best_a = a.copy()
    best_obj = _objective(a)
    while time.time() < deadline and t < max_iter:
        t += 1
        grad, max_val = _subgrad_max(a)  # returns gradient as described
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

def _subgrad_max(a):
    conv = _conv_self(a)
    max_val = conv.max()
    # find indices where conv == max_val (tolerate small tolerance)
    max_idx = np.where(conv >= max_val - 1e-12)[0]  # include all ties (floating)
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi+1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    grad /= max(1, len(max_idx))
    return grad, max_val

def _init_candidates(n, num_random=20):
    candidates = []
    # deterministic shapes
    x = np.arange(n) / (n-1)
    # tent
    tent = np.minimum(x, 1-x)
    candidates.append(tent / tent.sum())
    # cosine half-wave
    cos = np.sin(np.pi * x)  # sin shape goes from 0 to 0 with max at middle
    candidates.append(cos / cos.sum())
    # beta shape for some alphas
    for a,b in [(0.5,0.5), (2,2), (3,1), (1,3), (0.3,0.3)]:
        beta_shape = (x**a) * ((1-x)**b)
        if beta_shape.sum()>0:
            candidates.append(beta_shape / beta_shape.sum())
    # random dirichlet
    rng = np.random.default_rng()
    for i in range(num_random):
        alpha = rng.uniform(0.2, 2.0) # random concentration param
        # draw from Dirichlet with all alpha same for simplicity
        a_vec = rng.gamma(alpha, 1.0, size=n)
        a_vec = a_vec / a_vec.sum()
        candidates.append(a_vec)

    # evaluate each candidate's objective and keep best
    best_seq = None
    best_obj = np.inf
    for seq in candidates:
        obj = _objective(seq)
        if obj < best_obj:
            best_obj = obj
            best_seq = seq
    return best_seq

"""
Propose a low‑objective step function.

The objective for a non‑negative vector a (scaled arbitrarily) is

    2·n·max(conv(a,a)) / (sum(a)²)

which, after normalising sum(a)=1, is simply 2·n·max(conv(a,a)).
The algorithm works as follows:

1. **Candidate generation** – several simple families of symmetric
   non‑negative shapes (tent, cosine, beta‑like, powers) and a few
   random Dirichlet draws are built and the one with the smallest
   objective is kept as warm‑start.

2. **Projected sub‑gradient descent** – the gradient of the maximal
   convolution entry (or the average of the top‑k entries) is computed
   analytically (grad_i = 2·a_{k‑i} on the support).  An Adam‑style
   momentum update with a diminishing step size is applied and the
   iterate is projected onto the probability simplex.  The schedule
   starts with a smoother “top‑k=7’’ gradient and switches to the exact
   max‑gradient later on.

3. **Light simulated‑annealing polish** – once the time budget is
   almost exhausted a cheap Metropolis‑style annealer moves a small
   random packet of mass between two coordinates, accepting moves that
   improve the objective or with a temperature‑dependent probability.

All helpers are top‑level, the routine respects the 1000 s budget and
returns a plain `list[float]` that can be fed to `evaluate_sequence`.
"""

import time

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _generate_initial_candidates(n: int, rng: np.random.Generator):
    """Return a list of promising normalised vectors."""
    candidates = []

    # symmetric linear tent
    x = np.arange(n, dtype=float) / (n - 1)
    tent = np.minimum(x, 1.0 - x)
    if tent.sum() > 0:
        candidates.append(tent / tent.sum())

    # half‑sine wave (zero at ends, peak in the middle)
    sine = np.sin(np.pi * x)
    if sine.sum() > 0:
        candidates.append(sine / sine.sum())

    # cosine‑based bump (non‑negative)
    cosine = 1.0 - np.cos(np.pi * x)
    if cosine.sum() > 0:
        candidates.append(cosine / cosine.sum())

    for a, b in [(0.5, 0.5), (2.0, 2.0), (3.0, 1.0), (1.0, 3.0), (0.3, 0.3)]:

def _pick_best_candidate(candidates):
    """Select the candidate with the smallest objective."""
    best_seq = None
    best_score = np.inf
    for seq in candidates:
        score = _objective(seq)
        if score < best_score:

            best_seq = seq

# Projected sub‑gradient / Adam style optimisation
# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              deadline: float,
              start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimises the sequence using a projected Adam‑like update.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.6               # base learning rate

    m = np.zeros_like(a)    # first moment
    v = np.zeros_like(a)    # second moment

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # --- evaluate convolution and find top entries -----------------
        conv = _conv_self(a)
        max_val = conv.max()

        # schedule: smooth early, exact later
        if it < 20000:
            top_k = 7
        else:
            top_k = 1

        # indices of the top‑k entries (unsorted)
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # --- build (averaged) sub‑gradient ---------------------------
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # --- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / math.sqrt(it)        # diminishing global step
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # --- project back onto the simplex ---------------------------
        a = _project_simplex(a)

        # --- record improvement ---------------------------------------
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Iter {it:,} "
                  f"best={cur_score:.6f}")

        # safety net – stop if we are very close to the deadline
        if time.time() > deadline - 1.0:
            break

    return best_a, best_score

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis annealing that moves a tiny random packet of mass.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 50000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.05 * (1e-4 / 0.05) ** (it / total_steps)

        src = rng.choice(n, p=a)                # source weighted by mass
        dst = rng.integers(n)                    # destination uniform
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20      # move ≤20 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
