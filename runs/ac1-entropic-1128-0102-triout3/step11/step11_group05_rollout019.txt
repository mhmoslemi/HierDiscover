"""
Improved search for the minimum of

    2 * n * max(conv(a, a)) / (sum(a)**2)

for a non‑negative step‑function a.  

The new version

* starts from several analytically motivated seeds (uniform block,
  powers of the distance to the border, etc.) and keeps the best one,
* runs a longer smooth‑max mirror‑descent (exponentiated‑gradient) with
  a richer schedule of smoothing parameters,
* finishes with a *targeted* simulated‑annealing phase that always
  moves mass from an index belonging to the currently most violated
  convolution entry to a low‑mass index.  

Both phases respect the simplex (non‑negative, sum = 1) and the total
runtime never exceeds the supplied budget (default 1000 s).  The final
sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # only used if the LP fallback is needed

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 0️⃣  Initial seed generation
# ----------------------------------------------------------------------
def _generate_seeds(n: int) -> list[np.ndarray]:
    """Return a list of simple analytic candidates."""
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # powers of the distance to the border, a_i ∝ ((i+1)*(n-i))^{-p}
    i = np.arange(n, dtype=float)
    for p in (0.3, 0.5, 0.7, 1.0, 1.3, 1.6):
        raw = 1.0 / ((i + 1) * (n - i)) ** p
        raw = raw / raw.sum()
        seeds.append(raw)

    # a single “U‑shaped’’ seed: a_i ∝ 1/√((i+0.5)*(n‑i‑0.5))
    raw = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    raw = raw / raw.sum()
    seeds.append(raw)

    # if a historic construction exists, reuse it (trim / pad if needed)
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(height_sequence_1, dtype=float)
            if old.ndim == 1 and old.size > 0:
                old = np.maximum(old, 0.0)

                    seeds.append(old)
        except Exception:

    return seeds

# 1️⃣  Smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Longer version of the exponentiated‑gradient descent used earlier."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)   # avoid exact zeros
            a /= a.sum()

            # occasional real evaluation
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time()-start_time:.2f}s] "
                          f"smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score

# 2️⃣  Targeted incremental simulated‑annealing
# ----------------------------------------------------------------------
def _targeted_anneal(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     cur_best_score: float,
                     cur_best_seq: np.ndarray,
                     start_temp: float = 5e-2,
                     end_temp: float = 1e-12,
                     max_moves: int = 8_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that preferentially attacks the current maximum."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1, cheap for n≈4096
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max           # sum(a)==1 ⇒ denominator = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # temperature (exponential cooling)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        # most violated convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # source = heaviest element among those that influence the max entry
        src = idx[a[idx].argmax()]

        # if the chosen source is already tiny, fall back to a random source
        if a[src] < 1e-14:
            src = rng.integers(n)
            if a[src] < 1e-14:
                continue

        # destination = globally lightest element (avoid src)
        dst = np.argmin(a)
        if dst == src:
            dst = (dst + 1) % n
        # Occasionally (1 % of the time) pick a random destination to keep diversity
        if rng.random() < 0.01:
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

        # transfer up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.40
        if eps <= 0.0:
            continue

        # incremental convolution update
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis rule
        if delta_score <= 0.0 or math.exp(-delta_score / T) > rng.random():
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_score = new_score
            cur_max = new_max

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                if it % 10000 == 0:
                    print(f"[{time.time()-start_time:.2f}s] "
                          f"Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    The routine
    1. builds several analytic seeds,
    2. picks the best,
    3. performs a longer smooth‑max mirror descent,
    4. finishes with a targeted simulated‑annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # length of the vector – use the historic size if available
    try:
        n = int(np.asarray(height_sequence_1).size)
        if n == 0:
            raise ValueError
    except Exception:
        n = 4096

    # ------------------------------------------------------------------
    # 0️⃣  Choose the best analytical seed
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n)

    L_self = _next_pow2(2 * n - 1)
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(seeds):
        sc = _eval_fast(cand, L_self)
        print(f"[{time.time() - start_time:.2f}s] Seed {idx} score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Smooth‑max mirror descent (more iterations than the reference)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        smooth_deadline = deadline - 12.0
        best_seq, best_score = _smooth_descent(
            best_seq, n, smooth_deadline, start_time, max_iters=800_000)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best "
              f"score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Targeted simulated‑annealing polish (uses most of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        best_seq, best_score = _targeted_anneal(
            best_seq, n, anneal_deadline, start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=5e-2,
            end_temp=1e-12,
            max_moves=8_000_000)

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ "
          f"{final_score:.6f}")

    # Return as plain list of floats (the evaluator will clamp and normalise as needed)
    return [float(v) for v in best_seq]
