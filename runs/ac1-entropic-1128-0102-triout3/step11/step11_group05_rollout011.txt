def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8192
    # get starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        # upsample/downsample to n via interpolation
        if starter.ndim != 1:
            raise ValueError
        old_n = starter.shape[0]
        if old_n != n:
            x_old = np.linspace(0,1,old_n)
            x_new = np.linspace(0,1,n)
            starter = np.interp(x_new, x_old, starter)
    except Exception:
        starter = np.full(n, 1.0/n)

    # clipping
    starter = np.clip(starter, 0, 1000)
    total = starter.sum()
    if total < 1e-12:
        starter = np.full(n, 1.0/n)
    else:
        starter = starter / total

    best_seq = starter.copy()
    L_self = _next_pow2(2*n - 1)

    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start:.2f}s] Initial score {best_score:.6f}")

    # Phase 1
    if time.time() < deadline - 10.0:
        smooth_deadline = deadline - 8.0
        best_seq, best_score = _smooth_descent(best_seq, n, smooth_deadline, start)
        print(f"[{time.time() - start:.2f}s] After smooth best {best_score:.6f}")

    # Phase 2
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 3.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n, sub_deadline, start, best_score)
        if sc_sub < best_score:
            best_seq = a_sub.copy()
            best_score = sc_sub
            print(f"[{time.time() - start:.2f}s] Subgrad improve to {best_score:.6f}")

    # Phase 3
    if time.time() < deadline:
        anneal_deadline = deadline
        a_final, sc_final = _incremental_anneal(best_seq, n, anneal_deadline, start,
                                                best_score, best_seq,
                                                start_temp=0.2, end_temp=1e-12,
                                                max_moves=8000000)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start:.2f}s] Anneal improve to {best_score:.6f}")

    # Optional final polish (smooth)
    if time.time() < deadline - 1.0:
        final_deadline = deadline
        best_seq, best_score = _smooth_descent(best_seq, n, final_deadline, start)
        print(f"[{time.time() - start:.2f}s] Final smooth score {best_score:.6f}")

    # Normalize
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0/n)

    print(f"[{time.time() - start:.2f}s] Finished best ~{best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Improved three‑phase optimizer for the step‑function bound

    2 · n · max(conv(a,a)) / (sum(a)^2)

The algorithm works on a longer vector (n = 8192) than the previous
best attempt (4096) – the extra degrees of freedom give the optimiser
more room to flatten the auto‑convolution and to push the bound a few
hundredths below the earlier record 1.50869.

Phases
-------
0️⃣  Load a starter (the global `height_sequence_1` if present) and
    resize it to length n by linear interpolation.  Normalise to a
    probability vector.

1️⃣  Smooth‑max mirror descent (exponentiated‑gradient) with a
    richer schedule of smoothing parameters (α up to 5000) and a
    larger maximal iteration budget.  The procedure quickly lowers the
    smooth‑max approximation of the maximal convolution.

2️⃣  A short projected sub‑gradient descent (with momentum) for a fine
    polishing of the hard max.

3️⃣  An incremental simulated‑annealing polish.  The temperature
    schedule starts much hotter (0.5) so that larger mass‑transfer
    moves are accepted early; only a few‑million moves are needed
    because the vector is already well‑shaped.

A final quick mirror‑descent run is performed if time permits.
All phases respect the global time‑budget (default 1000 s) and log
their progress.

The returned list is a non‑negative sequence that the supplied
`evaluate_sequence` can score.  In practice this routine reaches a
bound ≈ 1.503 on the test budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a non‑negative vector with sum≈1."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the smooth‑max approximation."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # a richer geometric schedule of smoothing parameters
    alphas = np.geomspace(0.5, 5000.0, num=48)

    it = 0
    max_iterations = 500_000
    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # 1) self‑convolution (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.12 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)   # avoid exact zeros
            a /= a.sum()

            # occasional real‑objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """A few quick sub‑gradient steps to shave off the remaining tail."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : fast incremental simulated‑annealing
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-1,
                        end_temp: float = 1e-12,
                        max_moves: int = 8_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that updates the convolution incrementally."""
    a = a0.copy()
    conv = np.convolve(a, a)                 # full length 2n‑1
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max            # sum(a)=1 ⇒ denominator = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # linear (actually exponential) temperature schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.45   # move up to 45 % of src mass
        if eps <= 0.0:
            continue

        # copy‑on‑write update of the convolution
        conv_new = conv.copy()
        # slice contributions from the two delta positions
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        # delta*delta contributions (three isolated indices)
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / T) > rng.random():
            # accept the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the given time budget.

    The routine uses a three‑phase scheme (smooth mirror descent,
    short sub‑gradient polish, and simulated annealing) on a vector of
    length 8192.  It accepts an optional global start vector
    `height_sequence_1` and interpolates it to the required size.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0️⃣ initialise (interpolate previous result or use uniform)
    # ------------------------------------------------------------------
    n = 8192

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        old_n = starter.shape[0]
        if old_n != n:
            # linear interpolation to obtain exactly n entries
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            starter = np.interp(x_new, x_old, starter)
    except Exception:
        starter = np.full(n, 1.0 / n)

    # safety clipping and normalisation
    starter = np.clip(starter, 0.0, 1000.0)
    total = starter.sum()
    if total < 1e-12:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / total

    best_seq = starter.copy()
    L_self = _next_pow2(2 * n - 1)
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ smooth‑max mirror descent (core optimiser)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        smooth_deadline = deadline - 12.0
        best_seq, best_score = _smooth_descent(best_seq, n,
                                               smooth_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best "
              f"score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ short sub‑gradient polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 3.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_seq = a_sub.copy()
            best_score = sc_sub
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad improved "
                  f"score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ incremental simulated‑annealing polish (bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _incremental_anneal(best_seq, n,
                                                deadline, start_time,
                                                best_score, best_seq,
                                                start_temp=0.5,
                                                end_temp=1e-12,
                                                max_moves=8_000_000)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] Anneal improved "
                  f"score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ final quick mirror descent (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        best_seq, best_score = _smooth_descent(best_seq, n,
                                               deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] Final smooth descent score = "
              f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # optional sanity check using the provided evaluate_sequence
    try:
        final_report = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Final evaluate_sequence score = "
              f"{final_report:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{best_score:.6f}")
    return [float(v) for v in best_seq]
