"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a mixed strategy:
* fast parametric sweeps (including a new exp‑power family)
* Adam‑style smoothed gradient descent whose iterates are projected onto
  the decreasing simplex (isotonic regression + simplex normalisation)
* optional DE on a low‑dimensional Gaussian‑mixture model
* optional up‑sampling to a larger length and a second optimisation stage
* a tiny random hill‑climb polish

The algorithm respects the 1000 s budget, prints progress, and finally
returns a list compatible with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional imports – if unavailable the corresponding phase is skipped
try:
    from scipy.optimize import linprog, differential_evolution
    _has_scipy = True
except Exception:                     # pragma: no cover
    _has_scipy = False


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex / monotone‑simplex projection utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (v≥0, sum=1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _isotonic_regression_non_decreasing(y: np.ndarray) -> np.ndarray:
    """
    PAVA implementation for non‑decreasing isotonic regression.
    Returns the fitted vector (same length as y).
    """
    n = y.shape[0]
    x = y.astype(np.float64).copy()
    w = np.ones(n, dtype=np.float64)
    i = 0
    while i < n - 1:
        if x[i] <= x[i + 1]:
            i += 1
        else:
            # merge blocks i and i+1
            total_w = w[i] + w[i + 1]
            avg = (x[i] * w[i] + x[i + 1] * w[i + 1]) / total_w
            x[i] = avg
            x[i + 1] = avg
            w[i] = total_w
            w[i + 1] = total_w
            i = max(i - 1, 0)
    return x


def _project_decreasing_simplex(a: np.ndarray) -> np.ndarray:
    """
    Project onto the set of non‑negative, non‑increasing vectors that sum to one.
    First a decreasing isotonic regression (via PAVA on -a), then rescale to
    sum=1 (preserving order).  Returns a new ndarray.
    """
    a = np.maximum(a, 0.0)
    y = -a
    y_iso = _isotonic_regression_non_decreasing(y)
    proj = -y_iso
    proj = np.maximum(proj, 0.0)
    s = proj.sum()
    if s > 0:
        proj /= s
    else:
        proj = np.full_like(proj, 1.0 / proj.shape[0])
    return proj


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑style smoothed gradient descent with monotone‑simplex projection
# ----------------------------------------------------------------------
def _adam_smooth_descent_monotone(start_a: np.ndarray,
                                  deadline: float,
                                  start_time: float,
                                  L_fft: int,
                                  L_grad: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max smoothed objective.
    Each iterate is projected onto the decreasing simplex.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    tau = 0.08               # initial temperature
    tau_min = 1e-9
    tau_decay = 0.9993
    step0 = 0.12

    while time.time() < deadline:
        it += 1

        # convolution and soft‑max weights
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient (as in the original code)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto decreasing simplex
        a = _project_decreasing_simplex(a)

        # scoring
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamMon it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

def _gaussian_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """Build a non‑negative sequence of length n from a mixture of Gaussians."""
    seq = np.zeros(n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    K = len(params) // 3
    for i in range(K):
        amp = max(0.0, params[3 * i])
        cen = params[3 * i + 1]
        wid = max(1e-3, params[3 * i + 2])
        seq += amp * np.exp(- ((idx - cen) ** 2) / (2.0 * wid * wid))
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)
    return seq


def _de_optimise(n: int,
                 L_fft: int,
                 deadline: float,
                 start_time: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """Run differential evolution on a Gaussian‑mixture parameterisation."""
    if not _has_scipy:                     # pragma: no cover
        return best_seq, best_score

    K = 5                                 # number of Gaussian components
    dim = 3 * K
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 10.0))        # amplitude
        bounds.append((0.0, n - 1))       # centre
        bounds.append((1.0, n / 2.0))     # width

    def de_obj(params):
        seq = _gaussian_sequence(params, n)
        return _eval_fast(seq, L_fft)

    remaining = deadline - time.time()
    maxiter = max(1, int(remaining // 9))
    maxiter = min(maxiter, 35)

    print(f"[{time.time() - start_time:.2f}s] Starting DE (maxiter={maxiter})")
    result = differential_evolution(de_obj,
                                    bounds,
                                    maxiter=maxiter,
                                    popsize=12,
                                    tol=1e-7,
                                    polishing=False,
                                    seed=0,
                                    workers=1,
                                    updating='deferred')
    seq_de = _gaussian_sequence(result.x, n)
    score_de = _eval_fast(seq_de, L_fft)
    if score_de < best_score:
        best_score = score_de
        best_seq = seq_de.copy()
        print(f"[{time.time() - start_time:.2f}s] DE improved score={score_de:.6f}")
    return best_seq, best_score

def _guided_move(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 L_fft: int,
                 max_moves: int = 5000) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        conv = _conv_self(a, L=L_fft)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * 0.14
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] GuidedMove it={it:,} score={new_score:.6f}")

    return a, best_score


def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.09
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return a, best_score

    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full(new_n, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The pipeline:
    1️⃣  fast parametric families (geom, power, beta, exp‑power)
    2️⃣  Adam smoothing descent with decreasing‑simplex projection (multiple restarts)
    3️⃣  optional DE on a Gaussian mixture (low‑dimensional)
    4️⃣  optional up‑sampling to a longer length and a second descent stage
    5️⃣  guided move + hill‑climb polish
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Basic parameters
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # Baseline (uniform)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try to use a global starter if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:                               # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric families (including a new exp‑power family)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric family (already explored, but we make it a bit denser)
    eps_grid = np.logspace(-4, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = r ** idx
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        # blend with uniform – a few λ values
        for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.1f} score={sc:.6f}")

    # power law family
    p_grid = np.linspace(-1.0, 2.5, 55)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = (idx + 1) ** p
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type family (finer grid)
    beta_vals = np.linspace(0.0, 2.5, 26)
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")

    # new “exp‑power” family:   a_i ∝ exp( -α * (i/(n‑1))^β )
    alpha_grid = np.logspace(-4, -0.1, 30)      # α from 1e‑4 up to ~0.8
    beta_grid = np.linspace(0.2, 3.5, 30)
    norm_factor = n - 1
    for alpha in alpha_grid:
        for beta in beta_grid:
            if time.time() >= deadline - 30:
                break
            raw = np.exp(-alpha * (idx / norm_factor) ** beta)
            if raw.sum() == 0:
                continue
            raw /= raw.sum()
            sc = _eval_fast(raw, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = raw.copy()
                print(f"[{time.time() - start_time:.2f}s] ExpPower α={alpha:.2e} β={beta:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style smoothed gradient descent with monotone projection
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # generate a start point: noisy perturbation of current best,
        # or a fresh random decreasing simplex point
        if np.random.rand() < 0.6:
            noise = 0.03 * np.random.randn(n) * best_seq
            start = best_seq + noise
        else:
            # random decreasing vector: generate random values then isotonic regress.
            rand = np.random.rand(n)
            start = _project_decreasing_simplex(rand)

        start = np.maximum(start, 0.0)
        start = _project_decreasing_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 13.0)
        cand_seq, cand_score = _adam_smooth_descent_monotone(start,
                                                             sub_deadline,
                                                             start_time,
                                                             L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamMon new global best score={best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Gaussian‑mixture DE (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0 and best_score > 1.503:
        best_seq, best_score = _de_optimise(n, L_fft, deadline,
                                            start_time, best_seq, best_score)

    # ------------------------------------------------------------------
    # 4️⃣  Optional up‑sampling to a larger length and a second descent stage
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0 and best_score > 1.503:
        # double the length (keeping the same overall shape)
        n2 = n * 2
        L_fft2 = _next_pow2(2 * n2 - 1)
        L_grad2 = _next_pow2(3 * n2)

        up_seq = _upsample(best_seq, n2)
        print(f"[{time.time() - start_time:.2f}s] Upsampled to {n2} points, start score={_eval_fast(up_seq, L_fft2):.6f}")

        # a brief Adam descent on the larger grid
        while time.time() < deadline - 30.0:
            noise = 0.02 * np.random.randn(n2) * up_seq
            start = up_seq + noise
            start = np.maximum(start, 0.0)
            start = _project_decreasing_simplex(start)

            sub_deadline = min(deadline - 5.0, time.time() + 10.0)
            cand_seq, cand_score = _adam_smooth_descent_monotone(start,
                                                                 sub_deadline,
                                                                 start_time,
                                                                 L_fft2, L_grad2)
            if cand_score < best_score:
                best_score = cand_score
                up_seq = cand_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Upsampled AdamMon best={best_score:.6f}")

            if best_score <= 1.503:
                break

        # keep the up‑sampled version as the current best
        best_seq = up_seq.copy()
        n = n2
        L_fft = L_fft2
        L_grad = L_grad2

    # ------------------------------------------------------------------
    # 5️⃣  Guided sub‑gradient move (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0 and best_score > 1.503:
        best_seq, best_score = _guided_move(best_seq, deadline,
                                            start_time, L_fft)

    # ------------------------------------------------------------------
    # 6️⃣  Deterministic hill‑climb polish (final tiny tweaks)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline,
                                           start_time, L_fft)

    # ------------------------------------------------------------------
    # Normalisation and final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
