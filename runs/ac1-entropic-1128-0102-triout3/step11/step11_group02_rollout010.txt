"""
A fast smoothing‑gradient search for the step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:

1. **Initialisation** – start from the best known construction (global
   variable ``height_sequence_1``) or, if unavailable, from a symmetric
   triangular “tent’’ shape, normalised to the probability simplex.

2. **Smoothed projected gradient descent** – the non‑smooth objective
   ``max(conv)`` is approximated by a soft‑max
   ``f_σ(a) = (1/σ)·log Σ exp(σ·conv_k)``.
   The gradient of ``f_σ`` can be written in closed form as

        ∇f_σ(a) = a + (a ⋆ w) ,

   where ``w`` are the soft‑max weights (∑w=1) and ``⋆`` denotes the
   linear (non‑circular) correlation.  The correlation is computed via
   FFTs (O(n log n)).  At each iteration we move a small step opposite
   to the gradient and project back onto the simplex.  The smoothing
   parameter ``σ`` is increased linearly from a small value to a larger
   one, turning the smooth surrogate gradually into the true infinity‑norm.

3. **Light simulated‑annealing polish** – after the gradient phase a few
   seconds are spent moving tiny random packets of mass between indices
   (Metropolis rule) to escape any remaining shallow local minima.

All helpers are defined at the top level so that the judge can pickle
them if needed.  The routine respects the 1000 s budget and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a via a real FFT.  Result length = 2·n‑1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Equivalent to evaluate_sequence (without checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0 , Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient utilities
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, sigma: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns the gradient of the soft‑max surrogate of max(conv(a,a))
    together with the convolution itself.
    Gradient = a + (a ⋆ w)  where w_i ∝ exp(σ·conv_i).
    """
    n = a.shape[0]

    # 1) linear self‑convolution
    conv = _conv_self(a)               # length 2n‑1
    maxc = conv.max()

    # 2) soft‑max weights (numerically stable)
    exp_term = np.exp(sigma * (conv - maxc))
    w = exp_term / exp_term.sum()      # still length 2n‑1, sum(w)=1

    # 3) correlation a ⋆ w   (size = n)
    m = conv.shape[0]                  # = 2n‑1
    L = _next_pow2(n + m - 1)
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    # cross‑correlation via IFFT( conj(FA) * FB )
    corr_full = np.fft.irfft(np.conj(Fa) * Fw, n=L)[: n + m - 1]
    start = m - 1                       # index where lag = 0 starts
    corr = corr_full[start : start + n]

    # 4) gradient
    grad = a + corr
    return grad, conv


def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    sigma_start: float = 0.1,
                    sigma_end: float = 6.0,
                    sigma_ramp_iters: int = 5000,
                    eta0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed surrogate.
    Returns the best iterate observed and its (true) objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # ensure we start on the simplex
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # scheduling of the smoothing parameter
        if it <= sigma_ramp_iters:
            sigma = sigma_start + (sigma_end - sigma_start) * (it / sigma_ramp_iters)
        else:
            sigma = sigma_end

        # gradient of the surrogate
        grad, conv = _smooth_gradient(a, sigma)

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # gradient step + projection onto simplex
        a = a - eta * grad
        a = _project_simplex(a)

        # evaluate true objective (conv already computed)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)  # a.sum() is 1 after projection
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Descent it={it:,} "
                  f"sigma={sigma:.2f} score={cur_score:.6f}")

        # occasional tiny random jitter to escape plateaus
        if it % 2000 == 0:
            rng = np.random.default_rng()
            eps = 1e-6 * rng.random(n)
            a = a + eps
            a = _project_simplex(a)

    return best_a, best_score

# Light simulated‑annealing polish (unchanged from the reference)

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap annealing that repeatedly moves a small random packet of mass.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 40_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two yields fast FFTs.
    # --------------------------------------------------------------
    n = 16384                    # feel free to change (must fit in time)

    # --------------------------------------------------------------
    # 1) Warm start – try to use the globally stored best construction.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
                a0 = starter.copy()
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        # fallback: symmetric triangular “tent’’ on the simplex
        x = np.arange(n, dtype=float) / (n - 1)
        tent = np.minimum(x, 1.0 - x)
        if tent.sum() == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = tent / tent.sum()

    # --------------------------------------------------------------
    # 2) Smoothed projected gradient descent.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a cushion for polishing
        sub_deadline = time.time() + remaining * 0.85
        refined_seq, refined_score = _smooth_descent(
            a0,
            sub_deadline,
            start_time,
            sigma_start=0.05,
            sigma_end=8.0,
            sigma_ramp_iters=6000,
            eta0=0.3)
        if refined_score < _objective(a0):
            a0 = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(a0, deadline, start_time)
        if anneal_score < _objective(a0):
            a0 = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={anneal_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation & conversion.
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0:
        a0 = a0 / total
    else:
        a0 = np.full(n, 1.0 / n)

    return [float(v) for v in a0]
