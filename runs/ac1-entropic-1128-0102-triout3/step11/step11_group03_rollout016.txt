def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2*n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n - 1]
    return conv

def _conv_fft(a,b):
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:n+m-1]
    return conv

def _objective(seq):
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq**2))

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

"""
Search for a low‑score step function.

The algorithm works in three stages:

1. **Coarse family sweep** –  For a reasonably large n (8192) we
   generate U‑shaped families   a_i ∝ (x_i·(1−x_i)+ε)^{−α}  and a few
   mixtures with a uniform component.  The best candidate is kept.

2. **Smooth Adam optimisation** –  Starting from the best candidate we
   minimise a smooth surrogate of the true objective.
   The surrogate replaces the hard max by a soft‑max with an
   increasing temperature β, so that a differentiable gradient
   (derived from the convolution) is available.  Adam with simplex
   projection refines the sequence while respecting non‑negativity
   and the ℓ₁‑normalisation (the score is homogeneous).

3. **Light annealing polish** –  A few thousand random mass‑transfer
   moves are tried; a move is kept if it improves the exact objective
   (or is accepted with a tiny Metropolis probability).

All steps respect the given time budget.  The final sequence is
returned as a plain list of floats; scaling does not affect the
score because the objective is homogeneous.
"""

# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using FFT."""
    n, m = a.shape[0], b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (the one used by the verifier)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights of a step function that
    (hopefully) yields a low value of the evaluation function.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # 1) Coarse sweep of the U‑shaped family (α, ε) and uniform mixes
    # ------------------------------------------------------------------
    n = 8192                         # length of the final sequence
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    best_seq = None
    best_score = float("inf")

    alphas = np.linspace(0.30, 0.55, 26)      # step ≈ 0.01
    epsilons = [0.0, 1e-12, 1e-9, 1e-6]
    mixes = [0.0, 0.25, 0.5, 0.75, 1.0]       # λ in the mixture with uniform

    for eps in epsilons:
        base = x * (1.0 - x) + eps
        for alpha in alphas:
            w = base ** (-alpha)
            if np.any(np.isinf(w)):
                continue
            for lam in mixes:
                a = lam * w + (1.0 - lam) * np.ones_like(x)
                a = a / a.sum()
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] "
                        f"sweep α={alpha:.3f} eps={eps:g} λ={lam:.2f} score={score:.6f}"
                    )
                if best_score < 1.503:
                    break
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 2) Try the global starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur_score = _objective(starter)
                print(
                    f"[{time.time() - start_time:.2f}s] "
                    f"global starter score={cur_score:.6f}"
                )
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Fallback to uniform if nothing succeeded yet
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)
        print(
            f"[{time.time() - start_time:.2f}s] "
            f"fallback uniform score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4) Adam optimisation on a smooth surrogate of the max‑conv objective
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # already normalised to sum = 1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05                               # base learning rate
    it = 0
    beta = 1.0                               # soft‑max temperature
    max_beta = 4000.0

    while time.time() < deadline:
        it += 1

        # exact convolution and true score (kept for bookkeeping)
        conv = _conv_self(a)
        maxc = conv.max()
        true_score = 2.0 * n * maxc / (a.sum() ** 2)
        if true_score < best_score:
            best_score = true_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] "
                f"Adam iter={it:,} true_score={true_score:.6f}"
            )

        # increase the soft‑max temperature gradually
        if it > 10:
            beta = min(max_beta, beta * 1.002)

        # soft‑max weights (they sum to 1)
        exp_terms = np.exp(beta * (conv - maxc))          # stability by shifting
        w = exp_terms / exp_terms.sum()

        # gradient of the smooth surrogate:
        #   ∇O = 4·n·g  where g_i = Σ_j a_j·w_{i+j}
        w_rev = w[::-1]
        conv_aw = _conv_fft(a, w_rev)                    # length = 3n‑2
        Lw = 2 * n - 1
        idx = Lw - 1 - np.arange(n)                      # indices selecting g_i
        g = conv_aw[idx]
        grad = 4.0 * n * g

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step

        # keep a on the simplex (non‑negative, sum = 1)
        a = _project_simplex(a)

    # ------------------------------------------------------------------
    # 5) Very light simulated‑annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < start_time + budget_s - 0.2:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng(seed + 12345)
        total_steps = 20000
        for step_id in range(1, total_steps + 1):
            if time.time() >= start_time + budget_s - 0.1:
                break
            T = 0.02 * (0.5 / 0.02) ** (step_id / total_steps)   # exponential cooling
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps_move = a[src] * rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move
            new_score = _objective(a_prop)
            delta = new_score - cur_score
            if delta < 0.0 or math.exp(-delta / T) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] "
                        f"anneal step={step_id:,} best={new_score:.6f}"
                    )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(
                f"[{time.time() - start_time:.2f}s] "
                f"final anneal improved to {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6) Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # The verifier only cares about non‑negative numbers; scaling does not matter.
    return [float(v) for v in best_seq]
