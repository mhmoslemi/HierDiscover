```
def grad_softmax(a, alpha, L_self, L_grad):
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_self)
    conv = np.fft.irfft(Af * Af, n=L_self)[:2*n - 1]
    exp_arg = alpha * conv
    max_arg = exp_arg.max()
    w = np.exp(exp_arg - max_arg)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum
    # gradient via convolution of w with reversed a:
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L_grad)
    Revf = np.fft.rfft(a_rev, n=L_grad)
    grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[:3*n - 2]
    grad = 2.0 * grad_full[n-1:n-1+n]
    return grad
```

def _gs_step(a, n):
    # full FFT length n
    F = np.fft.fft(a)   # complex length n
    mag = np.abs(F)
    phase = np.angle(F)
    mag0 = mag[0]
    # compute target magnitude for non-DC components
    # avoid division by zero if n==1
    if n <= 1:
        return a
    total_power = np.sum(mag ** 2)
    # ensure non-negative target squared magnitude
    target_sq = (total_power - mag0 ** 2) / (n - 1)
    if target_sq < 0:
        target_sq = 0.0
    target_mag = np.sqrt(target_sq)
    # create new magnitude array
    mag_new = mag.copy()
    mag_new[1:] = target_mag
    # Reconstruct
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.real(np.fft.ifft(F_new))
    # clip negativity, normalize
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        a_new = np.full(n, 1.0 / n)
    return a_new

"""
Propose a step‑function (list of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm (≈ 1000 s total)
--------------------------------
1. Load the best known sequence (global ``height_sequence_1``) if available,
   otherwise start from the uniform distribution.

2. **Quick local polishing** – a few projected sub‑gradient steps to
   improve the current best point (this is cheap and never hurts).

3. **Random‑restart alternating projections** – each restart begins from a
   random Dirichlet vector.  We then run a Gerchberg‑Saxton style
   iteration that enforces *constant magnitude* of the full Fourier
   spectrum (except the DC term) while keeping the signal non‑negative.
   This operation tends to flatten the autocorrelation and typically
   reduces the max‑convolution value.

   The iteration is cheap (two FFTs per step) so we can afford many
   restarts; each restart stops after a fixed number of GS steps or when
   the wall‑clock budget forces us to quit.

4. **Hill‑climb polish** – after every restart we apply a fast
   mass‑transfer hill‑climber (the same routine used in the reference
   solution) for a few thousand moves.  This locally exploits any
   remaining improvement.

5. **Final hill‑climb** – when the time budget is almost exhausted we
   perform a longer hill‑climb on the global best sequence.

All phases respect a hard wall‑clock deadline; the best sequence found
so far is always returned (as a plain Python list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fast exact objective)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on the exact objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    stagn = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagn += 1

        if stagn >= 15_000:
            break

        # indices attaining the max (within a tiny tolerance)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_seq: np.ndarray,
                cur_score: float,
                max_moves: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves; keep a move only if it improves the exact objective.
    """
    a = a0.copy()
    best_seq = cur_seq.copy()
    best_score = cur_score
    cur_obj = cur_score
    rng = np.random.default_rng()

    moves = 0
    while time.time() < deadline and moves < max_moves:
        moves += 1
        src = rng.choice(n, p=a)          # source weighted by current mass
        dst = rng.integers(0, n)          # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 30 % of src mass
        eps = a[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_obj - 1e-12:
            a = a_prop
            cur_obj = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Gerchberg‑Saxton (alternating projections) – constant‑magnitude + non‑negativity
# ----------------------------------------------------------------------
def _gs_step(a: np.ndarray) -> np.ndarray:
    """
    One iteration of GS:
    – keep DC magnitude (preserves sum)
    – force all other FFT magnitudes to be equal (keeps total power)
    – enforce non‑negativity in the time domain
    The result is renormalised to sum = 1.
    """
    n = a.shape[0]
    # full FFT (complex)
    F = np.fft.fft(a)
    mag = np.abs(F)
    phase = np.angle(F)

    # preserve DC magnitude (sum = 1)
    mag0 = mag[0]

    # total power = sum |F_k|^2 = n * sum a_i^2
    total_power = np.sum(mag * mag)

    if n > 1:
        target_sq = (total_power - mag0 * mag0) / (n - 1)
        target_sq = max(target_sq, 0.0)
        target_mag = math.sqrt(target_sq)
    else:
        target_mag = 0.0

    mag_new = mag.copy()
    mag_new[1:] = target_mag                     # flatten all non‑DC components

    # reconstruct signal
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.real(np.fft.ifft(F_new))

    # enforce constraints
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full(n, 1.0 / n)
    return a_new


def _gs_optimize(a0: np.ndarray,
                 n: int,
                 L_self: int,
                 max_iters: int,
                 deadline: float,
                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Run a short Gerchberg‑Saxton loop starting from a0.
    Returns the best sequence seen inside the loop and its score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _eval_fast(a, L_self)

    it = 0
    stagn = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        a = _gs_step(a)
        cur_score = _eval_fast(a, L_self)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            stagn = 0
            print(f"[{time.time() - start_time:.2f}s] GS it={it:,} score={cur_score:.6f}")
        else:
            stagn += 1

        # stop early if nothing improves for a while
        if stagn >= 5000:
            break

    return best_seq, best_score
