"""
Proposes a low‑objective step sequence.

The method consists of three stages:

1. **Coarse parametric sweep** – explores the family  
   a_i ∝ (x_i·(1−x_i)+ε)^{−α} and mixtures of this shape with the uniform
   distribution.  The sweep is performed on a relatively large grid (n=8192)
   and the best candidate is kept.

2. **Projected Adam optimisation** – starting from the best sweep candidate
   (and a few random restarts) we minimise the exact objective
   2·n·max(conv(a,a))/sum(a)^2.
   A log‑sum‑exp smoothing of the max is used to obtain a smooth gradient;
   Adam updates are applied and the iterate is projected onto the simplex
   after each step.

3. **Light simulated‑annealing polish** – a temperature‑controlled
   mass‑transfer walk on the simplex is performed for a few‑seconds to
   escape possible local minima.

The whole search respects a 1000 s wall‑time limit and returns a plain
list of non‑negative floats (the sequence sum is normalised to 1, which
does not affect the score because the objective is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)

    Computed with an FFT‑based linear convolution for speed.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))

            elapsed = time.time() - start_time
            prop = min(1.0, elapsed / (budget_s - 10.0))
            beta = 5.0 + 395.0 * prop                      # final β ≈ 400

            # ---- soft‑max weights (log‑sum‑exp) ----------------------------
            beta_conv = beta * conv
            max_beta = beta_conv.max()
            exp_shift = np.exp(beta_conv - max_beta)       # stable exponentiation
            w = exp_shift / exp_shift.sum()               # shape (2n‑1,)

            # ---- gradient of the smooth surrogate -------------------------
            # B·a = cross‑correlation of w with reversed a
            a_rev = a[::-1]
            w_fft = np.fft.rfft(w, n=L)
            a_rev_fft = np.fft.rfft(a_rev, n=L)
            corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
            Ba = corr_full[n - 1:n - 1 + n]                # length n
            grad = 4.0 * n * Ba                           # ∇F_s

            # ---- Adam update ------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            a = a - step
            a = _project_simplex(a)

            # ---- keep the best exact objective seen so far -----------------
            if iteration % 50 == 0:
                cur_score = _objective(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] iter={iteration:,} best={cur_score:.6f}")

        return a

    # run Adam from the best sweep candidate
    _adam_optimize(best_seq)

    # a few random restarts (if time permits)
    rng = np.random.default_rng()
    for _ in range(3):
        if time.time() >= deadline - 5.0:
            break
        rand_vec = rng.random(n)
        rand_vec = _project_simplex(rand_vec)
        _adam_optimize(rand_vec)

    # --------------------------------------------------------------
    # 4) Very light simulated‑annealing polish (mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score

        max_moves = 50000
        T0 = 0.02                                 # initial temperature

        for move in range(1, max_moves + 1):
            if time.time() >= deadline:
                break

            # pick source proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            # move a modest fraction of the source mass
            eps = a[src] * rng.random() * 0.15        # up to 15 %
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            new_score = _objective(a_prop)
            delta = new_score - cur_score
            # linear cooling schedule
            T = T0 * (1.0 - move / max_moves)

            if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal move={move:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
