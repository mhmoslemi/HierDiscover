"""
Very fast three‑phase optimizer for the “max‑self‑convolution / squared‑sum’’
objective

    Φ(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works on a probability simplex (∑a = 1, a_i ≥ 0) and consists of

1️⃣  Smoothed‑max mirror descent (exponentiated‑gradient) with an aggressive
    geometric schedule for the smoothness parameter α and an Adam‑style
    adaptive step size.  This quickly pushes the peak of the convolution down.

2️⃣  A short projected sub‑gradient phase (with momentum) that refines the
    current point on the true L∞‑objective.

3️⃣  Incremental simulated‑annealing on the original (non‑smoothed) problem.
    The convolution is updated in‑place so that a few‑million cheap mass‑transfer
    moves can be performed within the 1000 s budget.

The length of the step‑function is doubled (n = 8192) compared to the
reference implementation – the larger dimensionality gives the optimiser
more freedom to flatten the self‑convolution and thus to reach a lower bound
(the theoretical limit for large n is ≈ 1).  All helpers are top‑level and the
routine never exceeds the time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """
    Exact objective for a probability vector a (sum ≈ 1).
    Returns 2·n·max(conv(a,a)) / (sum(a)^2).
    """
    conv = np.fft.irfft(np.fft.rfft(a, n=L_self) *
                        np.fft.rfft(a, n=L_self), n=L_self)[: 2 * a.shape[0] - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * a.shape[0] * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – smoothed‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # geometric schedule: start very smooth and end with a very sharp max
    alphas = np.geomspace(0.5, 4000.0, num=64)

    # Adam‑style accumulators for adaptive steps
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12
    it = 0

    for alpha in alphas:
        while time.time() < deadline - 0.2:
            it += 1

            # 1) self‑convolution (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights (softmax on α·conv)
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)               # stabilise
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

            # 3) gradient of the smoothed max (2·w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam‑style adaptive exponentiated‑gradient step
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # step size – tuned for the simplex geometry
            eta = 0.08 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps_adam))

            # keep positivity and re‑normalise onto the simplex
            a = np.maximum(a, 1e-18)
            a /= a.sum()

            # occasional evaluation of the true (non‑smoothed) objective
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

            if time.time() >= deadline - 0.2:
                break

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent (with momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        # full self‑convolution (FFT) – cheaper than np.convolve for this size
        conv = np.fft.irfft(np.fft.rfft(a, n=L_grad) *
                            np.fft.rfft(a, n=L_grad), n=L_grad)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

        # index where the current maximum occurs (first occurrence)
        k = int(conv.argmax())

        # indices that actually contribute to this peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max‑convolution w.r.t a (see paper for derivation)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected simple step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # simplex projection (non‑negativity + normalisation)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – fast incremental simulated‑annealing polish
# ----------------------------------------------------------------------
def _incremental_anneal(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-3,
                        end_temp: float = 1e-12,
                        max_moves: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealer that updates the convolution incrementally.
    a – current probability vector (∑a = 1)
    conv – full linear convolution a*a (length 2n‑1)
    """
    a = a0.copy()
    conv = np.convolve(a, a)               # cheap for n≈8192
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max          # denominator = 1 (sum = 1)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    it = 0

    # pre‑allocate a buffer for the updated convolution to avoid frequent allocations
    conv_buf = np.empty_like(conv)

    while time.time() < deadline and it < max_moves:
        it += 1

        # linear annealing schedule on log‑scale
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----------------
        conv_buf[:] = conv

        # contribution of the moved mass with the *old* distribution
        conv_buf[src:src + n] -= 2.0 * eps * a
        conv_buf[dst:dst + n] += 2.0 * eps * a

        # the tiny quadratic terms (only three positions change)
        conv_buf[2 * src] += eps * eps
        conv_buf[2 * dst] += eps * eps
        conv_buf[src + dst] -= 2.0 * eps * eps
        # -------------------------------------------------------------

        new_max = conv_buf.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / max(T, 1e-30)) > rng.random():
            # accept
            a[src] -= eps
            a[dst] += eps
            conv = conv_buf.copy()
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
