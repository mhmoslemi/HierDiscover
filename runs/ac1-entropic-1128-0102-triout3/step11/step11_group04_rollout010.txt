def compute_subgradient(a: np.ndarray, conv: np.ndarray, tol=1e-12):
    cur_max = conv.max()
    active = np.where(np.abs(conv - cur_max) <= tol)[0]
    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i in [lo, hi]; a_{k-i} corresponds to a[j] where j = k-i
        # Could vectorize: idx = np.arange(lo, hi+1); grad[idx] += 2*a[k - idx]
        idx = np.arange(lo, hi+1)
        grad[idx] += 2 * a[k - idx]
    return grad, cur_max, active
```

def gradient_mass_transfer(a, n, L_self, deadline, start_time, max_iters=500000):
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)
    cur_score = best_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        grad, cur_max, active = compute_subgradient(a, conv) # compute gradient
        # choose source idx = argmax(grad)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] < 1e-12:
            # maybe skip this iteration
            continue
        # choose epsilon
        # we might adapt epsilon = min(a[src], step_factor * something)
        eps = a[src] * 0.1  # 10% of source mass
        # try decreasing epsilon gradually if no improvement
        improved = False
        for factor in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]:
            eps = a[src] * factor
            if eps < 1e-12:
                break
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            # ensure nonnegative
            if a_prop[src] < -1e-15:
                continue
            new_score = _eval_fast(a_prop, L_self)
            if new_score < cur_score - 1e-12:
                a = a_prop
                cur_score = new_score
                improved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Grad-transfer iter={it} new best={best_score:.6f}")
                break
        if not improved:
            # optionally random move
            continue
    return best_a, best_score
```

"""
Gradient‑driven mass‑transfer optimizer for the step‑function problem.

The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a)**2)

for a non‑negative vector a (the sum is normalised to 1, so the objective
reduces to 2·n·max(conv)).  The algorithm works as follows:

1. **Initialisation** – start from the best known sequence
   (`height_sequence_1` if available, otherwise a uniform vector) and normalise
   to a probability simplex.

2. **Iterative directed mass transfer** – at each iteration we
   * compute the self‑convolution and locate the (possibly several) indices
     where the maximum is attained,
   * compute the sub‑gradient of the max‑convolution (for an active index
     k the gradient entry i is 2·a_{k‑i}),
   * choose a *source* index with the largest gradient (decreasing its value
     reduces the objective) and a *destination* index with the smallest
     gradient, and transfer a small fraction of the source mass to the
     destination,
   * accept the move only if the exact objective improves; otherwise shrink
     the transferred amount and retry.

   The step size (fraction of the source mass) decays like 1/√t so that the
   search becomes increasingly fine‑grained.

3. **Polishing** – after the directed phase a short random hill‑climb (as in
   the reference implementation) is performed to cleanup any remaining tiny
   improvements.

All operations respect a hard wall‑clock deadline (default 1000 s).  The final
vector is clipped to non‑negative values, renormalised to sum = 1 and returned
as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum(a) ≈ 1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (used for directed moves)
# ----------------------------------------------------------------------


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray, tol: float = 1e-12):
    """
    Returns:
        grad    – ∂ max(conv) / ∂ a   (size n)
        cur_max – current maximum value of conv
        active  – array of indices k where conv_k is (approximately) maximal
    """
    cur_max = conv.max()
    active = np.where(np.abs(conv - cur_max) <= tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)

    # For each active index k add the contribution 2·a_{k-i}
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Directed mass‑transfer optimisation
# ----------------------------------------------------------------------


def _directed_mass_transfer(a0: np.ndarray,
                            n: int,
                            L_self: int,
                            deadline: float,
                            start_time: float,
                            max_iters: int = 2_000_000):
    """
    Repeatedly moves a small amount of mass from a high‑gradient index
    (source) to a low‑gradient index (destination).  Moves are accepted only
    if they improve the exact objective.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_self)
    cur_score = best_score

    stagnation = 0           # counts iterations without improvement
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # 1) current convolution & sub‑gradient
        conv = _conv_self(a, L_self)
        grad, cur_max, active = _subgradient_maxconv(a, conv)

        # 2) pick source (largest gradient) and destination (smallest)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        if src == dst or a[src] < 1e-15:
            # nothing sensible to move – random perturbation for escape
            # (tiny Gaussian jitter followed by simplex projection)
            if it % 500 == 0:
                noise = np.random.default_rng().normal(scale=1e-4, size=n)
                a = _project_simplex(a + noise)
            continue

        # 3) step size decays like 1/sqrt(it)
        step_factor = max(0.2 / math.sqrt(it), 0.001)   # between 0.001 and ~0.02
        eps = a[src] * step_factor

        # 4) try the move; if not improving, shrink eps progressively
        improved = False
        for shrink in (1.0, 0.5, 0.25, 0.1):
            eps_try = eps * shrink
            if eps_try <= 0.0 or eps_try > a[src]:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps_try
            a_prop[dst] += eps_try

            new_score = _eval_fast(a_prop, L_self)
            if new_score < cur_score - 1e-14:     # strict improvement
                a = a_prop
                cur_score = new_score
                improved = True
                stagnation = 0
                if new_score < best_score:
                    best_score = new_score
                    best = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Iter {it:,} new best={best_score:.6f}")
                break

        if not improved:
            stagnation += 1
            # occasional random shake‑up to escape flat regions
            if stagnation >= 5000:
                rng = np.random.default_rng()
                noise = rng.normal(scale=5e-4, size=n)
                a = _project_simplex(a + noise)
                stagnation = 0

    return best, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climb (final polish – identical to the reference version)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_seq: np.ndarray,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climbing: move a random fraction of mass between two indices."""
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.choice(n, p=a)               # source weighted by current mass
        dst = rng.integers(0, n)               # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        eps = a[src] * rng.random() * 0.30     # up to 30 % of source mass
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    The optimisation consists of a directed mass‑transfer phase followed by
    a short hill‑climb polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the external harness
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)   # FFT length for self‑convolution

    # ---------------------------------------------------------------
    # Load a starter vector (global best if available, else uniform)
    # ---------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    # Adjust length if necessary
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')

    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()      # normalise to a simplex

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase‑1 : directed gradient‑driven mass transfers
    # ---------------------------------------------------------------
    if time.time() < deadline - 30.0:   # reserve some time for polishing
        transfer_deadline = deadline - 20.0
        best_seq, best_score = _directed_mass_transfer(
            best_seq, n, L_self, transfer_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After directed transfers best = {best_score:.6f}")

    # ---------------------------------------------------------------
    # Phase‑2 : gentle random hill‑climb (final polish)
    # ---------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _hill_climb(best_seq, n, L_self,
                                        deadline, start_time,
                                        best_seq, best_score)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved score = {sc_final:.6f}")

    # ---------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ---------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
