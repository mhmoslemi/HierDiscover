def _adam_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.5 # base learning rate
    t = 0
    L_fft = _next_pow2(2 * n - 1)
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # gradient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha0 / (np.sqrt(t) + 1) # maybe.
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a = _project_simplex(a)
        # occasional random noise injection
        if t % 5000 == 0:
            # compute improvement over last best?

                # add noise
                noise = 1e-4 * rng.standard_normal(n) * a
                a = a + noise
                a = np.maximum(a, 0.0)
                a = _project_simplex(a)
    return best_a, best_score

def _swap_anneal(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    cur_score = _eval_fast(a, _next_pow2(2 * n - 1))
    iterations = 0
    T0 = 0.01 * cur_score # initial temperature as fraction of score
    while time.time() < deadline:
        iterations += 1
        # gradually decrease temperature
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = T0 * (1 - elapsed / total) # linear cooling
        # propose swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a_prop = a.copy()
        a_prop[i], a_prop[j] = a_prop[j], a_prop[i]
        new_score = _eval_fast(a_prop, _next_pow2(2 * n - 1))
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                # print improvement
                print(f"[{time.time() - start_time:.2f}s] Swap SA iter={iterations} score={new_score:.6f}")
    return best_a, best_score

"""
A compact stochastic optimizer for the step‑function objective

    2·n·max(conv(a, a)) / (sum(a)**2)

The routine follows three phases:

1️⃣  **Broad parametric sweep** – geometric, power, beta and sinusoidal
    families are normalised to unit L¹ and evaluated with an FFT‑based
    convolution.  The best candidate from this cheap search seeds the
    optimiser.

2️⃣  **Projected Adam descent** – sub‑gradient of the active convolution
    entry is used and Adam‑style adaptive step sizes are applied.
    After each update the vector is projected onto the probability simplex.
    Random Gaussian jitter is occasionally injected to escape plateaus.

3️⃣  **Two local‑search refinements**
    * *Swap simulated‑annealing* – random swaps of two entries are accepted
      with a temperature schedule, allowing the ordering of the same
      magnitudes to be optimised.
    * *Tiny hill‑climb* – very small mass transfers between random indices
      polish the final solution.

All heavy work uses FFT (O(n log n)).  The algorithm respects the 1000 s
budget and finally returns a plain list of Python floats suitable for the
provided ``evaluate_sequence`` routine.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a * a (full length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective for a vector that (approximately) sums to 1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projected Adam descent (continuous optimisation)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on
        f(a)=2·n·max(conv(a,a))
    The sum is kept equal to 1 by projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.6                  # base learning rate
    it = 0
    L_fft = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cur_score:.6f}")

        # --- sub‑gradient of the active convolution entry ---
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]               # = 2·a_{k-i}

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha0 / (math.sqrt(it) + 1.0)
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # projection onto simplex
        a = _project_simplex(a)

        # --- occasional jitter to leave plateaus ---
        if it % 3000 == 0:
            # if no improvement for a while, perturb a bit
            if cur_score - best_score > 1e-6:
                noise = 5e-4 * rng.standard_normal(n) * a
                a = a + noise
                a = np.maximum(a, 0.0)
                a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Swap‑based simulated annealing (optimises ordering of the same magnitudes)
# ----------------------------------------------------------------------
def _swap_anneal(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Random swaps of two coordinates accepted with a simple linear
    temperature schedule.  The sum stays 1, no projection needed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    cur_score = _eval_fast(a, _next_pow2(2 * n - 1))
    rng = np.random.default_rng()
    it = 0

    # start temperature as a small fraction of the current score
    T0 = 0.02 * cur_score

    while time.time() < deadline:
        it += 1
        # linear cooling
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = T0 * max(0.0, 1.0 - elapsed / total)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap entries i and j
        a_prop = a.copy()
        a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        new_score = _eval_fast(a_prop, _next_pow2(2 * n - 1))
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Swap SA it={it:,} "
                      f"score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(start_a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int) -> tuple[np.ndarray, float]:
    """
    Transfer an infinitesimal amount of mass between two random indices.
    Only improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 5e-5   # a few · 10⁻⁵ of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb "
                  f"score={best_score:.6f}")

    return best_a, best_score

    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.3

    # problem size ---------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------------
    # baseline candidates
    # --------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # try to use the global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:

                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------------
    # 1️⃣  Broad parametric sweep
    # --------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    for r in np.linspace(0.90, 0.9995, 120):
        if time.time() >= deadline - 30:
            break
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        sc = _eval_fast(geo, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo r={r:.5f} score={sc:.6f}")

    # Power family a_i ∝ (i+1)^p
    for p in np.linspace(-2.0, 3.0, 81):
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1.0, p)
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # Beta‑type family a_i ∝ (i+1)^p * (n-i)^q
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1.0, p) *
                        np.power(n - idx, q))
            if beta_seq.sum() == 0:

            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc

                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")
