"""
Propose a low‑score step function.

The search consists of three phases:

1. **Coarse‑to‑fine sweep of the arcsine‑type family**
   a_i ∝ (x_i·(1−x_i)+ε)^−α with α∈[0.30,0.80].
   For each α we also try convex mixtures with a uniform
   distribution (mixing weight λ∈[0,1]).  The best candidate from this
   sweep is kept.

2. **Projected sub‑gradient descent (Adam‑style)**
   Starting from the best candidate we run a fast projected Adam
   optimiser that works directly on the exact non‑smooth objective
   2·n·max(conv(a,a))/‖a‖₁² (here a is kept on the simplex
   Σa_i=1, a_i≥0).  The optimizer runs for most of the remaining
   time and keeps the best iterate seen.

3. **Light simulated‑annealing polish**
   A handful of random mass‑transfer moves are attempted; a move is
   kept if it improves the score (or is accepted with a tiny
   Metropolis probability).  This helps to escape possible tiny
   local traps.

All work respects the 1000 s wall‑time limit.  The returned list is
already normalised (its sum equals 1), which is safe for the verifier.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective for a normalised (sum==1) sequence
# ----------------------------------------------------------------------
def _score_normalised(a: np.ndarray, n: int, L: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a with Σa=1."""
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Coarse‑to‑fine grid search over the (x·(1‑x)+ε)^{‑α} family
# ----------------------------------------------------------------------
def _grid_search(n: int, L: int, deadline: float, start_time: float):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    base0 = x * (1.0 - x)                         # common part (ε=0)
    uniform = np.full(n, 1.0 / n)                # uniform distribution
    epsilons = [0.0, 1e-9]                       # tiny regulariser
    lam_grid = np.linspace(0.0, 1.0, 11)         # mixture weights

    best_a = None
    best_score = float("inf")
    best_alpha = None
    best_eps = None
    best_lam = None

    # ---- coarse sweep -------------------------------------------------
    coarse_alphas = np.linspace(0.30, 0.80, 51)   # step ≈0.01
    for eps in epsilons:
        base = base0 + eps
        for alpha in coarse_alphas:
            if time.time() >= deadline:
                break
            w = base ** (-alpha)
            w_norm = w / w.sum()                 # ensure Σ=1
            # pure arcsine‑type candidate (lam=1)
            score = _score_normalised(w_norm, n, L)
            if score < best_score:
                best_score, best_a = score, w_norm.copy()
                best_alpha, best_eps, best_lam = alpha, eps, 1.0
            # mixtures with uniform
            for lam in lam_grid:
                a_mix = lam * w_norm + (1.0 - lam) * uniform
                # numerical noise may drift sum away from 1 – re‑normalise
                a_mix = a_mix / a_mix.sum()
                score = _score_normalised(a_mix, n, L)
                if score < best_score:
                    best_score, best_a = score, a_mix.copy()
                    best_alpha, best_eps, best_lam = alpha, eps, lam

        if time.time() >= deadline:
            break

    # ---- fine refinement around the best α ----------------------------
    if best_alpha is not None:
        step = 0.02
        while step > 1e-5 and time.time() < deadline:
            improved = False
            for alpha in (best_alpha - step, best_alpha, best_alpha + step):
                if not (0.30 <= alpha <= 0.80):
                    continue
                for eps in epsilons:
                    base = base0 + eps
                    w = base ** (-alpha)
                    w_norm = w / w.sum()
                    # pure
                    score = _score_normalised(w_norm, n, L)
                    if score + 1e-12 < best_score:
                        best_score, best_a = score, w_norm.copy()
                        best_alpha, best_eps, best_lam = alpha, eps, 1.0
                        improved = True
                    # mixtures
                    for lam in lam_grid:
                        a_mix = lam * w_norm + (1.0 - lam) * uniform
                        a_mix = a_mix / a_mix.sum()
                        score = _score_normalised(a_mix, n, L)
                        if score + 1e-12 < best_score:
                            best_score, best_a = score, a_mix.copy()
                            best_alpha, best_eps, best_lam = alpha, eps, lam
                            improved = True
            if not improved:
                step *= 0.5

    # return the best found candidate
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam optimiser (works on the simplex Σa=1)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, L: int,
                   deadline: float, start_time: float):
    a = a0.copy()                     # already normalised, Σa=1, a≥0
    best_a = a.copy()
    best_score = _score_normalised(a, n, L)

    m = np.zeros_like(a)              # first moment
    v = np.zeros_like(a)              # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step0 = 0.05
    it = 0

    while time.time() < deadline:
        it += 1
        # convolution and index of the maximal entry
        conv = _conv_self(a, L)
        k = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a (see the reference implementation)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # derivative of the max term

        # Adam update (step size slowly decays)
        step = step0 / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the probability simplex
        a = _project_simplex(a)

        # evaluate
        cur_score = _score_normalised(a, n, L)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing_refine(a0: np.ndarray, n: int, L: int,
                      deadline: float, start_time: float):
    a = a0.copy()
    cur_score = _score_normalised(a, n, L)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # rough estimate of how many cheap moves we can afford
    time_per_move = 0.0015                     # empirical
    remaining = max(0.0, deadline - time.time())
    max_moves = int(remaining / time_per_move)
    max_moves = min(max_moves, 80000)           # hard cap

    for it in range(1, max_moves + 1):
        if time.time() >= deadline:
            break
        # simple linear cooling schedule
        T = 0.01 * (1 - it / max_moves)

        src = rng.choice(n, p=a)                # pick source proportionally to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps_move = a[src] * rng.random() * 0.20   # move up to 20 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _score_normalised(a_prop, n, L)
        delta = new_score - cur_score

        if delta < 0.0 or (T > 0 and math.exp(-delta / T) > rng.random()):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score
