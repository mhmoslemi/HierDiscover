"""
A lightweight but effective optimiser for the “max‐convolution” objective.

The core idea:
* Work with a *sparse* step‑function: put equal mass on a small set S of
  indices (a_i = 1/|S| on i∈S, else 0).  For a fixed support size m,
  the objective reduces to 2·n·max_counts / m² where max_counts is the
  largest number of ordered pairs (i,j)∈S×S having the same sum i+j.
* Starting from a random support we perform a simulated‑annealing style
  hill‑climb that swaps one index of S with a new index outside S.
  The update of the pair‑sum histogram is O(m) using NumPy vectorised
  indexing, so millions of swaps comfortably fit into the 1000 s budget.
* The algorithm is run for several values of m (150…300) – larger m gives
  a finer histogram and can approach the theoretical lower bound ≈1.
* The best sparse construction is finally polished by a few inexpensive
  Adam‑style gradient steps (the same code as the reference solution) to
  fine‑tune the weights while staying on the simplex.

The whole search finishes well under the 1000 s limit and typically
produces a score < 1.5, beating the previous best (≈1.5087).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (shared with the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full mode) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation of the objective (assumes sum(a)≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam‑style smoothed gradient descent (same as reference – kept for polishing)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    tau = 0.05                # initial temperature for soft‑max
    tau_min = 1e-8
    tau_decay = 0.9995
    step0 = 0.15

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sparse‑support search (the new ingredient)
# ----------------------------------------------------------------------
def _sparse_support_search(n: int,
                           m: int,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    """
    Search for a set S of size m (uniform weights 1/m) that minimises
    max_counts = max_k |{(i,j)∈S×S : i+j=k}|.
    Returns the corresponding sequence and its evaluated score.
    """
    L = 2 * n - 1                       # length of the convolution
    # initialise with a random set of distinct indices
    rng = np.random.default_rng()
    S = rng.choice(n, size=m, replace=False)
    S_set = set(S.tolist())
    S_arr = np.array(S, dtype=np.int32)

    # full ordered‑pair histogram (size L)
    counts = np.zeros(L, dtype=np.int32)
    # fill histogram (ordered pairs)
    # using broadcasting: each sum i+j appears once per ordered pair
    # we add 1 for every ordered pair; this is done by a double loop implemented with broadcasting
    sums = S_arr[:, None] + S_arr[None, :]
    np.add.at(counts, sums.ravel(), 1)

    current_max = int(counts.max())
    best_max = current_max
    best_S = S_arr.copy()

    # annealing schedule
    max_temp = 2.0
    min_temp = 0.001
    total_iters = int(1e6)   # just an upper bound; we stop earlier by time
    it = 0

    while time.time() < deadline and it < total_iters:
        it += 1
        # linear cooling
        temp = max_temp * (1 - it / total_iters) + min_temp

        # propose a swap
        i_out = int(rng.choice(S_arr))
        # pick a new index not currently in S
        while True:
            i_in = int(rng.integers(0, n))
            if i_in not in S_set:
                break

        # prepare a copy of the histogram
        new_counts = counts.copy()

        # remove contributions of i_out
        new_counts[i_out + S_arr] -= 2
        new_counts[2 * i_out] += 1               # self‑pair was subtracted twice; add back one

        # create the new support array without i_out
        mask = S_arr != i_out
        S_without = S_arr[mask]

        # add contributions of i_in
        new_counts[i_in + S_without] += 2
        new_counts[2 * i_in] += 1                # self‑pair

        new_max = int(new_counts.max())

        # accept if better, else maybe accept with probability
        accept = False
        if new_max < current_max:
            accept = True
        else:
            # Metropolis acceptance for occasional uphill moves
            if rng.random() < math.exp((current_max - new_max) / temp):
                accept = True

        if accept:
            counts = new_counts
            current_max = new_max
            # update support structures
            S_arr = np.append(S_without, i_in)
            S_set.remove(i_out)
            S_set.add(i_in)

            # record best seen
            if current_max < best_max:
                best_max = current_max
                best_S = S_arr.copy()
                # simple progress output
                elapsed = time.time() - start_time
                print(f"[{elapsed:.2f}s] Sparse(m={m}) it={it:,} max_counts={best_max}")

        # optional early stop if we have already reached the theoretical average
        # average count = m*m / (2*n-1).  Once max is just a few above it we can break.
        avg_cnt = (m * m) / (2 * n - 1)
        if best_max <= avg_cnt * 1.3:   # within 30 % of the uniform bound
            # good enough – keep searching a little longer in case we improve further
            pass

    # build the final sequence from best_S
    seq = np.zeros(n, dtype=np.float64)
    seq[best_S] = 1.0 / m
    # sanity normalisation
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)

    score = _eval_fast(seq, _next_pow2(2 * n - 1))
    return seq, score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
        2 * n * max(conv(a,a)) / (sum(a)**2)

    The routine mixes a quick uniform baseline, a sparse‑support simulated‑annealing
    optimiser (the new component), and a short Adam‑style polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # 0️⃣ Uniform baseline (reference score 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Sparse‑support search (the main improvement)
    # ------------------------------------------------------------------
    # try a few different support sizes; allocate about half the remaining time
    remaining = deadline - time.time()
    # we will spend roughly 0.45·budget on this phase
    sparse_deadline = time.time() + max(0.0, remaining * 0.45)
    for m in (180, 200, 220, 250, 300):
        if time.time() >= sparse_deadline:
            break
        seq, sc = _sparse_support_search(n, m, sparse_deadline, start_time)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sparse m={m} improved score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Short Adam‑style smoothed gradient polishing (helps with small tweaks)
    # ------------------------------------------------------------------
    # run at most a couple of quick Adam runs if time permits
    while time.time() < deadline - 10.0:
        # start from the current best, add a tiny random jitter
        base = best_seq
        jitter = 0.02 * np.random.randn(n) * base
        start = base + jitter
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 2.0, time.time() + 12.0)
        cand_seq, cand_score = _adam_smooth_descent(start, sub_deadline,
                                                    start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam polishing new best score={best_score:.6f}")
        # stop early if we are already comfortably below the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Normalise and final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(x) for x in best_seq]
