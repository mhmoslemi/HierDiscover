def _adam_optimize(a0, n, L_fft, deadline, start_time, best_score_initial):

    a = a0.copy() # sum normalized to 1

    best_a = a.copy()

    best_score = best_score_initial

    m = np.zeros_like(a)

    v = np.zeros_like(a)

    beta1 = 0.9

    beta2 = 0.999

    epsilon = 1e-8

    lr = 0.05 # base learning rate

    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() **2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # maybe print

        # compute subgradient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n -1, k)
        idx = np.arange(lo, hi+1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]  # as earlier

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        a = _project_simplex(a)

    return best_a, best_score

def _local_refine(seq, obj_fn, param_type, param, step, L_fft, time_limit):
    # param_type: 'geom' expects r, 'pow' expects p
    # step: initial step scale

    The algorithm:
    1. Generate a diverse collection of parametric sequences (geometric,
       power, beta, random Dirichlet) and keep the best one.
    2. Refine the best geometric / power parameters with a simple
       stochastic local search.
    3. Run a projected Adam optimizer on the simplex for the bulk of the time.
    4. Perform a simulated‑annealing hill‑climb with moderate mass moves.
    5. Finish with a very fine‑grained hill‑climb (tiny moves) to polish.

"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of five stages:

1️⃣  Build a large pool of parametric seeds (geometric, power, beta, Dirichlet,
    plus the optional global starter).  Keep the best one.

2️⃣  Refine the best geometric / power seed with a tiny stochastic local
    search that adjusts the single shape parameter.

3️⃣  Run a *projected Adam* optimiser on the simplex for the bulk of the
    time.  The Adam update follows the sub‑gradient of the active (maximum)
    convolution entry and each step is projected back onto the probability
    simplex.

4️⃣  Perform a simulated‑annealing hill‑climb that moves a moderate amount
    of mass between two random positions; moves that improve the objective
    are always accepted and worsening moves are accepted with a decreasing
    temperature.

5️⃣  Finish with a very fine‑grained “tiny” hill‑climb (ε≈10⁻⁴·a_src) to
    polish the solution.

All heavy work uses FFT based self‑convolutions (O(n log n)).  The routine
respects the 1000 s budget and returns a plain list of Python floats ready
for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective function (fast, assumes sum≈1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    The vector a is expected to be non‑negative.
    """
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Small stochastic refinement of a single shape parameter
# ----------------------------------------------------------------------
def _local_refine(seq: np.ndarray,
                  obj_fn,
                  param_type: str,
                  param: float,
                  step: float,
                  L_fft: int,
                  time_limit: float):
    """
    Random walk on a single continuous parameter (either ratio r for a geometric
    sequence or exponent p for a power law).  Accepts a move if it improves the
    objective.  The step size is slowly adapted.  Returns the best sequence,
    its score and the final parameter value.
    """
    best_seq = seq.copy()
    best_score = obj_fn(best_seq, L_fft)
    start = time.time()
    while time.time() < time_limit:
        if param_type == "geom":
            # multiplicative perturbation of r ∈ (0,1)
            factor = math.exp(np.random.normal(scale=step))
            r_new = param * factor
            if not (0.0 < r_new < 1.0):
                continue
            cand = np.power(r_new, np.arange(seq.size, dtype=float))
        else:  # power law
            p_new = param + np.random.normal(scale=step)
            cand = np.power(np.arange(1, seq.size + 1, dtype=float), p_new)

        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        sc = obj_fn(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            param = r_new if param_type == "geom" else p_new
            step *= 0.9               # tighten step after improvement
            print(f"[{time.time() - start:.2f}s] Refine ({param_type}) score={sc:.6f}")
        else:
            step *= 1.02              # gently enlarge step if no improvement
    return best_seq, best_score, param


# ----------------------------------------------------------------------
# Projected Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_score_initial: float):
    """
    Adam optimisation of the sub‑gradient of max(conv(a,a)).
    The iterate is projected onto the simplex after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05
    t = 0
    last_improve = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = t
            print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} score={cur_score:.6f}")

        # sub‑gradient of the active (max) entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        # occasional tiny perturbation if we stall
        if t - last_improve > 5000:
            noise = 1e-4 * np.random.default_rng().standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            last_improve = t

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (moderate mass moves)
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray,
                n: int,
                L_fft: int,
                deadline: float,
                start_time: float):
    """
    Moves a fraction (up to 5 %) of mass from a source index to a destination.
    Accepts improvements always and accepts deteriorations with a temperature
    that slowly cools to zero.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = _objective(a, L_fft)
    rng = np.random.default_rng()
    max_factor = 0.05               # up to 5 % of the source mass per move

    while time.time() < deadline:
        remaining = deadline - time.time()
        total = deadline - start_time
        temperature = max(1e-5, remaining / total * 0.04)   # linear cooling

        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * max_factor
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop, L_fft)
        delta = new_score - best_score
        if delta < 0.0 or rng.random() < math.exp(-delta / temperature):
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(start_a: np.ndarray,
                     n: int,
                     L_fft: int,
                     deadline: float,
                     start_time: float):
    """
    Very small mass transfers (ε≈10⁻⁴·a_src) that are accepted only if they
    improve the true objective.  This fine‑tunes the solution after the
    previous phases.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _objective(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-4   # extremely tiny move
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={best_score:.6f}")

    return best_a, best_score
