def _smooth_max_gradient_descent(a0, deadline, start_time, init_beta=0.5, beta_inc_factor=1.05, max_beta=50, step0=0.5, momentum_beta=0.9):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    beta = init_beta
    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and smooth grad
        conv = _conv_self(a)  # length 2n-1
        # compute smoothed max estimator:
        # use np.exp( beta * (conv - max_conv) ) to avoid overflow
        max_conv = conv.max()
        # shift for stability
        exp_term = np.exp(beta * (conv - max_conv))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp
        # compute gradient: 2 * convolution(w, a_rev) at appropriate slice
        a_rev = a[::-1]  # reversed
        # determine FFT size:
        L = _next_pow2(len(w) + n - 1)
        # compute convolution via FFT:
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_w_a_rev = np.fft.irfft(W * A_rev, n=L)[:len(w) + n - 1]
        # gradient slice:
        grad = 2.0 * conv_w_a_rev[(n-1):(n-1 + n)]  # length n
        # momentum update
        momentum = momentum_beta * momentum + (1 - momentum_beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        # projection onto simplex
        a = _project_simplex(a)
        # evaluate objective:
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth iter={it} beta={beta:.2f} score={cur_score:.6f}")
        # increase beta gradually:
        if beta < max_beta:
            beta *= beta_inc_factor
            # ensure not exceeding deadline:
            if beta > max_beta:
                beta = max_beta
    return best_a, best_score
```

"""
Propose a step function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where conv(a,a) is the linear self‑convolution of a.
The algorithm works in three stages:

1. **Warm‑start** – several simple candidates (triangular “tent”, smooth
   sine shape and – if available – a previously found best sequence) are
   normalised to the probability simplex and the best of them is chosen.

2. **Smoothed max gradient descent** – we minimise a smooth approximation
   of the max‑convolution using the log‑sum‑exp (soft‑max).  The gradient
   of the smoothed objective can be computed efficiently with one FFT
   convolution per iteration.  A Euclidean projection onto the simplex
   keeps the iterate feasible.  The smoothing parameter β is increased
   slowly (annealing) so that the optimisation gradually focuses on the
   true maximum.

3. **Light simulated‑annealing polish** – finally a cheap Metropolis
   move set (transfer a random packet of mass between two indices) is
   applied for the remaining time budget.

The whole search respects the 1000 s time budget and always returns a
valid non‑negative list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed max gradient descent (soft‑max)
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    init_beta: float = 0.5,
                    max_beta: float = 30.0,
                    beta_inc_factor: float = 1.001,
                    step0: float = 1.0,
                    momentum_coeff: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of the
    max‑convolution (log‑sum‑exp).  Returns the best iterate and its
    exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)

    it = 0
    beta = init_beta

    while time.time() < deadline:
        it += 1

        # ----- convolution of the current iterate -----
        conv = _conv_self(a)                     # length 2n-1

        # ----- soft‑max weights -----
        max_conv = conv.max()
        # stability shift: subtract max before exponentiation
        exp_term = np.exp(beta * (conv - max_conv))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp                    # weights, sum to 1

        # ----- gradient of the smoothed objective -----
        a_rev = a[::-1]                           # reversed a
        L = _next_pow2(len(w) + n - 1)            # length for FFT
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        conv_w_a_rev = np.fft.irfft(W * A_rev, n=L)[: len(w) + n - 1]

        # gradient_i = 2 * Σ_k w_k a_{k-i}
        grad = 2.0 * conv_w_a_rev[(n - 1):(n - 1 + n)]

        # ----- momentum & step size -----
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad
        step = step0 / math.sqrt(it)

        a -= step * momentum

        # ----- projection onto the simplex -----
        a = _project_simplex(a)

        # ----- keep the best seen -----
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth iter={it:,} "
                  f"beta={beta:.2f} score={cur_score:.6f}")

        # ----- increase smoothing parameter -----
        if beta < max_beta:
            beta = min(max_beta, beta * beta_inc_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------


def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis annealing that moves a small random packet of mass.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 40_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        # pick source proportionally to its mass, destination uniformly
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25   # move up to 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance criterion
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,

    """

    start_time = time.time()
    # Slight safety margin to avoid a hard timeout

    # --------------------------------------------------------------
    # Problem size – a power of two yields fast FFTs.

    # --------------------------------------------------------------
    n = 16384

    # --------------------------------------------------------------

    candidates = []

    # 1) Symmetric linear “tent”.
    x = np.arange(n, dtype=float) / (n - 1)
    tent = np.minimum(x, 1.0 - x)          # triangular shape, zero at the ends
    if tent.sum() == 0:

    else:

    candidates.append(a_tent)

    i = np.arange(n, dtype=float) + 0.5
    sine = np.sin(np.pi * i / n)          # values in (0,1]
    a_sine = sine / sine.sum()
    candidates.append(a_sine)

    # 3) Global best sequence (if present) – resize if needed.
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # make it non‑negative and normalise
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                if starter.shape[0] != n:
                    # simple linear interpolation to required length
                    old_n = starter.shape[0]
                    xs_old = np.linspace(0, 1, old_n)
                    xs_new = np.linspace(0, 1, n)
                    starter = np.interp(xs_new, xs_old, starter)
                    starter = np.maximum(starter, 0.0)
                    starter /= starter.sum()
                candidates.append(starter)
    except Exception:
        pass  # ignore if the global variable is missing

    # --------------------------------------------------------------
    # Pick the best simple candidate as a starting point.
    # --------------------------------------------------------------
    best_start = min(candidates, key=_objective)
    print(f"[{time.time() - start_time:.2f}s] initial best score={_objective(best_start):.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed max gradient descent.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:        # keep a few seconds for polishing
        sub_deadline = time.time() + remaining * 0.85
        refined_seq, refined_score = _smooth_descent(best_start,
                                                     sub_deadline,
                                                     start_time,
                                                     init_beta=0.5,
                                                     max_beta=30.0,
                                                     beta_inc_factor=1.001,
                                                     step0=1.0,
                                                     momentum_coeff=0.9)
        if refined_score < _objective(best_start):
            best_start = refined_seq
            print(f"[{time.time() - start_time:.2f}s] refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – light annealing polish (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_start, deadline, start_time)
        if anneal_score < _objective(best_start):
            best_start = anneal_seq
            print(f"[{time.time() - start_time:.2f}s] final anneal improve score={anneal_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation & conversion to plain list.
    # --------------------------------------------------------------
    best_start = np.maximum(best_start, 0.0)
    total = best_start.sum()
    if total > 0:
        best_start /= total
    else:
        best_start = np.full(n, 1.0 / n)

    return [float(v) for v in best_start]
