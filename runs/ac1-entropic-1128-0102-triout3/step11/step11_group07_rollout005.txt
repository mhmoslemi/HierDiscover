def _mirrored_gradient_descent(a0, deadline, start_time, L_fft):
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    eta = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        k = int(conv.argmax())
        # compute gradient for this max index
        lo = max(0, k - (len(a)-1))
        hi = min(k, len(a)-1)
        idx = np.arange(lo, hi+1)
        # grad_i = 2*a[k-i]

        # multiplicative update
        a = a * np.exp(-eta * grad)
        # ensure positivity
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score

def _next_pow2(x: int) -> int:

def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.

    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation assuming sum(a)≈1: returns 2·n·max(conv)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (including new negative‑exponent beta & sine‑power)
# ----------------------------------------------------------------------
def _explore_parametric(n: int, L_fft: int,
                        best_seq: np.ndarray,
                        best_score: float,
                        start_time: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # ---- Geometric ----------------------------------------------------
    eps_grid = np.logspace(-4, -1, 80)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        sc = _eval_fast(geo, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} score={sc:.6f}")

    # ---- Power (including negative) ------------------------------------
    p_grid = np.concatenate(
        (np.linspace(-2.0, -0.2, 20), np.linspace(0.0, 2.0, 21))
    )
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1.0, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # ---- Negative‑beta family (edge‑heavy) ----------------------------
    # a_i ∝ (i+1)^p * (n-i)^q with p,q ∈ [-2,-0.1]
    beta_vals = np.linspace(-2.0, -0.2, 10)
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = np.power(idx + 1.0, p, dtype=np.float64) * \
                       np.power(n - idx, q, dtype=np.float64)
            s = beta_seq.sum()
            if s <= 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.2f} q={q:.2f} score={sc:.6f}")

    # ---- Sine‑power family (sin^(p) with p negative → edge spikes) --
    sin_vals = np.sin(np.pi * (idx + 0.5) / n)  # >0 for all idx
    sin_eps = 1e-12
    for p in np.linspace(-2.0, -0.2, 15):
        if time.time() >= deadline - 30:
            break
        shape = np.where(sin_vals > sin_eps,
                         np.power(sin_vals, p, dtype=np.float64),
                         0.0)
        s = shape.sum()
        if s == 0:
            continue
        shape /= s
        sc = _eval_fast(shape, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, shape.copy()
            print(f"[{time.time() - start_time:.2f}s] Sin^p p={p:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Mirror‑descent (multiplicative updates) on the active max index
# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int) -> tuple[np.ndarray, float]:
    """
    Projected multiplicative (mirror) descent using the exact sub‑gradient
    of the currently maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    n = a.shape[0]
    it = 0
    eta = 0.5                # initial step size (learning rate)
    decay_every = 500        # decay interval
    perturb_every = 4000     # occasional random jitter

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        max_val = conv.max()
        k = int(conv.argmax())            # one of the maximisers

        # ----- sub‑gradient of conv[k] w.r.t. a -----------------------
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        # grad_i = 2 * a_{k-i}
        grad[idx] = 2.0 * a[k - idx]

        # ----- multiplicative (mirror) update -------------------------
        factor = np.exp(-eta * grad)
        a = a * factor
        # keep tiny positive floor to avoid exact zeros
        a = np.maximum(a, 1e-15)
        a /= a.sum()                       # re‑normalise (simplex projection)

        # ----- keep the best -----------------------------------------
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Mirror it={it:,} score={cur_score:.6f}")

        # ----- schedule adjustments -----------------------------------
        if it % decay_every == 0:
            eta *= 0.9                     # slowly shrink step size

        if it % perturb_every == 0:
            # add a small random jitter (zero‑mean, proportional to current mass)
            rng = np.random.default_rng()
            jitter = 0.02 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            if a.sum() == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Tiny hill‑climb (local random tweaks, accept‑only‑improve)
# ----------------------------------------------------------------------
def _tiny_hill_climb(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # probability‑proportional source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.08   # tiny transfer
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop /= a_prop.sum()
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] Hill it={it:,} best={best_score:.6f}")
    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising the evaluation function.
    The strategy:
        • a broad parametric sweep (geometric, power, negative‑beta, sine‑power);
        • a long mirror‑descent on the active max‑convolution index;
        • a final tiny hill‑climb for polishing.
    Returns a list of non‑negative floats compatible with ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Uniform baseline (always a safe fallback)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if the environment provides one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score, best_seq = cur, starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (including new families)
    # ------------------------------------------------------------------
    best_seq, best_score = _explore_parametric(
        n, L_fft, best_seq, best_score, start_time, deadline
    )
    print(f"[{time.time() - start_time:.2f}s] After parametric sweep best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Mirror‑descent (multiplicative) – main workhorse
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        best_seq, best_score = _mirror_descent(
            best_seq, deadline, start_time, L_fft
        )
        print(f"[{time.time() - start_time:.2f}s] After mirror descent best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Tiny hill‑climb for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _tiny_hill_climb(
            best_seq, deadline, start_time, L_fft
        )
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalisation & final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Return a plain Python list as required by the external evaluator
    return [float(x) for x in best_seq]
