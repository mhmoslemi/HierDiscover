"""
Propose a step‑function that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

The idea is to treat the *soft‑max* of the convolution as a smooth
approximation of the L∞‑norm and run a long projected (Euclidean) gradient
descent on the probability simplex.  The gradient of the softened objective
has a simple form: it is proportional to the correlation of the current
vector with the soft‑max weights.  Both the self‑convolution and the
correlation are computed with FFTs (O(n log n)), so even for n≈8000 the loop
remains fast enough to consume most of the 1000 s budget.

The algorithm proceeds in three phases

1️⃣  **Quick parametric sweep** – a few geometric mixtures of the uniform
    distribution give a decent starting point.

2️⃣  **Smoothed gradient descent** – β (soft‑max temperature) is slowly
    increased from a modest value to a large one, turning the smooth
    objective into the true max‑convolution.  Momentum (β≈0.9) and a
    1/√t step‑size are used; after long periods without improvement a tiny
    random perturbation is added to escape plateaus.

3️⃣  **Microscopic hill‑climb** – finally we move an epsilon amount of mass
    between random indices and keep only improving moves.  This fine‑tunes
    the solution.

All heavy work uses FFTs and a fast Euclidean projection onto the simplex.
The routine respects the 1000 s time limit and returns a plain list of
Python floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the supplied
    evaluation function.  Returns a list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3           # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation and often
    # a smaller L∞‑ratio.  8192 is a good compromise between quality and
    # speed on a single CPU.
    # --------------------------------------------------------------
    n = 8192
    # one FFT length works for all convolutions/correlations we need
    L_fft = _next_pow2(3 * n)                       # ≥ 3 n for correlation

    # --------------------------------------------------------------
    # 1️⃣  Quick parametric sweep (geometric mixes) for a warm start
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _eval_fast(best_a, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # geometric families – a few points are enough for a quick boost
    eps_grid = np.logspace(-4, -1, 30)               # fewer points than before
    for eps in eps_grid:
        if time.time() >= deadline - 60:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        for lam in (0.0, 0.3, 0.6, 1.0):
            cand = lam * uniform + (1.0 - lam) * geo
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent on the simplex
    # --------------------------------------------------------------
    a = best_a.copy()
    momentum = np.zeros_like(a)
    beta_mom = 0.9                     # momentum coefficient
    step0 = 0.05                       # initial step magnitude
    it = 0
    last_improve = 0
    beta_soft = n * 2.0                # start with a modest temperature
    beta_soft_max = n * 30.0           # final (sharper) temperature
    beta_inc = 1.001                   # geometric increase per iteration

    while time.time() < deadline - 5.0:
        it += 1
        # ----- self‑convolution and current score (exact) -----
        Af = np.fft.rfft(a, n=L_fft)
        conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()                     # sum(a)=1 by construction
        if cur_score < best_score:
            best_score, best_a, last_improve = cur_score, a.copy(), it
            print(f"[{time.time() - start_time:.2f}s] it={it:,} score={cur_score:.6f}")

        # ----- soft‑max weights (smooth approximation of max) -----
        max_c = conv.max()
        # subtract max to avoid overflow; no risk of overflow because values are tiny
        w = np.exp(beta_soft * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        # correlation = (a_rev * w)  where * is linear convolution
        a_rev = a[::-1]
        Af_rev = np.fft.rfft(a_rev, n=L_fft)
        wf = np.fft.rfft(w, n=L_fft)
        corr_full = np.fft.irfft(Af_rev * wf, n=L_fft)      # length L_fft
        # relevant part: indices n‑1 … n‑1+n‑1
        start = n - 1
        corr = corr_full[start:start + n]                   # shape (n,)
        grad = 4.0 * n * corr                               # ∇ (2 n·softmax)

        # ----- momentum & projected update -----
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)                             # keep on simplex

        # ----- slowly sharpen the soft‑max -----
        beta_soft = min(beta_soft_max, beta_soft * beta_inc)

        # ----- occasional random jiggle if we stall -----
        if it - last_improve > 5000:
            rng = np.random.default_rng()
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            last_improve = it

    # --------------------------------------------------------------
    # 3️⃣  Final microscopic hill‑climb
    # --------------------------------------------------------------
    a = best_a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4   # extremely tiny move
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        sc = _eval_fast(a_prop, L_fft)
        if sc < best_score:
            a = a_prop
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={sc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (projected simplex already ensures sum≈1) and
    # return a plain Python list.
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
