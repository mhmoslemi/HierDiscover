def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear convolution of two real vectors u and v using FFT.
    Returns the full linear convolution (len(u)+len(v)-1 elements).
    """
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) is assumed to be 1)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray, L_self: int) -> float:
    """Return 2·n·max(a*a) for a having sum ≈ 1."""
    conv = _conv_self(a, L_self)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Phase‑1 : Adam on the smooth‑max (log‑sum‑exp) approximation
# ----------------------------------------------------------------------
def _adam_smooth_max(start_a: np.ndarray,
                     alphas: np.ndarray,
                     n: int,
                     L_self: int,
                     L_grad: int,
                     deadline: float,
                     start_time: float,
                     max_iters_per_alpha: int = 2500) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max objective.
    Parameters
    ----------
    start_a : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    alphas : np.ndarray
        Smoothing parameters, increasing geometrically.
    n, L_self, L_grad : int
        Problem size and FFT lengths.
    deadline, start_time : float
        Wall‑clock control.
    max_iters_per_alpha : int
        Upper bound on inner iterations for each α.
    Returns
    -------
    best_a, best_score
    """
    # initialise logits (log‑softmax parameters)
    v = np.log(start_a + 1e-30)          # avoid log(0)
    m = np.zeros_like(v)                # first moment
    v_hat = np.zeros_like(v)            # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = start_a.copy()
    best_score = _exact_score(best_a, L_self)

    it_total = 0
    for alpha in alphas:
        for it in range(max_iters_per_alpha):
            if time.time() >= deadline - 0.2:
                return best_a, best_score
            it_total += 1

            # ---- a = softmax(v) (keeps positivity & sum = 1) ----
            v_shift = v - np.max(v)               # for numerical stability
            a = np.exp(v_shift)
            a_sum = a.sum()
            a = a / a_sum

            # ---- self‑convolution (a*a) ----
            conv = _conv_self(a, L_self)

            # ---- soft‑max weights w_k ∝ exp(alpha·conv_k) ----
            scaled = alpha * conv
            max_scaled = scaled.max()
            exp_shifted = np.exp(scaled - max_scaled)
            w = exp_shifted / exp_shifted.sum()

            # ---- gradient wrt a : 2·(w ⋆ a_rev) ----
            a_rev = a[::-1]
            # convolution w * a_rev, full length = 3·n‑2
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]

            # ---- gradient wrt logits v (softmax Jacobian) ----
            grad_v = a * (grad_a - np.dot(grad_a, a))

            # ---- Adam update (bias‑corrected) ----
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** it_total)
            v_hat_corr = v_hat / (1.0 - beta2 ** it_total)

            # step size decays like 1/√t, multiplied by a small constant
            step = 0.2 / (math.sqrt(it_total) + 1e-8)
            v -= step * m_hat / (np.sqrt(v_hat_corr) + eps)

            # ---- occasional exact evaluation ----
            if it % 400 == 0:
                cur_score = _exact_score(a, L_self)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.1f} "
                          f"it={it_total:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (exact objective)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float,
                         max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact objective.
    Momentum with β=0.9 and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    stagnation = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        if stagnation >= 15_000:
            # early stop if no progress for a while
            break

        # indices attaining the current maximum (within a tiny tolerance)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + Euclidean projection onto the simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = 0.4 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : greedy hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_seq: np.ndarray,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: transfer a tiny fraction of mass from a source
    (chosen proportionally to its current height) to a uniformly random
    destination.  Keep the move only if it improves the exact objective.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.choice(n, p=a)               # source weighted by current mass
        dst = rng.integers(0, n)               # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 30 % of the source mass (random fraction)
        eps = a[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` for n = 4096 within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)                  # FFT length for a*a
    L_grad = _next_pow2(3 * n - 2)                  # length needed for grad w⋆a_rev

    # ------------------------------------------------------------------
    # Load an initial vector (global best if present, otherwise uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _exact_score(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter exact score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑1 : Adam on smooth‑max, several random restarts
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:               # reserve time for later phases
        alphas = np.geomspace(0.5, 2e4, num=35)    # smoothing schedule
        # number of restarts (including the starter)
        restarts = 5
        for r in range(restarts):
            if time.time() >= deadline - 30.0:
                break

            if r == 0:
                a0 = best_seq.copy()               # first restart uses current best
            else:
                # Dirichlet with low concentration => spikier starts
                conc = np.random.uniform(0.05, 0.5)
                a0 = np.random.gamma(conc, 1.0, size=n)
                a0 = a0 / a0.sum()

            a_opt, sc_opt = _adam_smooth_max(
                start_a=a0,
                alphas=alphas,
                n=n,
                L_self=L_self,
                L_grad=L_grad,
                deadline=deadline - 20.0,            # keep a margin
                start_time=start_time,
                max_iters_per_alpha=2400
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] Restart {r} improved score = {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : projected sub‑gradient descent (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(
            a0=best_seq,
            n=n,
            L_self=L_self,
            deadline=sub_deadline,
            start_time=start_time,
            init_best_score=best_score,
            max_iters=600_000
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 : greedy hill‑climbing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _hill_climb(
            a0=best_seq,
            n=n,
            L_self=L_self,
            deadline=deadline,
            start_time=start_time,
            cur_best_seq=best_seq,
            cur_best_score=best_score
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improvement = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best exact score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
