        grad = 2.0 * grad_full[n-1 : n-1 + n]

        m = beta1 * m + (1 - beta1) * grad

        m_hat = m / (1 - beta1 ** t)

        # project onto scaled simplex
        a = _project_scaled_simplex(a_new, np.sqrt(2.0 * n))

    return best_a, best_score

def _adam_smoothing_descent(a0, deadline, start_time, gamma_start, gamma_end, step0):
    a = a0.copy()
    n = a.shape[0]
    S = np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_score = _objective(a)
    while time.time() < deadline:
        t += 1
        # evaluate
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} score={cur_score:.6f}")

        # compute gamma schedule: linear from start to end per iteration
        gamma = gamma_start + (gamma_end - gamma_start) * min(1.0, t / 5000.0)  # ramp up
        # compute softmax weights
        shifted = conv - max_conv
        # to avoid overflow, shift then exponentiate scaled by gamma
        exp_vals = np.exp(gamma * shifted)  # max <= 0

"""
A fast continuous‑approximation optimizer for the step‑function bound.

The evaluation function to minimise is

    2·n·max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant.  We fix the ℓ₁‑norm to
    Σ a_i = √(2·n)
so that the objective becomes simply   max(conv(a,a)).
The algorithm works as follows

1.   Choose a moderately large canvas (n = 32768, a power of two).
2.   Generate a few analytic “tent‑with‑plateau’’ shapes and pick the best
     as a warm‑start.
3.   Perform a projected Adam optimisation on a **smooth** surrogate of the
     max‑convolution.  The surrogate is the log‑sum‑exp over the
     convolution entries; its gradient can be obtained by a single
     cross‑correlation (FFT) and the iteration is projected onto the
     simplex Σ a_i = √(2·n).
4.   After the allotted time a short simulated‑annealing polish moves tiny
     packets of mass between random positions.
5.   The best sequence seen is returned as a plain list.

All helper functions are top‑level so that the harness can pickle them.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (max‑convolution) – the value we finally want to minimise.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (identical to evaluate_sequence)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _project_scaled_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project onto {w ≥ 0, Σ w = S} using the unit‑simplex projector."""
    if S <= 0:
        raise ValueError("Scaling constant S must be positive.")
    # scale to unit simplex, project, then rescale back
    u = v / S
    u_proj = _project_simplex(u)
    return u_proj * S


# ----------------------------------------------------------------------
# Analytic shape generator (tent with optional plateau)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length n.
    p      – exponent of the side ramps (p>0).
    alpha  – relative plateau width ∈ [0,0.5]; 0 → pure tent,
             0.5 → uniform.
    Returned vector is normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# Smoothed Adam optimiser (log‑sum‑exp surrogate of max)
# ----------------------------------------------------------------------
def _adam_smooth_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         gamma_start: float = 0.5,
                         gamma_end: float = 20.0,
                         step0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth surrogate L( a ) = (1/γ)·log Σ exp(γ·conv(a,a)).
    The iteration is projected onto the scaled simplex Σ a = √(2·n).
    Returns the best iterate (and its true objective value).
    """
    a = a0.copy()
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_a = a.copy()
    best_score = _objective(a)

    # pre‑allocate FFT length for the gradient step (will stay constant)
    L_grad = _next_pow2((2 * n - 1) + n - 1)   # = 3n‑2 rounded up

    while time.time() < deadline:
        t += 1

        # current objective (true max‑convolution)
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} "
                  f"score={cur_score:.6f}")

        # ---- smooth surrogate ------------------------------------------------
        # raise the temperature (γ) slowly, allowing the optimiser to see a
        # smoother landscape early on.
        gamma = gamma_start + (gamma_end - gamma_start) * min(1.0, t / 5000.0)

        # softmax over convolution entries (max at zero, everything ≤0 after shift)
        shifted = conv - max_conv
        w = np.exp(gamma * shifted)
        w /= w.sum()                     # shape (2n‑1,)

        # ---- gradient of the surrogate ---------------------------------------
        # grad = 2 * cross‑correlation( w , a )
        Fw = np.fft.rfft(w, n=L_grad)
        Fa_rev = np.fft.rfft(a[::-1], n=L_grad)
        grad_full = np.fft.irfft(Fw * Fa_rev, n=L_grad)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]   # length n

        # ---- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / math.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto Σ a = S -----------------------------------------
        a = _project_scaled_simplex(a_new, S)

    return best_a, best_score
