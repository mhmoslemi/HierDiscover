import time

# ----------------------------------------------------------------------
# Utilities for FFT‑based convolutions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    """
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto { w ≥ 0 , Σ w = 1 }."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑tent with plateau (the classic family)."""
    alpha = max(0.0, min(0.5, alpha))

    if plateau_len % 2 == 1:

    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # rising from 0 to 1
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_cosine_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric cosine‑ramp with plateau.
    The side‑ramp follows (cos(π·t/2))**p where t∈[0,1] runs from the edge to
    the start of the plateau.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1:
        plateau_len -= 1
    plateau_len = max(0, plateau_len)
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        # t runs from 1 (edge) to 0 (plateau start)
        t = np.arange(side_len, 0, -1, dtype=float) / side_len
        ramp = (np.cos(np.pi * t / 2) ** p)
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_parabola_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric parabola‑ramp with plateau.
    The side‑ramp follows (1 - t**2)**p where t∈[0,1] runs from the edge to the
    plateau.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1:
        plateau_len -= 1
    plateau_len = max(0, plateau_len)
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        t = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = (1.0 - t ** 2) ** p
        ramp = np.clip(ramp, 0.0, None)
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (non‑smooth objective)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent until *deadline*.
    Returns best iterate seen and its (exact) score.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9            # momentum coefficient
    step0 = 0.5           # initial step size
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,}"
                  f"  score={cur_score:.6f}")

        # index where convolution attains its maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project onto the simplex (keeps a ≥ 0, Σ a = 1)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions; Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of annealing attempts – chosen to fit the remaining time.
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)                # more likely to move mass from larger slots
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25      # move ≤25 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}"
                      f"  best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence()
    score.  Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – larger canvases give a finer discretisation of the
    # continuous optimum (≈ 1.5).  2⁶ = 65536 is still cheap enough for many FFTs.
    # ------------------------------------------------------------------
    n = 65536  # power of two → fast FFTs

    # ------------------------------------------------------------------
    # 1) Sweep several analytical families.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    top_k = 5
    top_candidates = []   # list of (score, sequence)

    # parameters to explore
    p_grid = np.linspace(0.2, 3.0, 30)          # 30 values
    alpha_grid = np.linspace(0.0, 0.5, 11)      # 0 % … 50 %

    shape_funcs = [
        _make_power_shape,
        _make_cosine_shape,
        _make_parabola_shape,
    ]

    for make_shape in shape_funcs:
        for p in p_grid:
            for alpha in alpha_grid:
                a = make_shape(n, p, alpha)
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                # keep a small pool of top candidates
                if len(top_candidates) < top_k:
                    top_candidates.append((score, a.copy()))
                    top_candidates.sort(key=lambda x: x[0])
                else:
                    if score < top_candidates[-1][0]:
                        top_candidates[-1] = (score, a.copy())
                        top_candidates.sort(key=lambda x: x[0])

                # occasional progress report (every ~30 s)
                if (time.time() - start_time) % 30 < 0.2:
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"grid p={p:.2f} α={alpha:.2f} shape={make_shape.__name__}"
                          f" → score={score:.6f}")

    # a few random draws to diversify the pool
    rng = np.random.default_rng()
    for _ in range(300):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.1, 3.5)
        alpha = rng.uniform(0.0, 0.5)
        make_shape = rng.choice(shape_funcs)
        a = make_shape(n, p, alpha)
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        if len(top_candidates) < top_k:
            top_candidates.append((score, a.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if score < top_candidates[-1][0]:
                top_candidates[-1] = (score, a.copy())
                top_candidates.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] Best grid/random score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Incorporate the globally stored starter (if it exists).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    if len(top_candidates) < top_k:
                        top_candidates.append((cur, starter.copy()))
                        top_candidates.sort(key=lambda x: x[0])
                    else:
                        if cur < top_candidates[-1][0]:
                            top_candidates[-1] = (cur, starter.copy())
                            top_candidates.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Local refinement – projected sub‑gradient descent on the best few.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # allocate 80 % of the remaining budget to the descent phase
        descent_budget = time.time() + 0.80 * remaining
        per_candidate = (descent_budget - time.time()) / len(top_candidates)

        for idx, (score, cand) in enumerate(top_candidates):
            if time.time() >= descent_budget:
                break
            sub_deadline = time.time() + per_candidate
            refined_seq, refined_score = _subgradient_descent(cand,
                                                              sub_deadline,
                                                              start_time)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Refined candidate {idx}"
                      f" → score = {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light annealing polish (if any time remains).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve"
                  f" → score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
