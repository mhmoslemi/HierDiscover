"""
Optimiser for the step‑function upper‑bound.

The algorithm consists of three phases

1. Warm start with a triangular distribution (the classic “tent’’).
2. A continuation based projected “soft‑max’’ gradient descent.
   In every iteration the self‑convolution is computed,
   a soft‑max over the convolution entries (temperature τ) yields
   weights w_t.  The gradient of the smoothed objective is
        g = 2 · (w * a_rev)[n‑1 … 2n‑2] .
   An Adam‑like update is performed and the iterate is projected
   onto the probability simplex.
   τ is decreased exponentially from τ₀≈5·10⁻³ down to τ_end≈2·10⁻⁶,
   so the optimisation starts with a smooth landscape and finishes
   with an almost exact max‑operator.
3. A cheap simulated‑annealing polishing phase that moves tiny
   packets of mass between coordinates.

All heavy work is performed with FFTs, therefore a size of
n=16384 (a power of two) fits comfortably into the 1000 s budget.
The routine keeps the best vector seen (according to the exact
objective) and returns it as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    # linear convolution via FFT
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv

def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max over the convolution entries with temperature τ."""
    # shift for numerical stability
    scaled = conv / tau
    m = scaled.max()
    exps = np.exp(scaled - m)
    return exps / exps.sum()

def _gradient_from_weights(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the τ‑smoothed objective.
    g_i = 2·∑_t w_t·a_{t‑i}, i = 0…n‑1.
    Computed by a single FFT‑based convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w, n=L)
    conv_full = np.fft.irfft(Af * Wf, n=L)
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad

# ----------------------------------------------------------------------
# Core optimiser – soft‑max gradient descent with continuation
# ----------------------------------------------------------------------
def _softmax_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    tau0: float = 5e-3,
    tau_end: float = 2e-6,
    step0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Perform projected Adam‑style descent on the smoothed objective.
    Returns the best vector encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT length sufficient for both convolutions (a*a and w*a_rev)
    L = _next_pow2(3 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # ----- convolution and temperature schedule -----
        conv = _conv_self_fft(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] soft‑desc iter={it:,} score={cur_score:.6f}")

        # temperature (exponential decay)
        fraction = (time.time() - start_time) / (deadline - start_time)
        tau = tau0 * (tau_end / tau0) ** fraction

        # ----- gradient of smoothed max -----
        w = _softmax_weights(conv, tau)
        grad = _gradient_from_weights(a, w, L)

        # ----- Adam update + projection -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # projection onto simplex (keeps non‑negative & sum = 1)
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as the reference)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 80_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # simple linear cooling
        T = 0.015 * (1e-6 / 0.015) ** (it / total_steps)

        src = rng.choice(n, p=a)                 # source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.25       # move at most 25 % of source
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

        it += 1

        # ----- self‑convolution and current score -----
        conv = _conv_self_fft(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] soft‑desc iter={it:,} score={cur_score:.6f}")

        # ----- temperature schedule (exponential decay) -----
        fraction = (time.time() - start_time) / (deadline - start_time)
        tau = tau0 * (tau_end / tau0) ** fraction

        # ----- gradient of the smoothed max -----
        w = _softmax_weights(conv, tau)
        grad = _gradient_from_weights(a, w, L)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))

        # ----- projection onto simplex -----
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (as in the reference)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 80_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # simple exponential cooling
        T = 0.015 * (1e-6 / 0.015) ** (it / total_steps)

        src = rng.choice(n, p=a)                 # favour large masses
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25       # move ≤ 25 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin prevents an accidental timeout
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 16384

    # --------------------------------------------------------------
    # Warm start – symmetric triangular (tent) shape, normalised.
    # --------------------------------------------------------------
    x = np.arange(n, dtype=float) / (n - 1)
    tent = np.minimum(x, 1.0 - x)          # zero at the ends, peak at the centre
    a0 = tent / tent.sum() if tent.sum() > 0 else np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Optional global starter (if the environment supplies one).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1:
            if starter.size != n:
                # truncate or pad with zeros to match the required length
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size))
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
                if cur < _objective(a0):
                    a0 = starter.copy()
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – soft‑max based projected Adam descent.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining * 0.80
        refined_seq, refined_score = _softmax_descent(
            a0,
            sub_deadline,
            start_time,
            tau0=5e-3,
            tau_end=2e-6,
            step0=0.5,
        )
        if refined_score < _objective(a0):
            a0 = refined_seq
            print(f"[{time.time() - start_time:.2f}s] after soft‑desc best score={refined_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – light annealing polish (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(a0, deadline, start_time)
        if anneal_score < _objective(a0):
            a0 = anneal_seq
            print(f"[{time.time() - start_time:.2f}s] final anneal improvement score={anneal_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation onto the simplex (safety).
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0:
        a0 = a0 / total
    else:
        a0 = np.full(n, 1.0 / n)

    # Return a plain Python list for the harness.
    return [float(v) for v in a0]
