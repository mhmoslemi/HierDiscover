def _smooth_descent(...):
    a = a.copy()
    n = a.shape[0]
    L = _next_pow2(3*n - 2) # since w length = 2n-1
    # precompute indices for projection etc
    best_a = a.copy()
    best_score = _objective(a)
    t = 0
    # hyperparams
    base_lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # lambda schedule
    lam = 0.5
    steps_per_lam = max(500, n // 100)  # maybe 500.
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)  # length 2n-1
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth iter={t:,} score={cur_score:.6f} lam={lam:.2f}")
        # lambda schedule
        if t % steps_per_lam == 0:
            lam = min(lam * 2.0, 256.0)  # increase
        # softmax weights
        mmax = conv.max()
        # to avoid overflow, subtract max
        exp_terms = np.exp(lam * (conv - mmax))
        w = exp_terms / exp_terms.sum()
        # Compute correlation via FFT
        Af = np.fft.rfft(a, L)
        Wf = np.fft.rfft(w, L)
        corr_full = np.fft.irfft(Af * np.conj(Wf), L)[: (n + (2*n-1) -1)]  # length 3n-2
        grad = 2.0 * corr_full[:n]  # first n entries

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = base_lr / np.sqrt(t) * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_simplex(a)
    return best_a, best_score

"""
Propose a step function using a smooth “log‑sum‑exp” surrogate
and Adam updates on the probability simplex.  The surrogate
approximates the infinity‑norm of the self‑convolution;
its gradient is obtained analytically via two FFT‑based
correlations, allowing fast iterations on a large canvas.
The schedule starts with a very smooth surrogate (λ≈0.5) and
gradually sharpens (doubling λ every few thousand steps).
After the gradient phase a short simulated‑annealing polish
moves tiny mass packets between coordinates.  The best
iterate seen (according to the exact objective) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _upsample(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear up‑sample a 1‑D array to length n (preserving total mass)."""
    seq = np.asarray(seq, dtype=float)
    old_n = seq.size
    if old_n == n:
        return seq.copy()
    # map points to [0,1)
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, n, endpoint=False)
    up = np.interp(x_new, x_old, seq)
    # renormalise (the surrogate objective is scale‑invariant)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full(n, 1.0 / n)
    return up

# Core optimisation loop – smooth surrogate + Adam
# ----------------------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best vector (according to the exact objective) found.
    """
    a = a0.copy()
    n = a.shape[0]
    Lw = 2 * n - 1           # length of the convolution vector
    Lfft = _next_pow2(n + Lw - 1)   # length needed for correlation / convolution

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.5

    # lambda schedule (smoothness of the surrogate)
    lam = 0.5
    lam_max = 256.0
    steps_per_lam = max(2000, n // 150)   # how many iterations before λ is doubled

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # self‑convolution (exact) – also gives the true score
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f} λ={lam:.2f}")

        # Update λ periodically
        if it % steps_per_lam == 0 and lam < lam_max:
            lam = min(lam * 2.0, lam_max)

        # ------------------------------------------------------------------
        # Build the soft‑max weights w from the current convolution.
        # We use the numerically stable formulation w ∝ exp(λ·(conv−max)).
        # ------------------------------------------------------------------
        mmax = conv.max()
        # all entries are ≤ mmax, so the exponent is ≤ 0 -> safe
        exp_term = np.exp(lam * (conv - mmax))
        w = exp_term / exp_term.sum()   # shape (2n‑1,)

        # ------------------------------------------------------------------
        # Gradient of the surrogate w.r.t. a.
        # g = ∂/∂a (∑_k w_k·conv_k) = (a ⊛ w) + (a * w)   (⊛ = correlation, * = convolution)
        # Both quantities can be obtained with a single pair of FFTs.
        # ------------------------------------------------------------------
        Af = np.fft.rfft(a, Lfft)
        Wf = np.fft.rfft(w, Lfft)

        # convolution a * w  (full length)
        conv_full = np.fft.irfft(Af * Wf, Lfft)[: n + Lw - 1]

        # correlation a ⋆ w  (full length)
        corr_full = np.fft.irfft(Af * np.conj(Wf), Lfft)[: n + Lw - 1]

        # gradient (first n entries correspond to indices 0 … n‑1)
        grad = conv_full[:n] + corr_full[:n]

        # ------------------------------------------------------------------
        # Adam update (projected back onto the simplex afterwards)
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # keep a on the probability simplex
        a = _project_simplex(a)

    return best_a, best_score

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

# Core optimisation – smooth surrogate + Adam
# ----------------------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a log‑sum‑exp surrogate of the ‖a*a‖∞ term.
    Returns the best vector (according to the exact objective) found.
    """
    a = a0.copy()
    n = a.shape[0]
    Lw = 2 * n - 1                        # length of a*a
    Lfft = _next_pow2(n + Lw - 1)          # length needed for full correlation / convolution

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.5

    # λ schedule (smoothness of the surrogate)
    lam = 0.5
    lam_max = 256.0
    steps_per_lam = max(2000, n // 150)    # double λ every few thousand steps

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # exact convolution – also gives the true score for bookkeeping
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f} λ={lam:.2f}")

        # increase λ periodically (sharpen the surrogate)
        if it % steps_per_lam == 0 and lam < lam_max:
            lam = min(lam * 2.0, lam_max)

        # ------------------------------------------------------------------
        # Build soft‑max weights w from the current convolution.
        # w_k ∝ exp(λ·(conv_k − max(conv)))  – numerically stable.
        # ------------------------------------------------------------------
        mmax = conv.max()
        exp_term = np.exp(lam * (conv - mmax))   # every entry ≤ 1, safe from overflow
        w = exp_term / exp_term.sum()           # length 2·n‑1, symmetric automatically

        # ------------------------------------------------------------------
        # Gradient of the surrogate: g = (a * w) + (a ⋆ w)
        #   *  = linear convolution, ⋆ = cross‑correlation.
        # Both can be obtained from the same pair of FFTs.
        # ------------------------------------------------------------------
        Af = np.fft.rfft(a, Lfft)
        Wf = np.fft.rfft(w, Lfft)

        # convolution a * w (full length)
        conv_full = np.fft.irfft(Af * Wf, Lfft)[: n + Lw - 1]

        # correlation a ⋆ w (full length)
        corr_full = np.fft.irfft(Af * np.conj(Wf), Lfft)[: n + Lw - 1]

        # gradient (indices 0 … n‑1)
        grad = conv_full[:n] + corr_full[:n]

        # ------------------------------------------------------------------
        # Adam update + projection onto the simplex
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score

# Light simulated‑annealing polish (identical to the reference version)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (a probability vector)
    that minimises `evaluate_sequence`.  The routine runs three
    phases – initialisation, smooth Adam descent, and a short
    annealing polish – and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid breaching the hard limit
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a canvas size that is a power of two (fast FFTs).
    # ------------------------------------------------------------------
    n = 32768

    # ------------------------------------------------------------------
    # 1) Initialise – try to reuse the globally supplied construction.
    # ------------------------------------------------------------------
    init_seq = None
    try:
        init_raw = np.asarray(height_sequence_1, dtype=float)
        if init_raw.ndim == 1 and init_raw.size > 0:
            init_seq = _upsample(init_raw, n)
            print(f"[{time.time() - start_time:.2f}s] Upsampled global starter.")
    except Exception:
        pass

    if init_seq is None:
        # fallback: a symmetric tent shape (p=1, α=0)
        side = n // 2
        ramp = np.arange(1, side + 1, dtype=float) / side
        shape = np.concatenate([ramp, ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
        shape /= shape.sum()
        init_seq = shape
        print(f"[{time.time() - start_time:.2f}s] Using built‑in tent starter.")

    # ------------------------------------------------------------------
    # 2) Smooth Adam descent on the surrogate.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:               # reserve a few seconds for polishing
        sub_deadline = time.time() + 0.80 * remaining
        refined_seq, refined_score = _smooth_adam_descent(init_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < _objective(init_seq):
            init_seq = refined_seq
            print(f"[{time.time() - start_time:.2f}s] Adam stage finished, "
                  f"best score = {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Light annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polished_seq, polished_score = _anneal(init_seq, deadline, start_time)
        if polished_score < _objective(init_seq):
            init_seq = polished_seq
            print(f"[{time.time() - start_time:.2f}s] Anneal stage finished, "
                  f"best score = {polished_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to a plain Python list
    # ------------------------------------------------------------------
    init_seq = np.maximum(init_seq, 0.0)
    total = init_seq.sum()
    if total > 0:
        init_seq = init_seq / total
    else:
        init_seq = np.full(n, 1.0 / n)

    final_score = _objective(init_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in init_seq]
