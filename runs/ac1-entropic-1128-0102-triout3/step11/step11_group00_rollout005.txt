```python
"""
Step‑function optimizer.

The algorithm searches a probability vector `a` (non‑negative, sum = 1) that
minimises

    2·n·max(conv(a,a))

where `conv` is the linear self‑convolution.  The optimisation consists of

* a cheap parametric sweep (geometric, power, beta, sine, Gaussian) to
  obtain a few promising starting points,
* a *smooth‑max* projected gradient descent.  The non‑smooth max is
  replaced by `log‑sum‑exp`, whose gradient can be computed with two FFTs
  per iteration.  The smoothing parameter β is increased from a small
  value to a large one (homotopy), allowing the optimiser to escape
  shallow local minima.
* a final micro‑hill‑climb that moves tiny amounts of mass between
  random indices.

All updates are projected onto the probability simplex with the
Euclidean projection.  The routine respects the 1000 s budget and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _conv_full_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of x and y of lengths m,n using FFT of length L."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    full = np.fft.irfft(X * Y, n=L)
    return full

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Fast exact evaluation for a (sum≈1) vector
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector (any sum)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent (homotopy on β)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    max_iters: int,
                    step0: float,
                    beta_start: float,
                    beta_end: float,
                    L_fft_self: int,
                    L_fft_full: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed objective

        Jβ(a) = 2·n·Sβ(a) ,   Sβ = (log Σ exp(β·conv)) / β

    β is increased linearly from beta_start to beta_end.
    The simplex projection guarantees non‑negativity and ∑a = 1.
    """
    a = a0.copy()
    n = a.shape[0]
    C = 2.0 * n
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft_self)

    beta = beta_start
    beta_inc = (beta_end - beta_start) / max_iters if max_iters > 0 else 0.0

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # self‑convolution (full length 2n‑1)
        conv = _conv_self(a, L=L_fft_self)

        # soft‑max weights for the current β
        max_c = conv.max()
        # use log‑sum‑exp trick for numerical stability
        exp_vals = np.exp(beta * (conv - max_c))
        Z = exp_vals.sum()
        w = exp_vals / Z                     # length 2n‑1, sums to 1

        # gradient of the smoothed max:
        #   g0_i = 2 * Σ_j w_{i+j} * a_j  (valid indices only)
        # Compute via FFT: full = w * a (full convolution)
        full = _conv_full_fft(w, a, L_fft_full)
        offset = len(w) - 1                  # index where a_0 aligns
        g0 = 2.0 * full[offset:offset + n]   # length n

        # gradient of Sβ is just g0; the outer factor 2·n is constant,
        # we absorb it in the step size.
        grad = g0

        # step size schedule α_t = step0 / sqrt(t)
        step = step0 / math.sqrt(t)
        a = a - step * grad
        a = _project_simplex(a)

        # occasionally keep the best solution seen
        cur_score = _eval_fast(a, L_fft_self)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Smooth it={t:,} β={beta:.2f} score={cur_score:.6f}")

        # increase β gradually
        beta = min(beta_end, beta + beta_inc)

    return best_a, best_score

# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(start_a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft_self: int) -> tuple[np.ndarray, float]:
    """
    Very small mass transfers (ε≈10⁻⁶·a_src) accepted only if the true
    objective improves.  Acts as a micro‑hill‑climb after the main
    optimisation phases.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft_self)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4   # tiny move, ~10⁻⁴ of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft_self)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()

"""
Step‑function optimizer.

The algorithm searches a probability vector `a` (non‑negative, sum = 1) that
minimises

    2·n·max(conv(a,a))

where `conv` is the linear self‑convolution.  The optimisation consists of

* a cheap parametric sweep (geometric, power, beta, sine, Gaussian) to
  obtain a few promising starting points,
* a *smooth‑max* projected gradient descent.  The non‑smooth max is
  replaced by `log‑sum‑exp`, whose gradient can be computed with two FFTs
  per iteration.  The smoothing parameter β is increased from a small
  value to a large one (homotopy), allowing the optimiser to escape
  shallow local minima.
* a final micro‑hill‑climb that moves tiny amounts of mass between
  random indices.

All updates are projected onto the probability simplex with the Euclidean
projection.  The routine respects the 1000 s budget and returns the best
sequence found as a list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of x and y (length = len(x)+len(y)-1) using FFT."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    full = np.fft.irfft(X * Y, n=L)
    return full


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and Σ w = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum can be arbitrary)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent (β‑homotopy)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    max_iters: int,
                    step0: float,
                    beta_start: float,
                    beta_end: float,
                    L_fft_self: int,
                    L_fft_full: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed objective

        Jβ(a) = 2·n·Sβ(a) ,   Sβ = (log Σ exp(β·conv)) / β

    β is increased linearly from ``beta_start`` to ``beta_end``.
    The simplex projection guarantees non‑negativity and Σa = 1.
    """
    a = a0.copy()
    n = a.shape[0]
    C = 2.0 * n                     # outer constant (kept for reporting only)
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft_self)

    beta = beta_start
    beta_inc = (beta_end - beta_start) / max_iters if max_iters > 0 else 0.0

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # 1) self‑convolution (full length 2n‑1)
        conv = _conv_self(a, L=L_fft_self)

        # 2) soft‑max weights for the current β (log‑sum‑exp trick)
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        Z = exp_vals.sum()
        w = exp_vals / Z                     # length 2n‑1, Σw = 1

        # 3) gradient of the smoothed max:
        #    g0_i = 2 * Σ_j w_{i+j} * a_j   (valid indices only)
        full = _conv_full_fft(w, a, L_fft_full)   # length ≥ 3n‑2
        offset = len(w) - 1
        g0 = 2.0 * full[offset:offset + n]        # shape (n,)

        # 4) gradient step (we optimise Sβ; the factor 2·n is absorbed
        #    into the step size)
        grad = g0
        step = step0 / math.sqrt(t)
        a = a - step * grad
        a = _project_simplex(a)

        # 5) keep the best solution seen so far
        cur_score = _eval_fast(a, L_fft_self)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Smooth it={t:,} β={beta:.2f} score={cur_score:.6f}")

        # 6) increase β gradually
        beta = min(beta_end, beta + beta_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _tiny_hill_climb(start_a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft_self: int) -> tuple[np.ndarray, float]:
    """
    Very small mass transfers (ε≈10⁻⁴·a_src) accepted only if the true
    objective improves.  Acts as a micro‑hill‑climb after the main
    optimisation phases.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft_self)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4   # tiny transfer
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft_self)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={best_score:.6f}")

    return best_a, best_score
