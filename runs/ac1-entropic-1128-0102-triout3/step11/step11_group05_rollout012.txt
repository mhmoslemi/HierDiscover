def propose_candidate(seed=0, budget_s=1000, **kwargs):
    random seeding.
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096

    # Load previous best if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.shape[0] != n:
            if prev.shape[0] > n:
                prev = prev[:n]
            else:
                prev = np.pad(prev, (0, n - prev.shape[0]), mode='constant')
        prev /= prev.sum()
    except Exception:
        prev = None

    # Generate initial candidates
    rng = np.random.default_rng(seed)
    init_candidates = []
    # uniform
    init_candidates.append(np.full(n, 1.0/n))
    # Dirichlet with concentration 0.5,1,2
    for conc in [0.5, 1.0, 2.0]:
        a = rng.dirichlet(np.full(n, conc))
        init_candidates.append(a)
    # previous best if available
    if prev is not None:
        init_candidates.append(prev)
    # perturbed uniform: add small noise
    for sigma in [0.01, 0.05]:
        a = np.full(n, 1.0/n) + rng.normal(scale=sigma/n, size=n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        init_candidates.append(a)

    # Evaluate fast for each candidate and keep best
    L_self = _next_pow2(2 * n - 1)
    best_seq = None
    best_score = float('inf')
    for idx, a in enumerate(init_candidates):
        sc = _eval_fast(a, L_self)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[{time.time()-start_time:.2f}s] Init {idx} score={sc:.6f}")

    # Phase 1: smooth descent on best_seq for longer time portion
    if time.time() < deadline - 20.0:
        sd_deadline = deadline - 15.0
        best_seq, best_score = _smooth_descent_modified(best_seq, n, sd_deadline, start_time)
        print(f"[{time.time()-start_time:.2f}s] After smooth descent best={best_score:.6f}")

    # Phase 2: subgradient
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent_modified(best_seq, n, sub_deadline, start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time()-start_time:.2f}s] Subgrad improved to {best_score:.6f}")

    # Phase 3: anneal
    if time.time() < deadline:

        a_final, sc_final = _inplace_anneal(best_seq, n, anneal_deadline, start_time,

def _smooth_descent_modified(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)
    it = 0
    # alphas: values from 0.5 to 800, geometric spacing 50 points
    alphas = np.geomspace(0.5, 800.0, num=50)
    max_it = 600_000
    for alpha in alphas:
        while time.time() < deadline - 0.1 and it < max_it:
            it += 1
            # Compute self conv via FFT for speed
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]

            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[:3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]

            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            # to avoid underflow
            a = np.maximum(a, 1e-30)
            a /= a.sum()

def _inplace_anneal(a0, n, deadline, start_time, cur_best_score, cur_best_seq,

        it += 1
        # temperature schedule based on elapsed proportion
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue

        if src == dst:
            continue

        if eps <= 0.0:
            continue

        conv_dst = conv[dst:dst + n]

        conv_src -= 2.0 * eps * a
        conv_dst += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        new_max = conv.max()

        delta_score = new_score - cur_score

assistantfinal**Solution Overview**

The candidate is built in three stages  

1. **Diverse initialisation** – a few simple distributions (uniform, Dirichlet,
   a perturbed version of the previous best) are generated and the one with the
   lowest objective is kept.

2. **Smooth‑max mirror descent** – a mirror‑descent on the probability simplex
   using the log‑sum‑exp smoothing of the max‑convolution.  The step size and
   smoothing schedule are a little more aggressive than in the reference code
   and the loop runs for up to half a million iterations (time‑controlled).

3. **Projected sub‑gradient polishing** – a short momentum sub‑gradient phase
   reduces the remaining error.

4. **Fast in‑place simulated‑annealing** – a mass‑transfer annealer that updates
   the convolution *in‑place* (no copy of the full array).  This makes it
   possible to execute a few million moves inside the 1000 s budget, giving a
   finer polishing of the max‑convolution.

All phases respect the time budget and keep the best sequence seen so far.
The final vector is normalised to sum = 1 and returned as a plain Python list.

```python
"""Candidate generation for the step‑function optimisation problem.

The algorithm proceeds as follows

* generate several simple initial vectors (uniform, Dirichlet, a noisy
  version of the previous best) and keep the best according to a fast FFT‑based
  evaluation;
* run a smooth‑max mirror‑descent (exponentiated‑gradient) on the probability
  simplex – this quickly reduces the peak of the self‑convolution;
* perform a short projected sub‑gradient descent with momentum for a fine
  adjustment;
* finish with a fast *in‑place* simulated‑annealing that repeatedly moves a
  small amount of mass from a source index to a destination index while
  updating the convolution efficiently.  The annealer runs for as many moves
  as the remaining time allows.

The whole pipeline respects a 1000 s budget and always returns a non‑negative
step‑function (list of heights) whose objective value is printed for
monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast objective for a probability vector (sum ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent with a log‑sum‑exp smoothing of the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # a richer schedule of smoothing parameters
    alphas = np.geomspace(0.5, 800.0, num=50)

    it = 0
    max_iterations = 600_000
    for alpha in alphas:
        while time.time() < deadline - 0.1 and it < max_iterations:
            it += 1

            # 1) self‑convolution (FFT)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) smooth‑max weights (log‑sum‑exp)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2·(w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-30)               # avoid exact zeros
            a /= a.sum()

            # occasional evaluation of the exact objective
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.1:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """A short momentum sub‑gradient stage for fine polishing."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2                     # a bit smaller than the reference
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (first occurrence of the max)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of the max term

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : fast incremental simulated‑annealing (in‑place convolution update)
# ----------------------------------------------------------------------
def _inplace_anneal(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    cur_best_score: float,
                    cur_best_seq: np.ndarray,
                    start_temp: float = 5e-2,
                    end_temp: float = 1e-12,
                    max_moves: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer that updates the convolution in‑place."""
    a = a0.copy()
    conv = np.convolve(a, a)               # length 2n‑1, cheap for n=4096
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max          # sum(a)=1 ⇒ denominator = 1
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1

        # linear temperature schedule (exponential decay)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** progress

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 0.35   # up to 35 % of the source mass
        if eps <= 0.0:
            continue

        # --- in‑place update of the convolution ---------------------------------
        conv_src = conv[src:src + n]          # view, length n
        conv_dst = conv[dst:dst + n]          # view, length n

        conv_src -= 2.0 * eps * a
        conv_dst += 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        if delta_score <= 0.0 or math.exp(-delta_score / T) > rng.random():
            # accept the move
            a[src] -= eps
            a[dst] += eps
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")
        else:
            # reject – revert the convolution updates
            conv_src += 2.0 * eps * a
            conv_dst -= 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps

    return best_seq, best_score

    rng = np.random.default_rng(seed)

        a = np.maximum(a, 0.0)
        a /= a.sum()
        init_vectors.append(a)

        if prev.sum() > 0.0:
            prev /= prev.sum()
            init_vectors.append(prev)
    except Exception:

    # --------------------------------------------------------------
    # 0️⃣‑b  Pick the best starter according to the fast evaluation
```
