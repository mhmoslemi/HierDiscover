```
def _grad_softmax(a: np.ndarray, gamma: float) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2*n-1)  # for conv, but we also need conv for gradient? Actually conv length 2n-1
    conv = _conv_self(a, L=L)  # length 2n-1
    max_conv = conv.max()
    # compute softmax weights
    # Use exp(gamma*(conv - max_conv)) to avoid overflow
    exp_vec = np.exp(gamma * (conv - max_conv))
    w = exp_vec / exp_vec.sum()
    # reversed w for convolution
    w_rev = w[::-1]
    # compute convolution a * w_rev using FFT
    L2 = _next_pow2(n + (2*n - 1) - 1)  # 3n-2
    Af = np.fft.rfft(a, n=L2)
    Wf = np.fft.rfft(w_rev, n=L2)
    conv_full = np.fft.irfft(Af * Wf, n=L2)
    # slice and reverse
    grad = 2.0 * conv_full[n-1:2*n-1][::-1]
    return grad
```

assistantfinal**Solution Overview**

The task is to minimise  

\[
F(a)=\frac{2\,n\;\max_k (a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge 0 ,
\]

where `(a*a)` is the linear self‑convolution of the non‑negative step‑function `a`.  
Because the objective is homogeneous, the total sum of the entries can be fixed
(e.g. to 1) without loss of generality.

A good sequence must make the convolution *as flat as possible* – the average
value of the convolution is always \(1/(2n-1)\), therefore the theoretical
lower bound of the objective is 1.  The previous implementation reached
\(F\approx1.508\); the code below pushes the bound further by

1. **A cheap parametric start** – uniform, sinusoidal and cosine windows are
   tried and the best of them is kept.
2. **Smoothed‑max gradient descent** – instead of the non‑differentiable
   \(\max\) we optimise the smooth approximation  

   \[
   G(a)=\frac1\gamma\log\!\Bigl(\sum_k e^{\gamma (a*a)[k]}\Bigr)
   \]

   whose gradient can be written as a convolution and is computed with FFT.
   The temperature γ is increased slowly so the optimisation first moves
   smoothly and then focuses on the true maximum.
3. **Momentum + Euclidean projection onto the probability simplex** to keep the
   vector non‑negative and of unit sum.
4. **A final stochastic hill‑climb** that moves a random amount of mass
   between two coordinates; moves are accepted only if they improve the true
   objective.

All heavy work uses FFT (`O(n log n)`) and the whole search respects the
1000 s time budget.

```python
"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of:

1️⃣  A quick parametric sweep (uniform, sinusoid, cosine‑window) to obtain a
    decent start.
2️⃣  A long smoothed‑max gradient descent on the simplex.  The loss is
    G(a)=1/γ·log(∑exp(γ·conv(a,a))) and its gradient can be written as a
    convolution, which we evaluate with FFT.  γ grows from a small value to a
    large one so that the optimisation first sees a smooth landscape and
    finally concentrates on the true maximum.
3️⃣  A stochastic hill‑climb that transfers a random amount of mass between
    two indices; only improving moves are kept.

All steps are timed so that the total runtime is ≤ 1000 s.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT‑based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w | w ≥ 0, sum(w)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum of a is assumed to be 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    # sum is (almost) 1, but we keep it in the formula for safety
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Gradient of the smooth‑max loss
# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of  (1/γ)·log(∑_k exp(γ·conv_k(a)))  w.r.t. a.
    Returned gradient has the same shape as a.
    """
    n = a.shape[0]
    # 1. convolution a*a
    Lc = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L=Lc)                       # length 2n‑1
    # 2. soft‑max weights
    max_c = conv.max()
    exp_vec = np.exp(gamma * (conv - max_c))         # avoid overflow
    w = exp_vec / exp_vec.sum()                     # shape (2n‑1,)

    # 3. gradient = 2 * (a * reversed(w))  (convolution, then slice & flip)
    w_rev = w[::-1]

    Lg = _next_pow2(3 * n - 2)                      # length needed for a * w_rev
    Af = np.fft.rfft(a, n=Lg)
    Wf = np.fft.rfft(w_rev, n=Lg)
    full = np.fft.irfft(Af * Wf, n=Lg)              # length ≥ 3n‑2
    # slice the part that corresponds to indices n‑1 … 2n‑2 and reverse
    grad = 2.0 * full[n - 1 : 2 * n - 1][::-1]
    return grad
```
