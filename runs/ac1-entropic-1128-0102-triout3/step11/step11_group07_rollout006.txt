def subgradient_descent(a, deadline, start_time, L_fft, L_grad):
    # a is numpy array normalized to sum 1.
    best = a copy
    best_score = evaluate_fast
    eta = 0.1
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a, L_fft)
        M = conv.max()
        # active set mask
        tol = 1e-12
        active = (conv >= M - tol).astype(np.float64)  # float mask
        # compute subgradient: g = 2 * a * conv(active_rev)
        # compute conv of a with reversed active using FFT
        # act_rev = active[::-1]
        act_rev = active[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(act_rev, n=L_grad)
        gg_full = np.fft.irfft(Af * Wf, n=L_grad)
        # we need indices offset to get length 2n-1 conv orientation.
        offset = 2*n - 2
        grad = 2.0 * gg_full[offset - np.arange(n)]

        # mirror descent update:
        a_new = a * np.exp(- eta * grad)
        # project onto simplex: we can just renormalize (since product positive).
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.full_like(a, 1.0 / n)

def _subgradient_descent(a0, deadline, start_time, L_fft, L_grad):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _eval_fast(a, L_fft)
    eta = 0.2  # starting step (learning rate)
    iter = 0
    no_improve_iters = 0
    while time.time() < deadline:
        iter += 1
        conv = _conv_self(a, L_fft)
        M = conv.max()
        # create active mask as binary
        # tolerance: absolute: maybe 1e-12, relative relative tolerance 1e-12*max.
        tol = 1e-12
        active = (conv >= M - tol).astype(np.float64)
        # compute gradient
        act_rev = active[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(act_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)

def _hill_climb(a0, deadline, start_time, L_fft):
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = len(a)
    it = 0

        it += 1
        src = rng.choice(n, p=a)  # selecting index based on probability encourages moving from higher mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.05   # move up to 5% of src
        a_new = a.copy()

"""
Propose a non‑negative step function (a probability vector) that
minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages:

1️⃣  Fast parametric sweep (geometric, power, beta families) to obtain a
    good starting shape.

2️⃣  Mirror‑descent on the exact sub‑gradient of the max‑convolution.
    The update is
        a ← a * exp(-η·∇)  (mirrored onto the simplex).
    η is slowly decreased; the routine stops when the time‑budget is
    exhausted or when no improvement has been seen for many iterations.

3️⃣  Tiny random hill‑climb that transfers a small amount of mass between
    two indices and keeps an improvement.

All operations are performed with FFT‑based convolutions, the vector is
kept normalised (∑a = 1) and projected onto the probability simplex after
each change.  The code respects the 1000 s budget and prints intermediate
scores.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2).  a is assumed non‑negative."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweep – quick way to get a decent start
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, L_fft: int,
                      start_time: float, deadline: float):
    """Explore three low‑dimensional families and keep the best vector."""
    best_seq = np.full(n, 1.0 / n)
    best_score = _eval_fast(best_seq, L_fft)

    idx = np.arange(n, dtype=np.float64)

    # ----- geometric family (with optional uniform mixing) -----
    r_vals = np.exp(
        np.linspace(np.log(0.9), np.log(0.999999), 2000)  # finer near 1
    )
    lam_vals = np.linspace(0.0, 0.5, 6)                # 0,0.1,…,0.5
    uniform = np.full(n, 1.0 / n)

    for r in r_vals:
        if time.time() >= deadline - 30:
            break
        geo = np.power(r, idx, dtype=np.float64)
        s = geo.sum()
        if s == 0.0:
            continue
        geo /= s
        for lam in lam_vals:
            cand = lam * uniform + (1.0 - lam) * geo
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo r={r:.6f} lam={lam:.2f} → {sc:.6f}")

    # ----- power family -----
    p_vals = np.linspace(-0.9, 2.0, 200)
    for p in p_vals:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1.0, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0.0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} → {sc:.6f}")

    # ----- beta‑type family -----
    beta_vals = np.linspace(0.0, 2.0, 11)   # 0,0.2,…,2
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = np.power(idx + 1.0, p) * np.power(n - idx, q)
            s = beta_seq.sum()
            if s == 0.0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} → {sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Mirror‑descent on the exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the exact sub‑gradient of max(conv(a,a)).
    The update is a ← a·exp(−η·g)  (then renormalise).  η slowly decays.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _eval_fast(a, L_fft)

    eta = 0.2                     # initial learning‑rate
    it = 0
    stagnated = 0                # how many iterations without improvement

    offset = 2 * n - 2            # used for extracting the gradient slice

    while time.time() < deadline:
        it += 1
        # convolution and max
        conv = _conv_self(a, L=L_fft)
        M = conv.max()

        # active set (indices achieving (or very close to) the max)
        tol = 1e-12
        active = (conv >= M - tol).astype(np.float64)

        # sub‑gradient = 2 * a * (active reversed)  (convolution)
        act_rev = active[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(act_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # try a few step‑sizes, keep the first that improves the score
        factor = eta
        improved = False
        for _ in range(5):
            # mirror‑descent update
            exp_factor = np.exp(-factor * grad)
            # safeguard against overflow/underflow
            exp_factor = np.clip(exp_factor, 1e-50, 1e50)
            a_new = a * exp_factor
            s = a_new.sum()
            if s == 0.0:
                a_new = np.full_like(a, 1.0 / n)
            else:
                a_new = a_new / s

            cur_score = _eval_fast(a_new, L_fft)
            if cur_score < best_score - 1e-12:
                improved = True
                break
            factor *= 0.5  # shrink step if no improvement

        if improved:
            a = a_new
            best = a.copy()
            best_score = cur_score
            stagnated = 0
            # slightly increase learning‑rate after a good step
            eta = min(eta * 1.05, 1.0)
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} η={eta:.3e} score={cur_score:.6f}")
        else:
            # no improvement – decay η and count stagnation
            eta = max(eta * 0.8, 1e-6)
            stagnated += 1
            if stagnated > 8000:          # give up if we are stuck for a long while
                break

    return best, best_score


# ----------------------------------------------------------------------
# 3️⃣  Tiny random hill‑climb (final polish)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Randomly transfer a tiny amount of mass; keep only improvements."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        # pick source proportional to its mass (more likely to move from large entries)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # up to 5 % of the source mass
        eps = a[src] * rng.random() * 0.05
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # re‑normalise (the sum is unchanged but numerical drift may appear)
        a_prop = np.maximum(a_prop, 0.0)
        a_prop /= a_prop.sum()

        cur_score = _eval_fast(a_prop, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} score={cur_score:.6f}")

    return best, best_score
