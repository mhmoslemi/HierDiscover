"""
Simulated‑annealing / gradient‑guided local search for the step‑function
objective

    2·n·max(conv(a,a)) / (sum(a)**2)

with n = 4096.  The algorithm starts from the best known sequence (global
variable ``height_sequence_1`` if present, otherwise uniform), then repeatedly
moves a small amount of mass from an index that contributes strongly to the
current maximum convolution (estimated by a sub‑gradient) to an index that
contributes weakly.  Moves are accepted with a Metropolis probability that
gradually cools, allowing occasional uphill steps to escape local minima.
The search runs until the wall‑clock deadline (≈1000 s) and returns the
best sequence found, normalised to unit sum.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) using FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The function respects the time budget (default 1000 s) and
    returns a Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the external harness
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution

    # --------------------------------------------------------------
    # Load a starter vector (global best if available, else uniform)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        starter = None

    if starter is None or starter.ndim != 1 or starter.size == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = starter.astype(float)
        # adapt length
        if a.shape[0] != n:
            if a.shape[0] > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.shape[0]), mode='constant')
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

    # --------------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------------
    conv = _conv_self(a, L_self)
    cur_score = 2.0 * n * conv.max()            # sum(a) == 1
    best_seq = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing parameters
    # --------------------------------------------------------------
    max_temp = 1.0e-2          # start temperature
    min_temp = 1.0e-6          # final temperature
    max_eps0 = 0.30            # start max fraction of source mass moved
    min_eps0 = 0.001           # final max fraction
    it = 0
    max_iters = 10_000_000     # safety cap; real stop is the wall‑clock

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline and it < max_iters:
        # ----------------------------------------------------------
        # Current convolution, max value and active set
        # ----------------------------------------------------------
        conv = _conv_self(a, L_self)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # active indices where the max is attained (within a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # ----------------------------------------------------------
        # Sub‑gradient of the max‑convolution (used for source/dest bias)
        # ----------------------------------------------------------
        grad = np.zeros(n, dtype=float)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # source distribution – favour indices that contribute most
        src_weights = grad.copy()
        if src_weights.sum() == 0.0:
            src_weights = a
        src_weights /= src_weights.sum()

        # destination distribution – favour low‑gradient locations
        inv_grad = 1.0 / (grad + 1e-12)
        dest_weights = inv_grad / inv_grad.sum()

        # ----------------------------------------------------------
        # Temperature & step‑size schedule (linear cooling)
        # ----------------------------------------------------------
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (budget_s - 0.5))
        T = max_temp * (1.0 - frac) + min_temp * frac
        eps_factor = max_eps0 * (1.0 - frac) + min_eps0 * frac

        # ----------------------------------------------------------
        # Propose a mass‑transfer move
        # ----------------------------------------------------------
        src = rng.choice(n, p=src_weights)
        dst = rng.choice(n, p=dest_weights)
        if src == dst:
            it += 1
            continue

        eps = rng.random() * a[src] * eps_factor
        if eps <= 0.0:
            it += 1
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # a_prop stays non‑negative because eps ≤ a[src]

        # ----------------------------------------------------------
        # Evaluate the proposal
        # ----------------------------------------------------------
        conv_prop = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - cur_score

        # ----------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / max(T, 1e-15)):
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] new best = {best_score:.6f}")

        # occasional progress report
        if it % 5000 == 0:
            print(f"[{time.time() - start_time:.2f}s] it={it:,} cur={cur_score:.6f} best={best_score:.6f}")

        it += 1

    # --------------------------------------------------------------
    # Final normalisation (protect against tiny drift)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
