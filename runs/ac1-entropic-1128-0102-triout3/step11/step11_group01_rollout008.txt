```
def _adam_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    init_lr = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it} score={cur_score:.6f}")
        # gradient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = init_lr * (1.0 / (np.sqrt(v_hat) + eps))
        a = a - step * m_hat

        a = _project_simplex(a)
    return best_a, best_score
```

The program searches for a non‑negative step‑function (a list of heights) that minimises  

\[
\text{score}= \frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

the same quantity computed by `evaluate_sequence`.  
The search consists of five stages:

1. **Problem size** – a large power‑of‑two length (`n = 65536`) reduces the discretisation error of the continuous optimum (≈ 1.500).

2. **Rich family of analytic shapes** – several symmetric parametrised families are generated  
   * power‑ramp with optional flat plateau (the previous work)  
   * β‑distribution (`(x(1‑x))^p`)  
   * sine‑power (`sin(πx)^p`)  
   * exponential decay (`exp(-c·|x‑½|)`).  

   Every shape is normalised to unit ℓ₁‑norm and its score is evaluated with a fast FFT based convolution.  
   The ten best shapes are kept.

3. **Projected Adam descent** – starting from the current best shape we minimise the non‑smooth objective directly.  
   The sub‑gradient is taken from the convolution entry that attains the current maximum.  
   Adam‑style adaptive moments together with a diminishing learning‑rate, followed by a Euclidean projection onto the probability simplex, yields fast convergence.  
   Symmetry is enforced after each projection (`a ← (a+rev(a))/2`).

4. **Simulated‑annealing polishing** – a cheap “mass‑packet’’ move scheme (the same as before) searches the neighbourhood of the refined shape for further improvements.

5. **Final normalisation** – the result is clipped, re‑normalised to unit sum and returned as a plain Python list.

```python
"""
Search for a step‑function (non‑negative heights) that gives a low
value of the evaluation function.  The algorithm uses:

* a large canvas (n = 65536) – finer discretisation,
* several analytic symmetric families (ramp‑plateau, beta, sine‑power,
  exponential) to get good seeds,
* projected Adam descent on the exact non‑smooth objective,
* a short simulated‑annealing polish.

The code respects the 1000 s budget and returns a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σw = 1, w ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generators – all produce a symmetric non‑negative vector of length n
# ----------------------------------------------------------------------
def _make_ramp_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Power‑ramp with optional flat plateau (same as the previous work)."""
    alpha = max(0.0, min(alpha, 0.5))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, p: float) -> np.ndarray:
    """Symmetric beta‑like shape: (x(1‑x))^p, x∈[0,1]."""
    i = np.arange(n, dtype=float)
    x = i / (n - 1)
    shape = (x * (1.0 - x)) ** p
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sine_shape(n: int, p: float) -> np.ndarray:
    """Symmetric sine‑power shape: sin(πx)^p, x∈[0,1]."""
    i = np.arange(n, dtype=float)
    x = i / (n - 1)
    shape = np.sin(math.pi * x) ** p
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exp_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay: exp(-c·|x‑½|), x∈[0,1]."""
    i = np.arange(n, dtype=float)
    x = i / (n - 1)
    d = np.abs(x - 0.5) * 2.0        # distance in [0,1]
    shape = np.exp(-c * d)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Projected Adam (adaptive moment) descent on the exact non‑smooth objective
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected Adam descent until ``deadline``.
    Returns the best iterate (and its score) seen during the run.
    Symmetry is enforced after each projection.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    init_lr = 0.3

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = init_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project onto simplex and enforce symmetry
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions (Metropolis)."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of attempts – adapt to remaining time
    max_steps = 250_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
```
