"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Three‑stage exponential‑gradient optimisation on increasing problem
   sizes (≈800 → 2400 → 4800).  The objective is a smooth log‑sum‑exp
   approximation of the max‑convolution, and updates are performed with
   exponentiated‑gradient (multiplicative) steps – they stay automatically
   on the probability simplex.

2. A short projected‑gradient polishing phase that uses the exact sub‑gradient
   of the non‑smooth max‑convolution.

3. A fast *incremental* simulated‑annealing / mass‑transfer local search.
   The self‑convolution is updated in‑place using the closed‑form effect of
   moving a small ε‑mass from a source index to a destination index.  This
   allows millions of cheap moves in the remaining time budget and typically
   pushes the score a little further (target ≈ 1.503).

All numbers are kept non‑negative and normalised to sum = 1, therefore the
call to ``evaluate_sequence`` (which rescales internally) is always valid.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: nearest power of two (FFT length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) (active set)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a using all indices within `rel_tol` of the maximum."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) approximation of max(conv)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """Gradient of f(a)= (1/α)·log Σ_k exp(α·conv_k)."""
    n = a.shape[0]

    # weights w_k = exp(α·conv_k) / Σ exp(...)
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)          # stabilise
    w /= w.sum()

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (smoothed max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.3,
                              steps_per_alpha: int = 8000,
                              max_iter: int = 10**7):
    """
    Multiplicative updates with a smoothed max objective.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    fft_len = _next_pow2(3 * n - 2)       # enough for all convolutions
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent using the exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7):
    """
    Classical projected gradient descent with momentum.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum

        # Euclidean projection onto the simplex (O(n))
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) between stages
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Incremental simulated‑annealing / mass‑transfer local search
# ----------------------------------------------------------------------
def _run_incremental_local_search(a0: np.ndarray,
                                  deadline: float,
                                  start_time: float,
                                  start_temp: float = 5e-2,
                                  end_temp: float = 1e-6,
                                  recompute_every: int = 200_000):
    """
    Very cheap mass‑transfer moves that update the convolution in‑place.
    The move transfers a random ε‑fraction of mass from a source index to a
    destination index.  Acceptance follows a simple annealing schedule.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)                     # exact self‑convolution (length 2n‑1)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    accept_cnt = 0
    total_moves = 0

    # pre‑compute total budget for temperature scaling
    total_budget = deadline - start_time

    while time.time() < deadline:
        total_moves += 1

        # temperature (exponential cooling)
        elapsed = time.time() - start_time
        frac = elapsed / total_budget if total_budget > 0 else 1.0
        T = start_temp * ((end_temp / start_temp) ** frac)

        # choose source weighted by its current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move a random fraction of the source mass (5‑50 %)
        eps = a[src] * rng.uniform(0.05, 0.5)
        if eps <= 0.0:
            continue

        # ----- apply move to conv (in‑place) -----
        # store the old slices for a possible revert
        # (we will revert by applying the opposite delta if the move is rejected)
        conv[dst:dst + n] += 2.0 * eps * a
        conv[src:src + n] -= 2.0 * eps * a
        conv[2 * src] += eps * eps
        conv[2 * dst] += eps * eps
        conv[src + dst] -= 2.0 * eps * eps

        new_score = 2.0 * n * conv.max()
        delta = new_score - cur_score

        # ----- acceptance test -----
        accept = False
        if delta < 0:
            accept = True
        else:
            if math.exp(-delta / T) > rng.random():
                accept = True

        if accept:
            # keep the move: update a and cur_score
            a[src] -= eps
            a[dst] += eps
            cur_score = new_score
            accept_cnt += 1

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] inc‑search improve: score={cur_score:.6f}")

        else:
            # reject → revert conv (using the same a because a is unchanged)
            conv[dst:dst + n] -= 2.0 * eps * a
            conv[src:src + n] += 2.0 * eps * a
            conv[2 * src] -= eps * eps
            conv[2 * dst] -= eps * eps
            conv[src + dst] += 2.0 * eps * eps

        # occasional full recomputation to avoid drift
        if accept_cnt % recompute_every == 0 and accept_cnt > 0:
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] recomputed conv: score={cur_score:.6f}")

    return best_a, best_score

    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
