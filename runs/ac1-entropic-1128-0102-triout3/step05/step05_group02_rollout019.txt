"""
Propose a step‑function with a very low L∞ self‑convolution.
The search proceeds through increasing lengths, warm‑starts with a
variety of parametric families, then refines each candidate by a
projected (Euclidean) sub‑gradient descent that uses the *exact* gradient
of the active convolution entry (the correct gradient for
maxₖ (a * a)[k] is 2·a_{k‑i} when the index is inside the support).
Finally a cheap mass‑transfer simulated annealing finishes the
polish.  The routine respects the 1000 s time budget and prints progress.
"""
import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv

def score_of(a: np.ndarray) -> float:
    """Exact objective for a normalized (sum = 1) vector."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()

# ----------------------------------------------------------------------
# Upsampling / interpolation
# ----------------------------------------------------------------------
def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
# Parametric shape families for cheap warm‑starts
# ----------------------------------------------------------------------
def shape_product_power(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i+0.5)*(n‑i‑0.5) raised to power p.
    p>0 → centre‑heavy, p<0 → edge‑heavy.
    """
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term = np.where(term == 0, 1e-12, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def shape_power_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1)
    a = i ** (-p)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def shape_cosine(n: int) -> np.ndarray:
    """Cosine‑bell shape."""
    i = np.arange(n)
    a = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact ∞‑norm gradient)
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the simplex,
    using the correct gradient of the active convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.7               # base step size
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        idx = conv.argmax()
        cur_sc = 2.0 * n * conv[idx]
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # ---- exact sub‑gradient of the active entry ----
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        # derivative of max entry = 2 * a_{idx‑i} (if inside range)
        grad[mask] = 2.0 * a[j[mask]]
        # scale for the objective 2·n·max
        grad *= 2.0 * n

        # Euclidean projection step
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm
        step = step0 / math.sqrt(it)
        a = project_simplex(a - step * direction)

        # occasional isotropic jitter to escape flat plateaux
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc

# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing (polishing)
# ----------------------------------------------------------------------
def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 300000) -> tuple[np.ndarray, float]:
    """
    Transfer a small ε from a source index to a destination index.
    Acceptance uses a geometric temperature schedule.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2   # move up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}")

        # keep the best vector for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (the evaluator will clamp/normalise as needed)
    return [float(x) for x in best_seq]
