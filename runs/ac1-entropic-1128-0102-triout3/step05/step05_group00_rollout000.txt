"""
Multi‑scale optimisation for the self‑convolution peak problem.

The algorithm searches for a non‑negative step‑function (a probability
vector) that minimises

    2·n·max(conv(a,a))

which is exactly the value returned by `evaluate_sequence` when the
vector is normalised to sum = 1 (the objective is scale‑invariant).

Key ideas
---------

* **Warm‑start across sizes** – we start with a small sequence,
  optimise it thoroughly, then linearly up‑sample the best distribution
  to a larger length and continue optimisation.  Larger lengths have
  more degrees of freedom and can achieve a lower peak.

* **Adam on a smooth ℓ∞ surrogate** – the true max is approximated by a
  log‑sum‑exp (soft‑max) with a temperature λ that is decreased
  gradually.  Adam’s adaptive step‑sizes accelerate convergence.

* **Enforced symmetry** – most good solutions are symmetric
  (a_i = a_{n‑1‑i}); after each update we optionally symmetrise and then
  project onto the probability simplex.

* **Targeted simulated‑annealing polishing** – after every Adam phase
  a cheap mass‑transfer annealer moves a small ε from one index to
  another, accepting uphill moves with an exponentially decaying
  temperature.  This helps escape tiny plateaus.

The routine respects the given time budget (default 1000 s) and returns
the best sequence found as a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights – numerically stable
    weights = np.exp((conv - max_val) / lam)
    weights_sum = weights.sum()
    if weights_sum == 0.0:
        # should not happen, but protect against division by zero
        weights = np.ones_like(weights) / weights.size
    else:
        weights /= weights_sum

    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a handful of plausible starting vectors from a base.
    All vectors are normalised to sum = 1.
    """
    variants = []

    # 1. the base itself (typically a warm‑start)
    variants.append(base_seq.copy())

    n = base_seq.shape[0]

    # 2. sinusoidal shape (positive half‑wave)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)  # all positive
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3. triangular (ramp up then down)
    tri = np.concatenate([np.linspace(1, n // 2, n // 2, endpoint=False),
                          np.linspace(n // 2, 1, n - n // 2)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4. a random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants

def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed ℓ∞ objective.

    Returns the best vector encountered (on the simplex) and its exact
    objective value ``2·n·max(conv)``.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 0.5          # initial λ ≈ lam_factor * max(conv)
    lam_decay = 0.9
    lam_decay_interval = 500

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # exact objective (∑a = 1)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ (never let it become zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size (larger at start, decays ~1/√it)
        base_step = 0.8 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry (helps a lot for many lengths)
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # project back onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealer.

    Moves a small ε from a source index to a destination index.
    Accepts uphill moves with probability exp(-Δ/ T).  Returns the best
    sequence found (if any) and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 15 % of the mass at source
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
