"""
Improved search for a low‑score step function.

The core routine is a *mirror‑descent Adam* optimizer that works in the
log‑space of the simplex (i.e. a_i = exp(z_i)/∑exp(z)).  The gradient of the
objective  max_k (a⋆a)_k is obtained by a hard‑max sub‑gradient:
the active anti‑diagonals receive equal weight and the corresponding
gradient is 2·(a_rev ⋆ w).  Adam‑style first‑ and second‑moment
estimates are applied to the log‑variables, then the distribution is
re‑normalised, which keeps the iterates on the probability simplex while
adapting the step size per coordinate.

A short multi‑size scan (few hundred iterations each) finds a promising
length n and an initial candidate.  The remaining budget is spent on a
long mirror‑descent run followed by a final exponential‑gradient refinement
(the same update as in the reference solution).  All steps respect the
non‑negativity and normalisation requirements of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return the maximum entry of the linear convolution a⋆a and the full conv."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Implements   g = 2·(a_rev ⋆ w)   via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    # length of conv2 = len(w) + n - 1 = (2n‑1) + n‑1 = 3n‑2
    start = n - 1
    return 2.0 * conv2[start:start + n]


# ----------------------------------------------------------------------
# Mirror‑descent Adam on the log‑simplex using the *hard* max gradient
# ----------------------------------------------------------------------
def _adam_mirror_optimize(a0: np.ndarray,
                          max_iters: int,
                          deadline: float,
                          alpha_start: float = 0.5,
                          alpha_end: float = 0.01,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps_mask: float = 1e-12,
                          step_clip: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in log‑space (mirror descent) for the exact max‑convolution.
    Returns the best point seen (as a simplex vector) and its score
    ``2·n·max(conv)``.
    """
    n = a0.shape[0]
    # Pre‑compute FFT length for the gradient step (depends only on n)
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    # Initialise log‑variables; add a tiny epsilon to avoid log(0)
    tiny = 1e-20
    log_a = np.log(a0 + tiny)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_score = float("inf")
    best_a = None

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # Recover the simplex vector
        a = np.exp(log_a)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # Evaluate objective
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- hard‑max sub‑gradient ----
        mask = conv >= max_val - eps_mask
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum
        g = _gradient_from_weights(a, w, fft_len_grad)          # ∂f/∂a

        # Convert to gradient w.r.t. log‑variables:
        #   ∂f/∂z = a · (g − ⟨g,a⟩)
        inner = np.dot(g, a)
        grad_log = a * (g - inner)

        # Adam moments (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Learning‑rate schedule (geometric decay)
        tau = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** tau

        step = alpha * m_hat / (np.sqrt(v_hat) + 1e-12)

        # Update log‑variables (clipped to avoid overflow/underflow)
        log_a = log_a - np.clip(step, -step_clip, step_clip)

    # In case the loop never entered (deadline already passed)
    if best_a is None:
        best_a = np.exp(log_a)
        best_a = _as_simplex(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponential‑gradient (mirror descent) – kept from the reference code
# ----------------------------------------------------------------------
def _exp_gradient_optimize(a0: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps_mask: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent with a decreasing step‑size.
    Uses the exact max‑convolution sub‑gradient.
    """
    n = a0.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    a = a0.copy()
    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        mask = conv >= max_val - eps_mask
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric schedule for η
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # keep exp() stable
        a = a * np.exp(upd)

        # renormalise to simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

# Candidate construction utilities

def _upsample(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old vector (simplex) to a new length."""
    if old.shape[0] == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.shape[0])
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, old)
    return _as_simplex(up)


def _make_initial_pool(n: int, global_best: np.ndarray | None,
                       prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Construct a modest pool of diverse starting points for length ``n``."""
    pool = []

    # up‑sample the current global best (if any)
    if global_best is not None:
        pool.append(_upsample(global_best, n))

    # reuse the previously discovered construction if it exists
    if prev_seq is not None:
        pool.append(_upsample(prev_seq, n))

    # deterministic shapes
    pool.append(_as_simplex(np.full(n, 1.0 / n)))               # uniform
    idx = np.arange(n)
    tri = np.minimum(idx + 1, n - idx)                       # symmetric triangle
    pool.append(_as_simplex(tri))
    if n > 1:
        hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))  # Hann window
        pool.append(_as_simplex(hann))

    # random Dirichlet draws
    for _ in range(2):
        pool.append(_as_simplex(np.random.dirichlet(np.ones(n))))

    # sparse random spikes (3‑5 non‑zero entries)
    for _ in range(2):
        spikes = np.zeros(n)
        k = np.random.randint(3, 6)
        pos = np.random.choice(n, size=k, replace=False)
        spikes[pos] = np.random.rand(k) + 0.1
        pool.append(_as_simplex(spikes))

    return pool

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the score used by
    ``evaluate_sequence``.  The method consists of:

    1. a quick multi‑scale scan (few hundred iterations each) to pick a
       promising length ``n`` and an initial vector;
    2. a long mirror‑descent (Adam in log‑space) optimisation on that pair;
    3. a final exponential‑gradient refinement.

    The function respects the global time budget (default 1000 s) and prints
    intermediate scores.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Try to load the previously discovered construction if it exists
    # ------------------------------------------------------------------
    prev_seq = None
    try:
        prev_seq = np.array(height_sequence_1, dtype=float)
        prev_seq = _as_simplex(prev_seq)
    except Exception:
        prev_seq = None

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – cheap scan over a handful of lengths
    # ------------------------------------------------------------------
    scan_lengths = [1200, 1800, 2400, 3000, 3600]   # modest set
    for n in scan_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Scan stage – n = {n}")

        pool = _make_initial_pool(n, best_seq, prev_seq)

        # allocate a very small budget per candidate in this phase
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # we target roughly 0.05 s per candidate for the scan
        iters_per_cand = max(2000, int(0.05 / (n * 8e-6)))  # crude heuristic

        for cid, cand in enumerate(pool, start=1):
            if time.time() >= deadline:
                break
            print(f"   [{time.time() - start_time:5.2f}s]  cand {cid}/{len(pool)} – iters={iters_per_cand}")

            a_opt, score_opt = _adam_mirror_optimize(
                cand,
                max_iters=iters_per_cand,
                deadline=deadline,
                alpha_start=0.3,
                alpha_end=0.01,
                beta1=0.9,
                beta2=0.999,
                eps_mask=1e-12,
            )
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                best_n = n
                print(f"       *** NEW GLOBAL BEST (scan) – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If the scan did not produce anything, fall back to a default size
    # ------------------------------------------------------------------
    if best_seq is None:
        best_n = 1800
        best_seq = _as_simplex(np.full(best_n, 1.0 / best_n))
        print(f"[{time.time() - start_time:6.2f}s] No good scan result – using default n={best_n}")

    # ------------------------------------------------------------------
    # Phase 2 – long mirror‑descent optimisation on the chosen length
    # ------------------------------------------------------------------
    n = best_n
    remaining = deadline - time.time()
    if remaining > 0:
        # rough estimate of iteration time for this size
        est_iter = 8e-6 * n + 2e-4
        max_iters = int(min(400000, max(50000, remaining / est_iter)))
        print(f"[{time.time() - start_time:6.2f}s] Long mirror‑descent – n={n}, iters={max_iters}")

        a_refined, score_refined = _adam_mirror_optimize(
            best_seq,
            max_iters=max_iters,
            deadline=deadline,
            alpha_start=0.5,
            alpha_end=0.02,
            beta1=0.9,
            beta2=0.999,
            eps_mask=1e-12,
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = a_refined.copy()
            print(f"   post‑mirror best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final exponential‑gradient refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        # again estimate a sensible number of EG iterations
        est_iter = 8e-6 * n + 2e-4
        eg_iters = int(min(400000, max(80000, remaining / est_iter)))
        print(f"[{time.time() - start_time:6.2f}s] Final EG refinement – n={n}, iters={eg_iters}")

        a_final, score_final = _exp_gradient_optimize(
            best_seq,
            max_iters=eg_iters,
            deadline=deadline,
            eta_start=0.2,
            eta_end=5e-4,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"   FINAL BEST – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety checks & final reporting
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200)

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = float('nan')
    print(f"[{time.time() - start_time:6.2f}s] Finished – best score = {final_score:.6f}")

    return [float(x) for x in best_seq]
