    np.random.seed(seed)
    random.seed(seed)

    deadline = start_time + budget_s - 0.5

    # generate initial candidates
    init_candidates = []
    # uniform
    init_candidates.append(np.full(n, 1.0 / n))
    # sin wave (half wave)
    x = np.arange(n)
    sin_vals = np.sin(np.pi * (x + 0.5) / n)  # all positive
    sin_vals /= sin_vals.sum()
    init_candidates.append(sin_vals)
    # triangular
    tri = np.concatenate([np.linspace(1, n//2, n//2, endpoint=False),
                          np.linspace(n//2, 1, n - n//2)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    init_candidates.append(tri)
    # exponential decays random alpha
    for _ in range(2):
        alpha = np.random.uniform(0.0005, 0.005)
        exp_vals = np.exp(-alpha * np.arange(n))
        exp_vals /= exp_vals.sum()

    # random Dirichlet draws

        rnd = np.random.rand(n)

"""
Simple yet powerful optimiser for the step‑function self‑convolution problem.

The optimiser works on the probability simplex (∑aᵢ = 1) because the
objective returned by ``evaluate_sequence`` is scale‑invariant.
The core of the search is a projected Adam optimiser that minimises a
smooth “soft‑max” surrogate of the ℓ∞‑norm of the convolution.
A multi‑scale schedule is used: a small problem is solved first,
its solution is linearly interpolated (upsampled) to a larger size and
refined again.  The last (largest) size receives the bulk of the time
budget and is finalised by a cheap mass‑transfer simulated‑annealing
polish.  All iterates are forced to be symmetric (a = a[::-1]) which
is natural for the autocorrelation objective and halves the effective
search space.

The routine respects the user supplied ``budget_s`` (default 1000 s) and
returns a plain ``list[float]`` – the evaluator will normalise and clamp
the sequence automatically.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex ∑v = 1, v ≥ 0."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2n‑1
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate
        L_λ(a) = λ·log ∑_k exp(conv_k / λ)
    with respect to a (∑a = 1).  The gradient is derived via the
    chain rule and computed by a single convolution.
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights_sum = weights.sum()
    if weights_sum == 0.0:
        weights = np.full_like(weights, 1.0 / weights.size)
    else:
        weights /= weights_sum

    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    # length of the full convolution: (2n‑1) + n – 1 = 3n‑2
    full = _conv_fft(weights, a_rev)
    start = n - 1                     # the slice that corresponds to ∂L/∂a_i
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth ℓ∞ surrogate)
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best vector seen and its exact objective
        score = 2·n·max(conv)   (because we keep ∑a = 1).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 1.0                 # start with almost exact max
    lam_decay = 0.85
    lam_decay_interval = 500         # decay every 500 iterations
    lam_min = 1e-12

    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # decrease λ slowly – never let it become zero
        if it % lam_decay_interval == 0:
            lam_factor *= lam_decay
        lam = max(max_b * lam_factor, lam_min)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size (the 0.6 factor works well in practice)
        base_step = 0.6 / math.sqrt(it + 1.0)

        # update before projection
        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry enforcement (helps a lot for the autocorrelation problem)
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # project back onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 1e-1,
                   end_temp: float = 1e-6,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing style polish: move a small ε from a
    source index to a destination index.  The temperature is exponentially
    decayed, the acceptance rule is Metropolis.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move at most 15 % of the source mass (randomised)
        eps = a[i] * rng.random() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # evaluate the exact objective (fast FFT based)
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Shape generators for restarts
# ----------------------------------------------------------------------


def _shape_variants(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a small collection of plausible starting vectors of length n.
    All vectors are normalised to sum = 1.
    """
    variants = []

    # uniform
    variants.append(np.full(n, 1.0 / n))

    # half‑sine (positive on (0,π))
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # triangular (ramp up then down)
    half = n // 2
    tri = np.concatenate([np.linspace(1, half, half, endpoint=False),
                          np.linspace(half, 1, n - half)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # exponential families with random decay rates
    for _ in range(2):
        alpha = rng.uniform(0.0003, 0.003)
        exp_vals = np.exp(-alpha * np.arange(n))
        exp_vals /= exp_vals.sum()
        variants.append(exp_vals)

    # random Dirichlet draws (α = 1)
    for _ in range(2):
        rnd = rng.random(n)
        rnd /= rnd.sum()
        variants.append(rnd)

    return variants

    start_time = time.time()

    best_score = float('inf')

    # ------------------------------------------------------------------
