"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm
---------
* **Projected Adam** – Adam optimiser with Euclidean projection onto
  the simplex (x≥0, Σx=1).  The gradient is the (averaged) sub‑gradient
  of *all* convolution entries attaining the current maximum, which
  yields a more accurate descent direction than using a single index.
* **Multi‑stage refinement** – several short Adam runs on a modest size
  (n≈1024) with random restarts, then successive up‑sampling (linear
  interpolation) to larger dimensions (≈4 k → 12 k) while continuing the
  optimisation.  Larger n gives a finer discretisation of the optimal
  continuous shape.
* **Final simulated‑annealing** – a cheap mass‑transfer annealer that
  explores the neighbourhood of the best vector found by gradient
  descent.  The remaining time budget is used here.
* **Time‑budget aware** – the routine keeps track of the wall‑clock and
  stops before the supplied budget expires.  The best sequence seen so
  far (as a plain Python list) is returned.

The combination of a more accurate sub‑gradient, Adam’s adaptive step
sizes and a longer optimisation time typically improves the bound
obtained by the previous Nesterov‑style gradient descent, often
reaching scores < 1.503 on the benchmark.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT (fast)
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a.
# If several entries attain the maximum we average their gradients.
# ------------------------------------------------------------
def _subgradient_all_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return (averaged) sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    # indices that are (almost) maximal – tolerance avoids numerical noise
    idxs = np.where(np.isclose(conv, max_val, atol=1e-12))[0]

    grad = np.zeros_like(a, dtype=float)

    # vectorised contribution of one active index k
    for k in idxs:
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        # derivative of conv_k w.r.t a_i is a_j + a_i (where defined)
        # a_j term:
        grad[mask] += a[j[mask]]
        # a_i term:
        grad[mask] += a[i[mask]]
    # average over all active indices
    if len(idxs) > 0:
        grad /= len(idxs)
    return grad


# ------------------------------------------------------------
# Adam optimiser on the simplex
# ------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr0: float = 0.15,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for  F(a)=2·n·max(conv(a))   (sum(a)=1).
    Returns the best vector seen and its (un‑clamped) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val          # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={t:,}, score={cur_score:.6f}")

        # sub‑gradient (averaged over all max entries)
        grad = _subgradient_all_max(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing (mass‑transfer moves) for fine‑tuning
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale SA that moves ε from a high‑mass index to another.
    Keeps the globally best sequence found.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

        lr0 = np.random.uniform(0.10, 0.25)

        a_opt, score = _run_adam(
            a0, n0, slice_deadline, start_time,
            lr0=lr0,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: restart={r}, score={score:.6f}")

    # fallback – unlikely
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 1 – upsample to n≈4096 and longer Adam optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(30.0, 0.35 * remaining)          # about one third of what is left
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_adam(
        a1, n1, grad_deadline1, start_time,
        lr0=0.18,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upsample to n≈12288 and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(30.0, 0.30 * remaining)           # another third of what remains
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_adam(
        a2, n2, grad_deadline2, start_time,
        lr0=0.12,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement with the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-7,
            steps=400_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # optionally report the final (un‑clamped) score for information
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
