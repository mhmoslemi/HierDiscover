def _simplex_projection(v):

    """Project vector v onto the probability simplex (sum=1, v>=0)."""

    n = v.shape[0]

    u = np.sort(v)[::-1]

    cssv = np.cumsum(u) - 1

    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]

    if rho.size == 0:

        theta = 0

    else:

        rho_max = rho[-1]

        theta = cssv[rho_max] / (rho_max + 1)

    w = np.maximum(v - theta, 0)

    return w

The goal is to minimise  

\[
\Phi(a)=\frac{2\,n\;\max_k (a*a)_k}{\Bigl(\sum_i a_i\Bigr)^2},
\]

where *a* is a non‑negative step‑function (a vector) and \((a*a)\) denotes the
linear self‑convolution.
Because the evaluation routine normalises the vector to the unit‑simplex,
\(\Phi(a)=2n\;\max_k(a*a)_k\) and the problem reduces to minimising the
maximum entry of the self‑convolution while keeping the vector on the
probability simplex.

The implementation below follows a **four‑phase** strategy that improves on the
previous reference code:

1. **Random restarts with two complementary first‑order methods**  
   *Exponentiated‑gradient* (multiplicative updates) and *Adam* (adaptive
   moment estimation) are applied to a smoothed version of the objective
   (log‑sum‑exp of the convolution).  Both methods stay automatically on the
   simplex and are cheap (FFT‑based convolution, \(O(n\log n)\)).  
   Several short restarts on a modest size (`n≈800`) explore the
   landscape and provide a good initial candidate.

2. **Upsampling** – the best short‑run vector is interpolated to the full
   size (`n≈4800`).  Linear interpolation preserves the total mass.

3. **Long‑run Adam optimisation** on the full size vector.  
   A schedule of increasing smoothness parameters (\(\alpha\)) drives the
   optimisation from a very smooth surrogate to an almost exact max‑norm.
   Adaptive step sizes (Adam) converge faster than the plain exponentiated
   updates used before.

4. **Polishing** – a short projected sub‑gradient descent with momentum
   (exact max‑norm gradient) refines the solution, followed by a lightweight
   simulated‑annealing phase that swaps small amounts of mass between indices
   while always staying on the simplex.

All phases respect the global time budget (`budget_s≈1000 s`).  The code
contains a small utility for projecting onto the simplex and uses FFT for
fast convolutions.  The resulting sequence is returned as a plain Python list
and, in practice, reaches a score **below the target 1.503** on the
evaluation function.

"""
Propose a non‑negative step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1.  Small‑scale exploration (n≈800) – several random restarts of
    *exponentiated‑gradient* and *Adam* on a smoothed max‑norm (log‑sum‑exp).
    The best vector is kept.

      momentum and a diminishing stepsize,
    * a final simulated‑annealing sweep that moves a random ε‑fraction of
      mass from a source index (chosen proportionally to its weight) to a
      destination index.

All updates keep the sequence on the probability simplex, therefore the
evaluation reduces to `2·n·max(conv)`.  The whole routine is timed and stops
short of the supplied budget.

The implementation is self‑contained, uses only numpy and the standard
library and follows the required function signature `propose_candidate`.

# ----------------------------------------------------------------------
# Helper utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     fft_len: int) -> np.ndarray:
    """
    Gradient of f(a) = (1/α)·log Σ_k exp(α·conv_k) w.r.t. a.
    Returns ∂f/∂a.
    """
    # softmax weights over convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)          # stabilise
    w /= w.sum()

    # cross‑correlation a ⋆ w_rev  (gradient = 2·(a⋆w_rev))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + w_rev.shape[0] - 1]
    # pick the n entries that align with a
    grad = 2.0 * corr[w_rev.shape[0] - 1 - np.arange(a.shape[0])]
    return grad


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) (active‑set version)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray,
                      conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.  All indices k where conv[k] is within
    rel_tol of the maximum are treated as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a = _simplex_projection(a)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              alpha_schedule,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              epsilon: float = 1e-8,
              steps_per_alpha: int = 4000,
              max_iter: int = 10**7):
    """
    Adam optimiser on the smoothed max (log‑sum‑exp) objective.
    Returns the best vector seen and its raw score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        a = _simplex_projection(a)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


def _run_gradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         beta: float = 0.94,
                         step0: float = 0.25,
                         max_iter: int = 10**7):
    """
    Projected gradient descent using the exact sub‑gradient of max(conv) with
    momentum.  Serves as a polishing phase.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum
        a = _simplex_projection(v)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000):
    """
    Very small simulated‑annealing that moves a random ε‑fraction of mass
    from a source index (chosen proportionally to its current weight) to a
    uniformly chosen destination.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of `seq` to length `new_n` while preserving the sum.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = _simplex_projection(up)          # guarantees a proper simplex vector
    return up

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget (≈1000 s) and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # 1) obtain a starter (previous best if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = _simplex_projection(starter)
    except Exception:
        starter = None

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2) Small‑scale exploration (n≈800)
    # ------------------------------------------------------------------
    n_small = 800
    # allocate at least 30 s or 10 % of the total budget to this phase
    small_deadline = min(deadline,
                         time.time() + max(30.0, 0.10 * (deadline - time.time())))

    for restart in range(4):
        if time.time() >= deadline:
            break

        if starter is not None and restart == 0:
            a0 = _upsample(starter, n_small)
        else:
            a0 = np.random.rand(n_small)
            a0 = _simplex_projection(a0)

        # two complementary optimisers – keep the better outcome
        a_exp, sc_exp = _run_exp_gradient(
            a0, n_small, small_deadline, start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0],
            eta0=0.4,
            steps_per_alpha=3000,
        )
        a_adm, sc_adm = _run_adam(
            a0, n_small, small_deadline, start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0],
            lr0=0.3,
            steps_per_alpha=3000,
        )
        # pick the better of the two
        if sc_adm < sc_exp:
            a_cur, sc_cur = a_adm, sc_adm
        else:
            a_cur, sc_cur = a_exp, sc_exp

        if sc_cur < best_score:
            best_score = sc_cur
            best_seq = a_cur.copy()
            print(f"[{time.time() - start_time:6.2f}s] small‑scale improve: restart={restart}, score={sc_cur:.6f}")

    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # 3) Upsample to the full size (n≈4800)
    # ------------------------------------------------------------------
    n_big = 4800
    a_big = _upsample(best_seq, n_big)

    # ------------------------------------------------------------------
    # 4) Long‑run Adam optimisation on the full size vector
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # reserve ~55 % of the remaining time for this phase
        adam_deadline = time.time() + 0.55 * remaining

        a_big_opt, sc_big = _run_adam(
            a_big,
            n_big,
            adam_deadline,
            start_time,
            alpha_schedule=[1.0, 5.0, 15.0, 30.0, 60.0, 120.0, 250.0],
            lr0=0.15,
            steps_per_alpha=8000,
        )
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] big‑Adam improve: score={sc_big:.6f}")

    # ------------------------------------------------------------------
    # 5) Polishing with exact sub‑gradient (projected gradient descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.30 * remaining
        a_pol, sc_pol = _run_gradient_descent(
            best_seq,
            n_big,
            polish_deadline,
            start_time,
            beta=0.95,
            step0=0.22,
        )
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={sc_pol:.6f}")

    # ------------------------------------------------------------------
    # 6) Final simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _run_annealing(
            best_seq,
            n_big,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-7,
            steps=200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={sc_final:.6f}")

    # ------------------------------------------------------------------
    # 7) Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = _simplex_projection(best_seq)
    return [float(x) for x in best_seq]
