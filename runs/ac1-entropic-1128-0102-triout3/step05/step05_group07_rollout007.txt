"""
Propose a low‑score step‑function sequence.

The algorithm proceeds in three stages:

1. **Exploratory parametric sweep** – generate a diverse set of
   analytically‑parameterised shapes (uniform, triangle, Hann,
   exponential, Gaussian, power‑law, symmetric Beta) for a fixed length
   (default 1800).  Each candidate is normalised to the probability simplex
   (sum = 1) and its objective
       score = 2·n·max(conv(a,a))
   (the denominator sum(a)² cancels out) is evaluated with a fast FFT
   convolution.  The best candidate is retained.

2. **Heavy local optimisation** – the best candidate from step 1 is
   refined with a long Adam optimisation on a smooth‑max surrogate of the
   max‑norm, followed by an exponential‑gradient (mirror‑descent) phase
   that directly uses the true max‑convolution.  The number of iterations
   is chosen dynamically from the remaining time budget.

3. **Polishing** – a final exponential‑gradient run with a very slow
   decay of the step‑size is performed.  The resulting sequence is
   returned as a plain list of floats.

All operations respect the non‑negativity and simplex constraints, and
the routine stops before the supplied time budget (default 1000 s) to
avoid a timeout penalty.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution (real)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k * conv_k w.r.t. `a`.
    Implements g = 2 * (a_rev * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    alpha: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.05,
    temp_end: float = 0.0005,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best vector observed and its corresponding score
    (2·n·maxconv).
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # best observed
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current objective
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # smooth‑max weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient of the smooth‑max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    eta_start: float = 0.8,
    eta_end: float = 0.001,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser that directly minimises
    the supremum norm of the convolution.
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask of active anti‑diagonals (within eps of the max)
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient (exact max‑norm)
        g = _gradient_from_weights(a, w, fft_len_grad)

        # decreasing step size (geometric schedule)
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update with clipping
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)      # avoid overflow/underflow
        a = a * np.exp(upd)

        # renormalise to simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

# Shape generators (always return a vector on the simplex)
def _as_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

    idx = np.arange(n)
    tri = np.minimum(idx + 1, n - idx)

    idx = np.arange(n)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))

            if score_eg < best_score:
                best_score = score_eg
                best_seq = a_eg.copy()
                print(f"[{time.time() - start_time:6.2f}s] New best after EG -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional extra polishing (small random perturbations + EG)
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # tiny random perturbation of the current best
        noise = np.random.randn(n) * 0.02
        pert = best_seq + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_simplex(pert)

        # short EG run on perturbed point
        remaining = deadline - time.time()
        est_iter_time = 8e-4 * (n / 1200.0) + 1e-4
        iters = int(min(60000, max(5000, remaining / est_iter_time)))
        a_eg, score_eg = _exp_gradient_optimize(
            pert,
            max_iters=iters,
            deadline=deadline,
            eta_start=0.5,
            eta_end=0.0005,
        )
        if score_eg < best_score:
            best_score = score_eg
            best_seq = a_eg.copy()
            print(f"[{time.time() - start_time:6.2f}s] Polishing improvement -> {best_score:.6f}")
        else:
            # no improvement – break to avoid endless looping
            break

    # ------------------------------------------------------------------
    # Final sanity checks and reporting
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = float("nan")
    print(f"[{time.time() - start_time:6.2f}s] Finished – final score = {final_score:.6f}")

    return [float(x) for x in best_seq]
