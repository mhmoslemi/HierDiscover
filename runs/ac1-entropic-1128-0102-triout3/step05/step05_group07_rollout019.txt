"""
Simple‑yet‑robust optimiser for the “step‑function” benchmark.

The routine mixes three complementary strategies inside the fixed 1000 s
budget:

1. **Diverse initialisation** – a handful of uniform, Dirichlet‑drawn,
   up‑sampled previous best and simple exponential decays.
2. **Two‑phase continuous optimisation** – a short Adam run on a smooth‑max
   surrogate followed by a long exponential‑gradient (mirror‑descent)
   stage that works with the exact max‑convolution.
3. **Stochastic fine‑tuning** – repeated random perturbations of the
   current best vector, each refined by a fresh EG pass, plus a cheap
   hill‑climbing (random mass‑transfer) phase.

All steps respect the simplex (non‑negative entries, sum = 1) and use
FFT‑based convolution for speed.  The function returns the best sequence
found before the time‑budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k * conv_k(a,a)   w.r.t. ``a``.
    Implements   g = 2·(a_rev * w)   via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _score(a: np.ndarray) -> float:
    """Exact evaluation used during the search (no clipping)."""
    n = a.shape[0]
    max_b, _ = _conv_max(a)
    s = a.sum()
    if s == 0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.07,
                   temp_end: float = 0.001) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of the objective."""
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max over the convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        g = _gradient_from_weights(a, w, fft_len_grad)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser that uses the *hard* max
    of the convolution (sub‑gradient of the L∞‑norm).
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # indices where the max is (approximately) attained
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric decay of the step size
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # avoid overflow/underflow
        a = a * np.exp(upd)

        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score


# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                max_iters: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: repeatedly transfer a tiny amount of
    mass between two random coordinates and keep the move if it improves the
    objective.  Occasionally a worsening move is accepted to escape shallow
    plateaus.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # random direction of transfer
        src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)
        if best_a[src] == 0.0:
            continue

        # transfer at most 5 % of the source mass
        frac = np.random.rand() * 0.05
        delta = best_a[src] * frac

        new_a = best_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        sc = _score(new_a)
        if sc < best_sc:
            best_sc = sc
            best_a = new_a
        else:
            # weak simulated‑annealing acceptance
            temp = 1e-3 * (1.0 - it / max_iters) + 1e-6
            if np.random.rand() < math.exp((best_sc - sc) / temp):
                best_a = new_a

    return best_a, best_sc


# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 0, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative step heights that minimises

        score = 2·n·max(conv(a,a)) / (sum(a)²)

    inside the given time budget.  The routine returns the best sequence
    (as a plain list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Helper to normalise onto the simplex
    # ------------------------------------------------------------------
    def _as_simplex(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        return v / s if s > 0 else np.full_like(v, 1.0 / v.shape[0])

    # ------------------------------------------------------------------
    # Try to import the previously discovered construction (if any)
    # ------------------------------------------------------------------
    prev_seq = None
    try:
        prev_seq = np.array(height_sequence_1, dtype=float)
        prev_seq = _as_simplex(prev_seq)
    except Exception:
        prev_seq = None

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Size schedule – concentrate most effort on a medium‑size problem,
    # but also try a couple of larger instances.
    # ------------------------------------------------------------------
    sizes = [1800, 3600]          # two sizes keep the total runtime reasonable

    for n in sizes:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:6.2f}s] Working on size n = {n}")

        # --------------------------------------------------------------
        # Build a pool of diverse starting vectors
        # --------------------------------------------------------------
        pool = []

        # uniform
        pool.append(_as_simplex(np.full(n, 1.0 / n)))

        # up‑sample previous global best (if any)
        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, best_seq)
            pool.append(_as_simplex(up))

        # up‑sample the earlier construction (if any)
        if prev_seq is not None:
            if prev_seq.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, prev_seq.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                up = np.interp(x_new, x_old, prev_seq)
                pool.append(_as_simplex(up))
            else:
                pool.append(prev_seq.copy())

        # a few Dirichlet draws with low concentration (more variance)
        for _ in range(3):
            pool.append(_as_simplex(np.random.dirichlet(np.ones(n) * 0.5)))

        # simple exponential decays with different slopes
        for decay in (1.0 / n, 2.0 / n, 5.0 / n):
            idx = np.arange(n)
            exp_vec = np.exp(-decay * idx)
            pool.append(_as_simplex(exp_vec))

        # --------------------------------------------------------------
        # Optimise each candidate (Adam → EG)
        # --------------------------------------------------------------
        for cid, cand in enumerate(pool, 1):
            if time.time() >= deadline:
                break

            remaining = deadline - time.time()
            # cheap estimate of time per iteration for this n
            est_iter_time = 8e-4 * (n / 1200.0) + 1e-4
            # allocate a modest number of iterations – enough to feel the basin
            max_iters = int(min(120000, max(20000, remaining / est_iter_time)))
            if max_iters <= 0:
                break

            # short Adam phase
            a_adam, _ = _adam_optimize(
                cand.copy(),
                max_iters=int(0.35 * max_iters),
                deadline=deadline,
                alpha=0.12,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                temp_start=0.06,
                temp_end=0.001,
            )

            # exponential‑gradient refinement
            a_eg, _ = _exp_gradient_optimize(
                a_adam.copy(),
                max_iters=int(0.65 * max_iters),
                deadline=deadline,
                eta_start=1.0,
                eta_end=0.001,
            )

            sc = _score(a_eg)
            if sc < best_score:
                best_score = sc
                best_seq = a_eg.copy()
                print(f"   NEW GLOBAL BEST (after EG) – score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Additional stochastic fine‑tuning on the current best of this size
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            remaining = deadline - time.time()
            est_iter_time = 8e-4 * (n / 1200.0) + 1e-4
            passes = 0
            while remaining > 2.0 and passes < 5:
                # random perturbation of the current best
                pert = best_seq + 0.02 * np.random.randn(n)
                pert = _as_simplex(pert)

                # run a fresh EG pass (more aggressive start)
                a_pass, _ = _exp_gradient_optimize(
                    pert,
                    max_iters=int(min(80000, remaining / est_iter_time)),
                    deadline=deadline,
                    eta_start=0.5,
                    eta_end=0.0003,
                )
                sc = _score(a_pass)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_pass.copy()
                    print(f"   IMPROVEMENT via perturb‑pass – score = {best_score:.6f}")

                remaining = deadline - time.time()
                passes += 1

            # final cheap hill climbing
            if time.time() < deadline:
                remaining = deadline - time.time()
                iters_hc = int(min(60000, remaining / est_iter_time))
                if iters_hc > 0:
                    a_hc, sc_hc = _hill_climb(best_seq.copy(), iters_hc, deadline)
                    if sc_hc < best_score:
                        best_score = sc_hc
                        best_seq = a_hc.copy()
                        print(f"   AFTER HILL‑CLIMB – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # final sanity checks before returning
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    # report the final score with the provided evaluation routine
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = float('nan')
    print(f"[{time.time() - start_time:6.2f}s] Finished – final score = {final_score:.6f}")

    return [float(x) for x in best_seq]
