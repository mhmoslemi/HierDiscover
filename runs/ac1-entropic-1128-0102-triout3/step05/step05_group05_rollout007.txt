def simplex_projection(v):
    # ensures v >=0 and sum = 1
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u > (sv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (sv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a better step‑function by directly minimising the
L∞‑norm of the self‑convolution.

The problem is
    minimise   max_k (a * a)_k
    subject to  a_i ≥ 0 ,  Σ a_i = 1 .

The objective is scale‑invariant, therefore we work on the simplex.
We start from the best known sequence (the global variable
`height_sequence_1` if it exists) and apply a mirror‑descent /
exponentiated‑gradient scheme on the hard‑max sub‑gradient:

    • Compute b = a∗a and locate the indices where b attains its maximum.
    • The sub‑gradient of max(b) w.r.t. a is
          g_i = 2 Σ_{k∈argmax} a_{k‑i}   (zero when the index is out of range).
      This can be obtained by a single convolution of a binary mask of the
      maximal indices with the reversed vector a.
    • Perform a multiplicative update a_i ← a_i·exp(‑η·g_i) and renormalise
      to stay on the simplex.
    • η (the step size) is decreased slowly; occasional random jitter is
      added if no improvement is seen for many iterations.

The loop stops when the time budget (≈1000 s) is exhausted.
The best sequence encountered is returned as a plain Python list,
ready for `evaluate_sequence`.
"""

import time
import numpy as np

def _subgradient_of_max_convolution(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) w.r.t. a.
    The gradient is 2 * (mask ∗ a_rev)[::-1] where mask has 1 at all
    positions attaining the maximum of `conv`.
    """
    n = a.size
    max_val = conv.max()
    # mask of maximal entries (binary)
    mask = np.zeros_like(conv, dtype=float)
    # use a tolerance to catch all maxima that are numerically equal
    mask[np.isclose(conv, max_val, atol=1e-15)] = 1.0
    # convolution with reversed a gives the required sum of a_{k-i}
    grad = 2.0 * np.convolve(mask, a[::-1], mode='valid')[::-1]
    return grad


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (probability distribution)
    that improves on the current best bound.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    time_budget = budget_s - 0.1

    # ------------------------------------------------------------------
    # 1.  Initialise from the previously found sequence if possible.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim != 1 or init.size == 0:
            raise ValueError
        # apply the same clipping that evaluate_sequence does
        init = np.clip(init, 0.0, 1000.0)
        if init.sum() < 0.01:
            raise ValueError
        a = init / init.sum()
    except Exception:
        # fallback: a modest uniform vector
        n = 2000
        a = np.full(n, 1.0 / n, dtype=float)

    n = a.size
    best_a = a.copy()
    best_score = float('inf')
    # evaluate the starting point once
    try:
        best_score = evaluate_sequence(best_a.tolist())
        print(f"[0.00s] initial score {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2.  Mirror‑descent loop (exponentiated‑gradient on the hard max).
    # ------------------------------------------------------------------
    max_iter = 20000
    eta0 = 5.0               # initial step size for the multiplicative update
    decay = 0.9995           # exponential decay of the step size
    stagnation_patience = 2000   # iterations without improvement before jitter
    no_improve = 0

    for it in range(max_iter):
        # stop if we run out of the allotted time
        if time.time() - start_time > time_budget:
            break

        # current convolution and its maximal value
        conv = np.convolve(a, a)
        # compute the sub‑gradient of the max‑convolution
        grad = _subgradient_of_max_convolution(a, conv)

        # ----- multiplicative (exponentiated‑gradient) update -----
        eta = eta0 * (decay ** it)
        # shift exponents to avoid overflow/underflow
        exp_arg = -eta * grad
        exp_arg -= exp_arg.max()          # keep the largest exponent at 0
        a_new = a * np.exp(exp_arg)

        s = a_new.sum()
        if not np.isfinite(s) or s == 0.0:
            a_new = np.full(n, 1.0 / n, dtype=float)
        else:
            a_new = a_new / s

        # ----- evaluate the new candidate -----
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            no_improve = 0
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] iter {it:,} improved score to {best_score:.6f}")
        else:
            no_improve += 1

        a = a_new

        # ----- occasional random jitter to escape plateaus -----

            a = np.clip(a, 0.0, None)

            if tot < 1e-12 or not np.isfinite(tot):

    # ------------------------------------------------------------------

          f"with length {len(best_a)}")

    return [float(x) for x in best_a]
