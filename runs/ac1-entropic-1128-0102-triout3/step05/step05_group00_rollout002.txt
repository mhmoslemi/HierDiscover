"""
Propose a non‑negative step‑function with a low self‑convolution peak.

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1) of length 256.
   If a previous best solution (`height_sequence_1`) exists it is up‑sampled
   to this length, otherwise a random Dirichlet draw is used.

2. **Adam optimisation** on a smooth log‑sum‑exp surrogate of the ℓ∞‑norm.
   The surrogate temperature is decreased gradually.  The iterate is
   projected onto the probability simplex after each step.  This phase
   consumes roughly 60 % of the time budget and provides a high‑quality
   local minimum.

3. **Linear‑programming refinement** (alternating optimisation).
   For the current vector `a` let `M = max(a*a)`.  Solve the LP
       maximise  1ᵀb
       subject  (a * b) ≤ M,   b ≥ 0
   (the convolution `a*b` is linear in `b`).  If the optimal sum exceeds 1,
   move `a` part‑way towards the normalised LP solution.  This strictly
   reduces the maximum convolution value and is repeated until no
   improvement is possible or time runs out.

4. **Annealing polish** – a cheap mass‑transfer simulated‑annealing loop
   (move a small ε from one index to another) for the remaining seconds.

The best vector found is returned as a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (all top‑level – no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (exact, not circular)."""
    return np.convolve(a, a)


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Fast linear convolution using FFT (real inputs).
    """
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray,
                        conv: np.ndarray,
                        lam: float) -> np.ndarray:
    """
    Gradient of the smooth log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (numerically stable)
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()

    # cross‑correlation of the weight vector with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed ℓ∞ surrogate.
    Returns the best vector encountered and its exact evaluation
    ``2·n·max(conv)`` (the vector stays on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3          # λ ≈ lam_factor * max(conv)
    lam_decay = 0.7
    lam_decay_interval = 2000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(cur_max * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated annealing.
    Moves a random ε from one index to another, accepting uphill moves
    with a geometric temperature schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

        best_score = score_adam
        best_a = a_adam.copy()

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming refinement
    # ------------------------------------------------------------------
    a_lp = best_a.copy()
    while time.time() < deadline - 2.0:
        conv = _conv_self(a_lp)
        M = conv.max()
        rows = 2 * n - 1

        # Build the convolution matrix A where (A @ b) = a_lp * b
        A = np.zeros((rows, n))
        for j in range(n):
            A[j:j + n, j] = a_lp

        # Linear program: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=(0, None), method='highs')
        if not res.success:
            break

        sum_opt = res.x.sum()
        if sum_opt <= 1.0 + 1e-9:
            # No feasible increase – cannot improve further
            break

        g = res.x / sum_opt            # normalise to a probability vector
        alpha = 0.5                    # mixing coefficient
        a_lp = (1 - alpha) * a_lp + alpha * g
        a_lp = _project_simplex(a_lp)

        cur_score = 2.0 * n * _conv_self(a_lp).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_lp.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Annealing polish (uses whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_a, n, deadline, start_time,
            best_score, best_a,
            start_temp=5e-2, end_temp=1e-5, steps=200000)
        if score_final < best_score:
            best_score = score_final
            best_a = a_final.copy()

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    result = [float(x) for x in best_a]
    print(f"[{time.time() - start_time:6.2f}s] Final best score = {best_score:.6f}")
    return result
