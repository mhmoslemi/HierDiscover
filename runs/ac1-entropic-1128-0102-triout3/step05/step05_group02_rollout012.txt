"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer works on the probability simplex (∑a_i = 1) and consists of
four stages:

1. **Warm‑start** – a short stochastic search over several parametric
   families (power‑product, Gaussian, beta, cosine) to obtain a good
   initial shape for a relatively large length.
2. **Smoothed exponential‑gradient descent** – the ∞‑norm is approximated
   by a log‑sum‑exp with an increasing smoothing parameter α.
   Using a multiplicative (exponentiated) update automatically respects
   non‑negativity and the simplex constraint.
3. **Exact sub‑gradient descent** – after the smoothing phase we refine
   the solution with a classic projected sub‑gradient method that uses
   the true active convolution entry.
4. **Mass‑transfer annealing** – a final cheap Monte‑Carlo polishing
   phase that moves small ε‑mass between indices with a geometric
   temperature schedule.

All phases share the 1000 s time budget; the majority of the time is
spent on the large‑length optimisation (default length 32768).  The
routine prints progress information and finally returns a plain Python
list of non‑negative heights.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (FFT based convolutions)
# ----------------------------------------------------------------------


def fft_convolve_real(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real 1‑D arrays using an FFT of power‑of‑2 length."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 2).bit_length())
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    return np.fft.irfft(Z, size)[: n + m - 1]


def conv_self(a: np.ndarray) -> np.ndarray:
    """Exact self‑convolution a*a (length 2·n‑1)."""
    return fft_convolve_real(a, a)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def score_of(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a normalized sequence (∑a=1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for rapid warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term == 0] = 1e-12
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_cosine(n: int, power: float) -> np.ndarray:
    """Sinusoidal shape raised to a power."""
    x = np.arange(n) / (n - 1)
    a = np.sin(np.pi * x) ** power
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) exponential‑gradient descent
# ----------------------------------------------------------------------


def expgrad_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    alpha_start: float = 0.5,
                    alpha_end: float = 60.0,
                    step0: float = 0.6,
                    report_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponential) gradient descent on the smoothed objective
    f(a) = 2·n·log‑sum‑exp(α·(a*a)) / α.
    α is increased linearly from alpha_start to alpha_end.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    total_time = deadline - start_time

    while time.time() < deadline:
        it += 1
        # self‑convolution
        conv = conv_self(a)                     # length 2n‑1
        # schedule for α
        elapsed = time.time() - start_time
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, elapsed / total_time)

        # soft‑max weights
        max_c = conv.max()
        exp_vals = np.exp(alpha * (conv - max_c))   # numeric stability
        p = exp_vals / exp_vals.sum()               # sum(p)=1, length 2n‑1

        # compute term Σ_j p_{i+j} a_j via convolution of reversed p with a
        p_rev = p[::-1]
        conv_rev = fft_convolve_real(p_rev, a)      # length 3n‑2
        offset = len(p) - 1                         # = 2n‑2
        conv_term = conv_rev[offset - np.arange(n)]

        # gradient of the smoothed objective (before step scaling)
        base = a + conv_term                         # ∂/∂a_i of Σ_k p_k (a*a)_k
        grad = 2.0 * n * base

        # diminishing step size
        step = step0 / math.sqrt(it)

        # exponentiated (multiplicative) update; keeps positivity
        a = a * np.exp(-step * grad)
        a = np.maximum(a, 1e-20)                     # avoid exact zeros

        # renormalise onto simplex (actually just L1‑normalise)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional evaluation of the true (non‑smoothed) objective
        if it % report_every == 0:
            cur_sc = score_of(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] expgrad improve n={n} it={it} score={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (active‐set version)
# ----------------------------------------------------------------------


def subgrad_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 2.0,
                    report_every: int = 50) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(a*a).  The gradient is taken from the currently active
    convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        idx = conv.argmax()                      # active entry
        # gradient of the active entry
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        step = step0 / math.sqrt(it)
        a = project_simplex(a - step * grad)

        if it % report_every == 0:
            cur_sc = score_of(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-6,
              steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Randomly move a small ε of mass from one index to another.
    Accept moves that improve the true objective, otherwise accept with
    a Metropolis probability decreasing geometrically.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
