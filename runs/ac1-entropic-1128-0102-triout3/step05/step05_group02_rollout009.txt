def conv_full(x, y):
   m = len(x) + len(y) - 1
   fft_len = 1 << ((m-1).bit_length())
   X = np.fft.rfft(x, n=fft_len)
   Y = np.fft.rfft(y, n=fft_len)
   conv = np.fft.irfft(X * Y, n=fft_len)[:m]
   return conv

def smoothed_gradient_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # schedule for beta: e.g., geometric from 0.5 to 200
    betas = np.geomspace(0.5, 250.0, num=12)
    # iteration counter for step size schedule
    it = 0
    eta0 = 0.5  # base step size for exponentiated updates.
    for beta in betas:
        # inner loop until time or limit of iterations per beta
        # amount iteration per beta adapt based on remaining time and n.
        while time.time() < deadline:
            it += 1
            # compute conv
            conv = conv_self(a)
            # compute softmax weights
            max_c = conv.max()
            # stabilization using subtract max
            exp_vals = np.exp(beta * (conv - max_c))
            w = exp_vals / exp_vals.sum()
            # compute gradient
            grad_full = conv_full(w, a)  # length 3n-2
            grad = grad_full[:n]  # first n entries
            grad *= 2.0

            step = eta0 / math.sqrt(it)
            # exponentiated update
            a = a * np.exp(-step * grad)
            # renormalize
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full_like(a, 1.0/n)

            # evaluate actual score occasionally
            if it % 200 == 0:
                sc = score_of(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] smoothed improve n={n} beta={beta:.2f} it={it} score={sc:.6f}")

            # maybe break if convergence?
            # not needed

        # after inner loop for current beta, continue to next beta if time remains
        if time.time() >= deadline:
            break

    return best_a, best_sc

"""
Propose a low‑maximum self‑convolution step‑function.

The search proceeds in three stages:

1️⃣  Warm‑start – a short stochastic sweep over a few parametric families
    (product‑power, Gaussian, Beta‑shaped) provides a good initial shape.

2️⃣  Smoothed mirror‑descent – we minimise a smooth approximation of the
    ∞‑norm,
        f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k) .
    Its gradient is
        ∇f_β(a) = 2·softmax_β(a∗a)  ∗  a   (∗ = convolution).
    An exponentiated‑gradient (mirror descent w.r.t. KL) update
        a ← a·exp(−η·∇f)   ,   a ← a/∑a
    keeps the vector on the probability simplex.
    β is increased geometrically (coarse → fine) and the step size η
    follows a 1/√t schedule.

3️⃣  Cheap mass‑transfer annealing – after the gradient phase we perform a
    large number of tiny “ε‑moves” (transfer a small amount of mass from
    one index to another) with a geometric temperature schedule to polish
    the best sequence found.

The routine adapts the amount of work to the remaining time budget and
progressively upsamples the best vector to larger lengths.  The final
vector (a plain Python list) is returned; the evaluator will clamp and
normalise it as required.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x∗y (any lengths) via FFT."""
    m = len(x) + len(y) - 1
    fft_len = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:m]
    return conv


def score_of(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a∗a) for a normalised to sum = 1."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by re‑normalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for quick random warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i+0.5)*(n‑i‑0.5)  raised to power p.
    p>0 → centre‑heavy,  p<0 → edge‑heavy.
    """
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term == 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ expressed in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the unit interval."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed mirror‑descent on the simplex (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smoothed_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of the max self‑convolution using
    exponentiated‑gradient (mirror‑descent on the simplex).

    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # 1/√t step‑size schedule
    step0 = 0.7
    it = 0

    # geometric schedule for β (smoothness)
    betas = np.geomspace(0.3, 250.0, num=18)

    for beta in betas:
        # a rough budget per β (depends on remaining wall‑time)
        # we try at most ~2000 iterations for each β unless time runs out.
        max_iters_this_beta = 2000
        for _ in range(max_iters_this_beta):
            if time.time() >= deadline:
                break
            it += 1

            # ---- compute soft‑max weights ----
            conv = conv_self(a)                     # length 2n‑1
            max_c = conv.max()
            # stable exponentiation (all entries ≤ max_c)
            w_unnorm = np.exp(beta * (conv - max_c))
            w = w_unnorm / w_unnorm.sum()           # soft‑max, sums to 1

            # ---- gradient of the smooth objective ----
            # grad = 2 * (w ∗ a)  (take first n entries of the full convolution)
            grad_full = conv_full(w, a)              # length 3n‑2
            grad = 2.0 * grad_full[:n]

            # ---- exponentiated‑gradient update ----
            step = step0 / math.sqrt(it)
            # avoid overflow/underflow in the exponent
            a = a * np.exp(-step * grad)
            # renormalise onto the simplex (still non‑negative)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

            # ---- occasional logging & best‑track ----
            if it % 250 == 0:
                cur_sc = score_of(a)
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] "
                        f"smoothed improve n={n} β={beta:.2f} it={it} score={cur_sc:.6f}"
                    )

        if time.time() >= deadline:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Very cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """Transfer a random ε from one index to another; Metropolis acceptance."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score

    start_time = time.time()

    best_score = float("inf")

    # --------------------------------------------------------------

    except Exception:
        global_a = None

    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768, 65536]
    random.shuffle(lengths)  # avoid systematic bias when stopping early

    for idx, n in enumerate(lengths):

        if global_a is not None:

        else:

            a0 /= a0.sum()

        # ----- quick stochastic search over simple families -----
        shape_deadline = time.time() + min(3.0, (deadline - time.time()) * 0.25)
        local_best_a = a0
        local_best_sc = score_of(a0)

            kind = np.random.choice([0, 1, 2, 3])
            if kind == 0:  # product‑power
                p = np.random.uniform(-3.0, 4.0)

            elif kind == 1:  # Gaussian

            elif kind == 2:  # Beta

                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            else:  # random convex combo of two families

                cand1 = shape_product_power(n, np.random.uniform(-1.0, 1.0))

                cand = w * cand1 + (1 - w) * cand2

            sc = score_of(cand)

                start_time,

                best_seq,

                end_temp=1e-5,

                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"

        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (the evaluator will clamp/normalise as needed)
    return [float(x) for x in best_seq]
