```
    """
    Adam optimizer on smooth max approximation.

    a0: initial probability vector, sum=1.
    n: length.
    deadline: stop time.
    alpha_schedule: list of α values, increase gradually.
    lr0: initial learning rate.
    Returns best sequence and its (2*n*max(conv)) score.
    """
    a = a0.copy()

    y = np.log(a + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        # compute a via softmax (ensures sum=1)
        # Using stable softmax:
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()

        # bias correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

                best_a = a.copy()

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    # final check
    final_score = 2.0 * n * _conv_self(best_a).max()
```

The code below implements a new search routine for `evaluate_sequence`.  
It improves the previous approach by

1. **Adam optimisation on a smooth‑max surrogate** – the objective  
   `max(conv)` is replaced by a log‑sum‑exp (parameter α).  Adam gives an
   adaptive step size and works on the unconstrained variables  
   `y = log(a)` (soft‑max parametrisation guarantees `a_i≥0` and `∑a_i=1`).

2. **Gradual increase of the smoothing parameter** – we start with a very
   smooth surrogate (`α≈0.5`) and raise α during the run (up to 160).  
   This lets the optimiser first shape the distribution globally and later
   focus on the true max‑objective.

3. **Projected‑gradient polishing** – after Adam we run a short momentum
   projected‑gradient descent that uses the exact sub‑gradient of the
   (non‑smoothed) max‑convolution.  This removes the remaining bias from the
   surrogate.

4. **Simple convex mixing with the uniform distribution** – a cheap sweep
   over convex combinations `t·a + (1‑t)·u` often reduces the peak a little
   further.

5. **Final simulated‑annealing refinement** – a lightweight annealing pass
   moves mass between coordinates while preserving the simplex constraint.

All stages respect the supplied time budget (≈ 1000 s).  The routine can be
started from the previously discovered sequence `height_sequence_1` if it is
available, otherwise it begins from a random simplex point.

```python
"""Search for a non‑negative step‑function that minimises `evaluate_sequence`.

The algorithm consists of three phases:

1. Adam optimisation on a smoothed max‑convolution objective.
   The sequence is parameterised by a soft‑max vector `a = softmax(y)` so
   that the simplex constraints are satisfied automatically.
   A log‑sum‑exp with increasing temperature (α) provides a differentiable
   surrogate for `max(conv)`.  Adam’s adaptive step size speeds up convergence.

2. Projected gradient descent with the exact sub‑gradient of `max(conv)` for
   a short polishing stage.

3. A cheap convex mixing with the uniform distribution followed by a brief
   simulated‑annealing refinement.

The routine is time‑aware and returns the best sequence found before the
deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a*a."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max  f(a)= (1/α)·log Σ_k exp(α·conv_k)
    with respect to a (size n).  Returns a vector of size n.
    """
    # softmax weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w_rev  (gradient = 2·(a⋆w_rev))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: a.shape[0] + len(w_rev) - 1]

    # pick the central part of the correlation
    n = a.shape[0]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    return exp_y / exp_y.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to a new length `new_n` (preserves sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max objective
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  alpha_schedule,
                  lr0: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  steps_per_alpha: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a soft‑max parametrised vector.

    Returns the best sequence (sum = 1) and its true score
    `2·n·max(conv)`.
    """
    # initialise y = log(a)  (soft‑max ensures simplex)
    a = a0.copy()
    y = np.log(a + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        # current probabilities
        a = _softmax(y)

        # self‑convolution
        conv = _conv_self(a)

        # current smoothing parameter
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # gradient of the smoothed max w.r.t. a
        grad_a = _grad_smooth_max(a, conv, alpha, fft_len)

        # chain rule: ∂f/∂y = a * (∂f/∂a - ⟨a,∂f/∂a⟩)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (slow decay)
        lr = lr0 / math.sqrt(it)

        # update y
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if it % 100 == 0:
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # move to the next α after a fixed number of iterations
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    # final exact check
    final_score = 2.0 * n * _conv_self(best_a).max()
    return best_a, final_score


# ----------------------------------------------------------------------
# Projected gradient descent with exact sub‑gradient (polishing)
# ----------------------------------------------------------------------


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (using all almost‑active indices)."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Momentum projected‑gradient descent using the exact sub‑gradient."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad iter={it:,} score={cur_score:.6f}")

        # sub‑gradient
        grad = _subgradient_full(a, conv)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000) -> tuple[np.ndarray, float]:
    """Tiny annealing phase that swaps a random ε‑fraction of mass."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:,} score={score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
    # 4) Convex mixing with the uniform distribution
    # ------------------------------------------------------------------
    if best_seq is not None:
        uniform = np.full(n, 1.0 / n)
        for t in np.linspace(0.0, 1.0, 21):
            mixed = t * best_seq + (1.0 - t) * uniform
            mixed /= mixed.sum()
            score = 2.0 * n * _conv_self(mixed).max()
            if score < best_score:
                best_score = score
                best_seq = mixed.copy()
                print(f"[{time.time() - start_time:6.2f}s] mixing t={t:.2f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final annealing with the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and export
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] BEST FINAL SCORE = {2.0 * n * _conv_self(best_seq).max():.6f}")
    return [float(x) for x in best_seq]
```
