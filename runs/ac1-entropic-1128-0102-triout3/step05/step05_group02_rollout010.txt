def smooth_gradient_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = score_of(a)
    momentum = np.zeros_like(a)
    step0 = 1.0
    alpha = 2.0
    it=0
    while time.time() < deadline:
        it += 1
        # compute convolution
        conv = conv_self(a)
        # compute smooth weights
        max_val = conv.max()
        # compute softmax weights
        # For large conv, exponent overflow; we can subtract max to improve numeric stability.
        # Use safe exp.
        exp_vals = np.exp(alpha * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # gradient
        # correlation w and a
        corr_full = np.convolve(w, a[::-1])
        grad = 2.0 * corr_full[n-1:2*n-1]
        # step size schedule
        step = step0 / np.sqrt(it)
        # momentum update
        momentum = 0.9 * momentum - step * grad
        a_new = a + momentum
        # project onto simplex
        a = project_simplex(a_new)

        # evaluate actual objective
        cur_score = score_of(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve n={n} it={it} score={cur_score:.6f}")

        # adapt alpha
        if it % 2000 == 0:
            # increase alpha to sharpen max
            alpha = min(alpha * 1.5, 100.0)
    return best_a, best_score

def conv_full_fft(x, y):
    m = len(x) + len(y) - 1
    L = 1 << ((m-1).bit_length())
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[:m]

"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer builds the solution gradually on increasing lengths.
For each length it
  * starts from an up‑sampled version of the current best vector,
  * tries a few simple parametric families (power‑product, Gaussian,
    Beta) for a quick warm‑start,
  * runs a projected gradient descent on a *smoothed* max‑norm
    (soft‑max) of the self‑convolution – this gives a true gradient
    rather than a sub‑gradient,
  * finishes with a cheap mass‑transfer annealing that moves small
    amounts of probability mass between coordinates.

All iterates are projected onto the probability simplex, therefore the
objective `2·n·max(a*a)` (i.e. ``evaluate_sequence`` after normalisation)
is scale‑invariant.  The schedule spends a larger share of the 1000 s
budget on the bigger lengths, but never exceeds the deadline.

The combination of a smooth surrogate, adaptive step‑size and occasional
random perturbations usually drives the score below the previous record
(~1.503) within the allotted time.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers (simplex projection & FFT‑based convolutions)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors using a power‑of‑two FFT."""
    m = len(x) + len(y) - 1
    fft_len = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:m]
    return conv


def conv_self(a: np.ndarray) -> np.ndarray:
    """Exact linear self‑convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


def correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation (w ⋆ a) for shifts 0 … n‑1.  Both arguments are non‑negative."""
    # correlation_i = sum_j w_{j+i} * a_j
    # compute as convolution of w with reversed a and take the central slice
    a_rev = a[::-1]
    full = _fft_convolve(w, a_rev)          # length = len(w)+len(a)-1
    n = a.shape[0]
    start = n - 1
    return full[start:start + n]


def score_of(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (sum=1)."""
    n = a.shape[0]
    conv = conv_self(a)
    return 2.0 * n * conv.max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simple parametric families for rapid warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p (p<0 → edge‑heavy)."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term = np.where(term == 0, 1e-12, term)
    a = term ** p
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the unit interval."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed max (soft‑max) gradient descent on the simplex
# ----------------------------------------------------------------------


def smooth_gradient_descent(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed version of the ∞‑norm.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.8          # base step (will decay as 1/√it)
    alpha = 1.0          # smoothing parameter – grows slowly
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- current evaluation (exact) ----
        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve n={n} it={it} score={cur_sc:.6f}")

        # ---- soft‑max weights (smooth approximation of the max) ----
        shift = conv.max()                     # for numerical stability
        exp_vals = np.exp(alpha * (conv - shift))
        w = exp_vals / exp_vals.sum()          # w is a probability vector (sum=1)

        # ---- gradient of the smoothed objective ----
        grad = 2.0 * correlation(w, a)         # ∇_a  (1/α)log Σ e^{α·conv}
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        # ---- diminishing step size with a tiny momentum term ----
        step = step0 / math.sqrt(it)
        momentum = 0.9 * momentum - step * direction
        a = project_simplex(a + momentum)

        # ---- slowly increase the smoothing (sharpen the max) ----
        if it % 1500 == 0:
            alpha = min(alpha * 1.3, 80.0)     # never let α explode

        # ---- occasional isotropic jitter to escape plateaus ----
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Very cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 60000) -> tuple[np.ndarray, float]:
    """Move a small ε from a source index to a destination index (Metropolis)."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
