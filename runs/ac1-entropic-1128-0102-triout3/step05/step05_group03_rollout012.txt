"""
Propose a step‑function (non‑negative list) that minimises ``evaluate_sequence``.

Algorithm
---------
* **Multi‑stage refinement** – start with a modest size (≈2 k), then
  progressively up‑sample to larger sizes (≈8 k → 32 k).  Each stage is
  tackled with *smooth‑max* projected gradient descent, which uses a
  soft‑max weighting of the whole convolution instead of a single
  maximal index (the previous code used only a sub‑gradient at the
  unique maximum).  This gives a much richer descent direction and
  escapes plateaus more reliably.
* **Smooth‑max gradient** – for a vector ``a`` (∑a = 1) we compute the
  self‑convolution ``c = a * a``.  From ``c`` we form the soft‑max
  weights ``w_i ∝ exp(α·c_i)`` (α grows during the run).  The gradient
  of the smoothed objective is ``∇ = 2·(a ⋆ w)`` where ``⋆`` denotes the
  cross‑correlation; it can be obtained by a single convolution with the
  reversed weight vector.
* **Projected gradient descent with momentum** – the update is
  ``a ← Π_{simplex}(a – η·momentum)`` where the momentum follows the
  usual Nesterov style and the step‑size decays like ``η₀ / sqrt(t)``.
* **Final annealing** – a short simulated‑annealing phase (mass transfer
  moves) cleans up any remaining spikes.
* **Time‑aware** – the routine respects the supplied time budget
  (default 1000 s) and stops early if the deadline is reached.

The returned sequence is normalised to Σa = 1, which makes the
``evaluate_sequence`` value equal to ``2·n·max(conv)``.  The code is
self‑contained and uses only ``numpy``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT (real‑valued)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max weights from a convolution vector
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i ∝ exp(alpha·conv_i) with numerical stability.
    The returned ``w`` sums to 1.
    """
    # subtract the maximum to avoid overflow
    max_c = conv.max()
    shifted = conv - max_c
    # using np.exp on the shifted values
    exp_vals = np.exp(alpha * shifted)
    total = exp_vals.sum()
    if total == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Given a (length n) and soft‑max weights (length 2n‑1),
    return ∇ = 2·(a ⋆ weights), where ⋆ is the cross‑correlation.
    Implementation via a single convolution with the reversed weight vector.
    """
    n = a.shape[0]
    # reverse the weights for cross‑correlation
    w_rev = weights[::-1]
    # full convolution a * w_rev (length 3n‑2)
    conv_full = np.convolve(a, w_rev, mode="full")
    # indices (2n‑2)‑j for j=0…n‑1 give the desired gradient components
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent with smooth‑max and momentum
# ----------------------------------------------------------------------
def _run_smooth_gd(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   beta: float = 0.9,
                   step0: float = 0.3,
                   alpha_start: float = 0.5,
                   alpha_end: float = 12.0,
                   max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the smooth‑max surrogate.
    Returns the best vector seen and its (true) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    # true score (2·n·max(conv)) – we keep sum(a)=1 throughout
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ---- current convolution & true score ----
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"smooth‑gd improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # ---- smooth‑max weights ----
        # linearly increase alpha from start to end over the whole run
        alpha = min(alpha_end,
                    alpha_start + (alpha_end - alpha_start) * it / max_iter)

        w = _smooth_weights(conv, alpha)

        # ---- gradient of the smoothed objective ----
        grad = _smooth_gradient(a, w)

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- diminishing step size ----
        step = step0 / math.sqrt(it)

        # ---- projected gradient step ----
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing: move ε from a high‑mass index to a
    random one.  Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportional to its mass (bias to large entries)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
