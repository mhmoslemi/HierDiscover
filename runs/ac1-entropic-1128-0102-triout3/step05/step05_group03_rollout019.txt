def _gradient_softmax_max(a, conv, tau):
    n = a.shape[0]
    # compute softmax weights over conv
    c = np.max(conv)  # max for stability
    exp_vals = np.exp(tau * (conv - c))
    w = exp_vals / exp_vals.sum()
    # compute term1 = convolution of reversed w with a; slice appropriate
    w_rev = w[::-1]
    conv_len = len(w_rev) + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Wf = np.fft.rfft(w_rev, n=fft_len)
    conv_full = np.fft.irfft(Af * Wf, n=fft_len)[:conv_len]
    offset = 2 * n - 2
    term1 = conv_full[offset: offset + n]
    grad = term1 + a   # sum_w=1
    return grad

"""
Improved search for a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
* **Simplex projection** – keeps the vector non‑negative and Σa = 1.
* **Fast FFT convolutions** – O(n log n) for both the self‑convolution and the
  weighted convolution required by the soft‑max gradient.
* **Adam optimisation** – adaptive first/second‑moment estimates give a more
  reliable descent direction than plain (Nesterov) momentum.
* **Soft‑max smoothing of the max** – instead of using only the single
  maximal convolution entry we minimise a smooth approximation
  `log‑sum‑exp(τ·conv)`.  τ is increased during the run so that the optimiser
  first explores a smooth landscape and finally concentrates on the true
  maximum.
* **Multi‑stage refinement** – start with a modest size (≈1 k), optimise,
  then up‑sample (linear interpolation) to larger sizes (≈4 k → 12 k) and
  continue optimisation.
* **Final annealing** – a cheap mass‑transfer simulated‑annealing pass
  removes any remaining tiny peaks.

The whole routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2. Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# 3. Gradient of the soft‑max smoothed max‑convolution
# ----------------------------------------------------------------------
def _gradient_softmax_max(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of  g(a)= (1/τ)·log Σ_i exp(τ·conv_i) .
    It equals  Σ_i p_i·∇conv_i where p_i is the soft‑max weight.
    """
    n = a.shape[0]
    # soft‑max weights (stable)
    c = np.max(conv)
    exp_vals = np.exp(tau * (conv - c))
    w = exp_vals / exp_vals.sum()                     # shape (2n‑1,)

    # term1 = Σ_i w_i·a_{i‑k}  for every k (cross‑correlation)
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1                     # = 3n‑2
    fft_len = 1 << ((conv_len - 1).bit_length())
    Wf = np.fft.rfft(w, n=fft_len)
    Af_rev = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(Wf * Af_rev, n=fft_len)[:conv_len]

    # slice gives the needed cross‑correlation values
    term1 = full[n - 1 : n - 1 + n]                   # shape (n,)

    # because Σ_i w_i = 1 the second part of the gradient is just a
    grad = term1 + a
    return grad


# ----------------------------------------------------------------------
# 4. Adam optimiser on the simplex (soft‑max + final max phase)
# ----------------------------------------------------------------------
def _run_adam_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    lr: float = 0.5,
    tau0: float = 0.1,
    tau_max: float = 50.0,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of  F(a)=2·n·max(conv(a))  (a lives on the simplex).
    Returns the best vector seen and its (upper‑bound) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, "
                f"iter={it:,}, score={cur_score:.6f}"
            )

        # schedule τ (smooth→sharp)
        tau = tau0 + (tau_max - tau0) * min(1.0, it / (0.5 * max_iter))

        # gradient of the smoothed objective
        grad = _gradient_softmax_max(a, conv, tau)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # projected Adam step
        a = _project_simplex(a - step)

    return best_a, best_score


# ----------------------------------------------------------------------
# 5. Simple simulated‑annealing fine‑tuning (mass transfer)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing: move a random ε from a high‑mass index
    to a random destination.  Keeps the best sequence ever seen.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose source proportional to its mass (bias towards large values)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                f"iter={it:,}, score={score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# 6. Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    total = up.sum()
    if total > 0:
        up /= total
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to a large size (n≈12288) and final Adam run
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(15.0, 0.35 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_adam_descent(
        a2,
        n2,
        grad_deadline2,
        start_time,
        beta1=0.9,
        beta2=0.999,
        lr=0.2,
        tau0=2.0,
        tau_max=80.0,
        max_iter=400_000,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing refinement (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise (projected vectors already sum to 1, but guard against drift)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
