The algorithm is a hybrid optimiser that works in three phases  

1. **Warm‑start** – a discretised arcsine density (Beta(½,½)) is used as the
   initial shape because its self‑convolution is already very flat.  

2. **Simulated‑annealing on a medium size** – a *mass‑transfer* move
   (transfer a small ε from one position to another) is applied many
   times on a sequence of length `n_small` (≈ 3000).  The temperature is
   reduced continuously; occasional large random restarts and a tiny
   smoothing step help to escape local minima.  The best sequence found
   during this phase is kept.

3. **Progressive up‑sampling & refinement** – the best medium‑size sequence
   is linearly interpolated to the final length `n_final` (≈ 9600) and
   refined with a fast projected‑gradient descent (the same routine used
   by the reference implementation).  A short simulated‑annealing walk
   finishes the optimisation.

All work respects the global time‑budget (default 1000 s).  If nothing
is found a uniform distribution is returned as a safe fallback.

```python
"""
Hybrid optimiser for the step‑function evaluation problem.

* start with an arcsine shape (Beta(½,½));
* run a mass‑transfer simulated‑annealing on a medium‑size sequence;
* progressively up‑sample to the final size and finish with projected
  gradient descent + a short annealing walk.
The routine respects the given time budget and returns the best sequence
found.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project onto the probability simplex (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i≥0, ∑x_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a*a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using an FFT.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active constraints only)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Projected gradient descent with Nesterov momentum (as in the reference)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.8,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex.
    Returns the best point seen and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] gd n={n:5d} iter={it:7d} "
                  f"score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} "
                  f"iter={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Score of a sequence (the quantity returned by evaluate_sequence)
# ----------------------------------------------------------------------
def _score_of(seq: np.ndarray) -> float:
    """Return 2·n·max(conv) assuming the sequence is already normalised."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max() / (seq.sum() ** 2))


# ----------------------------------------------------------------------
# Arcsine initial shape (Beta(0.5,0.5))
# ----------------------------------------------------------------------
def _arcsine_init(n: int) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n          # points in (0,1)
    a = 1.0 / np.sqrt(xs * (1.0 - xs))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Hybrid optimiser:
    1) arcsine start,
    2) mass‑transfer simulated annealing on a medium size,
    3) up‑sample → projected gradient descent → short annealing.
    Returns a list of floats (the step‑function heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global variable height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # 0. Basic parameters
    # ------------------------------------------------------------------
    n_small = 3000                 # medium size for fast SA
    n_final = 9600                 # final length (same as the reference)
    # allocate roughly 70 % of the budget to the SA phase
    sa_deadline = start_time + 0.70 * (deadline - start_time)

    # ------------------------------------------------------------------
    # 1. Initialise a medium‑size sequence
    # ------------------------------------------------------------------
    if starter is not None:
        a_small = _upsample(starter, n_small)
    else:
        a_small = _arcsine_init(n_small)

    cur_seq = a_small.copy()
    cur_score = _score_of(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:6.2f}s] initial medium score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing on the medium size
    # ------------------------------------------------------------------
    # temperature schedule (geometric)
    start_temp = cur_score * 0.10
    end_temp = cur_score * 1e-6

    while time.time() < sa_deadline:
        # fraction of elapsed time used for temperature decay
        elapsed = time.time() - start_time
        prog = (elapsed) / (deadline - start_time)
        T = start_temp * (end_temp / start_temp) ** prog

        # ---- small mass‑transfer move ----
        i, j = np.random.randint(0, n_small, size=2)
        if i != j and cur_seq[i] > 0.0:
            eps = cur_seq[i] * np.random.rand() * 0.05   # up to 5 % of source mass
            prop = cur_seq.copy()
            prop[i] -= eps
            prop[j] += eps
            # sanity – keep non‑negative
            if prop[i] >= 0.0:
                new_score = _score_of(prop)
                delta = new_score - cur_score
                if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
                    cur_seq = prop
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = cur_seq.copy()
                        print(f"[{time.time() - start_time:6.2f}s] SA improve → {best_score:.6f}")

        # ---- occasional large random restart (helps escape deep local minima) ----
        if np.random.rand() < 0.001:
            rand_seq = np.random.rand(n_small)
            rand_seq = rand_seq / rand_seq.sum()
            rand_score = _score_of(rand_seq)
            if rand_score < cur_score or math.exp(-(rand_score - cur_score) / max(T, 1e-12)) > np.random.rand():
                cur_seq = rand_seq
                cur_score = rand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    print(f"[{time.time() - start_time:6.2f}s] SA random restart → {best_score:.6f}")

        # ---- very occasional smoothing (reduces high‑frequency noise) ----
        if np.random.rand() < 0.0005:
            smooth = np.convolve(cur_seq, np.ones(5) / 5.0, mode='same')
            smooth = np.maximum(smooth, 0.0)
            smooth /= smooth.sum()
            smooth_score = _score_of(smooth)
            if smooth_score < cur_score:
                cur_seq = smooth
                cur_score = smooth_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    print(f"[{time.time() - start_time:6.2f}s] SA smoothing → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to the final size and project‑gradient refinement
    # ------------------------------------------------------------------
    a_large = _upsample(best_seq, n_final)

    # Projected GD (re‑uses the routine from the reference solution)
    a_opt, gd_score = _run_gradient_descent(
        a_large, n_final, deadline, start_time,
        beta=0.9, step0=0.8, max_iter=10**7
    )
    if gd_score < best_score:
        best_score = gd_score
        best_seq = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] GD refinement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final cheap annealing (tiny mass moves) – only if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, anneal_score = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=300_000
        )
        if anneal_score < best_score:
            best_seq = a_ref.copy()
            best_score = anneal_score
            print(f"[{time.time() - start_time:6.2f}s] final anneal → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq.tolist()]
```
