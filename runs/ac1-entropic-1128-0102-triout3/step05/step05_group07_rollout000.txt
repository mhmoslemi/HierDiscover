def generate_uniform(n):
    return np.full(n, 1.0 / n)

def generate_triangle(n):
    idx = np.arange(n)
    tri = np.minimum(idx + 1, n - idx)
    return tri / tri.sum()

def generate_hann(n):
    idx = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * idx / (n-1))) if n>1 else np.array([1.0])
    return w / w.sum()

def generate_hamming(n):
    idx = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * idx / (n-1)) if n>1 else np.array([1.0])
    return w / w.sum()

def generate_blackman(n):
    idx = np.arange(n)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = a0 - a1 * np.cos(2 * np.pi * idx / (n-1)) + a2 * np.cos(4 * np.pi * idx / (n-1))
    return w / w.sum()
```

def generate_kaiser(n, beta):
    idx = np.arange(n)
    alpha = (n-1)/2.0
    x = (idx - alpha)/alpha
    w = i0(beta * np.sqrt(1 - x**2)) / i0(beta)
    return w / w.sum()

def propose_candidate(seed=0, budget_s=1000):
    import time, math, numpy as np
    from scipy.special import i0

    start = time.time()
    deadline = start + budget_s - 0.5

    np.random.seed(seed)

    best_seq = None
    best_score = float('inf')
    best_n = None

    # sizes to try
    sizes = [1200, 1500, 1800, 2400, 3000]  # adjust as needed
    # but ensure not exceed maybe 5000 to keep runtime.

    for n in sizes:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start:6.2f}s] Testing size {n}")

        # Candidate generator list (functions and param lists)
        candidates = []

        # uniform
        candidates.append((generate_uniform, {}))

        # triangle
        candidates.append((generate_triangle, {}))

        # hann
        candidates.append((generate_hann, {}))

        # hamming
        candidates.append((generate_hamming, {}))

        # blackman
        candidates.append((generate_blackman, {}))

        # kaiser with several betas
        for beta in [0.5, 2.0, 4.0, 8.0, 12.0]:
            candidates.append((generate_kaiser, {'beta': beta}))

        # beta distribution
        for alpha in [0.3, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]:
            candidates.append((generate_beta, {'alpha': alpha}))

        # sin power
        for gamma in [0.5, 1.0, 2.0, 3.0, 4.0, 6.0]:
            candidates.append((generate_sin_power, {'gamma': gamma}))

        # random Dirichlet
        for _ in range(4):
            candidates.append((lambda n: np.random.dirichlet(np.ones(n)), {}))

        # random sparse spikes
        for _ in range(4):
            candidates.append((lambda n: _random_spikes(n), {}))

        # Evaluate each candidate quickly
        for gen_func, params in candidates:
            if time.time() >= deadline:
                break
            try:
                # generate
                a = gen_func(n, **params) if params else gen_func(n)
            except Exception as e:
                continue
            # protect against nan
            if np.any(np.isnan(a)) or np.any(np.isinf(a)):
                continue
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0:
                continue
            a = a / s
            # compute max conv using FFT
            max_val, _ = _conv_max(a)
            score = 2.0 * n * max_val  # since sum=1
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                best_n = n
                print(f"    New best (scan) n={n} score={score:.6f}")
        # Optionally after scanning each n, we may run short EG refinement on best candidate for that size
        if best_seq is not None and best_n == n and time.time() < deadline:
            # short EG
            remaining = deadline - time.time()
            # allocate limited iters based on remaining and n
            iters = int(min(50000, remaining / (0.001 * (n/2000.0) + 1e-4)))
            if iters > 0:
                print(f"    Refine best for n={n} with EG iters={iters}")
                a_ref, ref_score = _exp_gradient_optimize(best_seq.copy(), max_iters=iters, deadline=deadline,
                                                          eta_start=0.6, eta_end=0.001)
                if ref_score < best_score:
                    best_score = ref_score
                    best_seq = a_ref.copy()
                    print(f"        Refined best score={best_score:.6f}")

    # final long refinement on best_seq
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        remaining = deadline - time.time()
        iters = int(min(300000, remaining / (0.001 * (n/2000.0) + 1e-4)))
        if iters > 0:
            print(f"[{time.time() - start:6.2f}s] Final long EG refinement, n={n}, iters={iters}")
            a_final, final_score = _exp_gradient_optimize(best_seq.copy(), max_iters=iters, deadline=deadline,
                                                         eta_start=0.3, eta_end=5e-4)
            if final_score < best_score:
                best_score = final_score
                best_seq = a_final.copy()
                print(f"    Final improvement score={best_score:.6f}")

    # final evaluation using evaluate_sequence for safety
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = float('nan')
    print(f"[{time.time() - start:6.2f}s] Finished, best score={final_score:.6f}")

    return best_seq.tolist()

"""
Propose a low‑score step‑function.

The algorithm consists of three phases:

1. **Parametric grid search** –  
   For several problem sizes (≈1200…3000) we generate many
   candidate probability vectors from classic windows (uniform, triangle,
   Hann, Hamming, Blackman, Kaiser), from simple parametric families
   (Beta(α,α) and sin^γ) and from random Dirichlet or sparse spikes.
   Each candidate is normalised to sum = 1 and evaluated with a fast
   FFT‑based convolution to obtain `score = 2·n·max(conv)`.
   The best candidate seen so far is kept.

2. **Exponential‑gradient (mirror‑descent) refinement** –  
   Starting from the current best vector we run a multiplicative
   update `a_i ← a_i·exp(−η·g_i)` where `g = 2·(a_rev * w)` and `w`
   is the (normalised) indicator of the active convolution entries.
   The step size η decays geometrically from `η_start` to `η_end`.
   This phase directly tackles the true L∞‑norm and quickly drives
   the objective lower.

3. **Final long‑run refinement** –  
   After the grid search we allocate the remaining time to a
   substantial EG optimisation of the global best, using a slower
   decay schedule for higher precision.

All operations respect the non‑negativity and normalisation constraints,
and the function returns the best sequence found within the time budget.
"""

import time
import math
import numpy as np
from scipy.special import i0   # Kaiser window

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max(conv(a, a)) and the full linear convolution.
    Uses a power‑of‑two FFT length for speed.
    """
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())      # next power of two ≥ m
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:m]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k * conv_k (where conv = a * a) w.r.t. a.
    Implements g = 2·(a_rev ⋆ w) via FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # should be 2·n‑1
    size = 1 << ((n + m - 1).bit_length())
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv2 = np.fft.irfft(A_rev * W, n=size)[:n + m - 1]
    start = n - 1
    return 2.0 * conv2[start:start + n]


# ----------------------------------------------------------------------
# Exponential‑gradient optimiser (hard‑max)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the exact max‑convolution.
    Returns the best vector encountered and its score.
    """
    n = a.shape[0]
    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # active anti‑diagonals (within eps of the maximum)
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        g = _gradient_from_weights(a, w)

        # geometric step‑size decay
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update (clipped to avoid overflow)
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # exp(±30) ≈ 1e13, safe for double
        a = a * np.exp(upd)

        # renormalise to the probability simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s

    return _as_simplex(w)

    alpha = (n - 1) / 2.0
    x = (idx - alpha) / alpha
    w = i0(beta * np.sqrt(1.0 - x * x)) / i0(beta)
    return _as_simplex(w)


def generate_beta(n: int, alpha: float = 1.0) -> np.ndarray:
    # Beta(α,α) on a regular grid, shifted by 0.5/n to avoid exact zeros
    x = (np.arange(n) + 0.5) / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    return _as_simplex(w)


def generate_sin_power(n: int, gamma: float = 2.0) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    w = np.sin(np.pi * x) ** gamma
    return _as_simplex(w)


def generate_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


def _random_spikes(n: int) -> np.ndarray:
    spikes = np.zeros(n)
    k = np.random.randint(2, 6)                     # 2‑5 spikes
    pos = np.random.choice(n, size=k, replace=False)
    spikes[pos] = np.random.rand(k) + 0.1           # avoid exact zero
    return _as_simplex(spikes)


def _upsample(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence to a new length."""
    old_n = seq.shape[0]
    if old_n == n:
        return _as_simplex(seq.copy())
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, n)
    up = np.interp(x_new, x_old, seq)
    return _as_simplex(up)
