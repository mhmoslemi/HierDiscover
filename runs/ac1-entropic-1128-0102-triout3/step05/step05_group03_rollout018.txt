def _aggregate_subgrad(a, conv, top_k=3):
    n = a.shape[0]
    idxs = np.argpartition(conv, -top_k)[-top_k:]  # unsorted
    grad = np.zeros_like(a)
    for k in idxs:
        i_range = np.arange(n)
        ip = k - i_range
        mask = (ip >= 0) & (ip < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[ip[mask]]
        diag_mask = mask & (2 * i_range == k)
        if np.any(diag_mask):
            g[diag_mask] = a[i_range[diag_mask]]  # derivative of square counted once
        grad += g
    grad /= top_k
    return grad

def _run_adam(a0, n, deadline, start_time, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8, top_k=1):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={t:,}, score={cur_score:.6f}")
        # compute gradient
        if top_k == 1:
            grad = _subgradient_max_correct(a, conv)
        else:
            # average subgrad for top_k values
            idxs = np.argpartition(conv, -top_k)[-top_k:]
            grad = np.zeros_like(a)
            for k in idxs:
                i_range = np.arange(n)
                ip = k - i_range
                mask = (ip >= 0) & (ip < n)
                grad[mask] += 2.0 * a[ip[mask]]
            grad /= top_k
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # Learning rate schedule: decay with sqrt(t) to ensure convergence
        lr = step0 / (np.sqrt(t) + 1e-6)
        a = a - lr * (m_hat / (np.sqrt(v_hat) + eps))
        a = _project_simplex(a)
    return best_a, best_score

def _conv_self_fft(a):
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(Af * Af, fft_len)[:2 * n - 1]

def _run_annealing(a0, n, deadline, start_time, cur_best_score, cur_best_seq,
                    start_temp=5e-2, end_temp=1e-6, steps=200_000):
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        # choose source weighted by mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        # transfer up to 80% of source
        eps = a[src] * rng.random() * 0.8
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")
    return best_seq, best_score

"""
Propose a step‑function (non‑negative list) that minimises
`evaluate_sequence`.  

Algorithm
---------
* **Diverse initialisations** – uniform, triangular, sinusoidal and
  sparse “spike’’ patterns (different numbers of spikes).  All candidates
  are re‑normalised to sum = 1 and the best is selected.
* **Projected Adam optimisation** – minimise 2·n·max(conv(a)) on the
  simplex using an Adam‑style optimiser.  The gradient is the exact
  sub‑gradient of the (unique) maximal convolution entry; a multi‑top‑k
  variant is also available.
* **Progressive up‑sampling** – the best vector is linearly interpolated
  to larger dimensions (≈8 k → 32 k) and further optimised, allowing a
  finer discretisation of the continuous optimum.
* **Intensive simulated‑annealing** – mass‑transfer moves (ε taken from a
  high‑mass index and added to a random one) with a geometric temperature
  schedule; the number of steps is limited only by the remaining budget.
* **Final polish** – a short Adam run after annealing to capture any
  remaining improvement.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (correct version)
# ------------------------------------------------------------
def _subgradient_max_correct(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) when the maximum is attained at a
    single index.  Derivative of conv_k wrt a_i is 2·a_{k‑i} whenever
    the index is valid (including the diagonal case).
    """
    n = a.shape[0]
    kmax = int(np.argmax(conv))          # one index attaining the maximum
    idx = np.arange(n)
    i = kmax - idx
    mask = (i >= 0) & (i < n)

    g = np.zeros_like(a)
    g[mask] = 2.0 * a[i[mask]]
    return g


# ------------------------------------------------------------
# Projected Adam optimiser on the simplex
# ------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              step0: float = 0.15,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              top_k: int = 1) -> tuple[np.ndarray, float]:
    """
    Adam optimiser (projected onto the simplex) that minimises
    F(a) = 2·n·max(conv(a)),  sum(a)=1.
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # gradient: either single max or average of top‑k entries
        if top_k == 1:
            grad = _subgradient_max_correct(a, conv)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]
            grad = np.zeros_like(a)
            for k in idxs:
                i_range = np.arange(n)
                ip = k - i_range
                mask = (ip >= 0) & (ip < n)
                grad[mask] += 2.0 * a[ip[mask]]
            grad /= top_k

        # Adam update (learning‑rate decays with sqrt(t) for stability)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = step0 / (math.sqrt(it) + 1e-6)
        a = a - lr * (m_hat / (np.sqrt(v_hat) + eps))
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-7,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated annealing: move ε from a heavy index to a
    random one.  The temperature decays geometrically; the best sequence
    seen is returned.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source index proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer up to 80 % of the source mass
        eps = a[src] * rng.random() * 0.80
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the time budget and returns
    the best sequence as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 0 – generate a palette of diverse initial candidates (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    candidates = []

    # uniform
    candidates.append(np.full(n0, 1.0 / n0))

    # triangular (symmetric)
    half = np.arange(1, n0 // 2 + 1)
    tri = np.concatenate([half, half[::-1][: n0 - len(half)]])
    candidates.append(tri / tri.sum())

    # sinusoidal bell (zero at the ends)
    x = np.linspace(0, np.pi, n0)
    sinbell = np.sin(x)
    candidates.append(sinbell / sinbell.sum())

    # sparse spike patterns with various numbers of spikes
    for k in (30, 60, 120, 180, 240):
        a = np.zeros(n0)
        idxs = np.random.choice(n0, size=k, replace=False)
        a[idxs] = 1.0 / k
        candidates.append(a)

    # Dirichlet draws with concentration < 1 (produces sparsity)
    for _ in range(5):
        a = np.random.dirichlet(np.full(n0, 0.3))
        candidates.append(a)

    # random dense restarts
    for _ in range(4):
        a = np.random.rand(n0)
        a /= a.sum()
        candidates.append(a)

    # evaluate all seeds and keep the best
    for i, a in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            continue
        a /= a.sum()
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            # fallback to our own formula if the harness version is unavailable
            score = 2.0 * n0 * _conv_self_fft(a).max()
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] seed {i:02d} improve: score={score:.6f}")

    # ------------------------------------------------------------------
    # If nothing survived (unlikely) fall back to uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self_fft(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – medium size optimisation (≈8 k)
    # ------------------------------------------------------------------
    n1 = 8192
    a1 = _upsample(best_seq, n1)
    remaining = deadline - time.time()
    if remaining > 10.0:
        stage1_time = min(200.0, 0.25 * remaining)
        stage1_deadline = time.time() + stage1_time
        a1_opt, score1 = _run_adam(
            a1, n1,
            stage1_deadline,
            start_time,
            step0=0.12,
            top_k=1,
        )
        if score1 < best_score:
            best_score = score1
            best_seq = a1_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – large size optimisation (≈32 k)
    # ------------------------------------------------------------------
    n2 = 32768
    a2 = _upsample(best_seq, n2)
    remaining = deadline - time.time()
    if remaining > 20.0:
        stage2_time = min(300.0, 0.30 * remaining)
        stage2_deadline = time.time() + stage2_time
        a2_opt, score2 = _run_adam(
            a2, n2,
            stage2_deadline,
            start_time,
            step0=0.08,
            top_k=3,           # use a few top indices for a smoother direction
        )
        if score2 < best_score:
            best_score = score2
            best_seq = a2_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # number of annealing steps grows with the remaining wall‑time
        remaining = deadline - time.time()
        steps = int(max(100_000, min(600_000, remaining * 250)))  # heuristic
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=6e-2,
            end_temp=1e-7,
            steps=steps,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Optional polishing pass (short Adam run) if a little time remains
    # ------------------------------------------------------------------
    if time.time() < deadline:
        brief_deadline = deadline
        a_final, score_final = _run_adam(
            best_seq,
            best_seq.shape[0],
            brief_deadline,
            start_time,
            step0=0.05,
            top_k=1,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final Adam polish improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
