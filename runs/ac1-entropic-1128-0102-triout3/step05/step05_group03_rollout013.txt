"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm
---------

* **Exact sub‑gradient** – compute the true sub‑gradient of `max(conv(a))`
  (no diagonal under‑counting).  When several convolution entries tie for
  the maximum we sum the contributions of all active indices via a fast
  FFT‑based convolution.
* **Projected gradient descent** – simple momentum with a 1/√t step–size,
  each iteration projects onto the probability simplex.
* **Multi‑stage refinement** – several stages of increasing size
  (≈2 k → 8 k → 12 k).  Each stage starts from the best vector of the
  previous stage (linear up‑sampling) and runs a longer descent.
* **Extended simulated annealing** – after the last gradient phase we
  perform many mass‑transfer moves with a geometric temperature schedule
  to escape possible local minima.
* **Random restarts** – the initial small‑scale stage uses many random
  seeds (and a possible starter from the previous run) to increase
  diversity.

All helpers are top‑level functions; the only public entry point is
`propose_candidate`.  The routine respects the supplied time budget
(default 1000 s) and returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via FFT (real‑to‑real)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear self‑convolution)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv(a))
# ----------------------------------------------------------------------
def _subgrad_max_exact(a: np.ndarray, conv: np.ndarray,
                       tol_rel: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(a)).
    If several indices are (almost) maximal we sum their contributions.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance to collect an active set (helps when the max is numerically
    # attained at several positions)
    tol = max(tol_rel * max_val, 1e-15)
    active = np.where(conv >= max_val - tol)[0]

    if active.size == 1:
        # Fast path: a single active index, O(n) shift
        k = active[0]
        i = k - np.arange(n)
        mask = (i >= 0) & (i < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[i[mask]]
        return grad
    else:
        # General case: build a mask and convolve (still cheap thanks to FFT)
        mask = np.zeros_like(conv, dtype=float)
        mask[active] = 1.0
        # gradient = 2 * (a * reverse(mask))_relevant_slice
        conv_full = _conv_fft(a, mask[::-1])
        grad = 2.0 * conv_full[n - 1:2 * n - 1]
        return grad


# ----------------------------------------------------------------------
# Projected gradient descent (with optional momentum)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on  F(a)=2·n·max(conv(a)),
    with simplex projection after each step.
    Returns the best vector seen and its (approximate) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: n={n:,} it={it:,} score={cur_score:.6f}")

        # exact sub‑gradient of max(conv)
        grad = _subgrad_max_exact(a, conv)

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing: move ε from a heavy index to a random one.
    Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source chosen proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # up to 30 % of the source mass is moved
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # record global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n} it={it:,} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                tmp = np.maximum(tmp, 0.0)
                s = tmp.sum()
                if s > 0:
                    starter = tmp / s
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short runs on a modest size (n≈2048)
    # ------------------------------------------------------------------
    n0 = 2048
    slice_time = max(15.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    restarts = 6
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        beta = np.random.uniform(0.80, 0.95)
        step0 = np.random.uniform(0.25, 0.45)

        a_opt, score = _run_gradient_descent(
            a0, n0, slice_deadline, start_time,
            beta=beta,
            step0=step0,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage0 restart {r} best score={score:.6f}")

    # fallback
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 1 – up‑sample to medium size (n≈8192) and longer GD
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.35 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_gradient_descent(
        a1, n1, grad_deadline1, start_time,
        beta=0.93,
        step0=0.28,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage1 improved score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to large size (n≈12288) and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(15.0, 0.30 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_gradient_descent(
        a2, n2, grad_deadline2, start_time,
        beta=0.94,
        step0=0.24,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage2 improved score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=400_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improved score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against accidental zero‑sum) and return list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
