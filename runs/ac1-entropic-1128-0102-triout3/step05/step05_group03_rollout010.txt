```
def adam_optimize(a0, n, deadline, start_time, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8, max_iters=10**7):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = np.inf
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam improve: n={n}, iter={t}, score={cur_score:.6f}")
        # gradient of max conv:
        kmax = int(np.argmax(conv))
        # compute gradient vector
        # gradient_i = 2*a[kmax - i] if 0 <= kmax-i < n else 0
        # no diag correction.
        idx = np.arange(n)
        i = kmax - idx
        mask = (i >= 0) & (i < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[i[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score
```

assistantfinal**Solution Explanation**

The objective  `evaluate_sequence`  (after the mandatory normalisation to a non‑negative vector with sum = 1) is  

```
F(a) = 2·n·max( a * a )                ( * = linear self‑convolution )
```

so the task is to find a probability vector `a` that makes the peak of its
self‑convolution as small as possible.
The function is non‑smooth (the maximum changes its active index) but a
sub‑gradient is easy to obtain:

*If the maximum of the convolution is attained at index `k`,
the sub‑gradient is*  

```
g_i = 2·a_{k-i}   for i s.t. 0 ≤ k-i < n , otherwise 0
```

(derivative of `a_i·a_{k-i}` – the diagonal term needs no special treatment).

The optimisation routine consists of three stages

1. **Adam‑style projected gradient descent** on a modest size (`n≈2 k`),
   many random restarts, and a small pool of handcrafted seeds
   (the previous best solution, a sine‑shaped start, uniform start).

2. **Upsampling** to a larger dimension (`n≈8 k` → `n≈24 k`) and continuing the
   same Adam optimiser.  The larger space allows a finer discretisation of the
   (continuous) optimal shape.

3. **Simulated‑annealing refinement** – cheap “mass‑transfer’’ moves are
   accepted according to a geometric temperature schedule; this helps to escape
   shallow local minima after the gradient phase.

All updates are projected onto the probability simplex (`a_i ≥ 0`,
`∑a_i = 1`).  The whole procedure respects the user supplied time budget
(`budget_s` = 1000 s by default) and continually remembers the best vector
found.

The code is completely self‑contained, uses only `numpy` and the standard
library, and returns the final sequence as a plain Python list of floats.



```python
"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm
---------
1.  Several random restarts on a modest size (n≈2048) using Adam‑style
    projected gradient descent on the objective
        F(a) = 2·n·max(conv(a))   (with sum(a)=1).
    The gradient is the exact sub‑gradient of the current maximal
    convolution entry (averaged over all almost‑maximal indices).

2.  Upsample the current best vector to a larger dimension (n≈8192) and
    continue the Adam optimiser.

3.  Upsample again to an even larger dimension (n≈24576) and finish the
    optimisation with the same optimiser.

4.  A short simulated‑annealing phase performs random mass‑transfer moves
    (ε transferred from a high‑mass index to a random one) with a
    geometric temperature schedule, keeping the global best.

All iterates are projected onto the probability simplex.  The routine
stops at the supplied time budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:m]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (averaged over almost‑maximal indices)
# ----------------------------------------------------------------------
def _max_conv_subgradient(a: np.ndarray, conv: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) when the maximum may be attained at several
    indices.  All indices with value within `tol` of the maximum are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # pick all indices that are numerically maximal
    active = np.where(conv >= max_val - max(tol, max_val * 1e-12))[0]

    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active:
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if active.size > 0:
        grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of  F(a)=2·n·max(conv(a)).
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient (average over near‑maximal indices)
        grad = _max_conv_subgradient(a, conv)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves) for fine tuning
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         cur_best_score: float,
                         cur_best_seq: np.ndarray,
                         start_temp: float = 4e-2,
                         end_temp: float = 1e-6,
                         steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Cheap mass‑transfer moves with a geometric temperature schedule.
    Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # temperature decay
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # choose a source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
```
