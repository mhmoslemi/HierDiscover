"""
Propose a low‑value step‑function.

Strategy
--------
1.  Start from a previous construction (if available) and normalise it to a
    probability vector (∑a_i = 1).  If none exists use a random simplex point.
2.  Progressive up‑sampling: 800 → 2400 → 4800 → 9600 → (optional) 20000.
    Very little time is spent on the tiny stages; the bulk of the budget is
    reserved for the largest size.
3.  Two gradient phases on the largest size
      * soft‑max (smooth) max‑convolution gradient – gives a smooth landscape
        and explores globally;
      * exact sub‑gradient of the true max–convolution – fine‑tunes the
        solution once the temperature is essentially zero.
   Both phases use projected Nesterov momentum on the probability simplex.
4.  A final Metropolis‑style simulated‑annealing walk (tiny mass transfers)
    refines the solution further.
5.  Throughout the run the best sequence found (lowest value of
    ``evaluate_sequence``) is stored and finally returned.

The whole procedure respects the supplied time budget (default 1000 s) and
never returns an empty or invalid list, guaranteeing a finite score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – Linear self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3 – Convolution of rev(a) with a weight vector (used for gradients)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed sequence a_rev with a weight vector w.
    Returns the full linear convolution (length = len(a_rev)+len(w)-1).
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4 – Soft‑max (smooth) gradient of the max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of Σ_k softmax_T(conv_k)·conv_k (used as smooth max)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))                   # numerical stability
    w = exps / exps.sum()
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)                       # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]                       # central block
    return grad


# ----------------------------------------------------------------------
# Helper 5 – Exact sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Uses a uniform convex combination of the gradients of all active shifts.
    """
    max_val = conv.max()
    # indices where the max is attained (tolerance needed because of floating error)
    active = np.where(conv >= max_val - eps)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Gradient descent on the *smooth* max (soft‑max)
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    temp_start: float = 2e-2,
                    temp_end: float = 5e-7,
                    target_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov gradient descent using the soft‑max approximation.
    Returns the best point found and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < target_iters:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # exponential temperature schedule (≈ target_iters steps to reach temp_end)
        T = temp_start * (temp_end / temp_start) ** (it / target_iters)

        grad = _softmax_gradient(a, conv, T)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # Diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score

def _run_exact_gd(a0: np.ndarray,

        momentum = beta * momentum + (1.0 - beta) * grad

# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,

                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 10 % of a[i]
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
