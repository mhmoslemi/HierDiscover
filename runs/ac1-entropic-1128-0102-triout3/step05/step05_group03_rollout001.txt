"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm
---------
1. **Simplex projection** – O(n) Euclidean projection onto
   {x ≥ 0, Σx = 1}.
2. **Fast self‑convolution** – linear convolution via real FFT.
3. **Adam‑style projected optimisation** – adaptive first/second‑moment
   updates on the sub‑gradient of the (unique) maximal convolution entry.
   This usually escapes shallow local minima faster than plain SGD.
4. **Multi‑stage up‑sampling** – start with a modest length (≈2 k),
   run several random restarts of Adam, keep the best, then linearly
   interpolate to larger dimensions (≈8 k → 16 k) and continue optimisation.
5. **Simulated‑annealing fine‑tuning** – mass‑transfer moves with a
   geometric temperature schedule for the remaining time budget.
6. **Time‑aware orchestration** – every stage respects the global
   1000 s budget; the routine always returns a feasible sequence.

The code is self‑contained, uses only NumPy and the provided
`evaluate_sequence`, and prints progress messages.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} (Euclidean projection)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (using a single maximal index)
# ------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) when the maximum is attained at a unique index."""
    n = a.shape[0]
    kmax = int(np.argmax(conv))       # index of the maximal convolution entry
    idx = np.arange(n)
    i = kmax - idx
    mask = (i >= 0) & (i < n)

    g = np.zeros_like(a)
    g[mask] = 2.0 * a[i[mask]]

    # diagonal correction: when i == idx (i.e. 2*idx == kmax) we counted a_i*a_i once
    diag = mask & (2 * idx == kmax)
    if np.any(diag):
        g[diag] = a[idx[diag]]
    return g


# ------------------------------------------------------------------
# Adam‑style projected optimisation on the simplex
# ------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              alpha: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam with projection onto the simplex.
    Returns the best vector seen and its score (2·n·max(conv)).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of the maximal convolution entry
        grad = _subgradient_max(a, conv)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # element‑wise step (clipped to avoid absurdly large jumps)
        step = alpha / (np.sqrt(v_hat) + eps)
        step = np.clip(step, 0.0, 0.5)

        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated annealing.
    Returns the best sequence encountered.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source index proportionally to its mass (bias towards large entries)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
