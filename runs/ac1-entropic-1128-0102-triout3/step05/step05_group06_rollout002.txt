"""
Propose a non‑negative step function (probability vector) that minimizes
`evaluate_sequence`.  

Algorithm
---------
1. **Initial shape** – a discretised arcsine (Beta(½,½)) density, which is already
   U‑shaped and yields a relatively flat self‑convolution.
2. **Progressive up‑sampling** – the sequence is repeatedly linearly interpolated
   to larger lengths (800 → 2400 → 4800 → 9600 → 12000).  
   At each resolution a dedicated optimiser runs.
3. **Projected Adam optimisation** – we minimise the quantity
   `2·n·max(conv)` (the denominator `sum(a)^2` is kept equal to 1 by working on
   the simplex).  The gradient of `max(conv)` is taken only on the active
   convolution indices (those attaining the current maximum).  An Adam update
   with per‑coordinate step sizes is applied and the iterate is projected back
   onto the probability simplex.
4. **Stochastic refinement** – after the last up‑sampled resolution a cheap
   simulated‑annealing walk moves a tiny amount of mass between two positions
   and accepts downhill or uphill moves with a geometric temperature schedule.
5. **Safety fallback** – if nothing was found, return a uniform distribution.

The routine respects the supplied time budget (`budget_s`), prints progress
messages, and finally returns the best sequence as a plain Python list.
"""
import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    # next power of two for efficient FFT
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (active constraints only)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    Only indices k with conv[k] >= max(conv)*(1‑rel_tol) are considered active.
    Returns the average sub‑gradient over the active set.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # gradient component for position i is a_{k-i}
        g[idx] += a[k - idx]

    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (projected)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   max_iter: int = 10**9) -> tuple[np.ndarray, float]:
    """
    Run Adam with simplex projection.
    Returns the best vector seen and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n:5d} "
                  f"iter={t:7d} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.2 / (np.sqrt(v_hat) + eps)   # per‑coordinate step
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   max_steps: int = 1_000_000) -> tuple[np.ndarray, float]:
    """Very cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} "
                  f"iter={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
