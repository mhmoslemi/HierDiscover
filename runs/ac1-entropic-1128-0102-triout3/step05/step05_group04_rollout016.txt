"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm (≈1000 s)
-------------------
1. **Candidate seeding**
   * Build a small portfolio of hand‑crafted families (uniform,
     edge‑heavy power laws, mixtures of the two) and the optional historic
     ``height_sequence_1``.  The best of these is taken as the start.

2. **L²‑flattening stage**
   * minimise the squared deviation of the self‑convolution from the
     uniform target µ=1/(2 n‑1) using a mirror‑descent update
     a←a·exp(‑η ∇L₂).  The gradient ∇L₂ is obtained analytically as a
     correlation and computed with a single FFT pair per iteration.
   * This stage quickly drives the convolution toward a flat shape and
     often improves the bound beyond the reference 1.513.

3. **Smoothed‑max exponentiated‑gradient stage**
   * Switch to the log‑sum‑exp surrogate of the max‑convolution.
   * Multiplicative (exponentiated) updates with an Adam‑like schedule
     keep the iterate in the probability simplex while aggressively
     reducing the peak value.

4. **Exact sub‑gradient polishing**
   * A classic projected gradient descent (momentum, Euclidean projection
     onto the simplex) refines the solution under the true max‑objective.

5. **Late‑stage simulated‑annealing**
   * Small random mass‑transfer moves finish the optimisation,
     accepting uphill moves with a geometric cooling schedule.

All stages respect the 1000 s wall‑clock budget; intermediate progress
(and the current best score) is printed for the harness output.

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv) using all active indices."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (same shape as a).
    """
    n = a.shape[0]
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)          # stabilise
    w /= w.sum()

    # correlation a ⋆ w_rev  (gradient = 2·(a⋆w_rev))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _grad_l2(a: np.ndarray, conv: np.ndarray, avg: float,
             fft_len: int) -> np.ndarray:
    """
    Gradient of 0.5·∥conv - avg·1∥²  w.r.t. a.
    Returns a vector of size n.
    """
    n = a.shape[0]
    diff = conv - avg
    rev = diff[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Rf = np.fft.rfft(rev, n=fft_len)
    full = np.fft.irfft(Af * Rf, n=fft_len)[: n + diff.size - 1]   # length 3n‑2
    idx = diff.size - 1 - np.arange(n)          # = 2n‑2‑i
    g = 2.0 * full[idx]
    return g


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _compute_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (sum(a) is assumed ≈1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Optimisation stages

def _run_l2_gradient(a0: np.ndarray, deadline: float, start_time: float,
                     max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Mirror‑descent minimisation of the L² deviation of the self‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    avg = 1.0 / (2 * n - 1)
    fft_len = _next_pow2(3 * n)   # enough for correlation
    best_a = a.copy()
    best_score = _compute_score(a)
    it = 0
    eta0 = 0.4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        grad = _grad_l2(a, conv, avg, fft_len)

        # exponentiated‑gradient update (mirror descent on the simplex)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a /= a.sum()

        cur_score = _compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] L2‑grad improve: iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


def _run_exponential_gradient(a0: np.ndarray, n: int, deadline: float,
                              start_time: float, alpha_schedule,
                              eta0: float = 0.3,
                              steps_per_alpha: int = 8000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the log‑sum‑exp surrogate of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        # Adam‑style adaptive steps (mirroring momentum)
        beta1, beta2 = 0.9, 0.999
        if it == 1:
            m = np.zeros_like(a)
            v = np.zeros_like(a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta = eta0 / (np.sqrt(v_hat) + 1e-8)

        upd = np.exp(np.clip(-eta * m_hat, -50.0, 50.0))
        a = a * upd
        a /= a.sum()

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


def _run_gradient_descent(a0: np.ndarray, deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """Projected gradient descent with the exact sub‑gradient of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    momentum = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum
        a = _simplex_projection(v)

    return best_a, best_score


def _run_annealing(a0: np.ndarray, deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing with mass‑transfer moves."""
    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 0.8

    # --------------------------------------------------------------
    # Problem size – we keep the classic length 4800 (reasonable compromise)
    # --------------------------------------------------------------
    n = 4800

    # --------------------------------------------------------------
    # Gather a few deterministic starters (uniform, historic, power‑law families)
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # historic if available
    try:
        hist = np.asarray(height_sequence_1, dtype=float)
        if hist.ndim == 1 and hist.size > 0:
            candidates.append(_upsample(hist, n))
    except Exception:
        pass

    # power‑law edge‑heavy families: a_i ∝ ((i+1)*(n-i))**(-p)
    powers = [0.1, 0.25, 0.5, 0.75, 1.0]
    for p in powers:
        idx = np.arange(n)
        a = 1.0 / ((idx + 1) * (n - idx)) ** p
        a /= a.sum()
        candidates.append(a)

    # mixtures of uniform and power‑law
    alphas = [0.0, 0.25, 0.5, 0.75, 1.0]
    for p in [0.5, 0.75]:
        idx = np.arange(n)
        a_pow = 1.0 / ((idx + 1) * (n - idx)) ** p
        a_pow /= a_pow.sum()
        for α in alphas:
            mix = α * np.full(n, 1.0 / n) + (1 - α) * a_pow
            mix /= mix.sum()
            candidates.append(mix)

    # random restarts (small batch)
    for _ in range(3):
        r = np.random.rand(n)
        r /= r.sum()
        candidates.append(r)

    # ------------------------------------------------------------------
    # Pick the best of the cheap candidates
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for i, cand in enumerate(candidates):
        score = _compute_score(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
        print(f"[{time.time() - start_time:.2f}s] seed‑candidate {i} score={score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – L²‑flattening (mirror descent)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        stage_deadline = time.time() + 0.25 * remaining
        a_l2, score_l2 = _run_l2_gradient(best_seq, stage_deadline, start_time)
        if score_l2 < best_score:
            best_score = score_l2
            best_seq = a_l2.copy()
            print(f"[{time.time() - start_time:.2f}s] L2 stage improve: score={score_l2:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient on smoothed max
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        stage_deadline = time.time() + 0.45 * remaining
        alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
        a_exp, score_exp = _run_exponential_gradient(
            best_seq, n, stage_deadline, start_time,
            alpha_schedule=alpha_sched,
            eta0=0.25,
            steps_per_alpha=6000,
        )
        if score_exp < best_score:
            best_score = score_exp
            best_seq = a_exp.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad final improve: score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        stage_deadline = time.time() + 0.25 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq, stage_deadline, start_time,
            beta=0.94, step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing refinement (uses any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq, deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
