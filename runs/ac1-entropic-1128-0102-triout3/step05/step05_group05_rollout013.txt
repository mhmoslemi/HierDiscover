def project_to_simplex(v):
    # v: np.ndarray positive maybe negative
    # Implementation: sort descending, find threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Gradient‑descent on the active convolution peak.
-------------------------------------------------
The evaluation function is scale‑invariant, so we keep the
sequence normalized to sum = 1.  For a fixed length *n* the
objective reduces to 2·n·max(a∗a).  The worst convolution entry
is a piecewise‑linear function of *a*.  Its gradient (when the
maximum is attained at a unique shift *k*) is

    ∂/∂a_i max(a∗a) = 2·a_{k‑i}      (zero if the index is out of range).

We repeatedly move the mass away from coordinates that contribute
most to the current peak, using a multiplicative (exponential) update
and projecting back onto the probability simplex.  The step size
decreases slowly so the method behaves like a Frank‑Wolfe/gradient
descent hybrid.  The algorithm starts from the previously found
sequence (global ``height_sequence_1``) if it exists and runs until
the 1000 s budget is exhausted, always keeping the best candidate
seen.

The returned list obeys all constraints of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper: projection of a vector onto the simplex {x>=0, sum(x)=1}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    # based on the algorithm of Wang & Carreira‑Perpiñán (2013)
    n = v.size
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the last index where u_i > (cumsum_i-1)/i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are <= 0 -> return a vertex of the simplex
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise `evaluate_sequence`.  The routine starts from the best-known
    sequence (global `height_sequence_1` if present) and performs a
    projected gradient descent on the active convolution peak.
    """
    np.random.seed(seed)
    start_time = time.time()
    TIME_LIMIT = budget_s - 0.5                # safety margin

    # --------------------------------------------------------------
    # 1.  Initialise a feasible sequence.
    # --------------------------------------------------------------
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        # apply the same clipping that evaluate_sequence does
        tmp = np.clip(tmp, 0.0, 1000.0)
        if tmp.ndim == 1 and tmp.size > 1 and tmp.sum() >= 0.01:
            seq = tmp / tmp.sum()               # normalise to sum = 1
            n = seq.size
        else:
            raise ValueError
    except Exception:
        # fallback: a uniform vector of moderate length
        n = 3000
        seq = np.full(n, 1.0 / n, dtype=float)

    # ensure feasibility (non‑negative, sum > 0)
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.sum() < 0.01:                         # degenerate case
        seq = np.full(n, 1.0 / n, dtype=float)

    # --------------------------------------------------------------
    # 2.  Helper to compute the (scale‑invariant) score.
    # --------------------------------------------------------------
    def _score(v: np.ndarray) -> float:
        # sum(v) is 1.0 (by construction) → denominator disappears
        conv = np.convolve(v, v)
        return 2.0 * n * conv.max()

    best_seq = seq.copy()
    best_score = _score(best_seq)

    # --------------------------------------------------------------
    # 3.  optimisation loop (projected gradient on the active peak).
    # --------------------------------------------------------------
    eta = 0.5                     # initial step size for the exponential update
    it = 0
    last_print = start_time

    while True:
        now = time.time()
        if now - start_time > TIME_LIMIT:
            break

        # current convolution and active shift(s)
        conv = np.convolve(seq, seq)
        max_b = conv.max()
        active_shifts = np.flatnonzero(conv == max_b)

        # compute gradient of the max‑convolution entry
        G = np.zeros_like(seq)
        for k in active_shifts:
            # i runs over all indices where both a[i] and a[k‑i] exist
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            G[i] += 2.0 * seq[j]               # ∂max/∂a_i = 2·a_{k-i}

        # multiplicative (exponential) update + simplex projection
        # (guarantees positivity and keeps sum = 1)
        new_seq = seq * np.exp(-eta * G)
        new_seq = _simplex_projection(new_seq)

        # evaluate candidate
        cur_score = _score(new_seq)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            # occasional log
            elapsed = now - start_time
            print(f"[{elapsed:.1f}s] new best score {best_score:.6f} (n={n})")

        # accept the candidate for the next iteration – even if not better
        seq = new_seq

        # step‑size decay (very mild)
        it += 1
        if it % 500 == 0:
            eta = max(eta * 0.9, 1e-4)          # keep eta bounded away from zero

        # occasional random perturbation to escape shallow plateaus
        if it % 2000 == 0:
            # pick a tiny random shuffle of 1 % of the entries
            idx = np.random.choice(n, size=max(1, n // 100), replace=False)
            np.random.shuffle(seq[idx])
            seq = _simplex_projection(seq)      # re‑project after the shuffle

        # periodic progress output (every ~30 s)
        if now - last_print > 30.0:
            print(f"[{now - start_time:.1f}s] iter {it}, best {best_score:.6f}")
            last_print = now

    # --------------------------------------------------------------
    # 4.  Final sanity checks and return.
    # --------------------------------------------------------------
    if best_seq.sum() < 0.01:
        # fallback to a safe uniform vector
        best_seq = np.full(200, 1.0 / 200.0, dtype=float)

    total_time = time.time() - start_time
    print(f"[{total_time:.2f}s] returning sequence of length {len(best_seq)} "
          f"with score {best_score:.6f}")

    # The harness expects a plain Python list of floats
    return [float(x) for x in best_seq]
