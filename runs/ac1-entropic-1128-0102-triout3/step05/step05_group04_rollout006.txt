```
        a_hill, score_hill = _run_hill_climb(best_seq, hill_deadline, start_time)
        if score_hill < best_score:
            best_score = score_hill
            best_seq = a_hill.copy()
            print(f"[{time.time() - start_time:6.2f}s] hill climb improve: score={score_hill:.6f}")
    
    # Stage 6: mix with smooth Gaussian and pick best
    if time.time() < deadline:
        n = best_seq.shape[0]
        # generate gaussian shape
        x = np.arange(n)
        centre = (n-1)/2.0
        sigma = n * 0.15
        gauss = np.exp(-0.5 * ((x-centre)/sigma)**2)
        gauss /= gauss.sum()
        best_mix_seq = best_seq.copy()
        best_mix_score = best_score
        for tau in np.linspace(0.0, 0.3, 13):
            mix = (1 - tau) * best_seq + tau * gauss
            mix /= mix.sum()
            score = compute_score(mix)
            if score < best_mix_score:
                best_mix_score = score
                best_mix_seq = mix.copy()
                print(f"[{time.time() - start_time:6.2f}s] gaussian mix tau={tau:.2f} improve: score={score:.6f}")
        best_seq = best_mix_seq
        best_score = best_mix_score
    
    # Final normalization (sum to 1)
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    
    # Return list of floats
    return [float(v) for v in best_seq]
```

assistantfinalBelow is a self‑contained implementation of `propose_candidate`.  
It builds on the previous multi‑stage scheme but adds many more random restarts, longer exponentiated‑gradient phases, a simple hill‑climbing re‑allocation step based on the exact sub‑gradient of the max‑convolution, and a final mixing with a smooth Gaussian profile.  The code respects the 1000 s budget, never writes to disk and only uses the standard library together with NumPy.

```python
"""
propose_candidate – minimise the evaluation function for a step‑function.

The algorithm works in several stages (all run under the same time budget):
1.  many short random restarts on a modest size (n≈600) using an
    exponentiated‑gradient optimiser applied to a smoothed max
    (log‑sum‑exp) of the self‑convolution.
2.  up‑sample the current best vector to n≈2400 and continue the same
    optimiser for a larger time slice.
3.  up‑sample again to n≈4800 and run the optimiser for the bulk of the
    remaining time.
4.  a short projected‑gradient descent that uses the *exact* sub‑gradient
    of the (non‑smoothed) max‑convolution.
5.  a cheap hill‑climbing “mass‑transfer” phase that moves mass from
    indices with large sub‑gradient to those with small sub‑gradient.
6.  finally try convex combinations with a smooth Gaussian profile and
    keep the best result.

All iterates are kept non‑negative and normalised to sum = 1, so the call
to ``evaluate_sequence`` is always valid.  Progress messages are printed
so that the harness can see the evolution of the bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a∗a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a using all indices where conv is
    within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth (log‑sum‑exp) approximation of max(conv):
        f(a) = (1/α)·log Σ_k exp(α·conv_k)

    Returns ∂f/∂a (same shape as a).
    """
    n = a.shape[0]

    # weights w_k = exp(α·conv_k) / Σ exp(...)
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)          # stabilise
    w /= w.sum()

    # cross‑correlation a ⋆ w_rev  (gradient = 2·(a⋆w_rev))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.4,
                              steps_per_alpha: int = 4000,
                              max_iter: int = 10**7):
    """
    Multiplicative (exponentiated) updates on the smoothed‑max objective.
    Returns the best vector seen and its score (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    # FFT length sufficient for all convolutions at this stage
    fft_len = _next_pow2(3 * n - 2)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the smoothed max
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        # exponentiated (multiplicative) update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # advance smoothing parameter
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7):
    """
    Projected gradient descent with momentum using the *exact* sub‑gradient
    of max(conv).  Returns the best vector seen and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        grad = _subgradient_full(a, conv)

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


def _run_hill_climb(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    eps_frac: float = 0.20,
                    min_eps_frac: float = 0.02):
    """
    Very cheap greedy mass‑transfer hill‑climbing.  At each step the
    index with largest sub‑gradient loses a fraction of its mass,
    which is given to the index with smallest sub‑gradient.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    it = 0

    while time.time() < deadline:
        it += 1
        # exact sub‑gradient of the current max
        grad = _subgradient_full(a, conv)
        if np.all(grad == 0):
            break

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if i_max == i_min or a[i_max] <= 1e-15:
            break

        eps = a[i_max] * eps_frac
        # try a full move
        a_prop = a.copy()
        a_prop[i_max] -= eps
        a_prop[i_min] += eps
        conv_new = _conv_self(a_prop)
        new_score = 2.0 * n * conv_new.max()

        if new_score < cur_score:
            a = a_prop
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] hill‑climb step improve: "
                      f"score={new_score:.6f}")
        else:
            # try a smaller move
            eps = a[i_max] * min_eps_frac
            if eps < 1e-15:
                break
            a_prop = a.copy()
            a_prop[i_max] -= eps
            a_prop[i_min] += eps
            conv_new = _conv_self(a_prop)
            new_score = 2.0 * n * conv_new.max()
            if new_score < cur_score:
                a = a_prop
                conv = conv_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] hill‑climb tiny improve: "
                          f"score={new_score:.6f}")
            else:
                # no improvement – stop early
                break
    return best_a, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``len(seq)`` to ``new_n`` (preserves sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _compute_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (sum is always 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine follows a multi‑stage scheme while respecting the supplied
    time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – many short restarts on a modest size (n≈600)
    # ------------------------------------------------------------------
    n_small = 600
    max_restarts = 12
    slice_time = 5.0                # seconds per restart

    for restart in range(max_restarts):
        if time.time() >= deadline:
            break

        # try to start from the previous global solution if available
        if restart == 0 and 'height_sequence_1' in globals():
            try:
                starter = np.asarray(height_sequence_1, dtype=float)
                if starter.ndim == 1 and starter.size > 0:
                    a0_small = _upsample(starter / starter.sum(), n_small)
                else:
                    raise ValueError
            except Exception:
                a0_small = rng.random(n_small)
                a0_small /= a0_small.sum()
        else:
            a0_small = rng.random(n_small)
            a0_small /= a0_small.sum()

        slice_deadline = min(deadline, time.time() + slice_time)

        a_small, score_small = _run_exponential_gradient(
            a0_small,
            n_small,
            slice_deadline,
            start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.5,
            steps_per_alpha=3000,
        )
        if score_small < best_score:
            best_score = score_small
            best_seq = a_small.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage1 restart {restart} improve: "
                  f"score={score_small:.6f}")

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = _compute_score(best_seq)

    # ------------------------------------------------------------------
    # Stage 2 – upscale to medium size (n≈2400)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_mid = 2400
        a_mid = _upsample(best_seq, n_mid)
        remaining = deadline - time.time()
        slice_time = max(20.0, 0.30 * remaining)
        slice_deadline = time.time() + slice_time

        a_mid_opt, score_mid = _run_exponential_gradient(
            a_mid,
            n_mid,
            slice_deadline,
            start_time,
            alpha_schedule=[1.0, 5.0, 15.0, 30.0, 60.0, 120.0, 240.0],
            eta0=0.35,
            steps_per_alpha=6000,
        )
        if score_mid < best_score:
            best_score = score_mid
            best_seq = a_mid_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage2 improve: score={score_mid:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – upscale to large size (n≈4800) and long optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = 4800
        a_big = _upsample(best_seq, n_big)
        remaining = deadline - time.time()
        slice_time = max(30.0, 0.45 * remaining)
        slice_deadline = time.time() + slice_time

        a_big_opt, score_big = _run_exponential_gradient(
            a_big,
            n_big,
            slice_deadline,
            start_time,
            alpha_schedule=[5.0, 15.0, 30.0, 60.0, 120.0, 240.0, 480.0],
            eta0=0.25,
            steps_per_alpha=8000,
        )
        if score_big < best_score:
            best_score = score_big
            best_seq = a_big_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage3 improve: score={score_big:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – short projected‑gradient polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + min(remaining * 0.30, 60.0)   # ≤ 60 s
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            best_seq.shape[0],
            polish_deadline,
            start_time,

    # ------------------------------------------------------------------
    # Stage 5 – cheap hill‑climbing mass‑transfer refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining * 0.20, 30.0)   # ≤ 30 s
        a_hill, score_hill = _run_hill_climb(
            best_seq,
            hill_deadline,
            start_time,
            eps_frac=0.20,
            min_eps_frac=0.02,
        )
        if score_hill < best_score:
            best_score = score_hill
            best_seq = a_hill.copy()
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: score={score_hill:.6f}")

    # ------------------------------------------------------------------
    # Stage 6 – convex combination with a smooth Gaussian profile
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n = best_seq.shape[0]
        x = np.arange(n)
        centre = (n - 1) / 2.0
        sigma = n * 0.15
        gauss = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        gauss /= gauss.sum()

        for tau in np.linspace(0.0, 0.30, 13):
            mix = (1.0 - tau) * best_seq + tau * gauss
            mix /= mix.sum()
            score = _compute_score(mix)
            if score < best_score:
                best_score = score
                best_seq = mix.copy()
                print(f"[{time.time() - start_time:6.2f}s] gaussian mix τ={tau:.2f} improve: "
                      f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(v) for v in best_seq]
```
