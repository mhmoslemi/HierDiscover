"""
Optimisation of a step‑function using Adam on a soft‑max approximation of the
max‑convolution, progressive up‑sampling and a final simulated‑annealing walk.
The algorithm starts from the previously best construction (``height_sequence_1``)
if available and then refines it at lengths 9600→19200 (and optionally 38400)
within the given time budget.
"""

import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_self(a):
    """Linear convolution a * a (full length = 2*len(a)-1) using FFT."""
    n = a.shape[0]
    # next power of two for zero‑padding (fast FFT)
    size = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Af * Af, n=size)[:2*n-1]
    return conv

def _conv_rev_weight(a_rev, w):
    """Convolution of rev(a) with weight vector w (full length)."""
    n = a_rev.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    B = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * B, n=size)[:n+m-1]
    return conv

def _softmax_gradient(a, conv, temperature):
    """Gradient of soft‑max approximation of max(conv)."""
    maxc = conv.max()
    # stable soft‑max
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    grad = 2.0 * full[n-1:n-1+n]
    return grad

def _run_adam_softmax(a0, n, deadline, start_time,
                     lr0=0.4, beta1=0.9, beta2=0.999,
                     eps=1e-8,
                     T_start=2e-2, T_end=5e-5,
                     max_iter=10**7):
    """Projected Adam optimisation using a soft‑max approximation of the max."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    # estimate number of effective iterations for temperature schedule
    # we simply use 20000 as reference; actual schedule will be based on it.
    T_ref_iters = 20000.0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it}, score={cur_score:.6f}")

        T = T_start * (T_end / T_start) ** (it / T_ref_iters)

        # gradient of the smoothed max
        grad = _softmax_gradient(a, conv, T)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

def _upsample(seq, new_n):

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – Linear self‑convolution via FFT (full length = 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Af * Af, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3 – Convolution of rev(a) with a weight vector w
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a_rev.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    B = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * B, n=size)[: n + m - 1]
    return conv
