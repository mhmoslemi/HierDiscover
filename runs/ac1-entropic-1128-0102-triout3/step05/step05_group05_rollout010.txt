Implementation of projection onto simplex: The standard algorithm:

def proj_simplex(v):
    # v is np.array (maybe not sum to 1). Returns w >=0, sum=1 closest in L2.
    # Implementation: sort v descending, compute cumulative sum.
    # Reference from (Duchi et al., 2008).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0,budget_s=1000,**kwargs):
    import time, numpy as np, random
    start = time.time()
    # set random seed to seed
    np.random.seed(seed)
    random.seed(seed)

    n = 2000 # length
    best_seq = np.full(n, 1.0/n)
    best_score = evaluate_sequence(best_seq.tolist())
    # incorporate previous if exists
    try:
        prev_arr = np.asarray(height_sequence_1, dtype=float)
        if prev_arr.ndim == 1 and prev_arr.size>0:
            prev_norm = np.clip(prev_arr, 0, 1000)
            if prev_norm.sum()>=0.01:
                prev_norm = prev_norm/prev_norm.sum()
                s = evaluate_sequence(prev_norm.tolist())
                if s < best_score:
                    best_seq = prev_norm
                    best_score = s
    except Exception:
        pass

    # search loop
    # generate shape candidates
    while time.time() - start < budget_s - 0.5: # keep margin to return
        # choose random exponent alpha in [0,2]
        alpha = np.random.rand() * 2.0   # Uniform
        # shape using inverse power
        i = np.arange(n)
        # use symmetric shape: weight = ( (i+1)**(-alpha) + (n-i)**(-alpha) )
        # add epsilon to avoid zero division when alpha=0 and i=0? for alpha=0 weight = 2 ; fine.
        # Potentially add 1e-12.
        w = ( (i+1)**(-alpha) + (n - i)**(-alpha) )
        seq = w / np.sum(w)
        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time()-start:.2f}s] New best {score:.6f} with alpha={alpha:.3f}")

        # occasional perturbation of best
        if np.random.rand() < 0.1:
            # small random jitter
            seq2 = best_seq.copy()
            # choose two random indices i,j
            i1, i2 = np.random.randint(0,n,2)
            delta = np.random.rand() * 0.01  # up to 1% mass shift
            # shift mass from i1 to i2 without violating nonnegativity
            if seq2[i1] > delta:
                seq2[i1] -= delta
                seq2[i2] += delta
                # renormalize to sum 1
                seq2 = seq2 / seq2.sum()
                score2 = evaluate_sequence(seq2.tolist())
                if score2 < best_score:
                    best_score = score2
                    best_seq = seq2.copy()
                    print(f"[{time.time()-start:.2f}s] Perturb improved to {score2:.6f}")

    # final
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.full(200, 1.0/200)

    elapsed = time.time() - start
    print(f"Finished in {elapsed:.2f}s, best score {best_score:.6f}, length {len(best_seq)}")
    return best_seq.tolist()

"""
Search for a step‑function that minimizes `evaluate_sequence`.

Idea
----
* Theoretical lower bound of the objective is 2·n/(2n‑1)≈1, attained when the
  self‑convolution `a*a` is flat (all entries equal).  Uniform vectors give a
  triangular convolution and score = 2, far from the bound.
* To flatten the convolution we try a simple parametric family that puts more
  weight near the edges:
      a_i(α) ∝ (i+1)^{-α} + (n‑i)^{-α}     (α≥0)
  For α=0 the vector is uniform; increasing α concentrates mass at the ends
  and reduces the peak of the convolution.  The best α is found by scanning
  many values.
* The best candidate from the scan is then refined by cheap stochastic
  “mass‑transfer’’ moves (pick two indices, move a tiny amount of mass from one
  to the other) while always keeping the vector non‑negative and on the
  simplex.  Accepted moves are those that improve the score; occasional
  uphill moves are allowed with a simple annealing schedule.
* The whole search runs for the allotted time budget (default 1000 s) and
  returns the best sequence found.  If a previous solution `height_sequence_1`
  exists it is taken into account.

The algorithm needs only NumPy and the already imported `evaluate_sequence`
function.
"""

import time
import numpy as np
import random
from typing import List

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0   # the “-1’’ makes the sum of the projected vector equal to 1
    ind = np.arange(n) + 1
    # find the rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return zero vector
        return np.zeros_like(v)
    rho = ind[cond][-1]          # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence (list → float)."""
    return evaluate_sequence(seq.tolist())

    Return a list of non‑negative floats that (hopefully) gives a low score for
    `evaluate_sequence`.  The function runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    TIME_LEFT = lambda: budget_s - (time.time() - start_time)

    # ----------------------------------------------------------------------
    # 1.  Basic parameters
    # ----------------------------------------------------------------------
    n = 3000                     # length of the step function (fast enough for many evals)
    best_seq = np.full(n, 1.0 / n, dtype=float)   # uniform start
    best_score = _evaluate(best_seq)

    # ----------------------------------------------------------------------
    # 2.  Use a previously‑found sequence if it exists
    # ----------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() >= 0.01:
                prev = prev / prev.sum()
                sc = _evaluate(prev)
                if sc < best_score:
                    best_seq = prev.copy()
                    best_score = sc
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # 3.  Scan the one‑parameter family a_i(α)
    # ----------------------------------------------------------------------
    alphas = np.linspace(0.0, 1.0, 51)          # 0 → uniform, 1.0 is fairly edge‑biased
    for alpha in alphas:
        if TIME_LEFT() <= 0.5:
            break
        i = np.arange(n, dtype=float)
        # avoid zero in the denominator – add 1 to the left term, use n-i for the right
        w = ( (i + 1.0) ** (-alpha) + (n - i) ** (-alpha) )
        seq = w / w.sum()
        sc = _evaluate(seq)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc
            print(f"[{time.time() - start_time:.2f}s] α={alpha:.3f} → new best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4.  Randomised local improvement (simple simulated annealing)
    # ----------------------------------------------------------------------
    # parameters for the annealing schedule
    T0 = 0.01                     # initial temperature (in objective units)
    decay = 0.999                 # geometric decay per iteration

    iteration = 0
    while TIME_LEFT() > 0.5:
        iteration += 1
        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # move a small amount of mass from i to j (or the opposite)
        if best_seq[i] < 1e-12:
            continue
        # step size proportional to the source mass, but never larger than 5 % of it
        delta = best_seq[i] * np.random.rand() * 0.05
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        # maintain feasibility (non‑negative, sum = 1) by a projection
        cand = _project_to_simplex(cand)

        sc = _evaluate(cand)
        # accept if improvement; otherwise accept with probability exp(-(Δ)/T)
        if sc < best_score:
            best_seq = cand
            best_score = sc
            print(f"[{time.time() - start_time:.2f}s] iter {iteration} ↓ {best_score:.6f}")
        else:
            # annealing acceptance
            T = T0 * (decay ** iteration)
            if np.random.rand() < np.exp(-(sc - best_score) / max(T, 1e-12)):
                best_seq = cand
                best_score = sc
                print(f"[{time.time() - start_time:.2f}s] iter {iteration} (uphill) {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5.  Final sanity checks and return
    # ----------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        # fall back to a safe short uniform vector
        best_seq = np.full(200, 1.0 / 200.0, dtype=float)

    elapsed = time.time() - start_time
    print(f"[{elapsed:.2f}s] finished – best score {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
