"""
Propose a step‑function (non‑negative sequence) that minimises the
`evaluate_sequence` score.

Algorithm
---------
1️⃣  **Initial candidates** – generate many Beta(α,β) densities on a modest
    grid (n≈2000).  The Beta family includes the arcsine (α=β=0.5) and
    covers a large variety of shapes.  Keep the best few candidates.

2️⃣  **Progressive refinement** – up‑sample the current best shape to larger
    sizes (2000 → 4000 → 8000 → 9600).  At each size run a projected Adam
    optimiser that minimises a *smooth* version of the max‑convolution.
    The smoothness is controlled by a temperature‑like parameter τ which is
    increased during the run, turning the objective from a soft‑max to the
    true max.

3️⃣  **Final stochastic walk** – a cheap simulated‑annealing walk that moves
    a tiny amount of mass between random positions.  It accepts downhill
    moves always and uphill moves with the usual Metropolis probability.

4️⃣  **Safety fallback** – if something goes wrong return a uniform
    distribution.

All heavy work respects the 1000 s budget (a 0.5 s safety margin is kept).
The best sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution a * a via FFT (real‑valued)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Score (2·n·max(conv)) – sum is assumed to be 1 (simplex)
# ----------------------------------------------------------------------
def _score_from_a(a: np.ndarray) -> float:
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Upsample by linear interpolation (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Beta‑family shape (continuous density discretised)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta_: float) -> np.ndarray:
    """Return a discretised Beta(α,β) density on n points, normalised to sum 1."""
    # avoid log(0) by keeping points away from the endpoints
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space evaluation for numerical stability
    logdens = (alpha - 1.0) * np.log(x) + (beta_ - 1.0) * np.log(1.0 - x)
    dens = np.exp(logdens)
    # if α or β < 1 the density blows up near the edges – clip to a huge but
    # finite number to keep things well‑behaved.
    dens[np.isnan(dens) | np.isinf(dens)] = 0.0
    if dens.sum() == 0.0:
        dens = np.full(n, 1.0 / n)
    else:
        dens /= dens.sum()
    return dens


# ----------------------------------------------------------------------
# Adam optimiser for the *smooth‑max* of the convolution.
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              deadline: float,
              start_time: float,
              max_iters: int = 300_000,
              step0: float = 0.12,
              tau_start: float = 0.5,
              tau_end: float = 30.0,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed max‑convolution.
    Returns the best sequence seen and its (unsmoothed) score.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = a.copy()
    best_score = _score_from_a(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear convolution
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n:5d} it={it:7d} "
                  f"score={cur_score:.6f}")

        # smooth‑max weights (softmax on the *relative* difference)
        if max_val <= 0.0:
            w = np.ones_like(conv) / conv.size
        else:
            tau = tau_start + (tau_end - tau_start) * (it / max_iters)
            # use a relative scale so that τ works independently of n
            rel = (conv - max_val) / max_val
            w = np.exp(tau * rel)
            w /= w.sum()

        # gradient of smooth‑max: 2 * correlation(w, a)
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule – decay like 1/√t
        step = step0 / math.sqrt(it)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # enforce simplex constraints
        a = _project_simplex(a)

    return best_a, best_score

# Cheap simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,

                   steps: int = 300_000,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """Move a small ε from i to j, Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    score = _score_from_a(a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:

        delta = new_score - score

            a = a_prop
            score = new_score

            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal it={it:7d} "
