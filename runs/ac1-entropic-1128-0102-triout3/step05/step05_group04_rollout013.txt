def _run_hill_climb(a0, n, deadline, start_time, start_temp=1e-2, end_temp=1e-6):
    a = a0.copy()
    conv = _conv_self(a)
    best_a = a.copy()
    best_score = 2 * n * conv.max()
    cur_score = best_score
    start = time.time()
    rng = np.random.default_rng()
    total = deadline - start_time
    while time.time() < deadline:
        elapsed = time.time() - start
        T = start_temp * (end_temp / start_temp) ** (elapsed / (deadline - start))
        # compute conv and max each iteration
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2 * n * max_val
        # locate max position
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi+1, dtype=int)
        j_idx = k - i_idx
        contrib = a[i_idx] * a[j_idx]
        # find index of max contribution
        idx = int(np.argmax(contrib))
        i = i_idx[idx]
        j = j_idx[idx]
        src = i if a[i] >= a[j] else j
        # choose destination weighted by inverse a
        probs = np.maximum(1.0 - a, 1e-12)
        probs_sum = probs.sum()
        if probs_sum == 0:
            probs = np.full_like(a, 1.0 / n)
        else:
            probs = probs / probs_sum
        dst = int(rng.choice(n, p=probs))
        if dst == src or a[src] < 1e-12:
            continue
        # magnitude to move
        eps = a[src] * rng.random() * 0.3  # up to 30% of src
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        if total_new > 0:

"""
Propose a step‑function (non‑negative sequence) that tries to beat the
previous best bound 1.513167 for ``evaluate_sequence``.

Strategy
--------
1. **Start** from the best known construction (global variable
   ``height_sequence_1``) – if unavailable we fall back to a random start.

2. **Exponentiated‑gradient stage** – minimise a smoothed version of the
   ∞‑norm of the self‑convolution while staying on the probability simplex.
   This quickly drives the objective close to a good local minimum.

3. **Directed hill‑climbing (annealed mass transfer)** – repeatedly
   identify the convolution index where the current maximum occurs,
   locate the pair of coordinates that contributes the most to this peak,
   and move a fraction of the larger coordinate’s mass to a low‑mass
   location.  An annealing schedule accepts occasional worsening moves
   to escape shallow local minima.  The method works directly on the exact
   max‑norm (no smoothing) and therefore can polish the solution further.

4. **Projected‑gradient polishing** – a short Nesterov‑style gradient
   descent with the exact sub‑gradient of ``max(conv)`` is performed as a
   final touch.

All stages respect the global time budget (≈ 1000 s).  The returned
sequence is normalised to sum = 1, clipped to non‑negative numbers and
conforms to the limits required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a*a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Uses all indices whose value is within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """Gradient of   (1/α)·log Σ exp(α·conv_k) ."""
    n = a.shape[0]
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)      # stabilise
    w /= w.sum()

    # cross‑correlation a ⋆ w_rev  (gradient = 2·(a⋆w_rev))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # extract the centred part
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.3,
                              steps_per_alpha: int = 8000,
                              max_iter: int = 10 ** 7):
    """
    Multiplicative (exponentiated) updates on the smoothed max‑objective.
    Returns the best vector seen and its (unsmoothed) score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    fft_len = _next_pow2(3 * n - 2)      # safe length for all FFTs
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10 ** 7):
    """
    Projected gradient descent with momentum using the exact sub‑gradient
    of ``max(conv)``.  Useful for a short polishing phase.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum
        # simplex projection (O(n))
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


def _run_hill_climb(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    start_temp: float = 5e-2,
                    end_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Directed annealed mass‑transfer hill‑climbing.
    At each step the convolution peak is identified; the pair contributing
    most to this peak is weakened by moving a fraction of its mass to a
    low‑mass location.  An exponential cooling schedule accepts occasional
    uphill moves.
    """
    a = a0.copy()
    conv = _conv_self(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    cur_score = best_score

    rng = np.random.default_rng()
    total_time = deadline - start_time
    # linear scaling of temperature in log‑space
    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = start_temp * (end_temp / start_temp) ** (elapsed / total_time)

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        k = int(conv.argmax())

        # indices participating in the maximal convolution entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        contrib = a[i_idx] * a[j_idx]

        # pair with the largest contribution
        idx = int(np.argmax(contrib))
        i = i_idx[idx]
        j = j_idx[idx]
        src = i if a[i] >= a[j] else j

        # destination – bias towards low‑mass coordinates
        probs = np.maximum(1.0 - a, 1e-12)
        probs_sum = probs.sum()
        if probs_sum == 0:
            probs = np.full_like(a, 1.0 / n)
        else:
            probs = probs / probs_sum
        dst = int(rng.choice(n, p=probs))

        if dst == src or a[src] < 1e-12:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop /= a_prop.sum()                     # re‑normalise

        # evaluate the proposal
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        if delta < 0 or rng.random() < math.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: "
                      f"score={new_score:.6f}")

    return best_a, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length ``len(seq)`` to ``new_n`` (preserves sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines an exponentiated‑gradient phase, a directed
    hill‑climbing annealer and a short projected‑gradient polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the surrounding harness
    deadline = start_time + budget_s - 0.8

    # ----- obtain a starter sequence (the best known construction) -----
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # target length – the previous best used 4800, we keep the same
    n = 4800

    if starter is not None and starter.size == n:
        a0 = starter.copy()
    else:
        # up‑sample whatever we have, or start from a random simplex vector
        if starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – exponentiated‑gradient optimiser (smooth max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0:                     # give it at least a minute
        slice_deadline = time.time() + 0.55 * remaining
        a_exp, score_exp = _run_exponential_gradient(
            best_seq,
            n,
            slice_deadline,
            start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.35,
            steps_per_alpha=6000,
        )
        if score_exp < best_score:
            best_score = score_exp
            best_seq = a_exp.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – directed hill‑climbing / annealed mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline - 0.5  # leave a tiny margin for final polish
        a_hill, score_hill = _run_hill_climb(
            best_seq,
            n,
            hill_deadline,
            start_time,
            start_temp=5e-2,
            end_temp=1e-6,
        )
        if score_hill < best_score:
            best_score = score_hill
            best_seq = a_hill.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score_hill:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – short projected‑gradient polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 5.0:
            polish_deadline = time.time() + 0.5 * remaining
            a_pol, score_pol = _run_gradient_descent(
                best_seq,
                n,
                polish_deadline,
                start_time,
                beta=0.94,
                step0=0.25,
            )
            if score_pol < best_score:
                best_score = score_pol
                best_seq = a_pol.copy()
                print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_pol:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
