def conv_self(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    return conv

def grad_softmax_max(a, tau):
    n = a.shape[0]
    conv = conv_self(a)  # length 2n-1
    max_conv = conv.max()
    # compute weights: exp(tau*(conv - max_conv))
    exps = np.exp(tau * (conv - max_conv))
    sum_exps = exps.sum()
    w = exps / sum_exps
    # compute gradient: 2 * sum_k w_k a_{k-i}
    # Compute conv of w with reversed a
    rev_a = a[::-1]
    # convolution length = len(w) + len(rev_a) -1 = (2n-1) + n -1 = 3n-2
    fft_len = 1 << ((len(w) + len(rev_a) - 1).bit_length())
    Wf = np.fft.rfft(w, n=fft_len)
    Af = np.fft.rfft(rev_a, n=fft_len)
    grad_full = np.fft.irfft(Wf * Af, n=fft_len)  # length fft_len
    # get slice
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]  # shape n
    return grad

    best_a = a.copy()
    best_score = float('inf')

        # schedule tau

"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.  The optimiser works in several stages:

1. **Initial shape** – a discretised arcsine (Beta(½,½)) density, which is
   already close to a flat auto‑convolution.

2. **Smooth‑max projected Adam** – minimise a smooth approximation of the
   maximum of the auto‑convolution.  The smoothness is gradually reduced
   (τ increases) so that the optimiser first flattens the whole
   convolution and then focuses on the true maximum.  The Adam updates are
   projected onto the probability simplex after each step.

3. **Progressive up‑sampling** – the best sequence from the previous stage
   is linearly interpolated to a larger size (800 → 1600 → 3200 → 6400 →
   12800).  Larger `n` gives a smaller theoretical lower bound
   (≈ 1), and the Adam optimiser is run again at each resolution.

4. **Final simulated‑annealing refinement** – cheap pair‑wise mass‑transfer
   moves (move a random fraction of mass from one index to another) are
   accepted with a Metropolis criterion.  This helps escape any remaining
   local minima.

All work respects a global time budget (default 1000 s).  The function
returns the best sequence observed as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero or negative – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Gradient of a smooth approximation of max(conv) w.r.t. a
# ----------------------------------------------------------------------
def _grad_softmax_max(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute the gradient of the smooth max (log‑sum‑exp) of the
    auto‑convolution.  `tau` controls the smoothness (large -> close to max).
    The gradient is:
        g_i = 2 * sum_k w_k * a_{k-i}
    where w_k = exp(tau * (conv_k - max_conv)) / sum_j exp(...).
    """
    n = a.shape[0]
    max_c = conv.max()
    # soft‑max weights, shifted for numerical stability
    exp_shift = np.exp(tau * (conv - max_c))
    w = exp_shift / exp_shift.sum()

    # convolution of w with reversed a gives the required sums
    rev_a = a[::-1]
    L = w.shape[0] + rev_a.shape[0] - 1          # = 3n‑2
    fft_len = 1 << ((L - 1).bit_length())        # next power of two
    Wf = np.fft.rfft(w, n=fft_len)
    Af = np.fft.rfft(rev_a, n=fft_len)
    grad_full = np.fft.irfft(Wf * Af, n=fft_len)
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad

# ----------------------------------------------------------------------
# Projected Adam optimiser (smooth‑max)
# ----------------------------------------------------------------------
def _run_adam_smooth(a0: np.ndarray, n: int,
                     deadline: float,
                     start_time: float,
                     step0: float = 0.5,
                     tau_max: float = 30.0,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     report_every: int = 200) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max of the auto‑convolution.
    Returns the best sequence seen and its *raw* score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # bias‑correction powers
    beta1_pow = beta1
    beta2_pow = beta2

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # convolution and smooth‑max gradient
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"adam n={n:5d} iter={it:7d} τ={min(tau_max, 0.01*it):5.2f} "
                  f"score={cur_score:.6f}")

        # schedule τ: start small, increase linearly up to tau_max
        tau = min(tau_max, 0.01 * it)

        grad = _grad_softmax_max(a, conv, tau)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # diminishing step size
        step = step0 / math.sqrt(it)

        # update and project onto the simplex
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # update bias‑correction powers
        beta1_pow *= beta1
        beta2_pow *= beta2

        # early exit if we have run a huge number of iterations (unlikely)
        if it % 100000 == 0 and time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   max_steps: int = 800_000) -> tuple[np.ndarray, float]:
    """Very cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} "
                  f"iter={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
