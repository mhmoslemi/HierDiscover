Propose a low‑score step‑function by:

1.  Building a diverse pool of initial probability vectors (uniform, triangular,
    Hann, Beta‑shaped, exponential, Dirichlet draws and the historic best if
    available).
2.  Selecting the best of those candidates using a fast FFT‑based evaluation.
3.  Running many rounds of **exponential‑gradient (mirror‑descent)**
    optimisation on the current best vector.  Each round consists of a few
    thousand iterations with a geometric step‑size schedule.
4.  Occasionally “shaking’’ the solution – a small random perturbation followed
    by a short mirror‑descent – to escape local minima.
5.  Performing a long final mirror‑descent refinement until the time budget
    (≈1000 s) is exhausted.

All updates keep the vector on the probability simplex (non‑negative,
sum = 1).  The objective that is minimised is exactly the value returned by
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a))/sum(a)²).  The routine respects
the global deadline and prints progress information.
"""

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full convolution (real)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    # tiny negative values can appear due to numerical noise – clip them
    conv = np.maximum(conv, 0.0)
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k   w.r.t. a.
    Implements   g = 2·(a_rev * w)   via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    # conv2 length = len(w)+len(a)-1 ; extract the middle n entries
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _objective(a: np.ndarray) -> float:
    """Exact objective that evaluate_sequence would return (up to clipping)."""
    n = a.shape[0]
    max_conv, _ = _conv_max(a)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (exponential‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent on the L∞‑norm of the convolution.
    Returns the best vector seen and its objective value.
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # indices that are (almost) active
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient for the exact max‑norm
        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric schedule for η
        tau = (it - 1) / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # avoid overflow/underflow
        a = a * np.exp(upd)

        # renormalise to the simplex
        s = a.sum()
        if s > 0.0 and np.isfinite(s):
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

# Small family generators -------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised symmetric Beta distribution (α,β) on n points."""
    x = (np.arange(n) + 0.5) / n
    val = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    return val / val.sum()


def _exp_shape(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i)."""
    val = np.exp(-lam * np.arange(n))
    return val / val.sum()
