```
    variants.append(lin)

    rnd /= rnd.sum()
    variants.append(rnd)
```

```python
"""Search for a low‑peak self‑convolution step function.

The optimiser works on the probability simplex (∑aᵢ = 1) because the
evaluation is scale‑invariant.  For each candidate length it

1. builds a small pool of diverse starting vectors (uniform, cosine‑edge,
   linear ramp, random Dirichlet and a warm‑start from the previous length);
2. refines each start with an Adam optimiser that minimises a smooth
   log‑sum‑exp surrogate of the ℓ∞‑norm of a∗a;
3. finally applies a cheap simulated‑annealing “mass‑transfer’’ polish.

The whole search respects a global 1000 s budget and returns the best
sequence found as a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1)."""
    n = a.shape[0]
    size = 2 * n - 1
    nfft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv_surrogate(a: np.ndarray,
                                  conv: np.ndarray,
                                  lam: float) -> np.ndarray:
    """
    Gradient of   L_λ(a) = λ·log Σ exp(conv/λ)   w.r.t. a.
    """
    max_val = conv.max()
    # soft‑max weights, numerically stable
    weights = np.exp((conv - max_val) / lam)
    w_sum = weights.sum()
    if w_sum == 0.0:
        weights = np.full_like(conv, 1.0 / conv.size)
    else:
        weights = weights / w_sum

    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length 3·n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = True,
                  max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed objective.

    Returns the best vector encountered and its exact objective value:
    2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.25          # initial λ ≈ lam_factor·max(conv)
    lam_decay = 0.92
    lam_decay_interval = 1500

    for it in range(1, max_iters + 1):
        if time.time() >= slice_deadline:
            break

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n} it={it} score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)
        lam = max(max_conv * lam_factor, 1e-15)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv_surrogate(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.8 / math.sqrt(it)   # diminishing base step
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry (helps a lot for the autocorrelation problem)
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap annealer that repeatedly moves a small ε from a source index
    to a destination index.  Returns an improved sequence (if any).
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_best_score

    it = 0
    # geometric temperature schedule – the constants were tuned empirically
    start_T = 5e-2
    end_T = 1e-6
    max_steps = 200000

    while time.time() < deadline and it < max_steps:
        it += 1
        T = start_T * (end_T / start_T) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 15 % of the mass at i
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_conv = _conv_self(a_prop)
        new_score = 2.0 * n * new_conv.max()
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n} it={it} score={cur_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) + normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s

"""Search for a low‑peak self‑convolution step function.

The optimiser works on the probability simplex (∑aᵢ = 1) because the
evaluation is scale‑invariant.  For each chosen length it

1. builds a small pool of diverse starting vectors (uniform, cosine‑edge,
   linear ramp, random Dirichlet and a warm‑start from the previous length);
2. refines each start with an Adam optimiser that minimises a smooth
   log‑sum‑exp surrogate of the ℓ∞‑norm of a∗a;
3. finishes with a cheap simulated‑annealing “mass‑transfer’’ polish.

The whole search respects a 1000 s budget and returns the best sequence
found as a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1)."""
    n = a.shape[0]
    size = 2 * n - 1
    nfft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv_surrogate(a: np.ndarray,
                                  conv: np.ndarray,
                                  lam: float) -> np.ndarray:
    """
    Gradient of   L_λ(a) = λ·log Σ exp(conv/λ)   w.r.t. a.
    """
    max_val = conv.max()
    # soft‑max weights, numerically stable
    weights = np.exp((conv - max_val) / lam)
    w_sum = weights.sum()
    if w_sum == 0.0:
        weights = np.full_like(conv, 1.0 / conv.size)
    else:
        weights = weights / w_sum

    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length 3·n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = True,
                  max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed objective.

    Returns the best vector encountered and its exact objective value:
    2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.25          # initial λ ≈ lam_factor·max(conv)
    lam_decay = 0.92
    lam_decay_interval = 1500

    for it in range(1, max_iters + 1):
        if time.time() >= slice_deadline:
            break

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n} it={it} score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-12)
        lam = max(max_conv * lam_factor, 1e-15)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv_surrogate(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.8 / math.sqrt(it)   # diminishing base step
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry (helps a lot for this problem)
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer simulated annealing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap annealer that repeatedly moves a small ε from a source index
    to a destination index.  Returns an improved sequence (if any).
    """
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a).max()
    best_seq = a.copy()
    best_score = cur_best_score

    it = 0
    start_T = 5e-2
    end_T = 1e-6
    max_steps = 200000

    while time.time() < deadline and it < max_steps:
        it += 1
        T = start_T * (end_T / start_T) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # move up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n} it={it} score={cur_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) + normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
```
