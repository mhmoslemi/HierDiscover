def _grad_smooth_max(a, conv, alpha, fft_len):
    # compute w = exp(alpha*conv) / sum
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= np.sum(w)
    # cross-correlation a ⋆ w  (grad = 2*(a*conv w? Actually earlier they compute 2*(a * rev(w)))
    w_rev = w[::-1]
    # compute a convolve w_rev
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)
    # The length of corr is fft_len, we need entries that correspond to gradient.
    # Earlier they computed: grad = 2*(corr[ len(w_rev)-1 - np.arange(n) ] )   # careful
    # Let's replicate.
    start = len(w_rev) - 1
    grad = 2.0 * corr[start - np.arange(len(a))]
    return grad

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **LP alternating step (small size)** – for n≈800 we repeatedly solve a
   linear programme that, for a given f, finds a g maximising its ℓ₁‑norm
   under the constraint   f*g ≤ M  (M = max(f*f)).  Afterwards we mix
   f←(1‑t)f+t g and repeat.  This monotone scheme quickly drives the
   convolution peak down.

2. **Adam optimisation (medium / large size)** – the LP‑improved vector
   is line‑arly up‑sampled to n≈2400 and finally to n≈4800.  A
   log‑softmax parametrisation a=softmax(L) guarantees non‑negativity
   and unit sum.  We minimise a smoothed version of the max‑convolution
   using Adam (first‑ and second‑moment estimates).  The smoothing
   parameter (α) is increased gradually, ending in the true max.

3. **Projected sub‑gradient polishing** – a short Euclidean‑projection
   gradient descent with the exact sub‑gradient of the max‑convolution
   removes any remaining tiny excess.

All stages respect the supplied time budget (≈1000 s).  The best
sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  (1/α)·log Σ_k exp(α·conv_k)  w.r.t. a.
    Uses the FFT‑based cross‑correlation a ⋆ w_rev, where
    w_k = softmax(α·conv_k).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)
    start = len(w_rev) - 1
    grad = 2.0 * corr[start - np.arange(len(a))]
    return grad


def _score_of(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (assumed non‑negative, sum=1)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from seq to length new_n (preserves sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) (active‑set)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a using all near‑max entries."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Projected gradient descent (exact sub‑gradient)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.94, step0: float = 0.25,
                         max_iter: int = 10_000_000):
    """Momentum projected gradient descent on the true max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = _score_of(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n} iter={it:,} score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum
        # Euclidean projection onto simplex
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming alternating improvement (small n)
# ----------------------------------------------------------------------
def _run_lp_alternating(a0: np.ndarray, n: int,
                        deadline: float, start_time: float,
                        max_iter: int = 150, t_init: float = 0.2):
    """
    Alternating scheme:
        given f, solve   maximise Σ g_j
        s.t.   f * g ≤ M  (M = max(f * f)),  g ≥ 0
    then update f ← (1‑t) f + t g.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_of(a)
    t = t_init
    it = 0
    # pre‑compute offsets for Toeplitz matrix
    offsets = np.arange(n)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = np.convolve(a, a)  # small n → dense convolution is cheap
        M = conv.max()

        # Build sparse Toeplitz matrix A where A_{k,j} = a_{k‑j} (0 if out of range)
        data = np.tile(a[:, None], (1, n))          # shape (n, n), rows = constant a_i
        A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')

        c = -np.ones(n)                              # maximise Σ g → minimise -Σ g
        b_ub = np.full(2 * n - 1, M)

        res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs')
        if not res.success:
            break

        g = res.x
        if g.sum() == 0:
            break

        # Normalise g to maintain unit sum
        g = g / g.sum()

        # Convex mixture
        a = (1.0 - t) * a + t * g
        t = max(t * 0.97, 0.01)     # slowly shrink step‐size

        score = _score_of(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP iter {it} n={n} score={score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on smoothed max‑convolution
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray, n: int,
                  deadline: float, start_time: float,
                  lr0: float = 0.5,
                  beta1: float = 0.9, beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_iter: int = 10_000_000):
    """
    Adam on logits L (softmax(L) = a) to minimise 2·n·max(a*a).
    The objective is approximated by a log‑sum‑exp smoothing whose
    parameter α grows from 0.5 to ≈200 during the run.
    """
    L = np.log(a0 + 1e-12)               # initialise logits
    m = np.zeros_like(L)
    v = np.zeros_like(L)

    best_a = np.exp(L - L.max())
    best_a /= best_a.sum()
    best_score = _score_of(best_a)

    fft_len = _next_pow2(3 * n - 2)      # length sufficient for all FFTs
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # softmax → a
        maxL = L.max()
        expL = np.exp(L - maxL)
        a = expL / expL.sum()

        # convolution and smoothed gradient
        conv = _conv_self_fft(a)
        # linear schedule for α (0.5 → 200)
        alpha = 0.5 + (200.0 - 0.5) * min(1.0, it / 5000.0)

        grad_f = _grad_smooth_max(a, conv, alpha, fft_len)   # ∂/∂a of smoothed max
        grad = 2.0 * n * grad_f                              # scale to true objective

        # chain rule: dF/dL = a * (grad - dot(grad, a))
        dot = np.dot(grad, a)
        gL = a * (grad - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * gL
        v = beta2 * v + (1.0 - beta2) * (gL ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        L = L - lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate
        score = _score_of(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n} iter={it:,} score={score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Search for the best step‑function under the given time budget."""
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.6

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Stage 1 – LP alternating improvement on a modest size (n≈800)
    # ------------------------------------------------------------------
    n_small = 800
    slice_deadline = time.time() + max(40.0, 0.15 * (deadline - time.time()))

    # starter from previous global if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    # first initial guess (down‑sample previous best if possible)
    if starter is not None:
        a0_small = _upsample(starter, n_small)
    else:
        a0_small = np.random.rand(n_small)
        a0_small /= a0_small.sum()

    # perform a few restarts
    for restart in range(3):
        if time.time() >= deadline:
            break
        if restart == 0:
            a0 = a0_small.copy()
        else:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()

        a_opt, score_opt = _run_lp_alternating(
            a0, n_small, slice_deadline, start_time,
            max_iter=140, t_init=0.2)

        if score_opt < best_score:
            best_seq = a_opt.copy()
            best_score = score_opt
            print(f"[{time.time() - start_time:6.2f}s] stage 1 restart={restart} improve: score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample and Adam optimisation (medium size)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = a0_small.copy()
        best_score = _score_of(best_seq)

    if time.time() >= deadline:
        # early exit – return whatever we have
        best_seq = best_seq / best_seq.sum()
        return [float(x) for x in best_seq]

    n_mid = 2400
    a_mid = _upsample(best_seq, n_mid)

    remaining = deadline - time.time()
    mid_deadline = time.time() + max(20.0, 0.45 * remaining)

    a_mid_opt, score_mid = _run_adam_opt(
        a_mid, n_mid, mid_deadline, start_time)

    if score_mid < best_score:
        best_seq = a_mid_opt.copy()
        best_score = score_mid
        print(f"[{time.time() - start_time:6.2f}s] stage 2 Adam improve: score={score_mid:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – up‑sample to full size and continue Adam optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = 4800
        a_big = _upsample(best_seq, n_big)

        # use all remaining budget for this last Adam run
        a_big_opt, score_big = _run_adam_opt(
            a_big, n_big, deadline, start_time)

        if score_big < best_score:
            best_seq = a_big_opt.copy()
            best_score = score_big
            print(f"[{time.time() - start_time:6.2f}s] stage 3 Adam improve: score={score_big:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – short projected‑gradient polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = time.time() + 0.5 * (deadline - time.time())
        a_polish, score_polish = _run_gradient_descent(
            best_seq, best_seq.shape[0],
            polish_deadline, start_time,
            beta=0.94, step0=0.25)
        if score_polish < best_score:
            best_seq = a_polish.copy()
            best_score = score_polish
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
