def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # i from lo to hi (inclusive)
        i_vals = np.arange(lo, hi+1)
        j_vals = k - i_vals
        rows.extend([k] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A
```

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.
Algorithm
---------

1. **Fast primitives**
   * Linear self‑convolution via FFT.
   * Build the linear convolution matrix A where (A·b)[k] = (f∗b)[k].

2. **Stage 0 – warm start**
   * If a previous construction ``height_sequence_1`` exists, use it,
     otherwise start from a random simplex vector.

3. **Stage 1 – cheap exponentiated‑gradient optimisation**
   * Work on a modest size (n≈800) with several random restarts.
   * Objective = smoothed max of the self‑convolution; updates are
     multiplicative (stay on the probability simplex).

4. **Stage 2 – up‑sample and continue EG optimisation**
   * Upsample the current best vector to n≈2000 and run EG for a larger
     time slice.

5. **Stage 3 – linear‑programming refinement**
   * Repeatedly (while time permits) solve the LP

        max   Σ_j b_j
        s.t.  (f ∗ b)[k] ≤ M   ∀k          (M = max(f∗f))
              b_j ≥ 0

     using ``scipy.optimize.linprog`` (HiGHS).  The returned b is
     scaled to the same ℓ₁‑norm as f and blended:
            f←(1‑t)·f + t·b_scaled   (t starts at 0.5 and is reduced).

6. **Stage 4 – projected gradient polishing**
   * Classic Euclidean projection onto the simplex with the exact
     sub‑gradient of ``max(conv)``.

7. **Stage 5 – simulated‑annealing fine tuning**
   * Small‑scale mass‑transfer moves that accept downhill or
     Metropolis‑type uphill steps.

8. **Normalisation**
   * Clip all entries to the allowed range [0, 1000] and renormalise
     so that the sum stays unchanged (the objective is scale‑invariant).

The whole routine respects the supplied time budget (≈1000 s) and prints
progress information.  The final sequence is returned as a plain Python
list of floats.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: next power of two for FFT length
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g

# ----------------------------------------------------------------------
# Gradient of a smooth max (log‑sum‑exp) approximation
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    n = a.shape[0]
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad

# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (smoothed max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.3,
                              steps_per_alpha: int = 8000,
                              max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score

# ----------------------------------------------------------------------
# Projected gradient descent (exact sub‑gradient)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] PG improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum
        # simplex projection (O(n))
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score

# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Linear up‑sampling (preserve ℓ₁‑norm)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
# Build the convolution matrix A such that (A·b) = f∗b  (linear, non‑circular)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # iterate over all possible output indices k
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_vals = np.arange(lo, hi + 1, dtype=int)
        j_vals = k - i_vals
        rows.extend([k] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

# ----------------------------------------------------------------------
# One LP refinement step (maximise ℓ₁‑norm of b under conv‑bound)
# ----------------------------------------------------------------------
def _lp_refine_step(a: np.ndarray, t: float) -> np.ndarray:
    """Return refined a; t is the blending weight (0<t≤1)."""
    n = a.shape[0]
    conv_a = _conv_self(a)
    M = conv_a.max()
    if M <= 0:
        return a  # nothing to do

    A = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)                # maximise sum(b) → minimise -sum(b)
    bounds = [(0, None)] * n

    # solve LP (HiGHS is the default high‑performance backend)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fall back to the current vector
        print("  LP failed:", res.message)
        return a

    b = res.x
    sb = b.sum()
    if sb == 0:
        return a

    # Scale b to have the same ℓ₁‑norm as a (preserves the objective scale)
    scale = a.sum() / sb
    b_scaled = b * scale

    # Convex combination
    new_a = (1.0 - t) * a + t * b_scaled

    # Keep the vector on the simplex (optional normalisation)
    s = new_a.sum()
    if s > 0:
        new_a = new_a / s * a.sum()
    else:
        new_a = np.full_like(new_a, 1.0 / n)

    return new_a

# ----------------------------------------------------------------------

    # ------------------------------------------------------------------

            if cur_score < best_score:

            # exponentially decay the blending weight

    # ------------------------------------------------------------------

            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & clipping
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # ensure the harness limits are respected
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(x) for x in best_seq]
