def _run_subgradient_opt(a0, n, slice_deadline, start_time, max_iters=inf) -> (np.ndarray, float):

    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    for it in range(1, max_iters+1):
        if time.time() > slice_deadline:
            break
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")
        # compute active mask weights
        # Using uniform weight for gradient
        active = (conv == max_b).astype(float)  # boolean mask may be multiple indices if ties
        # compute gradient: weighted sum across active maxima
        grad = _subgradient_from_active(a, active)
        # step size schedule
        eta0 = 0.5
        eta = eta0 / math.sqrt(it + 1.0)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        # renormalize to simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            # fallback to uniform
            a = np.full_like(a, 1.0 / n)
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    global_a = None
    # attempt to get initial from global height_sequence_1
    try:
        global_var = np.asarray(height_sequence_1, dtype=float)
        if global_var.ndim == 1 and global_var.size > 0:
            global_a = np.maximum(global_var, 0.0)
            if global_a.sum() > 0:
                global_a = global_a / global_a.sum()
    except Exception:
        global_a = None

    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    # Shuffle length order to avoid bias if early termination
    random.shuffle(lengths)

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # Determine slice time
        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # allocate fraction of remaining time
        slice_frac = 0.15 / (len(lengths) - idx)
        slice_deadline = time.time() + max(4.0, remaining * slice_frac)

        # Initialize a0
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # local best among candidates
        best_local_a = None
        best_local_score = float('inf')

        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            # Run subgradient optimizer
            a_opt, score_opt = _run_subgradient_opt(start_vec, n, slice_deadline, start_time)
            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] subgrad local improve n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        # record global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL new best after subgrad n={n} score={best_local_score:.6f}")

        # Run annealing after subgrad
        if time.time() < deadline:
            anneal_deadline = time.time() + max(2.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(best_local_a, n, anneal_deadline, start_time, best_score, best_seq)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # Update global_a for warm start
        global_a = best_seq.copy()

    # fallback if none
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    return [float(x) for x in best_seq]

def _run_subgradient_opt(a0, n, slice_deadline, start_time):
    a = a0.copy()
    # Keep values >0
    eps = 1e-12
    a = np.maximum(a, eps)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    eta0 = 0.8
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")
        # active mask
        active = (conv == max_b).astype(float)
        # gradient
        grad = _subgradient_from_active(a, active)
        # step size
        eta = eta0 / math.sqrt(it + 1.0)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        # renormalize
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)
        # occasionally symmetrize
        if np.random.rand() < 0.2:
            a = (a + a[::-1]) * 0.5
    return best_a, best_score

"""
Subgradient‑mirror‑descent optimizer for the self‑convolution peak.

The search works on the probability simplex (∑aᵢ = 1) because
`evaluate_sequence` is scale‑invariant.  For each candidate length we

1. initialise several plausible shapes (uniform, exponential decay,
   triangular, random and a warm‑start from the previous best);
2. run a projected exponentiated‑gradient (mirror descent) that uses the
   exact active set of the convolution peak to compute a sub‑gradient;
3. periodically symmetrise the distribution (helps because the objective
   is symmetric);
4. after the sub‑gradient phase perform a cheap annealing that moves a
   small ε of mass between two indices and accepts uphill moves with an
   exponential schedule.

The algorithm allocates its 1000 s budget over an increasing schedule of
lengths (up to 32768).  The best sequence found is returned as a plain
list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a handful of plausible starting vectors.
    All returned vectors are normalised to sum = 1.
    """
    n = base_seq.shape[0]
    variants = []

    # 1. warm‑start (if provided)
    variants.append(base_seq.copy())

    # 2. uniform
    uni = np.full(n, 1.0 / n)
    variants.append(uni)

    # 3. exponential decay from both ends (parameter β randomised)
    beta = random.uniform(0.3, 3.0)            # controls how sharp the decay is
    exp_vals = np.exp(-beta * np.arange(n)) + np.exp(-beta * np.arange(n)[::-1])
    exp_vals /= exp_vals.sum()
    variants.append(exp_vals)

    # 4. triangular (ramp up then down)
    tri = np.concatenate(
        [np.linspace(1, n // 2, n // 2, endpoint=False),
         np.linspace(n // 2, 1, n - n // 2)]
    )
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 5. a random Dirichlet draw (full support)
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants


def _subgradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution w.r.t. a.
    `active` is a (2n‑1)‑vector that holds weight = 1 for each index where
    the convolution attains its maximum and 0 elsewhere.
    The returned gradient is the average over the active set.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # convolution of the active mask with the reversed a
    full = _conv_fft(active, a_rev)               # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]            # d(max)/da_i up to a factor
    cnt = active.sum()
    if cnt > 0:
        grad = grad / cnt
    return grad


def _run_subgradient_opt(
    a0: np.ndarray,
    n: int,
    slice_deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns the best vector seen and its exact objective value
        2·n·max(a*a)   (sum(a)=1).
    """
    a = a0.copy()
    # ensure strict positivity to avoid zeros in the multiplicative update
    eps = 1e-12
    a = np.maximum(a, eps)
    a = a / a.sum()

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    eta0 = 0.8  # initial step‑size for the multiplicative update

    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}"
            )

        # active set (tolerance guards against floating‑point noise)
        tol = 1e-14
        active = (conv >= max_b - tol).astype(float)

        # sub‑gradient
        grad = _subgradient_from_active(a, active)

        # step size schedule (≈ 1/√t)
        eta = eta0 / math.sqrt(it + 1.0)

        # exponentiated (multiplicative) update
        a = a * np.exp(-eta * grad)

        # renormalise onto the probability simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)

        # occasional symmetrisation (helps because the objective is symmetric)
        if random.random() < 0.2:
            a = (a + a[::-1]) * 0.5

    return best_a, best_score


def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 80000,
) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated annealing.
    Moves a random fraction of the mass from a source index to a destination
    index and accepts the move with a temperature schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score
