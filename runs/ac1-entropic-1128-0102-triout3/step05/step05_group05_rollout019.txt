def generate_shape(p, lam, n):
    i = np.arange(n, dtype=np.float64)
    raw = (i+1)**p * np.exp(-lam * i)
    total = raw.sum()
    if total == 0:
        return None
    seq = raw / total
    return seq

"""
propose_candidate – heuristic search for a low‑Φ step function.

The evaluator
    Φ(a) = 2·n·max(conv(a,a)) / (sum(a)²)
is scale‑invariant, so we may normalise every candidate to sum = 1.
A simple one‑parameter family (exponential decay) reaches Φ≈1.8,
but a two‑parameter family

        a_i ∝ (i+1)^{p} · exp(-λ·i) ,   i = 0 … n‑1

offers a better trade‑off between the ℓ¹‑mass (which reduces the
denominator) and the peak of the autocorrelation (the numerator).
We perform a fast grid search over `(p, λ)` for a modest length
(`n≈4 000`), keep the best candidate, then locally refine the parameters
and finish with a tiny random‑swap hill‑climb.  All candidates are
clipped and normalised exactly as `evaluate_sequence` does, so the
returned list is ready for the harness.

The whole routine stops well before the 1000 s budget, printing progress
and the best Φ obtained.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) beats the current
    best known Φ≈1.513.  The search is completely self‑contained and respects
    the 1000 s time limit.
    """
    np.random.seed(seed)
    start = time.time()
    best_score = float('inf')
    best_seq = None
    best_p = None
    best_lam = None

    # ------------------------------------------------------------------
    # 1.  Use a previously found solution if it exists.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() >= 0.01:
                prev_norm = prev / prev.sum()
                s = evaluate_sequence(prev_norm.tolist())
                best_score = s
                best_seq = prev_norm
    except Exception:
        # No previous candidate – we start from scratch.
        pass

    # ------------------------------------------------------------------
    # 2.  Choose a working length.
    # ------------------------------------------------------------------
    # A length of a few thousand gives a good balance between search
    # granularity and evaluation speed.
    n = max(3000, len(best_seq) if best_seq is not None else 0)
    n = int(min(n, 6000))               # stay safely inside the 1 s / eval budget
    idx = np.arange(n, dtype=float)     # reusable index array

    # ------------------------------------------------------------------
    # 3.  Coarse grid over (p, λ).
    # ------------------------------------------------------------------
    p_grid = [0.0, 0.5, 1.0, 1.5]        # shape exponent
    lam_grid = np.logspace(-5, -2, 40)   # λ from 1e‑5 to 1e‑2 (log‑scale)

    for p in p_grid:
        # pre‑compute (i+1)^p once per p‑value
        pow_factor = (idx + 1) ** p
        for lam in lam_grid:
            # candidate vector (raw, not normalised)
            raw = pow_factor * np.exp(-lam * idx)
            total = raw.sum()
            if total == 0.0:
                continue
            seq = raw / total
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq
                best_p = p
                best_lam = lam
        # time guard
        if time.time() - start > budget_s * 0.85:
            break

    # ------------------------------------------------------------------
    # 4.  Local refinement of λ around the current best.
    # ------------------------------------------------------------------
    if best_p is not None and best_lam is not None:
        pow_factor = (idx + 1) ** best_p
        # explore a finer λ‑range (factor 5 on each side)
        lo = max(best_lam * 0.2, 1e-6)
        hi = best_lam * 5.0
        lam_fine = np.linspace(lo, hi, 120)
        for lam in lam_fine:
            raw = pow_factor * np.exp(-lam * idx)
            total = raw.sum()
            if total == 0.0:
                continue
            seq = raw / total
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq
                best_lam = lam
        # ------------------------------------------------------------------
        # 5.  Optional refinement of p (small neighbourhood).
        # ------------------------------------------------------------------
        p_fine = np.linspace(max(0.0, best_p - 0.5), best_p + 0.5, 31)
        for p in p_fine:
            pow_factor = (idx + 1) ** p
            raw = pow_factor * np.exp(-best_lam * idx)
            total = raw.sum()
            if total == 0.0:
                continue
            seq = raw / total
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq
                best_p = p

    # ------------------------------------------------------------------
    # 6.  Tiny random hill‑climb (swap a little mass between positions).
    # ------------------------------------------------------------------
    if best_seq is not None:
        seq = best_seq.copy()
        for _ in range(250):
            if time.time() - start > budget_s * 0.95:
                break
            # pick two distinct indices
            i, j = np.random.choice(n, size=2, replace=False)
            # transfer a small random amount from i to j
            delta = np.random.uniform(0.0, 0.1) * seq[i]   # up to 10 % of a_i
            new_seq = seq.copy()
            new_seq[i] = max(0.0, new_seq[i] - delta)
            new_seq[j] += delta
            # renormalise (preserves positivity)
            new_seq /= new_seq.sum()
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_seq
                seq = new_seq   # continue from the improved point

    # ------------------------------------------------------------------
    # 7.  Safety fall‑back (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.sum() < 0.01:
        # uniform fallback of length 2000
        n_fallback = 2000
        uniform = np.full(n_fallback, 1.0 / n_fallback, dtype=float)
        best_seq = uniform
        best_score = evaluate_sequence(uniform.tolist())

    elapsed = time.time() - start
    print(f"[{elapsed:.2f}s] best Φ = {best_score:.6f} with length {len(best_seq)} "
          f"(p={best_p:.3f}, λ={best_lam:.6f})")

    # Return plain Python list as required by the harness
    return [float(x) for x in best_seq]
