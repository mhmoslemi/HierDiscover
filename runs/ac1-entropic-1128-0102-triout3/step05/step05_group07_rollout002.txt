def _exp_gradient_optimize(a, max_iters, deadline, eta0=1.0, eta_end=0.0001, tol_factor=0.001, perturb_freq=5000, perturb_scale=0.001):
    n = a.shape[0]
    fft_len_conv = 1 << ((2*n-1).bit_length())
    fft_len_grad = 1 << ((3*n-2).bit_length())
    best_score = float('inf')
    best_a = a.copy()
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # mask tolerance:
        tolerance = max(1e-12, max_val * tol_factor)
        mask = conv >= max_val - tolerance
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)

"""
Propose a low‑score step‑function.

The optimizer works on a probability simplex (non‑negative entries that sum to 1).
For each chosen length `n` we generate a diverse pool of seeds (uniform,
triangular, Hann window, Dirichlet draws and sparse spikes).  Every candidate
is first refined with a few Adam steps on a smooth‑max surrogate of the
objective, then a long mirror‑descent (exponential‑gradient) phase that uses
the exact L∞‑norm of the convolution as a sub‑gradient.  The EG step uses a
geometric decay of the learning rate and a tolerance‑based mask that
covers all near‑maximal convolution entries (instead of a single index).
Occasional tiny random perturbations help to escape plateaus.  After all
sizes are processed the best sequence is refined one more time with a very
large EG run.  The routine respects the global time budget (default 1000 s)
and always returns the best vector found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Make a vector non‑negative and normalise it to sum 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Sub‑gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Implements   g = 2·(a_rev * w)   via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.07,
                   temp_end: float = 0.001) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate of 2·n·max(conv)."""
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        g = _gradient_from_weights(a, w, fft_len_grad)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score

def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 1.0,
                           eta_end: float = 1e-4,
                           tol_factor: float = 1e-3,
                           perturb_freq: int = 3000,
                           perturb_scale: float = 5e-4) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser for the L∞ convolution norm.

    The update is  a_i ← a_i·exp(−η·g_i)  with g the exact sub‑gradient.
    `tol_factor` determines how many near‑maximal entries are considered
    active (mask = conv ≥ max−tol).  Small random re‑distribution steps are
    inserted every `perturb_freq` iterations to avoid stagnation.
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask of near‑maximal entries
        tol = max(1e-12, max_val * tol_factor)
        mask = conv >= max_val - tol
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient
        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric decay of the learning rate
        eta = eta_start * (eta_end / eta_start) ** (it / max_iters)

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)   # avoid overflow in exp
        a = a * np.exp(upd)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional tiny random perturbation
        if perturb_freq and (it % perturb_freq == 0):
            eps = perturb_scale * np.random.rand()
            i = np.random.randint(n)
            j = np.random.randint(n)
            delta = min(eps, a[i])
            a[i] -= delta
            a[j] += delta
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score
