    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)

        grad = _subgradient_full(a, conv)
        # grad is subgradient of max(conv)
        # gradient of objective = 2*n*grad (since objective = 2*n*max(conv))
        # Actually gradient of cur_score w.r.t a is 2*n*grad (since cur_score = 2*n * max_b)
        # but we can use grad only and scale later.
        g = grad

        # Adam updates:
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule: step = step0 / sqrt(it)
        step = step0 / math.sqrt(it)
        update = step * m_hat / (np.sqrt(v_hat) + eps)
        a = a - update
        # ensure nonnegative and sum to 1 via simplex projection
        a = _project_simplex(a)

    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The optimiser works in several stages:

1. **Shape sweep** – a quick grid search over Beta(α,α) and Gaussian
   families (discretised on 800 points).  The best shape (already fairly
   flat in its self‑convolution) seeds the optimisation.

2. **Projected Adam** – a fast first‑order method on the probability
   simplex.  The sub‑gradient of the max‑convolution is used, Nesterov‑
   style momentum is replaced by Adam (β₁=0.9, β₂=0.999) and the iterate
   is symmetrised after each projection to enforce the natural left‑right
   symmetry of the problem.

3. **Progressive up‑sampling** – the current best sequence is linearly
   interpolated to larger lengths (800 → 1 600 → 3 200 → 6 400) and the
   Adam optimiser is run again on each scale.

4. **Final stochastic refinement** – a cheap simulated‑annealing walk
   that repeatedly moves a small amount of mass between two positions.
   The temperature schedule is geometric and the move size adapts to the
   current temperature, allowing both coarse and fine adjustments.

5. **Safety fallback** – uniform distribution if nothing else succeeded.

All stages respect the supplied time budget (default 1000 s).  The best
sequence discovered is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (active constraints only)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Symmetrisation – enforce a[i] = a[-i-1]
# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force left‑right symmetry."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (max‑convolution objective)
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              step0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex.
    Returns the best sequence seen and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    beta1_pow = 1.0
    beta2_pow = 1.0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n:5d} it={it:7d} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # diminishing step size
        step = step0 / math.sqrt(it)

        update = step * m_hat / (np.sqrt(v_hat) + eps)
        a = a - update

        # keep the iterate on the simplex and enforce symmetry
        a = _project_simplex(a)
        a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.2,
                   end_temp: float = 1e-6,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Cheap SA that moves a small ε from i to j, with a geometric temperature schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move size adapts to temperature (larger early, finer later)
        max_move = a[i] * 0.3   # up to 30 % of source mass
        eps = max_move * np.random.rand() * (T / start_temp)  # scale with T
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} it={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Utility: evaluate the (scaled) objective 2·n·max(conv) for sum‑normalised a
# ----------------------------------------------------------------------
def _eval_score(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,
    alternating between an analytical start, projected Adam optimisation,
    progressive up‑sampling and a final annealing phase, and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a one‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1. Quick sweep of analytically motivated shapes (Beta & Gaussian)
    # --------------------------------------------------------------
    n0 = 800
    best_init = None
    best_init_score = float("inf")

    # Beta(α,α) family (includes arcsine for α=0.5)
    alphas = np.concatenate((np.linspace(0.2, 0.9, 8), np.linspace(1.0, 5.0, 12)))
    xs = (np.arange(n0) + 0.5) / n0
    for alpha in alphas:
        pdf = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, alpha - 1.0)
        seq = pdf / pdf.sum()
        score = _eval_score(seq)
        if score < best_init_score:
            best_init_score = score
            best_init = seq.copy()

    # Gaussian‑like shapes (truncated and normalised)
    sigmas = [0.08, 0.10, 0.12, 0.15, 0.20]
    xs_centered = xs - 0.5
    for sigma in sigmas:
        gauss = np.exp(-0.5 * (xs_centered / sigma) ** 2)
        seq = gauss / gauss.sum()
        score = _eval_score(seq)
        if score < best_init_score:
            best_init_score = score
            best_init = seq.copy()

    # possibly use a starter supplied by the harness
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            s_score = _eval_score(starter)
            if s_score < best_init_score:
                best_init_score = s_score
                best_init = starter.copy()
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2. Initialise the optimisation
    # --------------------------------------------------------------
    if best_init is None:
        # fallback: uniform
        best_seq = np.full(n0, 1.0 / n0)
        best_score = _eval_score(best_seq)
    else:
        best_seq = best_init
        best_score = best_init_score

    print(f"[{time.time() - start_time:6.2f}s] initial best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Progressive up‑sampling + Adam optimisation
    # --------------------------------------------------------------
    stage_lengths = [800, 1600, 3200, 6400]          # growth is geometric
    # reserve ~20 % of the budget for the final annealing walk

    grad_budget = (deadline - start_time) - anneal_budget
    total_stage_weight = sum(stage_lengths)

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate time for this stage (proportional to its length)
        remaining = deadline - start_time - anneal_budget - (time.time() - start_time)
        stage_frac = n / total_stage_weight
        stage_time = max(5.0, remaining * stage_frac)
        stage_deadline = time.time() + stage_time

        # up‑sample the current best to the new length
        a0 = _upsample(best_seq, n)

        # ---------------------------------------------------------
        # Adam optimisation on the current scale
        # ---------------------------------------------------------

            a0, n, stage_deadline, start_time,
            step0=0.25, beta1=0.9, beta2=0.999, eps=1e-8,

            best_score = score_ref

    # --------------------------------------------------------------
    # 5. Optional short Adam polish after annealing if time permits
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:

            print(f"[{time.time() - start_time:6.2f}s] final Adam polish improve "
