def simplex_projection(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:
        return np.full_like(v, 1.0 / len(v))
    return v / s

def evaluate(seq):

def downsample(seq, n):
    seq = np.asarray(seq, dtype=float)
    L = len(seq)

    # linear interpolation

    # 2.  Try the analytic “arcsine” construction – it often flattens
    #     the autocorrelation and can beat the uniform baseline.
    # ----------------------------------------------------------------------
    candidate = arcsine_candidate(target_len)
    cand_score = evaluate(candidate)
    if cand_score < best_score:

    # 3.  Core greedy local search.
    # ----------------------------------------------------------------------
    a = best_seq.copy()
    n = len(a)

    # parameters

    step_frac = 0.02    # start moving at most 2 % of the source entry

        it += 1

        # Occasionally shrink step size
        if it % 5000 == 0:
            step_frac *= 0.9
            temperature *= 0.9
            step_frac = max(step_frac, 5e-4)

        # Compute the current convolution and locate the dominant shift
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # Build a contribution vector for that shift:
        # contrib[i] = a[i] * a[k_star-i] (zero if out of range)
        contrib = np.zeros(n, dtype=float)
        low = max(0, k_star - (n - 1))

        a_new[dst] += delta

        if new_score < best_score - 1e-9:

            best_score = new_score
            a = a_new

    if final.sum() < 0.01:

    print(f"Finished after {it} iterations, time {elapsed:.2f}s, best score {best_score:.6f}")

    return [float(v) for v in final]

"""
search for a low‑evaluation step‑function.

Algorithm
---------
1.  Initialise from the best known sequence (global ``height_sequence_1``) if it
    exists, otherwise from a uniform vector.
2.  Also evaluate an analytical “arcsine” shape
        a_i ∝ 1/√((i+½)(N‑½‑i))
    which tends to flatten the self‑convolution.
    Keep the better of the two as the starting point.
3.  Greedy local search:
    * compute the convolution a*a and locate the index k* with the largest
      entry (the current bottleneck);
    * each entry a_i contributes a_i·a_{k*‑i} to that bottleneck.
      Move a small amount of mass from the index with the largest contribution
      to the index with the smallest contribution;
    * re‑project onto the simplex (non‑negative, sum = 1) and keep the move
      if it improves the score (or occasionally accept it probabilistically).
    The step size is slowly decreased while the allotted time (``budget_s``)
    permits.
4.  Return the best vector found, clipped to the limits enforced by
    ``evaluate_sequence``.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def downsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linearly down‑sample a 1‑D array to ``new_len`` elements.
    ``seq`` is assumed to be a NumPy array.
    """
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq)


def arcsine_candidate(N: int) -> np.ndarray:
    """
    Analytic candidate whose entries are proportional to
        1 / sqrt( (i+0.5)*(N-0.5-i) )
    This shape is known to flatten the self‑convolution.
    The vector is returned already normalised to sum = 1.
    """
    i = np.arange(N, dtype=float)
    w = 1.0 / np.sqrt((i + 0.5) * (N - 0.5 - i))
    w = np.clip(w, 0.0, 1000.0)          # respect the clipping in evaluate_sequence
    return w / w.sum()


def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    # ``evaluate_sequence`` expects a plain Python list.
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Produce a step‑function (list of floats) with a low evaluation score.
    The routine combines a deterministic analytic start with a fast
    greedy local optimiser that moves probability mass away from the
    current convolution bottleneck.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # a small safety margin – we must return before the harness kills us
    time_limit = budget_s - 0.2

    # ------------------------------------------------------------------
    # 1.  Choose a working length.
    # ------------------------------------------------------------------
    # We keep the problem modest (≈1500 entries) – large enough to allow
    # improvement but cheap enough for many evaluations.
    target_len = 1500
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2.  Initialise from a previously found sequence if available.
    # ------------------------------------------------------------------
    try:
        prev_raw = np.asarray(height_sequence_1, dtype=float)
        if prev_raw.ndim == 1 and prev_raw.size > 0:
            # keep at most ``target_len`` entries (down‑sample if needed)
            if prev_raw.size != target_len:
                prev_raw = downsample(prev_raw, target_len)
            prev_raw = np.clip(prev_raw, 0.0, 1000.0)
            if prev_raw.sum() >= 0.01:
                prev = simplex_projection(prev_raw)
                score_prev = evaluate(prev)
                best_seq = prev
                best_score = score_prev
    except Exception:
        # No usable global sequence – ignore and fall back to uniform later.
        pass

    # ------------------------------------------------------------------
    # 3.  Uniform fallback (if we have not obtained a previous vector).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(target_len, 1.0 / target_len, dtype=float)
        best_score = evaluate(best_seq)

    # ------------------------------------------------------------------
    # 4.  Try the analytic arcsine candidate (often better than uniform).
    # ------------------------------------------------------------------
    cand = arcsine_candidate(target_len)
    cand_score = evaluate(cand)
    if cand_score < best_score:
        best_seq = cand.copy()
        best_score = cand_score
        print(f"[{time.time() - start_time:.2f}s] arcsine start improved score to "
              f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Greedy local search.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    N = len(a)

    # search parameters
    step_frac = 0.02          # start moving ≤2 % of the source entry each step
    temperature = 1e-3        # for occasional uphill moves (simulated annealing)
    it = 0
    max_iters = 200_000       # hard stop – the time check will break earlier

    while time.time() - start_time < time_limit and it < max_iters:
        it += 1

        # slowly shrink the step size and temperature
        if it % 5000 == 0:

            temperature *= 0.9

        # current convolution and the dominant index
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # contribution of each component to the bottleneck entry
        contrib = np.zeros(N, dtype=float)
        lo = max(0, k_star - (N - 1))
        hi = min(N - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i
            contrib[i] = a[i] * a[j]

        # choose source (largest contribution) and destination (smallest)
        src = int(np.argmax(contrib))
        dst = int(np.argmin(contrib))

        # if for any reason src == dst (degenerate case) fall back to random pair
        if src == dst:

        # amount of mass to move – a modest fraction of the source entry
        delta = step_frac * a[src]
        if delta <= 0.0:

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = simplex_projection(a_new)

        new_score = evaluate(a_new)

        # accept improvements outright; otherwise accept with a tiny probability
        if new_score < best_score - 1e-12:
            best_seq = a_new.copy()
            best_score = new_score
            a = a_new
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] iter {it:6d}  new best {best_score:.6f}")
        else:
            if temperature > 0 and np.random.rand() < np.exp(-(new_score - best_score) / temperature):
                a = a_new  # uphill move accepted

    # ------------------------------------------------------------------
    # 6.  Final sanity checks and return.
    # ------------------------------------------------------------------
    final = np.clip(best_seq, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.full(target_len, 1.0 / target_len, dtype=float)

    total_time = time.time() - start_time
    print(f"Finished after {it} iterations, time {total_time:.2f}s, best score {best_score:.6f}")

    # ``evaluate_sequence`` works with plain Python lists
    return [float(x) for x in final]
