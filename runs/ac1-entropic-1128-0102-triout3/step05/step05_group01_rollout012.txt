def _run_subgrad_gd(a0, n, deadline, start_time,
                    step0=0.5, beta=0.9, max_iter=1e7):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it=0
    while time.time() < deadline and it < max_iter:
        it+=1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] subgrad improve n={n} iter={it} score={cur_score:.6f}")
        # subgradient weight
        max_indices = np.where(conv >= max_val - 1e-12)[0]
        # uniform on them
        w = np.zeros_like(conv)
        w[max_indices] = 1.0 / len(max_indices)
        # gradient
        rev_a = a[::-1]
        full = _conv_rev_weight(rev_a, w)
        grad = 2.0 * full[n-1:n-1+n]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that reduces
the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv is the linear self‑convolution.
The algorithm proceeds in three stages:

1. **Exploratory restarts** on a modest size (≈2000).  
   Each restart runs a projected sub‑gradient descent on the
   true ℓ∞‑objective (gradient of the current maximum convolution
   entry).  The best candidate is kept.

2. **Progressive up‑sampling** – the best candidate is interpolated
   to larger sizes (≈4800 and finally 9600).  At each size a longer
   sub‑gradient descent refines the shape.

3. **Final simulated‑annealing** walk that moves a tiny fraction of
   mass between two positions; this often squeezes a little more
   improvement out of a near‑optimal point.

All operations respect the simplex constraint (non‑negative,
sum‑to‑one) by Euclidean projection.  The optimiser uses a simple
momentum scheme with a diminishing step size.  The whole routine
fits comfortably into the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3: convolution of rev(a) with an arbitrary weight vector w
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed sequence a_rev (length n) with a weight
    vector w (length m).  Returns the full linear convolution (length n+m-1).
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4: linear up‑sampling preserving total mass
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Core optimiser: projected sub‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _run_subgrad_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov‑style) sub‑gradient descent.
    Returns the best point seen and its score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        # full self‑convolution
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # --- sub‑gradient of the max (uniform weight on all ties) ---
        # find all indices attaining the maximum (within a tiny tolerance)
        # np.isclose works well for the tiny numbers we see.
        ties = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
        w = np.zeros_like(conv)
        if np.any(ties):
            w[ties] = 1.0 / np.count_nonzero(ties)
        else:               # fallback – should never happen
            w[np.argmax(conv)] = 1.0

        # gradient = 2 * (rev(a) * w) restricted to the central n entries
        rev_a = a[::-1]
        full = _conv_rev_weight(rev_a, w)          # length = n + (2n-1) - 1 = 3n-2
        grad = 2.0 * full[n - 1:n - 1 + n]         # central block

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # Diminishing step size (classic sqrt schedule)
        step = step0 / math.sqrt(it)

        # Gradient step + projection onto the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1     # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, performs several restarts,
    progressive up‑sampling and finishes with a short annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin for harness

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many cheap restarts on a small problem size
    # ------------------------------------------------------------------
    n_small = 2000
    n_medium = 4800
    n_large = 9600

    restarts = 12
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = starter.copy()
            if a0.shape[0] != n_small:
                a0 = _upsample(a0, n_small)
        else:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()

        a_opt, sc_opt = _run_subgrad_gd(
            a0, n_small, deadline, start_time,
            step0=1.0, beta=0.9, max_iter=10 ** 7
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] restart {r} improve: score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – medium size
    # ------------------------------------------------------------------
    if best_seq is not None:
        a0 = _upsample(best_seq, n_medium)
    elif starter is not None:
        a0 = _upsample(starter, n_medium)
    else:
        a0 = np.random.rand(n_medium)
        a0 /= a0.sum()

    a_opt, sc_opt = _run_subgrad_gd(
        a0, n_medium, deadline, start_time,
        step0=0.7, beta=0.9, max_iter=10 ** 7
    )
    if sc_opt < best_score:
        best_score = sc_opt
        best_seq = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] medium improve: score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – large size (final refinement)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a0 = _upsample(best_seq, n_large)
    else:
        a0 = np.random.rand(n_large)
        a0 /= a0.sum()

    a_opt, sc_opt = _run_subgrad_gd(
        a0, n_large, deadline, start_time,
        step0=0.5, beta=0.9, max_iter=10 ** 7
    )
    if sc_opt < best_score:
        best_score = sc_opt
        best_seq = a_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] large improve: score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Final annealing phase – cheap Metropolis walk
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        best_seq, best_score = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5,
            steps=300_000
        )
        print(f"[{time.time() - start_time:6.2f}s] final anneal score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
