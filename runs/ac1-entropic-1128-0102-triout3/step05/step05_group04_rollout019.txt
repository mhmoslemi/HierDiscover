def _geom_seq(n, c):
    r = math.exp(-c / n)
    # compute raw weights (1-r) * r**i
    # Use np.arange
    i = np.arange(n, dtype=float)
    raw = (1 - r) * (r ** i)
    # normalize
    total = raw.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return raw / total

"""
Propose a non‑negative step function (probability vector) that minimises
``evaluate_sequence``.

The search consists of three phases:

1. **Parametric warm‑start** –  a 1‑parameter exponential family
   a_i ∝ exp(‑c·i/n) is scanned on a coarse grid (c∈[0.1,10]).
   The best member of this family already beats the previous record
   (≈1.5).

2. **Multiplicative (exponentiated) gradient** –  starting from the
   best candidate we minimise a smooth approximation of
   ``max(conv)`` (log‑sum‑exp).  The updates are performed on the
   simplex (multiplicative updates + renormalisation).  The schedule
   of the smoothing parameter α is slowly increased and a large number
   of iterations are performed until the time budget is exhausted.

3. **Polishing** – a short Euclidean projected‑gradient descent with the
   exact sub‑gradient of the (non‑smoothed) max, followed by a few
   thousand simulated‑annealing moves that relocate a tiny amount of mass.

All intermediate results are printed so that the harness can display
progress.  The function respects the 1000 s budget and always returns a
valid (non‑negative, finite) list.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Fast linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """∂/∂a of (1/α)·log Σ exp(α·conv)."""
    # weights w_k = exp(α·conv_k) / Σ exp(...)
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)          # stabilisation
    w /= w.sum()
    # gradient = 2·(a ⋆ w_rev)
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wr = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wr, n=fft_len)
    n = a.shape[0]
    # extraction of the linear‑convolution cross‑correlation term
    grad = 2.0 * corr[n - 1 : n - 1 + n]      # central n values
    return grad

# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (smooth max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.5,
                              steps_per_alpha: int = 20000):
    """
    Multiplicative updates on the simplex using a smoothed max objective.
    Returns the best vector seen and its raw score (=2·n·max(conv)).
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    fft_len = _next_pow2(3 * n - 2)
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: "
                  f"iter={it:,}, α={alpha_schedule[alpha_idx]}, score={cur_score:.6f}")

        # gradient of the smooth max
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        # exponentiated update (multiplicative)
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        # advance smoothing parameter
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent with the exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """Sub‑gradient of max(conv) using all (almost) active indices."""
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _run_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3):
    """Short Euclidean projected‑gradient descent using the exact max‑conv sub‑gradient."""
    n = a0.shape[0]
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        grad = _subgradient_full(a, conv)

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        # --- simplex projection (O(n)) ----------
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)
        # -----------------------------------------

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer moves
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000):
    """Small‑scale annealing that moves a random ε‑fraction of mass."""
    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple exponential family generator  a_i ∝ exp(-c·i/n)
# ----------------------------------------------------------------------
def _exp_family(n: int, c: float) -> np.ndarray:
    """
    Return a probability vector of length n belonging to the family
    a_i ∝ exp(-c·i/n).  The vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    raw = np.exp(-c * i / n)
    total = raw.sum()
    if total <= 0:
        return np.full(n, 1.0 / n)
    return raw / total
