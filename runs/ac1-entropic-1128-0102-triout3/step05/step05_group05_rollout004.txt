Pseudo code:

def proj_simplex(v):
    """Project onto the probability simplex."""
    # v can have negative entries.
    n = len(v)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1
    rho = max(i for i in range(n) if u[i] > cssv[i] / (i+1))
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

Propose a low‑peak step‑function.

The evaluation is   Φ(a)=2·n·max(a*a)/(sum(a)²) .
Because the score is scale‑invariant we keep all candidates on the
probability simplex (sum = 1).  The algorithm works in three stages:

1.  Load a previous solution (if any) and keep the better of it and a
    plain uniform vector.
2.  Generate a large number of diverse random candidates
    (Beta‑shapes, sin‑powers, exponentials, polynomials, Dirichlet)
    and keep the best one.
3.  Refine the best candidate with a projected sub‑gradient method.
    At each iteration we locate the index of the maximum convolution,
    compute the sub‑gradient g_i=2·a_{k‑i} (zero outside the support),
    take a small step a←a−η·g, clip negatives to 0 and renormalise.
    The step size η decays as 1/√t.  The current best sequence and its
    score are printed regularly.  The whole search stops before the
    allotted time budget (default 1000 s) and the best sequence found
    is returned as a plain Python list.

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Return a list of non‑negative numbers (a step‑function) that minimizes
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and keeps the best sequence found.

    def proj_simplex(v):
        """Euclidean projection of a vector onto the probability simplex."""
        v = np.asarray(v, dtype=float)
        n = v.shape[0]

        ind = np.arange(n) + 1
        cond = u - cssv / ind > 0

        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w

        try:

        except Exception:
            return float("inf")

    # ----------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=float)
    best_seq = uniform.copy()
    best_score = eval_score(best_seq.tolist())

    if prev is not None:
        score_prev = eval_score(prev.tolist())
        if score_prev < best_score:
            best_seq = prev.copy()
            best_score = score_prev

    print(f"[{0.0:.2f}s] initial best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 1) Randomised exploration of a few analytic families
    # ----------------------------------------------------------------------
    def random_dirichlet(k):
        """Draw a point uniformly from the simplex (k dimensions)."""
        # using exponential variables
        x = -np.log(np.random.rand(k))
        x /= x.sum()
        return x

    families = ["dirichlet", "beta", "sin", "poly", "exp"]
    for _ in range(400):
        typ = np.random.choice(families)
        if typ == "dirichlet":
            a = random_dirichlet(n)
        elif typ == "beta":
            a = np.arange(1, n + 1, dtype=float) / n  # x in (0,1]
            alpha = np.random.uniform(0.1, 5.0)
            beta = np.random.uniform(0.1, 5.0)
            a = (a ** (alpha - 1.0)) * ((1.0 - a) ** (beta - 1.0))
        elif typ == "sin":
            gamma = np.random.uniform(0.2, 5.0)
            a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** gamma
        elif typ == "poly":
            p = np.random.uniform(-2.0, 2.0)
            i = np.arange(n, dtype=float)
            a = (i + 1) ** p + (n - i) ** p
        else:  # "exp"
            lam = np.random.uniform(0.0, 5.0)
            a = np.exp(-lam * np.arange(n, dtype=float))

        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a.sum()

        sc = eval_score(a.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] new best from {typ} {sc:.6f}")

    # ----------------------------------------------------------------------
    # 2) Projected sub‑gradient descent starting from the best sequence found.
    # ----------------------------------------------------------------------
    a = best_seq.copy()
    max_iters = 10000  # generous – we will break on time
    for it in range(max_iters):
        # stop if we are about to exceed the time budget
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # convolution and location of the current maximum
        conv = np.convolve(a, a)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of the active max‑entry w.r.t. a
        g = np.zeros_like(a)
        # i such that 0 ≤ k‑i < n  →  max(0, k-(n-1)) ≤ i ≤ min(k, n-1)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx_i = np.arange(lo, hi + 1, dtype=int)
            idx_j = k - idx_i
            g[idx_i] = 2.0 * a[idx_j]

        # step size (decaying)
        eta = 0.5 / np.sqrt(it + 1.0)

        # gradient step, clipping and renormalising
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:          # degenerate – fall back to uniform
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= s

        # occasional evaluation to keep track of improvements
        if it % 10 == 0:
            sc = eval_score(a_new.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a_new.copy()
                print(f"[{elapsed:.2f}s] iter {it:5d} improved score {sc:.6f}")

        a = a_new

    # ----------------------------------------------------------------------
    # 3) Final sanity check (very unlikely to be triggered)
    # ----------------------------------------------------------------------
    if best_seq.sum() < 0.01:
        best_seq = np.full(n, 1.0 / n)
        best_score = eval_score(best_seq.tolist())

    total_time = time.time() - start_time
    print(f"[{total_time:.2f}s] finished – best score {best_score:.6f} "
          f"length {len(best_seq)}")

    # Return a plain list as required by the harness
    return [float(x) for x in best_seq]
