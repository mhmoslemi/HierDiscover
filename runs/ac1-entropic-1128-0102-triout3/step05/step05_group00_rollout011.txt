def _solve_lp(f, max_conv):
    n = f.shape[0]
    # Build A_ub matrix
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)

    row_idx = np.concatenate([np.arange(i, i+n) for i in range(n)])
    col_idx = np.tile(np.arange(n), n)
    data = np.repeat(f, n)  # each f[i] repeated n times
    from scipy.sparse import csr_matrix
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))

    # constraints
    b = np.full(2*n-1, max_conv)

    # solve
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    # Avoid near-zero sum
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    # Normalize to sum=1 (or keep as is? We'll rescale later)
    g = g / sum_g
    return g

def _lp_refine(a, n, slice_deadline, start_time):
    a_curr = a.copy()
    best_score = 2*n * _conv_self(a_curr).max()
    best_a = a_curr.copy()

    return best_a, best_score

"""
Improved step‑function search.

The optimiser works on the probability simplex (∑aᵢ = 1) because the
`evaluate_sequence` metric is scale‑invariant.  It consists of three
stages:

1. **LP‑guided shape improvement** (only for modest lengths ≤ 2048).
   For a current distribution `f` we compute M = max(f*f).  An LP
   maximises the total mass of a new distribution `g` under the linear
   constraints `f*g ≤ M`.  After normalising `g` we mix it with `f`
   (convex combination) and project back onto the simplex.  Repeating
   this quickly reduces the convolution peak.

2. **Adam optimisation** – the same smooth‑max surrogate as in the
   previous baseline, but now started from the LP‑refined shape and
   with a more aggressive learning‑rate schedule.

3. **Mass‑transfer annealing** – a cheap simulated‑annealing polishing
   that moves a small ε from a source index to a destination index.
   Temperature is annealed geometrically; many moves are attempted
   (up to 150 k per length).

The algorithm walks a length schedule `[256, 512, 1024, 2048, 4096,
8192, 16384, 32768]`.  Small lengths receive the LP phase; larger
lengths are refined only with Adam + annealing.  The best sequence
found so far (or a supplied `height_sequence_1`) is always used as a
warm‑start for the next length.

The whole search respects the supplied time budget (default 1000 s)
and returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimisation on the smooth ℓ∞ surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best vector encountered and its exact objective value:
    2·n·max(conv) (the sequence is kept on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3          # λ ≈ lam_factor * max(conv)
    lam_decay = 0.7
    lam_decay_interval = 1500

    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score

def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from i to j.
    Returns an improved sequence (if any) and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

# LP‑based refinement (only for modest n)
# ----------------------------------------------------------------------


def _solve_lp(f: np.ndarray, max_conv: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j   subject to   f * g ≤ max_conv ,   g ≥ 0.
    Returns the optimal g (not normalised) or None on failure.
    """
    n = f.shape[0]

    # Build sparse Toeplitz matrix for convolution f * g.
    # Row k corresponds to Σ_i f_i * g_{k-i}  (indices outside [0,n) are ignored).
    i = np.arange(n)[:, None]                     # shape (n,1)
    j = np.arange(n)[None, :]                     # shape (1,n)
    row_idx = (i + j).ravel()                     # length n*n
    col_idx = np.tile(np.arange(n), n)            # length n*n
    data = np.repeat(f, n)                        # each f_i repeated n times

    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, max_conv)

    # linprog minimizes; we maximise Σ g ⇒ minimise -Σ g
    res = linprog(c=-np.ones(n),
                  A_ub=A,
                  b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return None
    return res.x


def _lp_refine(a0: np.ndarray,
               n: int,
               slice_deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑guided mixing for a small problem size (n ≤ 2048).
    Returns the best vector and its exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    # mixing factor – we keep it moderate to avoid overshooting
    t = 0.5

    while time.time() < slice_deadline:
        conv = _conv_self(a)
        M = conv.max()
        g = _solve_lp(a, M)
        if g is None:
            break
        S = g.sum()
        if S <= 0.0:
            break
        g_norm = g / S                      # back to simplex
        a = (1 - t) * a + t * g_norm
        a = _project_simplex(a)

        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP refine n={n} score={cur_score:.6f}")

        # stop after a few successful mixes if the improvement stalls
        if best_score < 1.51:   # heuristic early stop when already very good
            break

    return best_a, best_score

def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a small collection of plausible starting vectors from a base.
    The returned vectors are already normalised to sum = 1.
    """
    variants = []
    n = base_seq.shape[0]

    # 1. warm‑start itself
    variants.append(base_seq.copy())

    # 2. sinusoidal (positive half‑wave)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3. triangular ramp
    tri = np.concatenate([np.linspace(1, n // 2, n // 2, endpoint=False),
                          np.linspace(n // 2, 1, n - n // 2)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4. random Dirichlet
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants
