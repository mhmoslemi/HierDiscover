def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    def evaluate(a):
        m, _ = conv_max(a)
        return 2*len(a)*m

    # length schedule
    n_values = [500, 800, 1200, 1800, 2500, 3600, 5400, 8000]
    for n in n_values:
        if time.time() >= deadline:
            break
        print(f"Stage n={n}")

        # generate initial candidates
        init_candidates = []

        # include previous best upsample/dowmsample
        if best_seq is not None:
            # upsample or downsample to length n
            x_old = np.linspace(0,1,best_seq.shape[0])
            x_new = np.linspace(0,1,n)
            a_resampled = np.interp(x_new, x_old, best_seq)
            init_candidates.append(project_simplex(a_resampled))

        # uniform
        init_candidates.append(np.full(n, 1.0/n))
        # triangle
        idx = np.arange(n)
        tri = np.minimum(idx+1, n-idx)
        init_candidates.append(project_simplex(tri.astype(float)))
        # hann
        if n>1:
            hann = 0.5*(1-np.cos(2*np.pi*idx/(n-1)))
            init_candidates.append(project_simplex(hann))
        # beta shapes
        for alpha in [0.2, 0.5, 0.8, 1.0, 1.2, 2.0, 3.0]:
            a = ( (np.arange(1, n+1)-1 + 1e-12) ** (alpha-1) * (np.arange(n,0,-1) + 1e-12) ** (alpha-1) )
            init_candidates.append(project_simplex(a))
        # sin power
        for p in [0.5, 1, 2, 3, 5]:
            a = np.sin(np.pi * (np.arange(n)+0.5) / n) ** p
            init_candidates.append(project_simplex(a))

        # random Dirichlet draws
        for _ in range(3):
            init_candidates.append(project_simplex(np.random.dirichlet(np.ones(n))))

        # For each candidate Run optimization
        for cand in init_candidates:
            if time.time() >= deadline:
                break
            # allocate iteration budget based on remaining time approx
            remaining = deadline - time.time()
            # approximate time per iter = c * n log n, we can use: 0.0008 * (n/1200)
            est_iter = int(min(20000, max(3000, remaining/ (0.001 * n/1200.0))))

            # run Adam for half, EG for half
            a_adam, score_adam = adam_optimize(cand.copy(), max_iters=int(0.4*est_iter), deadline=deadline)
            a_exp, score_exp = exp_gradient_optimize(a_adam.copy(), max_iters=int(0.6*est_iter), deadline=deadline)
            # evaluate final
            cur_score = evaluate(a_exp)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_exp.copy()
                print(f"New best: score={best_score:.6f} at n={n}")

    # final refinement using EG with more iterations
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        remaining = deadline - time.time()
        iters = int(min(500000, max(50000, remaining/ (0.001 * n/1200.0))))
        if iters>0:
            a_ref, ref_score = exp_gradient_optimize(best_seq.copy(), max_iters=iters, deadline=deadline, eta_start=0.2, eta_end=0.001)
            cur_score = evaluate(a_ref)
            if cur_score < best_score:
                best_seq = a_ref
                best_score = cur_score
                print(f"Final refinement improved: score={best_score:.6f}")

    # final cleanup ensure positivity normalization
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished: best_score={final_score:.6f}")

    return best_seq.tolist()

"""
propose_candidate – search for a low‑score non‑negative step function.

The algorithm explores a collection of parametrised shapes (uniform,
triangular, Hann, Beta‑type, sin‑power) on several problem sizes.
Each shape is normalised to the probability simplex and then refined
by two complementary optimisation phases:

*  a smooth‑max Adam stage (soft‑max surrogate of the max‑convolution)
   to quickly reach a good basin;
*  a hard‑max exponential‑gradient (mirror‑descent) stage that follows
   the exact ℓ∞‑subgradient of the convolution max.

Optionally a short stochastic hill‑climbing phase is applied to
escape shallow local minima.  All steps respect the simplex constraint
by projection or normalisation.  The routine respects a 1000 s budget
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def gradient_from_weights(a: np.ndarray, w: np.ndarray,
                         fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) with respect to `a`.
    Implements  g = 2·(a_rev * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    # the linear convolution result length is len(w)+len(a)-1
    start = a.shape[0] - 1
    g = 2.0 * conv2[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def adam_optimize(a: np.ndarray,
                  max_iters: int,
                  deadline: float,
                  alpha: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  temp_start: float = 0.07,
                  temp_end: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best vector seen and its (un‑scaled) score 2·n·max(conv).
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    max_val, _ = conv_max(a)
    best_score = 2.0 * n * max_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # smooth‑max (soft‑max) weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        g = gradient_from_weights(a, w, fft_len_grad)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back to simplex
        a = project_simplex(a)

    return best_a, best_score


def exp_gradient_optimize(a: np.ndarray,
                          max_iters: int,
                          deadline: float,
                          eta_start: float = 0.8,
                          eta_end: float = 0.001,
                          eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser using the exact max‑convolution.
    Returns best vector encountered and its un‑scaled score.
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    max_val, _ = conv_max(a)
    best_score = 2.0 * n * max_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask of (approximately) active anti‑diagonals
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        g = gradient_from_weights(a, w, fft_len_grad)

        # decreasing step size (geometric schedule)
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)      # avoid overflow in exp
        a = a * np.exp(upd)

        # renormalise to simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score


def hill_climb_optimize(a: np.ndarray,
                        max_iters: int,
                        deadline: float,
                        init_eps: float = 0.01,
                        decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a small amount of mass from one index
    to another and accept the move if it lowers the objective.
    """
    n = a.shape[0]
    eps = init_eps
    cur_a = a.copy()
    max_val, _ = conv_max(cur_a)
    best_score = 2.0 * n * max_val
    best_a = cur_a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # pick source with enough mass
        src = np.random.randint(n)
        while cur_a[src] <= eps:
            src = np.random.randint(n)
        dst = np.random.randint(n)
        if src == dst:
            continue
        delta = eps * np.random.rand()
        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # evaluate
        max_val, _ = conv_max(new_a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = new_a.copy()
            cur_a = new_a
        # decay epsilon
        eps *= decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Shape factories
# ----------------------------------------------------------------------
def shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def shape_triangle(n: int) -> np.ndarray:
    idx = np.arange(n)
    tri = np.minimum(idx + 1, n - idx)
    return project_simplex(tri.astype(float))


def shape_hann(n: int) -> np.ndarray:
    if n == 1:
        return np.array([1.0])
    idx = np.arange(n)
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
    return project_simplex(hann)


def shape_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    # avoid exact zeros for α<1 by adding a tiny offset
    i = np.arange(1, n + 1) - 1
    a = (i + 1e-12) ** (alpha - 1) * (n - i + 1e-12) ** (alpha - 1)
    return project_simplex(a)


def shape_sin_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p(π(i+0.5)/n)   – zero at the ends."""
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    return project_simplex(a)


def shape_random_dirichlet(n: int) -> np.ndarray:
    return project_simplex(np.random.dirichlet(np.ones(n)))


def shape_random_spikes(n: int, k_min: int = 3, k_max: int = 5) -> np.ndarray:
    spikes = np.zeros(n)
    k = np.random.randint(k_min, k_max + 1)
    pos = np.random.choice(n, size=k, replace=False)
    spikes[pos] = np.random.rand(k) + 0.1
    return project_simplex(spikes)

                print(f"       NEW GLOBAL BEST – score = {best_score:.6f}")

        # extra refinement on the current best at this size (if any)
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            remaining = deadline - time.time()
            extra_iters = int(min(40_000, remaining / est_iter_time(n)))
            if extra_iters > 0:
                print(f"   [{time.time() - start_time:5.2f}s] extra EG refinement – iters={extra_iters}")
                a_ref, _ = exp_gradient_optimize(
                    best_seq.copy(),
                    max_iters=extra_iters,
                    deadline=deadline,
                    eta_start=0.4,
                    eta_end=0.0002,
                )
                ref_score = fast_score(a_ref)
                if ref_score < best_score:
                    best_score = ref_score
                    best_seq = a_ref.copy()
                    print(f"       IMPROVED AFTER EXTRA EG – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final long refinement on the overall best (hard‑max EG)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        remaining = deadline - time.time()
        final_iters = int(min(300_000, max(80_000, remaining / est_iter_time(n))))
        if final_iters > 0:
            print(f"[{time.time() - start_time:6.2f}s] Final EG refinement – n={n}, iters={final_iters}")
            a_final, _ = exp_gradient_optimize(
                best_seq.copy(),
                max_iters=final_iters,
                deadline=deadline,
                eta_start=0.2,
                eta_end=5e-4,
            )
            final_score = fast_score(a_final)
            if final_score < best_score:
                best_score = final_score
                best_seq = a_final.copy()
                print(f"   FINAL IMPROVEMENT – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback in the impossible case
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = shape_uniform(500)

    # ensure non‑negativity and normalise (required by evaluate_sequence)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    # final reporting using the supplied evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = float('nan')
    print(f"[{time.time() - start_time:6.2f}s] Finished – best score = {final_score:.6f}")

    return [float(x) for x in best_seq]
