def proj_simplex(v):
    # v is 1D np array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

        # step size
        eta = initial_eta / np.sqrt(it + 1)
        # update

"""
Propose a better step‑function sequence by stochastic local search.

The evaluation metric (with sum(a)=1) is
    Φ(a) = 2·n·max( a * a )                (a*a = discrete convolution)

Since Φ is scale‑invariant we keep the vector on the probability simplex.
The algorithm starts from the previously found sequence (if any) and repeatedly
moves a small amount of mass from a “high‑contributing” position to a random
position.  The move is kept if it reduces the current maximum of the
convolution (i.e. the objective).  A simple simulated‑annealing acceptance
criterion helps to escape shallow plateaus.  The best sequence seen during the
budget is returned.

The implementation uses only NumPy and the provided evaluate_sequence()
function; it respects the time budget (default 1000 s) and never returns a
negative‑inf score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("input must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Main routine expected by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Stochastic local search for a sequence that minimizes `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        RNG seed for reproducibility.
    budget_s : int
        Approximate time budget in seconds (the function will stop a little
        before this limit to avoid a hard timeout).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + 0.95 * budget_s   # a safety margin

    # --------------------------------------------------------------
    # 1. Initialise a candidate.
    # --------------------------------------------------------------
    # Try to use the previously discovered sequence; otherwise fall back
    # to a uniform distribution of modest size.
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim != 1 or prev.size == 0:
            raise ValueError
        # Apply the same clipping that evaluate_sequence does.
        a = np.clip(prev, 0.0, 1000.0)
        if a.sum() < 0.01:                     # protect against tiny sum
            raise ValueError
        a = a / a.sum()
        n = a.shape[0]
    except Exception:
        n = 2000                     # a safe default length
        a = np.full(n, 1.0 / n, dtype=float)

    # current best solution
    cur_a = a.copy()
    cur_score = 2.0 * n * np.max(np.convolve(cur_a, cur_a))
    best_a = cur_a.copy()
    best_score = cur_score

    print(f"[0.00s] start with length {n}, initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Stochastic local improvement loop.
    # --------------------------------------------------------------
    # Parameters controlling the search.
    max_no_improve = 200_000          # after this many failed moves we randomise a bit
    no_improve = 0
    iter_cnt = 0

    while time.time() < deadline:
        # ---- pick a shift that currently attains the maximum ----
        conv = np.convolve(cur_a, cur_a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))            # index of the maximal convolution entry

        # ---- compute a contribution weight for each index i ----
        # contribution of a_i to conv[k_max] is a_i * a_{k_max-i} (if the partner exists)
        contrib = np.zeros_like(cur_a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        # indices i in [lo, hi] have a valid partner j = k_max - i
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k_max - idx
        contrib[idx] = cur_a[idx] * cur_a[partner]

        # if for some reason all contributions are zero (should not happen) fall back to uniform sampling
        if contrib.sum() == 0.0:
            i = np.random.randint(0, n)
        else:
            probs = contrib / contrib.sum()
            i = np.random.choice(n, p=probs)

        # ---- choose a distinct target index j uniformly ----
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1  # ensure j != i

        # The fraction is sampled adaptively; we never move more than 30% of the source.
        max_transfer = cur_a[i] * 0.3

            continue

        new_a[j] += delta

        new_score = 2.0 * n * np.max(new_conv)

        # ---- acceptance test ----

            # temperature linearly decays to 0.01
            elapsed = time.time() - start_time
            T = 0.01 + 0.5 * max(0.0, (deadline - start_time - elapsed) / (deadline - start_time))
            prob = np.exp(-(new_score - cur_score) / T)
            if np.random.rand() < prob:
                cur_a = new_a
                cur_score = new_score
            no_improve += 1

        iter_cnt += 1

        # ---- random perturbation if stuck for a while ----
        if no_improve >= max_no_improve:
            # small Gaussian perturbation followed by projection onto simplex
            noise = np.random.randn(n) * 0.02
            cur_a = cur_a + noise
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = _project_onto_simplex(cur_a)
            cur_score = 2.0 * n * np.max(np.convolve(cur_a, cur_a))
            no_improve = 0
            # keep the global best unchanged here – we just shake the current point

    # --------------------------------------------------------------
    # 3. Return the best sequence found (normalise once more for safety).
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    # final sanity check via the official evaluator (mostly for debugging)
    try:
        final_eval = evaluate_sequence(best_a.tolist())
        print(f"Final evaluated score {final_eval:.6f}")
    except Exception:
        pass

    return best_a.tolist()
