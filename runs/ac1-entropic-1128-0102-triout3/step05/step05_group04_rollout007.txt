def _run_exp_subgrad(a0, n, deadline, start_time, step0=0.3):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it=0
    while time.time()<deadline:
        it+=1
        conv = _conv_self(a)
        cur_score = 2*n*conv.max()
        if cur_score<best_score:
            best_score=cur_score
            best_a=a.copy()
            print([...])
        # subgradient of max conv
        grad = _subgradient_full(a, conv)  # shape n
        eta = step0 / math.sqrt(it)
        # multiplicative update
        upd = np.exp(-eta * grad)

        s = a.sum()

    return best_a,best_score

"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Utility functions**
   • Fast linear convolution via FFT.
   • Exact sub‑gradient of the (non‑smoothed) max of the self‑convolution.
   • Gradient of the smooth (log‑sum‑exp) surrogate of the max.
   • Euclidean projection onto the probability simplex.

2. **Multi‑stage optimisation (time‑aware)**
   • *Stage 1* – exponentiated‑gradient on the smooth surrogate with a long
     schedule of increasing smoothing parameters (α up to ≈ 5000).  This
     quickly lowers the objective while keeping the iterate inside the
     simplex.
   • *Stage 2* – exponentiated‑gradient using the *exact* sub‑gradient of the
     true max (no smoothing).  Gives a more aggressive push on the current
     bottleneck entries.
   • *Stage 3* – short projected‑gradient descent with momentum (the classic
     polishing phase of the reference implementation).
   • *Stage 4* – simulated‑annealing “mass‑transfer’’ moves for a final
     refinement.

All stages stop when the global time budget (≈ 1000 s) is exhausted.
The best vector found is normalised to sum = 1 and returned as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) (active set)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a using all indices where conv is
    within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) approximation of max(conv)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (same shape as a).
    """
    n = a.shape[0]

    # soft‑max weights over the conv entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)          # stabilise
    w /= w.sum()

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w_rev))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    # extract the n entries that correspond to the gradient
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the *smooth* max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _run_exp_grad_smooth(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         alpha_schedule,
                         eta0: float = 0.4,
                         steps_per_alpha: int = 4000) -> tuple[np.ndarray, float]:
    """
    Multiplicative updates using the gradient of the smoothed max.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    # FFT length sufficient for the gradient computation
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑smooth improve: iter={it:,}, "
                  f"α={alpha_schedule[min(alpha_idx, n_alpha-1)]:.1f}, score={cur_score:.6f}")

        # current smoothing parameter
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        # exponential (multiplicative) update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a = _project_simplex(a)

        # move to the next smoothing parameter
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the *exact* max (sub‑gradient)
# ----------------------------------------------------------------------
def _run_exp_subgrad(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     step0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Multiplicative updates using the exact sub‑gradient of max(conv).
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑subgrad improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        eta = step0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent with momentum (exact sub‑gradient)
# ----------------------------------------------------------------------
def _run_proj_grad(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   beta: float = 0.94,
                   step0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Classic projected gradient descent with momentum using the exact
    sub‑gradient of max(conv).  Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient of the non‑smoothed max
        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that moves a random ε‑fraction of mass from a
    source index (chosen proportionally to its current weight) to a uniformly
    chosen destination.  Returns (best_seq, best_score).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.35
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: iter={it:,}, score={score:.6f}")

    return best_seq, best_score
