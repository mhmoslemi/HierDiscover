"""
Propose a low‑maximum self‑convolution step‑function.

Algorithm outline
-----------------
* Multi‑scale search: start with a modest length, gradually up‑sample the
  best vector to larger lengths.
* Warm‑starts from three simple parametric families (power‑product,
  Gaussian, Beta) plus a random start.
* Projected sub‑gradient descent on the probability simplex using the
  *exact* ∞‑norm of the self‑convolution.  The gradient is computed from
  all currently active (maximal) convolution entries.
* Cheap mass‑transfer annealing (random ε‑moves) for polishing.
* All work respects the global time budget (default 1000 s) and returns
  the best sequence found (as a plain Python list of non‑negative
  heights).
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    # Numerical noise may produce tiny negative values; clip to zero.
    np.maximum(conv, 0.0, out=conv)
    return conv


def score_of(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a sequence with sum=1."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation + renormalisation)
# ----------------------------------------------------------------------


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a sequence to length new_n and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for quick warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p (p>0 centre‑heavy, p<0 edge‑heavy)."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term <= 0, eps, term)
    a = term ** p
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ in index units)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact ∞‑norm)
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex.
    Returns the best vector seen (sum=1) and its exact score.
    """
    a = a0.copy()
    conv = conv_self(a)
    best_sc = 2.0 * n * conv.max()
    best_a = a.copy()

    step0 = 0.5  # initial step size
    it = 0
    tol = 1e-12

    while time.time() < deadline:
        it += 1

        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # Determine all active convolution indices (within tolerance)
        active = np.flatnonzero(conv >= max_val - tol)

        # Compute sub‑gradient: sum of gradients from all active indices
        grad = np.zeros_like(a)
        indices = np.arange(n)
        for k in active:
            j = k - indices
            mask = (j >= 0) & (j < n)
            grad[mask] += a[j[mask]]

        # Each active index contributes factor 4·n (see derivation)
        grad *= (4.0 * n)

        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * direction)

        # Periodic tiny noise to escape plateaus
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 80000) -> tuple[np.ndarray, float]:
    """
    Random epsilon‑transfer moves accepted with a geometric temperature schedule.
    Returns the best sequence found during annealing.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # Geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2  # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter = starter / s
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – from small to large.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    # Ensure deterministic order but still allow early exit.
    # No shuffling to keep a predictable up‑sampling cascade.
    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm start / upsample)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # Allocate a slice of the remaining time for this length.
        # Later (larger) lengths receive a larger share.
        fraction = 0.15 / (len(lengths) - idx)  # larger for later lengths
        slice_seconds = max(6.0, remaining * fraction)
        slice_deadline = time.time() + slice_seconds

        # --------------------------------------------------------------
        # Quick stochastic search over simple parametric families
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(4.0, slice_seconds * 0.15)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2])
            if kind == 0:  # product‑power family
                p = np.random.uniform(-2.0, 2.5)
                cand = shape_product_power(n, p)
            elif kind == 1:  # Gaussian family
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            else:  # Beta family
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}")

        # --------------------------------------------------------------
        # Projected sub‑gradient descent (exact ∞‑norm)
        # --------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = subgradient_descent(local_best_a, n, grad_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={sc_opt:.6f}")

        # --------------------------------------------------------------
        # Cheap annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.20)
            a_ref, sc_ref = annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=80000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}")

        # Keep the best vector for the next up‑sample.
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Final refinement on the largest length if time remains
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # Upsample to the largest length (may already be there)
        final_n = lengths[-1]
        a0 = upsample(best_seq, final_n)

        # Use remaining time for a final gradient + annealing phase
        remaining = deadline - time.time()
        final_grad_deadline = time.time() + max(5.0, remaining * 0.6)
        a_opt, sc_opt = subgradient_descent(a0, final_n, final_grad_deadline, start_time)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] final subgrad improve n={final_n} score={sc_opt:.6f}")

        if time.time() < deadline:
            final_anneal_deadline = deadline
            a_ref, sc_ref = annealing(
                a_opt,
                final_n,
                final_anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=3e-2,
                end_temp=1e-5,
                steps=120000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] final anneal improve n={final_n} score={sc_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return as plain Python list (the evaluator will clamp/normalise as needed)
    return [float(x) for x in best_seq]
