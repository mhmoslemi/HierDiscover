"""
Propose a non‑negative step function with a low self‑convolution peak.

The search proceeds on the probability simplex (∑aᵢ = 1) because the objective
2·n·max(a*a) / (∑a)² is scale‑invariant.  For each chosen length n we
  * start from several plausible shapes (uniform, sinusoidal, triangular,
    exponential, reversed‑exponential, random Dirichlet);
  * run an Adam optimiser on a smooth log‑sum‑exp surrogate of the ℓ∞‑norm;
    the temperature λ is reduced very slowly (continuation) so that the
    optimiser can escape the basin of the uniform solution;
  * polish the best candidate with a mass‑transfer simulated‑annealing phase
    that moves a random fraction (up to 40 %) of mass from a source index
    (sampled proportionally to its weight) to a random destination.
Time is split proportionally to the length (larger n receive more budget),
and the best distribution found for a given length is linearly interpolated
to initialise the next (larger) length.  The routine respects the 1000 s
budget and returns a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (auto‑convolution). Uses FFT for n>2048."""
    n = a.shape[0]
    if n <= 2048:
        return np.convolve(a, a)
    # FFT based convolution
    size = 2 * n - 1
    fft_len = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:size]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of L_λ(a) = λ·log Σ exp(conv/λ) .
    ``weights`` is the soft‑max over the convolution entries.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # full convolution length = (2n‑1) + n‑1 = 3n‑2
    full = _conv_fft(weights, a_rev)
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad

# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:

    Adam optimisation of the smoothed ℓ∞ surrogate.
    Returns the best vector encountered and its *exact* score
    (2·n·max(conv) because a lives on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters (standard defaults)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule (continuation)
    lam_factor = 0.6            # λ ≈ lam_factor * max(conv)
    lam_decay = 0.95
    lam_decay_interval = 500    # decay slowly (continuation)

    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # exact objective (∑a = 1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # Update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # soft‑max weights
        weights = np.exp((conv - max_b) / lam)
        weights /= weights.sum()

        # gradient of the surrogate
        grad = _gradient_from_weights(a, weights)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.5 / math.sqrt(it + 1.0)   # diminishing step size
        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # back to simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000,
                   max_move_frac: float = 0.40,
                   top_frac: float = 0.10) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a random ε (up to max_move_frac of source mass)
    from a source index (sampled proportionally to its weight, optionally
    restricted to the top_frac most massive indices) to a random destination.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # pre‑compute the list of “heavy” indices (updated every few thousand iters)
    heavy_indices = np.arange(n)

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # occasionally refresh heavy set (every 5000 iterations)
        if it % 5000 == 0:
            k = max(1, int(top_frac * n))
            heavy_indices = np.argpartition(-a, k - 1)[:k]

        # pick source: weighted by a, but restricted to heavy set for efficiency
        src = np.random.choice(heavy_indices, p=a[heavy_indices] / a[heavy_indices].sum())

        # pick destination uniformly (can be same as src – we reject)
        dst = np.random.randint(0, n)
        if src == dst or a[src] <= 0.0:
            continue

        # propose moving a random fraction of the source mass
        eps = a[src] * np.random.rand() * max_move_frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------
def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a small collection of plausible starting vectors from a base.
    All vectors are normalised to sum = 1.
    """
    n = base_seq.shape[0]
    variants = []

    # 1. warm‑start itself
    variants.append(base_seq.copy())

    # 2. half‑sine (positive on (0, π))
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3. triangular (ramp up then down)
    tri = np.concatenate([np.linspace(1, n // 2, n // 2, endpoint=False),
                          np.linspace(n // 2, 1, n - n // 2)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4. exponential decay
    lam = 3.0
    exp_vals = np.exp(-lam * np.arange(n) / n)
    exp_vals /= exp_vals.sum()
    variants.append(exp_vals)

    # 5. reversed exponential (mass at the right side)
    rev_exp_vals = np.exp(-lam * (n - np.arange(n) - 1) / n)
    rev_exp_vals /= rev_exp_vals.sum()
    variants.append(rev_exp_vals)

    # 6. random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Try to reuse a previously found sequence if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    global_a = starter

    # length schedule – increasing sizes (larger n gets more budget)
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    total_weight = sum(lengths)  # weight proportional to n

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm‑start or random)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # generate a few different starts
        init_candidates = _initial_variants(a0)

        # --------------------------------------------------------------
        # Allocate a time slice for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # proportional share of the remaining budget
        weight_left = sum(lengths[idx:])          # total weight of lengths not processed yet
        slice_time = max(6.0, remaining * (n / weight_left))
        slice_deadline = time.time() + slice_time

        # --------------------------------------------------------------
        # Adam optimisation (several starts)
        # --------------------------------------------------------------
        best_local_a = None
        best_local_score = float('inf')
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            # random choice whether to enforce symmetry (about 30 % of runs)
            sym = (np.random.rand() < 0.3)
            a_opt, score_opt = _run_adam_opt(start_vec, n, slice_deadline, start_time, symmetrize=sym)

            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        # update global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} score={best_local_score:.6f}")

        # --------------------------------------------------------------
        # Annealing refinement (if we still have time)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                best_local_a,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=200000,
                max_move_frac=0.40,
                top_frac=0.10,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # remember it for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list (the external evaluator will clamp / normalise)
    return [float(x) for x in best_seq]
