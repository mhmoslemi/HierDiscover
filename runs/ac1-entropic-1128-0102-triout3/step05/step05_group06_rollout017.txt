def _adam_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5
    t = 0
    best_a = a.copy()
    best_score = float('inf')
    # Precompute fft length
    conv_fft_len = 1 << ((2*n - 1).bit_length())
    corr_fft_len = 1 << ((3*n - 2).bit_length())
    # Use the larger length for both to reuse transforms maybe
    fft_len = max(conv_fft_len, corr_fft_len)
    # Preallocate padded arrays
    a_pad = np.zeros(fft_len)
    w_pad = np.zeros(fft_len)

    while time.time() < deadline:
        t += 1
        # Fill padded a
        a_pad[:n] = a
        # FFT of a
        A = np.fft.rfft(a_pad, n=fft_len)

        conv_full = np.fft.irfft(A * A, n=fft_len)

     auto‑convolution:
         f_λ(a)= (1/λ)·log Σ_k exp(λ·(a*a)_k) .
   * Its gradient is a cheap cross‑correlation that can be evaluated by FFT.
   * Adam updates are projected onto the probability simplex after each
     step, guaranteeing non‑negativity and a fixed sum (∑a_i=1), which makes
     the evaluation value equal to `2·n·max(a*a)`.
   * The soft‑max temperature λ is increased gradually so the optimiser first
     sees a smooth landscape and later focuses on the true maximum.
4. A short simulated‑annealing walk (tiny mass transfers) is performed with
   any remaining time to polish the result.
5. Return the best sequence found (as a plain Python list).

The whole procedure respects the 1000 s budget and prints progress info.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution of a with itself via FFT (full length)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate of max(conv) w.r.t. a
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, lam: float,
                     fft_len: int, a_pad: np.ndarray, w_pad: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a ( 1/λ log Σ_k exp(λ·conv_k) ) .
    The result is 2 * cross‑correlation of a and the soft‑max weights w.
    """
    # soft‑max weights (stable formulation)
    max_c = conv.max()
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / exp_term.sum()

    # pad a and w (re‑use the pre‑allocated buffers)
    n = a.shape[0]
    L = 2 * n - 1
    a_pad[:n] = a
    a_pad[n:] = 0.0
    w_pad[:L] = w
    w_pad[L:] = 0.0

    # FFT of a and w (same length)
    A = np.fft.rfft(a_pad, n=fft_len)
    W = np.fft.rfft(w_pad, n=fft_len)

    # cross‑correlation (using conjugate of A)
    corr_full = np.fft.irfft(np.conj(A) * W, n=fft_len)
    # first n entries correspond to lags 0 … n‑1
    grad = 2.0 * corr_full[:n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex with a smooth objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float, start_time: float):
    """
    Adam optimiser that minimises 2·n·max(a*a) on the simplex.
    Returns the best point seen and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5

    # lengths for the FFTs (use the larger one for both operations)
    conv_fft_len = 1 << ((2 * n - 1).bit_length())
    corr_fft_len = 1 << ((3 * n - 2).bit_length())
    fft_len = max(conv_fft_len, corr_fft_len)

    # reusable buffers for padding
    a_pad = np.zeros(fft_len)
    w_pad = np.zeros(fft_len)

    # bias‑correction powers
    beta1_pow = beta1
    beta2_pow = beta2

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # ----------- convolution a*a (full) -------------
        a_pad[:n] = a
        a_pad[n:] = 0.0
        A = np.fft.rfft(a_pad, n=fft_len)
        conv_full = np.fft.irfft(A * A, n=fft_len)
        conv = conv_full[: 2 * n - 1]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam iter={it:7d} "
                  f"score={cur_score:.6f}")

        # ----------- smooth gradient --------------------
        # λ grows linearly until it caps at 50
        lam = min(50.0, 0.01 * it)
        grad = _smooth_gradient(a, conv, lam, fft_len, a_pad, w_pad)

        # ----------- Adam update ------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto simplex
        a = _project_simplex(a)

        # update bias‑correction powers
        beta1_pow *= beta1
        beta2_pow *= beta2

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing walk (tiny mass transfers)
# ----------------------------------------------------------------------
def _anneal_refine(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA, only as a final polish."""
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # helper to evaluate the true objective (no soft‑max)
    def _score(x):
        return 2.0 * n * _conv_self_fft(x).max()

    cur_score = _score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # geometric temperature schedule
        T = 5e-2 * (1e-5 / 5e-2) ** (it / 500_000)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score(a_prop)
        delta = new_score - cur_score

        # Metropolis rule
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal iter={it:7d} "
                  f"best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Beta(α,α) density (discretised on a uniform grid)
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, n: int) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n
    # power law, safe for α<1 (xs>0)
    pdf = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, alpha - 1.0)
    pdf = np.nan_to_num(pdf, nan=0.0, posinf=0.0, neginf=0.0)
    s = pdf.sum()
    if s == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= s
    return pdf


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  A smooth Adam optimiser on a simplex together
    with a short simulated‑annealing fine‑tuning is used.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness itself
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1. Choose a working length (large enough to approach the asymptotic
    #    constant but still quick to FFT – 9600 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 9600

    # ------------------------------------------------------------------
    # 2. Scan a few Alpha‑Beta shapes and keep the best as a starter
    # ------------------------------------------------------------------
    alphas = [0.20, 0.30, 0.40, 0.50, 0.60, 0.70, 0.80, 1.00, 1.50, 2.00]
    best_seq = None
    best_score = float("inf")
    scan_deadline = start_time + (deadline - start_time) * 0.15

    for alpha in alphas:
        if time.time() > scan_deadline:
            break
        a0 = _beta_shape(alpha, n)
        conv = _conv_self_fft(a0)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = a0.copy()
            print(f"[{time.time() - start_time:6.2f}s] init α={alpha:.2f} "
                  f"score={score:.6f}")

    # Ensure we have *something*
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self_fft(best_seq).max()

    # ------------------------------------------------------------------
    # 3. Adam optimisation (major budget chunk)
    # ------------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 0:
        opt_deadline = time.time() + remain * 0.70
        a_opt, score_opt = _adam_optimize(best_seq, n, opt_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improvement "
                  f"score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 4. Final annealing polish (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _anneal_refine(best_seq, n, deadline,
                                          start_time, best_score, best_seq)
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal best "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # 5. Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny drift may have occurred)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # convert to plain Python list
    return [float(x) for x in best_seq]
