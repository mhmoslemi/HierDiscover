       # Solve LP
       g = solve_lp(a, M)
       sum_g = g.sum()
       if sum_g <= 1 + 1e-9:
           # cannot improve further, break
           break

       # step size schedule

def conv_max(a):
    n = len(a)
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
    return conv.max(), conv

"""
propose_candidate – LP‑guided + exponential‑gradient search for a low‑score step
function.

The algorithm works on a probability simplex (non‑negative entries that sum to 1)
and consists of two phases:

1. LP refinement.
   For a given sequence a we compute M = max(conv(a,a)).
   The linear program
        maximise   1ᵀg
        subject to conv(a,g) ≤ M ,  g ≥ 0
   yields a new direction g that can increase the total mass while keeping the
   L∞‑norm of the convolution unchanged.  By moving a a little towards the
   normalised solution g we strictly decrease M (or stop if no improvement is
   possible).  The LP is solved with SciPy’s HiGHS interior‑point method.
   The outer loop performs a few such updates with a decreasing convex‑combination
   weight.

2. Exponential‑gradient (mirror‑descent) fine‑tuning.
   Starting from the best sequence found in phase 1 we run a hard‑max
   sub‑gradient descent on the objective
        score = 2·n·max(conv(a,a))                     (∑a = 1)
   The sub‑gradient of the L∞‑norm is
        g = 2·(a_rev * 𝟙_{active})                    (FFT based)
   and the update is multiplicative:
        a_i ← a_i·exp(−η·g_i)  ,  renormalise to Σa_i = 1.
   The step size η decays geometrically from η_start to η_end.

Both phases respect the time budget (default 1000 s).  The routine also
accepts the previously discovered construction via the global variable
`height_sequence_1` and tries a few random Dirichlet initialisations.
The returned list is directly usable with the supplied `evaluate_sequence`
function.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution (real)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A of size (2n‑1)×n such that (A @ g)_k = conv(a,g)_k.
    Column j of A contains the vector a shifted by +j positions.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # Fill column‑wise – this is O(n²) but fast for n≈2000.
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution term.
    g = 2·(a_rev * w)  (linear convolution, computed by FFT).
    """
    n = a.shape[0]
    fft_len = 1 << ((n + w.shape[0] - 1).bit_length())
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A_rev * W, n=fft_len)
    start = n - 1
    return 2.0 * conv[start:start + n]


# ----------------------------------------------------------------------
# Phase 1 – LP refinement
# ----------------------------------------------------------------------
def _lp_refine_once(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program
        maximise   1ᵀg
        subject to conv(a,g) ≤ M , g ≥ 0 .
    Return the optimal g (or None if the LP fails).
    """
    n = a.shape[0]
    A = _build_constraint_matrix(a)
    c = -np.ones(n, dtype=np.float64)          # maximise → minimise –cᵀg
    b = np.full(A.shape[0], M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


def _lp_refine_sequence(a: np.ndarray,
                        deadline: float,
                        max_outer: int = 20,
                        t_start: float = 0.5,
                        t_end: float = 0.01) -> np.ndarray:
    """
    Repeatedly call the LP above and move a towards the normalised solution.
    The step size (convex‑combination weight) decays geometrically.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_M, _ = _conv_max(best_a)
    best_score = 2.0 * n * best_M

    for outer in range(1, max_outer + 1):
        if time.time() >= deadline:
            break

        M, _ = _conv_max(best_a)
        # Solve the LP for the current a
        g = _lp_refine_once(best_a, M)
        if g is None:
            print(f"[LP] iteration {outer}: LP failed – stop")
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # No possibility to increase total mass → converged
            print(f"[LP] iteration {outer}: sum(g) ≤ 1 – converged")
            break

        # convex combination weight (decreasing)
        tau = outer / max_outer
        t = t_start * ((t_end / t_start) ** tau)

        g_norm = g / sum_g
        cand = (1.0 - t) * best_a + t * g_norm
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / n)

        M_cand, _ = _conv_max(cand)
        score_cand = 2.0 * n * M_cand

        if score_cand < best_score - 1e-9:
            best_score = score_cand
            best_a = cand
            print(f"[LP] iteration {outer}: improved score = {best_score:.6f}")
        else:
            # no improvement – stop early
            print(f"[LP] iteration {outer}: no improvement – stop")
            break

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – Exponential‑gradient (hard‑max) refinement
# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best vector encountered and its score.
    """
    n = a.shape[0]
    best_a = a.copy()
    M, _ = _conv_max(a)
    best_score = 2.0 * n * M

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        M, conv = _conv_max(a)
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # active anti‑diagonals (within eps of the maximum)
        mask = conv >= M - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient for the L∞‑norm
        g = _gradient_from_weights(a, w)

        # geometric step‑size schedule
        tau = it / max_iters
        eta = eta_start * ((eta_end / eta_start) ** tau)

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # keep exp() stable
        a = a * np.exp(upd)

        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    final_score = 2.0 * n * _conv_max(a)[0]
    return best_a, final_score

        spikes[pos] = np.random.rand(k) + 0.1
        pool.append(_as_simplex(spikes))

            best_score = score_eg

        n = best_seq.shape[0]
        remaining = deadline - time.time()
        est_iter_time = 8e-4 * (n / 1200.0) + 1e-4
        final_iters = int(min(400000, max(80000, remaining / est_iter_time)))
        print(f"[{time.time() - start_time:6.2f}s] Final EG refinement – iters={final_iters}")
        best_seq, final_score = _exp_gradient_optimize(best_seq.copy(),
                                                       max_iters=final_iters,
                                                       deadline=deadline,
                                                       eta_start=0.3,
                                                       eta_end=5e-4)
        best_score = min(best_score, final_score)

    # ------------------------------------------------------------------
    # 4) sanitise and report
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector of length 200
        best_seq = np.full(200, 1.0 / 200)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    try:
        final_eval = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] Finished – best score = {final_eval:.6f}")
    except Exception:
        final_eval = None
        print(f"[{time.time() - start_time:6.2f}s] Finished – evaluation failed")

    return [float(x) for x in best_seq]
