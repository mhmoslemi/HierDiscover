def _smoothmax_grad(a, alpha):
    conv = _conv_self(a)    # length 2n-1

    # compute softmax weights
    maxc = conv.max()
    # avoid overflow: compute exp(alpha*(conv-maxc))
    exp_term = np.exp(alpha * (conv - maxc))
    sum_exp = exp_term.sum()
    p = exp_term / sum_exp

    # gradient: 2 * convolution of a with reversed p
    p_rev = p[::-1]
    grad_full = _conv_fft(a, p_rev)
    # extract indices
    n = a.shape[0]
    idx = 2*n - 2 - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, conv

"""
Smooth‐max gradient descent with progressive up‑sampling.

The algorithm builds on the previous approach but replaces the
sub‑gradient of the hard max by the gradient of a
log‑sum‑exp (smooth‑max) surrogate.  This yields a dense gradient
that guides mass away from the peaks of the self‑convolution.
A schedule of increasing smoothing (α) together with Nesterov
momentum is used, followed by a cheap simulated‑annealing walk.
The whole search respects a 1000 s budget and returns the best
sequence found (as a plain list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generic linear convolution via FFT (real input)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real 1‑D arrays using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    tot = n + m - 1
    fft_len = 1 << ((tot - 1).bit_length())      # next power of two
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:tot]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate of max(a*a)
# ----------------------------------------------------------------------
def _smoothmax_grad(a: np.ndarray, alpha: float):
    """
    Returns (grad, conv):
        grad – gradient of (1/α)·log Σ exp(α·conv) w.r.t. a,
        conv – the exact self‑convolution a*a.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1

    # stable softmax over the convolution entries
    maxc = conv.max()
    exp_term = np.exp(alpha * (conv - maxc))   # ≤ 1, no overflow
    sum_exp = exp_term.sum()
    p = exp_term / sum_exp                     # p_k ≥ 0, Σ p_k = 1

    # gradient = 2 * (a * p_rev) restricted to indices (2n‑2‑i)
    p_rev = p[::-1]
    grad_full = _conv_fft(a, p_rev)           # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)            # map to required slice
    grad = 2.0 * grad_full[idx]
    return grad, conv


# ----------------------------------------------------------------------
# Projected gradient descent with Nesterov momentum on the smooth‑max
# ----------------------------------------------------------------------
def _run_optimizer(a0: np.ndarray, n: int,
                   deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimises a on the simplex to minimise max(a*a) using a smooth‑max
    surrogate.  Returns the best sequence seen and its (scaled) score
    = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # schedule of smoothing parameters (α).  We start very smooth and
    # increase α as the optimisation proceeds.
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0]
    step0 = 0.6
    beta = 0.9

    # how many iterations per α before we move to the next level
    it_per_alpha = 5000

    while time.time() < deadline:
        it += 1
        # pick the current α
        alpha = alphas[min((it - 1) // it_per_alpha, len(alphas) - 1)]

        grad, conv = _smoothmax_grad(a, alpha)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size (√t schedule)
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] opt n={n:5d} "
                  f"it={it:7d} α={alpha:5.1f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Move a small ε from i to j and accept according to a geometric schedule."""
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # up to 10 % of the source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        # Metropolis rule
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} "
                  f"it={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to a new length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    progressively upsamples the current best shape, runs a smooth‑max
    gradient optimiser and finishes with a short simulated‑annealing
    walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 1‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter supplied by the harness (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Initial analytical shape – discretised arcsine (Beta(½,½))
    # ------------------------------------------------------------------
    n0 = 800
    xs = (np.arange(n0) + 0.5) / n0
    arcsine = 1.0 / np.sqrt(xs * (1.0 - xs))                # unnormalised arcsine density
    a0 = arcsine / arcsine.sum()                           # sum = 1
    if starter is not None:
        a0 = _upsample(starter, n0)

    # ------------------------------------------------------------------
    # Progressive up‑sampling schedule (geometric growth)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600, 19200]        # final length ≈ 20 k
    total_weight = sum(stage_lengths)

    # reserve ~20 % of budget for the final annealing walk
    anneal_budget = max(30.0, 0.20 * (deadline - start_time))
    grad_budget_total = (deadline - start_time) - anneal_budget

    for n in stage_lengths:
        if time.time() >= deadline - anneal_budget:
            break

        # allocate a slice of the remaining gradient budget proportionally
        remaining = deadline - start_time - anneal_budget - (time.time() - start_time)
        if remaining <= 0:
            break
        stage_frac = n / total_weight
        stage_time = max(5.0, remaining * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this stage
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = _upsample(a0, n)

        # --------------------------------------------------------------
        # Gradient descent on the smooth‑max surrogate
        # --------------------------------------------------------------
        a_opt, score_opt = _run_optimizer(
            a0, n, stage_deadline, start_time
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} "
                  f"improve: score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Final annealing phase – improves any remaining tiny imperfections
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, steps=800_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the harness
    return [float(x) for x in best_seq]
