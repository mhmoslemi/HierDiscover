"""
Improved step‑function optimizer.

Main ideas
~~~~~~~~~
* Progressive up‑sampling of the current best sequence (800 → 2400 → 4800 → 9600
  → 19200).  The up‑sampling is linear and preserves the total mass.
* Projected gradient descent on the *exact* max‑convolution (hard‑max
  sub‑gradient) after a short soft‑max “warm‑up”.  The hard‑max gradient is
  the true sub‑gradient of  max_k a*a (k)  and drives the optimisation more
  aggressively towards a flat convolution.
* Nesterov momentum and a 1/√t step‑size schedule.  The momentum coefficient
  (β) is 0.9.
* A cheap simulated‑annealing walk (tiny transfers of mass) that uses the
  remaining time budget.
* All updates are projected onto the probability simplex (the feasible set
  {x≥0, Σx=1}); the objective is scale‑invariant, therefore fixing the sum to
  1 does not affect the final score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: linear self‑convolution (full length = 2·n‑1) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3: convolution of rev(a) with an arbitrary weight vector w
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the smooth (soft‑max) approximation of max(conv)."""
    maxc = conv.max()
    # stable soft‑max (maxc removed, then scaled)
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    return 2.0 * full[n - 1:n - 1 + n]


def _hardmax_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient: average of the gradients of the active (max) entries."""
    max_val = conv.max()
    eps = max_val * 1e-12 if max_val > 0 else 1e-12
    mask = (conv >= max_val - eps).astype(float)
    cnt = mask.sum()
    w = mask / cnt if cnt > 0 else np.full_like(conv, 1.0 / conv.size)
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    return 2.0 * full[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Projected (Nesterov) gradient descent with hybrid soft‑max / hard‑max
# ----------------------------------------------------------------------
def _run_gd(a0: np.ndarray,
            n: int,
            deadline: float,
            start_time: float,
            step0: float = 0.6,
            beta: float = 0.9,
            max_iter: int = 10 ** 7,
            warmup_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Returns the best sequence found and its (scaled) score = 2·n·max(conv).
    The first `warmup_iters` iterations use a soft‑max gradient, after that the
    exact hard‑max sub‑gradient.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule for the warm‑up phase
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: n={n}, iter={it}, score={cur_score:.6f}")

        # ------- gradient computation ---------------------------------
        if it <= warmup_iters:
            # exponential annealing of the temperature
            T = T_start * (T_end / T_start) ** (it / warmup_iters)
            grad = _softmax_gradient(a, conv, T)
        else:
            grad = _hardmax_gradient(a, conv)

        # ------- Nesterov‑style momentum -------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ------- diminishing step size ---------------------------------
        step = step0 / math.sqrt(it)

        # ------- projection onto the simplex ----------------------------
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 10 % of the mass at i towards j
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
