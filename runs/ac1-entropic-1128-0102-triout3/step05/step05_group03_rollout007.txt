def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # get starter
    if 'height_sequence_1' in globals():
        try:
            starter = np.array(globals()['height_sequence_1'], dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = starter / starter.sum()
        except Exception:
            starter = None
    else:
        starter = None

    # Setup phase sizes
    n0 = 2048   # start moderate
    # time allocation for phase0
    phase0_time = max(20.0, 0.15 * (deadline - start))
    phase0_deadline = time.time() + phase0_time

    # Phase 0: Adam optimization with smooth max
    best_seq = None
    best_score = float('inf')
    # generate initial a0
    if starter is not None:
        a0 = _upsample(starter, n0)
    else:
        a0 = np.random.rand(n0)
        a0 /= a0.sum()
    a_opt, score_opt = _adam_optimize(a0, n0, phase0_deadline, start)
    if score_opt < best_score:
        best_score = score_opt
        best_seq = a_opt.copy()
        print(f"[{time.time() - start:.2f}s] phase0 best score {score_opt:.6f}")

    if time.time() >= deadline:
        return best_seq.tolist()

    # Phase 1 upscale to n1
    n1 = 8192
    a1 = _upsample(best_seq, n1)
    phase1_time = max(30.0, 0.35 * (deadline - time.time()))
    phase1_deadline = time.time() + phase1_time
    a_opt1, score_opt1 = _adam_optimize(a1, n1, phase1_deadline, start)
    if score_opt1 < best_score:
        best_score = score_opt1
        best_seq = a_opt1.copy()
        print(f"[{time.time() - start:.2f}s] phase1 best score {score_opt1:.6f}")

    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    # Phase 2 final annealing (like earlier) to refine
    remaining = deadline - time.time()
    a_ref, score_ref = _run_annealing(best_seq, best_seq.shape[0],
                                      deadline, start, best_score, best_seq,

"""
Advanced search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Smooth‑max Adam optimisation** – the true max in the objective is
   replaced by a log‑sum‑exp (smooth max) with a continuation on the
   smoothing parameter β.  The variables are optimised on the probability
   simplex via a soft‑max parametrisation and the Adam optimiser.

2. **Upsampling & further refinement** – the best vector from the first
   stage is linearly interpolated to a larger dimension and the Adam
   optimisation is repeated (with a larger β).

3. **Fine‑grained simulated annealing** – a mass‑transfer annealing
   improves the result locally.

All stages respect the overall time budget (default 1000 s).  The
function returns a plain Python list of non‑negative floats (normalised
to sum = 1) ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D real arrays using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    fft_len = 1 << ((n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Af * Bf, n=fft_len)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a (linear)."""
    return _conv(a, a)


# ----------------------------------------------------------------------
# Smooth‑max related helpers
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return the softmax weights for a smooth max with temperature 1/β."""
    # shift by max to avoid overflow
    shift = conv.max()
    exp_term = np.exp(beta * (conv - shift))
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  sum_k w_k * (a*a)_k  w.r.t. a.
    Implements  g_i = 2 * Σ_j a_j w_{i+j} - a_i * w_{2i}.
    Computed via one FFT‑based convolution.
    """
    n = a.shape[0]
    # reversed weight vector for correlation
    w_rev = w[::-1]
    conv = _conv(a, w_rev)                     # length = n + (2n-1) -1 = 3n-2
    idx = 2 * n - 2 - np.arange(n)             # positions that give Σ_j a_j w_{i+j}
    g = 2.0 * conv[idx] - a * w[2 * np.arange(n)]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the probability simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    beta_start: float = 5.0,
    beta_end: float = 80.0,
    beta_ramp_iters: int = 2500,
    lr_init: float = 0.12,
    max_iter: int = 10 ** 7,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of  F(a)=2·n·max(a*a)  on the simplex.
    Uses a smooth approximation of the max; the true objective is
    evaluated at every iteration and the best vector is retained.
    Returns the best vector (as a flat ndarray) and its *true* score.
    """
    # initialise y such that softmax(y) = a0
    a0 = np.clip(a0, 1e-15, None)
    y = np.log(a0)

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = None
    best_score = float("inf")

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ---------- map y -> a (softmax) ----------
        exp_y = np.exp(y - np.max(y))                 # shift for stability
        a = exp_y / exp_y.sum()

        # ---------- true objective (no smoothing) ----------
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # ---------- smooth‑max parameters ----------
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)

        # ---------- smooth gradient ----------
        w = _smooth_weights(conv, beta)               # length = 2n‑1
        g_a = _gradient_smooth(a, w)                  # ∂/∂a of the smooth objective

        # ---------- chain rule for y ----------
        avg = np.dot(g_a, a)                           # scalar = a·g_a
        grad_y = a * (g_a - avg)                       # ∂F/∂y

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr_init / (np.sqrt(v_hat) + eps)
        y = y - step * m_hat

    if best_a is None:                      # should never happen, but defensive
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_a).max()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (identical to the reference version)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-6,
    steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (keeps the shape, only changes resolution)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point (called by the evaluation harness)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising `evaluate_sequence`.
    Returns a list of non‑negative floats (sum ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global name `height_sequence_1`)
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Phase 0 – modest size, Adam optimisation with smooth max
    # ------------------------------------------------------------------
    n0 = 2048
    # allocate roughly 15 % of the total time (or at least 20 s)
    slice_time = max(20.0, 0.15 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    if starter is not None:
        a0 = _upsample(starter, n0)
    else:
        a0 = np.random.rand(n0)
        a0 /= a0.sum()

    best_seq, best_score = _adam_optimize(
        a0, n0, slice_deadline, start_time,
        beta_start=5.0,
        beta_end=50.0,
        beta_ramp_iters=2000,
        lr_init=0.12,
    )
    print(f"[{time.time() - start_time:6.2f}s] end phase 0, best score={best_score:.6f}")

    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Phase 1 – upscale to a larger resolution and continue optimisation
    # ------------------------------------------------------------------
    n1 = 8192
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    # spend about 40 % of the remaining budget on this stage
    grad_time = max(30.0, 0.40 * remaining)
    grad_deadline = time.time() + grad_time

    best_seq, best_score = _adam_optimize(
        a1, n1, grad_deadline, start_time,
        beta_start=10.0,
        beta_end=80.0,
        beta_ramp_iters=3000,
        lr_init=0.10,
    )
    print(f"[{time.time() - start_time:6.2f}s] end phase 1, best score={best_score:.6f}")

    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning (use whatever time is left)
    # ------------------------------------------------------------------
    anneal_deadline = deadline
    best_seq, best_score = _run_annealing(
        best_seq,
        best_seq.shape[0],
        anneal_deadline,
        start_time,
        best_score,
        best_seq,
        start_temp=3e-2,
        end_temp=1e-6,
        steps=250_000,
    )
    print(f"[{time.time() - start_time:6.2f}s] final score after annealing={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (ensure sum=1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # final safeguard – clip tiny negatives that can appear due to numerical noise
    best_seq = np.maximum(best_seq, 0.0)

    return [float(x) for x in best_seq]
