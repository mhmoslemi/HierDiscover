def _build_conv_matrix(f):
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows from i to i+n-1 inclusive: k = i + j
        rows_i = np.arange(i, i + n)  # length n
        cols_i = np.arange(0, n)      # j from 0 to n-1
        rows.append(rows_i)
        cols.append(cols_i)
        data.append(np.full(n, f[i]))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A
```

def _lp_mix(f, M, t=0.2):
    n = f.shape[0]
    A = _build_conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    if g.value is None:
        return f
    g_opt = np.array(g.value).flatten()
    sum_g = g_opt.sum()
    if sum_g > 0:
        g_norm = g_opt / sum_g
    else:
        g_norm = np.full_like(g_opt, 1.0 / n)
    # mixing
    new_f = (1 - t) * f + t * g_norm
    # ensure sum 1 by normalization
    new_f /= new_f.sum()
    return new_f
```

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm (≈1000 s)
-------------------
1. **Coarse search (n≈800).**  Several random restarts are optimised with
   exponentiated‑gradient updates on a smooth approximation of the
   ``max(conv)`` objective.  The best vector is kept.

2. **Medium optimisation (n≈2400).**  The best vector is up‑sampled
   (linear interpolation) and further refined with a longer exponentiated‑gradient
   run.  Afterwards a *single linear‑program* is solved:
   maximise ``∑ g`` subject to ``f⋆g ≤ M`` (where ``M`` is the current bound).
   The LP gives a direction that can reduce the bound when mixed with the
   current vector.

3. **Large scale optimisation (n≈4800).**  The mixed vector is up‑sampled
   again and optimised with exponentiated‑gradient.  A short projected‑gradient
   polish using the exact sub‑gradient of ``max(conv)`` follows.

4. **Fine‑tuning (annealing).**  Random mass‑transfer moves are accepted
   according to a simulated‑annealing schedule.

All vectors are kept non‑negative and normalised to sum = 1, which leaves the
objective invariant.  The routine respects the supplied time budget and prints
progress information which is shown in the harness output.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a using all indices where conv is
    within `rel_tol` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of f(a)= (1/α)·log Σ_k exp(α·conv_k).
    Returns ∂f/∂a (same shape as a).
    """
    n = a.shape[0]
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # correlation a ⋆ w_rev
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.3,
                              steps_per_alpha: int = 8000,
                              max_iter: int = 10 ** 7):
    """
    Multiplicative updates with a smoothed‑max objective.
    Returns (best_vector, best_score) where score = 2·n·max(conv)
    (the denominator is 1 because we keep sum(a)=1).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] EG improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # smoothed‑max gradient
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        # multiplicative update
        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10 ** 7):
    """
    Projected gradient descent with momentum using the *exact* sub‑gradient
    of max(conv).  Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] PG improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum

        # simplex projection (O(n))
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000):
    """
    Simple mass‑transfer simulated annealing.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Linear‑programming mixing step
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1) × n sparse matrix A such that (A @ g)[k]
    = Σ_i f[i]·g[k‑i]  (i.e. the linear convolution f⋆g).
    """
    n = f.shape[0]
    # total non‑zeros = n²
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    idx = 0
    for i in range(n):
        rows[idx:idx + n] = np.arange(i, i + n)        # k = i + j
        cols[idx:idx + n] = np.arange(n)               # j = 0 … n‑1
        data[idx:idx + n] = f[i]
        idx += n

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_mix(f: np.ndarray, M: float, mix_t: float = 0.2) -> np.ndarray:
    """
    Perform one LP‑based mixing step:
        maximise   ∑ g_j
        s.t.       A·g ≤ M    (A is the convolution matrix of f)
                    g ≥ 0
    Returns a new vector (1‑t)·f + t·ĝ, where ĝ is the LP solution
    normalised to have unit sum.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is fast for moderate‑size LPs
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)

    if g.value is None:
        # LP failed – fall back to the unchanged vector
        return f.copy()

    g_opt = np.asarray(g.value).flatten()
    sum_g = g_opt.sum()
    if sum_g <= 0:
        return f.copy()

    g_norm = g_opt / sum_g          # now ∑ g_norm = 1
    new_f = (1.0 - mix_t) * f + mix_t * g_norm
    # mixing already preserves the simplex, but enforce numerical safety
    new_f /= new_f.sum()
    return new_f

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (sequence) minimising
    ``evaluate_sequence``.  The routine follows a multi‑stage scheme
    (coarse EG, medium EG + LP, large EG, projected‑gradient polishing,
    annealing) while respecting the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Try to obtain a starter from the previous run (global variable)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
    except Exception:
        starter = None

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – many short EG runs on a modest size (n≈800)
    # ------------------------------------------------------------------
    n_small = 800
    # Allocate ~10 % of the total budget to stage 1 (but at least 30 s)
    slice_deadline = time.time() + max(30.0, 0.10 * (deadline - time.time()))
    restarts = 8
    for restart in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and restart == 0:
            a0 = _upsample(starter, n_small)
        else:
            a0 = np.random.rand(n_small)
            a0 /= a0.sum()

        a_opt, score_opt = _run_exponential_gradient(
            a0,
            n_small,
            slice_deadline,
            start_time,
            alpha_schedule=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            eta0=0.4,
            steps_per_alpha=4000,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: restart={restart}, score={score_opt:.6f}")

    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)
        best_score = 2.0 * n_small * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 2 – Up‑sample, longer EG, then a single LP mixing step
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return best_seq.tolist()

    n_mid = 2400
    a_mid = _upsample(best_seq, n_mid)

    # allocate roughly 25 % of the remaining time to this stage
    remaining = deadline - time.time()
    stage2_deadline = time.time() + max(20.0, 0.25 * remaining)

    a_mid_opt, score_mid = _run_exponential_gradient(
        a_mid,
        n_mid,
        stage2_deadline,
        start_time,
        alpha_schedule=[1.0, 5.0, 15.0, 30.0, 60.0, 120.0, 240.0],
        eta0=0.3,
        steps_per_alpha=8000,
    )
    if score_mid < best_score:
        best_score = score_mid
        best_seq = a_mid_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 EG improve: score={score_mid:.6f}")

    # LP mixing (uses the current bound)
    if time.time() < deadline:
        M_mid = _conv_self(best_seq).max()
        # do two quick mixing iterations – cheap and often beneficial
        for it in range(2):
            best_seq = _lp_mix(best_seq, M_mid, mix_t=0.2)
            M_mid = _conv_self(best_seq).max()
            cur_score = 2.0 * best_seq.shape[0] * M_mid
            if cur_score < best_score:
                best_score = cur_score
                print(f"[{time.time() - start_time:6.2f}s] stage 2 LP mix improve: iter={it+1}, score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Up‑sample to a large size (n≈4800) and run EG again
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return best_seq.tolist()

    n_big = 4800
    a_big = _upsample(best_seq, n_big)

    remaining = deadline - time.time()
    stage3_deadline = time.time() + max(30.0, 0.35 * remaining)

    a_big_opt, score_big = _run_exponential_gradient(
        a_big,
        n_big,
        stage3_deadline,
        start_time,
        alpha_schedule=[5.0, 15.0, 30.0, 60.0, 120.0, 240.0, 480.0],
        eta0=0.2,
        steps_per_alpha=12000,
    )
    if score_big < best_score:
        best_score = score_big
        best_seq = a_big_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 3 EG improve: score={score_big:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – short projected‑gradient polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            best_seq.shape[0],
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
