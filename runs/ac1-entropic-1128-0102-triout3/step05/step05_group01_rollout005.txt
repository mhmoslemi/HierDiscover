"""
Gradient‑descent + subgradient optimisation for step‑function minimisation.

The algorithm proceeds through progressive up‑sampling of the sequence.
For each size we run a fast Adam optimiser that uses a smoothed soft‑max
approximation of the max‑convolution (temperature annealed to ~0).  When the
temperature becomes very small we switch to the exact sub‑gradient of the
max‑convolution.  The iterates are kept on the probability simplex by a
cheap clip‑and‑renormalise projection.  After the last size a short
simulated‑annealing walk with larger mass‑transfer moves is performed.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1 (fast projection)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v / s
    else:
        # fall back to uniform distribution
        v = np.full_like(v, 1.0 / v.shape[0])
    return v

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution a * a using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of reversed sequence a_rev with a weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv

def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max approximation of max(conv)."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the true max‑convolution (averaged over all maximisers)."""
    max_val = conv.max()
    # tolerance to collect all entries that are numerically maximal
    mask = (conv >= max_val - 1e-14)
    cnt = mask.sum()
    if cnt == 0:
        return np.zeros_like(a)
    w = mask.astype(float) / cnt
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              step0: float = 0.8,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a temperature schedule that interpolates between a
    smooth soft‑max and the exact max.  Returns the best sequence seen.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    # temperature schedule – we aim at a very low final temperature
    T_start = 2.0e-2
    T_end = 5.0e-9
    # we target roughly 30 000 iterations for a full stage; the exponent
    # will be clamped automatically if the stage finishes earlier.
    target_iters = 30000.0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        sum_a = a.sum()
        cur_score = 2.0 * n * max_val / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter={it:6d}, n={n}, score={cur_score:.6f}")

        # temperature (linear in log‑scale)
        ratio = min(it / target_iters, 1.0)
        T = T_start * (T_end / T_start) ** ratio

        # gradient
        if T < 1e-11:
            grad = _exact_gradient(a, conv)
        else:
            grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score

def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 400_000) -> tuple[np.ndarray, float]:
    """Simple Metropolis walk that moves a random fraction of mass."""
    a = a0.copy()
    sum_a = a.sum()
    score = 2.0 * n * _conv_self(a).max() / (sum_a * sum_a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # transfer up to 30 % of the source mass
        eps_mass = a[i] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i] -= eps_mass
        a_prop[j] += eps_mass

        sum_a = a_prop.sum()
        new_score = 2.0 * n * _conv_self(a_prop).max() / (sum_a * sum_a)
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new length, followed by renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Exact sub‑gradient of the true max‑convolution.
# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Average gradient over all indices where conv attains its maximum."""
    max_val = conv.max()
    mask = conv >= max_val - 1e-14          # collect ties robustly
    cnt = mask.sum()
    if cnt == 0:
        return np.zeros_like(a)
    w = mask.astype(float) / cnt
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser with a temperature schedule that moves from soft‑max to
# exact sub‑gradient.  Returns the best point and its score.
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              step0: float = 0.8,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    # temperature schedule – from 2·10⁻² down to 5·10⁻⁹
    T_start = 2.0e-2
    T_end = 5.0e-9
    # we target ~30 000 iterations per stage; the schedule clamps
    # automatically if the stage stops earlier.
    target_iters = 30000.0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        sum_a = a.sum()
        cur_score = 2.0 * n * max_val / (sum_a * sum_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam iter={it:6d}, n={n}, score={cur_score:.6f}")

        # anneal temperature (exponential in iteration count)
        ratio = min(it / target_iters, 1.0)
        T = T_start * (T_end / T_start) ** ratio

        # gradient
        grad = _exact_gradient(a, conv) if T < 1e-11 else _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Short simulated‑annealing refinement – moves a random fraction of mass.
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 400_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    sum_a = a.sum()
    score = 2.0 * n * _conv_self(a).max() / (sum_a * sum_a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 30 % of the source mass
        eps_mass = a[i] * np.random.rand() * 0.3
        a_prop = a.copy()
        a_prop[i] -= eps_mass
        a_prop[j] += eps_mass

        sum_a = a_prop.sum()
        new_score = 2.0 * n * _conv_self(a_prop).max() / (sum_a * sum_a)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving total mass.
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` from its current length to `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, performs progressive up‑sampling,
    Adam optimisation with an annealed soft‑max gradient, and finishes with a short
    annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Optional starter from a previous run.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = _project_simplex(starter)
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage schedule (progressive enlargement).
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]
    total_len = sum(stage_lengths)

    # Allocate ~15 % of the whole budget for the final annealing phase.
    total_time = deadline - start_time
    anneal_budget = max(30.0, 0.15 * total_time)
    grad_budget = total_time - anneal_budget

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # time share proportional to the stage length
        stage_frac = n / total_len
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for the current size
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 = _project_simplex(a0)

        # Adam optimiser (soft‑max → exact gradient)
        a_opt, score_opt = _run_adam(
            a0, n, stage_deadline, start_time,
            step0=0.8, beta1=0.9, beta2=0.999, eps=1e-8,
            max_iter=10 ** 7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing refinement.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5,
            steps=500_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
