"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1.  Try to reuse the previous construction (``height_sequence_1``) if present.
2.  Perform a fast random search on a moderate size (≈2000) using
    symmetric Beta densities – a cheap way to explore the space of shapes.
3.  Upsample the best candidate to the final length (9600) while keeping the
    total mass equal to one.
4.  Refine the up‑sampled vector with a long projected Adam optimisation.
    The objective uses a *soft‑max* approximation of the max‑convolution;
    the temperature is annealed exponentially.
5.  A short Metropolis‑annealing walk moves tiny amounts of mass between
    positions to polish any remaining improvement.
6.  Return the final vector as a plain Python list.

The whole routine respects the given time budget (default 1000 s) and prints
progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – linear convolution a * a (full length = 2·len(a) – 1) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3 – convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution rev(a) * w (full length = len(a_rev)+len(w)-1)."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4 – gradient of a *soft‑max* of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution.
    Uses weights w_k ∝ exp(conv_k / T) and returns 2·(rev(a) * w) restricted
    to the central n entries.
    """
    # stable soft‑max
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    # subtract again the max for numerical stability (optional)
    scaled -= np.max(scaled)
    exps = np.exp(scaled)
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1 : n - 1 + n]       # central block
    return grad


# ----------------------------------------------------------------------
# Helper 5 – evaluate the objective (2·n·max(conv)) under sum=1
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1."""
    conv = _conv_self(a)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Helper 6 – generate a symmetric Beta density on the grid {0,…,n‑1}
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on n points (mid‑points), normalised to sum‑1."""
    # use mid‑point rule to avoid exact zeros at the boundaries
    x = (np.arange(n) + 0.5) / n
    # prevent overflow / underflow – clip exponents
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # replace NaNs (caused by 0**negative) by a large finite value
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    if s <= 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Helper 7 – linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Optimiser 1 – projected Adam with soft‑max gradient
# ----------------------------------------------------------------------
def _run_adam_gd(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 step0: float = 0.08,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex using the soft‑max approximation of the
    max‑convolution.  Returns the best point seen and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential annealing)
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] adam improve: n={n}, it={it}, score={cur_score:.6f}")

        # temperature for the soft‑max
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of the smoothed max
        grad = _softmax_gradient(a, conv, T)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # element‑wise Adam step, then project back to simplex
        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimiser 2 – cheap simulated‑annealing refinement (tiny mass moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that transfers a small fraction of mass between two slots."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.1   # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] anneal improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score
