def _run_subgradient_gd(a0, n, deadline, start_time, step0=1.0, max_iter=10**7):
    a = a0.copy()
    best_a = a.copy(); best_score = np.inf
    it=0; while time.time()<deadline and it<max_iter:
        it+=1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # identify active indices (multiple)
        # tolerance: maybe 1e-12 absolute, but due to floating errors, we may use relative tolerance.
        tol = 1e-12
        active = (conv >= max_val - tol)
        # weight vector w: uniform across active (if many)
        w = active.astype(float)
        # compute gradient
        rev_a = a[::-1]
        grad_full = _conv_rev_weight(rev_a, w)
        grad = 2.0 * grad_full[n-1:n-1+n]

        # step size schedule
        step = step0 / math.sqrt(it)

        # Mirror update
        # Use log domain
        log_a = np.log(a + 1e-20) - step * grad
        # renormalize
        max_log = np.max(log_a)  # stability
        a = np.exp(log_a - max_log)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            # fallback uniform
            a = np.full_like(a, 1.0/n)

    return best_a, best_score

"""
Propose Candidate – a convex sub‑gradient optimiser for the step‑function
problem.

Algorithm
---------
* Initialise from the best known construction (`height_sequence_1`) if it exists;
  otherwise start from a random probability vector.
* Normalise the vector to sum‑to‑one (the objective 2·n·max(conv) is scale‑invariant).
* Run a **projected sub‑gradient descent** on the exact ‑∞‑norm of the
  self‑convolution:
      – compute the full convolution a*a,
      – identify the set of indices attaining the maximal value,
      – obtain a sub‑gradient by convolving the reversed a with the indicator of
        the active set,
      – take a diminishing step (step₀/√t) and project back onto the probability
        simplex.
* The descent loop runs for as long as the time‑budget allows (≈80 % of the
  total time).  The best point seen so far (lowest 2·n·max(conv)) is kept.
* With the remaining time a short **simulated‑annealing** walk performs tiny
  mass‑transfer moves between random coordinates, accepting worsening moves
  with a temperature schedule.  This often removes the last few bits of
  sub‑optimality.
* The final sequence is returned as a plain Python list (all entries are
  non‑negative and sum to one).

The method works directly on the true max‑convolution (no soft‑max
approximation) and therefore converges to a true minimiser of the
evaluation function, given enough iterations.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3 – convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed sequence a_rev with a weight vector w.
    Returns the full linear convolution (length = len(a_rev)+len(w)-1).
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4 – linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (projected onto the simplex)
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.6,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   f(a) = max_k (a*a)_k .
    Returns the best point encountered and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    tol = 1e-12                     # tolerance for selecting the active set

    # pre‑compute a few objects to avoid attribute look‑ups in the loop
    proj = _project_simplex
    conv_self = _conv_self
    conv_rev_weight = _conv_rev_weight

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # ---- sub‑gradient of the exact max ----
        active = conv >= max_val - tol
        w = active.astype(float)                     # weight vector (0/1)
        rev_a = a[::-1]
        full = conv_rev_weight(rev_a, w)            # length = 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]          # central block

        # diminishing step size
        step = step0 / math.sqrt(it)

        # Euclidean projected gradient step
        a = proj(a - step * grad)

        # (optional) early exit if the improvement stalls
        if it % 10000 == 0:
            # safeguard against pathological stagnation
            if step * np.linalg.norm(grad) < 1e-12:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1    # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, runs a projected sub‑gradient
    descent on the true ``max`` of the self‑convolution and finishes with a
    simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Starting point – use the previously discovered construction if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0:
            starter /= s
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a target length.
    # The previous best used 9600 entries; we keep the same (larger lengths
    # would increase the linear factor 2·n and are not beneficial without a
    # proportional reduction of max(conv)).
    # ------------------------------------------------------------------
    target_n = 9600
    if starter is not None:
        if starter.shape[0] != target_n:
            a0 = _upsample(starter, target_n)
        else:
            a0 = starter.copy()
    else:
        a0 = np.random.rand(target_n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Allocate time: ~80 % for the sub‑gradient phase, the rest for annealing.
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    subgrad_time = total_time * 0.80
    anneal_time = total_time - subgrad_time

    # ------------------------------------------------------------------
    # Sub‑gradient optimisation
    # ------------------------------------------------------------------
    sub_deadline = time.time() + subgrad_time
    best_seq, best_score = _run_subgradient_gd(
        a0, target_n, sub_deadline, start_time,
        step0=0.6, max_iter=10 ** 8
    )
    print(f"[{time.time() - start_time:6.2f}s] sub‑gradient finished: "
          f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final annealing phase (uses whatever time remains)
    # ------------------------------------------------------------------
    remain = deadline - time.time()
    if remain > 5.0:
        anneal_deadline = time.time() + min(anneal_time, remain)
        best_seq, best_score = _run_annealing(
            best_seq, target_n,
            anneal_deadline, start_time,
            best_score, best_seq,
            start_temp=2e-2, end_temp=1e-6,
            steps=800_000
        )
        print(f"[{time.time() - start_time:6.2f}s] final anneal finished: "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (tiny numerical drift may have occured)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
