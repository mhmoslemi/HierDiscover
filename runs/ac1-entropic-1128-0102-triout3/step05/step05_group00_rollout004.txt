"""
Propose a step‑function with a low self‑convolution peak.

Algorithm (≈1000 s budget)
--------------------------------
1. **Warm‑start** – read the previous best sequence if available.
2. **Linear‑programming refinement** (n≈2048):
   * For a distribution `f` on the simplex compute `M = max(f*f)`.
   * Solve the LP   maximise Σ g_i   s.t. g≥0 and (f ∗ g) ≤ M  (linear constraints).
   * Normalise the LP solution `g` to sum = 1 and mix:
         f ← (1‑t)·f + t·g  (t≈0.2).
   * Iterate a few times, keeping the best vector.
   * A few Adam‑steps + cheap annealing polish the LP result.
3. **Multi‑scale up‑sampling** – repeatedly double the length
   (4096 → 8192 → 16384 → 32768).  At each scale:
   * Upsample the current best distribution (linear interpolation and re‑normalisation).
   * Run Adam‑style optimisation on a smooth log‑sum‑exp surrogate of the
     ℓ∞‑norm of the convolution (always project onto the probability simplex
     and enforce symmetry).
   * Finish with a targeted simulated‑annealing mass‑transfer polishing.
4. Return the best sequence found (as a plain list of floats).

The combination of an LP “direction‑finder” and a refined gradient/annealing
phase yields a lower convolution peak than the plain Adam+anneal pipeline
used previously.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – all top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v = 1, v ≥ 0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a ∗ a using FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights, numerically stable
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()

    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Adam‑style optimisation on the simplex (smooth ℓ∞ surrogate)
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best vector encountered and its exact objective value
    (2·n·max(conv) because the vector stays on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 0.3          # λ ≈ lam_factor * max(conv)
    lam_decay = 0.7
    lam_decay_interval = 2000

    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b          # exact objective (∑a = 1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # Update λ (never let it become zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # Gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Diminishing base step
        base_step = 0.6 / math.sqrt(it + 1.0)

        # Parameter update
        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # Optional symmetrisation (helps for many lengths)
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # Project back onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 100000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from i to j.
    Returns the best sequence found (if any) and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # Geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # Propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initial shape generators (different baselines)

def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a small collection of plausible starting vectors from a base.
    The returned vectors are already normalised to sum = 1.
    """
    variants = []

    # 1. the base itself (usually a warm‑start)
    variants.append(base_seq.copy())

    # 2. sinusoidal shape (positive on (0,π))
    n = base_seq.shape[0]
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)  # all positive
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3. triangular (ramp up then down)
    tri = np.concatenate([np.linspace(1, n // 2, n // 2, endpoint=False),
                          np.linspace(n // 2, 1, n - n // 2)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4. a random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants

# Linear‑programming refinement (fixed length, purely linear constraints)
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray,
               n: int,
               max_iters: int,
               mix_factor: float,
               slice_deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps.
    Returns the best vector seen and its exact objective value.
    """
    best_a = a.copy()
    best_score = float('inf')
    for it in range(max_iters):
        if time.time() >= slice_deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # Build dense constraint matrix A (rows = 2n‑1, cols = n)
        # A[k,i] = a[k-i] if 0 <= k-i < n else 0
        idx = np.arange(2 * n - 1)[:, None]   # (2n‑1,1)
        i = np.arange(n)[None, :]             # (1,n)
        diff = idx - i
        mask = (diff >= 0) & (diff < n)
        A = np.where(mask, a[diff], 0.0)       # shape (2n‑1, n)

        # Linear programme: maximise Σ g_i  ≡ minimise -Σ g_i
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – give up on further refinements
            break

        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            break

        # Normalise g to unit sum and mix
        g_norm = g / sum_g
        a = (1.0 - mix_factor) * a + mix_factor * g_norm
        a = _project_simplex(a)

        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Grab previous best if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑programming refinement on a moderate size
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_lp = 2048
        if starter is not None:
            a0 = _upsample(starter, n_lp)
        else:
            a0 = np.random.rand(n_lp)
            a0 /= a0.sum()

        n_restarts = 4
        for r in range(n_restarts):
            if time.time() >= deadline:
                break
            if r == 0:
                a = a0.copy()
            else:
                a = np.random.rand(n_lp)
                a /= a.sum()

            # 1) LP‑driven direction steps
            slice_deadline = time.time() + max(8.0, (deadline - time.time()) * 0.2)
            a, lp_score = _lp_refine(a, n_lp,
                                     max_iters=12,
                                     mix_factor=0.2,
                                     slice_deadline=slice_deadline,
                                     start_time=start_time)

            # 2) Adam polishing (with enforced symmetry – the problem is symmetric)
            if time.time() < deadline:
                adam_deadline = time.time() + max(6.0, (deadline - time.time()) * 0.2)
                a_adam, score_adam = _run_adam_opt(a, n_lp, adam_deadline,
                                                   start_time, symmetrize=True)
                if score_adam < lp_score:
                    a, lp_score = a_adam, score_adam

            # 3) cheap annealing polish
            if time.time() < deadline:
                anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.1)
                a_ann, score_ann = _run_annealing(a, n_lp, anneal_deadline,
                                                  start_time, best_score,
                                                  best_seq if best_seq is not None else a,
                                                  start_temp=5e-2,
                                                  end_temp=1e-5,
                                                  steps=120000)
                if score_ann < lp_score:
                    a, lp_score = a_ann, score_ann

            if lp_score < best_score:
                best_score = lp_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after LP n={n_lp} score={lp_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Multi‑scale up‑sampling + Adam + annealing
    # ------------------------------------------------------------------
    lengths = [4096, 8192, 16384, 32768]
    random.shuffle(lengths)   # avoid systematic bias when we stop early

    global_a = best_seq if best_seq is not None else starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # initialise a0 (warm start / several variants)
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # allocate a slice of remaining time for this length
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # larger lengths get a proportionally larger share
        fraction = 0.15 / (len(lengths) - idx)
        slice_deadline = time.time() + max(6.0, remaining * fraction)

        best_local_a = None
        best_local_score = float('inf')
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            # enforce symmetry about half of the time (helps convergence)
            sym = (np.random.rand() < 0.4)
            a_opt, score_opt = _run_adam_opt(start_vec, n,
                                             slice_deadline,
                                             start_time,
                                             symmetrize=sym)

            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        # final annealing refinement (if time permits)
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(best_local_a, n,
                                              anneal_deadline,
                                              start_time,
                                              best_score,
                                              best_seq if best_seq is not None else best_local_a,
                                              start_temp=1e-2,
                                              end_temp=1e-6,
                                              steps=150000)
            if score_ref < best_local_score:
                best_local_score = score_ref
                best_local_a = a_ref.copy()

        # update global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after up‑sample n={n} score={best_local_score:.6f}")

        # remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list (the evaluator will clip / normalise as needed)
    return [float(x) for x in best_seq]
