function project_simplex(v):
    # v is arbitrary real vector
    # ensures w >= 0, sum = 1
    # Implementation as per "Efficient projections onto the l1-ball".
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Project onto the probability simplex (sum 1, non-negative)."""
    # Based on algorithm from Duchi et al. 2008.
    # Input: 1-d numpy array of real numbers.
    # Output: projected array.
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that minimises `evaluate_sequence`.

The algorithm starts from the best known sequence (global `height_sequence_1`
if it exists, otherwise a uniform vector) and performs a large‐scale
sub‑gradient descent on the objective

    Φ(a) = 2·n·max(a*a)   (the sum of the coefficients is kept equal to 1)

The sub‑gradient of `max(a*a)` is obtained from the active convolution
indices (those attaining the current maximum).  At each iteration we move
in the negative sub‑gradient direction, project back onto the simplex
(`a_i ≥ 0, Σ a_i = 1`) and keep the best point ever seen.
Occasional random jitter is added to escape possible plateaus.
All work is done inside the time budget (default 1000 s).  The final
candidate is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1] - 1            # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: maximum of the full linear convolution a * a.
# Uses an FFT based convolution for speed.
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    C = A * A
    conv = np.fft.irfft(C, n=fft_len)[:m]
    # numerical noise may produce tiny negatives – clip them
    conv = np.maximum(conv, 0.0)
    return float(conv.max()), conv


# ----------------------------------------------------------------------
# Helper: sub‑gradient of max(a*a) at the current point.
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs from i_start … i_end (inclusive)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        g[i_range] += a[j_range]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) improves the
    current best score.  The routine runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 1) Choose a suitable length n and an initial vector a.
    # ------------------------------------------------------------------
    default_n = 3400                    # a modest size, enough for good ratios
    a = None
    n = default_n

    # try to reuse the previously found construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0 and prev.sum() >= 0.01:
            prev = prev / prev.sum()               # normalise to sum = 1
            a = prev.copy()
            n = prev.shape[0]
    except Exception:
        pass

    if a is None:
        # fall back to a uniform vector
        a = np.full(n, 1.0 / n, dtype=float)

    # keep track of the best solution encountered
    best_a = a.copy()
    max_c, _ = _max_convolution(best_a)
    best_score = 2.0 * n * max_c
    print(f"[0.00s] start length {n}, initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Main optimisation loop (sub‑gradient descent + occasional jitter)
    # ------------------------------------------------------------------
    iter_counter = 0
    eta0 = 1.0                     # initial step size
    epsilon = 1e-12                # tolerance for active set detection
    jitter_interval = 300          # add random noise every few iterations
    jitter_scale = 0.02            # size of the jitter (relative to the simplex)

    while time.time() - start_time < time_limit:
        iter_counter += 1

        # ---- compute convolution and current objective -----------------
        max_c, conv_full = _max_convolution(a)
        cur_score = 2.0 * n * max_c

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] new best score {best_score:.6f}")

        # ---- build the sub‑gradient ------------------------------------
        # active indices are those achieving the maximum (within a tiny tolerance)
        active = np.where(conv_full >= max_c - epsilon)[0]
        grad = _subgradient(a, conv_full, active)

        # ---- step size (simple diminishing schedule) -------------------
        eta = eta0 / np.sqrt(iter_counter)

        # ---- gradient step (project back onto the simplex) -------------
        a_candidate = _project_simplex(a - eta * grad)

        # occasionally try a small random perturbation to escape flat regions
        if iter_counter % jitter_interval == 0:
            noise = jitter_scale * (np.random.rand(n) - 0.5)
            a_candidate = _project_simplex(a_candidate + noise)

        a = a_candidate

        # safety‑net: if we have exhausted most of the budget, break early
        if time.time() - start_time > time_limit:
            break

    # ------------------------------------------------------------------
    # 3) Return the best vector found (as a plain Python list)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    total_time = time.time() - start_time
    print(f"[{total_time:.2f}s] finished, best score {final_score:.6f}")

    # The harness expects a list of floats
    return [float(x) for x in best_a]
