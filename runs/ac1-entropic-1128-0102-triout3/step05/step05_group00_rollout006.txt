"""
Propose a low‑peak step‑function.

The optimiser works on the probability simplex (∑aᵢ = 1) because the
evaluation is scale‑invariant.  It consists of three phases:

1. **Progressive warm‑start** – start with a tiny vector (≈256 entries),
   optimise it, then linearly up‑sample to larger sizes.  The time allocated
   to each size is proportional to the size, so the final (largest) size
   receives most of the budget.

2. **Adam on a log‑sum‑exp surrogate** – the ℓ∞ norm of the convolution is
   approximated by a soft‑max with a temperature λ that is gradually
   reduced.  Adam’s adaptive moments give fast convergence; after each
   update the vector is projected onto the simplex.

3. **Mass‑transfer simulated annealing** – a cheap neighbourhood move
   (transfer a small ε from one index to another) is accepted with a
   decreasing temperature.  This polishes the Adam result and can escape
   tiny plateaus.

The algorithm is deliberately lightweight (no external solvers) and

* respects the 1000 s budget,
* works for any length up to ≈65000,
* returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast for all n we use)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (full) convolution a * a using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


# ----------------------------------------------------------------------
# Upsampling / interpolation (preserves sum)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp surrogate for max(conv)
# ----------------------------------------------------------------------
def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of   L_λ(a) = λ log Σ exp(conv_i/λ)   w.r.t. a.

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter > 0

    Returns
    -------
    grad : ndarray, shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights, numerically stable
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # gradient = 2 * (w * a) convolved with reversed a
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best vector seen and its exact objective 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 0.30          # λ ≈ lam_factor * max(conv)
    lam_decay = 0.70
    lam_decay_every = 500      # iterations between decays

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam n={n} it={it} score={cur_score:.6f}")

        # update λ (never let it become zero)
        if it % lam_decay_every == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-15)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size (≈ 1/√it)
        step = 0.6 / math.sqrt(it + 1.0)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex
        a = _project_simplex(a)

    # return the best vector observed (not the final a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polishing
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray,
            n: int,
            deadline: float,
            start_time: float,
            cur_best_score: float,
            cur_best_seq: np.ndarray,
            start_temp: float = 5e-2,
            end_temp: float = 1e-5,
            max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealer: move a small ε from i to j.
    Returns an improved sequence (if any) and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (step / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # transfer up to 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal n={n} step={step} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initial shape generators (seed the optimiser)
# ----------------------------------------------------------------------
def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """Return a few plausible start vectors (already normalised)."""
    n = base_seq.shape[0]
    variants = [base_seq.copy()]

    # sinusoidal (positive half‑wave)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # raised‑cosine (Hann window)
    hann = 0.5 * (1 - np.cos(2 * np.pi * (np.arange(n) + 0.5) / n))
    hann /= hann.sum()
    variants.append(hann)

    # random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The routine respects the given time budget (default 1000 s) and returns a
    plain ``list[float]``.  It uses a progressive warm‑start, Adam on a smooth
    ℓ∞ surrogate and a final simulated‑annealing polishing phase.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously found sequence if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length schedule – the final (largest) length gets the lion’s share
    # ------------------------------------------------------------------
    # Very small start (fast to optimise)
    start_n = 256
    if starter is not None:
        a_current = _upsample(starter, start_n)
    else:
        a_current = np.random.rand(start_n)
        a_current /= a_current.sum()

    # Define a progressive schedule.  The total “weight” is the sum of the
    # lengths; each stage receives a proportion of the remaining time
    # equal to its length / total_remaining_length.
    schedule = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768, 65536]

    # Ensure we never exceed the deadline: stop when time runs out.
    best_score = float('inf')
    best_seq = None

    total_remaining = sum(schedule)
    for n in schedule:
        if time.time() >= deadline:
            break

        # allocate time slice proportional to length (at least 2 s)
        remaining = deadline - time.time()
        slice_time = max(2.0, remaining * n / total_remaining)
        slice_deadline = time.time() + slice_time

        # up‑sample the current vector to the new length
        a_up = _upsample(a_current, n)

        # generate a handful of distinct starts (warm‑start + random)
        init_candidates = _initial_variants(a_up)

        # ------------------------------------------------------------------
        # Run a few Adam optimisations (different starts)
        # ------------------------------------------------------------------
        best_local = None
        best_local_score = float('inf')
        # we give each candidate a tiny slice of the budget
        per_candidate_deadline = slice_deadline - time.time()
        per_candidate_deadline = per_candidate_deadline / max(len(init_candidates), 1)

        for cand in init_candidates:
            # break early if overall deadline is near
            if time.time() >= slice_deadline:
                break

            cand_deadline = min(slice_deadline, time.time() + per_candidate_deadline)
            a_opt, score_opt = _adam_optimize(cand, n, cand_deadline, start_time)
            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] Local Adam improve n={n} score={score_opt:.6f}")

        if best_local is None:
            # nothing improved – keep the up‑sampled version
            best_local = a_up.copy()
            best_local_score = 2.0 * n * _conv_self(best_local).max()

        # keep the globally best sequence seen so far
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} score={best_local_score:.6f}")

        # ------------------------------------------------------------------
        # Polishing via simulated annealing (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = min(deadline, time.time() + max(4.0, (deadline - time.time()) * 0.2))
            a_ref, score_ref = _anneal(best_local,
                                        n,
                                        anneal_deadline,
                                        start_time,
                                        best_score,
                                        best_seq,
                                        start_temp=5e-2,
                                        end_temp=1e-5,
                                        max_steps=200_000)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Anneal n={n} score={score_ref:.6f}")

        # Prepare for the next scaling step
        a_current = best_seq.copy()
        total_remaining -= n

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # uniform distribution over a modest size
        fallback_n = 512
        best_seq = np.full(fallback_n, 1.0 / fallback_n)

    # Return a plain Python list – the evaluator will clamp / normalise it.
    return [float(x) for x in best_seq]
