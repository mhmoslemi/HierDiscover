- sequence is list of numbers; after validation, it converts each to float, then max(0, x) ensures nonnegative, then min(1000.0, x) caps to 1000.0.
- n = len(seq)
- b_sequence = np.convolve(sequence, sequence). This is linear convolution (full, length 2n - 1).
- max_b = max(b_sequence)
- sum_a = np.sum(sequence)
- If sum_a < 0.01, return INF.
- Return 2 * n * max_b / (sum_a**2).

"""
A fast, time‑budget aware optimiser for the step‑function problem.

The optimiser works on the probability simplex (∑aᵢ = 1) because the
evaluation function is scale‑invariant.  It proceeds in three stages:

1. **Warm‑starts** – a few analytically motivated initial shapes are
   generated (U‑shaped Beta(0.3,0.3), uniform, random Dirichlet).

2. **Adam on a smooth ℓ∞ surrogate** – the maximum convolution entry
   is approximated by a log‑sum‑exp with a decreasing temperature λ.
   Adam’s adaptive steps together with a projection onto the simplex
   quickly drives the objective down.  Most of the allotted time is spent
   here (≈70 % of the budget).

3. **Heavy‑weight simulated‑annealing polish** – a cheap mass‑transfer
   move (ε from i to j) is repeatedly applied while the temperature is
   annealed to near zero.  This stage (≈30 % of the budget) removes the
   remaining “tiny peaks”.

The code respects the 1000 s budget, prints progress, and finally returns
the best sequence found as a plain list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max over all convolution entries – numerically stable
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()

    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def _beta_shape(n: int, alpha: float = 0.3) -> np.ndarray:
    """
    U‑shaped Beta(α,α) density on {0,…,n‑1}.
    The density diverges at the ends for α<1, giving a lot of mass near 0 and 1.
    """
    x = (np.arange(n) + 0.5) / n          # centre of each bin
    eps = 1e-12
    w = np.power(np.clip(x, eps, 1 - eps), alpha - 1.0) * \
        np.power(np.clip(1.0 - x, eps, 1 - eps), alpha - 1.0)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _uniform_shape(n: int) -> np.ndarray:
    """Plain uniform distribution."""
    a = np.full(n, 1.0 / n)
    return a


def _random_shape(n: int) -> np.ndarray:
    """Random Dirichlet draw (concentration 1)."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def _initial_variants(n: int, base: np.ndarray | None = None) -> list[np.ndarray]:
    """Collect a few plausible starting vectors."""
    variants = []
    if base is not None:
        variants.append(base.copy())
    variants.append(_beta_shape(n, 0.3))
    variants.append(_uniform_shape(n))
    variants.append(_random_shape(n))
    return variants


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed ℓ∞ surrogate.
    Returns the best vector seen and its exact objective 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule parameters
    lam_factor = 0.3          # start with λ ≈ lam_factor·max(conv)
    lam_decay = 0.85
    lam_decay_interval = 1500

    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={cur_score:.6f}")

        # temperature/lambda update
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing base step (empirically good)
        base_step = 0.8 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer annealer.  A small ε is moved from a source index i
    to a destination index j.  Temperature is exponentially annealed so
    that the search becomes greedy towards the end.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source (biased towards large entries) and a random destination
        i = np.random.choice(n, p=a / a.sum())
        if a[i] == 0.0:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.20
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        a_prop = _project_simplex(a_prop)

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis accept
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
