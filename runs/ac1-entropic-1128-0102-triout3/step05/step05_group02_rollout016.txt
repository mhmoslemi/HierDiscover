assistantfinal**Solution Overview**

The objective is  

\[
F(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where \((a*a)_k=\sum_i a_i\,a_{k-i}\) is the *linear* self‑convolution.
Because the function is homogeneous of degree 0 we may always work with a
probability vector \(\sum_i a_i = 1\).

The previous implementation used a sub‑gradient method that only followed the
currently active convolution entry.  
To obtain a stronger descent direction we minimise a **smooth approximation
of the maximum**:

\[
\phi_\beta(a)=\log\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr) .
\]

For \(\beta\rightarrow\infty\) the function converges to \(\max_k (a*a)_k\)
and its gradient can be written in closed form:

\[
\nabla\phi_\beta(a)=2\;{\rm corr}\bigl(w,a\bigr),\qquad
w_k=\frac{e^{\beta (a*a)_k}}{\sum_j e^{\beta (a*a)_j}} .
\]

The correlation can be computed in \(\mathcal O(n\log n)\) by a single FFT.
The algorithm therefore consists of

1. **Warm‑start** – random vectors from a few parametric families, the best
   of which is kept.
2. **Smooth gradient descent** – a schedule of increasing \(\beta\) values,
   diminishing step size and Euclidean projection onto the simplex.
3. **Optional sub‑gradient refinement** – the earlier method is applied for
   a few extra iterations.
4. **Cheap annealing** – mass‑transfer moves with a geometric temperature
   schedule for polishing.
5. **Upsampling** – the best vector is linearly interpolated to a larger
   length and the whole process repeats.

All steps respect the overall 1000 s time budget.

```python
"""
Improved search for a low‑maximum self‑convolution step‑function.

The core is a smooth‑max gradient descent: we minimise
    φ_β(a) = log Σ_k exp(β·(a*a)_k)
which converges to the true max as β grows.
The gradient is obtained analytically as a correlation
    ∇φ = 2·corr(w, a)   with w the soft‑max weights.
The correlation is computed via a single FFT, giving an
O(n log n) iteration that works well for lengths up to 2·10⁴.

The routine performs:
  * a quick random‑family warm‑start,
  * smooth‑gradient descent with an increasing β schedule,
  * a short sub‑gradient phase (the old method) for extra robustness,
  * a cheap mass‑transfer annealing for polishing,
  * progressive up‑sampling to larger lengths.

All iterates are projected onto the probability simplex
∑a_i = 1, a_i ≥ 0, guaranteeing scale‑invariance of the final score.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Exact linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def score_of(a: np.ndarray) -> float:
    """Exact objective value for a normalized sequence (∑a=1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Shape families for quick random warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n‑i‑0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term == 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ given in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate (β‑softmax of the convolution)
# ----------------------------------------------------------------------


def compute_gradient(weights: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute ∇φ = 2·corr(weights, a) where `weights` has length 2n‑1
    and `a` has length n.  The correlation is obtained by a single FFT.
    """
    n = a.shape[0]
    # correlation = (weights * a_rev)  (full)  -> take the 'valid' centre part
    a_rev = a[::-1]
    L = len(weights) + len(a_rev) - 1          # = 3n‑2
    fft_len = 1 << ((L - 1).bit_length())
    W = np.fft.rfft(weights, fft_len)
    A = np.fft.rfft(a_rev, fft_len)
    conv_full = np.fft.irfft(W * A, fft_len)[:L]
    start = n - 1                              # centre of the full convolution
    grad = 2.0 * conv_full[start:start + n]    # length n
    return grad


def smooth_gradient_descent(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth‑max surrogate.
    A schedule of increasing β values is used; after each step we project onto
    the probability simplex.  The best exact score seen is returned.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 1.0                 # initial step size (scaled by 1/√t)
    it = 0

    # β schedule – start smooth, finish very close to the true max
    beta_schedule = [0.3, 0.8, 2.0, 5.0, 12.0, 30.0]

    for beta in beta_schedule:
        while time.time() < deadline:
            it += 1
            conv = conv_self(a)                     # exact self‑convolution
            max_c = conv.max()

            # soft‑max weights (stable)
            shifted = conv - max_c
            exp_vals = np.exp(beta * shifted)
            w = exp_vals / exp_vals.sum()

            # gradient
            grad = compute_gradient(w, a)

            # normalise direction (independent of n)
            norm = np.linalg.norm(grad) + 1e-16
            direction = grad / norm

            step = step0 / math.sqrt(it)
            a = project_simplex(a - step * direction)

            # keep the best exact score
            sc = score_of(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] smooth‑grad improve n={n} it={it} β={beta:.1f} score={sc:.6f}")

            # occasional tiny random perturbation to escape flat plateaus
            if it % 800 == 0:
                a += 1e-5 * np.random.randn(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (the old method – used as a short refinement)
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """Exact ∞‑norm sub‑gradient descent (same code as before, kept for robustness)."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    step0 = 0.8
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        idx = conv.argmax()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # sub‑gradient of the active entry
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm
        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * direction)

        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Very cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 80000) -> tuple[np.ndarray, float]:
    """Move a small ε from a source index to a destination index (simulated annealing)."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # keep a small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from previous run (global variable `height_sequence_1`)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter = starter / s
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – start small, increase gradually.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    random.shuffle(lengths)   # avoid systematic bias when stopping early

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a0 (warm start / random)
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Time slice for this length
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # give a bit more budget to later (larger) lengths
        fraction = 0.11 / max(1, len(lengths) - idx)
        slice_deadline = time.time() + max(6.0, remaining * fraction)

        # ------------------------------------------------------------------
        # Quick stochastic search over simple parametric families
        # ------------------------------------------------------------------
        shape_deadline = time.time() + min(2.0, (slice_deadline - time.time()) * 0.2)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2])
            if kind == 0:                     # product‑power family
                p = np.random.uniform(-1.8, 2.2)
                cand = shape_product_power(n, p)
            elif kind == 1:                   # Gaussian family
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            else:                             # Beta family
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}")

        # ------------------------------------------------------------------
        # Smooth gradient descent (main optimisation)
        # ------------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = smooth_gradient_descent(local_best_a, n, grad_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after smooth‑grad n={n} score={sc_opt:.6f}")

        # ------------------------------------------------------------------
        # Optional short sub‑gradient refinement (helps escape occasional plateaus)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            sub_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.15)
            a_opt2, sc_opt2 = subgradient_descent(a_opt, n, sub_deadline, start_time)
            if sc_opt2 < best_score:
                best_score = sc_opt2
                best_seq = a_opt2.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={sc_opt2:.6f}")

        # ------------------------------------------------------------------
        # Cheap annealing refinement (polish)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.2)
            a_ref, sc_ref = annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=80000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}")

        # keep the best vector for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (the evaluator will clamp/normalise as needed)
    return [float(x) for x in best_seq]
```
