Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.
The algorithm is a hybrid of three inexpensive optimisation phases:

1. **Progressive up‑sampling** – start from a short vector and repeatedly
   double its length, keeping the distribution normalised (sum = 1).
2. **Smooth gradient descent** – minimise a *soft‑max* approximation of
   the max‑convolution (as in the reference code).  This gives a good
   initial basin.
3. **Exact sub‑gradient descent** – after the soft stage we descend on the
   true max‑convolution using the sub‑gradient  g_i = 2·a_{k‑i}
   (k = argmax conv).  This sharpens the solution.
4. **Long simulated‑annealing walk** – tiny random mass‑transfer moves
   with a slow cooling schedule to escape shallow local minima.

The whole routine respects the time budget (default 1000 s) and returns the
best sequence found.  If a previous construction (`height_sequence_1`)
is available it is used as a warm start; otherwise several random
restarts are performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n)).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`
    and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of max‑convolution).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of the soft‑max of the convolution."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    # weighted gradient = 2·rev(a) * w (central n entries)
    rev_a = a[::-1]
    # convolution of rev_a with the weight vector w
    n = a.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) gradient descent on the soft‑max objective."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent for the true max‑convolution.
# ----------------------------------------------------------------------
def _run_subgradient_gd(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        step0: float = 0.1,
                        beta: float = 0.9,
                        max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected (Nesterov) sub‑gradient descent on 2·n·max(conv)."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        k = conv.argmax()                     # index attaining the max

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        if k < n:
            # i = 0 … k  →  j = k‑i is in range
            grad[:k + 1] = 2.0 * a[k::-1]
        else:
            start = k - n + 1                 # first i with a valid partner
            grad[start:] = 2.0 * a[start:][::-1]

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves).
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 0.2,
                   end_temp: float = 1e-7,
                   steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 20 % of the mass at i
        eps = a[i] * np.random.rand() * 0.20
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score
