"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm works with a soft‑max parametrisation  a_i = exp(y_i)/∑exp(y)
so that the non‑negativity and normalisation constraints are built‑in.
It minimises a smooth approximation of the max‑convolution
  fα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
by Adam optimisation while slowly increasing the smoothing parameter α.
The search proceeds in three resolution stages (≈800 → 2400 → 4800) and
finishes with a short simulated‑annealing refinement.  All work is
time‑limited by the supplied ``budget_s`` argument (≈1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of  (1/α)·log Σ_k exp(α·conv_k)  w.r.t. a.
    Returned shape is (n,).
    """
    n = a.shape[0]
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)   # stabilise
    w /= w.sum()                      # w_k = softmax(α·conv_k)

    # correlation a ⋆ w_rev  (grad_i = 2· Σ_k w_k a_{k-i})
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

    # aligning the result to the original indices
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Loss & gradient for Adam (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _smooth_loss_and_grad(a: np.ndarray, conv: np.ndarray,
                          alpha: float, fft_len: int):
    """
    Returns (loss, grad_a) where loss = (1/α)·log Σ exp(α·conv)
    and grad_a is the gradient w.r.t. a.
    """
    # loss (log‑sum‑exp)
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_shift = np.exp(scaled - max_scaled)
    sum_exp = exp_shift.sum()
    loss = (max_scaled + math.log(sum_exp)) / alpha

    # gradient w.r.t. a
    grad_a = _grad_smooth_max(a, conv, alpha, fft_len)
    return loss, grad_a


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = y.max()
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (via logits y)
# ----------------------------------------------------------------------
def _adam_optimize(y0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   alpha_schedule,
                   steps_per_alpha: int = 4000,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best probability vector found and its raw score
    (2·n·max(conv)).
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = _softmax(y)
    conv = _conv_self(best_a)
    best_score = 2.0 * n * conv.max()

    # FFT length needed for the gradient (a ⋆ w_rev)
    fft_len = _next_pow2(3 * n - 2)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline:
        it += 1

        # ---------- current probability vector ----------
        a = _softmax(y)

        # ---------- convolution ----------
        conv = _conv_self(a)

        # ---------- keep best ----------
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # ---------- smoothing parameter ----------
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # ---------- loss & gradient (w.r.t. a) ----------
        _, grad_a = _smooth_loss_and_grad(a, conv, alpha, fft_len)

        # ---------- gradient w.r.t. logits ----------
        # dL/dy = a * (dL/da - <a, dL/da>)
        inner = np.dot(a, grad_a)
        grad_y = a * (grad_a - inner)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)          # simple decay
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- advance α ----------
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to length ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _anneal_refine(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   max_steps: int = 250_000):
    """
    Small‑scale annealing that moves a random ε‑fraction of mass from a
    source index (chosen proportionally to its current weight) to a uniform
    destination.  Returns the best sequence seen.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        # choose source proportional to current mass, destination uniformly
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30            # move up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

            )
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] stage improve (n={n_target}, restart={r}): score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – modest size (n≈800)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    t1 = max(30.0, 0.10 * remaining)          # at least 30 s or 10 % of remaining
    _run_stage(
        starter,
        n_target=800,
        time_slice=t1,
        restarts=3,
        alpha_sched=[0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
        steps_alpha=3000,
        lr0=0.35,
    )

    # ------------------------------------------------------------------
    # Stage 2 – medium size (n≈2400)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        t2 = max(20.0, 0.25 * remaining)
        _run_stage(
            best_seq,
            n_target=2400,
            time_slice=t2,
            restarts=2,
            alpha_sched=[1.0, 5.0, 15.0, 30.0, 60.0, 120.0, 240.0],
            steps_alpha=6000,
            lr0=0.30,
        )

    # ------------------------------------------------------------------
    # Stage 3 – large size (n≈4800)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        t3 = max(20.0, 0.30 * remaining)
        _run_stage(
            best_seq,
            n_target=4800,
            time_slice=t3,
            restarts=2,
            alpha_sched=[5.0, 15.0, 30.0, 60.0, 120.0, 240.0, 480.0],
            steps_alpha=8000,
            lr0=0.25,
        )

    # ------------------------------------------------------------------
    # Stage 4 – polishing with a very high smoothing factor (almost exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.45 * remaining
        # Use a single restart with a very aggressive α‑schedule
        a0 = best_seq if best_seq is not None else np.full(4800, 1.0 / 4800)
        y0 = np.log(a0 + 1e-12)
        a_polish, score_polish = _adam_optimize(
            y0,
            n_target=a0.shape[0],
            deadline=polish_deadline,
            start_time=start_time,
            alpha_schedule=[100.0, 300.0, 1000.0, 3000.0, 10000.0],
            steps_per_alpha=2000,
            lr0=0.15,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing refinement (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _anneal_refine(
            best_seq,
            n=best_seq.shape[0],
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            max_steps=300_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise (just in case) and return as a Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Optional: report the achieved bound (use the provided evaluate_sequence)
    try:
        val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final bound = {val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
