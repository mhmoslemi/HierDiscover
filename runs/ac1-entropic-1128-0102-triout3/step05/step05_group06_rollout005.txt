"""
Hybrid optimisation for minimising the convolution‐max objective.

* Initialise with a symmetric arcsine (Beta(½,½)) density.
* Progressive up‑sampling to larger problem sizes (up to ≈ 30 k points).
* At each size run an Adam‑style projected gradient descent on the
  smooth approximation  (soft‑max) of the max convolution.
  The gradient is obtained by a fast FFT correlation (O(n log n)).
* Enforce the simplex constraint by a O(n log n) projection and keep the
  distribution symmetric (a ← (a+rev(a))/2) – symmetry is known to be
  beneficial for flattening the self‑convolution.
* After the deterministic stage perform a cheap simulated‑annealing
  walk that moves a tiny amount of mass between two positions.
* Finally renormalise and return the best sequence found within the
  allotted time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation w ⋆ a  (full linear correlation) via FFT
# ----------------------------------------------------------------------
def _corr_wa(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the linear (full) correlation (w ⋆ a)[i] = Σ_k w_k a_{k-i}.
    Returned array length = len(w)+len(a)-1.
    """
    L = len(w) + len(a) - 1
    N = 1 << (L - 1).bit_length()          # next power of two
    Fw = np.fft.rfft(w, n=N)
    Fa = np.fft.rfft(a, n=N)
    corr = np.fft.irfft(Fw * np.conj(Fa), n=N)[:L]
    return corr


# ----------------------------------------------------------------------
# Smooth‑max of a convolution and its gradient on the simplex
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Given a probability vector a (Σ a = 1, a ≥ 0) and smoothing parameter β,
    compute
        s = (1/β)·log Σ_k exp(β·(a*a)[k])          (smooth max)
    and the gradient ∇_a s.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    maxc = conv.max()

    # soft‑max weights, stabilized by subtracting maxc
    exp_beta = np.exp(beta * (conv - maxc))
    w = exp_beta / exp_beta.sum()            # Σ w = 1, non‑negative

    # smooth max value
    smooth_max = maxc + np.log(exp_beta.sum()) / beta

    # gradient: ∇ s = 2·(w ⋆ a)  (first n entries)
    corr_full = _corr_wa(w, a)               # length 3n‑2
    grad = 2.0 * corr_full[:n]               # only the first n shifts are needed
    return smooth_max, grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the simplex (with symmetry)
# ----------------------------------------------------------------------
def _run_smooth_gd(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iter: int = 10**7,
                   beta_start: float = 5.0,
                   beta_end: float = 200.0,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best a seen and its *exact* score = 2·n·max(a*a).
    """
    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    # exact score (since Σa = 1) = 2·n·max(conv)
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # schedule β (log‑linear from start to end)
        beta = beta_start * ((beta_end / beta_start) ** ((it - 1) / max_iter))

        # smooth‑max and gradient
        _, grad, conv = _smooth_max_and_grad(a, beta)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size (helps later stages)
        step = lr0 / math.sqrt(it)

        a -= step * (m_hat / (np.sqrt(v_hat) + eps))

        # project onto the simplex and enforce symmetry
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5            # keep distribution symmetric

        # exact score using the (cheap) max of the convolution
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] gd n={n:5d} iter={it:7d} "
                  f"score={cur_score:.6f} β={beta:.2f}")

        # optional early stop when improvement stalls (not mandatory)
        if it % 2000 == 0 and time.time() > deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Very cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 5 % of the source mass
        eps = a[i] * np.random.rand() * 0.05
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # re‑symmetrise (keeps the distribution symmetric)
        a_prop = (a_prop + a_prop[::-1]) * 0.5

        # evaluate new score (exact)
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} iter={it:7d} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    # enforce symmetry after up‑sampling
    up = (up + up[::-1]) * 0.5
    return up


# ----------------------------------------------------------------------
# Initialise a symmetric arcsine (Beta(½,½)) density
# ----------------------------------------------------------------------
def _init_arcsine(n: int) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n               # points in (0,1)
    dens = 1.0 / np.sqrt(xs * (1.0 - xs))       # unnormalised arcsine
    a = dens / dens.sum()
    a = (a + a[::-1]) * 0.5                     # enforce symmetry explicitly
    return a
