def compute_score(a):
    n = a.shape[0]
    # conv length 2n-1 via FFT
    m = 1 << ((2*n - 2).bit_length())
    Af = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(Af*Af, n=m)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a**2)

"""
Optimisation of a step‑function for the self‑convolution bound.

The algorithm proceeds in three stages:

1. **Parametric warm‑start** – a one‑parameter family of U‑shaped
   distributions  a_i ∝ (x_i (1‑x_i))^{-p}  (with x_i = (i+0.5)/n) is
   explored on a coarse grid and then refined with a golden‑section
   search.  The best parameter p provides an excellent initial point
   (p≈0.2 gives a near‑uniform sum distribution).

2. **Sub‑gradient descent on the exact ℓ∞‑norm** – the current maximum
   entry of the self‑convolution is located, its sub‑gradient
   (g_i = a_{k*‑i}) is computed and a multiplicative (exponentiated‑gradient)
   update a_i ← a_i·exp(−η·g_i) is performed, followed by re‑normalisation
   to the probability simplex.  The step size η decays as 1/√t.
   This directly reduces the peak of the convolution.

3. **Optional simulated‑annealing polish** – a cheap mass‑transfer
   annealer (identical to the one used in the previous baseline) finishes
   the search if time permits.

All heavy work (convolution) is performed with FFTs, which makes the
method fast enough to evaluate thousands of candidates within the
1000 s budget.  The final sequence is returned as a plain list of
floats, ready for `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution via FFT (fast for the lengths we use)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    # length needed for linear convolution = 2n‑1; next power of two:
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (used only as a fallback)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score (exact objective) – identical to evaluate_sequence but faster
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a probability vector."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_self(a)
    max_b = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Parametric family a_i ∝ (x_i (1‑x_i))^{-p}
# ----------------------------------------------------------------------
def _powerlaw_distribution(p: float, n: int) -> np.ndarray:
    """Return a symmetric U‑shaped distribution with shape exponent p."""
    # centre points of sub‑intervals to avoid exact 0/1
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # protect against overflow at the boundaries
    eps = 1e-14
    prod = np.clip(x * (1.0 - x), eps, None)
    a = prod ** (-p)
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Golden‑section search on a one‑dimensional parameter
# ----------------------------------------------------------------------
def _golden_section_search(func, a, b, n, max_iter=12, rel_tol=1e-5):
    """Find p∈[a,b] minimising func(p) for a fixed length n."""
    phi = (1 + 5 ** 0.5) / 2
    invphi = 1 / phi
    c = b - (b - a) * invphi
    d = a + (b - a) * invphi
    fc = func(c, n)
    fd = func(d, n)
    for _ in range(max_iter):
        if abs(b - a) < rel_tol * (abs(a) + abs(b)):
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - (b - a) * invphi
            fc = func(c, n)
        else:
            a, fc = c, fd
            c = d
            d = a + (b - a) * invphi
            fd = func(d, n)
    # return the best of the two interior points
    return (c, fc) if fc < fd else (d, fd)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _run_subgradient(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     best_score: float,
                     best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated‑gradient) descent on the ℓ∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    # initial step size – tuned empirically
    step0 = 0.7

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val          # sum(a) == 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgradient improve it={it} score={cur_score:.6f}")

        # locate the (first) maximum index
        kstar = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a: g_i = a_{k*‑i}
        idx = kstar - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = a[idx[mask]]

        # step size schedule η_t = step0 / sqrt(t)
        eta = step0 / math.sqrt(it + 1.0)

        # multiplicative update (exponentiated‑gradient)
        a = a * np.exp(-eta * g)

        # re‑normalise to the simplex
        total = a.sum()
        if total > 0.0:
            a /= total
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional small random jitter to escape plateaus
        if it % 1000 == 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                eps = a[i] * np.random.rand() * 0.1
                a[i] -= eps
                a[j] += eps
                a = a / a.sum()
    return best_seq, best_score

# Simple mass‑transfer annealing (as in the previous baseline)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_score: float,
                   best_seq: np.ndarray,

    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * _conv_self(a).max()
    best_local_seq = best_seq.copy()
    best_local_score = best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)

            a = a_prop

            best_local_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve it={it} score={score:.6f}")

        idx = np.argmin(np.abs(grid - p))
        left = max(0.0, grid[max(0, idx - 1)] - 0.02)
        right = min(0.45, grid[min(len(grid) - 1, idx + 1)] + 0.02)

        best_p, best_p_score = _golden_section_search(_score, left, right, n,
                                                      max_iter=10)
        a = _powerlaw_distribution(best_p, n)
        score = _score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] golden p={best_p:.6f} score={score:.6f}")

    # ------------------------------------------------------------------
    # If we still have no sequence (should not happen) fall back to uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # Stage 2: sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:   # reserve a few seconds for anneal
        sub_deadline = deadline - 5.0
        best_seq, best_score = _run_subgradient(best_seq,
                                                sub_deadline,
                                                start_time,
                                                best_score,
                                                best_seq)

    # ------------------------------------------------------------------
    # Stage 3: annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        best_seq, best_score = _run_annealing(best_seq,
                                              anneal_deadline,
                                              start_time,
                                              best_score,
                                              best_seq,
                                              start_temp=5e-2,
                                              end_temp=1e-5,
                                              steps=120000)

    # ------------------------------------------------------------------
    # Return the final candidate as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
