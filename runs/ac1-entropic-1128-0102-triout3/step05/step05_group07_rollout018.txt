"""
Optimise a non‑negative step function (a probability vector) to minimise

        score = 2·n·max(conv(a,a)) / (sum(a)²)

using a two‑phase scheme:

1.  A short Adam optimisation on a smoothed max (soft‑max) of the
    convolution.  This quickly moves the vector into a good basin.
2.  A long exponential‑gradient (mirror‑descent) phase that works with the
    exact L∞‑norm of the convolution (hard‑max).  The update is

        a_i ← a_i·exp(−η·g_i) ,   g = 2·(a_rev * w)

    where *w* is a uniform distribution over the (almost) active
    anti‑diagonals.  The step size η is decreased geometrically.

All time is bounded by ``budget_s`` seconds; the routine keeps the best
vector seen (according to the supplied ``evaluate_sequence``) and returns
it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max(conv(a,a)), full linear convolution)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """Gradient of Σ_k w_k·conv_k w.r.t. a (i.e. 2·(a_rev * w))."""
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.07,
                   temp_end: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best vector seen and its (2·n·max_conv) score.
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max over the convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient of the smooth‑max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser that minimises the
    exact max‑convolution.  Returns the best vector and its (2·n·max_conv)
    score.
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # active anti‑diagonals (hard‑max sub‑gradient)
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient for the hard‑max norm
        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric step‑size schedule
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative mirror‑descent update
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # avoid overflow in exp
        a = a * np.exp(upd)

        # re‑normalise to simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score


# ----------------------------------------------------------------------
def _interpolate_sequence(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, old_seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq = new_seq / s
    else:
        new_seq = np.full(new_n, 1.0 / new_n)
    return new_seq
