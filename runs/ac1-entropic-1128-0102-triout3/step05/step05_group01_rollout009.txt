def _run_optimizer(a0, n, end_time, start_time):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5
    T_start = 0.1
    T_end = 1e-7
    max_iter = 10**7
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < end_time and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] optimizer improve: n={n}, iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end/T_start) ** (it / 50000.0) # more iterations for schedule

        # weighted softmax
        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)

        step = step0 / math.sqrt(it)  # or constant step?
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)
    return best_a, best_score

def _run_adam(a0, n, deadline, start_time, step0=0.5, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=10**7):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    # temperature schedule parameters
    T_start = 0.05
    T_end = 1e-7
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it}, score={cur_score:.6f}")
        # temperature decays
        T = T_start * (T_end / T_start) ** (it / 80000.0)  # slower schedule
        grad = _softmax_gradient(a, conv, T)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 / math.sqrt(it)   # diminishing step
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = _project_simplex(a)
    return best_a, best_score

"""
Propose a step‑function with minimal evaluation score.

Algorithm
---------
1.  Load a previous construction (``height_sequence_1``) if it exists and
    normalise it to a probability simplex.
2.  Repeatedly run a projected Adam optimiser on the *smooth* max of the
    self‑convolution.  The smoothness is provided by a soft‑max (log‑sum‑exp)
    with a temperature that is exponentially annealed from a relatively
    high start (≈0.05) down to ≈1e‑7.  This turns the nonsmooth objective
    into a well‑behaved one while still driving the true maximum down.
3.  After each optimisation run keep the best sequence obtained.
4.  When the allotted time for gradient optimisation is exhausted, perform
    a short simulated‑annealing walk that moves a small fraction of mass
    between two positions.  The walk accepts uphill moves with a geometric
    temperature schedule, allowing the final fine‑tuning of the solution.
5.  Return the best sequence found, converted back to a plain Python list.

All iterates are projected onto the probability simplex {x ≥ 0, Σx = 1},
so the evaluation value is invariant under scaling and the constraints
required by ``evaluate_sequence`` are satisfied automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: Linear self‑convolution via FFT (full length = 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (full length)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3: Convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution rev(a) * w.
    Result length = len(a_rev) + len(w) - 1.
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4: Gradient of a *soft‑max* of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k) · conv_k,  where
    w_k = exp(conv_k / T) / Σ_j exp(conv_j / T).
    The factor 2 appears because each unordered pair (i,j) contributes twice.
    """
    # stable soft‑max
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()

    # weighted gradient = 2·(rev(a) * w) and then take the central block
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)               # length = n + (2n‑1)‑1 = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]               # central n entries
    return grad


# ----------------------------------------------------------------------
# Optimiser: projected Adam on the smooth max‑convolution
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              step0: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with soft‑max approximation of the max.
    Returns the best iterate and its (scaled) score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 5.0e-2
    T_end = 1.0e-7
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # anneal temperature
        T = T_start * (T_end / T_start) ** (it / 80000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / math.sqrt(it)          # diminishing global step size
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 10 % of the mass at i towards j
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

    """
