"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the custom `evaluate_sequence` objective.
The optimiser works in three stages:

1. **Parametric warm‑start** – a quick sweep over several simple,
   analytically‑defined families (exponential, power‑law, beta,
   cosine, linear ramps, etc.).  The best member of the sweep is taken
   as the initial point for the next stage.

2. **Projected sub‑gradient descent** on the exact L∞ objective.
   At each iteration the shift producing the current maximal
   convolution value is identified; its gradient is simply
   `2 * a[shift‑i]` on the overlapping indices.  A diminishing
   step‑size and Euclidean projection onto the probability simplex
   keep the iterate feasible.  This step reliably pushes the max‑
   convolution below the uniform benchmark.

3. **Targeted annealing** – a cheap mass‑transfer simulated annealing
   that makes occasional ε‑moves between two coordinates to polish
   the result.

The algorithm is run for a sequence of increasing lengths
(1024 → 2048 → 4096 → 8192 → 16384 → 32768).  A warm‑start from the
previous length is obtained by linear interpolation and renormalisation.
All helper routines are top‑level (no closures) and only NumPy and the
standard library are used.  The final best sequence (as a plain Python
list) is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    # Make a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal (or negative); return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution (fast for the lengths we use)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (returns length 2n‑1)."""
    n = a.shape[0]
    # next power of two >= 2n‑1
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Af * Af, n=fft_len)
    return conv_full[: 2 * n - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


# ----------------------------------------------------------------------
# Utility: up‑sample (linear interpolation) and normalisation
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Exact objective (fast version): 2·n·max(conv)  (sum is 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Assumes sum(a)=1, a>=0."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families – generate a candidate distribution for a given n
# ----------------------------------------------------------------------


def _candidate_exponential(n: int, alpha: float, side: str) -> np.ndarray:
    """Exponential shape.  side='left','right','sym'."""
    idx = np.arange(n, dtype=float)
    if side == "left":
        vals = np.exp(-alpha * idx)
    elif side == "right":
        vals = np.exp(-alpha * (n - 1 - idx))
    else:  # symmetric
        centre = (n - 1) / 2.0
        vals = np.exp(-alpha * np.abs(idx - centre))
    return vals / vals.sum()


def _candidate_powerlaw(n: int, p: float, side: str) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^{-p} (or reversed)."""
    if side == "left":
        vals = 1.0 / ((np.arange(1, n + 1, dtype=float)) ** p)
    elif side == "right":
        vals = 1.0 / ((np.arange(1, n + 1, dtype=float))[::-1] ** p)
    else:  # symmetric (mirrored around centre)
        centre = (n - 1) / 2.0
        d = np.abs(np.arange(n, dtype=float) - centre) + 1.0
        vals = 1.0 / (d ** p)
    return vals / vals.sum()


def _candidate_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution‑like shape."""
    x = np.linspace(0.0, 1.0, n)
    vals = (x ** a) * ((1.0 - x) ** b)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _candidate_cosine(n: int, gamma: float) -> np.ndarray:
    """Cosine window raised to a power."""
    theta = np.pi * np.arange(n, dtype=float) / (n - 1)
    vals = (1.0 - np.cos(theta)) ** gamma
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _candidate_ramp(n: int, exponent: float, direction: str) -> np.ndarray:
    """Monotone ramp (optionally raised to a power)."""
    if direction == "up":
        vals = (np.arange(1, n + 1, dtype=float) ** exponent)
    else:  # "down"
        vals = ((np.arange(n, 0, -1, dtype=float) ** exponent))
    vals /= vals.sum()
    return vals


def _candidate_random(n: int) -> np.ndarray:
    """Uniform Dirichlet draw."""
    v = np.random.rand(n)
    v /= v.sum()
    return v


# ----------------------------------------------------------------------
# Quick sweep over the families – returns the best candidate found
# ----------------------------------------------------------------------


def _parametric_sweep(n: int,
                     slice_deadline: float,
                     start_time: float,
                     base_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Explore a modest number of simple families; keep the best."""
    best_a = base_seq.copy()
    best_score = _objective(best_a)

    # Helper to evaluate and possibly replace the current best
    def _maybe_update(a):
        nonlocal best_a, best_score
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] param sweep improve n={n} score={sc:.6f}")

    # --- Exponential -------------------------------------------------
    alphas = np.logspace(-4, 2, num=12)
    for alpha in alphas:
        if time.time() >= slice_deadline:
            return best_a, best_score
        _maybe_update(_candidate_exponential(n, alpha, "left"))
        _maybe_update(_candidate_exponential(n, alpha, "right"))
        _maybe_update(_candidate_exponential(n, alpha, "sym"))

    # --- Power law ---------------------------------------------------
    p_vals = np.linspace(0.0, 3.0, num=9)  # include 0 (uniform)
    for p in p_vals:
        if time.time() >= slice_deadline:
            return best_a, best_score
        _maybe_update(_candidate_powerlaw(n, p, "left"))
        _maybe_update(_candidate_powerlaw(n, p, "right"))

    # --- Beta‑like ---------------------------------------------------
    for _ in range(15):
        if time.time() >= slice_deadline:
            return best_a, best_score
        a = 10 ** random.uniform(-1, 2)   # random in [0.1,100]
        b = 10 ** random.uniform(-1, 2)
        _maybe_update(_candidate_beta(n, a, b))

    # --- Cosine windows -----------------------------------------------
    for g in (0.5, 1.0, 2.0, 3.0):
        if time.time() >= slice_deadline:
            return best_a, best_score
        _maybe_update(_candidate_cosine(n, g))

    # --- Linear ramps ------------------------------------------------
    for exp in (0.5, 1.0, 2.0):
        if time.time() >= slice_deadline:
            return best_a, best_score
        _maybe_update(_candidate_ramp(n, exp, "up"))
        _maybe_update(_candidate_ramp(n, exp, "down"))

    # --- Random draws ------------------------------------------------
    for _ in range(12):
        if time.time() >= slice_deadline:
            return best_a, best_score
        _maybe_update(_candidate_random(n))

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          slice_deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """Run sub‑gradient descent on  max(conv) with simplex projection."""
    a = a0.copy()
    a = _project_simplex(a)  # ensure feasibility
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    base_step = 0.6  # initial step magnitude
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={score:.6f}")

        # Gradient of the active shift (sub‑gradient of the max)
        grad = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        # vectorised fill
        idx = np.arange(lo, hi + 1, dtype=int)
        jdx = max_idx - idx
        grad[idx] = 2.0 * a[jdx]

        # Diminishing step size
        step = base_step / math.sqrt(it + 1.0)

        # Gradient step + projection
        a = a - step * grad
        a = np.maximum(a, 0.0)          # keep non‑negative before projection
        a = _project_simplex(a)

        # occasional random ε‑move to escape flat regions
        if it % 500 == 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                eps = a[i] * random.random() * 0.1
                a[i] -= eps
                a[j] += eps
                a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap mass‑transfer simulated annealing (polishing)
# ----------------------------------------------------------------------


def _anneal_polish(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  cur_best_score: float,
                  cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Simple annealing that moves a fraction of mass between two positions."""
    a = a0.copy()
    score = _objective(a)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    steps = 120_000
    start_temp = 5e-2
    end_temp = 1e-5
    for it in range(1, steps + 1):
        if time.time() >= slice_deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * random.random() * 0.2  # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > random.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

            starter = None
    except Exception:
        starter = None

    # Length schedule – we climb geometrically up to 32768.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]
    # shuffle order only when we are forced to stop early; otherwise go increasing
    # (larger lengths give more room for improvement)
    # random.shuffle(lengths)

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Warm‑start for this length
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.full(n, 1.0 / n)

        # allocate a time slice for this length
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # give a larger share to later (larger) lengths
        fraction = 0.18 / max(1, len(lengths) - idx)
        slice_deadline = time.time() + max(6.0, remaining * fraction)

        # --------------------------------------------------------------
        # 1) Parametric sweep (cheap, diversity)
        # --------------------------------------------------------------
        a_param, score_param = _parametric_sweep(n, slice_deadline, start_time, a0)

        # --------------------------------------------------------------
        # 2) Sub‑gradient optimisation (main workhorse)
        # --------------------------------------------------------------
        a_opt, score_opt = _subgradient_optimize(a_param, n, slice_deadline, start_time)

        # keep the best of the two (usually the sub‑gradient one)
        if score_opt < score_param:
            cur_best_a = a_opt
            cur_best_score = score_opt
        else:
            cur_best_a = a_param
            cur_best_score = score_param

        if cur_best_score < best_score:
            best_score = cur_best_score
            best_seq = cur_best_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after sub‑grad n={n} score={cur_best_score:.6f}")

        # --------------------------------------------------------------
        # 3) Polishing annealing (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _anneal_polish(
                cur_best_a,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # remember for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – very unlikely to be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(2048, 1.0 / 2048)

    # Return plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
