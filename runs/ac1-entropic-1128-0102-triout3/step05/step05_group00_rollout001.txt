"""
Hybrid optimizer for the self‑convolution L∞ problem.

The algorithm works on the probability simplex (∑a_i = 1) because the
objective returned by `evaluate_sequence` is scale‑invariant.
It combines several techniques that together fit into the 1000 s
budget:

1. **Multi‑scale warm‑starts** – a solution for a shorter vector is
   linearly interpolated to initialise the next larger length.

2. **Adam optimisation on a soft‑max surrogate** of the convolution
   peak.  The temperature λ is decreased continuously so the surrogate
   converges to the true L∞ norm.

3. **Deterministic sub‑gradient descent** on the exact max‑convolution
   (active‑set gradient).  This refines the solution after Adam.

4. **Targeted simulated annealing** – cheap mass‑transfer moves are
   accepted with a temperature schedule, providing a final polishing
   phase.

All vectors are projected onto the simplex after every update and the
distribution is symmetrised (a ← (a+a[::-1])/2) because optimal
solutions are believed to be symmetric.

The routine returns the best sequence found before the time limit.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) -------------------------------------------------
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based) -----------------------------------------
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (fast for all lengths)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – clip to 0.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    np.maximum(c, 0.0, out=c)
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Gradient of the smoothed L∞ surrogate (soft‑max) ------------------------
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing temperature λ > 0

    Returns
    -------
    grad: dense gradient of shape (n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (stable)
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()

    # cross‑correlation of weights with reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)  # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]   # dL/da_i
    return grad


# ----------------------------------------------------------------------
# Gradient of the exact max‑convolution (active‑set) -----------------------
# ----------------------------------------------------------------------


def _gradient_max_conv(a: np.ndarray, tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a * a)[k] at the current a.
    Returns (grad, max_value).
    """
    conv = _conv_self(a)
    max_val = conv.max()
    active = np.abs(conv - max_val) <= tol
    if not np.any(active):
        # fallback – treat the single max entry as active
        active = conv == max_val
    w = active.astype(float) / active.sum()
    a_rev = a[::-1]
    full = _conv_fft(w, a_rev)  # length 3n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate ---------------------------------
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate.
    Returns the best vector seen and its exact objective value
    (2·n·max(conv) because the vector stays on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule hyper‑parameters
    lam_factor = 0.3            # initial λ ≈ lam_factor·max(conv)
    lam_decay = 0.98           # per‑iteration decay
    lam_min_factor = 1e-9

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ
        lam = max(max_b * lam_factor, 1e-12)
        lam_factor = max(lam_factor * lam_decay, lam_min_factor)

        # gradient of smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing base step
        base_step = 0.8 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetrisation
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # project back onto simplex
        a = _project_simplex(a)

    return best_a, best_score

# Deterministic sub‑gradient descent on the exact max ------------------------
# ----------------------------------------------------------------------


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    step0 = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        grad, max_val = _gradient_max_conv(a)
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing) ---------------------------------
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Cheap annealing: move a small ε from a source index to a destination index.
    Returns the best sequence found during the run and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a handful of plausible starting vectors derived from a base
    distribution.  All returned vectors are normalised to sum = 1.
    """
    variants = []
    n = base_seq.shape[0]

    # 1. warm‑start copy
    variants.append(base_seq.copy())

    # 2. positive sine shape
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3. triangular shape
    left = np.linspace(1, n // 2, n // 2, endpoint=False)
    right = np.linspace(n // 2, 1, n - n // 2)
    tri = np.concatenate([left, right])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4. arcsine‑like shape (1/√(x(1‑x)))
    t = (np.arange(n) + 0.5) / n
    arc = 1.0 / np.sqrt(t * (1.0 - t) + 1e-12)
    arc /= arc.sum()
    variants.append(arc)

    # 5. random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants

            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} score={best_local_score:.6f}")

        # --------------------------------------------------------------
        # Deterministic sub‑gradient descent (refinement)
        # --------------------------------------------------------------
        if time.time() < deadline:
            subgrad_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.1)
            a_sub, score_sub = _run_subgradient(best_local_a, n, subgrad_deadline, start_time)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after subgrad n={n} score={score_sub:.6f}")

        # --------------------------------------------------------------
        # Final annealing polishing (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(
                best_local_a,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=150000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # Remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed (very unlikely to run out of time)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return as a plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
