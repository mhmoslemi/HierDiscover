"""
Propose a low‑score step‑function using a Frank‑Wolfe style linear‑oracle
(LP) combined with a multiplicative (exponential‑gradient) refinement.
The algorithm works on the probability simplex (sum = 1) so the objective
is scale‑invariant.

1. Initialise the vector from the previously found construction
   (height_sequence_1) or from a uniform distribution.
2. Repeatedly:
   * compute M = max(conv(f,f))
   * solve a linear programme  max ∑b_j  s.t.  f * b ≤ M,  b ≥ 0
   * obtain a feasible direction g = b / ∑b
   * mix f ← (1‑α) f + α g (small line‑search on α)
   * keep the best vector seen.
   The LP gives a sub‑gradient for the infinity‑norm of the auto‑convolution,
   guaranteeing monotone descent in practice.
3. After the LP phase a long multiplicative‑gradient (mirror‑descent)
   refinement is performed to fine‑tune the solution.
All heavy work (convolution, LP) is done with FFTs and sparse matrices,
so thousands of iterations fit easily into the 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as spopt

# ----------------------------------------------------------------------
# Helper: fast evaluation of the score (no clipping, we stay in the simplex)
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²).  Assumes a ≥ 0."""
    n = a.shape[0]
    # FFT based convolution
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    s = a.sum()
    if s <= 0.0:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Helper: return the maximum of the self‑convolution and the full conv array
def _max_conv_fft(a: np.ndarray) -> tuple[float, np.ndarray]:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


# ----------------------------------------------------------------------
# Pre‑compute the index structure for the convolution‑constraints matrix.
def _prepare_constraint_structure(n: int):
    """
    Return (row_indices, col_indices, shape) for the matrix A where
    (A b)_k = sum_i f_i * b_{k-i}  (0≤k≤2n‑2).
    The data array will be filled with the current f each iteration.
    """
    size = n * n
    rows = np.empty(size, dtype=int)
    cols = np.empty(size, dtype=int)

    base = np.arange(n, dtype=int)          # column numbers 0…n‑1
    offset = 0
    for i in range(n):
        rows[offset:offset + n] = i + base   # rows i … i+n‑1
        cols[offset:offset + n] = base
        offset += n

    shape = (2 * n - 1, n)
    return rows, cols, shape


# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float, rows, cols, shape):
    """
    Solve   max   ∑ b_j
    s.t.       (f * b)_k ≤ M   (for all k)
              b ≥ 0
    Returns the optimal b (numpy array, length n).
    """
    n = f.shape[0]
    # data = f_i repeated n times (once for each column of that i‑block)
    data = np.repeat(f, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=shape)

    c = -np.ones(n)                     # maximise sum → minimise –sum
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0, None)] * n

    res = spopt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'presolve': True})
    if not res.success:
        # In the unlikely event that the LP fails we fall back to a tiny step.
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Multiplicative (mirror‑descent) optimisation using the true
    L∞‑norm of the convolution as sub‑gradient.
    a is assumed to be on the simplex.
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = _score(a)
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _max_conv_fft(a)
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask of (almost) active anti‑diagonals
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient g = 2·(a_rev * w)  (via FFT)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=fft_len_grad)
        A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + n - 1]
        start = n - 1
        g = 2.0 * conv2[start:start + n]

        # decreasing step size (geometric schedule)
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)   # avoid overflow
        a = a * np.exp(upd)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

    return [float(x) for x in best_f]
