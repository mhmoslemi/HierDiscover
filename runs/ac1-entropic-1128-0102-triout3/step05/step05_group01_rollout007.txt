"""
Improved step‑function search.

Main ideas
~~~~~~~~~
* Progressive up‑sampling up to 19200 points (larger search space).
* Hybrid projected gradient descent:
      – a soft‑max approximation of the max–convolution with an aggressive
        temperature‑annealing;
      – once the temperature is very low we switch to the exact sub‑gradient of
        the max (average over all maximising shifts).
  Momentum (Nesterov) and a diminishing step size are used.
* A long simulated‑annealing refinement that moves a random fraction of mass
  between two positions; acceptance follows the Metropolis rule.
* Starts from the best known construction (``height_sequence_1``) if it is
  available, otherwise from a random simplex point.
* All time is budgeted (default 1000 s).  The routine returns the best sequence
  found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of reversed sequence a_rev with weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the smooth max: Σ_k w_k(conv_k)·conv_k with
    w_k = exp(conv_k/T)/Σ exp(conv/T).  The factor 2 comes from the
    derivative of aᵀJ_k a (symmetric Toeplitz matrix).
    """
    maxc = conv.max()
    # stable soft‑max
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)          # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]          # central block
    return grad


# ----------------------------------------------------------------------
# Exact sub‑gradient of the (hard) max‑convolution
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.  All maximizing shift indices are averaged.
    """
    maxv = conv.max()
    mask = (conv == maxv).astype(float)
    mask /= mask.sum()                          # average over ties
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, mask)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projected (Nesterov) gradient descent with hybrid gradient
# ----------------------------------------------------------------------
def _run_gradient(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent that uses a soft‑max gradient while the
    temperature is > 5e‑5 and switches to the exact max‑subgradient afterwards.
    Returns the best (lowest) 2·n·max(conv) seen and the corresponding vector.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # hyper‑parameters
    step0 = 0.6
    beta = 0.9
    T_start = 2.0e-2
    T_end = 5.0e-6           # very low final temperature

    # estimation of total needed iterations (for the schedule)
    # we keep a simple linear schedule in the exponent space.
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, it={it}, score={cur_score:.6f}")

        # temperature schedule (exponential decay)
        T = T_start * (T_end / T_start) ** (it / 30000.0)

        if T > 5.0e-5:
            grad = _softmax_gradient(a, conv, T)
        else:
            grad = _max_subgradient(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size (≈ 1/√t)
        step = step0 / math.sqrt(it)

        # projected update on the simplex
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that transfers a small random fraction of mass between two
    indices.  Temperature decays geometrically.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    # annealing schedule parameters
    start_temp = 5.0e-2
    end_temp = 1.0e-6
    # allow many steps but stop early if time runs out
    max_steps = 1_500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 30 % of the mass from i to j
        eps = a[i] * np.random.rand() * 0.30
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, it={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    The routine respects the given time budget, performs progressive up‑sampling,
    hybrid projected gradient descent and a long annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # normalise to a probability vector (does not affect the objective)
            s = starter.sum()
            if s > 0:
                starter = starter / s
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage definitions (length, fraction of remaining time)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600, 19200]   # progressive enlargement
    total_weight = sum(stage_lengths)

    total_time = deadline - start_time
    # reserve ~15 % for the final annealing phase (or at least 30 s)
    anneal_budget = max(30.0, 0.15 * total_time)
    grad_budget = total_time - anneal_budget

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # time share for this stage (proportional to its length)
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for the current size

            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:

            a0 /= a0.sum()

            a0, n, stage_deadline, start_time
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # update remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – works with the final length we ended up with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

            best_seq, n_final,

            best_score = score_ref

    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
