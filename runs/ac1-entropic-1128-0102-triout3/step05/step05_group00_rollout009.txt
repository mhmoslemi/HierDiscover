"""
Propose a low‑peak self‑convolution step function.

The optimiser works on the probability simplex (∑aᵢ = 1), because
the objective 2·n·max(conv) is scale‑invariant.  It consists of three
stages per length:

1. **Adam on a soft‑max surrogate** – a smooth approximation of the
   ℓ∞ norm (log‑sum‑exp) is minimised.  The temperature λ is driven
   down aggressively so the surrogate becomes a true max.

2. **Targeted simulated‑annealing** – cheap mass‑transfer moves
   (ε from i to j) are accepted with a fast geometric temperature
   schedule.  This quickly escapes tiny plateaus left by Adam.

3. **Sub‑gradient descent on the exact max** – after annealing the
   exact active lag(s) are identified; a sub‑gradient (2·shifted‑a)
   is followed with a diminishing step‑size and projection onto the
   simplex.

The search proceeds over a geometric ladder of lengths,
re‑using the best distribution found so far by linear interpolation.
More time is spent on the larger problems (where the extra degrees of
freedom give the biggest reductions).  Random restarts and a few
hand‑crafted initial shapes (sinusoidal, triangular, exponential)
help to avoid local minima.

The function ``propose_candidate`` respects the total time budget
(`budget_s`, default 1000 s) and returns the best sequence discovered.
"""

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (fast for all n)."""
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << ((size - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:size]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a*a   (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad : dense gradient vector (shape n,)
    """
    n = a.shape[0]
    max_val = conv.max()
    # numerically stable soft‑max weights
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()

    a_rev = a[::-1]
    # convolution of weights with the reversed a
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """Create a few plausible starting vectors from a base distribution."""
    n = base_seq.shape[0]
    variants = []

    # 0. warm start itself
    variants.append(base_seq.copy())

    # 1. sinusoidal (all positive)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 2. triangular (ramp up then down)
    tri = np.concatenate([np.linspace(1, n // 2, n // 2, endpoint=False),
                          np.linspace(n // 2, 1, n - n // 2)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 3. exponential decay (highly asymmetric)
    exp_vals = np.exp(-np.linspace(0, 5, n))
    exp_vals /= exp_vals.sum()
    variants.append(exp_vals)

    # 4. random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants

    v = np.zeros_like(a)

    lam_factor = 0.4          # initial λ ≈ lam_factor·max(conv)
    lam_decay = 0.7
    lam_decay_interval = 1500

    it = 0

        it += 1

        conv = _conv_self(a)
        max_b = conv.max()

            lam_factor = max(lam_factor * lam_decay, 1e-12)
        lam = max(max_b * lam_factor, 1e-14)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Diminishing base step – a bit larger than the previous version
        base_step = 0.9 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 200000) -> tuple[np.ndarray, float]:
    """Cheap mass‑transfer annealer (ε‑move from i→j)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the *exact* max‑convolution value.
    Uses the shifted‑a gradient (2·a_{L−k}) at the active lag(s).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    # initial step size (as a fraction of total mass moved each iteration)
    step0 = 0.15

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")

        # indices achieving the current max (within tiny tolerance)
        idxs = np.where(conv >= max_b - 1e-12)[0]
        inv = 1.0 / len(idxs)

        # compute the average sub‑gradient for these active lags
        grad = np.zeros_like(a)
        rng = np.arange(n)
        for L in idxs:
            shift = L - rng
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]] * inv

        # diminishing step size
        step = step0 / (1.0 + 1e-3 * it)

        a = a - step * grad
        a = _project_simplex(a)

    final_score = 2.0 * n * _conv_self(best_a).max()
    return best_a, final_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The function respects the time budget and returns a Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously found construction
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – larger lengths receive more iterations.
    lengths = [256, 1024, 4096, 16384, 65536]
    # If time permits we will even try 131072 after the loop.
    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm‑start / several variants)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # --------------------------------------------------------------
        # Allocate a time slice for this length (roughly 20 % of remaining)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        slice_deadline = time.time() + max(10.0, remaining * 0.20)

        # ------------------------------------------------------------------
        # Adam optimisation phase (multiple starts)
        # ------------------------------------------------------------------
        best_local_a = None
        best_local_score = float('inf')
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            # Occasionally enforce symmetry (∼30 % of runs)
            sym = (np.random.rand() < 0.3)
            a_opt, score_opt = _run_adam_opt(start_vec, n, slice_deadline,
                                             start_time, symmetrize=sym)

            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after Adam n={n} score={best_local_score:.6f}")

        # ------------------------------------------------------------------
        # Annealing refinement (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.25)
            a_ref, score_ref = _run_annealing(
                best_local_a,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=200000,
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient descent on the exact max (final polishing)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            subgrad_deadline = time.time() + max(3.0, (deadline - time.time()) * 0.25)
            a_sub, score_sub = _run_subgradient(best_seq, n, subgrad_deadline, start_time)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after subgrad n={n} score={score_sub:.6f}")

        # Remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Optional extra large length if still time left
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # try one more jump to 131072 (if memory permits)
        n = 131072
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
        slice_deadline = deadline  # use whatever time remains

        # a single quick Adam run followed by annealing/sub‑grad
        a_opt, score_opt = _run_adam_opt(a0, n, slice_deadline, start_time,
                                         symmetrize=False)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve at n={n} after final Adam score={score_opt:.6f}")

        # brief anneal
        if time.time() < deadline:
            anneal_deadline = deadline
            a_ref, score_ref = _run_annealing(
                best_seq, n, anneal_deadline, start_time,
                best_score, best_seq,
                start_temp=3e-2, end_temp=5e-6, steps=100000
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after final anneal score={score_ref:.6f}")

        # brief sub‑gradient polish
        if time.time() < deadline:
            subgrad_deadline = deadline
            a_sub, score_sub = _run_subgradient(best_seq, n, subgrad_deadline, start_time)
            if score_sub < best_score:
                best_score = score_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:6.2f}s] FINAL GLOBAL improve after subgrad score={score_sub:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list – the evaluator will clamp / normalise as needed
    return [float(x) for x in best_seq]
