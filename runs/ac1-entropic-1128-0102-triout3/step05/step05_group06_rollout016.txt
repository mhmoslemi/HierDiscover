"""Propose a candidate step-function that attempts to beat the arcsine benchmark.
Algorithm:
0. Simple analytic families (Beta, uniform) and random mixtures are sampled.
1. Each candidate is evaluated using fast FFT convolution and the evaluation metric.
2. The best candidate is refined by a short projected gradient descent.
3. If time permits, a cheap simulated‑annealing walk is performed.
4. The resulting sequence (normalised to sum‑1) is returned.
"""
import time, math, numpy as np

def _conv_self_fft(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n-1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af*Af, n=fft_len)[:2*n-1]
    return conv

def _score_from_seq(a):
    # sum normalized to 1 (otherwise scaling irrelevant)
    s = a.sum()
    if s == 0:
        return float('inf')
    a = a / s
    conv = _conv_self_fft(a)
    cur_max = conv.max()
    n = a.shape[0]
    # Evaluate metric (same as evaluate_sequence, but without clipping)
    return 2.0 * n * cur_max / (a.sum()**2)  # a.sum()==1 so just 2*n*cur_max

def _beta_shape(alpha, n):
    # symmetric Beta(alpha,alpha) density (unnormalised) on (0,1)
    xs = (np.arange(n) + 0.5) / n
    # avoid numerical overflow for alpha < 1: use log
    # compute log_vals = (alpha-1)*log(xs) + (alpha-1)*log(1-xs)
    log_vals = (alpha-1) * (np.log(xs) + np.log(1-xs))
    vals = np.exp(log_vals)
    # handle possible infinities if xs near 0 or 1 (should not happen)
    vals[np.isnan(vals)] = 0.0
    # normalise to sum 1
    total = vals.sum()
    if total == 0:
        # fallback uniform
        return np.full(n, 1.0/n)
    return vals / total

def _project_simplex(v):
    # Project onto probability simplex (non‑negative, sum 1)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _gradient_descent(a0, n, deadline, start_time, step0=0.5, beta=0.9, max_iter=200000):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_from_seq(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # subgradient from active set (tolerance)
        max_val = cur_max
        rel = 1e-12
        active = np.where(conv >= max_val * (1 - rel))[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n-1, k)
            i = np.arange(lo, hi+1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]
        if active.size > 0:
            g /= float(active.size)
        # momentum
        momentum = beta * momentum + (1 - beta) * g
        step = step0 / math.sqrt(it+1)
        a = a - step * momentum
        a = _project_simplex(a)
    return best_a, best_score

def _annealing_refine(a0, n, deadline, start_time, init_score=None):
    a = a0.copy()
    if init_score is None:
        score = _score_from_seq(a)
    else:
        score = init_score
    best_a = a.copy()
    best_score = score
    it = 0
    steps = 200000
    start_temp = 5e-2
    end_temp = 1e-5
    while time.time() < deadline and it < steps:
        it += 1
        T = start_temp * (end_temp / start_temp) ** (it / steps)
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0:
            continue
        eps = a[i] * np.random.rand() * 0.2    # move up to 20% of source
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        new_score = _score_from_seq(a_prop)
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    `evaluate_sequence`.  The routine blends a cheap random‑mixture search
    over Beta‑type families with a short gradient‑descent and annealing
    refinement.  It runs under the supplied time budget and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # 1. Base parameters
    # --------------------------------------------------------------
    base_n = 4000                     # size used during the random search
    best_seq = None
    best_score = float('inf')

    # Helper to evaluate and possibly update the best sequence
    def maybe_update(seq, score, descr=""):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time()-start_time:6.2f}s] new best {score:.6f} {descr}")

    # --------------------------------------------------------------
    # 2. Evaluate a few deterministic baselines
    # --------------------------------------------------------------
    uniform = np.full(base_n, 1.0/base_n)
    maybe_update(uniform, _score_from_seq(uniform), "uniform")

    arcsine = _beta_shape(0.5, base_n)
    maybe_update(arcsine, _score_from_seq(arcsine), "arcsine")

    # --------------------------------------------------------------
    # 3. Random mixture search
    # --------------------------------------------------------------
    # We sample three families:
    #   a) beta(alpha) alone,
    #   b) mixture beta(alpha) + uniform,
    #   c) mixture of two betas (alpha1,alpha2) + uniform.
    # The search stops when the time budget for this phase is exhausted.
    mix_phase_deadline = start_time + 0.45 * (deadline - start_time)  # first half of time
    while time.time() < mix_phase_deadline:
        # Decide which family to sample
        r = np.random.rand()
        if r < 0.4:
            # single beta
            alpha = np.random.uniform(0.15, 0.85)
            seq = _beta_shape(alpha, base_n)
        elif r < 0.8:
            # beta + uniform
            alpha = np.random.uniform(0.15, 0.85)
            w = np.random.rand()
            seq = w * _beta_shape(alpha, base_n) + (1-w) * uniform
        else:
            # two betas + uniform
            alpha1 = np.random.uniform(0.15, 0.85)
            alpha2 = np.random.uniform(0.15, 0.85)
            w1 = np.random.rand()
            w2 = np.random.rand() * (1-w1)
            seq = w1 * _beta_shape(alpha1, base_n) + w2 * _beta_shape(alpha2, base_n) + (1-w1-w2) * uniform
        # Normalise (numerical drift)
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0:
            continue
        seq = seq / s
        sc = _score_from_seq(seq)
        maybe_update(seq, sc, f"rand α={alpha:.3f}" if r<0.8 else "rand mix")

    # --------------------------------------------------------------
    # 4. Fine‑grained optimisation of the current best (if any)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_opt = best_seq.shape[0]
        # allocate a modest budget for projected gradient descent
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        seq_gd, score_gd = _gradient_descent(best_seq, n_opt, gd_deadline, start_time,
                                             step0=0.6, beta=0.85, max_iter=500000)
        maybe_update(seq_gd, score_gd, "gradient")
        # continue with annealing if time permits
        if time.time() < deadline:
            ann_deadline = deadline
            seq_ann, score_ann = _annealing_refine(seq_gd, n_opt, ann_deadline, start_time, score_gd)
            maybe_update(seq_ann, score_ann, "annealing")

    # --------------------------------------------------------------
    # 5. Upsample to a larger size (optional) – the evaluation is
    #    scale‑invariant, but a finer grid can reduce discretisation error.
    # --------------------------------------------------------------
    final_n = 9600
    if best_seq is not None and best_seq.shape[0] != final_n:
        # linear interpolation preserving total mass
        old_n = best_seq.shape[0]
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, final_n)
        up = np.interp(new_x, old_x, best_seq)
        up = np.maximum(up, 0.0)
        s = up.sum()
        if s > 0:
            up = up / s
        else:
            up = np.full(final_n, 1.0/final_n)
        best_seq = up

    # --------------------------------------------------------------
    # 6. Final safety normalisation and conversion to list
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s == 0:
        best_seq = np.full_like(best_seq, 1.0/best_seq.size)
    else:
        best_seq = best_seq / s

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  

Algorithm
---------

1. **Baseline families** – generate a uniform sequence and a
   discretised arcsine (Beta(½,½)) distribution.
2. **Random mixture search** – randomly combine Beta(α,α) densities
   (α∈(0.15,0.85)) with a uniform component.  Three families are tried:
   a single Beta, a Beta+uniform mix, and a mixture of two Betas+uniform.
   Each candidate is evaluated with a fast FFT‑based convolution and the
   best one is kept.
3. **Local refinement** – the current best sequence is improved by a
   short projected gradient descent on the simplex (sub‑gradient of the
   active constraints) and a cheap simulated‑annealing walk.
4. **Upsampling** – the final sequence is linearly interpolated to a
   larger size (9600) to reduce discretisation error.
5. **Safety fallback** – if nothing was found a uniform distribution is
   used.

The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution a * a using FFT (returns length 2·n‑1 array)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())        # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Score of a sequence (exactly the same as evaluate_sequence, but
# without the extra clipping that never matters for our candidates)
# ----------------------------------------------------------------------
def _score_from_seq(a: np.ndarray) -> float:
    s = a.sum()
    if s == 0.0:
        return float("inf")
    a = a / s                    # normalise – the objective is scale‑invariant
    conv = _conv_self_fft(a)
    cur_max = conv.max()
    n = a.shape[0]
    return 2.0 * n * cur_max     # denominator = 1 after normalisation


# ----------------------------------------------------------------------
# Symmetric Beta(α,α) density on (0,1), discretised on n points and
# normalised to sum = 1.  Handles α<1 (singular at the ends) safely.
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, n: int) -> np.ndarray:
    xs = (np.arange(n) + 0.5) / n                    # mid‑point rule
    # work in log‑space to avoid overflow for α≪1
    log_vals = (alpha - 1.0) * (np.log(xs) + np.log(1.0 - xs))
    # clip exponent to avoid infinities from exp()
    log_vals = np.clip(log_vals, -700.0, 700.0)
    vals = np.exp(log_vals)
    # replace any non‑finite entries (should be none after clipping)
    vals[~np.isfinite(vals)] = 0.0
    tot = vals.sum()
    if tot == 0.0:                                 # fallback – uniform
        return np.full(n, 1.0 / n)
    return vals / tot


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x_i≥0, Σx_i=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex, using the sub‑gradient of
# the active set of the max‑convolution.  Returns the best point seen.
# ----------------------------------------------------------------------
def _gradient_descent(a0: np.ndarray, n: int, deadline: float,
                     start_time: float,
                     step0: float = 0.6, beta: float = 0.85,
                     max_iter: int = 500_000):
    a = a0.copy()
    momentum = np.zeros_like(a)
    best_a = a.copy()
    best_score = _score_from_seq(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # --- sub‑gradient of the max‑convolution (active set) ---
        max_val = cur_max
        rel = 1e-12
        active = np.where(conv >= max_val * (1.0 - rel))[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]
        if active.size:
            g /= float(active.size)

        # --- Nesterov‑style momentum & diminishing step ----------
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _annealing_refine(a0: np.ndarray, n: int, deadline: float,
                      start_time: float, init_score: float = None):
    a = a0.copy()
    if init_score is None:
        score = _score_from_seq(a)
    else:
        score = init_score
    best_a = a.copy()
    best_score = score

    steps = 200_000
    start_temp = 5e-2
    end_temp = 1e-5

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # temperature schedule (geometric)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.20            # move up to 20 % of source
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score_from_seq(a_prop)
        delta = new_score - score
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine mixes a cheap random‑mixture search
    over Beta‑type families with a short projected gradient descent and a
    simulated‑annealing refinement.  It respects the given time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 1 s safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # 1. Basic parameters
    # --------------------------------------------------------------
    base_n = 4000                     # size used during the random search
    best_seq = None
    best_score = float("inf")

    # helper to update the incumbent
    def maybe_update(seq, score, label=""):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] new best {score:.6f} {label}")

    # --------------------------------------------------------------
    # 2. Deterministic baselines
    # --------------------------------------------------------------
    uniform = np.full(base_n, 1.0 / base_n)
    maybe_update(uniform, _score_from_seq(uniform), "uniform")

    arcsine = _beta_shape(0.5, base_n)
    maybe_update(arcsine, _score_from_seq(arcsine), "arcsine")

    # --------------------------------------------------------------
    # 3. Random mixture search (first ~45 % of the budget)
    # --------------------------------------------------------------
    mix_deadline = start_time + 0.45 * (deadline - start_time)
    while time.time() < mix_deadline:
        r = np.random.rand()
        if r < 0.40:                       # single Beta(α)
            alpha = np.random.uniform(0.15, 0.85)
            seq = _beta_shape(alpha, base_n)
            label = f"β(α={alpha:.3f})"
        elif r < 0.80:                     # mixture Beta(α) + uniform
            alpha = np.random.uniform(0.15, 0.85)
            w = np.random.rand()
            seq = w * _beta_shape(alpha, base_n) + (1.0 - w) * uniform
            label = f"mix β(α={alpha:.3f}) w={w:.3f}"
        else:                               # mixture of two Betas + uniform
            alpha1 = np.random.uniform(0.15, 0.85)
            alpha2 = np.random.uniform(0.15, 0.85)
            w1 = np.random.rand()
            w2 = np.random.rand() * (1.0 - w1)
            seq = (w1 * _beta_shape(alpha1, base_n) +
                   w2 * _beta_shape(alpha2, base_n) +
                   (1.0 - w1 - w2) * uniform)
            label = f"mix2 α1={alpha1:.3f} α2={alpha2:.3f}"
        # normalise (tiny drift may appear)
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0.0:
            continue
        seq = seq / s
        sc = _score_from_seq(seq)
        maybe_update(seq, sc, label)

    # --------------------------------------------------------------
    # 4. Local fine‑grained optimisation of the incumbent
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_opt = best_seq.shape[0]
        # allocate ~30 % of the remaining time to gradient descent
        gd_deadline = time.time() + 0.30 * (deadline - time.time())
        seq_gd, score_gd = _gradient_descent(best_seq, n_opt,
                                             gd_deadline, start_time,
                                             step0=0.6, beta=0.85,
                                             max_iter=500_000)
        maybe_update(seq_gd, score_gd, "gradient")

        # remaining time → simulated annealing
        if time.time() < deadline:
            seq_ann, score_ann = _annealing_refine(seq_gd, n_opt,
                                                   deadline, start_time,
                                                   init_score=score_gd)
            maybe_update(seq_ann, score_ann, "annealing")

    # --------------------------------------------------------------
    # 5. Upsample to a finer grid (optional – evaluation is scale‑invariant)
    # --------------------------------------------------------------
    final_n = 9600
    if best_seq is not None and best_seq.shape[0] != final_n:
        old_n = best_seq.shape[0]
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, final_n)
        up = np.interp(new_x, old_x, best_seq)
        up = np.maximum(up, 0.0)
        s = up.sum()
        if s > 0.0:
            up = up / s
        else:
            up = np.full(final_n, 1.0 / final_n)
        best_seq = up

    # --------------------------------------------------------------
    # 6. Safety fallback & final normalisation
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
