```python
def gradient_from_weights(a, w):
    n = a.shape[0]
    # a_rev length n
    a_rev = a[::-1]
    # length of convolution for gradient: len(w) + len(a_rev) - 1 = (2n-1) + n -1 = 3n-2 
    fft_len = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(W * A_rev, n=fft_len)
    # conv indices from 0 to ... ; we need part corresponding to length n
    start = len(a_rev) - 1
    g = 2.0 * conv[start:start + n]
    return g
```

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The search builds on the two‑phase scheme that was previously used:
    1. Adam optimisation on a smooth‑max surrogate of the max‑convolution.
    2. Mirror‑descent / exponential‑gradient optimisation that uses the
       exact max‑convolution (hard‑max sub‑gradient).

Improvements over the previous version
--------------------------------------
* A multi‑stage size schedule (800 → 1800 → 3600) that spends the bulk of
  the budget on the larger size – larger sequences have more freedom to
  flatten the auto‑correlation.
* More aggressive learning‑rate schedules (larger initial η for EG, larger
  α for Adam) and a slightly higher proportion of iterations devoted to
  the hard‑max phase.
* A final long exponential‑gradient refinement on the best (possibly
  up‑sampled) candidate that uses the remaining time budget.
* Small utilities (projection onto the simplex, fast score evaluation)
  avoid repeated calls to ``evaluate_sequence`` during the search.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -------------------------------------------------------
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Uniform vector is the projection in the degenerate case
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise onto the simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    if seq.shape[0] == new_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, seq.shape[0])
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, seq)


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute the full linear convolution a*a and return its maximum value.
    FFT length is chosen as the next power of two for speed.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of sum_k w_k * (a*a)_k with respect to a.
    Implements g = 2 * (a_rev * w) via FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # Length of the linear convolution of a_rev (len=n) and w (len=2n-1)
    conv_len = len(w) + n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(W * A_rev, n=fft_len)
    start = n - 1
    g = 2.0 * full[start:start + n]
    return g


def _fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (2·n·max(conv))/sum², assuming non‑zero sum."""
    n = a.shape[0]
    max_b, _ = _conv_max(a)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

# Optimisers ------------------------------------------------------------
# ----------------------------------------------------------------------


def _adam_optimize(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    alpha: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_start: float = 0.07,
    temp_end: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best iterate seen and its (fast) score.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _fast_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- evaluate current point -------------------------------------------------
        cur_score = _fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # --- smooth‑max weights ----------------------------------------------------
        max_val, conv = _conv_max(a)
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # --- gradient ---------------------------------------------------------------
        g = _gradient_from_weights(a, w)

        # --- Adam update ------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # --- projection back to simplex --------------------------------------------
        a = _project_simplex(a)

    return best_a, best_score


def _exp_gradient_optimize(
    a: np.ndarray,
    max_iters: int,
    deadline: float,
    eta_start: float = 0.9,
    eta_end: float = 0.001,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser using the exact max‑convolution.
    Returns the best iterate seen and its (fast) score.
    """
    best_a = a.copy()
    best_score = _fast_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)

        # mask of (approximately) active anti‑diagonals
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient for the hard‑max norm
        g = _gradient_from_weights(a, w)

        # decreasing geometric step‑size
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # avoid overflow/underflow in exp
        a = a * np.exp(upd)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / s

        cur_score = _fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score
