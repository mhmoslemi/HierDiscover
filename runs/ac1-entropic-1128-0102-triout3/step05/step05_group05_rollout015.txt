def solve_g(f, M):
    n = len(f)
    # Objective maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)

    # Build inequality matrix A_ub x <= b_ub
    # For each s in 0 .. 2n-2, we have sum_i f_i g_{s-i} <= M
    # Represent variables g_j.
    # Build A_ub as (2n-1, n) matrix.
    A = np.zeros((2*n-1, n))
    for s in range(2*n-1):
        # indices k where 0 <= s-k < n -> k ∈ [max(0, s-(n-1)), min(n-1, s)]
        start = max(0, s-(n-1))
        end = min(n-1, s)
        # For k in start..end:
        # coefficient = f[s - k]
        ks = np.arange(start, end+1)
        A[s, ks] = f[s - ks]
    # b = M * np.ones(2*n-1)
    b = np.full(2*n-1, M, dtype=float)

    # bounds for variables: (0, None)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
  
    if res.success:
        g = res.x
    else:
        # fallback: uniform
        g = np.full(n, 1.0/n)

    return g

def solve_lp_g(f, M):
    n = len(f)
    # Build coefficient matrix A
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for s in range(2*n-1):
        start = max(0, s-(n-1))
        end = min(n-1, s)
        ks = np.arange(start, end+1, dtype=int)
        A[s, ks] = f[s - ks]
    # Using scipy
    from scipy.optimize import linprog
    c = -np.ones(n, dtype=np.float64)  # maximize sum = minimize -sum
    b = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return res.x

def mix_sequences(f, g_scaled, t):

    # Ensure non-negative
    new_f = np.maximum(new_f, 0.0)

        # fallback uniform
        new_f = np.full_like(f, 1.0 / len(f))

            print(f"[{time.time() - start_time:.2f}s] new best {best_score:.6f}")

    # load prev if any
    try:

        if prev.ndim == 1 and prev.size > 0:

            if prev.sum() >= 0.01:
                prev = prev / prev.sum()
                evaluate_and_update(prev)
    except Exception:
        pass

    uniform = np.full(n, 1.0/n, dtype=float)
    evaluate_and_update(uniform)

    for _ in range(5):

        cand /= cand.sum()
        evaluate_and_update(cand)

    # Set current_seq to best_seq
    current_seq = best_seq.copy()
    current_score = best_score

    # hill climbing loop
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # compute convolution
        b = np.convolve(current_seq, current_seq)
        smax = int(np.argmax(b))
        # contributions
        idx = np.arange(n)
        partner = smax - idx
        mask = (partner >= 0) & (partner < n)
        contributions = np.zeros(n, dtype=float)
        contributions[mask] = current_seq[mask] * current_seq[partner[mask]]

        # pick i where contribution is max
        i = int(np.argmax(contributions))
        # pick random j different from i
        j = i
        while j == i:
            j = random.randint(0, n-1)

        # epsilon as small fraction of a[i] or fixed
        epsilon = min(current_seq[i] * 0.5, 0.01)  # cannot exceed 0.5*a_i to keep nonnegative
        if epsilon <= 1e-12:
            continue

        new_seq = current_seq.copy()
        new_seq[i] -= epsilon
        new_seq[j] += epsilon
        # enforce non-negativity after floating noise
        new_seq = np.clip(new_seq, 0.0, None)
        # normalize sum to 1
        total = new_seq.sum()
        if total < 1e-12:
            continue
        new_seq /= total

        new_score = evaluate_sequence(new_seq.tolist())
        # accept if better
        if new_score < current_score:
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_seq = new_seq
                best_score = new_score
                print(f"[{time.time() - start_time:.2f}s] iteration {iteration} improved best to {best_score:.6f}")
        else:
            # occasionally accept worse move to escape local minima (simulated annealing)
            # compute temperature schedule
            T = max(0.01, (budget_s - elapsed) / budget_s)  # decays to small
            # acceptance probability
            if np.exp((current_score - new_score) / T) > random.random():
                current_seq = new_seq
                current_score = new_score

        # Occasionally (every 1000 iter) random perturbation
        if iteration % 500 == 0:
            # random shuffle some mass: pick two indices and exchange random fraction
            i1, i2 = random.sample(range(n), 2)
            delta = min(current_seq[i1], 0.02) * random.random()
            new_seq2 = current_seq.copy()
            new_seq2[i1] -= delta
            new_seq2[i2] += delta
            new_seq2 = np.clip(new_seq2, 0.0, None)
            new_seq2 /= new_seq2.sum()
            sc2 = evaluate_sequence(new_seq2.tolist())
            if sc2 < best_score:
                best_seq = new_seq2.copy()
                best_score = sc2
                current_seq = new_seq2.copy()
                current_score = sc2
                print(f"[{time.time() - start_time:.2f}s] random perturb improved best to {best_score:.6f}")

    # final return best_seq
    if best_seq is None:
        best_seq = uniform
        best_score = evaluate_sequence(best_seq.tolist())

    result = [float(x) for x in best_seq.tolist()]
    print(f"[{time.time() - start_time:.2f}s] returning sequence length {len(result)} best score {best_score:.6f}")
    return result

"""
Hill‑climbing search for a step‑function that minimises

    Φ(a) = 2·n·max(a*a) / (sum(a)²)

The evaluation is scale‑invariant, so we keep the sequence normalised to
sum = 1.  Starting from the best known construction (``height_sequence_1``,
or a uniform vector if none is available) the algorithm repeatedly:

1.  Computes the full convolution a*a and finds the index ``s_max`` of the
    largest coefficient.
2.  Identifies the pair of positions that contributes most to that peak
    (the product a[i]·a[s_max‑i]).
3.  Moves a small amount of mass from the “heavy’’ index to a random
    ``light’’ index, re‑normalises, and accepts the move only if the
    evaluation score improves (a simple greedy step; occasional stochastic
    acceptance is added to escape shallow local minima).
4.  Occasionally performs a tiny random redistribution of mass to keep the
    search diverse.

The loop runs until the allocated time budget (≈ 95 % of the allowed 1000 s)
is exhausted, always keeping the best sequence found.  No external files or
network access are used; the only external call is ``evaluate_sequence``.
"""

import time
import random
import numpy as np


def _initial_candidates(n: int, seed: int = 0):
    """Generate a few simple start vectors (uniform and random Dirichlet)."""
    np.random.seed(seed)
    random.seed(seed)

    candidates = []

    # uniform vector (sum = 1)
    uniform = np.full(n, 1.0 / n, dtype=float)
    candidates.append(uniform)

    # a handful of random Dirichlet samples
    for _ in range(5):
        vec = np.random.rand(n).astype(float)
        if vec.sum() == 0.0:
            continue
        vec /= vec.sum()
        candidates.append(vec)

    return candidates


def _evaluate(seq: np.ndarray):
    """Thin wrapper around the provided evaluate_sequence."""
    # the harness guarantees evaluate_sequence is available globally
    return evaluate_sequence(seq.tolist())


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) improves
    on the current best score.  The routine respects the time budget and
    always returns a valid sequence (fallback to a uniform vector).
    """
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Choose a problem size.
    # ------------------------------------------------------------------
    # A moderate length keeps convolutions cheap while still allowing
    # enough degrees of freedom for improvement.
    n = 1800

    # ------------------------------------------------------------------
    # 2.  Load a previously known good sequence, if present.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    try:
        prev = np.asarray(height_sequence_1, dtype=float)  # global from harness
        if prev.ndim == 1 and prev.size > 0:
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() >= 0.01:
                prev = prev / prev.sum()          # normalise for fair comparison
                sc = _evaluate(prev)
                if sc < best_score:
                    best_seq = prev.copy()
                    best_score = sc
    except Exception:
        # No usable previous sequence – ignore.
        pass

    # ------------------------------------------------------------------
    # 3.  Add a few generic candidates (uniform + random Dirichlet).
    # ------------------------------------------------------------------
    for cand in _initial_candidates(n, seed):
        # If we already have a better sequence, skip the cheap uniform case.
        if best_seq is not None and best_score < 1.6:
            break
        sc = _evaluate(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc

    # If for any reason we still have nothing, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = _evaluate(best_seq)

    # ------------------------------------------------------------------
    # 4.  Hill‑climbing loop.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score

    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:          # leave a tiny safety margin
            break
        iteration += 1

        # Full convolution and location of the current worst coefficient.
        conv = np.convolve(current, current)
        s_max = int(np.argmax(conv))

        # Contribution of each index to that peak.
        idx = np.arange(n)
        partner = s_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = current[mask] * current[partner[mask]]

        # Heaviest contributor – we will steal a bit of its mass.
        i_heavy = int(np.argmax(contrib))
        if current[i_heavy] <= 1e-12:
            # Nothing to move; break out.
            break

        # Choose a random destination index that is currently light.
        i_light = i_heavy
        while i_light == i_heavy:
            i_light = random.randint(0, n - 1)

        # Transfer a modest amount of probability mass.
        eps = min(current[i_heavy] * 0.3, 0.02)   # keep non‑negativity robust
        new_seq = current.copy()
        new_seq[i_heavy] -= eps
        new_seq[i_light] += eps

        # Numerical safety – clip tiny negatives that could appear.
        new_seq = np.clip(new_seq, 0.0, None)
        total = new_seq.sum()
        if total == 0.0:
            continue
        new_seq /= total                           # renormalise to sum = 1

        new_score = _evaluate(new_seq)

        # Greedy acceptance (with a tiny simulated‑annealing fallback).
        if new_score < current_score:
            current = new_seq
            current_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(
                    f"[{elapsed:.2f}s] iter {iteration:5d}  new best ≈ {best_score:.6f}"
                )
        else:
            # Occasionally allow a worse move to escape plateaus.
            # Temperature falls linearly with remaining time.
            T = max(0.001, (budget_s - elapsed) / budget_s)
            if np.exp((current_score - new_score) / T) > random.random():
                current = new_seq
                current_score = new_score

        # Every few hundred steps perform a tiny random reshuffle
        # (helps to discover new directions).
        if iteration % 400 == 0:
            i1, i2 = random.sample(range(n), 2)
            delta = min(current[i1], 0.015) * random.random()
            reshuffle = current.copy()
            reshuffle[i1] -= delta
            reshuffle[i2] += delta
            reshuffle = np.clip(reshuffle, 0.0, None)
            reshuffle /= reshuffle.sum()
            rs_score = _evaluate(reshuffle)
            if rs_score < best_score:
                best_seq = reshuffle.copy()
                best_score = rs_score
                current = reshuffle.copy()
                current_score = rs_score
                print(
                    f"[{elapsed:.2f}s] iter {iteration:5d}  random boost → {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    result = [float(x) for x in best_seq.tolist()]
    total_time = time.time() - start
    print(
        f"[{total_time:.2f}s] returning sequence length {len(result)} "
        f"with best score {best_score:.6f}"
    )
    return result
