"""
Optimization of step‑function heights by minimizing the maximum of the self‑convolution.
The algorithm uses a projected exponential‑gradient descent on the L2‑flattening
objective `∑_k ( (a*a)_k - 1/(2n-1) )²`.  The update is multiplicative
(`a_i ← a_i·exp(−η·g_i)`) and the vector is projected onto the probability simplex.
A multi‑scale schedule gradually enlarges the problem size, while many random
restarts and a long final refinement are performed within the 1000 s budget.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of Σ_k w_k·conv_k with respect to a (a ≥ 0)."""
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]

def _flatten_optimize(a_init: np.ndarray,
                      max_iters: int,
                      deadline: float,
                      eta_start: float = 0.6,
                      eta_end: float = 0.001,
                      verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected exponential‑gradient optimisation that drives the self‑convolution
    of ``a`` towards a flat profile (its theoretical average 1/(2n‑1)).
    Returns the best vector seen and its corresponding score.
    """
    a = a_init.copy()
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())
    # constant target value for every entry of the convolution
    target = 1.0 / (2 * n - 1)

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full convolution
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]

        # current score (exact the same as evaluate_sequence up to clipping)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if verbose:
                print(f"       iter {it:6d} – new local best score = {best_score:.6f}")

        # weight = deviation from the flat target (signed)
        w = conv - target

        # gradient of Σ w_k·conv_k w.r.t. a
        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric step‑size schedule
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update, safe clipping before exponentiation
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # exp(±30) ≈ 1e13, safe
        a = a * np.exp(upd)

        # projection back onto the simplex (non‑negative, sum = 1)
        a = _project_simplex(a)

    # final evaluation of the best vector found
    final_score = 2.0 * n * _conv_full(best_a).max() / (best_a.sum() ** 2)
    return best_a, final_score

def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Make a non‑negative vector a probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s

"""
Optimization of step‑function heights by flattening the self‑convolution.
The method works on any length *n* and repeatedly applies a projected
exponential‑gradient descent that minimises the L2‑distance between the
convolution a∗a and its theoretical average 1/(2n‑1).  The update is
multiplicative (a_i ← a_i·exp(−η·g_i)) and after each step the vector is
projected onto the probability simplex (non‑negative, sum = 1).  A
multi‑scale schedule (starting from the size of the previously known best
construction and growing up to ~12 000) is used; for each size a small pool
of diverse initial vectors is generated and optimised.  After the main
search a long final refinement is performed on the current global best.
All work respects the 1000 s time budget; intermediate scores are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector normalised to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k  w.r.t. `a`.
    Implements g = 2·(a_rev * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _flatten_optimize(a_init: np.ndarray,
                      max_iters: int,
                      deadline: float,
                      eta_start: float = 0.6,
                      eta_end: float = 0.001,
                      verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Projected exponential‑gradient optimisation that pushes the convolution
    a∗a towards a flat profile (its average 1/(2n‑1)).
    Returns the best vector encountered and its exact score.
    """
    a = a_init.copy()
    n = a.shape[0]

    # FFT lengths (next powers of two)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    # Target value for every entry of the convolution
    target = 1.0 / (2 * n - 1)

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution a∗a (full) -----
        Af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]

        # ----- current objective value (identical to evaluate_sequence) -----
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if verbose:
                print(f"       iter {it:6d} – new local best score = {best_score:.6f}")

        # ----- weight = signed deviation from the flat target -----
        w = conv - target

        # ----- gradient of Σ w_k·conv_k -----
        g = _gradient_from_weights(a, w, fft_len_grad)

        # ----- geometric step‑size schedule -----
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # ----- multiplicative update (mirror‑descent) -----
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)   # keep exp(·) in a safe range
        a = a * np.exp(upd)

        # ----- project back onto the simplex -----
        a = _project_simplex(a)

    # final exact score for the best vector we have kept
    final_score = 2.0 * n * np.convolve(best_a, best_a).max() / (best_a.sum() ** 2)
    return best_a, final_score
