import numpy as np, cvxpy as cp, time

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    n = 2000  # adjustable
    f = np.full(n, 1.0 / n)
    # maybe start from height_sequence_1 if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # Clip and normalize to sum=1
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() >= 0.01:
                prev = prev / prev.sum()
                prev_phi = evaluate_sequence(prev.tolist())
                uniform_phi = evaluate_sequence(f.tolist())
                if prev_phi < uniform_phi:
                    f = prev
    except Exception:
        pass

    # compute current max
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best_phi = evaluate_sequence(f.tolist())
    best_f = f.copy()
    iteration = 0
    while time.time() - start < budget_s - 0.5: # keep margin
        # build matrix A
        # Since constraints many, but we can build sparse matrix
        # Use scipy.sparse for efficiency
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        data = []
        for k in range(2*n-1):
            # j runs from max(0, k-(n-1)) to min(n-1, k)
            j_start = max(0, k - (n-1))
            j_end = min(n-1, k)
            # For each j, i = k - j
            for j in range(j_start, j_end+1):
                i = k - j
                rows.append(k)
                cols.append(j)
                data.append(f[i])
        A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # solve LP using scipy.optimize.linprog
        from scipy.optimize import linprog
        c = -np.ones(n)  # maximize sum(g) => minimize -sum
        bounds = [(0, None)] * n
        # constraints: A @ g <= M
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': budget_s - (time.time()-start)})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            break
        # Combine
        f_new = (f + g) / (1 + sum_g)  # new sum=1
        # Evaluate new max
        conv_fnew = np.convolve(f_new, f_new)
        M_new = conv_fnew.max()
        phi_new = evaluate_sequence(f_new.tolist())
        if phi_new < best_phi:
            best_phi = phi_new
            best_f = f_new.copy()
            print(f"Iter {iteration}: phi={phi_new:.6f}, M={M_new:.6e}")
        # update for next iteration: always use f_new (even if phi not improved) to try reduce bound
        f = f_new
        M = M_new
        iteration += 1
        # maybe break if no improvement for many iterations
    print(f"Done after {iteration} iterations, best phi={best_phi:.6f}")
    return best_f.tolist()

    M = b.max()
    best_a = a.copy()
    best_phi = evaluate_sequence(best_a.tolist())
    iteration = 0

    # iteration loop
    while time.time() - start < budget_s * 0.95:
        # Build convolution matrix A for constraint conv(a,g) <= M
        # Using sparse matrix
        # Build data for each offset k
        rows = []
        cols = []
        data = []
        # For each k, compute range of j where 0<=j<n and i=k-j in [0,n-1]
        for k in range(2*n-1):
            j_start = max(0, k-(n-1))
            j_end = min(n-1, k)
            # i = k - j
            # For each j in range, coefficient = a[i] = a[k-j]
            # This loop is O(n^2); but okay for moderate n and few iterations
            for j in range(j_start, j_end+1):
                rows.append(k)
                cols.append(j)
                data.append(a[k-j])
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

        # Linear program: maximize sum(g) = - minimize -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        ub = np.full(2*n-1, M)  # RHS

        # Solve
        res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        if not res.success:
            print(f"Iter {iteration}: LP failed")
            break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g < 1e-12:
            print(f"Iter {iteration}: g sum negligible")
            break

        # Normalize g to sum 1
        g = g_raw / sum_g

        # Try a few mixing ratios
        improved = False
        for t in (0.2, 0.1, 0.05, 0.02):
            a_new = (1 - t) * a + t * g

"""
Propose a better step‑function by greedy local optimisation.
The objective Φ(a)=2·n·max(a*a)/(sum(a)^2) is scale‑invariant; we keep
the sequence normalised to a unit sum.  Starting from the previous best
(if available) we repeatedly:
    • locate the convolution offset with the largest value,
    • identify the pair of entries that contributes most to that offset,
    • move a small amount of mass from that pair to the currently smallest
      entry of the sequence,
    • accept the move only if the evaluation score improves.
The convolution is evaluated via FFT (O(n log n)) so many thousands of
iterations fit comfortably into the 1000 s budget.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft

def _phi_fft(a: np.ndarray) -> float:
    """Fast computation of the evaluation function (exactly the same
    as `evaluate_sequence` for non‑negative inputs)."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.size
    # linear convolution by zero‑padding and FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = rfft(a, n=size)
    conv = irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights whose Φ‑value is as low as possible.
    The routine uses a simple greedy local‑search that re‑balances mass
    between the most “dangerous’’ pair of entries (the pair that creates the
    current maximum of the self‑convolution) and the currently smallest entry.
    """
    np.random.seed(seed)
    start = time.time()
    # ----------------------------------------------------------------------
    # 1.  Initialise the candidate.
    # ----------------------------------------------------------------------
    try:
        # If the harness supplied a previous solution, use it.

        prev = np.clip(prev, 0.0, 1000.0)
        if prev.ndim != 1 or prev.size == 0 or prev.sum() < 0.01:
            raise ValueError
        a = prev / prev.sum()
    except Exception:
        # No usable previous solution → start from a random Dirichlet vector.
        n = 2500                     # a size that gives a good trade‑off
        a = np.random.rand(n)
        a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_score = _phi_fft(best_a)
    print(f"[{time.time() - start:.2f}s] start length {n}, score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2.  Main greedy loop.
    # ----------------------------------------------------------------------
    iteration = 0
    # temperature for occasional acceptance of worse moves (simulated annealing)
    temperature = best_score * 0.02   # small initial temperature
    while time.time() - start < budget_s * 0.95:
        # compute full convolution (FFT) – also yields the max offset
        size = 1 << ((2 * n - 1).bit_length())
        fa = rfft(a, n=size)
        conv = irfft(fa * fa, n=size)[: 2 * n - 1]
        k_max = int(conv.argmax())                # offset with the worst value

        # indices that contribute to this offset
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k_max - idx_range
        products = a[idx_range] * a[j_range]

        # the pair with the largest product (most responsible for the peak)
        max_idx = int(products.argmax())
        i_max = idx_range[max_idx]
        j_max = j_range[max_idx]

        # destination index – the currently smallest entry (helps to flatten)
        k_target = int(a.argmin())
        if k_target in (i_max, j_max):
            # avoid moving mass to an index already part of the offending pair
            # pick the second‑smallest entry instead
            candidates = np.where(a != a.max())[0]
            if candidates.size > 0:
                k_target = int(candidates[np.argmin(a[candidates])])

        # pick a step size (relative to the smaller of the two donors)
        donor = a[i_max] if i_max != j_max else a[i_max]
        step_frac = np.random.uniform(0.03, 0.15)   # 3‑15 % of the donor mass
        eps = donor * step_frac
        if eps <= 0.0:
            break

        # construct the perturbed vector
        a_new = a.copy()
        if i_max == j_max:
            # only one donor entry
            a_new[i_max] -= eps
            a_new[k_target] += eps
        else:
            a_new[i_max] -= eps
            a_new[j_max] -= eps
            a_new[k_target] += 2.0 * eps

        # numerical safety – enforce non‑negativity (tiny negatives may appear)
        np.maximum(a_new, 0.0, out=a_new)

        # renormalise to keep the sum exactly 1 (scale‑invariant, but avoids drift)
        s = a_new.sum()
        if s < 0.01:
            # something went wrong – abort the optimisation
            break
        a_new /= s

        # evaluate the new candidate (fast FFT version)
        new_score = _phi_fft(a_new)

        # accept if we improve, otherwise maybe accept probabilistically
        accept = False

        iteration += 1

    # ----------------------------------------------------------------------
    # 3.  Final sanity check and return.
    # ----------------------------------------------------------------------
    elapsed = time.time() - start
    print(
        f"[{elapsed:.2f}s] finished after {iteration} iterations, "
        f"best score {best_score:.6f}"
    )
    # Convert to plain Python list as required by the harness
    return [float(x) for x in best_a.tolist()]
