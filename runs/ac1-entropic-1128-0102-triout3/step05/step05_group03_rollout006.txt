"""
Propose a step‑function (non‑negative list) that minimises `evaluate_sequence`.

Algorithm
---------
1. **Arcsine‑law initialization** – the continuous arcsine density
   f(x)=1/(π√(x(1−x))) has the remarkable property that the sum of two
   independent copies is *uniform*.  Discretising this density gives a
   sequence whose self‑convolution is almost flat, yielding a very low
   value of the objective 2·n·max(conv)/(∑a)² ≈ 1.
2. **Simplex projection** – all iterates are projected onto the
   probability simplex {a_i≥0, Σa_i=1}.
3. **Projected gradient descent** – using the exact sub‑gradient of the
   maximal convolution entry (averaged over all active indices when the
   maximum is attained at several positions).  A Nesterov‑style momentum
   and a diminishing step size η_t = η₀/√t are used.
4. **Time‑aware schedule** – the whole routine respects the supplied
   time budget (default 1000 s).  If time runs out after the cheap
   arcsine construction, the current best sequence is returned immediately.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or numerical issues)
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())      # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of a single convolution entry (k)
# ------------------------------------------------------------
def _subgradient_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """Exact sub‑gradient of conv[k] w.r.t. a (linear convolution)."""
    n = a.shape[0]
    idx = np.arange(n)
    i = k - idx
    mask = (i >= 0) & (i < n)

    g = np.zeros_like(a)
    g[mask] = 2.0 * a[i[mask]]

    # diagonal correction (pair i=j counted only once)
    diag = mask & (2 * idx == k)
    if np.any(diag):
        g[diag] = a[idx[diag]]
    return g


# ------------------------------------------------------------
# Projected gradient descent on the simplex (max‑convolution objective)
# ------------------------------------------------------------
def _gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    beta: float = 0.9,
    step0: float = 0.1,
    max_iter: int = 10**7,
) -> tuple[np.ndarray, float]:
    """Run a few seconds of PGD and return the best vector seen."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"PGD iter={it:,} improve score={cur_score:.6f}"
            )

        # all indices attaining (approximately) the maximum
        tol = 1e-12
        max_inds = np.where(conv >= max_val - tol)[0]

        # average sub‑gradient over the active set
        grad = np.zeros_like(a)
        for k in max_inds:
            grad += _subgradient_at_k(a, k)
        grad /= len(max_inds)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        # projected update
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine builds an arcsine‑law distribution, optionally refines it with
    projected gradient descent and returns the best sequence it has found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size that comfortably fits into the remaining budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30:
        n = 32768               # largest size we can afford (2^15)
    elif remaining > 15:
        n = 16384
    else:
        n = 8192                # fallback for very short runs

    # ------------------------------------------------------------------
    # Build several arcsine‑law candidates (different small offsets)
    # ------------------------------------------------------------------
    deltas = [0.5, 0.75, 1.0, 1.25]   # offset to avoid division by zero
    best_seq = None
    best_score = float("inf")

    for delta in deltas:
        i = np.arange(n, dtype=np.float64)
        # un‑normalised arcsine shape: 1/√((i+δ)*(n‑1‑i+δ))
        a = 1.0 / np.sqrt((i + delta) * (n - 1 - i + delta))
        a /= a.sum()                 # normalise to Σa = 1

        # compute the objective directly (sum is 1, so evaluate_sequence = 2·n·max)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        print(
            f"[{time.time() - start_time:6.2f}s] "
            f"delta={delta:<4} initial score={cur_score:.6f}"
        )
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # If we still have a few seconds, perform a light gradient‑descent
    # refinement starting from the best arcsine candidate.
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 5.0:
        # modest number of iterations – we stop as soon as the budget expires
        grad_deadline = deadline
        a_ref, score_ref = _gradient_descent(
            best_seq,
            n,
            grad_deadline,
            start_time,
            beta=0.92,
            step0=0.12,
            max_iter=5_000_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"PGD final improve score={score_ref:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback: if for some reason we have no sequence, return a uniform one.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_seq).max()
        print(
            f"[{time.time() - start_time:6.2f}s] "
            f"fallback uniform score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Final normalisation (guards against tiny numerical drift)
    # ------------------------------------------------------------------
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # ------------------------------------------------------------------
    # Return as a plain Python list of floats
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
