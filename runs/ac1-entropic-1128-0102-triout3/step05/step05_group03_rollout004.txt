def _gradient_max(a, k):
    n = len(a)
    g = np.zeros_like(a)
    start = max(0, k - (n-1))
    # indices i from start to min(n-1, k)
    for i in range(start, min(k, n-1)+1):
        j = k - i
        g[i] = 2*a[j]
    # diagonal case is automatically counted as 2*a[i] when i=j (i = k/2)
    return g

Multi‑stage optimisation for the step‑function bound.

* Stage 0 – many short runs on a modest size (n≈1024) using a

directly (it reduces to 2·n·max(conv) because the sum is normalised to 1).

The soft‑max gradient approximates the true sub‑gradient of the
max‑convolution but incorporates information from near‑max entries,
allowing a smoother descent and typically reaching a lower bound than a
pure sub‑gradient method.

# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based linear convolutions
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y (length len(x)+len(y)-1)."""
    L = x.shape[0] + y.shape[0] - 1
    fft_len = 1 << ((L - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:L]
    return conv


# ----------------------------------------------------------------------
# Gradient of max‑convolution (exact sub‑gradient)
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) at the (unique) maximal index."""
    n = a.shape[0]
    kmax = int(np.argmax(conv))
    idx = np.arange(n)
    i = kmax - idx
    mask = (i >= 0) & (i < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[i[mask]]          # diagonal is also covered correctly
    return g


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max weighting)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed maximum:
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k)
    The overall objective is 2·n·f_α(a), so the returned gradient already
    incorporates the factor 2·n.
    """
    n = a.shape[0]
    # stabilise the exponential
    max_c = conv.max()
    shifted = conv - max_c               # max becomes 0
    # weights = exp(α·shifted) / Σexp(α·shifted)
    exp_terms = np.exp(alpha * shifted)
    Z = exp_terms.sum()
    w = exp_terms / Z                    # shape (2n‑1,)

    # gradient_i = 4·n· Σ_k w_k·a_{k‑i}
    rev_a = a[::-1]
    full = _conv_full_fft(w, rev_a)     # length 3n‑2
    start = n - 1
    grad = 4.0 * n * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Projected (Nesterov) gradient descent
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          method: str = "subgrad",
                          alpha: float = 10.0,
                          beta: float = 0.9,
                          step0: float = 0.30,
                          max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on  F(a)=2·n·max(conv(a)) (with sum(a)=1).
    `method` can be "subgrad" (exact sub‑gradient) or "softmax"
    (smooth approximation with parameter `alpha`).
    Returns the best vector seen and its (unsmoothed) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self_fft(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter {it:,} (n={n}) improve: {cur_score:.6f}")

        # compute gradient
        if method == "softmax":
            grad = _softmax_gradient(a, conv, alpha)
        else:
            grad = _subgradient_max(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Simple SA: move ε from a high‑mass index to a random one.
    Keeps the global best and returns it.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self_fft(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source bias – proportional to current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self_fft(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # record global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve (n={n}) iter {it:,}: {score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter (previous best) – normalise to sum = 1 if present
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short runs on a modest size (n≈1024) with random restarts
    # ------------------------------------------------------------------
    n0 = 1024
    slice_time = max(30.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time
    restarts = 5

    for r in range(restarts):
        if time.time() >= deadline:
            break
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        # random hyper‑parameters for diversity
        beta = np.random.uniform(0.80, 0.96)
        step0 = np.random.uniform(0.25, 0.45)
        alpha = np.random.uniform(5.0, 15.0)          # for soft‑max gradient
        method = "softmax" if np.random.rand() < 0.7 else "subgrad"

        a_opt, score = _run_gradient_descent(
            a0, n0, slice_deadline, start_time,
            method=method,
            alpha=alpha,
            beta=beta,
            step0=step0,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve (restart {r}) score={score:.6f}")

    # ------------------------------------------------------------------
    # Ensure we have a fallback if nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self_fft(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 1 – upscale to medium size (n≈4096) and longer gradient descent
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.40 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_gradient_descent(
        a1, n1, grad_deadline1, start_time,
        method="softmax",
        alpha=12.0,
        beta=0.92,
        step0=0.28,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to a large size (n≈12288) and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    # allocate most of the remaining budget to this stage; keep some for annealing
    grad_time2 = max(15.0, 0.55 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_gradient_descent(
        a2, n2, grad_deadline2, start_time,
        method="softmax",
        alpha=15.0,
        beta=0.94,
        step0=0.24,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve score={score2:.6f}")

    # ------------------------------------------------------------------
    # Optional Stage 3 – try an even larger grid if time still permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # try to push a little further; use the same vector up‑sampled to 16384
        n3 = 16384
        a3 = _upsample(best_seq, n3)
        remaining = deadline - time.time()
        grad_time3 = max(10.0, 0.30 * remaining)
        grad_deadline3 = time.time() + grad_time3

        a3_opt, score3 = _run_gradient_descent(
            a3, n3, grad_deadline3, start_time,
            method="softmax",
            alpha=18.0,
            beta=0.95,
            step0=0.22,
        )
        if score3 < best_score:
            best_score = score3
            best_seq = a3_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 3 improve score={score3:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-7,
            steps=300_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Very short polishing gradient run (few seconds) for final touch‑up
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = deadline
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            best_seq.shape[0],
            polish_deadline,
            start_time,
            method="subgrad",          # exact sub‑gradient for final tiny steps
            beta=0.96,
            step0=0.10,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: {score_polish:.6f}")

    # ------------------------------------------------------------------
    # Normalise to a proper probability vector and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
