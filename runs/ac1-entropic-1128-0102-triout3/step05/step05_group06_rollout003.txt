"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm is a continuation‑based optimisation:

1. **Simplex re‑parameterisation** – we store the vector on a probability
   simplex by using soft‑max logits.  This guarantees non‑negativity and
   sum = 1 (the objective is scale‑invariant).

2. **Smooth‑max surrogate** – the true objective `max(conv)` is non‑smooth.
   We replace it with a log‑sum‑exp (soft‑max) with a temperature τ that
   is gradually decreased (continuation).  The gradient of the surrogate
   is available in closed form and can be computed with two FFT‑based
   convolutions per iteration.

3. **Adam on logits** – Adam (adaptive moment estimation) is applied to the
   logits using the surrogate gradient.  Learning‑rate and τ are
   schedule‑controlled; we stop each stage when the time budget for that
   stage expires.

4. **Progressive up‑sampling** – we start with a modest length (≈ 800) and
   repeatedly up‑sample the best sequence to larger sizes
   (1600 → 3200 → 6400 → 9600).  Larger n gives a tighter theoretical lower
   bound 2 n · max / ∑² ≈ 1.

5. **Final simulated‑annealing walk** – a cheap mass‑transfer walk refines
   the best discrete solution found by the gradient phase.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence discovered as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via FFT (full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using FFT (returns length len(a)+len(b)-1)."""
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)-1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. a
# ----------------------------------------------------------------------
def _smoothmax_and_grad(a: np.ndarray, tau: float):
    """
    Returns (smooth_max, grad_a) where smooth_max = (1/τ) log Σ exp(τ·conv)
    and grad_a = Σ_k p_k ∂conv[k]/∂a, p_k = softmax(τ·conv).
    """
    conv = _conv_self(a)                         # length 2n‑1
    z = tau * conv
    z_max = np.max(z)
    expz = np.exp(z - z_max)
    p = expz / expz.sum()                        # soft‑max weights, sum=1

    # ∂conv[k]/∂a_i = 2·a_{k‑i} (zero outside the range)
    # Hence grad_a = 2 * (a_rev * p) evaluated at appropriate indices.
    a_rev = a[::-1]
    conv_rev = _conv_fft(a_rev, p)               # length 3n‑2
    start = a.shape[0] - 1
    grad_a = 2.0 * conv_rev[start:start + a.shape[0]]
    smooth = (z_max + np.log(expz.sum())) / tau   # = log‑sum‑exp / τ
    return smooth, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on logits (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(
    init_logits: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    tau_start: float = 5.0,
    tau_end: float = 0.2,
    max_iters: int = 10_000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    clip_grad: float = 10.0,
):
    """
    Runs Adam on the logits for the given length `n`.  Returns the best
    sequence (as a probability vector) and its exact objective
    2·n·max(conv).
    """
    x = init_logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = np.exp(x) / np.exp(x).sum()
    best_score = float("inf")
    best_iter = 0

    # schedule for τ – geometric decay towards tau_end
    tau = tau_start
    tau_decay = (tau_end / tau_start) ** (1.0 / max_iters)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute surrogate objective & gradient wrt a -----
        smooth, grad_a = _smoothmax_and_grad(np.exp(x) / np.exp(x).sum(), tau)

        # exact objective (using the current a)
        a = np.exp(x) / np.exp(x).sum()
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_iter = it
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"adam n={n:5d} it={it:6d} τ={tau:.4f} score={cur_score:.6f}"
            )

        # ----- gradient wrt logits (soft‑max Jacobian) -----
        # grad_obj_a = 2·n·grad_a (since objective = 2·n·smoothmax)
        grad_obj_a = 2.0 * n * grad_a
        dot = np.dot(a, grad_obj_a)
        grad_logits = a * (grad_obj_a - dot)  # ∂f/∂x

        # optional clipping to avoid explosions
        if clip_grad is not None:
            np.clip(grad_logits, -clip_grad, clip_grad, out=grad_logits)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- update temperature -----
        tau = max(tau_end, tau * tau_decay)

    # final exact evaluation of the best point
    return best_a, best_score, best_iter


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    steps: int = 300_000,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
):
    """Cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.08  # move up to ~8 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"anneal n={n:5d} it={it:7d} score={score:.6f}"
            )

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,
    uses a continuation‑based Adam optimiser on soft‑max logits,
    progressively upsamples the solution and finishes with a short SA walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a one‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional starter provided by the harness (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # 1. Initial analytical shape – arcsine (Beta(½,½)) density
    # ------------------------------------------------------------------
    n0 = 800
    xs = (np.arange(n0) + 0.5) / n0
    arcsine = 1.0 / np.sqrt(xs * (1.0 - xs))
    a0 = arcsine / arcsine.sum()
    if starter is not None:
        a0 = _upsample(starter, n0)

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling stages
    # ------------------------------------------------------------------
    stage_lengths = [800, 1600, 3200, 6400, 9600]  # geometric growth
    # Reserve ~20 % of the total budget for final annealing.
    anneal_budget = max(30.0, 0.20 * (deadline - start_time))
    grad_budget = (deadline - start_time) - anneal_budget

    # remaining time for gradient phases
    remaining_grad_time = grad_budget

    for stage_n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate a proportion of the remaining gradient budget
        portion = stage_n / sum(stage_lengths)
        stage_time = max(5.0, remaining_grad_time * portion)
        stage_deadline = time.time() + stage_time

        # initialise a for this stage
        if best_seq is not None:
            a_stage = _upsample(best_seq, stage_n)
        else:
            a_stage = _upsample(a0, stage_n)

        # logits = log(a)  (add tiny epsilon to avoid log(0))
        eps = 1e-12
        logits = np.log(a_stage + eps)

        # run Adam optimisation on logits
        a_opt, score_opt, _ = _adam_optimize(
            logits,
            stage_n,
            stage_deadline,
            start_time,
            tau_start=5.0,
            tau_end=0.2,
            max_iters=200_000,
            lr=0.07,
            clip_grad=15.0,
        )

        # keep the best solution found so far (exact objective)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] stage n={stage_n} "
                f"new best score={best_score:.6f}"
            )

        # update remaining gradient budget
        elapsed = time.time() - start_time
        remaining_grad_time = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # 3. Final annealing phase
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq,
            n_final,
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            steps=500_000,
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(
                f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                f"score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 4. Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny drift may happen)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
