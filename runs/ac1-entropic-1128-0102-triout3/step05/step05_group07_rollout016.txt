def _conv_max_full(a):

    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)

def gradient_max_conv(a, mask):
    # mask same shape as conv array; mask entries 1 for active anti-diagonals (approx), else 0, normalized to sum 1.
    # Compute gradient = 2 * (rev_a * mask) where conv of rev_a and mask.
    n = a.shape[0]
    # compute convolution of reversed a with mask using FFT length: conv_len = len(mask) + n - 1?
    # same as earlier: use fft_len = 1 << ((len(mask) + n - 1).bit_length())
    # note we need pad a_rev and mask to same length.
    a_rev = a[::-1]
    conv_len = len(mask) + n - 1
    fft_len = 1 << ((conv_len).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    M = np.fft.rfft(mask, n=fft_len)
    conv = np.fft.irfft(A * M, n=fft_len)[:conv_len]
    # extract gradient entries corresponding to original a indices: start = n-1; end = start + n
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

def upsample_to(a, new_n):
    x_old = np.linspace(0.0, 1.0, a.shape[0])
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, a)
    # ensure non-negative and normalised to simplex (sum=1)
    up = np.maximum(up, 0.0)
    if up.sum() == 0:
        up = np.full_like(up, 1.0/up.shape[0])
    else:
        up = up / up.sum()
    return up

"""
Propose a low‑score step‑function by a hybrid of
* simulated‑annealing on a small problem,
* up‑sampling the best shape,
* Adam (soft‑max) refinement and
* an adaptive exponential‑gradient (mirror‑descent) stage.

The routine respects the 1000 s budget, prints progress and finally
returns the best sequence found as a plain list of floats.

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and project onto the simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Convolution helpers (FFT based – works for any size)
# ----------------------------------------------------------------------
def _conv_max_full(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k * conv_k (a * a) w.r.t. a.
    Implements g = 2 * (rev(a) * w) via FFT.
    """
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    W = np.fft.rfft(w, n=fft_len_grad)
    conv2 = np.fft.irfft(A_rev * W, n=fft_len_grad)
    start = a.shape[0] - 1
    grad = 2.0 * conv2[start:start + a.shape[0]]
    return grad

def _score_of(a: np.ndarray) -> float:
    """Exact objective used for internal optimisation."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    max_b, _ = _conv_max_full(a)
    n = a.shape[0]
    return 2.0 * n * max_b / (s * s)

def _simulated_annealing_optimize(init: np.ndarray,
                                  deadline: float,
                                  max_iters: int = None) -> tuple[np.ndarray, float]:
    """

    Keeps the vector on the probability simplex.
    """
    a = init.copy()
    n = a.shape[0]
    cur_score = _score_of(a)
    best = a.copy()
    best_score = cur_score

    # linear temperature schedule
    T_start, T_end = 1.0, 1e-4
    it = 0
    # time‑based fallback for max_iters
    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break

        i = np.random.randint(n)

        cand[j] += delta

# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 0.0005) -> tuple[np.ndarray, float]:
    """Adam optimiser on a soft‑max approximation of the L∞‑norm."""
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_score = _score_of(a)
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and soft‑max weights
        max_val, conv = _conv_max_full(a)
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule (linear in iteration count)
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)

        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        # gradient of the smooth‑max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score

# Exponential‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with the exact L∞‑subgradient.
    The step size η decays geometrically from eta_start to eta_end.
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = _score_of(a)
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max_full(a)
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # active anti‑diagonals (within eps of the max)
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient of the true max‑norm
        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric schedule for η
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update (clipped to avoid overflow)
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)
        a = a * np.exp(upd)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

def _upsample_to(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a onto a new length and normalise."""
    if a.shape[0] == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.shape[0])
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, a)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        return np.full_like(up, 1.0 / new_n)
    return up / s

        reported = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] Finished – best reported score = {reported:.6f}")
    except Exception as exc:
        print(f"Evaluation error: {exc}")

    return [float(x) for x in best_seq]
