def log_adam_optimize(w_init, max_iters, deadline, ...):
    # w is log-unnormalized potentials.
    m, v = zeros_like(w)
    best_score = inf
    best_w = w.copy()
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break

        # compute a
        a = np.exp(w - np.max(w)) # for numerical stability
        a = a / a.sum()
        # compute conv and max
        max_val, conv = _conv_max(a)
        cur_score = 2*len(a)*max_val
        # update best
        if cur_score < best_score: best_score, best_w = cur_score, w.copy()

        # schedule temperature
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)

        # compute softmax weights
        w_soft = np.exp((conv - max_val)/tau)
        if w_soft.sum() > 0:
            w_soft /= w_soft.sum()
        else:
            w_soft.fill(1.0 / w_soft.size)

        # compute gradient of smooth max w.r.t a
        g_a = _gradient_from_weights(a, w_soft, fft_len_grad)
        # convert to gradient w.r.t w: dF/dw_i = (g_a_i - <g_a, a>) * a_i   (since a = softmax(w))
        # Actually derivative: a_i = exp(w_i)/Z; dF/dw_i = sum_j dF/da_j * da_j/dw_i = g_a_i * a_i - (Sum_j g_a_j * a_j) * a_i (due to derivative of normalization)
        # compute average = np.dot(g_a, a)
        avg = np.dot(g_a, a)
        g_w = (g_a - avg) * a

        # Adam update w
        m = beta1*m + (1-beta1)*g_w
        v = beta2*v + (1-beta2)*(g_w**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = alpha/(np.sqrt(v_hat) + eps)

        w = w - step * m_hat
    best_a = np.exp(best_w - np.max(best_w))
    best_a /= best_a.sum()
    return best_a, best_score
```

def random_transfer_optimize(a, max_iters, deadline, eps=1e-12):
    n = len(a)
    a = a.copy()
    best_a = a.copy()

def _exp_gradient_optimize(a_init, max_iters, deadline, eta_start=0.6, eta_end=5e-5, eps=1e-12):
    a = a_init.copy()
    n = a.shape[0]
    fft_len_conv = 1 << ((2*n - 1).bit_length())
    fft_len_grad = 1 << ((3*n - 2).bit_length())
    best_a = a.copy()
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask near max
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient
        g = _gradient_from_weights(a, w, fft_len_grad)

        # step size schedule geometric
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)
        a = a * np.exp(upd)

        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    # return best found
    max_val, _ = _conv_max(best_a)
    best_score = 2.0 * n * max_val
    return best_a, best_score

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score

        # temperature schedule linear
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        if tau < 1e-14:
            tau = 1e-14
        w = np.exp((conv - max_val) / tau)

"""
Propose a low‑score step‑function sequence.

The search proceeds in three phases for a single problem size (default 1800).
1) A diverse pool of initial vectors (uniform, triangle, Gaussians,
   Dirichlet draws and sparse spikes) is built, optionally up‑sampling the
   previously discovered construction `height_sequence_1`.
2) Each start is refined by a smooth‑max Adam optimiser, then by a hard‑max
   exponentiated‑gradient (mirror‑descent) optimiser.  Both operate on the
   probability simplex and repeatedly evaluate the exact convolution
   (FFT‑based) to obtain the true max‑norm.
3) A cheap random‑transfer local search (move a small amount of mass from
   one coordinate to another) is applied to the best vector from step 2.
   The whole pipeline runs under the global time budget (default 1000 s) and
   keeps the best sequence encountered.  A final long mirror‑descent
   refinement and random fine‑tuning are performed before returning the
   result.

All helper functions are defined at top‑level; no closures are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv.max(), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k * conv_k  w.r.t. `a`.
    Implements g = 2 * (a_rev * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a_init: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.1,
                   temp_end: float = 0.0005) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    The surrogate uses a soft‑max over the convolution entries.
    Returns best vector (as a numpy array) and its score 2·n·max(conv).
    """
    a = a_init.copy()
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # temperature schedule (linear annealing)
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)

        # soft‑max weights
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient of the smooth‑max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update (Euclidean space)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_optimize(a_init: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.6,
                           eta_end: float = 5e-5,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) optimiser that uses the exact
    max‑convolution (hard‑max sub‑gradient).  The update rule is
        a_i ← a_i·exp(−η·g_i)
    followed by normalisation to the simplex.
    Returns the best vector seen and its score.
    """
    a = a_init.copy()
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_a = a.copy()
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask of (approximately) active anti‑diagonals
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient of the L∞‑norm
        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric schedule for η
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update, with clipping to avoid overflow/underflow
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # exp(±30) ≈ 1e13, safe for double
        a = a * np.exp(upd)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    # final score for the best point
    max_val, _ = _conv_max(best_a)
    best_score = 2.0 * n * max_val
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_transfer_optimize(a_init: np.ndarray,
                              max_iters: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a small random amount of mass from one
    coordinate to another (sum stays constant).  Accept the move only if it
    improves the objective.  Returns the best sequence seen.
    """
    a = a_init.copy()
    n = a.shape[0]
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()
    rng = np.random.default_rng()
    eps = 1e-14

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # choose a source index with non‑zero mass
        i = rng.integers(n)
        if a[i] <= eps:
            continue

        # choose a distinct target index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a small fraction of the source mass
        frac = rng.uniform(0.001, 0.02)               # between 0.1 % and 2 %
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # compute new score
        max_val_new, _ = _conv_max(a_new)
        cur_score = 2.0 * n * max_val_new

        if cur_score < best_score:
            a = a_new
            best_a = a_new.copy()
            best_score = cur_score

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Main entry point required by the evaluation harness.

    Runs a timed multi‑stage optimisation (Adam → mirror‑descent → random
    local search) on a single problem size (default 1800).  Returns the best
    sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # safety margin

    # ------------------------------------------------------------------
    # Retrieve the previously discovered construction, if available.
    # ------------------------------------------------------------------
    prev_seq = None
    try:
        prev_seq = np.array(height_sequence_1, dtype=float)
        prev_seq = np.maximum(prev_seq, 0.0)
    except Exception:
        prev_seq = None

    # ------------------------------------------------------------------
    # Problem size (chosen based on earlier experiments)
    # ------------------------------------------------------------------
    n = 1800

    # Upsample previous construction to the current size, if it exists.
    if prev_seq is not None and prev_seq.shape[0] != n:
        x_old = np.linspace(0.0, 1.0, prev_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        prev_seq = np.interp(x_new, x_old, prev_seq)

    # ------------------------------------------------------------------
    # Build a small but diverse pool of initial vectors.
    # ------------------------------------------------------------------
    pool = []

    if prev_seq is not None:
        pool.append(prev_seq.copy())

    # uniform
    pool.append(np.full(n, 1.0 / n))

    # symmetric triangle
    idx = np.arange(n)
    tri = np.minimum(idx + 1, n - idx)
    pool.append(tri / tri.sum())

    # a few centred Gaussian shapes
    for sigma_factor in (0.15, 0.20, 0.25):
        sigma = sigma_factor * n
        gauss = np.exp(-0.5 * ((idx - (n - 1) / 2) / sigma) ** 2)
        pool.append(gauss / gauss.sum())

    # random Dirichlet draws
    for _ in range(5):
        pool.append(np.random.dirichlet(np.ones(n)))

    # random sparse spikes
    for _ in range(4):
        spikes = np.zeros(n)
        k = np.random.randint(3, 7)          # 3‑6 spikes
        pos = np.random.choice(n, size=k, replace=False)
        spikes[pos] = np.random.rand(k) + 0.1
        pool.append(spikes / spikes.sum())

    best_seq = None
    best_score = float("inf")
    cand_id = 0

    # ------------------------------------------------------------------
    # Optimise every candidate in the pool.
    # ------------------------------------------------------------------
    for a0 in pool:
        if time.time() >= deadline:
            break
        cand_id += 1
        print(f"[{time.time() - start_time:6.2f}s] Candidate {cand_id}/{len(pool)}")

        remaining = deadline - time.time()
        # Rough per‑iteration budget (≈5·10⁻⁵ s per iteration on a typical machine)
        # Allocate a generous amount of iterations while staying inside the deadline.
        iters_adam = int(min(200_000, max(50_000, remaining * 30)))
        iters_eg   = int(min(200_000, max(50_000, remaining * 25)))
        iters_rand = int(min(300_000, max(50_000, remaining * 20)))

        # ---- Adam (smooth‑max) -------------------------------------------------
        a_adam, score_adam = _adam_optimize(
            a0.copy(),
            max_iters=iters_adam,
            deadline=deadline,
            alpha=0.2,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            temp_start=0.1,
            temp_end=0.0005,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"   NEW BEST after Adam – score = {best_score:.6f}")

        # ---- Exponentiated‑gradient (hard‑max) ---------------------------------
        a_eg, score_eg = _exp_gradient_optimize(
            a_adam.copy(),
            max_iters=iters_eg,
            deadline=deadline,
            eta_start=0.6,
            eta_end=5e-5,
        )
        if score_eg < best_score:
            best_score = score_eg
            best_seq = a_eg.copy()
            print(f"   NEW BEST after EG – score = {best_score:.6f}")

        # ---- Random local transfer --------------------------------------------
        a_rand, score_rand = _random_transfer_optimize(
            a_eg.copy(),
            max_iters=iters_rand,
            deadline=deadline,
        )
        if score_rand < best_score:
            best_score = score_rand
            best_seq = a_rand.copy()
            print(f"   NEW BEST after local search – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final long refinements on the overall best vector.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        iters_final_eg = int(min(600_000, max(80_000, remaining * 30)))
        if iters_final_eg > 0:
            print(f"[{time.time() - start_time:6.2f}s] Final EG refinement (iters={iters_final_eg})")
            a_final, score_final = _exp_gradient_optimize(
                best_seq.copy(),
                max_iters=iters_final_eg,
                deadline=deadline,
                eta_start=0.4,
                eta_end=1e-5,
            )
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"   IMPROVEMENT after final EG – score = {best_score:.6f}")

        remaining = deadline - time.time()
        if remaining > 1.0:
            iters_final_rand = int(min(400_000, max(70_000, remaining * 25)))
            print(f"[{time.time() - start_time:6.2f}s] Final random refinement (iters={iters_final_rand})")
            a_fin, score_fin = _random_transfer_optimize(
                best_seq.copy(),
                max_iters=iters_final_rand,
                deadline=deadline,
            )
            if score_fin < best_score:
                best_score = score_fin
                best_seq = a_fin.copy()
                print(f"   IMPROVEMENT after final random search – score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback and final normalisation.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200)

    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    # ------------------------------------------------------------------
    # Report the final evaluation (optional, for logging only).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] Finished – final score = {final_score:.6f}")
    except Exception:
        final_score = None
        print(f"[{time.time() - start_time:6.2f}s] Finished – evaluation failed")

    return [float(x) for x in best_seq]
