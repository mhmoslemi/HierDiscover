    Search for a non‑negative step‑function minimizing evaluate_sequence.
    Uses multi‑stage optimisation:
      - initialise from previous run or random simplex point,
      - progressive projected gradient descent with a soft‑max surrogate,
      - refinement with a high‑order power surrogate (sharp max approximation),
      - short simulated‑annealing walk for final polishing.
    Returns a list of floats (probability simplex).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin
    
    # Load starter if available
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None
    
    # Desired final length
    final_n = 9600
    
    # initial candidate
    if starter is not None:
        a0 = _upsample(starter, final_n)
    else:
        a0 = np.random.rand(final_n)
        a0 /= a0.sum()
    
    # Allocate time: 80% to gradient, 20% to annealing
    total_time = deadline - start
    grad_time = total_time * 0.80
    anneal_time = total_time - grad_time

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  The optimiser works in three stages:

1.  Projected (Nesterov) gradient descent on a *soft‑max* approximation of
    the max‑convolution.  This gives a fast smooth landscape.
2.  A refinement using a *power‑weight* surrogate (large p) which is a
    sharper approximation of the true max.
3.  A short simulated‑annealing walk that swaps tiny amounts of mass between
    two positions – it often squeezes a few extra points out of the tail.

The routine respects the given time budget, normalises the final vector to
the probability simplex, and returns it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with a weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of the max)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k)·conv_k with soft‑max weights
    w_k = exp(conv_k / T) / Σ.
    """
    maxc = conv.max()
    # stable soft‑max
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled - np.max(scaled))  # extra stability
    w = exps / exps.sum()
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    return 2.0 * full[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Power‑gradient (sharp approximation of the max, p >> 1)
# ----------------------------------------------------------------------
def _power_gradient(a: np.ndarray, conv: np.ndarray, p: float) -> np.ndarray:
    """
    Gradient of Σ_k conv_k^p.
    Uses normalised weights w_k = conv_k^{p‑1} / Σ conv^{p‑1}.
    """
    # avoid underflow / zero‑to‑zero problems
    eps = 1e-30
    w = np.maximum(conv, eps) ** (p - 1.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    return 2.0 * full[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Projected gradient descent using the soft‑max surrogate
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the smooth soft‑max objective."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # temperature schedule (exponential decay)
    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: "
                  f"n={n}, iter={it}, score={cur_score:.6f}")

        # temperature decay
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        # gradient of the smooth max
        grad = _softmax_gradient(a, conv, T)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent using the power‑p surrogate (p grows)
# ----------------------------------------------------------------------
def _run_power_gd(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  step0: float = 0.4,
                  beta: float = 0.95,
                  p_start: float = 4.0,
                  p_end: float = 128.0,
                  max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """Projected gradient descent where the loss approximates the max via a large power."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # number of iterations over which p will be increased
    it_target = 20000.0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] power improve: "
                  f"n={n}, iter={it}, p≈{p_start * (p_end / p_start) ** (it / it_target):.1f}, "
                  f"score={cur_score:.6f}")

        # schedule exponent p (geometric growth)
        frac = min(it / it_target, 1.0)
        p = p_start * (p_end / p_start) ** frac

        grad = _power_gradient(a, conv, p)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The optimiser proceeds through:
        * a soft‑max projected gradient phase,
        * a high‑p power‑gradient refinement,
        * a short simulated‑annealing walk.
    Returns a list of floats (probability simplex, sum = 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Try to load a previous construction if it exists
    # ------------------------------------------------------------------
    starter = None
    try:
        tmp = np.asarray(height_sequence_1, dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            starter = tmp / tmp.sum()
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Target length (the previous best run used 9600)
    # ------------------------------------------------------------------
    final_n = 9600

    # ------------------------------------------------------------------
    # Initialise a starting point on the simplex
    # ------------------------------------------------------------------
    if starter is not None:
        a_initial = _upsample(starter, final_n)
    else:
        a_initial = np.random.rand(final_n)
        a_initial /= a_initial.sum()

    # ------------------------------------------------------------------
    # Divide the time budget
    # ------------------------------------------------------------------
    total_time = max(0.0, deadline - start_time)
    grad_time = total_time * 0.80          # 80 % for gradient‑based optimisation
    anneal_time = total_time - grad_time    # the rest for final annealing

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent (coarse optimisation)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + grad_time * 0.60   # first 60 % of gradient budget
    a_soft, score_soft = _run_softmax_gd(
        a_initial, final_n,
        deadline=phase1_deadline,
        start_time=start_time,
        step0=0.6,
        beta=0.9,
        max_iter=10 ** 7,
    )

    # ------------------------------------------------------------------
    # Phase 2 – power‑p refinement (sharper approximation of the max)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + grad_time
    a_ref, score_ref = _run_power_gd(
        a_soft, final_n,
        deadline=phase2_deadline,
        start_time=start_time,
        step0=0.4,
        beta=0.95,
        p_start=4.0,
        p_end=128.0,
        max_iter=10 ** 7,
    )

    best_seq = a_ref
    best_score = score_ref
    print(f"[{time.time() - start_time:6.2f}s] best after gradient phases: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing (tiny mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_final, score_final = _run_annealing(
            best_seq,
            final_n,
            deadline=anneal_deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=5e-2,
            end_temp=1e-5,
            steps=800_000,
        )
        if score_final < best_score:
            best_seq = a_final
            best_score = score_final
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise to the simplex (tiny drift may have occurred)
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:6.2f}s] final score reported: {best_score:.6f}")

    # Return a plain Python list for the harness
    return [float(x) for x in best_seq]
