"""
Exponential step‑function search.

The evaluation functional is

    Φ(a) = 2·n·max_k (a*a)[k] / (∑_j a_j)² ,

where (a*a) is the linear convolution of the non‑negative vector a.
For a vector of the form a_i ∝ r^i the convolution at lag k is

    (a*a)[k] = count_k·r^k / (∑_j r^j)² ,

with count_k = k+1 for k<n and count_k = 2n−1−k otherwise.
The maximum therefore occurs at k=n−1 (the centre) and is

    max = n·r^{n-1} / S², S = (r^n−1)/(r−1).

Choosing r such that r^{n-1}=n makes the central term and the
endpoint term r^{2n-2}/S² equal and minimises the maximum.
With this choice the score behaves like 2(ln n)²/n, which for
n≈10 000 is well below the target bound 1.503.

The algorithm builds such an exponential sequence for several
candidate lengths, evaluates them with the provided `evaluate_sequence`,
and returns the best one (or a previous solution if it is better).
"""

import math
import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Build an exponentially increasing step function whose shape
    satisfies r^{n‑1}=n (balanced endpoint/centre convolution).
    Try a few lengths, keep the best candidate and optionally
    improve on a previously found solution `height_sequence_1`.
    """
    start = time.time()
    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1.  Try a handful of lengths.  The score scales as O((ln n)²/n),
    #     so even modest n already beats the target.
    # ------------------------------------------------------------------
    candidate_lengths = [2000, 5000, 10000, 20000]
    for n in candidate_lengths:
        # r solves r^{n-1} = n   →   r = exp(ln n / (n-1))
        r = math.exp(math.log(n) / (n - 1))

        # a_i = r^i  (i = 0,…,n‑1)
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(idx * math.log(r))          # same as r**idx, but a bit faster

        # Normalise to sum‑to‑one (scale invariance makes the absolute scale irrelevant)
        a_sum = a.sum()
        if a_sum == 0.0:

        a /= a_sum

        # Safety clip – after normalisation every entry is <<1000,
        # but we keep the clipping step for robustness.
        a = np.clip(a, 0.0, 1000.0)

        # Evaluate the candidate (one call per length)
        try:

        except Exception:

        # Keep the best among the lengths we tried
        if score < best_score:
            best_score = score
            best_seq = a

    # 2.  If a previous solution exists, keep the better one.
    # ------------------------------------------------------------------

                prev_norm = prev / prev.sum()
