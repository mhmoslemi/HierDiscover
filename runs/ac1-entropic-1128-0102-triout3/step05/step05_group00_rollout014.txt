def _subgrad_max(a, conv):
    k = conv.argmax()
    n = a.shape[0]
    g = np.zeros_like(a)
    start_i = max(0, k - (n-1))
    end_i = min(k, n-1)
    # i runs from start_i to end_i
    # for each i, j = k - i
    # j is also in [0, n-1]
    # set g[i] = 2 * a[j]
    i_range = np.arange(start_i, end_i+1)
    j_range = k - i_range
    g[i_range] = 2 * a[j_range]
    return g, k
```

def _run_subgradient(a0, n, slice_deadline):
    a = a0.copy()
    best = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # compute subgradient
        g = np.zeros_like(a)
        k = conv.argmax()
        start_i = max(0, k - (n - 1))

        i_idx = np.arange(start_i, end_i+1)

        a = a - step * g
        # projection onto simplex
        a = _project_simplex(a)

"""
Propose a step‑function with a low self‑convolution peak.

Algorithm
---------
1. **Warm‑starts** – if a previous best (`height_sequence_1`) is
   available it is up‑sampled to the current length.

2. **Diverse initial shapes** – for each length we generate a small
   portfolio of candidates: uniform, triangular, Hann window,
   Gaussian, a few sine‑power windows and several Dirichlet draws.

3. **Sub‑gradient descent on the exact max‑norm** – the objective
   `2·n·max(conv)` is non‑smooth.  For the index `k*` where the
   convolution attains its maximum the sub‑gradient is known in closed
   form and we descend with a diminishing step size
   `η_t = 0.5 / sqrt(t)`, always projecting back onto the probability
   simplex.

4. **Annealing polish** – after sub‑gradient optimisation a cheap
   mass‑transfer simulated‑annealing phase makes small ε‑moves
   (source → destination) and accepts worsening moves with a geometric
   temperature schedule.

5. **Time management** – the 1000 s budget is split over a geometric
   schedule of lengths (256 … 32768).  Early lengths receive a modest
   fixed slice; the final (largest) length consumes the remaining time.

The routine returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())      # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(seq) to new_n
    and re‑normalisation to sum = 1.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _run_subgradient(a0: np.ndarray,
                     n: int,
                     slice_deadline: float,
                     start_time: float,
                     symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective 2·n·max(conv).
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")

        # sub‑gradient of the max entry (index k*)
        k = int(conv.argmax())
        # build gradient g_i = 2 * a_{k-i} when the partner index lies in range
        low_i = max(0, k - (n - 1))
        high_i = min(k, n - 1)
        i_idx = np.arange(low_i, high_i + 1, dtype=int)
        j_idx = k - i_idx
        g = np.zeros_like(a)
        g[i_idx] = 2.0 * a[j_idx]

        # diminishing step size
        step = 0.5 / math.sqrt(it)

        a = a - step * g

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150000) -> tuple[np.ndarray, float]:
    """
    Cheap mass‑transfer annealer.
    Moves a small ε from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # propose a move
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12   # up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


def _generate_initial_variants(base_seq: np.ndarray | None, n: int) -> list[np.ndarray]:
    """
    Create a collection of plausible starting vectors for length n.
    The returned vectors are already normalised to sum = 1.
    """
