"""
A fast projected Adam‑style optimiser for the step‑function problem.

Main ideas
~~~~~~~~~
* Initialise from the best known construction (`height_sequence_1`) if it exists,
  otherwise from a random simplex point.
* Optimise the normalised sequence (sum = 1) with a smooth soft‑max
  approximation of the max‑convolution.  The temperature is annealed so that
  the gradient eventually becomes the exact sub‑gradient of the max.
* Adam momentum (bias‑corrected first and second moments) together with a
  diminishing step size provides a stable descent.
* After the smooth phase the algorithm switches to the true max‑sub‑gradient,
  which yields a sharper direction close to a local optimum.
* A short deterministic greedy “mass‑transfer” refinement finishes the run.
* All updates are projected onto the probability simplex, guaranteeing
  non‑negativity and a unit sum (the objective is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed sequence a_rev with a weight vector w.
    Returns the full linear convolution (length = len(a_rev)+len(w)-1).
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of a *soft‑max* of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of the smooth approximation  log‑sum‑exp_T(conv)
    (the factor 2 comes from the symmetry of the convolution).
    """
    maxc = conv.max()
    # stable soft‑max, shifting by maxc before scaling
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()                     # shape (2n‑1,)

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)        # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]       # central block corresponds to a‑indices
    return grad


# ----------------------------------------------------------------------
# Helper: exact sub‑gradient of the max‑convolution (active‑set version)
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Returns a sub‑gradient of max(conv) w.r.t. a.
    Uses the equally‑weighted active set of indices attaining the maximum.
    """
    maxc = conv.max()
    # tolerance to collect all “almost” active positions
    eps = maxc * 1e-12
    active = conv >= maxc - eps
    cnt = np.count_nonzero(active)
    if cnt == 0:                     # fallback – should never happen
        w = np.zeros_like(conv)
    else:
        w = active.astype(float) / cnt
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, runs a fast Adam optimiser
    with a soft‑max approximation followed by an exact max sub‑gradient phase,
    and finishes with a deterministic greedy mass‑transfer refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Choose a working length. 9600 gave the best published result;
    # we stay with it to keep per‑iteration cost modest.
    # ------------------------------------------------------------------
    n = 9600

    # ------------------------------------------------------------------
    # Initialise the sequence (normalised to sum = 1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # up‑sample / down‑sample to the chosen length
            if starter.size != n:
                old_x = np.linspace(0.0, 1.0, starter.size)
                new_x = np.linspace(0.0, 1.0, n)
                a = np.interp(new_x, old_x, starter)
            else:
                a = starter.copy()
        else:
            raise ValueError
    except Exception:
        a = np.random.rand(n)

    a = a.astype(float)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Adam optimiser parameters
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n)          # first moment
    v = np.zeros(n)          # second moment

    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    # Temperature schedule for the soft‑max phase
    T_start = 2.0e-2
    T_end = 5.0e-4
    softmax_iters = 20000     # after this we switch to the exact max sub‑gradient

    while time.time() < deadline:
        it += 1

        # current convolution and objective
        conv = _conv_self(a)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc

        # record best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] iter={it:7d}  score={cur_score:.6f}")

        # gradient choice
        if it <= softmax_iters:
            # smooth approximation – temperature decays exponentially
            T = T_start * (T_end / T_start) ** (it / softmax_iters)
            grad = _softmax_gradient(a, conv, T)
        else:
            # exact max sub‑gradient (active‑set)
            grad = _max_subgradient(a, conv)

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step size – moderate and slowly decreasing
        step = 0.5 / math.sqrt(it)   # base step, can be tuned

        # Parameter update and projection onto the simplex
        a = _project_simplex(a - step * m_hat / (np.sqrt(v_hat) + epsilon))

        # safety break if we are very close to the deadline
        if time.time() >= deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # Greedy mass‑transfer refinement (deterministic hill‑climb)
    # ------------------------------------------------------------------
    # Use whatever time is still left (but no more than a few seconds per move)
    remaining = deadline - time.time()
    if remaining > 2.0:
        # decide on a modest number of attempts; each attempt costs O(n log n)
        # approximate cost ~0.004 s per attempt for n=9600 → about 200 attempts/second
        max_attempts = int(min(20000, remaining / 0.004))
        cur_score = best_score
        a = best_a.copy()
        for _ in range(max_attempts):
            # pick a source with relatively large mass and a sink with small mass
            i = np.argmax(a)
            j = np.argmin(a)
            if a[i] <= 0.0:
                break
            eps = a[i] * 0.07          # move up to 7 % of the source mass
            a_prop = a.copy()
            a_prop[i] -= eps
            a_prop[j] += eps

            conv_prop = _conv_self(a_prop)
            score_prop = 2.0 * n * conv_prop.max()
            if score_prop < cur_score - 1e-12:
                a = a_prop
                cur_score = score_prop
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] greedy improve: score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (the optimisation works on a simplex, but rounding
    # may have introduced a tiny drift)
    # ------------------------------------------------------------------
    best_a = np.asarray(best_a, dtype=float)
    s = best_a.sum()
    if s > 0:
        best_a /= s
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    # Return a plain Python list – the evaluation harness expects that.
    return [float(x) for x in best_a]
