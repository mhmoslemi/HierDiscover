"""
Propose a step‑function (a non‑negative list) that minimises `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – start from the supplied `height_sequence_1` (if any)
   or a random probability vector.  Normalise to sum = 1.

2. **Multi‑scale optimisation** – the vector is gradually up‑sampled
   (linear interpolation) to larger sizes (≈8 k → 32 k).  At each scale we run
   two optimisation phases:
   * **Phase A – sub‑gradient descent** on the exact L∞‑norm
     (max‑convolution entry).  This gives fast global improvements.
   * **Phase B – smoothed‑max descent** where the objective
     `log‑sum‑exp(ε·conv)` is minimised.  The smoothing parameter ε is
     increased from a small value to a large one, allowing the method to
     first move globally and then fine‑tune the true max.

   Both phases use Euclidean projection onto the probability simplex and a
   Nesterov‑style momentum with a step size η₀/√t.

3. **Late‑stage random annealing** – after the last scale we run a long,
   aggressive simulated‑annealing phase that repeatedly transfers a random
   ε‑mass from a high‑value index to a low‑value index; moves are accepted
   with the Metropolis rule.

4. **Return** the best vector seen (projected onto the simplex, converted to
   plain Python floats).  The routine respects the user‑supplied time budget
   (default 1000 s) and prints progress information.

The combination of a deterministic gradient phase and a global stochastic
phase usually pushes the score below the previous best ~1.513, aiming for the
target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT (real‑FFT)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the (exact) max‑convolution entry
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) (uses the unique maximal index)."""
    n = a.shape[0]
    kmax = int(np.argmax(conv))
    # indices i such that 0 ≤ i < n and 0 ≤ kmax‑i < n
    i = np.arange(n)
    mask = (kmax - i >= 0) & (kmax - i < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[kmax - i[mask]]
    # diagonal correction (i == kmax‑i) – derivative of a_i² is 2·a_i,
    # but the convolution contains the term only once.
    diag = mask & (2 * i == kmax)
    if np.any(diag):
        g[diag] = a[i[diag]]          # half of the full 2·a_i
    return g


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp) – ε controls smoothness
# ----------------------------------------------------------------------
def _grad_smoothed_max(a: np.ndarray, conv: np.ndarray, eps: float) -> np.ndarray:
    """
    Compute ∇  ( 1/eps * log Σ_k exp(eps·conv[k]) )
    using the correlation trick: ∂conv[k]/∂a_i = 2·a_{k‑i}.
    """
    # stable softmax on conv
    cmax = conv.max()
    exp_vals = np.exp(eps * (conv - cmax))
    w = exp_vals / exp_vals.sum()          # weights sum to 1

    # gradient = 2 * correlation(w, a)  (valid mode gives length n)
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


# ----------------------------------------------------------------------
# Projected (sub‑)gradient descent with Nesterov momentum
# ----------------------------------------------------------------------
def _run_descent(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 use_smooth: bool = False,
                 eps_start: float = 10.0,
                 eps_end: float = 200.0,
                 step0: float = 0.4,
                 beta: float = 0.92,
                 max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient (or sub‑gradient) descent on
        F(a) = 2·n·max(a*a)        (if use_smooth=False)
    or the smoothed version if use_smooth=True.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # pre‑compute total number of iterations we can afford
    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] desc improve: n={n}, it={it:,}, score={cur_score:.6f}")

        # gradient / sub‑gradient
        if use_smooth:
            # linearly anneal ε from eps_start to eps_end
            frac = min(1.0, it / max_iter)
            eps = eps_start + frac * (eps_end - eps_start)
            grad = _grad_smoothed_max(a, conv, eps)
        else:
            grad = _subgradient_max(a, conv)

        # momentum update (Nesterov style)
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 4e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealing: move a random ε‑fraction from a source index
    (chosen proportionally to its mass) to a random destination.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # source biased by mass, destination uniform
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30          # up to 30 % of source
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, it={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) + renormalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Initial vector (use previous if available)
    # --------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # --------------------------------------------------------------
    # Stage 0 – modest size, many short restarts (n≈2048)
    # --------------------------------------------------------------
    n0 = 2048
    slice_time = max(20.0, 0.10 * (deadline - time.time()))   # allocate ~10 %
    slice_deadline = time.time() + slice_time

    for r in range(5):                     # a few restarts
        if time.time() >= deadline:
            break
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        beta = np.random.uniform(0.80, 0.95)
        step0 = np.random.uniform(0.30, 0.45)

        a_opt, score = _run_descent(
            a0, n0, slice_deadline, start_time,
            use_smooth=False,
            step0=step0,
            beta=beta,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: restart={r}, score={score:.6f}")

    if best_seq is None:                     # fallback
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # --------------------------------------------------------------
    # Stage 1 – medium size (n≈8192) + smooth‑max descent
    # --------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)
    remaining = deadline - time.time()
    grad_time1 = max(30.0, 0.45 * remaining)      # 45 % to gradient phases
    grad_deadline1 = time.time() + grad_time1

    # Phase A: sub‑gradient descent (coarse)
    a1_mid, score_mid = _run_descent(
        a1, n1, grad_deadline1, start_time,
        use_smooth=False,
        step0=0.35,
        beta=0.91,
    )

    # Phase B: smoothed‑max descent (fine)
    if time.time() < grad_deadline1:
        a1_opt, score1 = _run_descent(
            a1_mid, n1, grad_deadline1, start_time,
            use_smooth=True,
            eps_start=8.0,
            eps_end=120.0,
            step0=0.20,
            beta=0.94,
        )
    else:
        a1_opt, score1 = a1_mid, score_mid

    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – large size (n≈32768) + long descent + annealing
    # --------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 32768
    a2 = _upsample(best_seq, n2)
    remaining = deadline - time.time()
    grad_time2 = max(30.0, 0.55 * remaining)   # give most of the rest to optimisation
    grad_deadline2 = time.time() + grad_time2

    # Sub‑gradient phase
    a2_mid, score_mid2 = _run_descent(
        a2, n2, grad_deadline2, start_time,
        use_smooth=False,
        step0=0.30,
        beta=0.92,
    )

    # Smoothed‑max phase (if time permits)
    if time.time() < grad_deadline2:
        a2_opt, score2 = _run_descent(
            a2_mid, n2, grad_deadline2, start_time,
            use_smooth=True,
            eps_start=5.0,
            eps_end=150.0,
            step0=0.18,
            beta=0.95,
        )
    else:
        a2_opt, score2 = a2_mid, score_mid2

    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final simulated‑annealing refinement (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-7,
            steps=800_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # --------------------------------------------------------------
    # Normalise to a proper probability vector and return as list
    # --------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
