"""
Search for a low‑maximum self‑convolution step‑function.

The optimizer works on the probability simplex (∑a_i = 1) because the
evaluation function is scale‑invariant.  It combines several ideas that
are complementary to the previous baseline:

1. **Parametric warm‑starts** – a fast random search over three simple
   families (power‑product, truncated Gaussian, beta‑shaped) quickly
   yields good “shapes” for each length.

2. **Continuation‑style smooth gradient descent** – the ∞‑norm of the
   self‑convolution is replaced by a smooth log‑sum‑exp
   approximation with a temperature β that is slowly increased.
   The gradient of the smooth objective is computed analytically
   (a + cross‑correlation of the weight vector with a) using FFTs.
   A momentum term and Euclidean projection onto the simplex are used.
   After every update the vector is symmetrised (a[i]=a[n‑1‑i]) which
   halves the effective dimension and respects the known symmetry of the
   optimum.

3. **Exact projected sub‑gradient descent** – a short phase that uses the
   true active convolution entry to polish the result.

4. **Mass‑transfer annealing** – a cheap Monte‑Carlo move set (transfer
   ε from one index to another) with a geometric temperature schedule
   for final polishing.

The routine runs a length schedule (256 → … → 16384), always up‑sampling
the current best vector before refining it at the next scale.  All
phases respect the global time budget (default 1000 s).  The best
plain‑Python list of heights found is returned.
"""

# ----------------------------------------------------------------------
# Imports & globals
# ----------------------------------------------------------------------
import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v=1, v≥0)."""
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero or negative – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce v[i] = v[n‑1‑i] and re‑normalise to sum = 1."""
    v = (v + v[::-1]) / 2.0
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a * a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation Σ_k w_k a_{k‑i} for i=0…n‑1.
    w has length 2·n‑1 (self‑convolution length),
    a has length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    m = len(w) + len(a_rev) - 1         # = 3·n‑2
    fft_len = 1 << ((m - 1).bit_length())
    Wf = np.fft.rfft(w, n=fft_len)
    Af_rev = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(Wf * Af_rev, n=fft_len)[:m]
    # extract indices n‑1 … n‑1+n‑1 (inclusive) → length n
    return full[n - 1:n - 1 + n]


def score_of(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(a * a) for a normalised (∑a=1) sequence."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for quick warm‑starts
# ----------------------------------------------------------------------
def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term == 0] = eps
    a = term ** p
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ given in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Continuation (smooth) gradient descent
# ----------------------------------------------------------------------
def smooth_gradient_descent(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of the ∞‑norm.
    The smoothness parameter β grows from β_start to β_end.
    Returns the best vector seen and its exact (non‑smooth) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    beta_start = 0.2
    beta_end = 20.0
    max_iter_est = 3000                # used only for the β schedule
    step0 = 0.6
    momentum = np.zeros_like(a)
    gamma = 0.9                         # momentum decay

    while time.time() < deadline:
        it += 1
        # ---- exact scores for reporting ----
        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] smooth improve n={n} it={it} score={cur_sc:.6f}"
            )

        # ---- compute self‑convolution and smooth weights ----
        conv = conv_self(a)                     # length 2·n‑1
        # increase β gradually (continuation)
        ratio = min(1.0, it / max_iter_est)
        beta = beta_start * (beta_end / beta_start) ** ratio

        # soft‑max weights (avoid overflow)
        shift = conv - conv.max()
        w = np.exp(beta * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth objective ----
        cross = cross_corr(w, a)                # length n
        grad = a + cross                        # raw gradient of Σ w·conv
        grad *= (2.0 * n)                       # scale to match original objective

        # ---- momentum & step ----
        momentum = gamma * momentum + grad
        step = step0 / math.sqrt(it)
        a = project_simplex(a - step * momentum)

        # enforce symmetry after projection
        a = symmetrize(a)

        # occasional tiny isotropic noise to escape plateaus
        if it % 800 == 0:
            a += 1e-5 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc

# Projected sub‑gradient descent on the exact ∞‑norm
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent (active max entry) with momentum.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    step0 = 0.8
    momentum = np.zeros_like(a)
    gamma = 0.9

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        idx = conv.argmax()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}"
            )

        # ----- sub‑gradient of the active entry -----
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        # momentum+step
        momentum = gamma * momentum + grad
        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * momentum)
        a = symmetrize(a)

        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------
def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 200000) -> tuple[np.ndarray, float]:
    """Transfer a small ε from one index to another with a geometric temperature schedule."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score
