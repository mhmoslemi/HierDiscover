"""
Propose a step‑function with very low max‑convolution.
Algorithm
---------

* **Progressive up‑sampling** – start with a short vector (800 entries)
  and double its size step by step (2400, 4800, 9600, 19200).  The current
  best candidate is linearly interpolated to the next size.

* **Smooth max gradient descent** – for each size we run a large number of
  projected Nesterov steps on the *soft‑max* approximation of the max‑convolution.
  The temperature is exponentially annealed, giving a gradual transition to
  the true objective.

* **Exact sub‑gradient refinement** – after the biggest size we switch to
  a pure sub‑gradient method that uses the true max (no soft‑max).  This
  eliminates the smoothing bias and can push the solution slightly further.

* **Final simulated annealing** – a cheap Metropolis walk moves a tiny
  fraction of mass between two positions; it helps escape any remaining
  local plateau.

* **Symmetrisation** – after each stage the candidate is forced to be
  symmetric (`a[i] = a[n‑1‑i]`).  The problem is symmetric, and this step
  never worsens the objective while reducing the search space.

The whole procedure respects the time budget (default 1000 s).  Multiple
random starts are used for the first stage, and the best result is kept.
All updates are projected onto the probability simplex (non‑negative,
unit‑sum) – exactly the feasible set of the evaluation function.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3 – convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with w (full length)."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4 – gradient of a soft‑max of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """Gradient of Σ_k w_k(conv_k)·conv_k with soft‑max weights."""
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)   # length = 3·n‑2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]   # central block
    return grad


# ----------------------------------------------------------------------
# Helper 5 – soft‑max (Nesterov) gradient descent
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.8,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov gradient descent on the smooth max (soft‑max).
    Returns the best vector seen and its 2·n·max(conv) score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, iter={it}, score={cur_score:.6f}")

        # temperature schedule (exponential)
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper 6 – exact sub‑gradient descent on the true max
# ----------------------------------------------------------------------
def _run_subgradient(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     step0: float = 0.2,
                     max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on f(a)=max_k (a*a)[k].
    Uses a random active index at each iteration.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    eps_active = 1e-12

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # pick a random active shift
        active = np.where(conv >= M - eps_active)[0]
        k = np.random.choice(active)

        # gradient of conv[k] w.r.t a: g_i = 2*a_{k-i} (if index in range)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # step with classic diminishing schedule
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * grad)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper 7 – simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1    # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper 8 – linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to unit sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Helper 9 – force symmetry a[i] = a[n‑1‑i]
# ----------------------------------------------------------------------
def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Replace seq by (seq+rev(seq))/2 and re‑project onto the simplex."""
    sym = (seq + seq[::-1]) * 0.5
    sym = _project_simplex(sym)
    return sym

    total_time = deadline - start_time
    # allocate about 12 % of total time to the final annealing phase
    anneal_budget = max(30.0, 0.12 * total_time)
    # allocate about 10 % to the exact sub‑gradient refinement after the biggest stage
    subgrad_budget = max(30.0, 0.10 * total_time)
    grad_budget = total_time - anneal_budget - subgrad_budget

    # ------------------------------------------------------------------
    # Early stages (soft‑max GD)
    # ------------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # time share for this stage (proportional to its length)
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for the current size
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            # for the very first stage we try a few random starts
            if n == stage_lengths[0]:
                # a few random restarts (here 5)
                best_local_seq = None
                best_local_score = float("inf")
                for restart in range(5):
                    a_guess = np.random.rand(n)
                    a_guess /= a_guess.sum()
                    a_opt, score_opt = _run_softmax_gd(
                        a_guess, n, stage_deadline, start_time,
                        step0=0.8, beta=0.9, max_iter=10 ** 6
                    )
                    if score_opt < best_local_score:
                        best_local_score = score_opt
                        best_local_seq = a_opt.copy()
                a0 = best_local_seq if best_local_seq is not None else np.full(n, 1.0 / n)
            else:
                a0 = np.random.rand(n)
                a0 /= a0.sum()

        # soft‑max gradient descent
        a_opt, score_opt = _run_softmax_gd(
            a0, n, stage_deadline, start_time,
            step0=0.8, beta=0.9, max_iter=10 ** 7
        )

        # enforce symmetry before comparing / storing
        a_opt = _symmetrize(a_opt)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={score_opt:.6f}")

        # recompute the remaining gradient budget
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - subgrad_budget - elapsed)

    # ------------------------------------------------------------------
    # Exact sub‑gradient refinement on the final size
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        subgrad_deadline = time.time() + subgrad_budget
        a_ref, score_ref = _run_subgradient(
            best_seq, n_final,
            subgrad_deadline, start_time,
            step0=0.2, max_iter=5 * 10 ** 6
        )
        a_ref = _symmetrize(a_ref)
        # compute the true score for the refined candidate
        conv = _conv_self(a_ref)
        score_ref = 2.0 * n_final * conv.max()
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] sub‑gradient improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final annealing phase
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5,
            steps=800_000
        )
        # recompute exact score
        conv = _conv_self(a_ref)
        score_ref = 2.0 * n_final * conv.max()
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (project onto simplex one last time)
    best_seq = _project_simplex(best_seq)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
