"""
Propose a low‑maximum self‑convolution step function.

The algorithm is a hybrid of three stages:

1. **Parametric warm‑starts** – a quick random search over three simple
   families (power‑product, Gaussian, Beta) provides a good initial shape.
2. **Projected gradient descent on a smoothed objective** – we minimise the
   squared‑error between the self‑convolution and the perfectly flat target
   (the uniform distribution).  The gradient can be computed analytically
   via a single convolution, and after each step the iterate is projected
   back onto the probability simplex.
3. **Polishing** – a short sub‑gradient descent on the true ∞‑norm followed
   by a cheap mass‑transfer simulated‑annealing phase.

The schedule starts with small lengths and progressively upsamples the
current best vector to larger sizes, always respecting the overall time
budget (≈1000 s).  The final sequence is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_linear(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a power‑of‑two FFT (no circular wrap‑around)."""
    m = x.size + y.size - 1
    fft_len = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:m]
    return conv


def conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution of a with itself (linear, not circular)."""
    return conv_linear(a, a)


def score_of(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) for a vector that sums to 1."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for quick random warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p (centre‑heavy for p>0)."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term == 0, eps, term)
    a = term ** p
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ given in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected gradient descent on the *squared* convolution error
# ----------------------------------------------------------------------


def grad_descent_sse(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise ½‖conv(a) – uniform‖² via projected gradient descent.
    Returns the best sequence seen and its exact (max‑conv) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # target is a perfectly flat self‑convolution:
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1))

    step0 = 0.8  # base step; will be divided by √iteration
    it = 0

    while time.time() < deadline:
        it += 1
        # current self‑convolution
        conv = conv_self(a)

        # gradient of ½‖conv-target‖²  w.r.t. a :
        #   g = 4 * ( (conv - target) * a_rev )  (linear convolution)
        residual = conv - target
        grad_full = conv_linear(residual, a[::-1])
        g = 4.0 * grad_full[n - 1:n - 1 + n]   # slice of length n

        # normalise step size (direction length independent of n)
        norm = np.linalg.norm(g) + 1e-16
        direction = g / norm
        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * direction)

        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] sse‑grad improve n={n} it={it} score={cur_sc:.6f}")

    return best_a, best_sc

def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method where the sub‑gradient is taken from the
    active self‑convolution entry.  Returns the best vector seen and its
    exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.8
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # active index
        idx = conv.argmax()
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm
        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * direction)

        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc

# Cheap mass‑transfer annealing (polishing)

def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 60000) -> tuple[np.ndarray, float]:
    """Move a small ε from a source index to a destination index, accepting
    downhill moves and occasionally uphill moves according to a geometric schedule."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2   # transfer at most 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

        if time.time() < deadline:
            subgrad_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.15)
            a_opt2, sc_opt2 = subgradient_descent(a_opt, n, subgrad_deadline, start_time)
            if sc_opt2 < best_score:
                best_score = sc_opt2
                best_seq = a_opt2.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={sc_opt2:.6f}")

        # --------------------------------------------------------------
        # Cheap annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, sc_ref = annealing(
                best_seq,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=60000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}")

        # keep the best vector for the next length (upsample later)
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return as a plain Python list; the evaluation routine will clamp/normalise.
    return [float(x) for x in best_seq]
