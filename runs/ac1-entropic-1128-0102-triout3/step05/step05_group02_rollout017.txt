    """

    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

"""
Propose a step‑function with a low value of `evaluate_sequence`.

The algorithm is a simple multi‑stage optimiser that works on the
probability simplex (∑a_i = 1).  The stages are:

1. **Random warm‑starts** – a large number of vectors drawn from a
   few parametric families (power‑product, gaussian, beta, triangular)
   are generated and the best of them is kept.

2. **Projected sub‑gradient descent** – the exact ℓ∞‑norm of the
   self‑convolution is used.  At each iteration the sub‑gradient of the
   active convolution entry (or the average of all active entries) is
   computed analytically, a diminishing step is taken and the iterate is
   projected back onto the simplex.  Nesterov momentum is added to speed
   up convergence.

3. **Intensive simulated‑annealing polish** – after the gradient phase a
   long series of cheap mass‑transfer moves are performed.  The
   temperature follows a geometric schedule and Metropolis acceptance
   is used.  This stage is able to escape the tiny plateaux that the
   sub‑gradient method often gets stuck in.

The optimiser runs a length schedule
`[256, 512, 1024, 2048, 4096, 8192, 16384]`.  For each length the best
vector found is up‑sampled (linear interpolation) and used as a warm‑
start for the next, larger, problem.  Time is split between the three
stages keeping a small safety margin before the global deadline.

The returned list contains plain Python floats; `evaluate_sequence`
will clamp / normalise them internally.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of power‑of‑two size."""
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for zero‑padding (ensures exact linear conv)
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def objective(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` when ∑a = 1:
        2·n·max(a*a)
    """
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


# ----------------------------------------------------------------------
# Shape families (fast random warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n‑i‑0.5)  raised to p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = term ** p
    a /= a.sum()
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the unit interval."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a /= a.sum()
    return a


def shape_triangular(n: int) -> np.ndarray:
    """Simple symmetric triangle that peaks in the middle."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iter: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method with Nesterov momentum.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = objective(a)

    # momentum parameters
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        # compute convolution and locate all almost‑maximal entries
        conv = conv_self(a)
        max_val = conv.max()
        # indices where the entry is within 1e-12 of the max
        active = np.where(conv >= max_val - 1e-12)[0]

        # sub‑gradient: contribution from every active offset
        grad = np.zeros_like(a)
        for off in active:
            # a_i * a_{off‑i}
            idx = np.arange(n)
            j = off - idx
            mask = (j >= 0) & (j < n)
            grad[idx[mask]] += a[j[mask]]

        # plus the diagonal term (i=j) which is already covered when off==0
        # the definition above already adds the diagonal because for off==0
        # we have j = -idx, mask only true for idx==0… Actually the diagonal
        # is added twice when off==0, which matches the true gradient
        # d/da_i Σ a_i*a_i = 2*a_i.
        # Therefore we simply scale by 2.
        grad *= 2.0

        # normalise to obtain a direction independent of magnitude
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        # step size schedule (diminishing)
        step = 0.6 / math.sqrt(t)

        # Nesterov momentum update
        v_new = 0.9 * v - step * direction
        a_candidate = a + v_new
        # projection back onto the simplex
        a_candidate = project_simplex(a_candidate)

        # update state
        a = a_candidate
        v = v_new

        cur_val = objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={t} score={cur_val:.6f}"
            )

        # occasional tiny random perturbation to escape plateaus
        if t % 800 == 0:
            a += 1e-5 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def anneal(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    init_temp: float = 5e-2,
    final_temp: float = 1e-6,
    max_steps: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap mass‑transfer annealing.
    Moves a random ε from a source index to a destination index.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = objective(a)
    cur_val = best_val

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = init_temp * (final_temp / init_temp) ** (step / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass (biased towards small moves)
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_val = objective(a_prop)
        delta = new_val - cur_val

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            cur_val = new_val

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={step} score={cur_val:.6f}"
            )

    return best_a, best_val


# ----------------------------------------------------------------------
# Upsampling (linear interpolation + re‑normalisation)
# ----------------------------------------------------------------------


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    np.random.seed(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # If a previous construction is available, use it as a warm‑start.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        if starter.sum() > 0.0:

    except Exception:
        starter = None

    # Length schedule – start modest, finish with a fairly large vector.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
