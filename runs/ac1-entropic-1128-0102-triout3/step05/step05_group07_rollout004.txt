"""
Propose a low‑score step function.

The routine builds a modest schedule of problem sizes and, for each size,
generates a diverse pool of candidates (uniform, triangular, cosine,
Beta‑family shapes, Dirichlet draws, sparse spikes and up‑sampled
previous best).  Each candidate is projected onto the probability simplex
and a cheap “fast” objective (2·n·max(conv(a,a))) is evaluated.

The most promising candidates are refined by an exponential‑gradient
(mirror‑descent) optimiser that works with the *exact* max‑convolution
sub‑gradient:

    a_i ← a_i·exp(‑η·g_i) ,   g = 2·(a_rev * indicator_of_max)

with a geometric schedule η∈[η_start,η_end] and re‑normalisation onto the
simplex after every update.  Throughout the search the best sequence seen
is kept.  After the size schedule a final long refinement is performed.
All computation respects a global time budget (default 1000 s) and the
function returns the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Make a non‑negative vector sum to 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return (max(conv(a,a)), full linear convolution).
    Uses FFT with a power‑of‑two length for speed.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k(a,a) w.r.t. a.
    Implements g = 2·(a_rev * w) via FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = len(w) + n - 1           # = (2n‑1)+n‑1 = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    W = np.fft.rfft(w, n=fft_len)
    conv2 = np.fft.irfft(A_rev * W, n=fft_len)[:conv_len]
    start = n - 1
    return 2.0 * conv2[start:start + n]


def _fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective when a is already normalised.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    maxv, _ = _conv_max(a)
    s = a.sum()
    if s == 0.0:
        return float("inf")
    return 2.0 * n * maxv / (s * s)


def _generate_beta(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta density on n points (mid‑point rule) – not normalised.
    """
    # avoid exact 0/1 to keep finite values when a<1 or b<1
    x = (np.arange(n) + 0.5) / n
    v = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    # replace NaNs that can appear for extreme a,b
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    return _as_simplex(v)


def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponential‑gradient) optimiser on the simplex.
    Uses the true max‑convolution sub‑gradient.
    Returns the best vector found and its score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _fast_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        maxval, conv = _conv_max(a)
        cur_score = 2.0 * n * maxval  # sum(a) is 1 by construction
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # active set (indices where conv is within eps of the max)
        mask = conv >= maxval - eps
        w = mask.astype(float)
        ws = w.sum()
        if ws > 0.0:
            w /= ws
        else:
            w.fill(1.0 / w.size)

        # sub‑gradient of the max‑norm
        g = _gradient_from_weights(a, w)

        # geometric schedule for η
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update with safe clipping
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)       # exp(±30) ≈ 1e13
        a = a * np.exp(upd)

        # project back onto the simplex (renormalise)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score
