import time
import math
import random
import numpy as np

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def conv_self(a):
    """Full self‑convolution using FFT (real‑valued)."""
    n = a.shape[0]
    m = 2*n - 1
    fft_len = 1 << ((m-1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv

def score_of(a):
    """Exact score used by the evaluator (sum of a is assumed to be 1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()

def upsample(seq, new_n):
    """Linear interpolation and re‑normalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------
# Simple parametric families – used for cheap warm‑starts
# ----------------------------------------------------------------------
def shape_product_power(n, p):
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term == 0] = 1e-12
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def shape_gaussian(n, sigma):
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def shape_beta(n, alpha, beta_par):
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def shape_exponential(n, r):
    i = np.arange(n)
    a = r ** i
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

def shape_sine(n):
    i = np.arange(n)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

# ----------------------------------------------------------------------
# Projected sub‑gradient (or Adam‑style) descent on the simplex
# ----------------------------------------------------------------------
def projected_subgradient(a0, n, deadline, start_time):
    """Simple projected sub‑gradient descent. Returns best vector seen."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        idx = conv.argmax()
        cur_sc = 2.0 * n * conv[idx]
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad n={n} it={it} score={cur_sc:.6f}")

        # ----- sub‑gradient of the active entry -----
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        # normalise gradient for a robust step size
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        step = step0 / math.sqrt(it + 1.0)

        a = project_simplex(a - step * direction)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best_a, best_sc

# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (as a final polishing phase)
# ----------------------------------------------------------------------
def annealing(a0, n, deadline, start_time, cur_best_sc, cur_best_seq,
              start_temp=5e-2, end_temp=1e-5, steps=200000):
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_sc

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.3  # up to 30 % transfer
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n} it={it} score={score:.6f}")

    return best_seq, best_score

"""
Search for a non‑negative step function with a small self‑convolution
∞‑norm.

The strategy:
  1. Choose a single “large” length (4096–16384 depending on the time
     budget) and work only on that size – the score does not depend on the
     absolute scale of the vector, only on its shape.
  2. Build a cheap warm‑start by drawing a handful of parametric families
     (product‑power, Gaussian, Beta, exponential, sine, random Dirichlet)
     and keep the best one.
  3. Run an aggressive projected sub‑gradient descent on the simplex.
     Each iteration computes the exact active convolution entry,
     builds its sub‑gradient, takes a diminishing step and projects back
     onto the probability simplex.
  4. Finish with a long low‑temperature annealing phase that transfers a
     small epsilon of mass between random sites.  Moves that improve the
     objective are always accepted; worsening moves are accepted with

        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)                     # derivative of 2·n·max_conv

        # normalise direction – makes the step size schedule independent of scale
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        step = step0 / math.sqrt(it + 1.0)
        a = project_simplex(a - step * direction)

        # tiny isotropic noise every few hundred steps to escape plateaus
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------
def annealing(a0, n, deadline, start_time, cur_best_sc, cur_best_seq,
              start_temp=5e-2, end_temp=1e-5, steps=400000):
    """Move a small ε from a source index to a destination index."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_sc

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.3   # transfer up to 30 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    provided ``evaluate_sequence`` objective.  The algorithm follows the
    outline described at the top of the file and respects the 1000 s
    wall‑clock budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Attempt to reuse the sequence from the previous run (if available)
    # ------------------------------------------------------------------
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter = starter / s
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Choose a length that balances quality and speed.
    # Longer vectors give more freedom for a lower constant.
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 16384
    elif budget_s >= 400:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time() - start_time:6.2f}s] target length n={n}")

    # ------------------------------------------------------------------
    # Initialise a0 (warm start)
    # ------------------------------------------------------------------
    if starter is not None:
        a0 = upsample(starter, n)
    else:
        a0 = np.random.gamma(1.0, 1.0, size=n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Very quick stochastic search over simple families
    # ------------------------------------------------------------------
    shape_deadline = time.time() + min(8.0, (deadline - time.time()) * 0.12)
    local_best_a = a0
    local_best_sc = score_of(a0)

    while time.time() < shape_deadline:
        kind = np.random.choice(8)
        if kind == 0:
            p = np.random.uniform(-2.0, 2.5)
            cand = shape_product_power(n, p)
        elif kind == 1:
            sigma = np.random.uniform(n / 30.0, n / 2.0)
            cand = shape_gaussian(n, sigma)
        elif kind == 2:
            alpha = np.random.exponential(1.0) + 0.05
            beta_par = np.random.exponential(1.0) + 0.05
            cand = shape_beta(n, alpha, beta_par)
        elif kind == 3:
            r = np.random.uniform(0.3, 0.99)
            cand = shape_exponential(n, r)
        elif kind == 4:
            cand = shape_sine(n)
        else:
            cand = np.random.gamma(1.0, 1.0, size=n)
            cand /= cand.sum()

        sc = score_of(cand)
        if sc < local_best_sc:
            local_best_sc = sc
            local_best_a = cand
            print(f"[{time.time() - start_time:6.2f}s] shape kind={kind} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation: projected sub‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return [float(x) for x in local_best_a]

    grad_deadline = time.time() + max(4.0, remaining * 0.65)
    a_opt, sc_opt = projected_subgradient(local_best_a, n, grad_deadline, start_time)

    best_seq = a_opt
    best_score = sc_opt
    print(f"[{time.time() - start_time:6.2f}s] after sub‑gradient best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Polishing with a long annealing phase (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, sc_ref = annealing(
            a_opt,
            n,
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-6,
            steps=500000,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref
            print(f"[{time.time() - start_time:6.2f}s] after anneal best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return a plain Python list – the external evaluator will clamp/normalise.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
