def _run_lp_opt(a, n, M, t=0.2):
    # Build A_ub
    A = _build_A_ub(a)
    b_ub = np.full(2*n - 1, M)
    c = -np.ones(n)  # minimize -sum b => maximize sum b
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return a
    b = res.x
    sum_b = b.sum()
    if sum_b <= 1.0 + 1e-12:
        # No improvement
        return a
    g = b / sum_b
    # convex combination
    a_new = (1 - t) * a + t * g
    # ensure non-neg and project simplex (should sum=1)
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_simplex(a_new)
    return a_new

"""
Advanced hybrid optimisation for the self‑convolution peak.

The search combines:
* Projected Adam optimisation on a log‑sum‑exp surrogate of  max(a*a).
* A linear‑programming “swap” step (the classic LP‑based attack) that
  increases the total mass while keeping the convolution bounded.
* A simple sub‑gradient descent on the true max‑norm.
* A light mass‑transfer simulated‑annealing polishing phase.
* Multi‑scale warm‑starts and optional symmetry.

All operations stay on the probability simplex (∑aᵢ=1) so the
evaluation 2·n·max(conv) is scale‑invariant.  The routine respects a
hard 1000 s time‑budget and returns the best sequence found.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real, fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact evaluation (without the safety clamp).  Sum(a) is assumed = 1."""
    n = a.shape[0]
    max_b = _conv_self(a).max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Adam optimisation on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a) = λ·log Σ exp(conv/λ).
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max weights
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    # cross‑correlation of w with reversed a
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)            # length 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]   # central part
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth surrogate.  Returns best encountered."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3          # λ = lam_factor * max(conv)
    lam_decay = 0.8
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming swap step (the classic LP attack)
# ----------------------------------------------------------------------


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A with entries
        A_{k,i} = f_i  if   i ≤ k < i + n,
                = 0    otherwise.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _run_lp_improve(a: np.ndarray,
                    n: int,
                    step: float = 0.2,
                    max_iters: int = 3,
                    deadline: float = None,
                    start_time: float = None) -> np.ndarray:
    """Repeated LP‑based improvement of a."""
    a_cur = a.copy()
    for _ in range(max_iters):
        if deadline is not None and time.time() >= deadline:
            break
        conv = _conv_self(a_cur)
        M = conv.max()
        # Build constraint matrix for current a_cur
        A = _build_A_ub(a_cur)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximize sum(b)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if not res.success:
            # LP failed – fall back to the current vector
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # No gain
            break
        g = b / sum_b
        # convex combination
        a_cur = (1.0 - step) * a_cur + step * g
        # tiny numeric drift may break the simplex – re‑project
        a_cur = _project_simplex(a_cur)
        # optional logging
        if start_time is not None:
            score = _objective(a_cur)
            print(f"[{time.time() - start_time:6.2f}s] LP improve n={n} step={step:.2f} score={score:.6f}")
    return a_cur

# Sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         slice_deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        # indices where the maximum is attained
        max_idx = np.flatnonzero(conv == M)
        # compute sub‑gradient (sum of contributions from all maximal shifts)
        grad = np.zeros_like(a)
        for k in max_idx:
            # first part : a_{k-i}
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]
            # second part : a_{i-k}
            i_start2 = max(k, 0)
            i_range2 = np.arange(i_start2, n)
            j_range2 = i_range2 - k
            grad[i_range2] += a[j_range2]

        # step size schedule – diminishing
        step = 0.3 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap annealing (mass transfer) for final polishing
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing."""
    a = a0.copy()
    score = _objective(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.15   # up to 15 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _objective(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """Generate a set of plausible starting vectors."""
    n = base_seq.shape[0]
    variants = [base_seq.copy()]

    # sinusoidal (positive)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # triangular
    half = n // 2
    tri = np.concatenate([np.linspace(1, half, half, endpoint=False),
                          np.linspace(half, 1, n - half)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # random Dirichlet
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants

"""
Advanced hybrid optimisation for the self‑convolution peak.

The search combines:
* Projected Adam optimisation on a log‑sum‑exp surrogate of max(a*a).
* A linear‑programming “swap” step (the classic LP‑based attack) that
  increases the total mass while keeping the convolution bounded.
* A simple sub‑gradient descent on the true max‑norm.
* A light mass‑transfer simulated‑annealing polishing phase.
* Multi‑scale warm‑starts and optional symmetry.

All operations stay on the probability simplex (∑aᵢ=1) so the
evaluation 2·n·max(conv) is scale‑invariant.  The routine respects a
hard 1000 s time‑budget and returns the best sequence found.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real, fast)."""
    n = a.shape[0] + b.shape[0] - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:n]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact evaluation (sum(a)≈1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_self(a).max()


# ----------------------------------------------------------------------
# Adam optimisation on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate
    L_λ(a) = λ·log Σ exp(conv/λ).
    """
    n = a.shape[0]
    max_val = conv.max()
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    a_rev = a[::-1]
    tmp = _conv_fft(w, a_rev)            # length 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]   # central slice
    return grad


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """Projected Adam on the smooth surrogate; returns best encountered."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.3          # λ = lam_factor * max(conv)
    lam_decay = 0.8
    lam_decay_interval = 3000

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        lam = max(conv.max() * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.6 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score

def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A with entries
        A_{k,i} = f_i   if   i ≤ k < i + n,
                = 0      otherwise.
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

    # sinusoidal (positive)
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # triangular
    half = n // 2
    tri = np.concatenate([np.linspace(1, half, half, endpoint=False),
                          np.linspace(half, 1, n - half)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # random Dirichlet
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Global warm‑start (if a previous solution is available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Length schedule – we start modest and grow, retaining the longest
    # length we can afford for a final fine‑tune.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)

    global_a = starter
    best_score = float('inf')
    best_seq = None

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a0 (warm‑start + several variants)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # Allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        fraction = 0.12 / (len(lengths) - idx)           # more time for later lengths
        slice_deadline = time.time() + max(6.0, remaining * fraction)

        # ------------------------------------------------------------------
        # Run several Adam optimisations (different starts, possibly symmetric)
        # ------------------------------------------------------------------
        best_local_a = None
        best_local_score = float('inf')
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            sym = (np.random.rand() < 0.3)
            a_opt, score_opt = _run_adam_opt(start_vec, n,
                                             slice_deadline, start_time,
                                             symmetrize=sym)
            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] adam local best n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        # ------------------------------------------------------------------
        # Optional LP “swap” phase – short, cheap but often beneficial.
        # ------------------------------------------------------------------
        lp_deadline = min(slice_deadline, time.time() + 8.0)
        best_local_a = _run_lp_improve(best_local_a, n,
                                       step=0.2,
                                       max_iters=2,
                                       deadline=lp_deadline,
                                       start_time=start_time)
        best_local_score = _objective(best_local_a)

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after LP n={n} score={best_local_score:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient refinement (if we still have time)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            sg_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.15)
            a_sg, score_sg = _subgradient_descent(best_local_a, n,
                                                  slice_deadline=sg_deadline,
                                                  start_time=start_time)
            if score_sg < best_score:
                best_score = score_sg
                best_seq = a_sg.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after sub‑grad n={n} score={score_sg:.6f}")

        # ------------------------------------------------------------------
        # Final annealing polish for this length
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, score_ref = _run_annealing(best_local_a, n,
                                              deadline=anneal_deadline,
                                              start_time=start_time,
                                              cur_best_score=best_score,
                                              cur_best_seq=best_seq,
                                              start_temp=5e-2,
                                              end_temp=1e-5,
                                              steps=150000)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # remember the best distribution for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Last‑stage heavy polishing on the largest length we managed to keep.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        final_deadline = deadline - 0.5
        a_final, score_final = _run_adam_opt(best_seq, n,
                                             slice_deadline=final_deadline,
                                             start_time=start_time,
                                             symmetrize=True)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final Adam improve score={score_final:.6f}")

        if time.time() < deadline - 0.5:
            anneal_deadline = deadline - 0.2
            best_seq, best_score = _run_annealing(best_seq, n,
                                                  deadline=anneal_deadline,
                                                  start_time=start_time,
                                                  cur_best_score=best_score,
                                                  cur_best_seq=best_seq,
                                                  start_temp=2e-2,
                                                  end_temp=1e-5,
                                                  steps=200000)

    # ------------------------------------------------------------------
    # Fallback – ensures a valid return even if everything fails.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list (the evaluator will clamp / normalise as needed)
    return [float(x) for x in best_seq]
