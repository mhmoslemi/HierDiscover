```
def _convolve_full(x, y):
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:n+m-1]
    return conv
```

```
def _softmax_weights(conv, eps):
    # conv is array length 2n-1
    max_c = conv.max()
    # compute stable exponentials
    exp_shift = np.exp((conv - max_c) / eps)
    p = exp_shift / exp_shift.sum()
    return p
```

assistantfinal**Solution Overview**

The goal is to find a non‑negative sequence `a` (a step‑function) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear convolution of the sequence with itself.  
The score is *scale‑invariant* – only the shape of the normalised vector matters.

The optimiser works in three phases  

1. **Initial shapes** – a small set of analytical candidates (Beta‑densities) and a few random Dirichlet vectors.  
2. **Progressive optimisation** – a fast projected‑gradient descent on the probability simplex, first on a modest size (`≈2000`) and then on a large size (`≈20000`).  Momentum (Nesterov) and an adaptive step‑size are used.  
3. **Final refinement** – a cheap simulated‑annealing walk that moves a tiny amount of mass between two positions.  This removes the last local peaks of the convolution.

All heavy work is performed with FFT‑based convolutions (`O(n log n)`).  
The algorithm respects the supplied time budget (`budget_s`) and always falls back to a uniform distribution.

```python
"""
Fast multi‑stage optimiser for the max‑convolution bound.

1. generates a handful of analytical (Beta) and random simplex starts,
2. runs a projected Nesterov gradient descent (sub‑gradient of the max‑conv)
   on a small size and then on a large size (≈20 k points),
3. finishes with a tiny simulated‑annealing mass‑transfer walk.

The code stays inside the 1000 s budget, projects onto the probability simplex
after each update and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution via FFT (full length)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return conv


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y using FFT."""
    n, m = x.shape[0], y.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) (active constraints only)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    Only indices with conv[k] ≥ max(conv)*(1‑rel_tol) are treated as active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Projected gradient descent with Nesterov momentum
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.8,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent minimizing `max(conv)`.
    Returns the best point seen and its *scaled* score = 2·n·max(conv)
    (the denominator Σa² disappears because of the normalisation Σa=1).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] gd n={n:5d} "
                  f"iter={it:7d} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution (active set)
        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Very cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} "
                  f"iter={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Helper – generate a Beta(α,α) density on a discrete grid
# ----------------------------------------------------------------------
def _beta_density(n: int, alpha: float) -> np.ndarray:
    """Return a normalised symmetric Beta(α,α) density on n points."""
    xs = (np.arange(n) + 0.5) / n
    # avoid zeroes when α<1
    dens = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, alpha - 1.0)
    dens = np.nan_to_num(dens)          # replace NaNs (0⁰) by 0
    if dens.sum() == 0.0:
        dens = np.full(n, 1.0 / n)
    else:
        dens /= dens.sum()
    return dens
```
