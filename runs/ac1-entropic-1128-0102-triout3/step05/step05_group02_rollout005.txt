def conv_fft(x, y):
    m = len(x) + len(y) - 1
    L = 1 << ((m-1).bit_length())

"""
Low‑maximum self‑convolution search.

The optimizer works on the probability simplex (∑a_i = 1) because the
evaluation function is scale‑invariant.  It combines three ideas that were
not used together before:

1. **Fast parametric warm‑starts** – a short stochastic search over a few
   simple families (product‑power, Gaussian, Beta, cosine) quickly finds
   a good shape for a given length.
2. **Smoothed ∞‑norm gradient descent** – instead of the sub‑gradient of
   the exact max, we minimise a smooth soft‑max (log‑sum‑exp) of all
   convolution entries.  The gradient is obtained analytically via a
   single FFT‑based convolution, and each step is projected onto the
   simplex.
3. **Two‑phase annealing** – after the smooth descent a cheap
   mass‑transfer annealing with a high initial temperature and a long
   schedule finishes the optimisation.

The routine gradually increases the sequence length, always up‑sampling
the current best vector, and respects the global 1000 s time budget.
It returns a plain Python list of non‑negative heights.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using a power‑of‑two FFT."""
    m = len(x) + len(y) - 1
    L = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:m]
    return conv


def conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT."""
    return conv_fft(a, a)


def score_of(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) (a is already normalised to sum = 1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for quick random warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i+0.5)*(n-i-0.5) raised to power p.
    p>0 ⇒ centre‑heavy, p<0 ⇒ edge‑heavy.
    """
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term == 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑shaped family a_i ∝ sin(π(i+0.5)/n)^p, p>0."""
    i = np.arange(n) + 0.5
    a = np.sin(np.pi * i / n) ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max of the convolution)
# ----------------------------------------------------------------------


def smooth_gradient_descent(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of max(conv).
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # smoothing parameter λ (soft‑max temperature)
    lam = 5.0
    lam_target = 30.0
    lam_step = (lam_target - lam) / max(1, int((deadline - time.time()) / 0.05))

    step0 = 0.6   # base step size
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve n={n} it={it} score={cur_sc:.6f}")

        # soft‑max weights on the convolution entries
        max_conv = conv.max()
        # avoid overflow/underflow
        exp_vals = np.exp(lam * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient = 2 * (w * a_rev) convolved, slice the middle part
        grad_full = conv_fft(w, a[::-1])
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # projected gradient step (additive)
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = project_simplex(a)

        # increase λ slowly – the surrogate becomes sharper
        if lam < lam_target:
            lam = min(lam_target, lam + lam_step)

        # occasional isotropic jitter to escape plateaus
        if it % 400 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            else:
                a = np.full(n, 1.0 / n)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-6,
              steps: int = 120000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing: move a small ε from one index to another.
    The temperature decays geometrically.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.3   # up to 30 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_score = float("inf")
    best_seq = None
    global_a = None

    # ------------------------------------------------------------------

    try:

        s = starter.sum()

    except Exception:
        pass

    # Length schedule – start modest, increase gradually.
    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0 (warm start / random)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # allocate a slice of the remaining time for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # give a tiny fixed budget to the smallest lengths, the rest proportionally
        min_slice = 12.0
        slice_time = min_slice + remaining * (n / sum(lengths))
        slice_deadline = time.time() + min(slice_time, remaining)

        # --------------------------------------------------------------
        # quick stochastic search over simple parametric families
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(2.0, (slice_deadline - time.time()) * 0.2)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2, 3])
            if kind == 0:                     # product‑power family
                p = np.random.uniform(-2.5, 2.5)
                cand = shape_product_power(n, p)
            elif kind == 1:                   # Gaussian family
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:                   # Beta family
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            else:                             # Cosine family
                p = np.random.uniform(0.5, 5.0)
                cand = shape_cosine(n, p)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}")

        # --------------------------------------------------------------
        # smoothed gradient descent (soft‑max surrogate)
        # --------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = smooth_gradient_descent(local_best_a, n, grad_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after smooth n={n} score={sc_opt:.6f}")

        # --------------------------------------------------------------
        # cheap annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.2)
            a_ref, sc_ref = annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=0.5,
                end_temp=1e-6,
                steps=120000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}")

        # keep the best vector for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (the evaluator will clamp/normalise as needed)
    return [float(x) for x in best_seq]
