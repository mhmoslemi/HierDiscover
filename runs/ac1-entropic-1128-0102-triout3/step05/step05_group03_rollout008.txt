```
       a = _project_simplex(a)
   return best_a, best_score
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the score is invariant under a global scaling we may keep the
sequence on the probability simplex (`∑a = 1`).  
For a fixed length `n` the problem becomes

* minimise `max(conv(a))` with `conv = a * a` (linear convolution).

`max(conv)` is a non‑smooth function – the previous approach used a plain
sub‑gradient of the *single* maximal entry.  
Here we improve the descent direction by **softening the max**:

```
w_i =  softmax_β ( conv_i )               (β ↑  →  max )
```

and use the **exact gradient of  Σ_i w_i·conv_i**.  
The gradient can be computed in *O(n log n)* by a single FFT‑based
convolution, then an Adam‑style optimiser together with an Euclidean
projection onto the simplex is applied.  
A geometric schedule for `β` (starting with a very smooth objective and
ending with a sharp one) guides the optimiser from the “average”
convolution to the true maximum.

The optimisation is performed in three stages:

1. **Coarse stage** – `n≈2048`, many random restarts, short budget.
2. **Medium stage** – up‑sample the best vector to `n≈8192`,
   continue the Adam‑descent.
3. **Fine stage** – up‑sample to `n≈16384`, finish with a short
   simulated‑annealing “mass‑transfer’’ refinement.

All stages respect the global time limit (`budget_s`).  
The routine returns the best sequence found (as plain Python floats);
the harness will evaluate it with `evaluate_sequence`.

```python
"""
propose_candidate – minimise evaluate_sequence by smoothed‑max Adam descent
==========================================================================

We treat the sequence as a probability vector (∑a = 1) and minimise
2·n·max(a*a).  The non‑smooth max is approximated by a softmax with a
temperature that is lowered during optimisation.  The resulting smooth
objective has a closed‑form gradient that can be evaluated with a single
FFT‑based convolution.  An Adam optimiser together with a Euclidean
projection onto the simplex drives the search.  The algorithm works in
three stages (coarse → medium → fine) and finishes with a short
simulated‑annealing mass‑transfer refinement.

The implementation respects a user supplied time budget (default 1000 s)
and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Softmax with inverse temperature β (larger β → sharper, approaches max)."""
    # numerically stable
    scaled = conv * beta
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    return exp_scaled / np.sum(exp_scaled)


def _gradient_softmax(a: np.ndarray, conv: np.ndarray,
                     weights: np.ndarray) -> np.ndarray:
    """
    Exact gradient of  f(a)= Σ_i w_i·conv_i  with respect to a,
    where conv = a*a (linear) and w_i are fixed softmax weights.
    """
    n = a.shape[0]
    # term 2 * Σ_i w_i * a_{i-j}
    # This is a cross‑correlation between w and a.
    # Compute by convolving the reversed weight vector with a.
    wr = np.convolve(weights[::-1], a)          # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)            # positions needed
    grad = 2.0 * wr[idx]

    # diagonal correction: for i = 2*j we counted a_j twice but derivative
    # should contain it only once.
    diag_pos = 2 * np.arange(n)
    mask = diag_pos < conv.size
    grad[mask] -= weights[diag_pos[mask]] * a[mask]
    return grad


# ----------------------------------------------------------------------
# Optimisation core – smoothed‑max Adam descent
# ----------------------------------------------------------------------


def _run_smoothed_adam(a0: np.ndarray, n: int,
                       deadline: float, start_time: float,
                       lr: float = 0.05,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8,
                       beta_start: float = 1e2,
                       beta_end: float = 1e6,
                       beta_factor: float = 1.002) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of 2·n·max(a*a) using a soft‑max approximation.
    Returns the best vector encountered and its (exact) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    it = 0
    beta_cur = beta_start

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # soft‑max weights (smooth max)
        w = _softmax_weights(conv, beta_cur)
        # gradient of the smooth objective
        grad = _gradient_softmax(a, conv, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)               # keep on the simplex

        # increase the sharpness of the soft‑max
        beta_cur = min(beta_end, beta_cur * beta_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale annealing: move a random ε from a heavy index to a random one.
    """
    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) ------------------------------------------------
# ----------------------------------------------------------------------


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length and renormalisation to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The method consists of three Adam‑descent stages of increasing resolution
    (≈2 k → 8 k → 16 k) followed by a brief simulated‑annealing fine‑tune.
    It respects a user‑supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈2048)
    # ------------------------------------------------------------------
    n0 = 2048
    slice_time = max(20.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    restarts = 5
    for r in range(restarts):
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        a_opt, score = _run_smoothed_adam(
            a0,
            n=n0,
            deadline=slice_deadline,
            start_time=start_time,
            lr=0.08,
            beta_start=5e1,
            beta_end=1e5,
            beta_factor=1.005,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

        if time.time() >= deadline:
            break

    if best_seq is None:
        # very unlikely, but keep a safe default
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 1 – upscale to n≈8192 and longer Adam descent
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 8192
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.45 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_smoothed_adam(
        a1,
        n=n1,
        deadline=grad_deadline1,
        start_time=start_time,
        lr=0.05,
        beta_start=1e2,
        beta_end=5e5,
        beta_factor=1.003,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to n≈16384 and final Adam refinement
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 16384
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(15.0, 0.30 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_smoothed_adam(
        a2,
        n=n2,
        deadline=grad_deadline2,
        start_time=start_time,
        lr=0.04,
        beta_start=2e2,
        beta_end=1e6,
        beta_factor=1.002,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            deadline=deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
```
