def max_conv_subgrad(a):
    conv = np.convolve(a, a)
    max_val = conv.max()
    k = conv.argmax()
    # compute gradient g
    # g[j] = 2* a[k - j] if 0 <= k - j < n else 0
    n = a.shape[0]
    g = np.zeros_like(a)
    # Determine overlap region
    start_j = max(0, k - (n-1))
    end_j = min(n-1, k)  # inclusive
    # Equivalent: j indices from start_j to end_j
    # We'll assign g[start_j:end_j+1] = 2*a[k - np.arange(start_j, end_j+1)]
    idx = np.arange(start_j, end_j+1)
    g[idx] = 2 * a[k - idx]
    return max_val, g

```python
"""
Propose a step‑function sequence by projected sub‑gradient descent.
The method works on the simplex (non‑negative entries summing to one) and
optimises the objective 2·n·max(conv(a,a)).  A simple index‑wise gradient
for the maximal convolution entry is computed in O(n) and combined with
Nesterov momentum and a 1/√t stepsize.  The routine first tries to warm‑up
with the smooth soft‑max gradient used previously, then switches to the
exact sub‑gradient.  The best sequence found before the time budget runs
out is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of the maximal convolution entry (exact sub‑gradient)
# ----------------------------------------------------------------------
def _max_conv_and_gradient(a: np.ndarray):
    """
    Returns (max_val, grad, conv) where max_val = max(a*a),
    grad = ∂ max_conv / ∂a (sub‑gradient) and conv is the full convolution.
    """
    conv = _conv_self(a)
    max_val = conv.max()
    k = conv.argmax()
    n = a.shape[0]
    # gradient g_j = 2 * a_{k-j} whenever the index is valid, else 0
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    if start <= end:
        idx = np.arange(start, end + 1)
        g[idx] = 2.0 * a[k - idx]
    return max_val, g, conv


# ----------------------------------------------------------------------
# Soft‑max gradient (kept from the previous solution – used for warm‑up)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of  Σ_k w_k(conv_k) * conv_k  with  w_k = exp(conv_k / T) / Σ.
    The factor 2 comes from the derivative of aᵀJ_k a.
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    # convolution rev_a * w, then take central block
    n = a.shape[0]
    # use FFT for convolution
    fft_len = 1 << ((n + (2 * n - 1) - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + (2 * n - 1) - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Optimiser: warm‑up using soft‑max GD, then exact sub‑gradient GD
# ----------------------------------------------------------------------
def _run_optimisation(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    warmup_frac: float = 0.15,
    step0: float = 0.6,
    beta: float = 0.9,
    max_iter_soft: int = 200_000,
    max_iter_sub: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Runs a two‑phase optimisation.
    * Phase 1 (soft‑max): smooth approximation of the max for a limited number
      of iterations; useful to get a reasonably balanced shape.
    * Phase 2 (exact sub‑gradient): directly minimises the max convolution.
    Both phases use projected Nesterov momentum and a 1/√t stepsize.
    Returns (best_sequence, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # Phase‑1 – soft‑max GD
    T_start = 2.0e-2
    T_end = 5.0e-4
    phase1_iters = max_iter_soft
    for _ in range(phase1_iters):
        if time.time() >= deadline:
            break
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, iter={it}, score={cur_score:.6f}")

        # annealing temperature
        T = T_start * (T_end / T_start) ** (it / (phase1_iters or 1))
        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    # Phase‑2 – exact sub‑gradient GD
    # continue iteration count from previous phase
    while time.time() < deadline and it < max_iter_sub:
        it += 1
        max_val, grad, conv = _max_conv_and_gradient(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple up‑sampling that preserves total mass
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from current length to new_n, then renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

"""
Propose a step‑function sequence by projected sub‑gradient descent.

The objective (after fixing the sum of the coefficients to 1) is
    score = 2·n·max(conv(a,a)),
where conv denotes the linear self‑convolution.
The algorithm works on the probability simplex (non‑negative entries,
summing to one) and consists of two phases for each resolution:

1. A short “soft‑max” gradient descent (the same smooth surrogate that
   the reference solution used) – this quickly yields a balanced shape.
2. An exact sub‑gradient descent on the true max‑convolution, using
   Nesterov momentum and a 1/√t stepsize.  The sub‑gradient for the
   maximal entry can be computed in O(n) by a simple index shift.

The optimisation is performed on increasingly larger discretisations
(800 → 2400 → 4800 → 9600).  The best sequence discovered before the
time budget expires is returned.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: exact sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_conv_and_gradient(a: np.ndarray):
    """
    Compute max(conv(a,a)) together with its sub‑gradient.
    Returns (max_val, grad, full_conv) where grad_j = 2·a_{k−j}
    for the lag k attaining the maximum.
    """
    conv = _conv_self(a)
    max_val = conv.max()
    k = conv.argmax()
    n = a.shape[0]
    grad = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    if start <= end:
        idx = np.arange(start, end + 1)
        grad[idx] = 2.0 * a[k - idx]
    return max_val, grad, conv


# ----------------------------------------------------------------------
# Helper: gradient of the smooth soft‑max surrogate (kept for warm‑up)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k)·conv_k with w_k = exp(conv_k/T)/Σ.
    The factor 2 comes from differentiating aᵀJ_k a.
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()
    rev_a = a[::-1]
    n = a.shape[0]
    fft_len = 1 << ((n + (2 * n - 1) - 1).bit_length())
    A = np.fft.rfft(rev_a, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(A * B, n=fft_len)[: n + (2 * n - 1) - 1]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Optimiser for one resolution (soft‑max warm‑up → exact sub‑gradient)
# ----------------------------------------------------------------------
def _run_optimisation(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    step0: float = 0.7,
    beta: float = 0.9,
    max_iter_soft: int = 30_000,
    max_iter_sub: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Two‑phase projected gradient descent.
    Returns the best sequence and its score (2·n·max(conv)).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # ---------- Phase 1: soft‑max gradient descent ----------
    T_start = 2.0e-2
    T_end = 5.0e-4
    for _ in range(max_iter_soft):
        if time.time() >= deadline:
            break
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, iter={it}, score={cur_score:.6f}")

        T = T_start * (T_end / T_start) ** (it / max_iter_soft)
        grad = _softmax_gradient(a, conv, T)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    # ---------- Phase 2: exact sub‑gradient descent ----------
    while time.time() < deadline and it < max_iter_sub:
        it += 1
        max_val, grad, _ = _max_conv_and_gradient(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve: n={n}, iter={it}, score={cur_score:.6f}")

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate `seq` linearly to length `new_n` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
