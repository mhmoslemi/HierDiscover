"""
Multi‑scale optimiser for the self‑convolution peak.

* Warm‑start the search: start with a tiny vector (256 entries) and
  progressively double its length.  At each scale the current best 
  distribution is linearly interpolated and refined.

* For the first few scales we run a few Adam steps on a smooth
  log‑sum‑exp surrogate of the ℓ∞‑norm (the same surrogate used in the
  reference implementation).  The sequence is kept symmetric and
  projected onto the probability simplex after every update.

* The largest scale (32768 entries) is tackled with a dedicated
  exponentiated‑gradient optimiser that uses the exact sub‑gradient of
  the max‑convolution (a simple convolution of the active‑set mask with
  the reversed sequence).  The step size decays as η₀/√t and the
  distribution is symmetrised and normalised after each update.

* A cheap mass‑transfer simulated‑annealing polishing is performed at
  the end to escape tiny plateaus.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of a linear combination of convolution entries.
    `weights` must have length 2·len(a)‑1 and sum to 1.
    Returns 2·(weights * a_rev)_{centered}.
    """
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length 3·n‑2
    start = a.shape[0] - 1
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a).

    Parameters
    ----------
    a   : current probability vector (sum = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter λ > 0
    """
    max_val = conv.max()
    # stable soft‑max
    w = np.exp((conv - max_val) / lam)
    w /= w.sum()
    return _gradient_from_weights(a, w)


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a (a on the simplex).
    The active set consists of all indices attaining the maximum.
    """
    max_val = conv.max()
    mask = (conv == max_val).astype(float)
    total = mask.sum()
    if total == 0.0:
        return np.zeros_like(a)
    mask /= total                     # any convex comb. of active gradients works
    return _gradient_from_weights(a, mask)

# Optimisers

def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth LSE surrogate.
    Returns the best vector encountered and its exact objective
    (2·n·max(conv) because sum(a)=1).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # λ‑schedule
    lam_factor = 0.35          # initial λ ≈ lam_factor·max(conv)
    lam_decay = 0.7
    lam_decay_interval = 1500

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ (never let it hit zero)
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        # gradient of the smoothed surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size (diminishing with √t)
        base_step = 0.55 / math.sqrt(it + 1.0)

        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # optional symmetry (helps a lot)
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score


def _run_expgrad_opt(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimiser on the exact max‑convolution.
    Uses the sub‑gradient of max(conv) and a √t‑decaying step size.
    Always keeps the distribution symmetric.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve n={n} it={it} score={cur_score:.6f}")

        # exact sub‑gradient of max(conv)
        grad = _subgradient_maxconv(a, conv)

        # step size η_t = η₀ / √t
        eta = eta0 / math.sqrt(it + 1.0)

        # multiplicative update (avoid overflow/underflow)
        factor = np.exp(-eta * grad)
        a = a * factor

        # enforce symmetry and normalise
        a = (a + a[::-1]) * 0.5
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / a.size)

        # occasional tiny random jitter to escape flat regions
        if it % 4000 == 0:
            jitter = np.random.rand(a.size) * 1e-4
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) * 0.5
            a /= a.sum()

    return best_a, best_score


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """
    Simple mass‑transfer simulated annealing.
    Moves a small ε from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.13    # move up to 13 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a handful of plausible starting vectors from a base sequence.
    All vectors are normalised to sum = 1.
    """
    variants = []

    # 1) warm‑start itself
    variants.append(base_seq.copy())

    # 2) sinusoidal shape
    n = base_seq.shape[0]
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3) triangular ramp
    half = n // 2
    tri = np.concatenate([np.linspace(1, half, half, endpoint=False),
                          np.linspace(half, 1, n - half)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4) random Dirichlet‑like draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a plain
    ``list[float]``.  The strategy is:

    1. Warm‑start with several short Adam runs on increasing lengths
       (256 → 16384).
    2. Upsample the best distribution to the final length (32768) and
       refine it for the remaining time using an exponentiated‑gradient
       optimiser that works on the exact max‑convolution sub‑gradient.
    3. Finish with a short simulated‑annealing polish.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Optional warm start from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # 1) Multi‑scale Adam warm‑up (limited budget ~30% of total time)
    # ------------------------------------------------------------------
    warm_lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    warm_budget = min(300.0, (deadline - start_time) * 0.3)
    warm_deadline = start_time + warm_budget
    global_a = starter

    for idx, n in enumerate(warm_lengths):
        if time.time() >= warm_deadline:
            break

        # initialise a0 (warm‑start / several variants)
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # allocate a slice of the remaining warm‑up time
        remaining = warm_deadline - time.time()
        # give later (larger) lengths a slightly bigger share
        slice_deadline = time.time() + max(4.0, remaining * (0.12 / (len(warm_lengths) - idx)))

        best_local_a = None
        best_local_score = float('inf')

        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break
            a_opt, score_opt = _run_adam_opt(start_vec, n, slice_deadline, start_time, symmetrize=True)
            if score_opt < best_local_score:
                best_local_score = score_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] warm‑up improve n={n} score={score_opt:.6f}")

        if best_local_a is None:
            continue

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after warm‑up n={n} score={best_local_score:.6f}")

        global_a = best_seq.copy()    # remember for the next up‑sample

    # ------------------------------------------------------------------
    # 2) Large‑scale exponentiated‑gradient refinement (majority of time)
    # ------------------------------------------------------------------
    final_n = 32768
    if global_a is None:
        a_start = np.full(final_n, 1.0 / final_n)
    else:
        a_start = _upsample(global_a, final_n)

    # allocate the rest of the time to the EG optimiser
    remaining = deadline - time.time()
    if remaining <= 0.0:
        return [float(x) for x in best_seq] if best_seq is not None else [1.0 / final_n] * final_n

    eg_deadline = time.time() + max(20.0, remaining * 0.85)

    a_refined, refined_score = _run_expgrad_opt(a_start, final_n, eg_deadline, start_time, eta0=0.6)

    if refined_score < best_score:
        best_score = refined_score
        best_seq = a_refined.copy()
        print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after exp‑grad score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final annealing polish (if we still have a bit of budget left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.15)
        a_final, final_score = _run_annealing(
            best_seq,
            final_n,
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-5,
            steps=150000,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(final_n, 1.0 / final_n)

    return [float(x) for x in best_seq]
