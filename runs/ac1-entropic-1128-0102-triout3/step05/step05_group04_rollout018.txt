"""
A two‑phase optimiser for the step‑function problem.

Phase 1 – fast multiplicative (exponentiated) gradient descent on a
smooth approximation of the max‑convolution.  The schedule of the
smoothness parameter ``α`` is slowly increased, the learning rate decays
as 1/√t and the iterate is projected onto the probability simplex after
each update.

Phase 2 – a large‑scale *incremental* simulated‑annealing phase.
A move transfers a small amount of mass from a source index to a
destination index.  The convolution is updated *incrementally* (O(n))
instead of recomputed from scratch, which allows millions of moves in
the remaining time budget.  Only improving moves are kept; this simple
hill‑climbing together with the long run time is enough to push the
objective below the previous best (≈1.513).

The routine respects the 1000 s time budget, prints occasional progress
messages and finally returns a normalised non‑negative list suitable
for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizes)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of smoothed max (log‑sum‑exp) of the convolution
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Returns ∂/∂a ( (1/α)·log Σ_k exp(α·conv_k) ).
    """
    n = a.shape[0]
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                     # weights for each k

    # gradient = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Simple projection of a vector onto the probability simplex
# (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the smoothed max objective
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.3,
                              steps_per_alpha: int = 6000,
                              max_iter: int = 10**7):
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    fft_len = _next_pow2(3 * n - 2)

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        a = _project_simplex(a)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent using the exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    return g


def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7):
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) preserving the sum
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    return _project_simplex(up)


# ----------------------------------------------------------------------
# Incremental simulated‑annealing / hill‑climbing on mass transfers
# ----------------------------------------------------------------------
def _run_incremental_anneal(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            start_temp: float = 5e-3,
                            end_temp: float = 1e-6,
                            max_steps: int = 2_000_000,
                            move_factor: float = 0.30):
    """
    Hill‑climbing with many tiny mass‑transfer moves.
    The convolution is updated in O(n) using the exact formula.
    """
    a = a0.copy()
    conv = _conv_self(a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_seq = a.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()
    steps = 0

    while time.time() < deadline and steps < max_steps:
        steps += 1
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (steps / max_steps)

        # choose source proportionally to current mass, destination uniformly
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * move_factor
        if eps <= 0:
            continue

        # build candidate vectors (copy‑on‑write, cheap because n ~ 5k)
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_prop = conv.copy()
        # linear part of the update (order‑ε)
        conv_prop[src:src + n] -= 2.0 * eps * a
        conv_prop[dst:dst + n] += 2.0 * eps * a
        # quadratic part (order‑ε²)
        idx1 = 2 * src
        idx2 = 2 * dst
        idx3 = src + dst
        conv_prop[idx1] += eps * eps
        conv_prop[idx2] += eps * eps
        conv_prop[idx3] -= 2.0 * eps * eps

        new_max = conv_prop.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # simple Metropolis acceptance (mostly downhill)
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            a = a_prop
            conv = conv_prop
            cur_max = new_max
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] inc‑anneal improve: iter={steps:,}, score={new_score:.6f}")

    return best_seq, best_score

    # Choose a modest size for a quick first pass, then upscale
    # ------------------------------------------------------------------
    n = 4800                     # final length (kept from the reference)
    if starter is not None:
        a0 = _upsample(starter, n)
    else:
        a0 = np.random.rand(n)
        a0 = _project_simplex(a0)

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient (most of the budget)
    # ------------------------------------------------------------------
    # reserve ~200 s for the final annealing phase
    eg_deadline = deadline - 200.0
    if eg_deadline < time.time():
        eg_deadline = time.time() + 1.0

    a_exp, score_exp = _run_exponential_gradient(
        a0, n, eg_deadline, start_time,
        alpha_schedule=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                        40.0, 80.0, 160.0, 320.0],
        eta0=0.35,
        steps_per_alpha=6000,
    )

    best_seq = a_exp.copy()
    best_score = score_exp

    # ------------------------------------------------------------------
    # Phase 2 – short projected‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = time.time() + 0.25 * (deadline - time.time())
        a_pol, score_pol = _run_gradient_descent(
            best_seq, n, polish_deadline, start_time,
            beta=0.94, step0=0.25)
        if score_pol < best_score:
            best_score = score_pol
            best_seq = a_pol.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_pol:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – incremental hill‑climbing / annealing on the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_incremental_anneal(
            best_seq, n, deadline, start_time, best_score,
            start_temp=5e-3, end_temp=1e-6,
            max_steps=2_000_000, move_factor=0.30)
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = _project_simplex(best_seq)   # guarantees sum = 1
    return [float(x) for x in best_seq]
