    m = conv.size
    # compute sum_window = moving sum of w over windows length n, offset i: sum_{k=i}^{i+n-1} w[k]
    cumsum = np.concatenate(([0.0], np.cumsum(w)))
    sum_window = cumsum[n:] - cumsum[:n]   # size n

    # compute term1 via convolution
    # conv_rev = np.convolve(w[::-1], a, mode='full')
    # term1 = conv_rev[(m-1) - np.arange(n)]
    # Use FFT:
    N = 1 << ((m + n - 2).bit_length())  # next power of two for length m+n-1.
    conv_rev_fft = np.fft.rfft(w[::-1], n=N)

def term1_via_fft(a, w):
    n = a.shape[0]
    m = w.shape[0]   # = 2n-1
    # compute convolution of reversed w with a (full)
    # length needed = m + n -1
    size = 1 << ((m + n - 2).bit_length())

    conv_full = np.fft.irfft(w_rev_fft * a_fft, n=size)[:m + n - 1]
    term = conv_full[(m - 1) - np.arange(n)]
    return term

"""
propose_candidate
-----------------

A multi‑scale stochastic/gradient optimizer for the ∞‑norm self‑convolution
objective used by `evaluate_sequence`.  The algorithm works in three stages
for each chosen length `n`:

1. **Warm‑start** – a short random search over a few parametric families
   (product‑power, Gaussian, Beta) yields a decent initial shape.
2. **Smooth gradient descent** – the exact max‑convolution is replaced by a
   smooth log‑sum‑exp surrogate with an increasing temperature parameter
   `β`.  The gradient of this surrogate can be written in closed form and
   is computed with FFT‑based convolutions.  After each β‑phase the best
   vector (according to the exact objective) is remembered.
3. **Polishing** – a cheap mass‑transfer simulated‑annealing phase followed
   by a few steps of the classic sub‑gradient method tighten the solution.

The search proceeds on an increasing schedule of lengths
`[1024, 2048, 4096, …]`.  The best vector found so far is up‑sampled and
used as a warm‑start for the next larger length.  All operations respect
the simplex constraint `∑a_i = 1` and `a_i ≥ 0`, which makes the objective
scale‑invariant and matches the pre‑processing performed inside
`evaluate_sequence`.

The implementation stays completely within the 1000 s time budget,
periodically checks the deadline and prints progress messages.
"""

import time
import math
import random
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolutions (FFT based)
# ----------------------------------------------------------------------
def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (real FFT, exact)."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:m]
    return conv


def term1_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute  t_i = Σ_{l=0}^{n-1} w_{i+l}·a_l   for i=0…n‑1
    using a single FFT‑based convolution (w reversed).
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    # length needed for full linear convolution of length m and n
    full_len = m + n - 1
    fft_len = 1 << ((full_len - 1).bit_length())
    w_rev_fft = np.fft.rfft(w[::-1], n=fft_len)
    a_fft = np.fft.rfft(a, n=fft_len)
    full = np.fft.irfft(w_rev_fft * a_fft, n=fft_len)[:full_len]
    # extract the slice (m‑1‑i) for i=0…n‑1
    idx = (m - 1) - np.arange(n)
    return full[idx]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant version)
# ----------------------------------------------------------------------
def score_of(a: np.ndarray) -> float:
    """Exact evaluation:  2·n·max(a*a)  (sum(a) = 1)."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


# ----------------------------------------------------------------------
# Upsampling / interpolation
# ----------------------------------------------------------------------
def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to new length, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric families – cheap random warm‑starts
# ----------------------------------------------------------------------
def shape_product_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term == 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a

# Random shape search (very short)
# ----------------------------------------------------------------------
def random_shape_search(a0: np.ndarray, n: int, deadline: float, start_time: float):
    best_a = a0.copy()
    best_sc = score_of(a0)
    while time.time() < deadline:
        kind = np.random.randint(3)
        if kind == 0:
            p = np.random.uniform(-2.0, 2.5)
            cand = shape_product_power(n, p)
        elif kind == 1:
            sigma = np.random.uniform(n / 30.0, n / 2.0)
            cand = shape_gaussian(n, sigma)
        else:
            alpha = np.random.exponential(1.0) + 0.05
            beta_par = np.random.exponential(1.0) + 0.05
            cand = shape_beta(n, alpha, beta_par)

        sc = score_of(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand
            print(f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑norm (fallback / fine‑tune)
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    step0 = 0.8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        idx = conv.argmax()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # gradient of the active entry
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm
        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * direction)

        # occasional isotropic noise to escape plateaus
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def grad_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of 2·n·log‑sum‑exp(beta·conv)/beta w.r.t. a."""
    n = a.shape[0]
    # softmax weights
    maxc = conv.max()
    exp_vals = np.exp(beta * (conv - maxc))
    w = exp_vals / exp_vals.sum()

    # sliding window sum of w (length n)
    cumsum = np.concatenate(([0.0], np.cumsum(w)))
    sum_window = cumsum[n:] - cumsum[:n]

    # term Σ_{l} w_{i+l} a_l
    term = term1_from_weights(a, w)

    grad = 2.0 * n * (term + a * sum_window)
    return grad


def smooth_gradient_descent(a0: np.ndarray,
                            n: int,
                            betas: list[float],
                            iters_per_beta: list[int],
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """Multi‑β phase gradient descent on the smooth surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    it_total = 0

    for beta, iters in zip(betas, iters_per_beta):
        for _ in range(iters):
            if time.time() >= deadline:
                break
            it_total += 1
            conv = conv_self(a)
            grad = grad_softmax(a, conv, beta)

            norm = np.linalg.norm(grad) + 1e-16
            direction = grad / norm
            step = 0.5 / math.sqrt(it_total)   # diminishing step
            a = project_simplex(a - step * direction)

            cur_sc = score_of(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] smooth n={n} β={beta:.2f} it={it_total} score={cur_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------
def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 60000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
