def simulated_annealing(a, deadline, max_iters=None):
    best_a = a.copy()
    best_max = _conv_max(a)[0]

    cur_a = a.copy()
    cur_max = best_max

    n = len(a)

    # estimate iteration count based on remaining time
    if max_iters is None:
        # default 100k or based on remaining time
        max_iters = int(200000) # maybe too many; adjust later

    # Step size for moving mass: start with step_range = 0.05 maybe; adapt
    step_scale = 0.2

    # Temperature schedule: start T0 = best_max * 0.1 maybe; end ~ best_max*1e-4
    T0 = best_max * 0.2
    Tend = best_max * 1e-6

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        # temperature schedule: linear or exponential
        tau = it / max_iters
        T = T0 * (Tend / T0) ** tau

        # propose move
        # pick i from indices with a_i > epsilon0
        i = np.random.randint(n)
        while cur_a[i] < 1e-12:
            i = np.random.randint(n)
        j = np.random.randint(n)
        # ensure i != j
        if i == j:
            j = (j+1) % n

        # delta fraction of a_i
        # choose delta = cur_a[i] * step_scale * random_uniform()
        delta = cur_a[i] * step_scale * np.random.rand()
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # maintain sum=1 automatically

        # compute new max
        max_new, _ = _conv_max(new_a)
        # prefer low max
        if max_new < cur_max or np.random.rand() < np.exp(-(max_new - cur_max) / T):
            cur_a = new_a
            cur_max = max_new
            # update best
            if cur_max < best_max:
                best_max = cur_max
                best_a = cur_a.copy()

        # adapt step_scale occasionally: if acceptance rate too low, reduce step; if high, increase.

    return best_a, best_max

"""
Propose a low‑score step‑function.

The algorithm proceeds in three phases:

1. **Initialisation** – start from the previous best construction (if
   available) or from a few generic shapes (uniform, triangular,
   Hann, random Dirichlet and sparse spikes).  All candidates are
   projected onto the probability simplex (sum = 1).

2. **Local optimisation** – each candidate is refined by a short Adam
   optimisation on the smooth‑max surrogate of the objective, followed
   by a mirror‑descent (exponential‑gradient) phase that uses the true
   ℓ∞‑norm of the auto‑convolution.  The best vector of the stage is
   kept globally.

3. **Multiscale annealing** – the current global best is up‑sampled
   repeatedly (≈ × 1.5) up to a configurable maximum length.  At each
   scale an extra simulated‑annealing loop (random mass transfers) is
   performed to escape possible local minima, followed by a final
   long exponential‑gradient refinement.  The whole search respects a
   global time budget (default 1000 s).  The final sequence is returned
   as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max(conv(a,a)), full linear convolution)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k * conv_k  w.r.t. a.
    Implements g = 2 * (a_rev * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.05,
                   temp_end: float = 0.0005) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of the ℓ∞‑objective."""
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # best observed score (2·n·max_conv)
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update (Euclidean)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser that directly uses the
    hard ℓ∞‑norm (max of the convolution).  Update:
        a_i ← a_i·exp(−η·g_i)   ,   g = 2·(a * indicator_of_max)
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric schedule for η
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)      # keep exp stable
        a = a * np.exp(upd)

        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the ℓ∞‑objective.
    Randomly transfers a small amount of mass from one index to another.
    Accepts uphill moves with probability exp(−Δ/Τ).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val, _ = _conv_max(a)

    cur_a = a.copy()
    cur_val = best_val

    # temperature schedule – from about 20 % of the current max down to a tiny value
    T0 = best_val * 0.2 + 1e-12
    T_end = best_val * 1e-6 + 1e-12

    # step‑size factor (relative to the source mass)
    step_factor = 0.2
    accept_counter = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear interpolation of temperature on a log‑scale
        tau = it / max_iters
        T = T0 * (T_end / T0) ** tau

        # propose a random transfer
        i = np.random.randint(n)
        while cur_a[i] < 1e-12:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        delta = cur_a[i] * step_factor * np.random.rand()
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_val, _ = _conv_max(new_a)

        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-16)):
            cur_a = new_a
            cur_val = new_val
            accept_counter += 1
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()

        # adapt step_factor every 2000 proposals
        if it % 2000 == 0:
            if accept_counter < 100:          # too few acceptances → reduce step
                step_factor *= 0.9
                step_factor = max(step_factor, 0.02)
            else:
                step_factor *= 1.1
                step_factor = min(step_factor, 0.5)
            accept_counter = 0

    return best_a, best_val

    Search for a non‑negative step function that minimises
        score = 2·n·max(conv(a,a)) / (sum(a)²)

    The routine respects the global time budget and returns the best
    sequence found (as a plain list of floats).
    """

    start_time = time.time()

    deadline = start_time + max(0.0, budget_s - safety)

    # helpers
    # --------------------------------------------------------------
    def _as_simplex(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            return np.full_like(v, 1.0 / v.shape[0])
        return v / s

    # --------------------------------------------------------------
    # initialise from the previously discovered construction, if any
    # --------------------------------------------------------------
    best_seq = None
    try:
        prev = np.array(height_sequence_1, dtype=float)
        best_seq = _as_simplex(prev)
    except Exception:                 # pragma: no cover
        best_seq = None

    if best_seq is None:
        # fallback: uniform of length 1800 (the size of the previous best)
        n0 = 1800
        best_seq = np.full(n0, 1.0 / n0)

    best_score = float('inf')
    n_best = len(best_seq)

    # --------------------------------------------------------------
    # primary optimisation loop – multiscale refinement
    # --------------------------------------------------------------
    max_n = 12000                     # hard upper bound on length
    scale_factor = 1.5                # growth per stage

    n_current = n_best
    while time.time() < deadline and n_current <= max_n:

        # ----------------------------------------------------------

            break

        x_old = np.linspace(0.0, 1.0, n_current)
        x_new = np.linspace(0.0, 1.0, n_next)
        upsampled = np.interp(x_new, x_old, best_seq)
        upsampled = _as_simplex(upsampled)

        # ----------------------------------------------------------
        # build a tiny pool around the up‑sampled vector
        # ----------------------------------------------------------
        pool = [upsampled]

        # a few random Dirichlet draws (different concentration parameters)
        for alpha in (0.5, 1.0, 2.0):
            pool.append(_as_simplex(np.random.dirichlet(alpha * np.ones(n_next))))

        # sparse random spikes (5–9 spikes)
        for _ in range(2):
            spikes = np.zeros(n_next)
            k = np.random.randint(5, 10)
            pos = np.random.choice(n_next, size=k, replace=False)
            spikes[pos] = np.random.rand(k) + 0.1
            pool.append(_as_simplex(spikes))

        # small perturbations of the up‑sampled vector
        for _ in range(2):
            jitter = np.random.randn(n_next) * 0.01
            cand = upsampled + jitter
            pool.append(_as_simplex(cand))

        # ----------------------------------------------------------
        # optimise each candidate in the pool
        # ----------------------------------------------------------
        for cid, cand in enumerate(pool, 1):
            if time.time() >= deadline:
                break

            remaining = deadline - time.time()
            # rough estimate of time per basic iteration for this n
            est_it_time = 8e-4 * (n_next / 1200.0) + 1e-4
            # we allocate a fraction of the remaining budget to this candidate
            alloc = min(remaining * 0.7, max(5.0, remaining * 0.1))
            max_iters = int(max(5000, min(300000, alloc / est_it_time)))

            # split the budget: 30 % Adam, 30 % exponential‑gradient, 40 % SA
            adam_iters = int(0.30 * max_iters)
            eg_iters   = int(0.30 * max_iters)
            sa_iters   = max_iters - adam_iters - eg_iters

            # ---- Adam (smooth‑max) ----
            a_adam, _ = _adam_optimize(
                cand.copy(),
                max_iters=adam_iters,
                deadline=deadline,
                alpha=0.12,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                temp_start=0.07,
                temp_end=0.001,
            )
            # ---- Exponential‑gradient (hard‑max) ----
            a_eg, _ = _exp_gradient_optimize(
                a_adam.copy(),
                max_iters=eg_iters,
                deadline=deadline,
                eta_start=0.9,
                eta_end=0.001,
            )
            # ---- Simulated‑annealing refinement ----
            a_sa, max_val = _simulated_annealing(
                a_eg.copy(),
                deadline=deadline,
                max_iters=sa_iters,
            )
            cur_score = 2.0 * n_next * max_val

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_sa.copy()
                n_current = n_next
                print(f"[{time.time() - start_time:6.2f}s] NEW GLOBAL BEST – n={n_current}, score={best_score:.6f}")

        # after processing the pool, continue to the next larger scale
        n_current = n_next

    # --------------------------------------------------------------
    # final long refinement on the overall best (hard‑max EG)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = len(best_seq)
        remaining = deadline - time.time()
        est_it_time = 8e-4 * (n_final / 1200.0) + 1e-4
        final_iters = int(min(500000, remaining / est_it_time))
        if final_iters > 0:
            print(f"[{time.time() - start_time:6.2f}s] Final EG refinement – n={n_final}, iters={final_iters}")
            a_final, _ = _exp_gradient_optimize(
                best_seq.copy(),
                max_iters=final_iters,
                deadline=deadline,
                eta_start=0.2,
                eta_end=5e-4,
            )
            max_val, _ = _conv_max(a_final)
            final_score = 2.0 * n_final * max_val
            if final_score < best_score:
                best_score = final_score
                best_seq = a_final.copy()
                print(f"[{time.time() - start_time:6.2f}s] IMPROVED AFTER FINAL EG – score={best_score:.6f}")

    # --------------------------------------------------------------
    # safety normalisation before returning
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    # final reporting via the supplied routine (may clip values, etc.)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:          # pragma: no cover
        final_score = float('nan')
    print(f"[{time.time() - start_time:6.2f}s] Finished – best score = {final_score:.6f}")

    return [float(x) for x in best_seq]
