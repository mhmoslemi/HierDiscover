def compute_conv_and_grad(p):
    conv = conv_self(p)
    max_val = conv.max()
    idx = conv.argmax()
    # gradient: grad = p + reversed(p) shifted.
    n = len(p)
    indices = np.arange(n)
    j = idx - indices
    mask = (j >= 0) & (j < n)
    grad = p.copy()
    grad[mask] += p[j[mask]]
    # Note: grad_i = p_i + p_{k-i}
    return conv, max_val, idx, grad
```

"""
Hybrid shape‑bootstrap + Adam + annealing optimizer.

The objective 2·n·max(conv(a,a)) is scale‑invariant, therefore we work on the
probability simplex (∑a_i = 1, a_i ≥ 0).  For a given length n the algorithm
proceeds in three phases:

1. **Warm‑start** – generate many candidate shapes from a rich family
   (power‑product, Gaussian, Beta, sine‑power and mixtures).  Keep the best.

2. **Projected Adam** – minimise the exact ∞‑norm of the self‑convolution.
   The active convolution entry gives a sub‑gradient; Adam’s adaptive step
   sizes together with Euclidean projection onto the simplex provide fast
   convergence.

3. **Mass‑transfer annealing** – cheap stochastic moves (transfer ε from one
   index to another) with a geometric temperature schedule polish the final
   vector.

The search is performed on a schedule of increasing lengths (512 → 32768).
The current best vector is up‑sampled after each stage, so progress on a
smaller problem is inherited by larger ones.  The routine respects the global
1000 s budget and prints intermediate scores.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT (real)."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def max_conv_and_grad(p: np.ndarray):
    """
    Return (max_val, max_idx, gradient) where gradient is a sub‑gradient of
    max(conv(p,p)) w.r.t. p.
    """
    n = p.size
    conv = conv_self(p)
    max_val = conv.max()
    idx = int(conv.argmax())
    indices = np.arange(n)
    j = idx - indices
    mask = (j >= 0) & (j < n)
    grad = p.copy()
    grad[mask] += p[j[mask]]
    return max_val, idx, grad


def score_of(p: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(p,p)) for a probability vector."""
    n = p.shape[0]
    max_val = conv_self(p).max()
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) and normalisation
# ----------------------------------------------------------------------
def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for fast warm‑starts
# ----------------------------------------------------------------------
def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term == 0, eps, term)
    a = term ** p
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_sine_power(n: int, q: float) -> np.ndarray:
    """a_i ∝ sin^q(π (i+0.5)/n)."""
    i = np.arange(n) + 0.5
    a = np.sin(np.pi * i / n) ** q
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_mixture_gaussian(n: int, sigma1: float, sigma2: float,
                           w: float) -> np.ndarray:
    """Convex combination of two centred Gaussians."""
    a1 = shape_gaussian(n, sigma1)
    a2 = shape_gaussian(n, sigma2)
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected Adam optimiser on the simplex (exact max‑norm)
# ----------------------------------------------------------------------
def adam_optimize(p0: np.ndarray, n: int,
                  deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam with projection onto the simplex.  Returns the best vector seen
    (and its exact objective value).
    """
    p = p0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    best_p = p.copy()
    best_sc = score_of(p)

    # base step size (will decay like 1/sqrt(it))
    base_eta = 0.5

    while time.time() < deadline:
        it += 1
        max_val, idx, grad = max_conv_and_grad(p)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing learning rate
        eta = base_eta / math.sqrt(it)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        p_new = p - step
        p_new = project_simplex(p_new)

        p = p_new
        sc = 2.0 * n * max_val
        if sc < best_sc:
            best_sc = sc
            best_p = p.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve n={n} it={it} score={sc:.6f}")

        # occasional isotropic jitter to escape plateaus
        if it % 400 == 0:
            jitter = np.random.randn(n) * 1e-4
            p = project_simplex(p + jitter)

    return best_p, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------
def annealing(p0: np.ndarray, n: int,
              deadline: float, start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 60000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic mass‑transfer with a geometric temperature schedule.
    Returns the best sequence found during the annealing phase.
    """
    p = p0.copy()
    score = score_of(p)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or p[i] <= 0.0:
            continue

        eps = p[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        prop = p.copy()
        prop[i] -= eps
        prop[j] += eps

        new_score = score_of(prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            p = prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = p.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve n={n} it={it} score={score:.6f}")

        # tiny jitter every few thousand steps (helps mixing)
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            p = project_simplex(p + jitter)

    return best_seq, best_score
