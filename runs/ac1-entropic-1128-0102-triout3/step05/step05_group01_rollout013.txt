"""
Search for a non‑negative step–function that minimises ``evaluate_sequence``.

Algorithm
---------
* Work on the probability simplex (sum = 1) – the objective is homogeneous.
* Use several *stages* with increasing length (800 → 2400 → 4800 → 9600).
  Each stage gets a share of the remaining time proportional to its size.
* Inside a stage run a *multiplicative (exponentiated) gradient descent* on the
  exact max‑convolution (sub‑gradient of the max).  The update is
      a_i ← a_i·exp(-η·g_i)   and then renormalise.
  Step size η decays as η₀/√t.  The sub‑gradient is obtained by choosing a
  random active shift (one of the indices where the convolution attains its
  maximum) and convolving the reversed probability vector with a one‑hot
  weight vector – a single FFT‑based convolution per iteration.
* After the last stage a cheap *Metropolis simulated annealing* walk
  (tiny mass transfers) is performed to polish the best solution.
* Multiple random restarts are performed automatically: for each stage the
  optimisation is run several times while time permits, and the best of all
  candidates is kept.
* The final sequence is normalised to sum = 1 and returned as a plain Python
  list (the harness will evaluate it with ``evaluate_sequence``).

The method differs from the previous soft‑max gradient descent and often
reaches a lower bound within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: projection onto the probability simplex (fallback)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed sequence a_rev with a weight vector w.
    Returns the full linear convolution (length = len(a_rev)+len(w)-1).
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4: compute max‑convolution and a sub‑gradient of the max
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Returns
        conv      : full linear convolution a*a
        max_val   : maximum entry of conv
        subgrad   : a sub‑gradient of max(conv) w.r.t. a
    The sub‑gradient is obtained by picking a *random* active shift where
    conv attains its maximum and computing the corresponding linear gradient.
    """
    conv = _conv_self(a)                     # shape (2n-1,)
    max_val = conv.max()
    # indices where conv is (numerically) maximal
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]
    # pick one active index uniformly at random – stochastic sub‑gradient
    k = np.random.choice(active)
    w = np.zeros_like(conv)
    w[k] = 1.0
    # gradient = 2 * (rev(a) * w) restricted to the central n entries
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)       # length = 3n-2
    n = a.shape[0]
    subgrad = 2.0 * full[n - 1 : n - 1 + n]
    return conv, max_val, subgrad


# ----------------------------------------------------------------------
# Exponentiated (multiplicative) gradient descent on the exact max‑conv.
# ----------------------------------------------------------------------
def _run_expgrad(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 eta0: float = 0.5,
                 max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Runs exponentiated gradient descent on the objective 2·n·max(a*a).
    Returns the best point seen (as a probability vector) and its scaled
    objective value 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv, max_val, grad = _maxconv_and_subgrad(a)
        score = 2.0 * n * max_val

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

        # step size decays like 1/√t
        eta = eta0 / math.sqrt(it)

        # multiplicative update, keep everything positive
        factor = np.exp(-eta * grad)
        # avoid overflow / under‑flow
        factor = np.clip(factor, 1e-20, 1e20)
        a = a * factor

        # re‑normalise to stay on the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            # rare numerical collapse – fall back to uniform
            a = np.full(n, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 10 % of a[i] (could be tuned)
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine obeys the time budget, performs
    progressive up‑sampling, exponentiated gradient descent and a final
    simulated‑annealing polish.  The best sequence found is returned as a
    plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run (if the variable exists)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Stage definitions (length, time share)
    # ------------------------------------------------------------------
    stage_lengths = [800, 2400, 4800, 9600]      # progressive enlargement
    total_weight = sum(stage_lengths)

    total_time = deadline - start_time
    # allocate ~15 % of the time for the final annealing walk
    anneal_budget = max(30.0, 0.15 * total_time)
    grad_budget = total_time - anneal_budget

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # time share proportional to stage size
        stage_frac = n / total_weight
        stage_time = max(5.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for the current size
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None:
            a0 = _upsample(starter, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # Run a few random restarts inside the stage while time permits
        local_best_a = a0
        local_best_score = float("inf")
        while time.time() < stage_deadline:
            # random restart (Dirichlet with α=0.5 tends to produce a moderately
            # spread initialization)
            a_restart = np.random.rand(n)
            a_restart /= a_restart.sum()

            a_opt, score_opt = _run_expgrad(
                a_restart, n, stage_deadline, start_time,
                eta0=0.5, max_iter=10 ** 7
            )
            if score_opt < local_best_score:
                local_best_score = score_opt
                local_best_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] stage n={n} new local best: score={score_opt:.6f}")

        # after the local restarts, compare with the global best
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} update global best: score={best_score:.6f}")

        # update the remaining gradient budget after this stage
        elapsed = time.time() - start_time
        grad_budget = max(0.0, deadline - start_time - anneal_budget - elapsed)

    # ------------------------------------------------------------------
    # Final annealing phase – works with the final length we ended up with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5,
            steps=800_000   # a bit more steps than the previous version
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(200, 1.0 / 200.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_seq]
