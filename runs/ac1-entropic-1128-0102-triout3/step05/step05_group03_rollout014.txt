"""
Enhanced search for a step‑function minimizing ``evaluate_sequence``.

Main ideas
----------
* Use a *smooth* surrogate for the ℓ∞‑norm of the self‑convolution:
  F(a) = ( Σ (a*a)[k]**p )**(1/p)  (p≈8–16).  Its gradient is cheap to compute
  via a single FFT‑based correlation.
* Optimise the surrogate with Adam (adaptive moment) projected onto the
  probability simplex.  The optimiser is run on several problem sizes,
  progressively up‑sampling the best vector.
* After the Adam phase a long simulated‑annealing phase (mass‑transfer
  moves and occasional swaps) fine‑tunes the solution using the exact
  objective.
* The whole routine respects the supplied time budget (default 1000 s) and
  returns the best sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth ℓ_p surrogate of max(conv)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, p: int = 8) -> np.ndarray:
    """
    Return the gradient of   ( Σ (a*a)[k]**p )**(1/p)   w.r.t. a.
    Computation uses a single FFT‑based correlation.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length L = 2n-1
    # raise to the power p, safe because conv ≥ 0
    conv_p = conv ** p
    S = conv_p.sum()
    if S == 0:
        return np.zeros_like(a)

    # weight w_k = conv_k**(p-1) * S**(-(p-1)/p)
    scaling = S ** (-(p - 1) / p)
    w = conv ** (p - 1) * scaling            # length L

    # correlation w ⋆ a  (full mode).  For shift ℓ we have
    #   corr[ℓ] = Σ_i w[i] * a[i+ℓ].
    #   The required gradient component is   g_j = 2 * Σ_k w_k * a_{k-j}
    #   i.e. shift ℓ = -j, so use index (n‑1)-j in the full correlation.
    corr = np.correlate(w, a, mode='full')   # length L + n - 1
    start = n - 1
    g = 2.0 * corr[start - np.arange(n)]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate, with projection onto simplex
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              lr: float = 0.12,
              beta1: float = 0.90,
              beta2: float = 0.999,
              eps: float = 1e-8,
              p: int = 8,
              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth ℓ_p objective.
    Returns the best vector seen and its (exact) score according to
    ``evaluate_sequence`` (i.e. 2·n·max(conv) because the sum is 1).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    # pre‑compute bias‑correction denominators for speed
    beta1_pow = 1.0
    beta2_pow = 1.0

    while time.time() < deadline and it < max_iter:
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve: n={n}, iter={it:,}, "
                  f"p={p}, score={cur_score:.6f}")

        # gradient of the smoothed max
        grad = _grad_smooth_max(a, p=p)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

        # optional: slowly decay learning rate
        lr *= 0.9999995

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the exact objective
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 4e-2,
                   end_temp: float = 1e-6,
                   steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer + occasional swap moves.
    Returns the best sequence encountered.
    """
    a = a0.copy()
    conv = _conv_self(a)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # decide move type
        if rng.random() < 0.8:                     # mass‑transfer move
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue
            eps = a[src] * rng.random() * 0.35
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
        else:                                       # swap move
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            a_prop = a.copy()
            a_prop[i], a_prop[j] = a_prop[j], a_prop[i]

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) and renormalisation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate ``seq`` to length ``new_n`` and re‑normalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine follows a multi‑stage scheme:
      1. Small‑scale Adam optimisation on a smooth surrogate (several restarts).
      2. Upsample to larger sizes and continue Adam (p is increased gradually).
      3. Length‑final simulated annealing for fine‑tuning.
    The best vector found within the budget is returned as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Optional starter (previous best) – normalised to sum = 1
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short Adam runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    slice_time = max(30.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    restarts = 6
    for r in range(restarts):
        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        # Adam with a modest p (4) for quick exploration
        a_opt, score = _run_adam(
            a0, n0, slice_deadline, start_time,
            lr=0.15, p=4, max_iter=10_000,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: restart={r}, "
                  f"score={score:.6f}")

        if time.time() >= deadline:
            break

    # fallback if nothing was produced
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 1 – upsample to medium size (n≈4096) and longer Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(25.0, 0.35 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_adam(
        a1, n1, grad_deadline1, start_time,
        lr=0.12, p=8, max_iter=80_000,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upsample to a larger size (n≈12288) and continue Adam
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(30.0, 0.30 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_adam(
        a2, n2, grad_deadline2, start_time,
        lr=0.10, p=12, max_iter=150_000,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – intensive simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-7,
            steps=300_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final short projected‑gradient pass on the exact max (clean‑up)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a couple of hundred simple gradient‑descent steps
        a_final, score_final = _run_adam(
            best_seq, best_seq.shape[0], deadline, start_time,
            lr=0.08, p=2, max_iter=2_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final PGD improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and convert to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
