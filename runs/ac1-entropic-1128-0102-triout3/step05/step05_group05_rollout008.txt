"""
propose_candidate – heuristic search for a step‑function with a small
self‑convolution peak.

The evaluation metric is
    Φ(a) = 2·n·max(a*a) / (sum(a)^2)          (scale‑invariant)

Goal: make the self‑convolution a*a as flat as possible.
The algorithm proceeds in three stages:

1.  Build a small set of deterministic seeds (uniform, power‑law,
    symmetric exponential, and, if available, the previous best
    sequence).  All seeds are normalised to have sum = 1.

2.  Choose the best seed according to `evaluate_sequence`.

3.  Refine the chosen seed by a projected multiplicative gradient
    descent on the quantity max(a*a).  The gradient of the peak entry
    `b_k = (a*a)[k]` w.r.t. `a_i` is `∂b_k/∂a_i = 2·a_{k‑i}`.
    For the current peak location(s) we accumulate this gradient,
    update the vector multiplicatively (`a ← a·exp(‑η·g)`) to keep it
    non‑negative, renormalise, and repeat.  The step size η is
    annealed linearly; occasional tiny random jitter helps to escape
    plateaus.  The loop stops when the time budget is exhausted or when
    a very low score has been reached.

The whole routine runs well within the 1000 s budget and returns the
best sequence found as a plain list of floats.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) improves
    on the current best bound.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility (default 0).
    budget_s : int, optional
        Approximate CPU time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    max_time = budget_s - 0.5

    # ------------------------------------------------------------------
    # 1.  Determine the working length n.
    # ------------------------------------------------------------------
    n = 3200                     # default length used by the previous run
    try:
        prev_raw = np.asarray(height_sequence_1, dtype=float)
        if prev_raw.ndim == 1 and prev_raw.size > 0 and prev_raw.sum() >= 0.01:
            n = len(prev_raw)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2.  Generate a few deterministic initial candidates.
    # ------------------------------------------------------------------
    candidates = []

    # 2.1 uniform
    candidates.append(np.full(n, 1.0 / n, dtype=float))

    # 2.2 power‑law shapes  a_i ∝ ((i+1)*(n-i))**alpha   (α<0 → heavy ends)
    alphas = [-0.1, -0.25, -0.4, -0.5, -0.6, -0.8, -1.0]
    idx = np.arange(n)
    for alpha in alphas:
        # add a small epsilon to avoid exact zeros
        val = ((idx + 1) * (n - idx)) ** alpha
        candidates.append(val / val.sum())

    # 2.3 symmetric exponential decays  a_i ∝ r^{i} (mirrored)
    rs = [0.5, 0.6, 0.7, 0.8, 0.85, 0.9, 0.95]
    for r in rs:
        if n % 2 == 0:
            half = n // 2
            left = r ** np.arange(half, dtype=float)
            shape = np.concatenate([left, left[::-1]])
        else:
            half = n // 2
            left = r ** np.arange(half + 1, dtype=float)
            shape = np.concatenate([left, left[:-1][::-1]])
        candidates.append(shape / shape.sum())

    # 2.4 the previous best sequence (if length matches)
    try:
        prev = np.clip(prev_raw, 0.0, 1000.0)
        if prev.sum() >= 0.01:
            prev_norm = prev / prev.sum()
            if prev_norm.shape[0] == n:
                candidates.append(prev_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Pick the best seed according to the official evaluator.
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for a in candidates:
        # each candidate already sums to 1 → direct score = 2·n·max(b)
        # but we use the provided evaluator for safety.
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:

            best_a = a.copy()
    print(f"[{time.time() - start_time:.2f}s] initial best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the best seed by projected multiplicative gradient descent.
    # ------------------------------------------------------------------
    # Settings
    max_iter = 3000                     # overall iteration budget
    eta0 = 0.5                          # initial step size
    tolerance = 1e-12                   # equality tolerance for max entry

    a = best_a
    n = a.size
    best_seq = a.copy()
    best_val = best_score

    for it in range(max_iter):
        # ------------------------------------------------------------------
        # 4.1 time check
        # ------------------------------------------------------------------
        if time.time() - start_time > max_time:
            break

        # 4.2 convolution a*a via FFT (fast, O(n log n))
        # ------------------------------------------------------------------

        b = np.fft.irfft(A * A, n=m)[: 2 * n - 1]    # self‑convolution
        max_b = b.max()
        current_score = 2.0 * n * max_b               # sum(a)=1 ⇒ this is Φ(a)

        # keep the global best
        if current_score < best_val:

            print(f"[{time.time() - start_time:.1f}s] iter {it:4d} new best {best_val:.6f}")

        if best_val <= 1.4:

            g[i_start : i_end + 1] += 2.0 * a_slice

        # ------------------------------------------------------------------
        # linear annealing of the step size
        eta = eta0 * (1.0 - it / max_iter)
        if eta <= 0:
            break
        a_new = a * np.exp(-eta * g)

        # ------------------------------------------------------------------
        # 4.6 renormalise to keep the sum equal to one
        # ------------------------------------------------------------------
        s = a_new.sum()
        if s == 0 or np.isnan(s):
            a_new = a  # fallback to the previous iterate
        else:
            a_new /= s

        # ------------------------------------------------------------------
        # 4.7 occasional tiny random jitter to escape plateaus
        # ------------------------------------------------------------------

            a_new = a_new + jitter
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s > 0:
                a_new /= s

        a = a_new

    # ------------------------------------------------------------------
    # 5.  Final sanity check and return.
    # ------------------------------------------------------------------
    if best_seq.sum() < 0.01 or np.any(np.isnan(best_seq)):

        best_seq = np.full(200, 1.0 / 200, dtype=float)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:.2f}s] final best score {final_score:.6f} "
          f"length {len(best_seq)}")
    return [float(x) for x in best_seq.tolist()]
