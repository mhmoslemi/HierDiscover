def _run_adam_opt(a0, n, deadline, start_time, alpha_schedule, eta=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8, steps_per_alpha=2000):
    # Convert a0 to logits
    a = a0.copy()
    # to avoid log(0), clip small values
    eps_small = 1e-20
    z = np.log(np.clip(a, eps_small, None))
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)
    fft_len = _next_pow2(3 * n - 2)
    while time.time() < deadline and it < 10**7:
        it += 1
        # compute a via softmax
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        # compute convolution
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")
        # get smoothing parameter
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        # compute smooth max and weight w
        # compute smooth max for gradient but we only need gradient w
        scaled = alpha * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        sum_exp = exp_scaled.sum()
        # w = softmax(alpha*conv)
        w = exp_scaled / sum_exp
        # gradient w.r.t a
        grad_a = _grad_smooth_max_from_w(a, w, n, fft_len)
        # gradient w.r.t z (softmax chain)
        s = np.dot(a, grad_a)
        grad_z = a * (grad_a - s)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z -= eta * m_hat / (np.sqrt(v_hat) + epsilon)
        # advance smoothing parameter schedule
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1
    return best_a, best_score

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Initialise a feasible vector (sum = 1).  If a historic solution
    ``height_sequence_1`` is available it is used, otherwise a random
    simplex point is generated.

2.  **Adam optimisation on the simplex**  
    The variables are the logits ``z`` of a soft‑max parametrisation
    ``a = softmax(z)`` – this guarantees non‑negativity and unit sum.
    We minimise a smoothed version of the objective

        f(z) = 2·n·  (1/α)·log Σₖ exp(α·(a*a)ₖ)

    The smoothing parameter ``α`` is increased gradually (schedule
    ``[5, 20, 80, 200]``).  The gradient of the smooth maximum is
    obtained analytically via FFT‑based cross‑correlation, giving an
    inexpensive O(n log n) update.  Adam’s adaptive moments provide
    fast, stable convergence.

3.  **Polishing** – a short projected‑gradient descent with the exact
    sub‑gradient of ``max(conv)`` removes any remaining slack.

4.  **Final refinement** – a lightweight simulated‑annealing pass that
    moves a small fraction of mass between coordinates.

The whole procedure respects the supplied time budget (≈1000 s) by
checking the wall‑clock after each iteration.  All helper functions are
top‑level, there are no closures, and only ``numpy`` and ``scipy`` are
used.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize  # only used for a tiny LBFGS stub (optional)

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def _smooth_max_and_weights(conv: np.ndarray, alpha: float):
    """
    Returns (smooth_max, w) where
        smooth_max = (1/α)·log Σₖ exp(α·convₖ)
    and w_k = exp(α·convₖ) / Σ_j exp(α·conv_j)  (the soft‑max weights).
    """
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    smooth = (max_scaled + np.log(sum_exp)) / alpha
    w = exp_scaled / sum_exp
    return smooth, w


def _grad_smooth_max_from_w(a: np.ndarray, w: np.ndarray,
                            n: int, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. ``a`` given the weights ``w``.
    Implements   ∂/∂a   (1/α)·log Σₖ exp(α·convₖ)   = 2·(a⋆w_rev).
    """
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the L1‑norm."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  alpha_schedule,
                  eta: float = 0.01,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  epsilon: float = 1e-8,
                  steps_per_alpha: int = 3000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed objective.
    Returns (best_vector, best_score) with ``score = 2·n·max(conv)``.
    """
    # Initialise logits from the starting point (avoid log(0))
    a = a0.copy()
    eps_small = 1e-20
    z = np.log(np.clip(a, eps_small, None))

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline and it < 10**7:
        it += 1

        # ----- current iterate -------------------------------------------------
        a = _softmax(z)                       # always on the simplex
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # ----- smoothing parameter --------------------------------------------
        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]

        # ----- smooth max & weights -------------------------------------------
        _, w = _smooth_max_and_weights(conv, alpha)

        # ----- gradient w.r.t. a -----------------------------------------------
        grad_a = _grad_smooth_max_from_w(a, w, n, fft_len)

        # ----- chain rule to logits (softmax Jacobian) -------------------------
        s = np.dot(a, grad_a)
        grad_z = a * (grad_a - s)

        # ----- Adam update -----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= eta * m_hat / (np.sqrt(v_hat) + epsilon)

        # ----- schedule shift --------------------------------------------------
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent with the exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7):
    """Polishing phase using the exact sub‑gradient of the max‑convolution."""
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        max_val = conv.max()
        thresh = max_val * (1.0 - 1e-12)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum and step size
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
