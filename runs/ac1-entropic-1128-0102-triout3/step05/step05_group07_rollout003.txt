"""
Propose a low‑score step‑function sequence.

The search combines several ideas that proved useful for the previous
run:

*   A multi‑scale schedule (256 → … → 4096) with linear‑interpolation
    of the current best solution when moving to a larger size.
*   A diverse pool of starting vectors (uniform, triangular, Hann,
    Dirichlet draws, sparse spikes and the historic best `height_sequence_1`).
*   Two complementary gradient phases:
    – Adam on a smooth‑max surrogate of the L∞‑norm.
    – Exponential‑gradient (mirror‑descent) on the exact max‑convolution.
*   A cheap stochastic “perturb‑and‑refine’’ loop that repeatedly
    jitters the best vector and re‑optimises it, allowing the search to
    escape shallow local minima.
*   Careful time‑budget handling – the remaining time is split between
    candidates and the iteration budget is chosen from an empirical
    per‑iteration cost model.

All updates respect the probability simplex (non‑negative, sum = 1) by
projecting after each step.  The routine stops before the given budget
and returns the best sequence found as a plain list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return max(conv(a,a)) and the full linear convolution."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k w.r.t. a.
    Implements g = 2·(rev(a) * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)
    # keep only the part that aligns with a (length = n)
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.07,
                   temp_end: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best vector seen and its raw (2·n·max_conv) score.
    """
    n = a.shape[0]
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # best observed score (2·n·max_conv)
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights over convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        g = _gradient_from_weights(a, w, fft_len_grad)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser that uses the exact
    ℓ∞‑norm sub‑gradient.  The update is
        a_i ← a_i·exp(−η·g_i)  (followed by simplex projection).
    Returns the best vector seen and its raw score.
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # active set – indices whose convolution is within eps of the maximum
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric decay of the step size
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)   # avoid overflow in exp
        a = a * np.exp(upd)

        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score


def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Force non‑negative and normalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _upsample(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, v)
    return _as_simplex(up)


def _perturb(v: np.ndarray, magnitude: float = 0.02) -> np.ndarray:
    """
    Small random redistribution of mass.
    Each coordinate receives an additive Gaussian noise of scale `magnitude`,
    then the vector is projected back onto the simplex.
    """
    n = v.shape[0]
    noise = (np.random.rand(n) - 0.5) * magnitude
    v2 = v + noise
    return _as_simplex(v2)


def _raw_score(a: np.ndarray) -> float:
    """Raw score = 2·n·max(conv(a,a)) (sum is 1)."""
    n = a.shape[0]
    max_val, _ = _conv_max(a)
    return 2.0 * n * max_val
