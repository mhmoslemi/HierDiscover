def gradient_smooth_max(a, beta):
    n = a.size
    conv = conv_self(a)   # length L = 2n-1
    # compute softmax probabilities
    expb = np.exp(beta*conv - np.max(beta*conv)) # for numerical stability
    Z = expb.sum()
    p = expb / Z
    smooth_max = np.dot(conv, p)  # or sum conv*expb/Z
    w = p * (1 + beta * (conv - smooth_max))
    # compute convolution of a and w using FFT
    L = conv.shape[0]
    m = n + L - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(w, n=fft_len)
    conv_ab = np.fft.irfft(Af * Bf, n=fft_len)[:m]
    # sliding sum of length n across conv_ab
    # sliding_sum[k] = sum_{i=k}^{k+n-1} conv_ab[i]
    cumsum = np.cumsum(conv_ab)
    # sliding sum vector size m - n + 1 = (n+L-1) - n +1 = L = 2n-1
    # we need for k 0..n-1: sliding = cumsum[k+n-1] - (cumsum[k-1] if k>0 else 0)
    g = np.empty(n)
    g[0] = cumsum[n-1]
    g[1:] = cumsum[np.arange(1, n) + n - 1] - cumsum[:n-1]
    g = 2.0 * g
    return smooth_max, g

Low‑max self‑convolution optimizer.

The algorithm proceeds in stages for a series of lengths (doubling schedule):
1. Warm‑start from a parametric family (product‑power, Gaussian, Beta,
   arcsine, exponential) – a cheap stochastic search finds a good shape.
2. Projected sub‑gradient descent with momentum on the exact ∞‑norm.
3. Smooth‑max gradient descent (soft‑max approximation of the max) with an
   increasing β schedule – this refines the distribution beyond the
   active‑index sub‑gradient.
4. Very cheap mass‑transfer simulated‑annealing polishing.
All stages respect the global 1000 s budget and the solution is kept on the
probability simplex (∑a_i = 1).  The returned list is plain Python floats,
the evaluator will clamp/normalise automatically.
"""

import math

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def score_of(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(a*a) for a normalized (∑a=1) sequence."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(seq) to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families (quick warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term == 0] = 1e-12
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    """PDF of the arcsine law on the discrete grid."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = 1.0 / np.sqrt(term)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - centre))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (with momentum)
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 1.0
    it = 0
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        idx = conv.argmax()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # sub‑gradient of the active entry
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)
        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)                # scaling does not affect direction after projection

        # momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = project_simplex(a)

        # occasional isotropic noise to escape plateaus
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) gradient descent
# ----------------------------------------------------------------------


def gradient_smooth_max(a: np.ndarray, beta: float):
    """
    Returns (smooth_max, grad) where smooth_max = soft‑max_β(conv_self(a))
    and grad = ∂ smooth_max / ∂ a (exact, using FFT for speed).
    """
    n = a.shape[0]
    conv = conv_self(a)                     # length L = 2n‑1

    # soft‑max of β*conv (numerically stable)
    z = beta * conv
    z_max = np.max(z)
    expz = np.exp(z - z_max)
    Z = expz.sum()
    p = expz / Z
    smooth = np.dot(conv, p)                # soft‑max value

    # weight vector for the gradient
    w = p * (1.0 + beta * (conv - smooth))

    # convolution a * w (full) – use FFT
    L = conv.shape[0]
    m = n + L - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(w, n=fft_len)
    conv_ab = np.fft.irfft(Af * Bf, n=fft_len)[:m]

    # gradient = 2 * sliding sum of length n over conv_ab
    csum = np.cumsum(conv_ab)
    grad = np.empty(n, dtype=np.float64)
    grad[0] = csum[n - 1]
    grad[1:] = csum[np.arange(1, n) + n - 1] - csum[:n - 1]
    grad *= 2.0
    return smooth, grad


def smooth_descent(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Refine a using a smooth approximation of the max.
    Uses a schedule of β values that grow, each with a few hundred iterations.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    total_it = 0

    for beta in betas:
        # allocate a modest slice for each β (but respect global deadline)
        phase_deadline = time.time() + min(6.0, max(1.0, (deadline - time.time()) * 0.12))
        it = 0
        while time.time() < phase_deadline:
            it += 1
            total_it += 1
            _, grad = gradient_smooth_max(a, beta)

            # step size – simple 1/√it schedule, normalised direction
            norm = np.linalg.norm(grad) + 1e-16
            direction = grad / norm
            step = 0.8 / math.sqrt(total_it)

            a = a - step * direction
            a = project_simplex(a)

            cur_sc = score_of(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] smooth ({beta}) improve n={n} it={total_it} score={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-5,
              steps: int = 80000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2  # move up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of heights (sum normalisation is handled by the evaluator).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_score = float("inf")
    best_seq = None
    global_a = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter = starter / s
            global_a = starter
    except Exception:
        pass

    # Length schedule – start modest, double each step.
    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)   # avoid systematic bias when we stop early

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a0 (warm start / up‑sample)
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Allocate a slice of remaining time for this length
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # later (larger) lengths receive a slightly larger share
        slice_frac = 0.12 / max(1, (len(lengths) - idx))
        slice_deadline = time.time() + max(8.0, remaining * slice_frac)

        # ------------------------------------------------------------------
        # QUICK SHAPE SEARCH (few seconds)
        # ------------------------------------------------------------------
        shape_deadline = time.time() + min(2.0, (slice_deadline - time.time()) * 0.20)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2, 3, 4])
            if kind == 0:  # product‑power
                p = np.random.uniform(-1.8, 2.2)
                cand = shape_product_power(n, p)
            elif kind == 1:  # Gaussian
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 2:  # Beta
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            elif kind == 3:  # Arcsine
                cand = shape_arcsine(n)
            else:  # Exponential
                lam = np.random.uniform(0.001, 0.02) * n
                cand = shape_exponential(n, lam)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}")

        # ------------------------------------------------------------------
        # PROJECTED SUB‑GRADIENT DESCENT (with momentum)
        # ------------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = subgradient_descent(local_best_a, n, grad_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={sc_opt:.6f}")

        # ------------------------------------------------------------------
        # SMOOTH‑MAX REFINEMENT
        # ------------------------------------------------------------------
        if time.time() < deadline:
            smooth_deadline = time.time() + min(8.0, (deadline - time.time()) * 0.25)
            a_opt2, sc_opt2 = smooth_descent(a_opt, n, smooth_deadline, start_time)
            if sc_opt2 < best_score:
                best_score = sc_opt2
                best_seq = a_opt2.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after smooth n={n} score={sc_opt2:.6f}")
        else:
            a_opt2 = a_opt

        # ------------------------------------------------------------------
        # CHEAP ANNEALING POLISHING (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.30)
            a_ref, sc_ref = annealing(
                a_opt2,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=80000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}")

        # --------------------------------------------------------------
        # Keep the best vector (upsampled) as the starter for the next length
        # --------------------------------------------------------------
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (the evaluator will handle clamping/normalisation)
    return [float(x) for x in best_seq]
