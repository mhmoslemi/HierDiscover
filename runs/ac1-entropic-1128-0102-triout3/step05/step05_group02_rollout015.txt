    v = np.zeros_like(a)

    it = 0

        max_val = conv.max()
        idx = conv.argmax()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time()-start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # gradient for active conv idx
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)
        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

"""
Propose a step‑function with a low self‑convolution supremum.

The algorithm works on the probability simplex (∑a_i = 1) because the
objective 2·n·max(conv(a,a)) / (∑a_i)^2 is scale‑invariant.
It improves on the reference implementation by

* adding an exponential‑decay shape family (the theoretical analysis
  suggests a mild decay is close to optimal),
* using a momentum‑accelerated projected sub‑gradient method,
* performing a cheap “exponent‑transform” local search,
* allocating a larger share of the time budget to the final (largest)
  length and running a longer annealing phase.

The search proceeds on a schedule of increasing lengths
(256 → … → 4096).  At each length we warm‑start from the up‑sampled
best vector, sample many simple parametric shapes, run the momentum
sub‑gradient descent, try a few exponent‑re‑weightings,
and finally polish with a mass‑transfer simulated‑annealing loop.
The best sequence found is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simple linear algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT (real‑to‑real)."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def score_of(a: np.ndarray) -> float:
    """Exact objective value for a normalised to sum = 1."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Parametric shape families (warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^p  (p>0 centre‑heavy, p<0 edge‑heavy)."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term = np.where(term == 0, eps, term)
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ given in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the unit interval."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_exponential(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)  (rate > 0).  Slight decay is often beneficial."""
    i = np.arange(n)
    a = np.exp(-rate * i)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_linear_decrease(n: int) -> np.ndarray:
    """Deterministic linear decay from 1 to 0."""
    a = np.linspace(1.0, 0.0, n) + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Momentum projected sub‑gradient descent (exact ∞‑norm)
# ----------------------------------------------------------------------


def subgradient_descent_momentum(a0: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method with simple momentum.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    beta = 0.9               # momentum coefficient
    v = np.zeros_like(a)     # velocity
    step0 = 0.8              # base step size
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        idx = conv.argmax()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # ---- sub‑gradient of the active convolution entry ----
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        # normalise direction
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        # momentum update
        v = beta * v + (1.0 - beta) * direction

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * v)

        # occasional isotropic jitter to escape plateaus
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-6,
              steps: int = 150000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing: move a small ε from a source index to a destination index.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the runner
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # small safety margin for housekeeping
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from the previous run (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter /= s
    except Exception:
        starter = None

    # Length schedule – we finish with the largest length we can afford.
    lengths = [256, 512, 1024, 2048, 4096]
    random.shuffle(lengths)   # avoid ordering bias if we stop early

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0 (warm start / up‑sample)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # allocate a slice of the remaining time for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # larger lengths receive a larger share of the remaining budget
        if idx == len(lengths) - 1:
            slice_deadline = time.time() + remaining   # all remaining time for the final length
        else:
            # give about 8–12 % of the remaining time to early lengths
            fraction = 0.12 / (len(lengths) - idx)
            slice_deadline = time.time() + max(6.0, remaining * fraction)

        # --------------------------------------------------------------
        # quick stochastic search over a few simple parametric families
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(3.0, (slice_deadline - time.time()) * 0.25)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = random.choice(['prod', 'gauss', 'beta', 'exp', 'lin'])
            if kind == 'prod':
                p = random.uniform(-2.0, 2.5)
                cand = shape_product_power(n, p)
            elif kind == 'gauss':
                sigma = random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 'beta':
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            elif kind == 'exp':
                # rate roughly O(1/n); sample a modest range
                rate = random.uniform(0.5 / n, 5.0 / n)
                cand = shape_exponential(n, rate)
            else:   # linear decreasing
                cand = shape_linear_decrease(n)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}")

        # --------------------------------------------------------------
        # momentum sub‑gradient descent (exact ∞‑norm)
        # --------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = subgradient_descent_momentum(local_best_a, n, grad_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={sc_opt:.6f}")

        # --------------------------------------------------------------
        # exponent‑transform local refinement (quick power‑re‑weighting)
        # --------------------------------------------------------------
        exp_deadline = time.time() + min(2.0, (deadline - time.time()) * 0.10)
        a_refine = a_opt.copy()
        sc_refine = sc_opt
        while time.time() < exp_deadline:
            p = 10.0 ** random.uniform(-0.5, 0.5)   # p ∈ [≈0.32, ≈3.16]
            cand = a_refine ** p
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            sc = score_of(cand)
            if sc < sc_refine:
                sc_refine = sc
                a_refine = cand
                print(f"[{time.time() - start_time:6.2f}s] exp‑transform improve n={n} p={p:.3f} score={sc:.6f}")

        # --------------------------------------------------------------
        # cheap annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.20)
            a_ref, sc_ref = annealing(
                a_refine,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=200000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}")

        # keep the best vector for the next up‑sample step
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list; the evaluator will clamp/normalise as needed
    return [float(x) for x in best_seq]
