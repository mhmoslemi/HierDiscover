"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence().

Algorithm
---------
1. **Linear‑programming phase (n = 800).**  
   Starting from a simplex point, repeatedly
   * compute M = max(a * a),
   * solve the LP  max Σ b_j  s.t. a * b ≤ M  and  b ≥ 0,
   * mix the current a with the normalised LP solution (convex combination).
   The LP is tiny (≈800 × 800) and drives the shape toward a lower peak.

2. **Progressive up‑sampling + soft‑max gradient descent.**  
   The LP‑output is linearly interpolated to larger lengths
   (2400 → 4800 → 9600).  For each length a projected Nesterov‑gradient
   descent on a *soft‑max* approximation of the max‑convolution is run.
   The temperature is annealed to a very small value, giving an accurate
   gradient of the true max.  Several random restarts are performed only
   for the final length.

3. **Final cheap simulated‑annealing walk** to fine‑tune the best
   candidate.

All optimisation respects the 1000 s budget and returns a plain Python
list of floats (sum normalised to 1, which is scale‑invariant for the
evaluation function).
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper 1: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Convolution of the reversed sequence a_rev with a weight vector w.
    Returns the full linear convolution (length = len(a_rev)+len(w)-1).
    """
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4: gradient of a *soft‑max* of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of  Σ_k w_k(conv_k) * conv_k  with  w_k = exp(conv_k / T) / Σ.
    The factor 2 comes from the derivative of aᵀJ_k a.
    """
    maxc = conv.max()
    # stable soft‑max
    scaled = (conv - maxc) / temperature
    exps = np.exp(scaled)
    w = exps / exps.sum()

    # weighted gradient: 2 * (rev(a) * w) restricted to the central n entries
    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)   # length = n + (2n-1) - 1 = 3n-2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]   # central block corresponds to i‑range
    return grad


# ----------------------------------------------------------------------
# Optimiser: projected (Nesterov) gradient descent with soft‑max gradient
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    T_start: float = 2.0e-2,
                    T_end: float = 5.0e-4,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected Nesterov GD on the smooth max‑convolution.
    Returns the best point found (as a simplex vector) and its scaled score
    (2·n·max(conv)).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve n={n} "
                  f"iter={it} score={cur_score:.6f}")

        # exponential temperature decay
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: linear convolution matrix (Toeplitz) for a * b (full length)
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix T such that T @ b == a * b (full conv).
    Each diagonal offset k (0 ≤ k < n) contains the constant value a[k].
    """
    n = a.shape[0]
    # data[i] is a constant vector of length n for diagonal offset i
    data = [np.full(n, a_i, dtype=np.float64) for a_i in a]
    offsets = np.arange(n, dtype=int)
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return T

# Linear‑programming stage (n ≈ 800)
# ----------------------------------------------------------------------
def _run_lp_stage(n: int, deadline: float, start_time: float,

    problem statement.  Returns the best sequence (still on the simplex).
    """

    else:
        a = np.random.rand(n)
        a = _project_simplex(a)

    best_a = a.copy()
    best_M = _conv_self(a).max()
    iteration = 0

        bounds = [(0, None)] * n
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = opt.linprog(c, A_ub=T, b_ub=ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        if not res.success:
            # fall back to plain gradient descent if LP fails
            print(f"[{time.time() - start_time:6.2f}s] LP failed at iter {iteration}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        # normalise b to a simplex vector (sum = 1)
        g = b / sum_b

        # convex combination with a
        a = (1.0 - t_mix) * a + t_mix * g

        cur_M = _conv_self(a).max()
        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve iter={iteration} M={cur_M:.6f}")

        # simple stop criterion
        if iteration >= 250:
            # enough progress for this small problem
            break

    return best_a


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    # re‑normalise to the simplex
    up = _project_simplex(up)
    return up


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Metropolis walk that moves a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1    # move up to 10 % of a[i]
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: "
                  f"n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget, performs a small LP phase,
    progressive up‑sampling, soft‑max gradient descent, and finishes with a
    short annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness itself
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    try:
        starter_full = np.asarray(height_sequence_1, dtype=float)
        if starter_full.ndim == 1 and starter_full.size > 0:
            starter_full = _project_simplex(starter_full)
        else:
            starter_full = None
    except Exception:
        starter_full = None

    # ------------------------------------------------------------------
    # Phase 1 – tiny LP optimisation (n = 800)
    # ------------------------------------------------------------------
    n_lp = 800
    total_time = deadline - start_time
    lp_budget = min(180.0, total_time * 0.22)          # roughly 20 % of the total budget
    lp_deadline = start_time + lp_budget

    init_lp = None
    if starter_full is not None:
        init_lp = _upsample(starter_full, n_lp)

    best_a = _run_lp_stage(n_lp, lp_deadline, start_time, init_seq=init_lp)
    best_score = 2.0 * n_lp * _conv_self(best_a).max()
    print(f"[{time.time() - start_time:6.2f}s] LP phase finished: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Remaining time is split between gradient descent stages and final annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        # fallback – return whatever we have
        return [float(x) for x in best_a.tolist()]

    anneal_budget = max(30.0, remaining * 0.15)      # reserve ~15 % for annealing
    grad_budget = remaining - anneal_budget

    # stage definitions (length, share of gradient budget)
    stage_lengths = [800, 2400, 4800, 9600]          # progressive enlargement
    total_len = sum(stage_lengths)

    # run GD for each stage, starting from the LP result
    for n in stage_lengths:
        if n == n_lp:
            # we already have an answer for n=800; still give it a quick GD polish
            stage_time = max(5.0, grad_budget * (n / total_len))
            stage_deadline = time.time() + stage_time
            a_opt, sc_opt = _run_softmax_gd(
                best_a, n, stage_deadline, start_time,
                step0=0.6, beta=0.9,
                T_start=2.0e-2, T_end=5.0e-4
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={sc_opt:.6f}")
            continue

        # up‑sample to the required size
        a0 = _upsample(best_a, n)

        # allocate time proportionally to length
        stage_frac = n / total_len
        stage_time = max(8.0, grad_budget * stage_frac)
        stage_deadline = time.time() + stage_time
        if stage_deadline > deadline - anneal_budget:
            stage_deadline = deadline - anneal_budget

        # for the biggest stage we launch a few random restarts
        if n == stage_lengths[-1]:
            # keep the best over a few short runs
            best_local_a = a0.copy()
            best_local_score = 2.0 * n * _conv_self(best_local_a).max()
            n_restarts = 3
            for r in range(n_restarts):
                # small random perturbation of the current best
                pert = best_local_a + 0.02 * np.random.randn(n)
                pert = _project_simplex(np.maximum(pert, 0.0))
                sub_deadline = time.time() + (stage_deadline - time.time()) / (n_restarts - r)
                a_opt, sc_opt = _run_softmax_gd(
                    pert, n, sub_deadline, start_time,
                    step0=0.6, beta=0.9,
                    T_start=2.0e-2, T_end=5.0e-5
                )
                if sc_opt < best_local_score:
                    best_local_score = sc_opt
                    best_local_a = a_opt.copy()
                    print(f"[{time.time() - start_time:6.2f}s] restart {r+1}/{n_restarts} "
                          f"improve: score={sc_opt:.6f}")
            a_opt, sc_opt = best_local_a, best_local_score
        else:
            a_opt, sc_opt = _run_softmax_gd(
                a0, n, stage_deadline, start_time,
                step0=0.6, beta=0.9,
                T_start=2.0e-2, T_end=5.0e-4
            )

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Final annealing phase (works on the final length)
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - anneal_budget:
        n_final = best_a.shape[0]
        a_ref, score_ref = _run_annealing(
            best_a, n_final,
            deadline - anneal_budget, start_time,
            best_score, best_a,
            start_temp=5e-2, end_temp=1e-5,
            steps=500_000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_a = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (ensure we return a proper simplex point)
    # ------------------------------------------------------------------
    best_a = _project_simplex(best_a.astype(float))

    # return a plain Python list for the evaluation harness
    return [float(x) for x in best_a.tolist()]
