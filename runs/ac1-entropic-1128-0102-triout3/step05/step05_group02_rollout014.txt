def smooth_gradient_descent(a, max_iter, deadline, start_time):
    best_a = a.copy()
    best_score = score_of(a)
    alpha = 0.5
    step = 0.5
    beta = 0.9 # momentum
    vel = np.zeros_like(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline: break
        conv = conv_self(a)
        # compute p softmax
        maxc = conv.max()
        # use alpha smoothing param
        exp_vals = np.exp(alpha * (conv - maxc))
        p = exp_vals / exp_vals.sum()
        # gradient
        # compute correlation a ⋆ p (length n) via convolution a * p_rev
        # p_rev = p[::-1]
        # we need correlation result for i in [0,n-1]: sum_j a_j * p_{i+j}
        # compute conv_corr = np.convolve(a, p[::-1]) # length n + 2n-2 = 3n-2
        corr_full = np.convolve(a, p[::-1])
        start = len(p)-1
        grad = 2 * alpha * corr_full[start:start+n] # shape (n,)
        # momentum update
        vel = beta * vel + (1 - beta) * grad
        # step size schedule
        step_size = step / np.sqrt(it)
        a_new = a - step_size * vel
        a_new = project_simplex(a_new)
        a = a_new
        # compute score
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] smooth GD improve it={it} score={sc:.6f}")
        # adjust alpha gradually
        if it % 200 == 0:
            alpha = min(alpha * 1.2, 20.0) # increase smoothness
    return best_a, best_score

def cross_corr_fft(p, a):
    n = a.shape[0]
    m = p.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())
    Af = np.fft.rfft(a, n=size)
    Pf = np.fft.rfft(p[::-1], n=size)
    conv = np.fft.irfft(Af * Pf, n=size)[:n+m-1]
    start = m - 1
    return conv[start:start+n]

    n = a.shape[0]
    best_a = a.copy()
    best_score = score_of(a)
    # parameters
    step0 = 0.5
    alpha = 1.0  # smoothing parameter
    beta = 0.9   # momentum factor
    vel = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        maxc = conv.max()

        exp_vals = np.exp(alpha * (conv - maxc))

"""
Propose a low‑maximum self‑convolution step‑function.

The optimizer works on the probability simplex (∑a_i = 1) because the
objective 2·n·max(a∗a) is scale‑invariant.  It combines three ideas
that are not used together in the reference implementation:

1. **Richer parametric warm‑starts** – besides the simple power‑product,
   truncated Gaussian and beta families we also try edge‑Gaussian,
   double‑exponential and random mixtures of an edge shape with the
   uniform distribution.  A quick random sweep (a few seconds) finds a
   promising seed for each chosen length.

2. **Smoothed gradient descent on the simplex** – instead of a pure
   sub‑gradient (which only looks at the single most active convolution
   entry) we minimise the log‑sum‑exp of the convolution:
      F(a)=log∑_k exp(α·(a∗a)_k) .
   For α→∞ this becomes the original ∞‑norm, while for modest α the
   objective is smooth and its gradient can be obtained analytically.
   We use a diminishing step‑size, Euclidean projection onto the simplex
   and a simple momentum term.  α is increased geometrically during the
   run so that the optimisation first moves in a smooth landscape and
   then sharpens around the true maximum.

3. **Cheap mass‑transfer annealing** – a final Metropolis‑style polishing
   phase that repeatedly moves a tiny ε from one index to another with a
   geometric temperature schedule, accepting only moves that do not
   increase the exact objective.

The search proceeds over a schedule of lengths (starting small,
progressively up‑sampling) and respects a hard 1000 s wall‑time.
The best vector found (converted to a plain Python list) is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen, but fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Af * Af, n=size)[:m]
    return conv


def cross_corr(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ p)[i] = Σ_j a_j·p_{i+j},
    where a has length n and p has length 2n‑1.
    Returned array has length n (indices 0 … n‑1).
    """
    n = a.shape[0]
    m = p.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())
    Af = np.fft.rfft(a, n=size)
    Pf = np.fft.rfft(p[::-1], n=size)      # reverse p for correlation
    conv = np.fft.irfft(Af * Pf, n=size)[:n + m - 1]
    start = m - 1
    return conv[start:start + n]


def score_of(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(a*a) for a normalized (sum=1) vector."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families for warm‑starts
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i+0.5)*(n-i-0.5) raised to power p.
    p>0 → centre‑heavy, p<0 → edge‑heavy.
    """
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term <= 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ given in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_edge_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two symmetric edge‑centred Gaussian bumps."""
    i = np.arange(n)
    left = np.exp(-0.5 * (i / sigma) ** 2)
    right = np.exp(-0.5 * ((n - 1 - i) / sigma) ** 2)
    a = left + right
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_exponential_decay(n: int, tau: float) -> np.ndarray:
    """Two symmetric exponential decays from the ends."""
    i = np.arange(n)
    left = np.exp(-i / tau)
    right = np.exp(-(n - 1 - i) / tau)
    a = left + right
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smoothed_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(a*a) using the smooth surrogate
    F(a)=log∑exp(α·conv) with a growing α.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.7           # base step size
    alpha = 0.8           # initial smoothness
    beta = 0.9            # momentum coefficient
    vel = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        # --- soft‑max weights for the current α ---
        maxc = conv.max()
        exp_vals = np.exp(alpha * (conv - maxc))   # shift for numerical stability
        p = exp_vals / exp_vals.sum()             # p_k are the soft‑max probabilities

        # --- gradient of the smooth objective ---
        # ∇F = 2·α·(a ⋆ p)   (cross‑correlation)
        grad = 2.0 * alpha * cross_corr(a, p)

        # momentum + diminishing step
        vel = beta * vel + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = project_simplex(a - step * vel)

        sc = score_of(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth GD improve n={n} it={it} score={sc:.6f}")

        # increase smoothness gradually – after many iterations we approach the true max
        if it % 250 == 0:
            alpha = min(alpha * 1.25, 20.0)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Very cheap mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-6,
              steps: int = 40000) -> tuple[np.ndarray, float]:
    """Simple Metropolis style moves that transfer ε from one index to another."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2   # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
