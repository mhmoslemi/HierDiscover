        arr = np.maximum(arr, 0)
        arr = arr / arr.sum()
        return arr

    for n in stage_ns:
        if time.time() >= deadline:
            break
        # compute allowed time for this stage, allocate proportionally
        remaining = deadline - time.time()
        # allocate for stage proportionally to n (larger stage gets more)

        v = np.zeros_like(x)

        lr = 0.05

def _fft_convolve(a, b):
    n = a.shape[0] + b.shape[0] - 1
    fft_len = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:n]
    return conv

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.

Algorithm
---------
1. **Parameterisation** – we work with logits `x` and obtain the
   candidate heights by a soft‑max (`a = softmax(x)`).  This guarantees
   non‑negativity and unit L1‑norm.

2. **Smoothed max objective** – the true objective is
   `2·n·max(a⋆a)`.  Instead of the non‑smooth max we minimise a
   temperature‑controlled soft‑max:
        fβ(a) = 2·n·log‑sum‑exp(β·(a⋆a))/β .
   The gradient of the smoothed term can be written analytically and
   computed by a pair of FFT‑based convolutions.

3. **Adam optimiser** – an Adam update is performed on the logits `x`.
   The temperature `β` is increased linearly during optimisation,
   so the algorithm first smooths the landscape and then focuses on
   the true max.

4. **Progressive up‑sampling** – optimisation is run on a sequence of
   increasing lengths `[2000, 5000, 12000]`.  After each stage the best
   sequence is linearly interpolated to the next size, preserving the
   total mass, and the optimisation continues.

5. **Final simulated annealing** – a cheap mass‑transfer walk refines
   the best solution found.

The routine respects the given time budget (default 1000 s) and prints
progress information.  The returned sequence is a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (O(N log N))
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    fft_len = 1 << ((n - 1).bit_length())       # next power of two
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Compute a * a (linear convolution, length 2·len(a)‑1)."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Simplex projection (used only for a fallback uniform case)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a simplex‑vector to length new_n."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed‑max objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    n: int,
    stage_deadline: float,
    start_time: float,
    max_iters: int = 200_000,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
    lr: float = 0.05,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n starting from a_init (already unit‑sum).
    Returns the best sequence seen and its exact score = 2·n·max(conv).
    """
    # logits for the soft‑max parametrisation
    eps = 1e-12
    x = np.log(a_init + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = a_init.copy()
    best_score = float("inf")
    it = 0

    while time.time() < stage_deadline and it < max_iters:
        it += 1

        # soft‑max (stable)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()                     # size n, sum = 1

        # self‑convolution
        conv = _conv_self(a)                        # length 2·n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] n={n:5d} iter={it:7d} "
                  f"score={cur_score:.6f}")

        # ----- smoothed max -------------------------------------------------
        # linear schedule for the temperature β
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iters // 2))

        # softmax weights over the convolution entries
        # shift by max_conv for numerical stability
        w_raw = np.exp(beta * (conv - max_conv))
        w = w_raw / w_raw.sum()                     # length 2·n‑1, sum = 1

        # gradient of the smoothed max w.r.t. a
        a_rev = a[::-1]
        grad_conv = _fft_convolve(a_rev, w)         # length 3·n‑2
        # slice the central n entries (indices n‑1 … 2·n‑2 inclusive)
        g_a = 2.0 * grad_conv[n - 1 : n - 1 + n]

        # gradient w.r.t. logits x (using Jacobian of soft‑max)
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # ----- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing refinement
# ----------------------------------------------------------------------
def _anneal_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_seq: np.ndarray,
    cur_best_score: float,
    steps: int = 300_000,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """Move a tiny ε of mass between two positions (Metropolis walk)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.08  # move up to 8 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} "
                      f"iter={it:7d} score={score:.6f}")

    return best_seq, best_score
