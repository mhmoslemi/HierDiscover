def smooth_descent(a0, n, slice_deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)
    it = 0
    start = time.time()
    while time.time() < slice_deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth improve n={n} it={it} score={cur_sc:.6f}")

        # temperature schedule
        total = slice_deadline - start
        elapsed = time.time() - start
        # avoid division by zero
        if total <= 0:
            temp = 1e-4
        else:
            temp = 0.5 * (1e-4 / 0.5) ** (elapsed / total)  # exponential from 0.5 to 1e-4
        # compute softmax p
        # subtract max for numerical stability
        conv_max = conv.max()
        p = np.exp((conv - conv_max) / temp)
        p_sum = p.sum()
        if p_sum == 0 or np.isnan(p_sum):
            p = np.full_like(p, 1.0 / p.size)
        else:
            p = p / p_sum

        # gradient of smooth max
        p_rev = p[::-1]
        L = 1 << ((3*n - 2 - 1).bit_length())
        A = np.fft.rfft(a, n=L)
        P = np.fft.rfft(p_rev, n=L)
        conv_rev = np.fft.irfft(A * P, n=L)[:3*n - 2]
        idx = (2*n - 2) - np.arange(n)
        grad_f = 2.0 * conv_rev[idx]   # derivative of smooth max
        grad = 2.0 * n * grad_f         # derivative of objective O

        # normalized direction
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm
        step = 0.8 / math.sqrt(it)   # adjust step0 maybe

            # small noise
            a += 1e-5 * np.random.randn(n)
            a = np.maximum(a, 0)
            a /= a.sum()

    return best_a, best_sc

"""
Propose a low‑maximum self‑convolution step‑function.

The optimiser works on the probability simplex (∑a_i = 1) because the
objective 2·n·max(a*a) is scale‑invariant.
The new pipeline differs from the previous implementation in two ways:

1. **Smoothed‑max gradient descent** – the exact ∞‑norm is replaced by a
   temperature‑controlled soft‑max (T·log∑exp(conv/T)).  Its gradient is
   computed analytically via a second convolution (a * softmax‑weights)
   using FFTs, and the step direction is projected onto the simplex.
   The temperature is annealed exponentially, which gives a smooth path
   from a convex surrogate to the true max and helps escape local minima.

2. **Multi‑scale schedule with richer warm‑starts** – we still start from
   simple parametric families (power‑product, Gaussian, Beta) but the
   parameter ranges are widened (including heavy edge‑heavy shapes).  The
   best vector from a shorter length is up‑sampled for the next length.

After the smoothed descent we polish the solution with a cheap
mass‑transfer annealing (the same routine as before) and keep the best
vector seen across all lengths.  The whole search respects a 1000 s
budget and prints progress messages.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Exact self‑convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def score_of(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(a*a) for a simplex‑normalised vector."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Shape families for quick random warm‑starts (wider parameter ranges)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)*(n-i-0.5) raised to power p (p can be negative)."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    term[term <= 0] = 1e-12
    a = term ** p
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian (σ given in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid; α,β may be <1 (edge‑heavy)."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a

def smooth_max_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    T_start: float = 0.5,
    T_end: float = 1e-4,
    step0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed max (T·log‑sum‑exp) of the
    self‑convolution.  The temperature is annealed from T_start to T_end.
    Returns the best vector seen and its exact (non‑smoothed) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    it = 0
    descent_start = time.time()
    total_allowed = max(1e-9, deadline - descent_start)

    while time.time() < deadline:
        it += 1

        # exact convolution (used also for the soft‑max)
        conv = conv_self(a)                     # length 2n‑1
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] smooth improve n={n} it={it} score={cur_sc:.6f}"
            )

        # ---- temperature schedule (exponential) ----
        elapsed = time.time() - descent_start
        temp = T_start * (T_end / T_start) ** (elapsed / total_allowed)

        # ---- soft‑max weights (stable version) ----
        conv_max = max_val
        w = np.exp((conv - conv_max) / temp)   # max weight = 1
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum                         # soft‑max (probability vector)

        # ---- gradient of smooth max ----
        # grad_f_i = 2 * Σ_j a_j * p_{i+j}
        p_rev = p[::-1]
        L = 1 << ((3 * n - 2 - 1).bit_length())   # next power of two ≥ 3n‑2
        A = np.fft.rfft(a, n=L)
        P = np.fft.rfft(p_rev, n=L)
        conv_rev = np.fft.irfft(A * P, n=L)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)          # indices to pick from conv_rev
        grad_f = 2.0 * conv_rev[idx]              # d(smooth_max)/da_i

        # objective is 2·n·smooth_max, so total gradient:
        grad = 2.0 * n * grad_f

        # ---- normalized step ----
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm
        step = step0 / math.sqrt(it)

        a = project_simplex(a - step * direction)

        # occasional isotropic jitter to escape plateaus
        if it % 400 == 0:
            a += 1e-5 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (polishing – unchanged)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """Move a small ε from a source index to a destination index."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2  # up to 20 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns a plain Python list of heights (normalised, sum=1).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # safety margin for the environment
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from previous run (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter = starter / s
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – start modest, grow progressively.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    random.shuffle(lengths)  # avoid bias if we run out of time early

    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0 (warm start)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # allocate a slice of remaining time for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # later (larger) lengths receive a slightly larger share
        fraction = 0.12 / (len(lengths) - idx)
        slice_deadline = time.time() + max(8.0, remaining * fraction)

        # --------------------------------------------------------------
        # quick stochastic search over simple parametric families
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(2.0, (slice_deadline - time.time()) * 0.15)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice([0, 1, 2])
            if kind == 0:  # product‑power family
                p = np.random.uniform(-2.0, 3.0)
                cand = shape_product_power(n, p)
            elif kind == 1:  # Gaussian family
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            else:  # Beta family – allow heavy edge shapes
                alpha = np.random.uniform(0.02, 3.0)
                beta_par = np.random.uniform(0.02, 3.0)
                cand = shape_beta(n, alpha, beta_par)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(
                    f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}"
                )

        # --------------------------------------------------------------
        # smoothed‑max gradient descent (core optimisation)
        # --------------------------------------------------------------
        smooth_deadline = time.time() + (slice_deadline - time.time()) * 0.60
        a_opt, sc_opt = smooth_max_descent(
            local_best_a, n, smooth_deadline, start_time, T_start=0.5, T_end=1e-4, step0=0.8
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] global improve after smooth n={n} score={sc_opt:.6f}"
            )

        # --------------------------------------------------------------
        # cheap annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(4.0, (deadline - time.time()) * 0.25)
            a_ref, sc_ref = annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-5,
                steps=80000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}"
                )

        # keep the best vector for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – never should be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return plain Python list (the evaluator will clip/normalise as needed)
    return [float(x) for x in best_seq]
