```python
def solve_lp_for_b(f, M):
    n = len(f)
    # Build constraints matrix T_f shape (2n-1,n)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)  # inclusive
        # i ranges from i_start to i_end inclusive
        for i in range(i_start, i_end+1):
            j = k - i
            # coefficient f[i] for b[j]
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # Build sparse matrix
    import scipy.sparse as sp
    T = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Solve LP: maximize sum(b) => minimize -sum(b)
    c = np.full(n, -1.0)
    # bounds: b >= 0
    bounds = [(0, None)] * n
    # inequality: T @ b <= M
    # Use scipy.optimize.linprog
    from scipy.optimize import linprog
    res = linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        b = res.x
    else:
        b = None
    return b
```

```
    a = a0.copy()
    best_a = a.copy()

    momentum = np.zeros_like(a)
    beta = 0.9
```

def solve_lp_max_sum(f, M, max_time):
    n = len(f)
    # Build constraint matrix rows in CSR format
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i_start ... i_end inclusive
        # for each i, j = k-i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    import scipy.sparse as sp
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # set up linprog
    from scipy.optimize import linprog
    c = -np.ones(n)  # maximize sum -> minimize -sum
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # Use 'highs' which is fast. We can set options for time limit (maybe maxiter)
    # Not sure if SciPy's high-level solver supports time limit; but we check we won't exceed.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Hybrid optimiser for minimise 2·n·max(conv(a,a))/sum(a)^2.

The optimiser proceeds in several loosely coupled phases:

1. **Beta‑family sweep** – a cheap search over symmetric Beta(α,α)
   densities (including the classical arcsine α=½).  The best discrete
   shape for a modest size (n=800) is kept as a starter.

2. **Projected gradient descent** – Nesterov‑style momentum on the
   non‑convex objective  max(conv).  Each iteration projects onto the
   probability simplex (∑a_i = 1, a_i ≥ 0).  The step size decays as
   1/√t.  A simple sub‑gradient of the active constraints is used.
   The descent is run for a time‑budgeted slice on each up‑sampled
   resolution.

3. **Linear‑programming refinement** – for moderate lengths (n ≤ 2000)
   we solve a linear programme
        max ∑b_i  s.t.  (f * b)_k ≤ M  ∀k ,  b_i ≥ 0
   where M = max(conv(f,f)).  Increasing the ℓ₁‑mass while keeping the
   cross‑convolution ≤ M typically reduces the final bound.
   The LP solution is normalised back to sum = 1 and fed into the next
   stage.

4. **Progressive up‑sampling** – the current best sequence is linearly
   interpolated to larger sizes (800 → 1600 → 3200 → 6400 → 9600); the
   gradient‑descent + LP cycle is repeated at each resolution.

5. **Final simulated‑annealing walk** – a cheap mass‑transfer walk
   (tiny ε moved between two positions) that accepts downhill moves
   and occasional uphill moves with a geometric temperature schedule.

The whole pipeline respects the supplied time budget (default 1000 s).
If the time runs out early the best sequence found so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used only for the LP stage)
# ----------------------------------------------------------------------
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _LP_AVAILABLE = True
except Exception:                 # pragma: no cover – SciPy may be missing
    _LP_AVAILABLE = False

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT (fast for many calls)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (active constraints only)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    # Average over the active set – a standard sub‑gradient choice.
    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Projected gradient descent with Nesterov‑style momentum
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.8,
                         max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns the best point seen and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] gd n={n:5d} it={it:7d} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution (active constraints)
        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximise ℓ₁‑mass under cross‑conv constraints)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Given a non‑negative vector f (∑f = 1) and a bound M ≥ max(conv(f,f)),
    solve    maximise 1ᵀb   s.t.   conv(f,b) ≤ M,   b ≥ 0.
    Returns the optimal b or None if the LP fails.
    """
    if not _LP_AVAILABLE:
        return None

    n = f.shape[0]
    # Build the (2n‑1)×n constraint matrix A where (A b)_k = (f*b)_k
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # A b ≤ M  (vectorised)
    b_ub = np.full(2 * n - 1, M * 1.000001)   # tiny slack for numeric safety
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={"presolve": True})
    if res.success:
        return np.maximum(res.x, 0.0)   # guard tiny negatives
    return None


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass before re‑normalisation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer SA that moves up to 10 % of the source mass.
    Returns the best sequence seen and its (scaled) score.
    """
    a = a0.copy()
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
        s = 1.0

    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max() / (s * s)

    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # move up to 10 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        s_prop = a_prop.sum()
        conv_prop = _conv_self(a_prop)
        new_score = 2.0 * n * conv_prop.max() / (s_prop * s_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            cur_score = new_score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} it={it:7d} "
                  f"score={cur_score:.6f}")

    return best_seq, best_score

    """
    Hybrid optimiser that searches for a non‑negative step‑function
    (list of heights) with a low value of `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a one‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional starter supplied by the harness (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # 1. Beta‑family sweep on a modest resolution (n0 = 800)
    # ------------------------------------------------------------------
    n0 = 800
    xs = (np.arange(n0) + 0.5) / n0           # mid‑point locations in (0,1)

    # beta parameters to try – symmetric Beta(α,α).  Include arcsine (α=0.5).
    alphas = np.concatenate((
        np.linspace(0.15, 0.45, 7),   # below arcsine
        np.linspace(0.55, 1.20, 7),   # slightly above
        np.linspace(1.4, 3.0, 5)      # much more interior‑mass
    ))
    alphas = np.unique(np.round(alphas, 6))

    for α in alphas:
        # pdf ∝ x^{α‑1} (1‑x)^{α‑1}
        a = np.power(xs, α - 1.0) * np.power(1.0 - xs, α - 1.0)
        # Guard against NaNs (should not happen for α>0)
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        conv = np.convolve(a, a)
        score = 2.0 * n0 * conv.max()     # sum=1 → scaled score = 2·n·max
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] α={α:5.3f} "
                  f"init score={score:.6f}")

    # if the harness supplied a starter, try it as well
    if starter is not None:
        conv = np.convolve(starter, starter)
        score = 2.0 * n0 * conv.max()
        if score < best_score:
            best_score = score
            best_seq = starter.copy()
            print(f"[{time.time() - start_time:6.2f}s] using harness starter "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling + gradient descent + LP refinement
    # ------------------------------------------------------------------
    stage_lengths = [800, 1600, 3200, 6400, 9600]

    # allocate roughly 80 % of the budget to the optimisation loop,
    # the rest will be kept for the final anneal.

    for n in stage_lengths:
        if time.time() >= loop_deadline:
            break

        # up‑sample the current best sequence to the new resolution
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = _upsample(best_seq, n)   # will never happen, defensive

        a0 = _project_simplex(a0)  # ensure exact sum = 1

        # ----------- gradient descent phase -----------
        # allocate a slice of the remaining budget proportionally
        remaining = loop_deadline - time.time()
        # ensure at least a few seconds for each stage

        stage_deadline = time.time() + stage_time

        a_opt, score_opt = _run_gradient_descent(
            a0, n, stage_deadline, start_time,
            beta=0.9, step0=0.8, max_iter=10**7

        if score_opt < best_score:

        # -------------- LP refinement (only for modest n) -----------------
        if _LP_AVAILABLE and n <= 2000 and time.time() < loop_deadline:

            b = _solve_lp_max_sum(best_seq, M)

                        best_seq = b / s
                        print(f"[{time.time() - start_time:6.2f}s] LP n={n} "
                              f"improve score={lp_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final simulated‑annealing refinement (uses whatever final length)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        best_seq, best_score = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-5, max_steps=300_000
        )
        print(f"[{time.time() - start_time:6.2f}s] final anneal best score="
              f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Safety fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny numerical drift may happen)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()

        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
