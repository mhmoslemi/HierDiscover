        it+=1
        conv = _conv_self(a)

"""
Propose a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  The method is a multi‑stage search:

* Stage 0 – many short runs on n≈1024 with random restarts.
  Uses projected gradient descent with a *correct* sub‑gradient of the
  self‑convolution maximum (the previous code halved the diagonal term).
  Momentum (Nesterov) and a √t‑schedule are employed.

* Stage 1 – up‑sample the best vector to a medium size (≈4096) and run a
  longer gradient descent.

* Stage 2 – up‑sample again to a large size (≈12288) and continue the
  descent.

* Stage 3 – a light simulated‑annealing phase for fine tuning.

All stages respect the overall time budget (default 1000 s) and the
final vector is normalised to sum = 1 (the exact scaling does not affect
`evaluate_sequence`).  Helper functions are kept at module level as
required by the harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x_i ≥ 0, Σ x_i = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – correct diagonal handling.
# The gradient of conv[k] w.r.t a_i is 2*a_{k-i} if the index is in range.
# For several (almost) maximal k we average the gradients.
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, conv: np.ndarray,
                     rel_tol: float = 1e-12) -> np.ndarray:
    """Return an (averaged) sub‑gradient of max(conv)."""
    n = a.shape[0]
    max_val = conv.max()
    # indices whose value is within rel_tol of the maximum
    if max_val == 0:
        # avoid division by 0 – gradient is zero
        return np.zeros_like(a)
    eps = max(rel_tol * max_val, 0.0)
    max_inds = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    # vector of all positions
    idx = np.arange(n)

    for k in max_inds:
        # i runs over all components, k-i must be a valid index
        i = k - idx
        mask = (i >= 0) & (i < n)
        grad[mask] += 2.0 * a[i[mask]]

    if max_inds.size > 1:
        grad /= max_inds.size          # average over the active set
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent (Nesterov style) with the corrected sub‑gradient
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Minimise F(a) = 2 * n * max(conv(a)) under a ≥ 0, Σ a = 1.
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] grad improve: n={n}, "
                  f"iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient (average over all almost‑maximal entries)
        grad = _subgradient_max(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Small‑scale simulated annealing: move ε from a high‑mass index to a
    random one, using a geometric temperature schedule.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source index proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative list) that minimises
    `evaluate_sequence`.  The routine follows a multi‑stage scheme
    (random restarts → up‑sampling → long gradient descent → annealing)
    while respecting the supplied time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=float)
            if tmp.ndim == 1 and tmp.size > 0:
                starter = tmp / tmp.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    slice_time = max(30.0, 0.10 * (deadline - time.time()))   # about 10 % of the budget
    slice_deadline = time.time() + slice_time

    restarts = 8
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = np.random.rand(n0)
            a0 /= a0.sum()

        # randomise a few hyper‑parameters for diversity
        beta = np.random.uniform(0.80, 0.95)
        step0 = np.random.uniform(0.20, 0.45)

        a_opt, score = _run_gradient_descent(
            a0, n0, slice_deadline, start_time,
            beta=beta,
            step0=step0,
        )
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: "
                  f"restart={r}, score={score:.6f}")

    # fallback in the unlikely case that no vector was produced
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * (_conv_self(best_seq).max())

    # ------------------------------------------------------------------
    # Stage 1 – upscale to medium size (n≈4096) and longer gradient descent
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.35 * remaining)          # about 35 % of what is left
    grad_deadline1 = time.time() + grad_time1

    a1_opt, score1 = _run_gradient_descent(
        a1, n1, grad_deadline1, start_time,
        beta=0.92,
        step0=0.28,
    )
    if score1 < best_score:
        best_score = score1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={score1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to a large size (n≈12288) and continue optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(15.0, 0.30 * remaining)          # about 30 % of the rest
    grad_deadline2 = time.time() + grad_time2

    a2_opt, score2 = _run_gradient_descent(
        a2, n2, grad_deadline2, start_time,
        beta=0.94,
        step0=0.24,
    )
    if score2 < best_score:
        best_score = score2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={score2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_ref, score_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=4e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    return [float(x) for x in best_seq]
