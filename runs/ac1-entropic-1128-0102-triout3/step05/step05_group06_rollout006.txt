"""
A fast two‑phase optimiser for the L∞‑norm of a self‑convolution.

Phase 1 – Linear‑programming refinement:
    Starting from a smooth arcsine profile we repeatedly
    * compute M = max(f∗f)
    * solve the LP   max ‖g‖₁   s.t.   f∗g ≤ M,  g ≥ 0
    * renormalise g to sum = 1 and mix f ← (1‑t)·f + t·g
  The LP spreads mass into positions where f is small, flattening the
  convolution and decreasing the evaluation score.

Phase 2 – Progressive up‑sampling + projected gradient descent
    The refined vector is linearly interpolated to larger sizes
    (800 → 2400 → 4800 → 9600) and a few thousand steps of
    projected gradient descent on the simplex are performed.
    Finally a cheap simulated‑annealing walk fine‑tunes the result.

All helpers are top‑level; the routine respects the 1000 s budget
and returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i ≥ 0, Σx_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (active constraints only)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # derivative of c_k = Σ a_i a_{k-i} is 2·a_{k-i}
        g[i] += a[j]
        g[j] += a[i]

    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Projected gradient descent with Nesterov‑style momentum
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.8,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns the best point seen and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] gd n={n:5d} it={it:7d} score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)          # sub‑gradient of max(conv)
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)               # diminishing step size
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Very cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} it={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Linear‑programming refinement (maximise ℓ₁ norm under convolution bound)
# ----------------------------------------------------------------------
def _lp_maximize_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   a∗g ≤ M  (element‑wise) , g ≥ 0 .
    Returns the optimal g (not normalised).
    """
    n = a.shape[0]
    m = 2 * n - 1
    # dense matrix A where A[k, j] = a[i] with i = k−j (if i in range)
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        A[k, j] = a[i]           # vectorised assignment

    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximise Σ g  → minimise −Σ g
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0. Use a starter if the harness supplied one (global variable)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)

            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:

    # --------------------------------------------------------------

    else:
        xs = (np.arange(n0) + 0.5) / n0

        # normalise g to sum = 1 (feasible region always contains the current a,
        # therefore Σ g ≥ 1)

        if cur_score < best_score:
            best_score = cur_score

    stage_lengths = [800, 2400, 4800, 9600]          # geometric growth

    s = best_seq.sum()

    else:

    # return plain Python list as required by the harness
    return [float(x) for x in best_seq]
