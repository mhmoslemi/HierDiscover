"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Initialise from the previous construction (`height_sequence_1`) if it exists,
    otherwise from a random point on the probability simplex.
2.  Progressive up‑sampling: 800 → 2400 → 4800 → 9600.
3.  For every resolution
    * a) smooth “soft‑max” gradient descent (the code that gave the
       1.513… bound) – many cheap FFT‑based steps,
    * b) exact projected sub‑gradient descent on the true max‑convolution.
    The best point seen is kept.
4.  A final simulated‑annealing walk with tiny mass‑transfer moves.
5.  Return the best sequence found (as a plain Python list).

All steps respect the time budget (default 1000 s).  The implementation
uses only NumPy, ``math`` and the built‑in ``time`` module.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3: convolution of rev(a) with a weight vector w (both real)
# ----------------------------------------------------------------------
def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Convolution of the reversed sequence a_rev with a weight vector w."""
    n = a_rev.shape[0]
    m = w.shape[0]
    fft_len = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=fft_len)
    B = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 4: gradient of a *soft‑max* of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Gradient of Σ_k w_k(conv_k)·conv_k with
    w_k = exp(conv_k / T) / Σ exp(...).
    """
    maxc = conv.max()
    scaled = (conv - maxc) / temperature          # stability shift
    exps = np.exp(scaled)
    w = exps / exps.sum()

    rev_a = a[::-1]
    full = _conv_rev_weight(rev_a, w)            # length = n + (2n-1) - 1 = 3n-2
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]           # central block
    return grad


# ----------------------------------------------------------------------
# Helper 5: exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray):
    """
    Returns (max_val, grad) where max_val = max(a*a) and grad is a
    sub‑gradient of that maximum.  grad is projected onto the simplex
    later by the optimiser.
    """
    conv = _conv_self(a)
    max_val = conv.max()
    max_idx = np.where(conv == max_val)[0]

    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # a[k-idx] is safe because k-idx is in [0, n-1]
        grad[idx] += 2.0 * a[k - idx]

    grad /= len(max_idx)          # any convex combination of active gradients
    return max_val, grad


# ----------------------------------------------------------------------
# Optimiser 1: projected soft‑max gradient descent
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.6,
                    beta: float = 0.9,
                    max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov) GD on a smooth soft‑max surrogate of max(conv).
    Returns the best point seen and its (scaled) evaluation score.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    T_start = 2.0e-2
    T_end = 5.0e-4

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax improve: n={n}, iter={it}, score={cur_score:.6f}")

        # exponential temperature schedule
        T = T_start * (T_end / T_start) ** (it / 20000.0)

        grad = _softmax_gradient(a, conv, T)

        # Nesterov momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimiser 2: projected exact sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------
def _run_exact_gd(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  step0: float = 0.3,
                  beta: float = 0.9,
                  max_iter: int = 10 ** 7) -> tuple[np.ndarray, float]:
    """
    Projected (Nesterov) sub‑gradient descent on the true max‑convolution.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        max_val, grad = _subgradient_maxconv(a)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exact improve: n={n}, iter={it}, score={cur_score:.6f}")

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        step = step0 / math.sqrt(it)
        a = _project_simplex(a - step * momentum)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper 6: linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate ``seq`` linearly to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 800_000) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a small fraction of mass between two positions."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 10 % of the donor mass
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it}, score={score:.6f}")

    return best_seq, best_score

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional starter from a previous run

    try:
