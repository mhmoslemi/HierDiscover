def propose_candidate(seed=0, budget_s=1000):
    import time
    import numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    # get previous sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        # clip
        prev = np.clip(prev, 0.0, 1000.0)
        # remove negligible negatives done
        if prev.sum() < 0.01:
            raise ValueError
        # normalize to sum=1
        a = prev / prev.sum()
        # trim trailing zeros for efficiency
        eps = 1e-12
        nz = np.nonzero(a > eps)[0]
        if nz.size == 0:
            # fallback uniform
            n = 500
            a = np.full(n, 1.0 / n)
        else:
            a = a[:nz[-1] + 1]
    except Exception:
        # fallback uniform
        n = 500
        a = np.full(n, 1.0 / n)
    n = a.shape[0]
    # ensure at least 2 elements
    if n < 2:
        n = 2
        a = np.full(n, 1.0 / n)
    # compute score
    def score_vec(v):
        # length n of vector matters
        m = v.shape[0]
        # compute max conv via FFT
        size = 1 << ((2*m - 1).bit_length())
        A = np.fft.rfft(v, n=size)
        conv = np.fft.irfft(A * A, n=size)[:2*m-1]
        max_b = conv.max()
        return 2 * m * max_b
    best_a = a.copy()
    best_score = score_vec(best_a)
    # keep current a
    cur_a = best_a.copy()
    # precompute size for current length
    size = 1 << ((2*n - 1).bit_length())
    # set parameters
    step_fraction = 0.1
    # annealing schedule: reduce step over time
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break
        # update step size linearly decreasing
        # step decays from 0.1 to 0.001 over time
        step = step_fraction * (1 - elapsed / budget_s) + 0.001
        # pick donor index i with large value
        # choose from top 10% of values
        topk = max(1, int(0.1 * n))
        # indices sorted by value descending
        idx_desc = np.argpartition(-cur_a, topk-1)[:topk]
        i = rng.choice(idx_desc)
        # choose recipient from bottom 10%
        botk = max(1, int(0.1 * n))
        idx_asc = np.argpartition(cur_a, botk-1)[:botk]
        j = rng.choice(idx_asc)
        if i == j:
            continue
        delta = min(step * cur_a[i], cur_a[i])
        if delta < 1e-12:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # if any negative clamp
        if new_a[i] < -1e-12:
            continue
        # compute score
        # length unchanged
        A = np.fft.rfft(new_a, n=size)
        conv = np.fft.irfft(A * A, n=size)[:2*n-1]
        max_b = conv.max()
        new_score = 2 * n * max_b
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = new_a.copy()
            cur_a = new_a
            # print progress occasionally
            if int(elapsed) % 30 == 0:
                print(f"[{elapsed:.1f}s] New best score {best_score:.6f}")
        else:
            # optionally accept with small probability
            # compute temperature
            temperature = 0.01 * (1 - elapsed / budget_s)
            if temperature > 0 and rng.random() < np.exp((best_score - new_score) / temperature):

    # final best sequence length may be trimmed of trailing zeros again
    nz = np.nonzero(best_a > eps)[0]
    if nz.size > 0:

    # final check score
    final_score = score_vec(best_a)
    print(f"Finished after {time.time()-start:.2f}s, best score {final_score:.6f}, length {len(best_a)}")
    # return list of floats
    return [float(x) for x in best_a]

"""
propose_candidate

Search for a step‑function (non‑negative vector) that minimises
evaluate_sequence.  The algorithm starts from the best known
construction (the global variable ``height_sequence_1``) and performs a
guided stochastic local search:

* At each iteration we compute the discrete convolution a*a and locate
  the index k where the value is maximal (the current bottleneck).
* The contribution a[i]·a[k−i] of every component to that maximal value
  is evaluated; the index with the largest contribution is chosen as a
  *donor* – moving mass away from this entry is most likely to lower the
  peak.
* Mass is transferred (by a small, annealed step size) from the donor to
  the current minimal entry (the *receiver*).  The move is accepted if
  it does not increase the score, or – with a temperature‑dependent
  probability – if it yields a slight increase (simulated‑annealing).
* The best vector ever seen is remembered.  Trailing zeros are trimmed
  before returning, which reduces the factor ``n`` in the objective.

The search runs for the allotted time (default 1000 s) and returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1.  Load a previous construction (if any) and normalise it.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        prev = np.clip(prev, 0.0, 1000.0)               # same clipping as evaluate_sequence
        if prev.sum() < 0.01:
            raise ValueError
        # normalise to a probability vector (the score is scale‑invariant)
        a = prev / prev.sum()
        # remove trailing negligible entries – they only increase the factor n
        eps = 1e-12
        nz = np.nonzero(a > eps)[0]
        if nz.size:
            a = a[: nz[-1] + 1]
    except Exception:
        # Fallback: a modest uniform vector (length 2000 → score = 2)
        n0 = 2000
        a = np.full(n0, 1.0 / n0, dtype=float)

    n = a.shape[0]
    if n < 2:                      # guard against absurdly short vectors
        a = np.full(2, 0.5, dtype=float)
        n = 2

    # ------------------------------------------------------------------
    # Helper: fast evaluation of  Φ(a) = 2·n·max(a*a) / (sum(a)^2)
    # For our normalised vectors sum(a)=1, so Φ = 2·n·max_conv.
    # ------------------------------------------------------------------
    def score_of(vec: np.ndarray) -> float:
        m = vec.shape[0]
        size = 1 << ((2 * m - 1).bit_length())          # next power of two
        A = np.fft.rfft(vec, n=size)
        conv = np.fft.irfft(A * A, n=size)[: 2 * m - 1]
        max_conv = conv.max()
        return 2.0 * m * max_conv

    # ------------------------------------------------------------------
    # Initial statistics
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_score = score_of(best_vec)

    cur_vec = best_vec.copy()
    cur_score = best_score

    # pre‑computed FFT size for the current length (kept constant)
    fft_size = 1 << ((2 * n - 1).bit_length())

    # annealing / step‑size schedule parameters
    max_temp = 0.02          # initial temperature for uphill moves
    min_step = 0.001         # smallest transfer fraction
    max_step = 0.15          # start with relatively bold updates

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:
            break

        # ------------------------------------------------------------------
        # 2.  Identify the current bottleneck (largest convolution entry)
        # ------------------------------------------------------------------
        A_cur = np.fft.rfft(cur_vec, n=fft_size)
        conv_cur = np.fft.irfft(A_cur * A_cur, n=fft_size)[: 2 * n - 1]
        max_conv_cur = conv_cur.max()
        k_max = int(conv_cur.argmax())

        # contribution of each coordinate to the peak at k_max
        idx = np.arange(n)
        jdx = k_max - idx
        mask = (jdx >= 0) & (jdx < n)
        contrib = np.zeros_like(cur_vec)
        contrib[mask] = cur_vec[idx[mask]] * cur_vec[jdx[mask]]

        # donor = index with maximal contribution to the peak
        donor = int(np.argmax(contrib))
        # receiver = index with minimal current mass (prefer a genuine zero)
        receiver = int(np.argmin(cur_vec))

        if donor == receiver:
            # pick a different receiver at random
            possible = np.where(np.arange(n) != donor)[0]
            if possible.size == 0:
                continue
            receiver = int(rng.choice(possible))

        # ------------------------------------------------------------------
        # 3.  Propose a transfer of mass from donor to receiver
        # ------------------------------------------------------------------
        # step size shrinks linearly with time, never below min_step
        step_frac = max_step * (1.0 - elapsed / budget_s) + min_step
        delta = step_frac * cur_vec[donor]
        if delta <= 0.0:
            continue

        new_vec = cur_vec.copy()
        new_vec[donor] -= delta
        new_vec[receiver] += delta

        # safeguard against negative values due to round‑off
        if new_vec[donor] < -1e-12:
            continue

        # ------------------------------------------------------------------
        # 4.  Compute the new score
        # ------------------------------------------------------------------
        A_new = np.fft.rfft(new_vec, n=fft_size)
        conv_new = np.fft.irfft(A_new * A_new, n=fft_size)[: 2 * n - 1]
        new_score = 2.0 * n * conv_new.max()

        # ------------------------------------------------------------------
        # 5.  Acceptance rule (simulated annealing)
        # ------------------------------------------------------------------
        if new_score <= cur_score:
            accept = True
        else:
            temperature = max_temp * (1.0 - elapsed / budget_s)
            # probability to accept a worsening move
            accept = rng.random() < np.exp(-(new_score - cur_score) / (temperature + 1e-12))

        if accept:
            cur_vec = new_vec
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_vec = new_vec.copy()

        # ------------------------------------------------------------------
        # 6.  Logging
        # ------------------------------------------------------------------
        iteration += 1
        if iteration % 2000 == 0:
            print(f"[{elapsed:.1f}s] iter {iteration:,} – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final tidy‑up: trim trailing zeros (they only increase n)
    # ------------------------------------------------------------------
    eps = 1e-12
    nz = np.nonzero(best_vec > eps)[0]
    if nz.size:
        best_vec = best_vec[: nz[-1] + 1]

    final_score = score_of(best_vec)
    print(f"Finished after {time.time() - start_time:.2f}s – final score {final_score:.6f}, length {len(best_vec)}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_vec]
