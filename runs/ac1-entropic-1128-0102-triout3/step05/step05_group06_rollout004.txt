def _subgradient_active(a, conv, rel_tol=0.01):
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)
    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi+1)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]
    if active.size > 0:
        g /= float(active.size)
    return g

"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.  The optimiser combines three ideas:

1. **Progressive up‑sampling** – start with a modest size (≈2000) and
   repeatedly linearly interpolate the best sequence to a larger size
   (≈4000 → 8000 → 16000).  Larger n gives a lower theoretical lower bound
   2·n·max/∑² → 1, so we aim for the largest n that fits the time budget.

2. **Adam‐style projected optimisation** – we minimise the maximum of the
   self‑convolution on the probability simplex using an Adam optimiser.
   After each update we project back onto the simplex (non‑negative,
   sum = 1).  The sub‑gradient is taken as the average of the gradients of
   all “almost maximal’’ convolution entries (within 1 % of the current
   maximum).  This yields a smoother descent direction than the plain
   sub‑gradient used previously.

3. **Final simulated‑annealing walk** – a cheap mass‑transfer SA cleans up
   any remaining peaks.

The routine respects a 1000 s budget (one‑second safety margin) and
returns the best sequence discovered as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) – average over near‑max entries
# ----------------------------------------------------------------------
def _subgradient_active(a: np.ndarray, conv: np.ndarray,
                        rel_tol: float = 0.01) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _run_adam_optimize(a0: np.ndarray, n: int,
                       deadline: float, start_time: float,
                       rel_tol: float = 0.01,
                       step0: float = 0.5,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8,
                       max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of  max(conv(a,a))  on the probability simplex.
    Returns the best sequence seen and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n:5d} "
                  f"iter={it:7d} score={cur_score:.6f}")

        # compute sub‑gradient using a modest tolerance
        grad = _subgradient_active(a, conv, rel_tol)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate decays like 1/√t (optional)
        lr = step0 / math.sqrt(it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 600_000) -> tuple[np.ndarray, float]:
    """Very cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} "
                  f"iter={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,
    runs a progressive up‑sampling + Adam optimiser, finishes with a short
    simulated‑annealing refinement and finally returns the best sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a one‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. If the harness provided a starter, use it (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # 1. Initial analytical shape – discretised arcsine (Beta(½,½))
    # ------------------------------------------------------------------
    n0 = 2000
    xs = (np.arange(n0) + 0.5) / n0
    arcsine = 1.0 / np.sqrt(xs * (1.0 - xs))
    a0 = arcsine / arcsine.sum()                     # sum = 1
    if starter is not None:
        a0 = _upsample(starter, n0)

    # ------------------------------------------------------------------
    # 2. Stage list – progressive up‑sampling (geometric growth)
    # ------------------------------------------------------------------
    stage_lengths = [2000, 4000, 8000, 16000]
    total_weight = sum(stage_lengths)

    # Reserve about 20 % of the overall budget for the final annealing walk.
    anneal_budget = max(30.0, 0.20 * (deadline - start_time))
    grad_budget = (deadline - start_time) - anneal_budget

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining gradient budget proportionally
        time_left = deadline - start_time - anneal_budget
        # avoid negative allocation if we are already low on time
        if time_left <= 0:
            break
        stage_frac = n / total_weight
        stage_time = max(5.0, time_left * stage_frac)
        stage_deadline = time.time() + stage_time

        # initialise a0 for this stage
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        else:
            a0 = _upsample(a0, n)

        # ------------------------------------------------------------------
        # 2a. Adam optimiser on the simplex
        # ------------------------------------------------------------------
        a_opt, score_opt = _run_adam_optimize(
            a0, n,
            deadline=stage_deadline,
            start_time=start_time,
            rel_tol=0.01,          # consider the top 1 % of the conv histogram
            step0=0.6,             # initial learning rate (tuned empirically)
            max_iter=10**7
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: "
                  f"score={score_opt:.6f}")

        # ------------------------------------------------------------------
        # 2b. Optional short random jitter + quick Adam run (helps escape)
        # ------------------------------------------------------------------
        # Give a tiny extra boost if we still have a few seconds left.
        extra_time = stage_deadline - time.time()
        if extra_time > 8.0:
            # create a noisy copy of the current best
            jitter = best_seq * (1.0 + 0.02 * np.random.randn(n))
            jitter = np.maximum(jitter, 0.0)
            jitter = jitter / jitter.sum()
            sub_deadline = time.time() + min(8.0, extra_time)
            a_jit, score_jit = _run_adam_optimize(
                jitter, n,
                deadline=sub_deadline,
                start_time=start_time,
                rel_tol=0.015,
                step0=0.3,
                max_iter=10**6
            )
            if score_jit < best_score:
                best_score = score_jit
                best_seq = a_jit.copy()
                print(f"[{time.time() - start_time:6.2f}s] jitter improve: "
                      f"score={score_jit:.6f}")

    # ------------------------------------------------------------------
    # 3. Final annealing phase – use whatever length we ended with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline=anneal_deadline,
            start_time=start_time,
            cur_best_score=best_score,
            cur_best_seq=best_seq,
            start_temp=5e-2,
            end_temp=1e-5,
            steps=600_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # 4. Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
