# ----------------------------------------------------------------------
# Basic linear‑algebra helpers

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i≥0, ∑x_i=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector (sum = 1)."""
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution a * a using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _corr_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid cross‑correlation of w (len = 2n‑1) with a (len = n):
    returns length‑n array corr[i] = Σ_j w[i+j]·a[j].
    Implemented via FFT for speed.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2n‑1
    total_len = m + n - 1
    fft_len = 1 << ((total_len - 1).bit_length())
    W = np.fft.rfft(w, fft_len)
    A_rev = np.fft.rfft(a[::-1], fft_len)
    full = np.fft.irfft(W * A_rev, fft_len)[:total_len]
    start = n - 1
    return full[start:start + n]

# Smooth p‑norm objective (f = (∑ conv^p)^{1/p}) and its gradient
# ----------------------------------------------------------------------
def _obj_and_grad(z: np.ndarray, p: int) -> tuple[float, np.ndarray]:
    """
    Returns (f, ∇_z f) where f is the p‑norm of the self‑convolution of
    a = softmax(z).  The gradient is derived analytically.
    """
    a = _softmax(z)                     # probability vector, sum=1
    conv = _conv_self(a)                # length 2n‑1
    conv_p = conv ** p
    S = conv_p.sum()
    # Guard against underflow (should not happen for reasonable a)
    if S == 0.0:
        return np.inf, np.zeros_like(z)

    f = S ** (1.0 / p)                  # p‑norm
    # weight w_k = ∂f/∂conv_k = S^{1/p-1} * conv_k^{p-1}
    # This can be written w = conv^{p-1} / S^{(p-1)/p}
    factor = S ** (-(p - 1) / p)
    w = conv ** (p - 1) * factor        # length 2n‑1

    # ∂f/∂a_i = Σ_k w_k ∂conv_k/∂a_i = 2 Σ_k w_k a_{k-i}
    grad_a = 2.0 * _corr_valid(w, a)    # length n

    # Chain rule through soft‑max:
    # ∂f/∂z_j = a_j (∂f/∂a_j - Σ_i a_i ∂f/∂a_i)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    return f, grad_z


# ----------------------------------------------------------------------
# Optimisation of a single starting shape via L‑BFGS‑B
# ----------------------------------------------------------------------
def _optimize_shape(a0: np.ndarray,
                    p: int = 8,
                    max_iter: int = 300,
                    deadline: float = 1e30,
                    start_time: float = 0.0) -> np.ndarray:
    """
    Optimises the probability vector a0 (sum=1) over the smooth p‑norm.
    Returns the improved probability vector (still on the simplex).
    """
    # Convert to log‑space (soft‑max parametrisation)
    z0 = np.log(a0 + 1e-30)

    # Wrapper to respect time budget for the optimiser
    def callback(zk):
        if time.time() >= deadline:
            raise RuntimeError("time budget exceeded inside optimiser")

    # SciPy minimise with analytic gradient
    if _SCIPY_AVAILABLE:
        try:
            res = minimize(_obj_and_grad, z0,
                           args=(p,),
                           method='L-BFGS-B',
                           jac=True,
                           callback=callback,
                           options={'maxiter': max_iter,
                                    'ftol': 1e-12,
                                    'disp': False})
            a_opt = _softmax(res.x)
            return a_opt
        except RuntimeError:
            # time limit – return the current best (softmax of last z)
            return _softmax(z0)
    else:
        # Fallback: simple projected gradient descent (very limited)
        a = a0.copy()
        momentum = np.zeros_like(a)
        beta = 0.9
        step0 = 0.8
        for it in range(1, max_iter + 1):
            if time.time() >= deadline:
                break
            conv = _conv_self(a)
            # gradient of the max (sub‑gradient) – identical to earlier code
            max_val = conv.max()
            rel_tol = 1e-12
            active = np.where(conv >= max_val * (1 - rel_tol))[0]
            g = np.zeros_like(a)
            n = a.shape[0]
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                i = np.arange(lo, hi + 1, dtype=int)
                j = k - i
                g[i] += a[j]
                g[j] += a[i]
            if active.size:
                g /= float(active.size)
            momentum = beta * momentum + (1 - beta) * g
            step = step0 / math.sqrt(it)
            a = a - step * momentum
            a = _project_simplex(a)
        return a


# ----------------------------------------------------------------------
# Tiny mass‑transfer simulated annealing (fast O(n) updates)
# ----------------------------------------------------------------------
def _anneal(a: np.ndarray,
            max_moves: int = 100_000,
            start_temp: float = 5e-2,
            end_temp: float = 1e-5,
            deadline: float = 1e30,
            start_time: float = 0.0) -> np.ndarray:
    """
    Perform a cheap SA walk that moves a small ε from i to j.
    Convolution is updated incrementally in O(n) per move.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    for it in range(1, max_moves + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / max_moves)
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.1  # move up to 10 % of source mass

        # Incremental convolution update (linear shift)
        conv[i:i + n] -= 2.0 * eps * a
        conv[j:j + n] += 2.0 * eps * a
        a[i] -= eps
        a[j] += eps

        cur_score = 2.0 * n * conv.max()
        delta = cur_score - best_score
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # Metropolis acceptance (allow uphill moves)
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            # keep the move (already applied)
            pass
        else:
            # revert
            conv[i:i + n] += 2.0 * eps * a
            conv[j:j + n] -= 2.0 * eps * a
            a[i] += eps
            a[j] -= eps

        # occasional progress report
        if it % 5000 == 0:
            print(f"[{time.time() - start_time:6.2f}s] anneal move={it} "
                  f"best_score={best_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Helper: build a symmetric Beta(α,α) density on the discrete grid
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Return a discretised symmetric Beta(α,α) density (sum=1)."""
    xs = (np.arange(n) + 0.5) / n
    # beta pdf ∝ x^{α-1} (1-x)^{α-1}
    pdf = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, alpha - 1.0)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0:
        pdf = np.full_like(pdf, 1.0 / n)
    else:
        pdf = pdf / pdf_sum
    return pdf


def _mixture_shape(n: int, lam: float) -> np.ndarray:
    """Mixture λ·arcsine + (1-λ)·uniform, normalised."""
    xs = (np.arange(n) + 0.5) / n
    arcsine = 1.0 / np.sqrt(xs * (1.0 - xs))
    arcsine /= arcsine.sum()
    uniform = np.full(n, 1.0 / n)
    mix = lam * arcsine + (1.0 - lam) * uniform
    mix /= mix.sum()
    return mix


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget, tries several analytically
    inspired initial shapes, refines them with a smooth‑p‑norm L‑BFGS
    optimiser and finishes with a short annealing walk.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 1 s safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Candidate initial shapes (all normalised)
    # ------------------------------------------------------------------
    n_opt = 2400                    # optimisation resolution (fast enough)
    shapes = []

    # a) arcsine (Beta(0.5,0.5))
    shapes.append(_beta_shape(n_opt, 0.5))

    # b) a few other Beta shapes
    for a in [0.3, 0.35, 0.4, 0.6, 0.7, 0.8]:
        shapes.append(_beta_shape(n_opt, a))

    # c) mixtures of arcsine & uniform
    for lam in [0.2, 0.4, 0.6, 0.8]:
        shapes.append(_mixture_shape(n_opt, lam))

    # d) optionally use a starter supplied by the harness
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            shapes.append(_upsample(starter, n_opt))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Optimise each shape with the smooth p‑norm objective
    # ------------------------------------------------------------------
    p_initial = 8               # base smoothness parameter
    for idx, a0 in enumerate(shapes):
        if time.time() >= deadline:
            break
        # optimise
        stage_deadline = min(deadline,
                             time.time() + 0.12 * (deadline - start_time))  # ~12 % budget per shape
        a_opt = _optimize_shape(a0,
                                 p=p_initial,
                                 max_iter=250,
                                 deadline=stage_deadline,
                                 start_time=start_time)
        # upsample to a larger size for the final scoring
        n_final = 9600
        a_big = _upsample(a_opt, n_final)

        # evaluate using the provided black‑box (fast, O(n log n))
        try:
            score = evaluate_sequence(a_big.tolist())
        except Exception:      # pragma: no cover
            score = float("inf")

        if score < best_score:
            best_score = score
            best_seq = a_big.copy()
            print(f"[{time.time() - start_time:6.2f}s] shape {idx+1}/{len(shapes)} "
                  f"improved score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final refinement – short annealing on the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        anneal_deadline = deadline
        refined = _anneal(best_seq.copy(),
                         max_moves=120_000,
                         start_temp=5e-2,
                         end_temp=1e-6,
                         deadline=anneal_deadline,
                         start_time=start_time)
        # final evaluation
        try:
            score_ref = evaluate_sequence(refined.tolist())
        except Exception:      # pragma: no cover
            score_ref = float("inf")
        if score_ref < best_score:
            best_score = score_ref
            best_seq = refined.copy()
            print(f"[{time.time() - start_time:6.2f}s] annealing improved score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # 4. Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny rounding drift may have occurred)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return plain Python list
    return [float(x) for x in best_seq]
