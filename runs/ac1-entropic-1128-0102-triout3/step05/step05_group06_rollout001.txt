"""
Propose a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  The optimiser works in three phases:

  1. **Fast shape sweep** – a small set of symmetric Beta‑pdfs
     (including the arcsine law) and a simple triangular shape are
     discretised, normalised to sum = 1 and their scores are evaluated.
     The best among them (or the provided `height_sequence_1`) is kept.

  2. **Large‑scale descent** – the best seed is up‑sampled to a larger
     dimension (≈ 8000) and refined with two complementary simplex‑
     constrained gradient methods:
        * projected gradient descent with Nesterov momentum,
        * mirror descent (exponentiated gradient) on the probability simplex.
     Both run until the allocated time slice expires.

  3. **Stochastic refinement** – a cheap simulated‑annealing walk moves a
     tiny amount of mass between two positions, accepting downhill moves
     always and uphill moves with a geometric temperature schedule.
     This phase uses the remaining time budget.

All helper routines are top‑level; the algorithm respects the supplied
time budget (default 1000 s) and always returns a valid normalised list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i >= 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (active constraints only)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    All indices k with conv[k] ≥ max(conv)*(1‑rel_tol) are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0.0:
        return np.zeros_like(a)

    threshold = max_val * (1.0 - rel_tol)
    active = np.where(conv >= threshold)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Projected gradient descent with Nesterov‑style momentum
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         beta: float = 0.9,
                         step0: float = 0.6,
                         max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Perform many iterations of projected gradient descent.
    Returns the best point seen and its *scaled* score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD n={n:5d} iter={it:7d} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_full(a, conv)

        # Nesterov‑style momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) on the simplex
# ----------------------------------------------------------------------
def _run_mirror_descent(a0: np.ndarray, n: int,
                        deadline: float, start_time: float,
                        eta0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with a decaying learning rate.
    Returns the best point seen and its scaled score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] MD n={n:5d} iter={it:7d} "
                  f"score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)
        eta = eta0 / math.sqrt(it)
        # exponential update (mirror step for KL divergence)
        a = a * np.exp(-eta * grad)
        # renormalise onto simplex
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Cheap SA that moves a small ε from i to j."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move up to 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal n={n:5d} iter={it:7d} "
                  f"score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the given time budget, performs a rapid
    shape sweep, a large‑scale gradient/mirror descent, and a final
    simulated‑annealing refinement.  Returns the best sequence found as
    a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # one‑second safety margin for the harness itself
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional starter from harness (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # 1. Fast shape sweep on a modest size (n_base)
    # ------------------------------------------------------------------
    n_base = 2000
    xs = (np.arange(n_base) + 0.5) / n_base  # mid‑point discretisation

    # a small grid of Beta shape parameters
    alphas = np.concatenate((
        np.linspace(0.25, 0.8, 12, endpoint=False),
        np.linspace(0.8, 2.0, 13)
    ))

    uniform_weights = [0.0, 0.02, 0.05, 0.10, 0.15, 0.20]

    for alpha in alphas:
        if time.time() >= deadline:
            break
        # symmetric Beta(alpha,alpha) density (unnormalised)
        beta_shape = xs ** (alpha - 1) * (1.0 - xs) ** (alpha - 1)
        for w in uniform_weights:
            if time.time() >= deadline:
                break
            shape = (1.0 - w) * beta_shape + w * np.ones_like(beta_shape)
            shape /= shape.sum()
            conv = _conv_self(shape)
            score = 2.0 * n_base * conv.max()
            if score < best_score:
                best_score = score
                best_seq = shape.copy()
                print(f"[{time.time() - start_time:6.2f}s] init beta α={alpha:.3f} w={w:.2f} "
                      f"score={score:.6f}")

    # also try a simple triangular shape (linearly increasing then decreasing)
    tri = np.minimum(np.arange(n_base) + 1, n_base - np.arange(n_base))
    tri = tri.astype(float)
    tri /= tri.sum()
    conv = _conv_self(tri)
    score = 2.0 * n_base * conv.max()
    if score < best_score:
        best_score = score
        best_seq = tri.copy()
        print(f"[{time.time() - start_time:6.2f}s] init triangular score={score:.6f}")

    # incorporate possible harness starter
    if starter is not None:
        s_up = _upsample(starter, n_base)
        conv = _conv_self(s_up)
        score = 2.0 * n_base * conv.max()
        if score < best_score:
            best_score = score
            best_seq = s_up.copy()
            print(f"[{time.time() - start_time:6.2f}s] init starter score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Large‑scale refinement (upsample + gradient + mirror descent)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 30.0:
        # target size – a few‑thousand points give a better theoretical bound
        n_big = max(4000, min(12000, int(4 * n_base)))
        a_big = _upsample(best_seq, n_big)

        # allocate ~60 % of the remaining time to deterministic descent
        gd_deadline = time.time() + 0.60 * remaining

        # projected gradient descent
        a_opt, score_opt = _run_gradient_descent(
            a_big, n_big, gd_deadline, start_time,
            beta=0.9, step0=0.6, max_iter=10**7
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] GD improve: score={score_opt:.6f}")

        # mirror descent (exponentiated gradient) – continue if time permits
        if time.time() < gd_deadline:
            a_opt2, score_opt2 = _run_mirror_descent(
                best_seq, n_big, gd_deadline, start_time, eta0=0.2
            )
            if score_opt2 < best_score:
                best_score = score_opt2
                best_seq = a_opt2.copy()
                print(f"[{time.time() - start_time:6.2f}s] MD improve: score={score_opt2:.6f}")

        # update remaining time after deterministic stage
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 3. Final stochastic refinement (annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-6, steps=600_000
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # 4. Safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)
        print("[fallback] using uniform distribution")

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return plain Python list
    return [float(x) for x in best_seq]
