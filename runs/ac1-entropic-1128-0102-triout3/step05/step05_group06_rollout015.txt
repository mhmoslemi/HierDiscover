def _optimize_via_adam(a0, n, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05  # base learning rate
    lr = lr0 / n  # scale
    t = 0
    best_a = a.copy()
    best_score = float('inf')
    # schedule for beta for smoothing
    beta_start = 5.0
    beta_end = 200.0

    while time.time() < deadline:
        t += 1
        # compute conv self
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam n={n} it={t} score={cur_score:.6f}")

        # smoothing beta schedule by progress
        progress = (time.time() - start) / (deadline - start)  # need start param
        beta = beta_start + progress * (beta_end - beta_start)
        # compute smoothed weights
        maxC = cur_max
        # using conv directly
        exp_term = np.exp(beta * (conv - maxC))
        w = exp_term / exp_term.sum()
        # gradient: conv_rev = convolution of rev_a and w
        rev_a = a[::-1]
        # compute convolution via FFT
        L = 1 << ((3 * n - 2).bit_length())
        Fa = np.fft.rfft(rev_a, n=L)
        Fw = np.fft.rfft(w, n=L)
        conv_rev = np.fft.irfft(Fa * Fw, n=L)[:3*n-2]
        grad = 4.0 * n * conv_rev[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_simplex(a)
    return best_a, best_score

def _run_annealing(a0, n, deadline, global_start, cur_best_score, cur_best_seq, start_temp=1e-1, end_temp=1e-6, max_moves=500000):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        # temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_moves)
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.2  # up to 20% transfer
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score
        if delta < 0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score
        if score < best_score:

            best_seq = a.copy()
            print(f"[{time.time() - global_start:6.2f}s] anneal n={n:5d} step={it:7d} score={score:.6f}")
    return best_seq, best_score

"""
Advanced optimizer for the step‑function problem.

The algorithm works in three phases:

1. **Progressive up‑sampling** – start from a simple arcsine shape on a
   modest size (800).  The current best sequence is linearly interpolated
   to larger lengths (2400 → 4800 → 9600 → 14400).  Larger dimensions give a
   finer discretisation of a continuous optimum.

2. **Smoothed Adam descent** – at each stage we minimise a smooth
   approximation of the objective 2·n·max(a*a) using the Adam optimiser.
   The smoothing is a log‑sum‑exp with a temperature β that is increased
   from 5 to 200 during the stage, turning the smooth loss into the true
   max‑convolution.  Gradients are obtained analytically via an FFT‑based
   convolution, projected onto the probability simplex after each step.

3. **Final simulated‑annealing refinement** – a cheap mass‑transfer walk
   (move a small ε from one index to another) with a geometric temperature
   schedule cleans up the remaining irregularities.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a feasible non‑negative sequence (fallback is uniform).
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i ≥ 0, Σ x_i = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution a * a via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << ((size).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:size]
    return conv


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Initial analytical shape – discretised arcsine (Beta(½,½))
# ----------------------------------------------------------------------
def _initial_arcsine(n: int) -> np.ndarray:
    """Return a probability‑simplex vector obtained from the arcsine density."""
    xs = (np.arange(n) + 0.5) / n
    arcs = 1.0 / np.sqrt(xs * (1.0 - xs))
    arcs /= arcs.sum()
    return arcs

# Adam optimiser with smoothed max‑convolution loss
# ----------------------------------------------------------------------
def _optimize_adam_stage(a0: np.ndarray, n: int,
                         stage_deadline: float,

    Adam optimisation for a single stage.
    Returns (best_a, best_score) where best_score = 2·n·max(conv).
    """

    m = np.zeros_like(a)

    beta_start, beta_end = 5.0, 200.0

    L = 1 << ((3 * n - 2).bit_length())     # ≥ 3n‑2

    best_score = float("inf")
    it = 0

    while time.time() < stage_deadline:
        it += 1

        conv = _conv_self(a)                # a * a
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - global_start:6.2f}s] adam n={n:5d} "
                  f"iter={it:7d} score={cur_score:.6f}")

        # ---- smoothing schedule (linear in elapsed fraction) ----
        elapsed = time.time() - stage_start
        total = stage_deadline - stage_start
        prog = elapsed / total if total > 0 else 1.0
        beta = beta_start + prog * (beta_end - beta_start)

        # ---- compute the soft‑max weights w_k ----
        # subtract cur_max for numerical stability
        exp_term = np.exp(beta * (conv - cur_max))
        w = exp_term / exp_term.sum()       # Σ w_k = 1, w ≥ 0

        # ---- gradient of the smoothed loss ----
        rev_a = a[::-1]                     # reversed a
        Fa = np.fft.rfft(rev_a, n=L)
        Fw = np.fft.rfft(w, n=L)
        conv_rev = np.fft.irfft(Fa * Fw, n=L)[:3 * n - 2]

        # gradient = 4·n· Σ_j a_j w_{i+j}  (i = 0 … n‑1)
        grad = 4.0 * n * conv_rev[n - 1:n - 1 + n]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)             # keep a on the simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray, n: int,
                   deadline: float, global_start: float,
                   cur_best_score: float, cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   max_moves: int = 500_000):
    """
    Very cheap SA walk that moves a small ε from i to j.
    Returns (best_seq, best_score).
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, max_moves + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / max_moves)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - global_start:6.2f}s] anneal n={n:5d} "
                  f"move={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget, performs progressive up‑sampling,
    Adam optimisation with a smoothed max‑convolution loss, and a final annealing
    refinement.  The best sequence discovered is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness itself
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 0. Optional starter from a previous run (if the harness defines it)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
        else:
            starter = None
    except Exception:
        starter = None

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Define the stage schedule
    # ------------------------------------------------------------------
    stage_sizes = [800, 2400, 4800, 9600, 14400]
    # give larger stages more time (weights)
    stage_weights = [1, 1, 1, 2, 3]
    total_weight = float(sum(stage_weights))

    # Reserve roughly 20 % of the budget for the final annealing walk.
    anneal_budget = max(30.0, 0.20 * (deadline - start_time))
    grad_deadline = deadline - anneal_budget
    remaining_grad_time = grad_deadline - start_time

    # ------------------------------------------------------------------
    # 2. Loop over stages
    # ------------------------------------------------------------------
    for n, w in zip(stage_sizes, stage_weights):
        if time.time() >= grad_deadline:
            break

        # allocate time proportionally to the stage weight
        allocated = remaining_grad_time * (w / total_weight)
        stage_start = time.time()
        stage_deadline = stage_start + allocated

        # initialise the starting point for this stage
        if best_seq is not None:
            a0 = _upsample(best_seq, n)
        elif starter is not None and starter.size == n:
            a0 = starter.copy()
        else:
            # use arcsine on the smallest size and up‑sample
            a0 = _upsample(_initial_arcsine(stage_sizes[0]), n)

        a0 = a0 / a0.sum()                     # enforce simplex

        # ------------------------------------------------------------------
        # 2a. Adam optimisation on the simplex
        # ------------------------------------------------------------------
        a_opt, score_opt = _optimize_adam_stage(
            a0, n, stage_deadline, stage_start, start_time
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} "
                  f"improved score={score_opt:.6f}")

        # update remaining time accounting for what we actually spent
        now = time.time()
        remaining_grad_time = max(0.0, grad_deadline - now)
        total_weight -= w               # remove weight of the completed stage

    # ------------------------------------------------------------------
    # 3. Final annealing phase – use whatever length we ended with.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        a_ref, score_ref = _run_annealing(
            best_seq, n_final,
            deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-6, max_moves=500_000
        )
        if score_ref < best_score:
            best_seq = a_ref.copy()
            best_score = score_ref
            print(f"[{time.time() - start_time:6.2f}s] final anneal improved "
                  f"score={score_ref:.6f}")

    # ------------------------------------------------------------------
    # 4. Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny drift may happen because of rounding)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
