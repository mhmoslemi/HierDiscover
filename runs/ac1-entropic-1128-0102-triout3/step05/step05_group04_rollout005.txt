"""
propose_candidate
=================

Search for a non‑negative step‑function (probability vector) that minimises
``evaluate_sequence``.  The algorithm is a multi‑stage, time‑bounded optimisation
that combines three complementary heuristics:

1. **Adam optimisation of a smooth max‑convolution surrogate**  
   The non‑smooth objective `2·n·max(conv(a,a))` is approximated by
   a log‑sum‑exp (smooth‑max) with a schedule of increasing smoothing
   parameters `α`.  Adam (adaptive moment estimation) performs additive
   updates; after each update the vector is projected onto the probability
   simplex to enforce non‑negativity and unit sum.

2. **Linear‑programming mixing (small scale)**  
   For a modest size (`n≈800`) we solve a linear programme
   ``max sum(g)  s.t.  a * g ≤ M`` (convolution bound `M` from the current
   vector).  The solution `g` is mixed with the current iterate and re‑projected.
   This step corresponds to the “LP‑boost” in the reference method and
   often yields a noticeable drop in the max‑convolution.

3. **Polishing** – projected gradient descent with the exact sub‑gradient of
   the max‑convolution, followed by a short simulated‑annealing phase that
   transfers random mass between coordinates.

The routine respects the supplied time budget (default 1000 s) and prints
progress messages.  The final vector is normalised to sum = 1 and returned
as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution and smoothed‑max gradient
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) self‑convolution a*a."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth‑max
        f(a) = (1/α)·log Σ_k exp(α·conv_k)
    where conv = a*a (linear convolution).
    Returns ∂f/∂a (same shape as a).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w_rev  (gradient = 2·a⋆w_rev)
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    # Pick the central n entries (reverse indexing trick)
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed‑max objective
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              alpha_schedule,
              eta0: float = 0.12,
              steps_per_alpha: int = 5000,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10**7):
    """
    Adam optimisation of the smoothed‑max surrogate.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)
    fft_len = _next_pow2(3 * n - 2)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # update smoothing parameter
        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1
        alpha = alpha_schedule[alpha_idx]

        # gradient of the smooth max (scaled by 2·n)
        grad = _grad_smooth_max(a, conv, alpha, fft_len)
        g = 2.0 * n * grad

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑correction
        t = it
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # diminishing step size (helps after many iterations)
        step = eta0 / (math.sqrt(t) + 1e-16)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max constraint)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.94,
                          step0: float = 0.25,
                          max_iter: int = 10**7):
    """
    Projected gradient descent using the exact sub‑gradient of max(conv).
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # exact sub‑gradient of max(conv)
        max_val = conv.max()
        active = np.where(conv >= max_val * (1.0 - 1e-12))[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        # diminishing step size
        step = step0 / math.sqrt(it)

        # Euclidean projection onto the simplex
        v = a - step * momentum
        a = _project_simplex(v)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250_000):
    """
    Simple annealing that moves a random ε‑fraction of mass from a source
    index (sampled proportionally to its weight) to a uniformly chosen target.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming mixing (small size only)
# ----------------------------------------------------------------------
def _lp_mix(f: np.ndarray,
            n: int,
            M: float,
            mix_factor: float = 0.4,
            max_iter: int = 3):
    """
    Perform up to ``max_iter`` LP mixing steps:
    max Σ g_j   s.t.   f * g ≤ M   ,   g ≥ 0
    The solution g is mixed with f and re‑projected onto the simplex.
    Returns the mixed vector and its new max‑convolution.
    """
    a = f.copy()
    current_M = M

    for _ in range(max_iter):
        # Build the linear constraint matrix A (size (2n‑1) × n)
        # (f * g)[k] = Σ_i f_i * g_{k-i}
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f  # adds f_i to rows i+j, column j

        b = np.full(2 * n - 1, current_M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximise Σ g_j  == minimise -Σ g_j

        # Solve the LP (HiGHS is the default method in recent SciPy)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            break

        g = res.x
        if g.sum() < 1e-12:
            break

        # Mix and re‑project
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _project_simplex(a)

        # Update the bound
        current_M = _conv_self(a).max()
        # early exit if bound does not improve
        if current_M >= M:
            break
        M = current_M

    return a, M


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length while preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up_sum = up.sum()
    if up_sum > 0:
        up /= up_sum
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
