def gradient_step(seq, step):
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    # compute gradient as earlier: For each i, grad_i = 2*seq[k - i] if valid else 0
    grad = np.zeros_like(seq)
    low = max(0, k-(len(seq)-1))
    high = min(k, len(seq)-1)
    idx = np.arange(low, high+1)
    grad[idx] = 2*seq[k-idx]
    new_seq = seq - step * grad
    new_seq = np.clip(new_seq, 0, None)
    # renormalize
    if new_seq.sum() < 1e-8:
        return seq
    new_seq = new_seq / new_seq.sum()
    return new_seq

    occasionally accepts worsening moves with a temperature that cools

most to the current maximal convolution entry to an index that contributes
least.  Improving moves are always accepted; worsening moves are accepted
with a temperature‑controlled probability that cools linearly with the
remaining wall‑clock budget.  Every few thousand iterations a short
gradient step on the current maximal shift is tried – this is the exact
sub‑gradient of the max‑convolution term (∂max_b/∂a_i = 2·a_{k−i} for the
shift k attaining the maximum).

The best sequence seen before the time limit (default 1000 s) is returned.
All numbers are clipped to the range required by ``evaluate_sequence``.
"""

import time
import numpy as np

def _normalise(seq: np.ndarray) -> np.ndarray:
    """Clip to the allowed range, enforce non‑negativity and unit sum."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 1e-12:                     # guard against a zero vector
        seq[:] = 1.0 / len(seq)
        total = 1.0
    return seq / total


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a step‑function (list of floats) with a low evaluation score.

    The routine runs a lightweight simulated‑annealing / gradient‑descent
    loop for at most ``budget_s`` seconds and returns the best sequence
    it discovered.  If a global variable ``height_sequence_1`` exists it
    is used as the starting point; otherwise a uniform sequence is used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1.  Choose a length and an initial sequence.
    # ------------------------------------------------------------------
    # a modest default; we allow a longer vector if a previous one is
    # already larger (more degrees of freedom may help the optimiser).
    n = 5000
    try:
        prev = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if prev.ndim == 1 and prev.size > 0:
            n = max(n, prev.size)
    except Exception:
        prev = None

    if prev is not None:
        seq = prev.copy()
        # adjust length: truncate or pad with zeros
        if seq.size > n:
            seq = seq[:n]
        elif seq.size < n:
            seq = np.concatenate([seq, np.zeros(n - seq.size)])
    else:
        seq = np.full(n, 1.0 / n, dtype=float)

    seq = _normalise(seq)
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    cur_seq = best_seq.copy()
    cur_score = best_score

    print(f"[{time.time() - start_time:.2f}s] start length={n}, "
          f"initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop (simulated annealing + occasional gradient).
    # ------------------------------------------------------------------
    it = 0
    # linear cooling: temperature is 1.0 at the beginning and 0.0 at the end
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:               # leave a little margin
            break

        # Convolution and the shift that attains the maximum.
        conv = np.convolve(cur_seq, cur_seq)
        max_idx = int(np.argmax(conv))

        # Contributions of each index to this maximal shift.
        low = max(0, max_idx - (n - 1))
        high = min(max_idx, n - 1)
        i_range = np.arange(low, high + 1)
        j_range = max_idx - i_range
        contrib = cur_seq[i_range] * cur_seq[j_range]

        # Source = index with the largest product (most responsible for the peak)
        src = int(i_range[np.argmax(contrib)])

        # Target = index with the smallest product (least responsible)
        # If several minima exist we pick one at random.
        mins = np.where(contrib == contrib.min())[0]
        tgt = int(i_range[np.random.choice(mins)])

        if src == tgt:                # fallback to a random different target
            tgt = (src + np.random.randint(1, n)) % n

        # Move a small chunk of probability mass from src to tgt.
        eps = min(cur_seq[src] * 0.5, 0.01)          # never exceed the source value
        if eps <= 0.0:
            eps = 1e-8

        new_seq = cur_seq.copy()
        new_seq[src] -= eps
        new_seq[tgt] += eps
        new_seq = np.maximum(new_seq, 0.0)
        new_seq = _normalise(new_seq)

        new_score = evaluate_sequence(new_seq.tolist())

        # Acceptance test (always accept improvements)
        if new_score < cur_score:
            accept = True
        else:
            # temperature falls linearly from 1.0 to a tiny epsilon
            T = max(1e-6, (budget_s - elapsed) / budget_s)
            prob = np.exp((cur_score - new_score) / T)
            accept = np.random.rand() < prob

        if accept:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"[{elapsed:.2f}s] new best score {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3.  Every few thousand iterations perform a deterministic gradient step.
        # ------------------------------------------------------------------
        if it % 2000 == 0 and it > 0:
            # gradient of the current maximal convolution entry:
            grad = np.zeros_like(cur_seq)
            low_g = max(0, max_idx - (n - 1))
            high_g = min(max_idx, n - 1)
            p = np.arange(low_g, high_g + 1)
            grad[p] = 2.0 * cur_seq[max_idx - p]

            step_size = 0.02 / (1 + it // 2000)
            tmp_seq = cur_seq - step_size * grad
            tmp_seq = np.maximum(tmp_seq, 0.0)
            tmp_seq = _normalise(tmp_seq)

            tmp_score = evaluate_sequence(tmp_seq.tolist())
            if tmp_score < cur_score:
                cur_seq = tmp_seq
                cur_score = tmp_score
                if tmp_score < best_score:
                    best_seq = tmp_seq.copy()
                    best_score = tmp_score
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"gradient improvement {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 4.  Final sanity checks and return.
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    print(f"[{elapsed:.2f}s] optimisation finished, best score {best_score:.6f}")

    best_seq = _normalise(best_seq)
    if best_seq.sum() < 0.01:
        # a safe fallback (should never happen)
        best_seq = np.full(200, 1.0 / 200.0, dtype=float)

    return [float(x) for x in best_seq]
