"""
Propose a low‑peak step function.

The optimiser works on the probability simplex (∑aᵢ = 1) because
`evaluate_sequence` is scale‑invariant.  The method consists of

1. **Multi‑scale warm‑starts** – a good distribution for a short length
   is linearly interpolated to the next larger length, preserving
   the already discovered structure.

2. **Adam‑style projected gradient descent on a log‑sum‑exp surrogate**
   of the ℓ∞‑norm of the self‑convolution.  The temperature λ is
   decayed aggressively so the surrogate converges to the true maximum.

3. **Targeted simulated‑annealing polishing** – after gradient descent a
   cheap mass‑transfer annealer moves a small ε from a source index to a
   destination index.  The temperature schedule is long enough to escape
   tiny plateaus and settle near a local optimum.

The search explores several initial shapes (uniform, exponential,
triangular, edge‑concentrated and random Dirichlet draws) and runs a
few restarts at each length.  The largest length reached is 65536,
still comfortably fitting into the 1000 s budget.  The best sequence
found is returned as a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v_i ≥ 0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (fast for large n)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n, re‑normalised to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate L_λ(a) = λ·log∑exp(conv/λ).

    Parameters
    ----------
    a   : current probability vector (∑a = 1)
    conv: a * a   (length 2n‑1)
    lam : smoothing parameter λ > 0

    Returns
    -------
    grad: dense gradient vector (shape n,)
    """
    n = a.shape[0]
    # stable soft‑max over convolution entries
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights_sum = weights.sum()
    if weights_sum == 0.0:
        # fall back to uniform weights (should not happen)
        weights = np.full_like(conv, 1.0 / conv.size)
    else:
        weights /= weights_sum

    # cross‑correlation of the weight vector with the reversed a
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]          # dL/da_i
    return grad


def _score_from_vector(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv) (∑a = 1)."""
    n = a.shape[0]
    max_b = _conv_self(a).max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """A small collection of plausible starting vectors."""
    n = base_seq.shape[0]
    variants = [base_seq.copy()]

    # uniform
    uni = np.full(n, 1.0 / n)
    variants.append(uni)

    # exponential decay (left‑heavy)
    beta = np.random.uniform(0.001, 0.02)
    exp = np.exp(-beta * np.arange(n))
    exp /= exp.sum()
    variants.append(exp)

    # reversed exponential (right‑heavy)
    variants.append(exp[::-1])

    # triangular
    tri = np.concatenate(
        [np.linspace(1, n // 2, n // 2, endpoint=False),
         np.linspace(n // 2, 1, n - n // 2)]
    )
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # edge‑concentrated (mass at both ends)
    edge_frac = min(0.07, 5 / n)                     # at most a few percent
    L = max(1, int(edge_frac * n))
    edge = np.concatenate([np.ones(L), np.zeros(n - 2 * L), np.ones(L)])
    if edge.sum() > 0:
        edge = edge / edge.sum()
        variants.append(edge)

    # random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    return variants


# ----------------------------------------------------------------------
# Adam‑style optimisation on the simplex (smooth ℓ∞ surrogate)
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best vector encountered and its exact objective value
    2·n·max(conv) (the sequence stays on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    # aggressive λ‑decay parameters
    lam_factor = 0.5          # initial multiplier of max(conv)
    lam_decay = 0.95          # multiplicative decay per 500 iterations
    lam_decay_interval = 500

    while time.time() < slice_deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = _score_from_vector(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor *= lam_decay
        lam = max(conv.max() * lam_factor, 1e-12)

        # gradient of the smooth surrogate
        grad = _gradient_from_conv(a, conv, lam)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing base step (√it schedule works well)
        base_step = 0.6 / math.sqrt(it + 1.0)

        # parameter update
        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple mass‑transfer annealing (polishing)
# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 250000) -> tuple[np.ndarray, float]:
    """
    Cheap annealer: move a small ε from a source index to a destination index.
    Returns an improved sequence (if any) and its exact score.
    """
    a = a0.copy()
    score = _score_from_vector(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # propose moving up to 20 % of the source mass
        eps = a[i] * np.random.rand() * 0.20
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = _score_from_vector(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score
