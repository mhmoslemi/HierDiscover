def _subgradient_multi(a, conv, tol=1e-12):
    """
    Compute subgradient of max(conv) using all indices within tolerance.
    Returns averaged subgradient vector.
    """
    max_val = np.max(conv)
    # active indices: conv >= max_val - tol (absolute)
    # but to avoid many due to floating error, we can use relative tolerance.
    active = np.where(conv >= max_val - max(tol, max_val*1e-12))[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        # base contribution 2*a_j
        g[mask] += 2.0 * a[j[mask]]
        # diagonal where i=j (2*i = k)
        diag = mask & (2 * idx == k)
        if np.any(diag):
            # subtract a_i (since we added 2*a_i)
            g[diag] -= a[idx[diag]]
    # average
    if len(active) > 0:
        g /= len(active)
    return g

# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------

                       abs_tol: float = 1e-12, rel_tol: float = 1e-12) -> np.ndarray:

    it is a_i.

    max_val = conv.max()
    # indices whose value is within tolerance of the maximum
    tol = max(abs_tol, rel_tol * max_val)
    active = np.where(conv >= max_val - tol)[0]

    n = a.shape[0]
    g = np.zeros_like(a)

    # vector of all i indices
    idx = np.arange(n)

    for k in active:
        j = k - idx                     # partner index for each i
        mask = (j >= 0) & (j < n)       # valid pairs (i,j)
        # base contribution 2*a_j
        g[mask] += 2.0 * a[j[mask]]
        # diagonal correction: when i == j (i.e. 2*i == k) we counted twice
        diag = mask & (2 * idx == k)
        if np.any(diag):
            g[diag] -= a[idx[diag]]     # change 2*a_i -> a_i
    if active.size > 0:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _run_expgrad(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 step0: float = 0.5,
                 decay: float = 0.5,
                 max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on F(a)=2·n·max(conv(a)),
    keeping a on the probability simplex (sum=1, a_i≥0).  Returns the
    best vector seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] expgrad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient (averaged over all active max entries)
        grad = _subgradient_multi(a, conv)

        # diminishing step size η_t = step0 / (t^decay)
        step = step0 / (it ** decay)

        # exponentiated update, staying non‑negative
        a = a * np.exp(-step * grad)

        # renormalise to sum = 1  (simplex projection for a positive vector)
        s = a.sum()
        if s == 0.0:
            # fallback to uniform if underflow occurs
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves guided by gradient)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 4e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple annealing: move a fraction of mass from a gradient‑large index
    to a random destination.  Accepts moves with Metropolis criterion.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # compute current gradient (helps bias source selection)
        grad = _subgradient_multi(a, _conv_self(a))

        # source: pick proportional to |grad| (large gradient ⇒ reduction needed)
        prob = np.abs(grad)
        prob_sum = prob.sum()
        if prob_sum == 0.0:
            prob = a                       # fall back to mass‑biased
            prob_sum = prob.sum()
        prob = prob / prob_sum
        src = rng.choice(n, p=prob)

        # destination: uniform random
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 30 % of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        # keep global best
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) from old length to new length
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine follows a multi‑stage scheme (random restarts → up‑sampling
    → exponentiated‑gradient descent → annealing) and respects the supplied
    time budget.  Returns the best sequence as a plain Python list of floats.
    """
    # Initialise RNG
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optionally bootstrap from a previously stored solution
    # ------------------------------------------------------------------
    starter = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(globals()["height_sequence_1"], dtype=float)
            if arr.ndim == 1 and arr.size > 0:
                starter = arr / arr.sum()
        except Exception:
            starter = None

    # ------------------------------------------------------------------
    # Stage 0 – many short runs on a modest size (n≈1024)
    # ------------------------------------------------------------------
    n0 = 1024
    slice_time = max(30.0, 0.12 * (deadline - time.time()))
    slice_deadline = time.time() + slice_time

    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline:
            break

        if starter is not None and r == 0:
            a0 = _upsample(starter, n0)
        else:
            a0 = rng.random(n0)
            a0 /= a0.sum()

        # Small randomisation of hyper‑parameters
        step0 = rng.uniform(0.45, 0.55)    # initial learning rate
        decay = 0.5                       # 1/√t decay

        a_opt, scr = _run_expgrad(
            a0, n0, slice_deadline, start_time,
            step0=step0,
            decay=decay,
            max_iter=5_000_000,
        )
        if scr < best_score:
            best_score = scr
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 0 improve: restart={r}, score={scr:.6f}")

    # Fallback if nothing produced
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0)
        best_score = 2.0 * n0 * _conv_self(best_seq).max()

    # ------------------------------------------------------------------
    # Stage 1 – medium size (n≈4096) and longer optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n1 = 4096
    a1 = _upsample(best_seq, n1)

    remaining = deadline - time.time()
    grad_time1 = max(20.0, 0.35 * remaining)
    grad_deadline1 = time.time() + grad_time1

    a1_opt, scr1 = _run_expgrad(
        a1, n1, grad_deadline1, start_time,
        step0=0.5,
        decay=0.5,
        max_iter=10_000_000,
    )
    if scr1 < best_score:
        best_score = scr1
        best_seq = a1_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 1 improve: score={scr1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – large size (n≈12288) and continued optimisation
    # ------------------------------------------------------------------
    if time.time() >= deadline:
        return [float(x) for x in best_seq]

    n2 = 12288
    a2 = _upsample(best_seq, n2)

    remaining = deadline - time.time()
    grad_time2 = max(15.0, 0.30 * remaining)
    grad_deadline2 = time.time() + grad_time2

    a2_opt, scr2 = _run_expgrad(
        a2, n2, grad_deadline2, start_time,
        step0=0.45,
        decay=0.5,
        max_iter=12_000_000,
    )
    if scr2 < best_score:
        best_score = scr2
        best_seq = a2_opt.copy()
        print(f"[{time.time() - start_time:6.2f}s] stage 2 improve: score={scr2:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing refinement (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        a_ref, scr_ref = _run_annealing(
            best_seq,
            best_seq.shape[0],
            anneal_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if scr_ref < best_score:
            best_score = scr_ref
            best_seq = a_ref.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={scr_ref:.6f}")

    # ------------------------------------------------------------------
    # Normalise to sum = 1 and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Optional: log the final evaluation (does not affect returned value)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final evaluate_sequence score = {final_val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
