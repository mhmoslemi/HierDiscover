def _conv_self_fft(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2*n - 1]
    return conv

def _local_search(a, conv, deadline, start_time,
                  max_move_frac=0.3, start_temp=0.01, end_temp=1e-6,
                  max_iter=10**7):
    n = a.shape[0]
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # temperature schedule based on elapsed fraction
        elapsed = time.time() - start_time
        total_time = deadline - start_time
        if total_time <= 0:
            break
        T = start_temp * (end_temp / start_temp) ** ((elapsed) / total_time)
        # select source i weighted by a
        # If a is all zero maybe not; but will have non-zero
        i = np.random.choice(n, p=a)  # probability proportional to mass
        # ensure a[i] >0, else skip
        if a[i] <= 0:
            continue
        # choose dest j
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1  # ensure j != i
        # epsilon up to max_move_frac * a[i], random fraction
        eps = a[i] * np.random.rand() * max_move_frac
        if eps <= 0:
            continue
        # compute temporary conv with updates
        conv_new = conv.copy()
        # subtract and add slices
        conv_new[i:i+n] -= 2*eps * a
        conv_new[j:j+n] += 2*eps * a
        # delta-delta term
        conv_new[2*i] += eps*eps
        conv_new[2*j] += eps*eps
        conv_new[i+j] -= 2*eps*eps
        # compute new max
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        # acceptance criteria
        delta = new_score - cur_score
        if delta < 0 or np.exp(-delta / (T+1e-15)) > np.random.rand():
            # accept
            a[i] -= eps
            a[j] += eps
            conv = conv_new
            cur_max = new_max
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_conv = conv.copy()
                print(f"[{time.time() - start_time:6.2f}s] local i={i} j={j} eps={eps:.6e} score={cur_score:.6f} it={it}")
        # optionally short-circuit if it > some threshold
    return best_a, best_conv, best_score

"""
Lightweight hill‑climbing optimizer for the step‑function problem.

The algorithm:
1. Initialise a probability vector on a modest size (2000) – a discretised
   arcsine (Beta(½,½)) density or, if provided, the previous best sequence.
2. Repeatedly move a small amount of mass from one random position to another.
   The effect on the full self‑convolution is computed analytically in O(n)
   using slice updates, so thousands of moves per second are possible.
3. Moves are accepted by a simple simulated‑annealing schedule; downhill moves
   are always kept.
4. After a fixed time budget on the current size we up‑sample the best
   vector (linear interpolation, re‑normalised) to a larger size (4000, 8000)
   and repeat the hill‑climbing phase.
5. The best vector found is returned as a plain Python list.

The implementation respects the 1000 s budget, never uses I/O and works on
the simplex (sum = 1, all entries ≥ 0), which matches the invariance of
``evaluate_sequence``.
"""

import time
import numpy as np

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _local_search(a: np.ndarray,
                  conv: np.ndarray,
                  deadline: float,
                  start_time: float,
                  max_move_frac: float = 0.3,
                  start_temp: float = 5e-2,
                  end_temp: float = 1e-6,
                  max_iter: int = 10**7) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Hill‑climbing / simulated‑annealing on the simplex.
    Moves a small amount of probability mass between two indices and updates
    the self‑convolution analytically (O(n)).  The best point seen is returned.
    """
    n = a.shape[0]
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = cur_score

    total_time = deadline - start_time
    it = 0
    # small helper to occasionally re‑normalise and recompute the convolution
    def _recalc():
        nonlocal a, conv, cur_max, cur_score
        s = a.sum()
        if s != 0.0:
            a /= s
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

    while time.time() < deadline and it < max_iter:
        it += 1
        elapsed = time.time() - start_time
        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (elapsed / total_time)

        # choose source index weighted by current mass
        if np.all(a <= 0):
            break
        i = np.random.choice(n, p=a)
        if a[i] <= 0.0:
            continue

        # choose destination index uniformly (different from i)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # amount to transfer – up to max_move_frac of the source mass
        eps = a[i] * np.random.rand() * max_move_frac
        if eps <= 0.0:
            continue

        # ---- compute the new convolution -------------------------------------------------
        conv_new = conv.copy()
        # the linear part of the update (2*eps*a shifted)
        conv_new[i:i + n] -= 2.0 * eps * a
        conv_new[j:j + n] += 2.0 * eps * a
        # quadratic delta‑delta terms (at three indices)
        conv_new[2 * i] += eps * eps
        conv_new[2 * j] += eps * eps
        conv_new[i + j] -= 2.0 * eps * eps

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # ---- Metropolis acceptance --------------------------------------------------------
        delta = new_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if T > 0.0 and np.exp(-delta / T) > np.random.rand():
                accept = True

        if accept:
            # apply the mass transfer
            a[i] -= eps
            a[j] += eps
            # update the stored convolution
            conv = conv_new
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_conv = conv.copy()
                print(f"[{time.time() - start_time:6.2f}s] new best {best_score:.6f} it={it}")

            # occasional re‑normalisation to combat rounding errors
            if it % 1000 == 0:
                _recalc()
        # end of iteration
    return best_a, best_conv, best_score

    # ----------------------------------------------------------------------
    stage_ns = [2000, 4000, 8000]
    num_stages = len(stage_ns)

        remaining = deadline - time.time()
        stages_left = num_stages - stage_idx
        stage_deadline = time.time() + remaining / stages_left

        # initialise the probability vector for this stage
        if best_seq is not None:
            a = _upsample(best_seq, n)
        elif starter is not None:
            a = _upsample(starter, n)
        else:
            # discretised arcsine density (Beta(0.5,0.5))
            xs = (np.arange(n) + 0.5) / n
            a = 1.0 / np.sqrt(xs * (1.0 - xs))
            a = a / a.sum()

        # compute the exact self‑convolution for the start point
        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} init score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # 2. Local hill‑climbing / annealing phase
        # ------------------------------------------------------------------
        a_opt, conv_opt, score_opt = _local_search(
            a, conv, stage_deadline, start_time,
            max_move_frac=0.3,
            start_temp=5e-2,
            end_temp=1e-6,
            max_iter=10**7,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage n={n} improve: score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 3. Safety fallback – uniform distribution if nothing was found
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(500, 1.0 / 500.0)

    # final normalisation (tiny drift may appear)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
