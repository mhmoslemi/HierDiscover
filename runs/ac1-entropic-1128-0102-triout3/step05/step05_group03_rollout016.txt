"""
Propose a non‑negative step function that (approximately) minimizes
`evaluate_sequence`.  

The algorithm consists of three phases:

1. **Initialization** – a mixture of a deterministic “arcsine” shape
   (a_i ∝ 1/√((i+½)(n‑i‑½))) and a random simplex point.
2. **Flattening** – minimise the squared deviation of the self‑convolution
   from its average value (which is 1/(2n‑1) after normalising the sum to 1).
   The loss gradient is computed by a single FFT‑based convolution, and the
   iterates are projected onto the probability simplex.
3. **Max‑optimisation** – after the convolution is roughly flat we switch to a
   sub‑gradient descent that directly reduces `max(conv)`.  A Nesterov momentum
   scheme with a diminishing step‑size is used.  Finally a short simulated‑annealing
   move set is applied for fine‑tuning.

The routine respects a user‑supplied time budget (default 1000 s).  It tries
two problem sizes (≈8 k and ≈16 k) and keeps the best sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) computed with FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (unique maximum case)
# ----------------------------------------------------------------------
def _subgradient_max_correct(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) (assumes a unique maximiser)."""
    n = a.shape[0]
    kmax = int(np.argmax(conv))
    idx = np.arange(n)
    i = kmax - idx
    mask = (i >= 0) & (i < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[i[mask]]
    return g


# ----------------------------------------------------------------------
# Upsample (linear interpolation) from one length to another
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` and renormalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Very cheap annealing that moves a small ε from a source index to a destination."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (geometric)
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30   # up to 30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
