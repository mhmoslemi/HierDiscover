"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1.  **Parameterisation** – work in the log‑domain:
        a_i = softmax(y)_i = exp(y_i)/∑_j exp(y_j).  This automatically
        enforces a_i ≥ 0 and ∑ a_i = 1.

2.  **Smooth objective** – minimise the smoothed max of the self‑convolution
        f(y) = 2·n·(1/α)·log∑_k exp(α·conv_k(a(y))).
    The gradient w.r.t. the logits y is obtained by the chain rule:
        ∂f/∂a = 2·n·∂(log‑sum‑exp)/∂a
        ∂f/∂y = a ∘ (∂f/∂a – (∂f/∂a)·a)

3.  **Adam optimiser** – run a few Adam passes with an *increasing*
    smoothing parameter α (α‑schedule).  The optimiser stays in the simplex
    without any explicit projection.

4.  **Exact‑subgradient polishing** – a short projected‑gradient descent
    on the true (non‑smoothed) max‑convolution, using the sub‑gradient
    routine from the reference implementation.

5.  **Simulated‑annealing fine‑tuning** – mass‑transfer moves that explore
    the neighbourhood of the polished solution.

6.  **Time management** – the whole routine respects the supplied time budget
    (≈1000 s).  It aborts early if the deadline is reached.

The combination of a smooth Adam phase, exact sub‑gradient polishing and
a final annealing stage typically yields a lower bound than the previous
baseline (≈1.513) and often reaches the target ≈1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of      f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (shape (n,)).
    """
    n = a.shape[0]

    # soft‑max weights over the convolution entries
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)
    w /= w.sum()

    # cross‑correlation a ⋆ w   (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Projected gradient descent (exact sub‑gradient)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7):
    """
    Simple projected gradient descent with momentum on the exact max‑convolution.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        # sub‑gradient of max(conv)
        max_val = conv.max()
        # active set (within relative tolerance)
        rel_tol = 1e-12
        thresh = max_val * (1.0 - rel_tol)
        active = np.where(conv >= thresh)[0]

        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            g[i] += a[j]
            g[j] += a[i]

        # momentum update (Nesterov‑style)
        momentum = beta * momentum + (1.0 - beta) * g

        step = step0 / math.sqrt(it)

        # Euclidean projection onto the probability simplex
        v = a - step * momentum
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score

# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000):
    """
    Small‑scale simulated annealing that moves a random ε‑fraction of mass
    from a source index to a destination index.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score

            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---------- occasionally evaluate the exact score ----------
            if i % 200 == 0:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, iter={i:,}, score={sc:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Exact sub‑gradient projected gradient polishing (short)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            n,
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.25,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing fine‑tuning (use any remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=5e-2,
            end_temp=1e-6,
            steps=250_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against accidental drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    return [float(x) for x in best_seq.tolist()]
