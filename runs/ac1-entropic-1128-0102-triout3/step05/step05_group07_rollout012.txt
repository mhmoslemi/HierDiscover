    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    n = 1800
    best_seq = None
    best_score = float('inf')
    iteration_counter = 0

    # Generate initial candidates: include height_sequence_1 if available, uniform, Hann, Dirichlet draws, random spikes.
    pool = []
    # ... as before

    def score_of(seq):
        # seq is numpy array already normalized sum=1
        max_conv, _ = _conv_max(seq) # defined earlier
        return 2.0 * n * max_conv

    # define modifications: mutation function
    def mutate(seq, scale=0.05):
        # add Gaussian noise then project to simplex
        noise = np.random.randn(n) * scale
        new = seq + noise
        new = np.maximum(new, 0.0)
        if new.sum() == 0:
            new = np.full_like(new, 1.0/n)
        else:
            new = new/ new.sum()
        return new

    # main loop
    while time.time() < deadline:
        # choose a starting point: either from pool or mutate best
        if best_seq is None:
            cand = pool[np.random.randint(len(pool))]

            # with probability 0.5 start from best mutated, else random pool
            if np.random.rand() < 0.7:

            else:
                cand = pool[np.random.randint(len(pool))]
        # Now refine cand using a mixture of Adam and EG
        # Determine available time for this iteration:
        remaining = deadline - time.time()
        # allocate some number of iterations based on remaining time.

                                            deadline=deadline,

        cand_refined = a_adam
        # Then EG refinement
        a_eg, score_eg = _exp_gradient_optimize(cand_refined.copy(), max_iters=int(0.5*max_iters),

        cand_refined = a_eg
        # compute score w/ conv_max (fast)
        cur_score = score_of(cand_refined)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand_refined.copy()
            print(f"New best score {best_score:.6f} at time {time.time()-start:.2f}s")
        # add cand to pool for later use?
        # Could also keep local best for later mutations.

    # after time exhausted: final refinement of best_seq with many EG iterations
    # allocate any leftover time
    remaining = deadline - time.time()

        est_iter_time = 8e-4 * (n / 1200.0) + 1e-4
        final_iters = int(min(500000, remaining / est_iter_time))
        if final_iters > 0:
            a_final, score_final = _exp_gradient_optimize(best_seq.copy(), max_iters=final_iters,

    # Convert best_seq to list, ensure positivity and sum normalization:

implementation re‑uses fast FFT‑based convolution and gradient
computations and adapts the number of iterations to the remaining time,
so it never exceeds the 1000 s budget.
"""

import time

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (max(conv(a,a)), full linear convolution)."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·conv_k w.r.t. `a`.  Implements
    g = 2·(a_rev * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   max_iters: int,
                   deadline: float,
                   alpha: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   temp_start: float = 0.07,
                   temp_end: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best vector seen and its (raw) score = 2·n·max(conv).
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # track the best seen
    max_val, _ = _conv_max(a)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights over the convolution entries
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # sub‑gradient of the smooth‑max
        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_optimize(a: np.ndarray,
                           max_iters: int,
                           deadline: float,
                           eta_start: float = 0.8,
                           eta_end: float = 0.001,
                           eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimiser that uses the *hard* max
    of the convolution (sub‑gradient of the L∞‑norm).  Returns the best
    vector encountered and its raw score.
    """
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        max_val, conv = _conv_max(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask of (approximately) active anti‑diagonals
        mask = conv >= max_val - eps
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient for the exact max‑norm
        g = _gradient_from_weights(a, w, fft_len_grad)

        # decreasing step size (geometric schedule)
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        # multiplicative update with safe clipping before the exponent
        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)          # exp(±30) ~ 1e13, safe
        a = a * np.exp(upd)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score


# ----------------------------------------------------------------------
