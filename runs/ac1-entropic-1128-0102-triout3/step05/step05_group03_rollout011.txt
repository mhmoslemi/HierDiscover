"""
Improved search for a non‑negative step function that minimises
`evaluate_sequence`.  The core idea is a *smooth approximation* of the
max‑convolution objective (log‑sum‑exp) and a projected Adam optimiser.
A continuation on the smoothing parameter λ is used: we start with a
very smooth landscape (small λ) so that the gradient points in a useful
direction, then increase λ so that the optimiser focuses on the true
max‑norm.  The routine works in three increasing resolutions (≈1k,
≈4k, ≈12k) and finishes with a short simulated‑annealing fine‑tune.
All helpers are defined at top level and respect the time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x≥0, Σx=1})
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based linear convolution (full length)
# ----------------------------------------------------------------------
def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    conv = np.fft.irfft(A * B, L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Self‑convolution (a * a) – used many times
# ----------------------------------------------------------------------
def self_conv(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution, length 2·len(a)‑1)."""
    return fft_convolve(a, a)


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def grad_softmax_conv(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of   F(a) = (1/lam) * log Σ_k exp(lam * (a*a)_k)
    with respect to a (projected gradient will be taken later).
    """
    n = a.shape[0]
    conv = self_conv(a)                     # length 2n-1
    # soft‑max weights (stable)
    maxc = conv.max()
    w_exp = np.exp(lam * (conv - maxc))
    w = w_exp / w_exp.sum()                 # Σ w = 1, length 2n‑1

    # gradient = 2 * correlation(a, w)
    # correlation can be obtained from convolution with w reversed
    w_rev = w[::-1]
    corr_full = fft_convolve(a, w_rev)      # length 3n‑2
    # indices 2n‑2‑i give Σ_j a_j w_{i+j}
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * corr_full[idx]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def run_smooth_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    lam0: float = 5.0,
    lam_final: float = 5e4,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    step0: float = 0.3,
    max_iter: int = 10 ** 7,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    Returns the best vector seen and its true (unsmoothed) score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = self_conv(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"adam improve: n={n}, iter={it:,}, score={cur_score:.6f}"
            )

        # continuation on λ  (log‑linear schedule)
        lam = lam0 * (lam_final / lam0) ** (it / max_iter)

        # gradient of the smoothed surrogate
        g = grad_softmax_conv(a, lam)

        # Adam update (with simple diminishing global step)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size  (step0 / sqrt(it))
        step = step0 / math.sqrt(it)

        a = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a = project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def run_annealing(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 4e-2,
    end_temp: float = 1e-6,
    steps: int = 250_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    score = 2.0 * n * self_conv(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick a source weighted by current mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.40
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * self_conv(a_prop).max()
        delta = new_score - score

        # Metropolis test
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve: "
                f"n={n}, iter={it:,}, score={score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample (linear interpolation) + re‑normalisation
# ----------------------------------------------------------------------
def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` to length `new_n` and re‑normalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
