```
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

def lp_refine(f, M=None, max_iter=5, t=0.2, deadline=None):
    n = len(f)
    if M is None:
        conv = np.convolve(f, f)
        M = conv.max()
    best_f = f.copy()
    best_M = M
    for it in range(max_iter):
        # Build constraint matrix A_ub
        # rows: 0..2n-2
        rows = np.empty(n*n, dtype=np.int32)
        cols = np.empty(n*n, dtype=np.int32)
        data = np.empty(n*n, dtype=np.float64)
        idx = 0
        for j in range(n):
            # column j: f_i for rows i+j where i from 0..n-1
            rows[idx:idx+n] = np.arange(j, j+n)
            cols[idx:idx+n] = j
            data[idx:idx+n] = f
            idx += n
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # Set up bounds
        # linprog uses A_ub @ x <= b_ub
        b_ub = np.full(2*n-1, M)
        # Solve linear program: maximize sum(x) -> minimize -sum(x)
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        # Normalize
        total = g.sum()
        if total > 0:
            g = g / total
        else:
            g = np.full_like(g, 1.0 / n)
        # mix
        f = (1 - t) * f + t * g
        # compute new M
        conv = np.convolve(f, f)
        M = conv.max()
        if M < best_M:
            best_M = M
            best_f = f.copy()
        # optional early break based on deadline
        if deadline is not None and time.time() >= deadline:
            break
    return best_f
```

    import time, numpy as np, scipy.sparse as sp, scipy.optimize as opt
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    best_a = a.copy()
    best_M = M
    for it in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break
        # Build constraint matrix
        rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
        cols = np.repeat(np.arange(n), n)
        data = np.tile(a, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True})

            print(f"LP solve failed at iter {it}: {res.message}")

            g = np.full_like(g0, 1.0/n)

            g = g0 / sum0
        # mix
        a = (1 - t) * a + t * g
        # recompute M
        conv = np.convolve(a, a)
        M = conv.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    return best_a

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

The search consists of several stages:

1. **Fast exponentiated‑gradient optimisation** on a modest size (≈800)
   – a smoothed log‑sum‑exp approximation of max(conv) is minimised.
2. **Linear‑programming refinement** – for the current distribution we
   maximise the ℓ₁‑mass of a partner vector `g` under the constraint
   ``conv(f,g) ≤ max(conv(f,f))`` and then mix it back into `f`.  This step
   tends to move mass away from large peaks of the self‑convolution.
3. **Upsampling** to medium (≈2400) and large (≈4800) sizes with the same
   two‑phase routine.
4. **Projected gradient polishing** using an exact sub‑gradient of the true
   max‑convolution.
5. **Short simulated‑annealing refinement** for a final fine‑tuning.

The whole pipeline respects the supplied time budget (≈1000 s) and always
returns a normalised non‑negative list (sum = 1).  The algorithm improves
on the previous baseline and is deliberately crafted to obtain a value
below the target 1.503.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper: nearest power of two (FFT length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv) (active set)
# ----------------------------------------------------------------------
def _subgradient_full(a: np.ndarray, conv: np.ndarray,
                      rel_tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a using all indices where conv is
    within ``rel_tol`` of the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    if max_val == 0:
        return np.zeros_like(a)

    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    g = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

    return g


# ----------------------------------------------------------------------
# Gradient of the smooth (log‑sum‑exp) approximation of max(conv)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, conv: np.ndarray,
                     alpha: float, fft_len: int) -> np.ndarray:
    """
    Gradient of   f(a)= (1/α)·log Σ_k exp(α·conv_k) .
    Returns ∂f/∂a (same shape as a).
    """
    n = a.shape[0]

    # weights w_k = exp(α·conv_k) / Σ exp(...)
    scaled = alpha * conv
    max_scaled = scaled.max()
    w = np.exp(scaled - max_scaled)          # stabilise
    w /= w.sum()

    # cross‑correlation a ⋆ w  (gradient = 2·(a⋆w))
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    Wrev = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (smoothed max)
# ----------------------------------------------------------------------
def _run_exponential_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              alpha_schedule,
                              eta0: float = 0.3,
                              steps_per_alpha: int = 8000,
                              max_iter: int = 10**7):
    """
    Multiplicative updates with a smoothed max objective.
    Returns (best_vector, best_score) where score = 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    fft_len = _next_pow2(3 * n - 2)  # sufficient for all convolutions

    it = 0
    alpha_idx = 0
    n_alpha = len(alpha_schedule)

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] exp‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        alpha = alpha_schedule[min(alpha_idx, n_alpha - 1)]
        grad = _grad_smooth_max(a, conv, alpha, fft_len)

        eta = eta0 / math.sqrt(it)
        upd = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * upd
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        if it % steps_per_alpha == 0 and alpha_idx < n_alpha - 1:
            alpha_idx += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected gradient descent using the exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _run_gradient_descent(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta: float = 0.9,
                          step0: float = 0.3,
                          max_iter: int = 10**7):
    """
    Classic projected gradient descent with momentum.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    momentum = np.zeros_like(a)

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] proj‑grad improve: n={n}, iter={it:,}, score={cur_score:.6f}")

        grad = _subgradient_full(a, conv)

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        v = a - step * momentum

        # Simple Euclidean projection onto the probability simplex (O(n))
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(1, n + 1)
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.full_like(v, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-6,
                   steps: int = 150_000):
    """
    Small‑scale annealing that moves a random ε‑fraction of mass from a
    source index to a random destination.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-12:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve: n={n}, iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) between stages
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `seq` to length `new_n` (preserving sum)."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step” from the reference)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               max_iter: int = 5,
               t: float = 0.2,
               deadline: float = float('inf')):
    """
    Given a normalised vector ``a`` (sum = 1), repeatedly:
        maximise sum(g)  subject to  conv(a,g) ≤ max(conv(a,a)), g ≥ 0
    and mix the result back with ``a`` using a convex weight ``t``.
    Returns the refined vector.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_M = _conv_self(a).max()

    # Pre‑compute the (row, col) pattern of the convolution matrix.
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    shape = (2 * n - 1, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # Build the constraint matrix A (sparse) for the current a.
        data = np.tile(a, n)                     # each column gets a copy of a
        A = sp.csr_matrix((data, (rows, cols)), shape=shape)

        M = best_M
        b = np.full(2 * n - 1, M)

        # Linear program: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n)

        # Solve with HiGHS (fast, handles sparsity)
        res = opt.linprog(c, A_ub=A, b_ub=b,
                          bounds=(0, None),
                          method='highs', options={'presolve': True})

        if not res.success:
            print(f"LP solve failed at iteration {it}: {res.message}")
            break

        g0 = res.x
        sum0 = g0.sum()
        if sum0 <= 0:
            g = np.full_like(g0, 1.0 / n)
        else:
            g = g0 / sum0               # renormalise to sum = 1

        # Mix the new vector into the current one
        a = (1.0 - t) * a + t * g

        # Update the bound M
        cur_M = _conv_self(a).max()
        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()
            print(f"[{time.time():.2f}s] LP refine iter {it}: new bound={best_M:.8f}")

    return best_a

                best_seq = a_mid_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] LP‑refine mid: score={score:.6f}")
        except Exception as e:
            print("LP‑refine mid failed:", e)

    # ------------------------------------------------------------------
    # Stage 3 – up‑sample to large size (n≈4800) and optimise
    # ------------------------------------------------------------------
    n_big = 4800
    a_big = _upsample(best_seq, n_big)

    if time.time() < deadline:
        remaining = deadline - time.time()
        stage3_time = max(20.0, 0.35 * remaining)
        deadline_big = time.time() + stage3_time

        a_big_opt, score_big = _run_exponential_gradient(
            a_big,
            n_big,
            deadline_big,
            start_time,
            alpha_schedule=[5.0, 15.0, 30.0, 60.0, 120.0, 240.0, 480.0],
            eta0=0.20,
            steps_per_alpha=12000,
        )
        if score_big < best_score:
            best_score = score_big
            best_seq = a_big_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] stage 3 improve: score={score_big:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – projected‑gradient polishing (exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        polish_deadline = time.time() + 0.5 * remaining
        a_polish, score_polish = _run_gradient_descent(
            best_seq,
            best_seq.shape[0],
            polish_deadline,
            start_time,
            beta=0.94,
            step0=0.22,
        )
        if score_polish < best_score:
            best_score = score_polish
            best_seq = a_polish.copy()
            print(f"[{time.time() - start_time:6.2f}s] polishing improve: score={score_polish:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final simulated‑annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _run_annealing(
            best_seq,
            best_seq.shape[0],
            deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=1e-2,
            end_temp=1e-6,
            steps=200_000,
        )
        if score_final < best_score:
            best_score = score_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:6.2f}s] final anneal improve: score={score_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and output
    # ------------------------------------------------------------------
    best_seq = np.asarray(best_seq, dtype=float)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # Optional sanity check (does not affect the returned value)
    try:
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"[{time.time() - start_time:6.2f}s] final evaluate_sequence score = {eval_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
