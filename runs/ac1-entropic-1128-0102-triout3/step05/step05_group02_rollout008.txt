def convolve_fft(x, y): # returns linear conv using FFT
    L = 1 << ((len(x)+len(y)-2).bit_length())
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X*Y, n=L)[:len(x)+len(y)-1]
    return conv

    a = a0.copy()

"""
Low‑maximum self‑convolution optimizer.

The algorithm runs for a given time budget (default 1000 s) and returns a
non‑negative step‑function (list of heights) that minimizes

    2·n·max(a∗a) / (sum(a)²)

where *∗* denotes the linear convolution.  The optimisation proceeds in
stages:

1. **Warm‑starts** – short random search in a few analytically‑defined
   families (power‑product, truncated Gaussian, Beta‑shaped) supplies a
   good initial shape for each length.
2. **Smoothed sub‑gradient descent** – the ∞‑norm of the self‑convolution
   is approximated by a smooth soft‑max (weighted by exp(α·(c–c_max))).
   The resulting gradient is projected onto the probability simplex.
   The smoothing parameter α is increased gradually, turning the smooth
   problem into the original one.
3. **Exact sub‑gradient polish** – a classic projected sub‑gradient step
   on the true max‑entry (as in the reference code) refines the result.
4. **Cheap mass‑transfer annealing** – a final simulated‑annealing pass
   makes tiny ε‑moves between coordinates to squeeze out a few more
   improvements.

Lengths are processed in increasing order; after each length the best
vector is up‑sampled (linear interpolation) and used as a starter for
the next larger size.  The time budget is split dynamically among the
stages, always guaranteeing a return before the deadline.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real‑valued, fast)."""
    size = len(x) + len(y) - 1
    fft_len = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:size]
    return conv


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·n‑1)."""
    return convolve_fft(a, a)


def score_of(a: np.ndarray) -> float:
    """Exact objective for a normalised (sum=1) vector."""
    n = a.shape[0]
    max_conv = conv_self_fft(a).max()
    return 2.0 * n * max_conv


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length seq.shape[0] to new_n, re‑normalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Simple parametric families (warm‑starts)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term == 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (
        np.maximum(1.0 - x, eps) ** (beta_par - 1.0)
    )
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smoothed sub‑gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def smoothed_subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on a smooth approximation of the
    max‑convolution.  The smoothing parameter α grows over time, turning
    the problem into the true ∞‑norm.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.8
    it = 0
    alpha = 1.0  # start smoothing

    while time.time() < deadline:
        it += 1

        # current convolution and its maximum
        conv = conv_self_fft(a)
        max_conv = conv.max()

        # soft‑max weights (max weight = 1)
        w = np.exp(alpha * (conv - max_conv))
        W = w.sum()

        # gradient of the smooth max (a_i + weighted shifted a)
        a_rev = a[::-1]
        conv_rev = convolve_fft(w, a_rev)  # length = 3n‑2
        weighted_shift = conv_rev[n - 1 : n - 1 + n] / W

        grad = a + weighted_shift  # direction of increase
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        # diminishing step size
        step = step0 / math.sqrt(it)
        a = project_simplex(a - step * direction)

        # keep the best
        cur_sc = score_of(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] smoothed improve n={n} it={it} score={cur_sc:.6f}"
            )

        # slowly sharpen the approximation
        if it % 200 == 0:
            alpha = min(alpha * 1.2, 40.0)

        # occasional tiny isotropic noise (helps escape plateaus)
        if it % 800 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (as in the reference)
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient method on the true max‑convolution entry."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    step0 = 0.8
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self_fft(a)
        max_val = conv.max()
        idx = conv.argmax()
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}"
            )

        # active‑set sub‑gradient
        indices = np.arange(n)
        j = idx - indices
        mask = (j >= 0) & (j < n)

        grad = a.copy()
        grad[mask] += a[j[mask]]
        grad *= (2.0 * n)

        # normalise direction (step size independent of ‖grad‖)
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        step = step0 / math.sqrt(it)
        a = project_simplex(a - step * direction)

        # occasional noise
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer annealing (final polishing)
# ----------------------------------------------------------------------


def annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    start_temp: float = 5e-2,
    end_temp: float = 1e-5,
    steps: int = 60000,
) -> tuple[np.ndarray, float]:
    """Simple simulated annealing on the simplex (mass transfer moves)."""
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.2  # move up to 20 % of source
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}"
            )

    return best_seq, best_score

    Search for a non‑negative step‑function minimising ``evaluate_sequence``.

    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from a previous run (global variable ``height_sequence_1``)
    # ------------------------------------------------------------------
    try:

        if s > 0.0:

    except Exception:

    # Process in increasing order so that up‑sampling works naturally.

        total_future = sum(lengths[idx:])

        slice_deadline = time.time() + share
        if slice_deadline > deadline:

    return [float(x) for x in best_seq]
