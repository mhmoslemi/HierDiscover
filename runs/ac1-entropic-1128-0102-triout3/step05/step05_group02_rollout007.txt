def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (real‑to‑real)."""
    n = a.shape[0]
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[:m]
    return conv


def score_of(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a vector normalised to sum=1."""
    n = a.shape[0]
    return 2.0 * n * conv_self(a).max()


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from old length to new_n and re‑normalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Shape families (both symmetric and asymmetric)
# ----------------------------------------------------------------------


def shape_product_power(n: int, p: float) -> np.ndarray:
    """Symmetric product‑power family: a_i ∝ ((i+0.5)*(n-i-0.5))^p."""
    i = np.arange(n) + 0.5
    term = i * (n - i)
    eps = 1e-12
    term[term == 0] = eps
    a = term ** p
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian (σ in index units)."""
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_beta(n: int, alpha: float, beta_par: float) -> np.ndarray:
    """Beta‑like shape on the discrete grid."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    a = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta_par - 1.0))
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_power_decay(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^(-p)."""
    i = np.arange(n) + 1.0
    a = i ** (-p)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_exponential(n: int, lam: float) -> np.ndarray:
    """Monotone decreasing exponential: a_i ∝ exp(-lam·i)."""
    i = np.arange(n)
    a = np.exp(-lam * i)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_linear(n: int) -> np.ndarray:
    """Simple linear ramp decreasing from left to right."""
    a = np.arange(n, 0, -1).astype(float)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def shape_half_cosine(n: int, power: float = 1.0) -> np.ndarray:
    """Half‑cosine on [0,1]: sin(π·x)^power."""
    x = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * x) ** power
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum & active set
# ----------------------------------------------------------------------


def subgradient_descent_momentum(a0: np.ndarray,
                                 n: int,
                                 deadline: float,
                                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Momentum‑accelerated projected sub‑gradient descent.
    The sub‑gradient is taken as the average of the gradients of all
    convolution entries that are within a tiny tolerance of the maximum.
    Returns the best vector observed and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = score_of(a)

    # momentum parameters
    beta = 0.9
    v = np.zeros_like(a)

    step0 = 0.9        # initial step (will be divided by sqrt(iter))
    it = 0
    tol = 1e-12        # tolerance for active set

    while time.time() < deadline:
        it += 1
        conv = conv_self(a)
        max_val = conv.max()
        # active indices (within absolute tolerance)
        act_idx = np.where(conv >= max_val - tol)[0]

        # aggregate gradient over the active set
        grad = np.zeros_like(a)
        for idx in act_idx:
            j = idx - np.arange(n)
            mask = (j >= 0) & (j < n)
            g = a.copy()
            g[mask] += a[j[mask]]
            grad += g
        grad = grad * (2.0 * n) / max(1, len(act_idx))

        # normalise direction
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        # diminishing step size
        step = step0 / math.sqrt(it)

        # momentum update
        v = beta * v + (1.0 - beta) * direction

        a = project_simplex(a - step * v)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] subgrad improve n={n} it={it} score={cur_sc:.6f}")

        # occasional isotropic jitter to escape plateaus
        if it % 500 == 0:
            a += 1e-4 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            v = np.zeros_like(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Cheap mass‑transfer simulated annealing (polishing)
# ----------------------------------------------------------------------


def annealing(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              cur_best_score: float,
              cur_best_seq: np.ndarray,
              start_temp: float = 5e-2,
              end_temp: float = 1e-6,
              steps: int = 200000) -> tuple[np.ndarray, float]:
    """
    Very cheap mass‑transfer annealing.
    Moves a small ε from a source index to a destination index.
    The schedule is geometric; the number of steps is capped by the time budget.
    """
    a = a0.copy()
    score = score_of(a)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # take up to 20% of the source mass
        eps = a[i] * np.random.rand() * 0.2
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = score_of(a_prop)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) minimising
    `evaluate_sequence`.  The algorithm:
        * tries several monotone and symmetric parametric families,
        * runs momentum‑based sub‑gradient descent on the exact ∞‑norm,
        * finishes with a long simulated‑annealing polishing phase.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Optional starter from the previous run (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter = starter / s
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – start modest, finish with a large vector.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    random.shuffle(lengths)            # guard against early termination bias
    global_a = starter

    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # initialise a0 (warm start / random)
        # --------------------------------------------------------------
        if global_a is not None:
            a0 = upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # allocate a time slice for this length
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # later (larger) lengths receive a slightly larger share
        frac = 0.07 + 0.13 * (idx + 1) / len(lengths)   # between 7% and 20%
        slice_deadline = time.time() + max(6.0, remaining * frac)

        # --------------------------------------------------------------
        # quick stochastic search over parametric families
        # --------------------------------------------------------------
        shape_deadline = time.time() + min(3.0, (slice_deadline - time.time()) * 0.25)
        local_best_a = a0
        local_best_sc = score_of(a0)

        while time.time() < shape_deadline:
            kind = np.random.choice(['prod', 'gauss', 'beta',
                                      'powdec', 'expdec', 'lin', 'cos'])
            if kind == 'prod':
                p = np.random.uniform(-1.8, 2.2)
                cand = shape_product_power(n, p)
            elif kind == 'gauss':
                sigma = np.random.uniform(n / 30.0, n / 2.0)
                cand = shape_gaussian(n, sigma)
            elif kind == 'beta':
                alpha = np.random.exponential(1.0) + 0.05
                beta_par = np.random.exponential(1.0) + 0.05
                cand = shape_beta(n, alpha, beta_par)
            elif kind == 'powdec':
                p = np.random.uniform(0.0, 4.0)
                cand = shape_power_decay(n, p)
            elif kind == 'expdec':
                # log‑uniform λ ∈ [1e‑3, 5]
                lam = 10 ** np.random.uniform(-3, np.log10(5))
                cand = shape_exponential(n, lam)
            elif kind == 'lin':
                cand = shape_linear(n)
            else:   # 'cos'
                power = np.random.uniform(0.5, 3.0)
                cand = shape_half_cosine(n, power)

            sc = score_of(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand
                print(f"[{time.time() - start_time:6.2f}s] shape improve n={n} kind={kind} score={sc:.6f}")

        # --------------------------------------------------------------
        # momentum sub‑gradient descent (exact ∞‑norm)
        # --------------------------------------------------------------
        grad_deadline = time.time() + (slice_deadline - time.time()) * 0.55
        a_opt, sc_opt = subgradient_descent_momentum(local_best_a, n,
                                                     grad_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] global improve after subgrad n={n} score={sc_opt:.6f}")

        # --------------------------------------------------------------
        # cheap annealing refinement (if time permits)
        # --------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.2)
            a_ref, sc_ref = annealing(
                a_opt,
                n,
                anneal_deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=5e-2,
                end_temp=1e-6,
                steps=200000,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] global improve after anneal n={n} score={sc_ref:.6f}")

        # keep the best vector for the next up‑sample
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # final fine‑tuning on the longest vector (if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_final = best_seq.shape[0]
        final_deadline = deadline
        print(f"[{time.time() - start_time:6.2f}s] final polishing on length {n_final}")
        a_opt, sc_opt = subgradient_descent_momentum(best_seq, n_final,
                                                     final_deadline, start_time)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:6.2f}s] final subgrad improvement score={sc_opt:.6f}")

        if time.time() < deadline:
            a_ref, sc_ref = annealing(
                best_seq,
                n_final,
                deadline,
                start_time,
                best_score,
                best_seq,
                start_temp=2e-2,
                end_temp=1e-6,
                steps=300000,
            )
            if sc_ref < best_score:
                best_score, best_seq = sc_ref, a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] final anneal improvement score={sc_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return a plain Python list (the evaluator will clamp/normalise as needed)
    return [float(x) for x in best_seq]
