def conv_max(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    return conv.max(), conv

"""
Propose a low‑score step‑function using a two‑phase mirror‑descent scheme.

The algorithm works on a fixed length `n` (chosen according to the time
budget) and optimises a probability vector `a` (non‑negative, sum = 1).
First a short Adam phase minimises a smooth approximation of the
∞‑norm of the auto‑convolution (soft‑max).  Then a long exponential‑gradient
(mirror‑descent) phase minimises the exact max‑convolution using the true
sub‑gradient (hard‑max).  Several diverse initial vectors are tried;
the best result is refined until the time budget (≈1000 s) expires.
All operations are performed with FFTs for O(n log n) speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _as_simplex(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length) using FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k * conv_k(a,a) w.r.t. a.
    Implements g = 2 * (a_rev * w) via FFT.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len_grad)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(W * A_rev, n=fft_len_grad)[: len(w) + a.shape[0] - 1]
    start = a.shape[0] - 1
    return 2.0 * conv2[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_phase(a: np.ndarray,
                max_iters: int,
                deadline: float,
                alpha: float = 0.2,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                temp_start: float = 0.07,
                temp_end: float = 0.001) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max approximation of max(conv)."""
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights
        tau = temp_start + (temp_end - temp_start) * (it / max_iters)
        tau = max(tau, 1e-14)
        w = np.exp((conv - max_val) / tau)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.shape[0])

        g = _gradient_from_weights(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


def _exp_gradient_phase(a: np.ndarray,
                        max_iters: int,
                        deadline: float,
                        eta_start: float = 0.8,
                        eta_end: float = 0.001,
                        eps_mask: float = 1e-12) -> tuple[np.ndarray, float]:
    """Exponential‑gradient (mirror descent) on the exact max‑convolution."""
    n = a.shape[0]
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    best_score = float("inf")
    best_a = a.copy()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # mask for (almost) active anti‑diagonals
        mask = conv >= max_val - eps_mask
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        g = _gradient_from_weights(a, w, fft_len_grad)

        # geometric decay of η
        tau = it / max_iters
        eta = eta_start * (eta_end / eta_start) ** tau

        upd = -eta * g
        upd = np.clip(upd, -30.0, 30.0)            # keep exp() stable
        a = a * np.exp(upd)

        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score


def _optimize_candidate(a0: np.ndarray,
                        max_iters: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Runs a short Adam warm‑up followed by a longer exponential‑gradient
    refinement.  Returns the best vector seen (and its 2·n·max_conv score).
    """
    # keep about 15 % of the budget for Adam, the rest for EG
    adam_iters = max(1, int(0.15 * max_iters))
    eg_iters = max_iters - adam_iters

    a_adam, score_adam = _adam_phase(a0.copy(),
                                     max_iters=adam_iters,
                                     deadline=deadline,
                                     alpha=0.18,

                                            eta_end=0.001)

    # pick the better of the two phases
    if score_adam < score_eg:
        return a_adam, score_adam
    else:
        return a_final, score_eg
