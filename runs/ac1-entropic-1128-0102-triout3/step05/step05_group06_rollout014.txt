def _gradient_smooth(a, conv, alpha):
    max_val = conv.max()
    exp_weights = np.exp(alpha * (conv - max_val))
    exp_weights_sum = exp_weights.sum()
    w = exp_weights / exp_weights_sum
    # compute grad = 2 * convolution of w and reversed a
    a_rev = a[::-1]
    # full convolution using FFT:
    len_a = a.shape[0]
    len_w = w.shape[0]
    fft_len = 1 << ((len_a + len_w - 1).bit_length())
    Fa = np.fft.rfft(a_rev, n=fft_len)  # reversed a
    Fw = np.fft.rfft(w, n=fft_len)
    conv_full = np.fft.irfft(Fa * Fw, n=fft_len)
    # conv_full length = fft_len; relevant portion size = len_a + len_w -1
    # we need elements index offset = len_w - 1 (since conv_rev = w * a_rev)
    start = len_w - 1
    grad = 2 * conv_full[start:start + len_a]   # shape n
    return grad

"""
Optimise a non‑negative step‑function (probability vector) for the
`evaluate_sequence` score.

Key ideas
---------
* Initialise with a Beta(½,½) (arcsine) density – a good analytic start.
* Optimise a smooth surrogate of the max‑convolution using an Adam‑style
  optimiser.  The surrogate is the log‑sum‑exp of the convolution,
  whose gradient can be obtained by a single weighted convolution.
* After each optimisation stage the vector is up‑sampled (linear
  interpolation) to a larger support; larger n gives a tighter theoretical
  bound while the optimiser keeps the shape.
* A cheap simulated‑annealing walk refines the final candidate.
* Time is split automatically between the gradient stages and the final
  annealing; the routine always returns the best sequence found before the
  deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x_i≥0, Σx_i=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based linear convolution (real input)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT (real‐to‐real)."""
    n = x.shape[0] + y.shape[0] - 1
    # next power of two for speed
    N = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:n]
    return conv


# ----------------------------------------------------------------------
# Self‑convolution a*a (length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Gradient of the smooth approximation of max{conv}
#    f(a) = 2·n·LSE_α(conv)   with   LSE_α(z) = (1/α)·log Σ exp(α·z)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of 2·n·LSE_α(conv) w.r.t. a.
    Returns a vector of length len(a).
    """
    max_val = conv.max()
    # soft‑max weights (stable)
    exp_vals = np.exp(alpha * (conv - max_val))
    w = exp_vals / exp_vals.sum()          # shape (2·n‑1,)

    # gradient = 2 * (w * a_rev)   where * denotes linear convolution
    a_rev = a[::-1]
    # full convolution w * a_rev has length len(w)+len(a)-1
    full = _fft_convolve(w, a_rev)
    start = len(w) - 1                     # index where the length‑n part begins
    grad = 2.0 * full[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray,
              n: int,
              deadline: float,
              start_time: float,
              alpha: float = 8.0,
              lr: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using the smooth max objective.
    Returns (best_sequence, best_score) where best_score = 2·n·max(conv).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    # initialise best_score with the true max‑convolution (not the smooth one)
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # 1) exact convolution (needed for the score)
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"adam n={n:5d} iter={it:7d} score={cur_score:.6f}")

        # 2) smooth gradient
        grad = _smooth_gradient(a, conv, alpha)

        # 3) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # 4) projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing refinement (tiny mass transfers)
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 500_000) -> tuple[np.ndarray, float]:
    """Move a tiny fraction ε of mass from i→j, accepting with a geometric schedule."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        eps = a[i] * np.random.rand() * 0.1   # move ≤10 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"anneal n={n:5d} iter={it:7d} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (preserves total mass)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` from its current length to `new_n`."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up
