def simplex_projection(v):
    # returns projection of v onto simplex {x >=0, sum x =1}
    # QP solution: sort v descending, find rho = max{i | v_i - (cum_sum_i -1)/(i+1) >0}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv/(np.arange(1, len(v)+1)) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho]/(rho+1)
    w = np.maximum(v - theta, 0)
    return w

Propose a (non‑negative, sum‑normalised) step‑function that beats the
uniform solution.  The objective to minimise is

    Φ(a) = 2·n·max(a∗a) / (∑a)²   (∑a is fixed to 1).

The algorithm uses a smooth approximation of the max (soft‑max) and
performs projected (multiplicative) gradient descent on the simplex.
Fast FFT based convolutions give O(n log n) per iteration,
allowing thousands of iterations in the 1000 s budget.
The best vector found (including a possible previously stored
`height_sequence_1`) is returned.

import time

# --------------------------------------------------------------
# helper: Euclidean projection onto the probability simplex
# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x≥0, sum x = 1}."""
    # based on the algorithm of Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            # g_i = 2 * sum_j w_{i+j} * a_j  (indices out of range are ignored)
            # we compute it as a convolution of w with reversed a
            a_rev = a[::-1]
            W_f = np.fft.rfft(w, n=fft_len)
            A_rev_f = np.fft.rfft(a_rev, n=fft_len)
            conv_wr = np.fft.irfft(A_rev_f * W_f, n=fft_len)
            # slice the part that corresponds to g_i
            start_idx = n - 1
            g = 2.0 * conv_wr[start_idx:start_idx + n]

            # ---------- multiplicative update on the simplex ----------
            # a <- a * exp(-η g)   then renormalise
            mult = np.exp(-eta * g)
            a = a * mult
            a = a / a.sum()          # enforce sum‑to‑one (keeps positivity)

            # ---------- schedule updates ----------
            # increase temperature (sharpen the approximation of max)
            if temperature < temp_max:
                temperature = min(temp_max, temperature * temp_factor)

            # if we have not improved for a while, shrink the step size
            if patience >= 120:
                eta = max(1e-5, eta * eta_decay)
                patience = 0

        # end of optimisation for this length
        print(f"[{time.time()-start_time:.2f}s] finished length {n} "
              f"best score for this block {best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback: if nothing was found (unlikely) use a modest uniform vector
    # ------------------------------------------------------------------
    if best_vec is None:
        n_fallback = 200
        best_vec = np.full(n_fallback, 1.0 / n_fallback, dtype=float)
        best_score = evaluate_sequence(best_vec.tolist())
        print(f"[{time.time()-start_time:.2f}s] fallback uniform score {best_score:.6f}")

    # final safety checks (the harness also does them, but we keep them)
    if best_vec.sum() < 0.01:
        best_vec = np.full(200, 1.0 / 200.0, dtype=float)

    # final evaluation using the provided function (ensures identical handling)
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"[{time.time()-start_time:.2f}s] returning sequence length {len(best_vec)} "
          f"final evaluated score {final_score:.6f}")

    return best_vec.tolist()
