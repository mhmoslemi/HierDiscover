        g = cp.Variable(len(f), nonneg=True)
        constraints = [A @ g <= M * np.ones(A.shape[0])]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)

            break

            break

        f = (1 - t) * f + t * g_scaled
        # Normalize to sum 1 for safety (should stay 1)
        f_sum = f.sum()
        if f_sum != 0:

"""
Iterative LP‐based flattening of the autocorrelation.

The score to minimise is
    Φ(a) = 2·n·max(a∗a) / (sum(a)^2)        (scaling‑invariant)

We keep the sequence normalised to sum = 1.
Given a current candidate `f`, the largest convolution value is `M = max(f∗f)`.
For any non‑negative `g` we have the linear constraints

    (f∗g)[k] = Σ_i f[i]·g[k‑i] ≤ M   for all k.

The linear programme

    maximise   Σ_j g[j]
    subject to g ≥ 0 and (f∗g) ≤ M

always admits the feasible point `g = f` (so the optimum sum ≥ 1).
If the optimum has sum S > 1 we scale it to unit sum:
    ĝ = g / S .
Because `f∗ĝ ≤ M/S < M`, a convex blend
    f_new = (1‑t)·f + t·ĝ
strictly reduces the peak `max(f_new∗f_new)`.
Repeating these steps quickly drives the autocorrelation towards a flat
distribution, lowering the evaluation score.

The implementation below
* starts from the best known global sequence (`height_sequence_1`) if
  available, otherwise from a uniform vector,
* limits the length to at most 2000 elements (to keep the LP tractable),
* builds the convolution matrix once per iteration (sparse, O(n²) entries),
* solves the LP with CVXPY/ECOS,
* mixes the new vector with the old one (t≈0.5, slowly decayed),
* stops when the time budget is exhausted or the target score (1.503)
  is reached.

The function returns the best sequence found within the allotted time.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp


def _build_conv_index_arrays(n: int):
    """
    Return the row and column index arrays for the convolution matrix
    of a length‑n vector.
    The matrix A satisfies  (A @ g)[k] = Σ_i f[i]·g[k‑i]
    for a given fixed vector f.
    """
    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    pos = 0
    arange_n = np.arange(n, dtype=np.int64)
    for i in range(n):
        # row = i + j, column = j
        rows[pos:pos + n] = i + arange_n
        cols[pos:pos + n] = arange_n
        pos += n
    return rows, cols


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimise
    `evaluate_sequence`.  The routine performs a few iterations of the LP
    flattening scheme described above and stops before the allotted time budget.
    """
    # ------------------------------------------------------------------
    # 1.  Prepare the initial sequence.
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()

    # Try to use the previously discovered sequence.
    prev_seq = None
    try:
        # `height_sequence_1` is provided as a global if a previous run succeeded.
        cand = globals().get("height_sequence_1", None)
        if cand is not None:
            cand = np.asarray(cand, dtype=float)
            # Apply the same clipping that evaluate_sequence does.
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.size > 0 and cand.sum() >= 0.01:
                prev_seq = cand
    except Exception:
        prev_seq = None

    # Limit the length to keep the LP reasonably fast.
    MAXLEN = 2000
    if prev_seq is not None:
        # Truncate if necessary, then renormalise.
        if prev_seq.size > MAXLEN:
            prev_seq = prev_seq[:MAXLEN]
        f = prev_seq.copy()
    else:
        # Uniform fallback.
        n = MAXLEN
        f = np.full(n, 1.0 / n, dtype=float)

    # Ensure strict positivity (required for bounded LP) and normalise.
    eps = 1e-12
    f = np.maximum(f, eps)
    f /= f.sum()

    n = len(f)
    rows_idx, cols_idx = _build_conv_index_arrays(n)

    # Store the best sequence seen so far.
    best_f = f.copy()
    try:
        best_score = evaluate_sequence(best_f.tolist())
    except Exception:
        best_score = float("inf")

    print(f"[0.00s] start length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Iterative LP flattening.
    # ------------------------------------------------------------------
    # Mixing factor (starts relatively large, then decays).
    t = 0.5
    iteration = 0
    # Run until we run out of budget (reserve a small safety margin).
    while time.time() - start_time < budget_s - 5:
        iteration += 1

        # Current worst convolution value.
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Build the convolution matrix A for the current f.
        # A @ g gives f∗g (componentwise).
        data = np.repeat(f, n)               # shape (n*n,)
        A = sp.csr_matrix((data, (rows_idx, cols_idx)),
                          shape=(2 * n - 1, n))

        # Linear program: maximise Σ g_j   subject to g≥0 and A g ≤ M.
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # Solve – ECOS is fast for this modest size LP.
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, abstol=1e-9, reltol=1e-9)
        except Exception:
            print(f"[{time.time()-start_time:.2f}s] LP solve failed, breaking.")
            break

        if g.value is None:
            print(f"[{time.time()-start_time:.2f}s] LP infeasible, breaking.")
            break

        g_val = np.asarray(g.value, dtype=float)
        sum_g = g_val.sum()
        if sum_g <= 0:
            print(f"[{time.time()-start_time:.2f}s] LP gave zero sum, breaking.")
            break

        # Scale to unit sum.
        g_hat = g_val / sum_g

        # Convex combination with the current point.
        f = (1.0 - t) * f + t * g_hat

        # Re‑enforce positivity and normalise (tiny numerical drift may appear).
        f = np.maximum(f, eps)
        f /= f.sum()

        # Compute the new evaluation score.
        try:
            cur_score = evaluate_sequence(f.tolist())
        except Exception:
            cur_score = float("inf")

        # Log progress.
        elapsed = time.time() - start_time
        print(f"[{elapsed:.2f}s] iter {iteration:2d}: score={cur_score:.6f}, "
              f"M={np.max(np.convolve(f, f)):.6e}, t={t:.3f}")

        # Keep the best.
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            # If we have already beaten the target, we can stop early.
            if best_score <= 1.503:
                print(f"[{elapsed:.2f}s] target reached (score={best_score:.6f})")
                break

        # Decay the mixing factor a little to improve stability.
        if iteration % 3 == 0:
            t = max(0.05, t * 0.9)

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found.
    # ------------------------------------------------------------------
    total_time = time.time() - start_time
    print(f"[{total_time:.2f}s] finished – best score {best_score:.6f} "
          f"with length {len(best_f)}")
    # The harness expects a plain Python list.
    return [float(x) for x in best_f]
