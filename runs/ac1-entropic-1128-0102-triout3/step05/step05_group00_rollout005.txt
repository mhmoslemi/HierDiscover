"""
Improved step‑function optimiser.

The optimiser works on the probability simplex (∑ a_i = 1) – the
evaluation function is scale‑invariant.  For each length n we
* warm‑start from the previous best (linear interpolation)
* run several independent searches:
    – Adam optimisation on a soft‑max surrogate of the ℓ∞ norm.
    – Mirror‑descent (multiplicative‑update) on the exact
      sub‑gradient of max (a∗a) (λ→0).
* finish with a cheap mass‑transfer simulated‑annealing polish.
The time budget (default 1000 s) is split proportionally to the
sequence length, giving larger problems more iterations.
Both gradient‑based phases use FFT‑based convolutions for speed and
project onto the simplex (or re‑normalise) after each update.
The code is fully self‑contained, uses only NumPy and the provided
`evaluate_sequence` function, and prints intermediate scores.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, fast for all n)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full, length 2·n‑1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    c = np.fft.irfft(X * Y, n=nfft)[:size]
    return c


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from `seq` (length old_n) to `new_n`, renormalised."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Gradient / sub‑gradient helpers
# ----------------------------------------------------------------------


def _gradient_from_conv(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max (log‑sum‑exp) surrogate of max(conv).
    """
    n = a.shape[0]
    max_val = conv.max()
    weights = np.exp((conv - max_val) / lam)
    weights /= weights.sum()
    a_rev = a[::-1]
    full = _conv_fft(weights, a_rev)          # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution functional (λ = 0).
    Returns a dense gradient of shape (n,).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_b = conv.max()
    mask = (conv == max_b).astype(float)      # length 2n‑1
    a_rev = a[::-1]
    full = _conv_fft(mask, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed ℓ∞ surrogate
# ----------------------------------------------------------------------


def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  slice_deadline: float,
                  start_time: float,
                  symmetrize: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate.
    Returns the best vector seen and its exact score 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam_factor = 0.5          # initial λ ≈ lam_factor * max(conv)
    lam_decay = 0.6
    lam_decay_interval = 1500

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] adam improve n={n} it={it} score={cur_score:.6f}")

        # update λ
        if it % lam_decay_interval == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)
        lam = max(max_b * lam_factor, 1e-12)

        grad = _gradient_from_conv(a, conv, lam)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        base_step = 0.5 / math.sqrt(it + 1.0)
        a = a - base_step * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a = (a + a[::-1]) * 0.5

        a = _project_simplex(a)

    return best_a, best_score

def _run_mirror_descent(a0: np.ndarray,
                        n: int,
                        slice_deadline: float,
                        start_time: float,
                        eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Exponential‑update (mirror descent) using the exact sub‑gradient.
    Returns best vector and its exact score.
    """
    a = np.maximum(a0, 1e-15)               # avoid exact zeros
    a = a / a.sum()
    best_a = a.copy()
    best_score = float('inf')

    it = 0
    while time.time() < slice_deadline:
        it += 1
        conv = _conv_self(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] mirror improve n={n} it={it} score={cur_score:.6f}")

        # sub‑gradient for the current max‑convolution
        grad = _subgradient_maxconv(a)

        # optional symmetry enforce (helps for many lengths)
        if np.random.rand() < 0.15:
            grad = (grad + grad[::-1]) * 0.5

        # step size schedules like 1/√t
        eta = eta0 / math.sqrt(it)

        grad_clip = np.clip(grad, -12.0, 12.0)

        # multiplicative update and re‑normalisation
        a = a * np.exp(-eta * grad_clip)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

    return best_a, best_score

# ----------------------------------------------------------------------


def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float,
                   cur_best_seq: np.ndarray,
                   start_temp: float = 5e-2,
                   end_temp: float = 1e-5,
                   steps: int = 120000) -> tuple[np.ndarray, float]:
    """
    Cheap annealer that moves a small ε from a source index to a destination index.
    Returns an improved sequence (if any) and its exact score.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # geometric temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.12   # move up to 12 % of source mass
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max()
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > np.random.rand():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] anneal improve n={n} it={it} score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initialisation helpers – a few sensible starting shapes
# ----------------------------------------------------------------------


def _initial_variants(base_seq: np.ndarray) -> list[np.ndarray]:
    """
    Generate a small collection of plausible starting vectors from a base.
    All vectors are already normalised to sum = 1.
    """
    n = base_seq.shape[0]
    variants = []

    # 1 – the base itself (warm‑start)
    variants.append(base_seq.copy())

    # 2 – pure sinusoid (positive on (0,π))
    sin_vals = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_vals = np.maximum(sin_vals, 0.0)
    sin_vals /= sin_vals.sum()
    variants.append(sin_vals)

    # 3 – triangular ramp
    half = n // 2
    tri = np.concatenate([np.linspace(1.0, half, half, endpoint=False),
                          np.linspace(half, 1.0, n - half)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    variants.append(tri)

    # 4 – random Dirichlet draw
    rnd = np.random.rand(n)
    rnd /= rnd.sum()
    variants.append(rnd)

    # 5 – a slightly smoothed version of the base
    smooth = np.convolve(base_seq, np.ones(3) / 3, mode='same')
    smooth = np.maximum(smooth, 0.0)
    smooth /= smooth.sum()
    variants.append(smooth)

    return variants

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the `budget_s` time limit and returns a Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = None
        else:
            starter = None
    except Exception:
        starter = None

    # Length schedule – we increase gradually and give more time to larger n
    lengths = [128, 256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    total_len = sum(lengths)

    global_a = starter

    for n in lengths:
        if time.time() >= deadline:
            break

        # Allocate a time slice proportional to the length
        slice_time = max(5.0, (budget_s * 0.85) * (n / total_len))
        slice_deadline = time.time() + slice_time
        if slice_deadline > deadline:
            slice_deadline = deadline

        # ------------------------------------------------------------------
        # initialise a0 (warm start / several variants)
        # ------------------------------------------------------------------
        if global_a is not None:
            a0 = _upsample(global_a, n)
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        init_candidates = _initial_variants(a0)

        # add a few extra random Dirichlet draws for extra diversity
        for _ in range(3):
            rnd = np.random.rand(n)
            rnd = np.maximum(rnd, 0.0)
            rnd /= rnd.sum()
            init_candidates.append(rnd)

        best_local_a = None
        best_local_score = float('inf')

        # ------------------------------------------------------------------
        # Run a mixture of Adam and mirror‑descent searches
        # ------------------------------------------------------------------
        for start_vec in init_candidates:
            if time.time() >= slice_deadline:
                break

            # randomly decide which optimiser to use for this start
            if np.random.rand() < 0.5:
                a_opt, sc_opt = _run_adam_opt(start_vec, n, slice_deadline, start_time,
                                               symmetrize=(np.random.rand() < 0.25))
            else:
                a_opt, sc_opt = _run_mirror_descent(start_vec, n, slice_deadline, start_time,
                                                    eta0=0.45)

            if sc_opt < best_local_score:
                best_local_score = sc_opt
                best_local_a = a_opt.copy()
                print(f"[{time.time() - start_time:6.2f}s] local improve n={n} score={sc_opt:.6f}")

        if best_local_a is None:
            continue

        # global improvement check
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after optimiser n={n} score={best_local_score:.6f}")

        # ------------------------------------------------------------------
        # final annealing polish (if time permits)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            anneal_deadline = time.time() + max(3.0, (deadline - time.time()) * 0.15)
            a_ref, score_ref = _run_annealing(best_local_a, n, anneal_deadline,
                                              start_time,
                                              best_score,
                                              best_seq,
                                              start_temp=4e-2,
                                              end_temp=8e-6,
                                              steps=150000)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:6.2f}s] GLOBAL improve after anneal n={n} score={score_ref:.6f}")

        # remember for the next (larger) length
        global_a = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(512, 1.0 / 512.0)

    # Return as plain Python list (evaluate_sequence will handle clamping)
    return [float(x) for x in best_seq]
