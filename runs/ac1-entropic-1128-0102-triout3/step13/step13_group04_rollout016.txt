```
def _smooth_grad(a, conv, T):
    # conv is array length 2n-1.
    n = a.shape[0]
    max_conv = conv.max()
    # compute logits offset
    logits = (conv - max_conv) / T
    w = np.exp(logits)
    w_sum = w.sum()
    w = w / w_sum
    G = np.dot(w, conv)
    # compute c = w * (1 + (conv - G) / T)
    c = w * (1.0 + (conv - G) / T)
    # compute gradient = 2 * correlation of c with a
    # correlation via convolution with reversed a
    a_rev = a[::-1]
    L = _next_pow2(3*n)
    fft_c = np.fft.rfft(c, n=L)
    fft_arev = np.fft.rfft(a_rev, n=L)
    conv_rev = np.fft.irfft(fft_c * fft_arev, n=L)[:3*n-2]
    grad = 2.0 * conv_rev[n-1:2*n-1]  # length n
    # optionally multiply by factor 2n (since objective factor)
    grad = grad * (2.0 * n)   # because objective = 2*n* G, gradient scaled accordingly
    return grad, conv.max()
```

```
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_self = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n)  # ensure enough length
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # may adjust
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    # schedule parameters
    temp_start = 0.5   # initial factor relative to max_conv
    temp_min = 0.01
    decay_iters = 20000  # after this temp close to min
    while time.time() < deadline:
        it += 1
        # compute self-convolution
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={cur_score:.6f}")
        # compute temperature
        temp_factor = max(temp_min, temp_start * np.exp(-it / decay_iters))
        T = max(cur_max * temp_factor, 1e-12)
        # compute softmax weights
        # shift by max for stability
        logits = (conv - cur_max) / T
        w = np.exp(logits)
        w_sum = w.sum()
        w = w / w_sum
        G = np.dot(w, conv)
        # compute c vector
        c = w * (1.0 + (conv - G) / T)
        # compute gradient
```

```
"""
Propose a non‑negative step function that minimizes the evaluation metric.

Algorithm:
1. Initialise a probability vector (normalize sum=1).  Use the global
   `height_sequence_1` if present, otherwise a uniform distribution.
2. Run a parametric sweep over geometric‑plus‑uniform mixtures to obtain a
   good starting point.
3. Refine the vector with a projected Adam optimiser that works on a
   *smooth* approximation of the max‑convolution.
   The smooth approximation uses a temperature‑controlled soft‑max
   over the self‑convolution, yielding a dense gradient.
4. Perform a guided hill‑climb that moves a small amount of mass from
   indices that most increase the current maximal convolution to indices
   that contribute the least, always keeping the best vector seen.
5. Return the final normalised vector as a plain Python list.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) obtained from the single active shift."""
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def _adam_smooth_opt(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_score_initial: float,
                     max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the true objective but computes a
    *smooth* gradient using a soft‑max over the convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)   # enough for linear correlation length

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # moderately aggressive

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    # temperature schedule parameters
    temp_start = 0.5          # temperature = temp_start * max_conv at iteration 0
    temp_min = 0.01
    decay_iters = 20000       # after this many iters the temperature ≈ temp_min

    while time.time() < deadline and it < max_iter:
        it += 1

        # convolution and current objective
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam‑smooth it={it:,} score={cur_score:.6f}")

        # temperature (relative to current maximal convolution)
        temp_factor = max(temp_min, temp_start * np.exp(-it / decay_iters))
        T = max(cur_max * temp_factor, 1e-14)

        # soft‑max over the convolution
        # shift by cur_max for numerical stability
        logits = (conv - cur_max) / T
        w = np.exp(logits)
        w_sum = w.sum()
        w = w / w_sum
        G = np.dot(w, conv)          # smoothed max

        # coefficient vector for the gradient of the smoothed objective
        c = w * (1.0 + (conv - G) / T)   # length 2n‑1

        # correlation(c, a) → gradient of G (times 2)
        a_rev = a[::-1]
        fft_c = np.fft.rfft(c, n=L_grad)
        fft_arev = np.fft.rfft(a_rev, n=L_grad)
        corr_full = np.fft.irfft(fft_c * fft_arev, n=L_grad)[:3 * n - 2]
        # The needed slice corresponds to lags –i, i = 0…n‑1
        grad_G = 2.0 * corr_full[n - 1:2 * n - 1]   # length n

        # gradient of the *true* objective (2·n·G)
        grad = grad_G * (2.0 * n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex
        a = _project_simplex(a)

        # occasional noise to escape plateaus
        if it - last_improve > 5000:
            noise = 1e-6 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score

def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Guided hill‑climb: move a tiny fraction of mass from the index that
    contributes most to the current maximum convolution towards the index
    that contributes the least.  Accept only improving moves.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        # compute active shift and associated sub‑gradient
        conv = _conv_self(a, L_fft)
        k = int(conv.argmax())
        # sub‑gradient of the maximal entry
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # choose source (large positive gradient & non‑zero mass)
        candidate_src = idx[grad[idx] > 0.0]
        if candidate_src.size == 0:
            continue
        src = rng.choice(candidate_src)
        # choose destination (small gradient, possibly zero mass)
        dst_candidates = np.setdiff1d(np.arange(n), [src])
        dst = rng.choice(dst_candidates)

        # random tiny fraction of src mass (up to 0.5 %)
        frac = rng.random() * 0.005 + 1e-7
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Guided hill‑climb score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2 · n · max(conv(a,a)) / (sum(a)²).

    The procedure consists of:
        * an optional geometric‑uniform sweep (quick warm‑start);
        * a long projected Adam optimiser that uses a *smooth* soft‑max
          gradient;
        * a final guided hill‑climb that transfers tiny amounts of mass.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size (use the size of a provided global solution if any)
    # ------------------------------------------------------------------
    n = 4096
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if starter is not None:

"""
Propose a non‑negative step function that minimises
        2·n·max(conv(a,a)) / (sum(a)²).

Algorithm
---------
1. Initialise a probability vector (sum = 1).  Use the global
   ``height_sequence_1`` if it exists, otherwise a uniform distribution.
2. Run a cheap geometric‑plus‑uniform sweep to obtain a decent warm‑start.
3. Refine the vector with a projected Adam optimiser that works on a
   *smooth* approximation of the max‑convolution.  The smooth objective
   uses a temperature‑controlled soft‑max over the self‑convolution,
   giving a dense gradient that is more informative than the
   “single‑active‑shift’’ sub‑gradient.
4. Perform a guided hill‑climb that moves a tiny amount of mass from the
   index that most increases the current maximal convolution towards the
   index that contributes the least.  Only improving moves are kept.
5. Return the final normalised vector as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) obtained from the single active shift."""
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser using a *smooth* max‑convolution gradient
# ----------------------------------------------------------------------
def _adam_smooth_opt(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_score_initial: float,
                     max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the true objective but computes a
    *smooth* gradient using a soft‑max over the convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)    # for the self‑convolution
    L_grad = _next_pow2(3 * n)        # enough for the linear correlation

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                         # fairly aggressive

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    # temperature schedule (relative to the current maximal convolution)
    temp_start = 0.5
    temp_min = 0.01
    decay_iters = 20_000

    while time.time() < deadline and it < max_iter:
        it += 1

        # current convolution and true objective value
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam‑smooth it={it:,} score={cur_score:.6f}")

        # temperature (a decreasing fraction of the maximal convolution)
        temp_factor = max(temp_min, temp_start * np.exp(-it / decay_iters))
        T = max(cur_max * temp_factor, 1e-14)

        # soft‑max over the convolution (stable version)
        logits = (conv - cur_max) / T            # ≤ 0
        w = np.exp(logits)
        w_sum = w.sum()
        w = w / w_sum
        G = np.dot(w, conv)                      # smoothed max

        # coefficient vector for the gradient of the smoothed objective
        c = w * (1.0 + (conv - G) / T)           # length 2·n‑1

        # correlation(c, a) → gradient of G (times 2)
        a_rev = a[::-1]
        fft_c = np.fft.rfft(c, n=L_grad)
        fft_arev = np.fft.rfft(a_rev, n=L_grad)
        corr_full = np.fft.irfft(fft_c * fft_arev, n=L_grad)[:3 * n - 2]
        # slice corresponds to lags –i, i = 0…n‑1
        grad_G = 2.0 * corr_full[n - 1:2 * n - 1]

        # gradient of the *true* objective (2·n·G)
        grad = grad_G * (2.0 * n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _project_simplex(a)

        # occasional tiny random perturbation to escape plateaus
        if it - last_improve > 5_000:
            noise = 1e-6 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Guided hill‑climb (tiny mass transfers guided by the active shift)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Guided hill‑climb: move a very small fraction of probability mass from
    the index that most contributes to the current maximal convolution
    towards an index that contributes the least.  Keep only improving
    moves; the best vector seen is returned.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        k = int(conv.argmax())

        # sub‑gradient of the maximal entry (provides a contribution map)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # source = index with large positive contribution and non‑zero mass
        cand_src = idx[grad[idx] > 0.0]
        if cand_src.size == 0:
            continue
        src = rng.choice(cand_src)

        # destination = any other index (preferably with small contribution)
        dst_candidates = np.setdiff1d(np.arange(n), [src])
        dst = rng.choice(dst_candidates)

        # random tiny fraction of src mass (up to 0.5 %)
        frac = rng.random() * 0.005 + 1e-7
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop[dst] += eps

        if cur_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Guided hill‑climb score={best_score:.6f}")

    return best_a, best_score
```
