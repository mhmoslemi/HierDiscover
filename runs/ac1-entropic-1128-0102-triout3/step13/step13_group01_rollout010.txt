    w_rev = w[::-1]

    end = start + len(a)  # = n-1 + n = 2n-1
    grad = conv_rev[start:end][::-1]

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Choose n

    # Initialize a candidate: maybe use best among analytical families via grid search (similar to previous) but faster, maybe fewer combos.
    # We'll generate initial a as uniform plus small random perturbation, then run gradient descent.

                # If length differs from n, upsample or downsample
                if starter.size != n:
                    # Rescale using interpolation
                    x_old = np.linspace(0, 1, starter.size)
                    x_new = np.linspace(0, 1, n)
                    starter = np.interp(x_new, x_old, starter)
                a = starter
    except Exception:
        pass

    # Evaluate initial best
    best_a = a.copy()
    best_score = _objective(a)
    print(f"[{time.time() - start:.2f}s] initial score = {best_score:.6f}")

    # Prepare parameters
    iter = 0
    beta = 0.5
    beta_max = 500.0
    beta_inc_iter = 2000      # increase beta every this many iterations
    # Step size schedule param
    eta0 = 0.5

    conv_len = 2 * n - 1

    # Main loop
    while time.time() < deadline:

        # compute conv via real FFT (use precomputed length)
        Af = np.fft.rfft(a, n=L_conv)

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (1.0)   # sum=1
        # track best
        if cur_score < best_score:

            best_a = a.copy()

    else:

    return [float(v) for v in best_a]

"""
Multi‑scale smoothed max‑convolution optimisation.

1.  Build a high‑resolution candidate (n_big = 65536) and minimise a
    smoothed version of the maximum of the self‑convolution using
    exponentiated‑gradient (mirror descent) on the probability simplex.
    The smoothing is a soft‑max with a temperature β that is slowly
    increased, turning the problem from a smooth convex one into the
    original non‑smooth objective.

2.  Down‑sample the best high‑resolution shape to the target length
    (4096) by summing contiguous blocks – this preserves the total mass
    and keeps the low autocorrelation already discovered.

3.  Refine the down‑sampled vector with the same mirror‑descent loop
    (now cheaper because n is small) and finish with a light
    simulated‑annealing polish.

All operations use FFT‑based convolutions, so each iteration is
≈O(n log n).  The routine respects the 1000 s budget, prints progress,
and finally returns a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT)
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution a*a using a real FFT. Returns length 2·n‑1 array."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    if L is None:
        L = _next_pow2(conv_len)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:conv_len]
    return conv

# ----------------------------------------------------------------------
# Smoothed max (soft‑max) of a vector
# ----------------------------------------------------------------------
def _softmax_weights(x: np.ndarray, beta: float) -> np.ndarray:
    """Return softmax weights w_i ∝ exp(beta * x_i)."""
    # numerical stability
    mx = np.max(beta * x)
    w = np.exp(beta * x - mx)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum

# ----------------------------------------------------------------------
# Gradient of the smoothed max of the self‑convolution.
# For w = softmax_weights(conv),   grad_i = 2 * Σ_j w_{i+j} a_j .
# ----------------------------------------------------------------------
def _grad_from_weights(a: np.ndarray, w: np.ndarray,
                       L_grad: int) -> np.ndarray:
    """Compute 2 * Σ_j w_{i+j} a_j for all i (valid part) via FFT."""
    # w has length 2·n‑1, a has length n
    w_rev = w[::-1]
    # convolution length = (2·n‑1) + n – 1 = 3·n‑2
    Wf = np.fft.rfft(w_rev, n=L_grad)
    Af = np.fft.rfft(a, n=L_grad)
    conv_rev = np.fft.irfft(Wf * Af, n=L_grad)
    conv_rev = conv_rev[: (2 * a.size - 1) + a.size - 1]   # 3n‑2 entries
    start = (2 * a.size - 1) - a.size    # = n‑1
    grad = conv_rev[start:start + a.size][::-1]
    return 2.0 * grad

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (not used in the
# exponentiated‑gradient version, but kept for possible fall‑backs).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Fast objective (scale‑invariant; assumes Σa = 1)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray, L_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = _conv_self_fft(a, L_conv)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Core optimisation loop (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _optimize_on_grid(a0: np.ndarray,
                      max_time: float,
                      start_time: float,
                      beta_start: float = 0.5,
                      beta_max: float = 200.0,
                      beta_inc: float = 1.5,
                      beta_every: int = 5000,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation of the smoothed objective.
    Returns the best vector found and its (exact) objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # pre‑allocate FFT lengths
    conv_len = 2 * n - 1
    L_conv = _next_pow2(conv_len)
    L_grad = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = _fast_objective(a, L_conv)

    beta = beta_start
    it = 0

    while time.time() < max_time:
        it += 1

        # compute convolution once (used for both objective and weights)
        conv = _conv_self_fft(a, L_conv)

        # exact objective for logging
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,}  best={best_score:.6f}")

        # soft‑max weights (smoothed max)
        w = _softmax_weights(conv, beta)

        # gradient of the smoothed max
        grad = _grad_from_weights(a, w, L_grad)

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (mirror descent)
        exponents = -eta * grad
        # avoid overflow / underflow
        exponents -= np.max(exponents)
        factor = np.exp(exponents)
        a = a * factor
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # increase smoothing temperature periodically
        if it % beta_every == 0:
            beta = min(beta * beta_inc, beta_max)

        # small random jitter every few thousand iterations to help escape plateaux
        if it % 8000 == 0:
            rng = np.random.default_rng()
            i, j = rng.integers(n, size=2)
            if i != j:
                eps = min(a[i], 0.02 * rng.random())
                a[i] -= eps
                a[j] += eps  # total mass conserved, positivity preserved

    return best_a, best_score

# ----------------------------------------------------------------------
# Light simulated‑annealing polish (for the final, small‑n stage)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap Metropolis annealing that respects the simplex."""
    a = seq.copy()
    n = a.size
    cur_score = _fast_objective(a, _next_pow2(2 * n - 1))
    best_a, best_score = a.copy(), cur_score
    rng = np.random.default_rng()
    max_steps = 300_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # linear cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # more mass ⇒ more likely to move
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 40 %) of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _fast_objective(a_prop, _next_pow2(2 * n - 1))
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,}  best={best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    2. Down‑sample the best high‑resolution vector to the target length
       (4096) by blockwise summation, re‑normalise, and run a short
       refinement on the smaller problem.

    3. Finish with a tiny simulated‑annealing polish.

    The routine respects the 1000 s time budget and returns a plain
    Python list of non‑negative heights.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a tiny safety margin

    # 0) Target length (the final sequence must be reasonably small
    #    because `evaluate_sequence` uses a naïve O(n²) convolution).
    # --------------------------------------------------------------
    n_target = 4096

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    n_big = n_target * blocksize       # 65536

    a_big = np.full(n_big, 1.0 / n_big)

            if starter.sum() > 0.0:
                # up‑sample or down‑sample to n_big as needed

                    x_new = np.linspace(0.0, 1.0, n_big)
                    starter = np.interp(x_new, x_old, starter)
                starter /= starter.sum()
                a_big = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Using global starter (up‑sampled).")
    except Exception:
        pass

    # allocate a reasonable slice of the budget for the large‑scale phase
    remaining = deadline - time.time()
    big_time = start_time + 0.75 * remaining   # ~75 % of the remaining time

                                                 small_deadline,
                                                 start_time,
                                                 beta_start=0.2,
                                                 beta_max=500.0,
                                                 beta_every=3000,
                                                 eta0=0.8)

    # --------------------------------------------------------------
    # 3) Light simulated‑annealing polish (if any time left)
    # --------------------------------------------------------------

    else:
        a_final, final_score = a_refined, refined_score

    # --------------------------------------------------------------
    # 4) Final normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    a_final = np.maximum(a_final, 0.0)
    total = a_final.sum()
    if total > 0.0:
        a_final /= total
    else:
        a_final = np.full_like(a_final, 1.0 / a_final.size)

        # evaluate_sequence is available in the execution environment
        true_score = evaluate_sequence([float(v) for v in a_final])
        print(f"[{time.time() - start_time:.2f}s] Final true score = {true_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished, returning sequence of length {a_final.size}.")
    return [float(v) for v in a_final]
