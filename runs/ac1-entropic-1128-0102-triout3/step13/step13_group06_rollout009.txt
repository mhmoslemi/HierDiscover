"""
Guided hill‑climbing with simulated‑annealing.

* Warm‑start: a large pool of analytic shapes (tent, exponential, power‑law,
  Gaussian, random Dirichlet, and the previously discovered vector).  The best
  (lowest exact objective) is kept.

* Main loop:  repeatedly move a small packet of mass from a “bad’’ index to a
  “good’’ one.  The bad index is chosen with probability proportional to the
  contribution a[i]·a[k*‑i] to the current maximal convolution entry
  (k* = argmax (a∗a)).  The destination is biased toward indices that have a
  small overlap with the reversed vector.  The move is accepted if it reduces
  the objective, otherwise it may be accepted with a temperature‑dependent
  probability (simulated annealing) to escape local minima.

* The convolution is maintained incrementally (O(n) per move) using the same
  formula as the reference implementation, allowing millions of moves within
  the 1000 s budget.  The best vector seen during the whole run is returned.

The algorithm respects all constraints of ``evaluate_sequence`` (non‑negative,
finite values, sum≥0.01) and works for any length; the default length is
4096 which gave the previous best result.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _exact_objective(a: np.ndarray) -> float:
    """Exact objective for an arbitrary non‑negative vector."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`.

    The convolution (a*a) is updated in‑place using the linear formula.
    """
    n = a.shape[0]

    # linear term 2 * a * Δ
    conv[src : src + n] -= 2.0 * eps * a
    conv[dst : dst + n] += 2.0 * eps * a

    # quadratic term Δ·Δ (only three positions change)
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # update the probability vector (sum stays constant)
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Analytic shape generators (all produce a non‑negative vector of length n)
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of unnormalised analytic shapes."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Exponential decay from the centre (symmetric)
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]                     # symmetric
    candidates.append(shape)

    # 5) Half‑cosine
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random draws (Dirichlet‑like)
    rng = np.random.default_rng(0)
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # normalise (sum to 1) and discard the all‑zero case
    out = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            out.append(c / s)
    return out

# Main entry point
# ----------------------------------------------------------------------

    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """

    start_time = time.time()
    # small safety margin (the harness adds ~0.5 s)

            best_vec = ws.copy()
    # best_vec now holds the warm‑start with the smallest exact objective

    conv = _conv_fft(a)
    cur_obj = 2.0 * n * conv.max()      # sum(a) == 1, so denominator = 1
    best_vec = a.copy()

    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Guided hill‑climbing with simulated annealing
    # ------------------------------------------------------------------
    it = 0
    # initial temperature (10 % of the current objective works well)
    T0 = cur_obj * 0.1
    # counters for stagnation detection
    no_improve_iters = 0
    stagnation_limit = 200_000

    while time.time() < deadline:
        it += 1

        # locate the index of the current maximal convolution entry
        k_star = int(conv.argmax())               # 0 … 2n‑2
        # --------------------------------------------------------------
        # construct the overlap vector rev[i] = a[k_star‑i] (0 if out of range)
        # --------------------------------------------------------------
        idx = np.arange(n)
        j = k_star - idx
        mask = (j >= 0) & (j < n)
        rev = np.zeros_like(a)
        rev[mask] = a[j[mask]]

        # source weighting = contribution to the current maximum
        contrib = a * rev
        sum_contrib = contrib.sum()
        if sum_contrib > 0.0:
            src = rng.choice(n, p=contrib / sum_contrib)
        else:
            # fallback: pick proportionally to the mass itself
            src = rng.choice(n, p=a)

        # destination weighting – prefer positions with small overlap
        inv_rev = 1.0 - rev
        sum_inv = inv_rev.sum()
        if sum_inv > 0.0:
            dst = rng.choice(n, p=inv_rev / sum_inv)
        else:
            dst = rng.integers(n)

        if src == dst or a[src] < 1e-15:
            continue

        # ------------------------------------------------------------------
        # step size schedule – large early, shrinking later
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)          # 0 … 1
        max_move_frac = 0.30 * (1.0 - frac) + 0.02         # from 0.30 → 0.02
        eps = a[src] * rng.random() * max_move_frac
        if eps <= 0.0:
            continue

        # --------------------------------------------------------------
        # apply the move (incremental update)
        # --------------------------------------------------------------
        _apply_move(a, conv, src, dst, eps)

        new_obj = 2.0 * n * conv.max()
        delta = new_obj - cur_obj

        # temperature schedule – linear cooling
        T = T0 * (1.0 - frac) + 1e-12

        accept = False
        if delta <= 0.0:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_obj = new_obj
            no_improve_iters = 0
            if new_obj < best_score:
                best_score = new_obj
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # reject – undo the move
            _apply_move(a, conv, dst, src, eps)
            no_improve_iters += 1

        # ------------------------------------------------------------------
        # occasional random perturbation if we are stuck
        # ------------------------------------------------------------------
        if no_improve_iters >= stagnation_limit:

            noise = rng.random(n) * 0.01
            a += noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            conv = _conv_fft(a)
            cur_obj = 2.0 * n * conv.max()
            no_improve_iters = 0

    # 3) Return the best vector found (renormalised to sum=1)
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
