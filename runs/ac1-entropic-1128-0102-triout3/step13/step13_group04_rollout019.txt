"""
Propose a step‑function (non‑negative list) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three ideas that were not present in the reference
implementation:

1️⃣  A symmetric parametric family – a mixture of the uniform distribution
    and two exponential tails (left‑hand and right‑hand).  By sweeping the
    decay factor and the mixing weight we can reach regions of the search
    space that the previous one‑sided sweep never visits.

2️⃣  A cheap random‑Dirichlet exploration.  Sampling a large number of
    probability vectors with random concentration parameters often discovers
    “lucky” patterns that a deterministic sweep misses.

3️⃣  A projected Adam optimiser (single‑active sub‑gradient) followed by a
    stochastic hill‑climb.  Adam refines the best candidate found by the
    two preceding stages, and the hill‑climb finalises the solution with
    tiny mass transfers.

All steps respect the 1000 s time budget and work entirely on the simplex
(sum = 1, a_i ≥ 0).  The returned list can be fed directly into
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:               # numerically impossible, but keep a safe fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Scoring utility (the same expression that evaluate_sequence uses)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray, n: int, L_fft: int) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    conv = _conv_self(seq, L_fft)
    cur_max = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # should never happen but keep safe
        return np.inf
    return float(2.0 * n * cur_max / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (single active shift)
# ----------------------------------------------------------------------
def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) obtained from the *single* active shift:
    the shift k where the convolution attains its maximum.
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser (projected onto the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_score_initial: float,
    n: int,
    L_fft: int,
    max_iter: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that uses the *single‑active* sub‑gradient.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15      # a little softer than the reference code

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a) == 1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient from the (single) active shift
        grad = _subgrad_one_active(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _project_simplex(a)

        # occasional tiny Gaussian shake when we have stalled
        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(
    start_a: np.ndarray,
    start_time: float,
    deadline: float,
    n: int,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Tiny random mass transfers between two indices.  Only moves that improve
    the *true* objective are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _score(a, n, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction of source mass, up to 5 % (larger moves allow escape)
        frac = rng.random() * 0.05 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        cur_score = _score(a_prop, n, L_fft)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point – the function the grader will call
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The search consists of:
        * a sweep over a *symmetric* exponential‑plus‑uniform family,
        * a large random‑Dirichlet exploration,
        * a projected Adam optimisation,
        * a final hill‑climb.
    The best sequence found before the deadline is returned as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the very last hill‑climb
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # problem size (large enough to give a smooth shape)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # uniform baseline (already normalised to sum = 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score(best_seq, n, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Symmetric exponential + uniform sweep
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # decay rates (log‑spaced, focusing on the region that usually helps)
    eps_grid = np.logspace(-5, -2, 200)          # ε = 1‑r
    lam_grid = np.linspace(0.0, 1.0, 101)       # mixing weight of the uniform part

    for eps in eps_grid:
        # stop early if we are close to the final deadline
        if time.time() >= deadline - 90:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo = geo / geo.sum()                    # normalised left‑hand exponential
        geo_rev = geo[::-1]                      # reflected copy (right‑hand tail)
        geo_sym = 0.5 * (geo + geo_rev)          # symmetric mixture, still sums to 1

        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo_sym
            # extra safety normalisation (numerical noise)
            seq = seq / seq.sum()
            cur = _score(seq, n, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] "
                    f"sweep eps={eps:.1e} λ={lam:.2f} score={cur:.6f}"
                )

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet exploration (wide variety of shapes)
    # ------------------------------------------------------------------
    # Allow ~200 s for this stage (adjustable depending on time left)
    rand_deadline = deadline - 150.0
    while time.time() < rand_deadline:
        # concentration parameter α drawn log‑uniformly in [10⁻³, 10]
        log_alpha = rng.uniform(-3.0, 1.0)
        alpha = 10.0 ** log_alpha

        # Dirichlet via Gamma draws
        g = rng.gamma(alpha, 1.0, size=n)
        if g.sum() == 0:
            continue
        seq = g / g.sum()
        cur = _score(seq, n, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Random α={alpha:.3e} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimiser (single‑active sub‑gradient)
    # ------------------------------------------------------------------
    adam_deadline = deadline - 40.0
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        a_opt, sc_opt = _adam_optimize(
            best_seq, start_time, adam_deadline, best_score, n, L_fft
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb (tiny mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline, n, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
