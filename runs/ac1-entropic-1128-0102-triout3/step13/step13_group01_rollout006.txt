"""
A fast, gradient‑based optimiser for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

using a soft‑max parametrisation and a smoothed “log‑sum‑exp”
approximation of the max.  The optimiser runs several low‑dimensional
analytic seeds (uniform, triangular, power‑ramps, beta‑densities) and a
handful of random starts, refines each with a continuation of
L‑BFGS‑B runs for increasing smoothing parameters β, and finishes with a
light Metropolis‑type annealing.  The whole search respects the 1000 s
budget and returns a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)-1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors (full length)."""
    n = u.shape[0] + v.shape[0] - 1
    L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Soft‑max parametrisation (guarantees a_i ≥ 0, Σa_i = 1)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    s = np.sum(e)
    return e / s


# ----------------------------------------------------------------------
# Exact objective (no safety checks) for a normalised sequence
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a already normalised (sum=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective + gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
class _ObjGrad:
    """
    Callable returning (value, gradient) for the smoothed objective.
    The variable x lives in ℝⁿ and the actual sequence is a = softmax(x).
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, x: np.ndarray):
        """Return (f, g) where g is ∂f/∂x."""
        a = _softmax(x)                         # a ≥ 0, Σa = 1
        conv = _conv_self(a)                    # length 2n‑1
        m = conv.max()
        # soft‑max weights over the convolution entries
        w = np.exp(self.beta * (conv - m))
        w_sum = w.sum()
        w /= w_sum                               # Σ w = 1
        # smoothed max = m + log w_sum / β
        smoothed_max = m + np.log(w_sum) / self.beta
        f = 2.0 * self.n * smoothed_max          # smoothed objective

        # ---- gradient w.r.t. a ----
        # we need   g_a = 2·n·(conv(w,a) + a)
        # where conv(w,a)[t] = Σ_i w[t+i]·a[i]  (i from 0..n‑1, t from 0..n‑1)
        # This is a convolution of w with the *reversed* a.
        a_rev = a[::-1]
        full = _conv_full(w, a_rev)             # length 3n‑2
        # slice the needed n entries (indices n‑1 … 2n‑2)
        conv_w_a = full[self.n - 1 : self.n - 1 + self.n]
        grad_a = 2.0 * self.n * (conv_w_a + a)

        # ---- chain rule for the soft‑max (x → a) ----
        s = np.dot(a, grad_a)                    # scalar ⟨a,∇ₐf⟩
        grad_x = a * (grad_a - s)                # ∇ₓf = diag(a)(∇ₐf - ⟨a,∇ₐf⟩)

        return f, grad_x


# ----------------------------------------------------------------------
# Analytic families of shapes (always normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with a flat plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau = int(round(alpha * n))
    # enforce even plateau length (makes the shape perfectly symmetric)
    if plateau % 2 == 1:
        plateau -= 1
    side = (n - plateau) // 2
    if side <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side + 1, dtype=float) / side
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, a: float, b_: float) -> np.ndarray:
    """Discretised beta distribution on {0,…,n‑1} (mid‑point rule)."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b_ - 1.0))
    shape /= shape.sum()
    return shape


def _make_triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangle peaking at the centre."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=float)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=float)
        right = left[:-1][::-1]
    shape = np.concatenate([left, right])
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Light Metropolis‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> np.ndarray:
    """Very cheap annealer moving tiny packets of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective_exact(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # pick a source index proportional to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 50 %) of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective_exact(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a
