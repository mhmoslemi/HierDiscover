def optimize_for_n(n, deadline):
    L_fft = _next_pow2(2*n -1)  # for conv

    # Baseline uniform
    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = _objective(best_seq)
    
    # parametric sweep: geometric+uniform
    idx = np.arange(n)
    eps_grid = np.logspace(-4, -1, 200)  # more coarse maybe
    lam_grid = np.linspace(0, 1, 21)
    for eps in eps_grid:
        # break if time
        if time.time() > deadline - 30: break
        r = 1 - eps
        geo = r**idx
        geo /= geo.sum()
        for lam in lam_grid:
            seq = lam*uniform + (1-lam)*geo
            seq /= seq.sum()
            score = _objective(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    # random Dirichlet seeds
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    for alpha in alphas:
        for _ in range(6):
            if time.time() > deadline - 30: break
            seq_raw = np.random.default_rng().gamma(alpha, 1.0, size=n)
            seq = seq_raw / seq_raw.sum()
            score = _objective(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    # run Adam from best_seq multiple times with random perturbations
    num_restarts = 3
    for restart in range(num_restarts):
        if time.time() > deadline - 30: break
        # perturb best seq slightly
        a0 = best_seq.copy()
        # add small random noise and project
        noise = 0.01 * np.random.default_rng().normal(size=n) * a0
        a0 = np.maximum(a0 + noise, 0)
        a0 = _project_simplex(a0)
        a_opt, sc_opt = _adam_optimize(a0, start_time, deadline, best_score, max_iter=200000)

    # final hill climb
    a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
    if sc_hc < best_score:
        best_score = sc_hc
        best_seq = a_hc.copy()

def _conv_self_fft(a, L=None):
    n = len(a)
    L = L or _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    return conv

"""
Propose a non‑negative step‑function (list of floats) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines several cheap global‑search ideas with a
projected Adam optimiser and a stochastic hill‑climb:

1.  Try several problem sizes (n = 8192, 4096, 2048) – larger n
    gives more freedom, but the term 2·n grows, so we keep a modest
    range.

2.  For each n we start from:
    • the uniform distribution,
    • a fine 2‑D sweep of a geometric family mixed with a uniform part,
    • a few Dirichlet‑drawn random simplexes (different concentration α).

3.  The best sequence found so far is refined by a projected Adam optimiser
    that follows the sub‑gradient of the *single* maximising convolution shift.
    A small amount of Gaussian noise is injected when the optimiser plateaus.

4.  Finally a stochastic hill‑climb moves a tiny (up‑to‑5 %) fraction of mass
    between two random positions and accepts only improving moves.

All operations respect the given time budget (default 1000 s).  The returned
list is normalised to sum to 1, contains only non‑negative numbers and is
ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using real FFT (returns length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # all zero or negative – return uniform
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (single active shift)
# ----------------------------------------------------------------------
def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the *single* most‑violating convolution entry.
    For shift k with maximal value,   ∂/∂a_i (a⋆a)[k] = 2·a_{k‑i}
    for all i that contribute to that shift.
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Objective helper (sum OK, we keep sum = 1 after projection)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)^2)."""
    seq = np.asarray(seq, dtype=float)
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self_fft(seq)
    return float(2.0 * len(seq) * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Projected Adam optimiser using the single‑active sub‑gradient
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float,
                   max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the simplex; returns best seen sequence and its score."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            # occasional progress report
            if it % 5000 == 0:
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # sub‑gradient from the maximising shift
        grad = _subgrad_one_active(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the probability simplex
        a = _project_simplex(a)

        # plateau handling – a tiny Gaussian shake
        if it - last_improve > 3000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass between two positions, keep improvement."""
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self_fft(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer up to 5 % of source mass
        frac = rng.random() * 0.05 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        cur_score = float(2.0 * n * _conv_self_fft(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {best_score:.6f}")

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
        2 * n * max(conv(a, a)) / (sum(a)**2).

    The routine explores several lengths, performs a parametric sweep,
    random Dirichlet initialisations, a projected Adam optimiser and a
    final stochastic hill‑climb.  The best sequence found before the
    time budget expires is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the outer wrapper
    deadline = start_time + budget_s - 0.15

    # try a few lengths – larger n gives more flexibility, but 2·n also grows
    candidate_ns = [8192, 4096, 2048]
    global_best_seq = None
    global_best_score = float("inf")

    for n in candidate_ns:
        if time.time() >= deadline - 5:
            break
        seq_n, score_n = _optimize_n(n, start_time, deadline, rng)
        if score_n < global_best_score:
            global_best_score = score_n
            global_best_seq = seq_n.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best with n={n}: {score_n:.6f}")

    # fallback – should never happen but keeps the function safe
    if global_best_seq is None:
        n = 4096
        global_best_seq = np.full(n, 1.0 / n)
        global_best_score = _objective(global_best_seq)

    # final normalisation (make sure sum = 1 and no negatives)
    global_best_seq = np.maximum(global_best_seq, 0.0)
    s = global_best_seq.sum()
    if s > 0:
        global_best_seq = global_best_seq / s
    else:
        global_best_seq = np.full_like(global_best_seq, 1.0 / global_best_seq.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {global_best_score:.6f} (length {len(global_best_seq)})")
    return [float(x) for x in global_best_seq]
