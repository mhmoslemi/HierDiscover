        # compute gradient from top K shifts

"""
Propose a low‑objective step‑function.

The algorithm consists of three stages:
1.  Generate many analytic “warm‑start” shapes (power‑tents,
    exponentials, Beta‑PDFs, random Dirichlet draws, …), normalise them
    to sum‑1 and keep the best according to the exact objective
    `2·n·max(conv(a,a))`.
2.  Refine the best shape by a simple *exponentiated‑gradient* descent.
    At each iteration the convolution is recomputed (FFT), the
    `K` largest convolution entries are selected and a sub‑gradient
    of their sum with respect to the coefficients is built.
    The coefficients are multiplied by ``exp(-lr·grad)`` and re‑normalised.
    The learning rate decays from a relatively large value to a small
    one; occasional tiny Gaussian perturbations are added to escape
    possible plateaus.
3.  Return the best vector encountered during the search as a plain
    Python list – ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns a vector of length 2·n‑1 (full convolution).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective for a normalised (sum≈1) vector."""
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.shape[0])


def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of analytic shapes (unnormalised)."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    cand = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        cand.append(shape)

    # 2) Exponential decay from the centre (symmetric)
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        cand.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        cand.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]                     # symmetric
    cand.append(shape)

    # 5) Half‑cosine
    cand.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        cand.append(gauss)

    # 7) Beta‑like shapes (x^(a‑1)*(1‑x)^(b‑1))
    for a in (0.3, 0.5, 1.0, 2.0, 5.0):
        for b in (0.3, 0.5, 1.0, 2.0, 5.0):
            left = np.where(x > 0, x ** (a - 1), 0.0)
            right = np.where(x < 1, (1.0 - x) ** (b - 1), 0.0)
            shape = left * right
            cand.append(shape)

    # 8) Random Dirichlet‑like draws
    rng = np.random.default_rng(0)
    for _ in range(12):
        cand.append(rng.random(n))

    # normalise all candidates (sum to 1) and keep only those with positive mass
    normalised = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised


def _gradient_from_shifts(a: np.ndarray, shift_idxs: np.ndarray,
                         weights: np.ndarray) -> np.ndarray:
    """
    Build the (sub)gradient of the weighted sum of selected convolution
    entries with respect to the coefficient vector a.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector (sum = 1).
    shift_idxs : np.ndarray, shape (k,)
        Indices in the full convolution array (0 … 2n‑2) that are used.
        The corresponding shift is s = idx - (n‑1).
    weights : np.ndarray, shape (k,)
        Positive weights for each selected shift.

    Returns
    -------
    g : np.ndarray, shape (n,)
        Gradient vector.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    for idx, w in zip(shift_idxs, weights):
        s = idx - (n - 1)          # actual shift (may be negative)
        s_abs = abs(s)
        if s_abs >= n:
            # shift outside possible range – ignore
            continue
        # contribution: for shift s_abs,
        #   g[0:n-s]   += w * a[s:n]
        #   g[s:n]     += w * a[0:n-s]
        if s_abs == 0:
            # zero shift – adds 2·a (both terms coincide)
            g += 2.0 * w * a
        else:
            # first part
            g[: n - s_abs] += w * a[s_abs:]
            # second part
            g[s_abs:] += w * a[: n - s_abs]
    return g

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises
    ``2·n·max(conv(a,a)) / (sum(a)**2)``.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # a safety margin – the harness adds ~0.5 s of its own overhead
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    max_n = 16384                     # cap to keep the search fast
    try:
        n0 = len(height_sequence_1)   # type: ignore[name-defined]
        n = int(min(max_n, max(4096, n0)))  # at least 4 k, never exceed cap
    except Exception:
        n = 8192                      # default – good resolution & speed

    # --------------------------------------------------------------
    # 1) Warm‑start – diverse analytic shapes
    # --------------------------------------------------------------
    warm = _make_candidate_shapes(n)

    # include the previous best vector, if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm.append(starter / starter.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for cand in warm:
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    if best_vec is None:
        # Fallback – uniform distribution
        best_vec = np.full(n, 1.0 / n)

    a = best_vec.copy()
    best_vec = a.copy()
    best_score = _objective(a)
    print(f"[{time.time() - start:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Exponentiated‑gradient refinement
    # --------------------------------------------------------------
    total_time = deadline - start
    init_lr = 1.0        # start with a relatively aggressive step
    final_lr = 0.01      # final step size after the budget is exhausted
    K = 12               # number of top convolution entries used for the gradient
    it = 0
    last_improve = start

    while time.time() < deadline:
        it += 1
        # current convolution & objective
        conv = _conv_fft(a)
        cur_score = _objective_from_conv(conv, n)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = time.time()
            print(f"[{time.time() - start:.2f}s] iter={it:,} new best={best_score:.6f}")

        # select the K largest convolution entries (their indices)
        if K < conv.size:
            top_idx = np.argpartition(conv, -K)[-K:]
        else:
            top_idx = np.arange(conv.size)

        # equal weights (could be refined with a softmax, but equal works well)
        weights = np.ones_like(top_idx, dtype=float)

        # build gradient
        g = _gradient_from_shifts(a, top_idx, weights)

        # learning‑rate schedule (linear decay)
        elapsed = time.time() - start
        frac = min(1.0, elapsed / total_time)
        lr = init_lr * (1.0 - frac) + final_lr * frac

        # exponentiated‑gradient update (keeps positivity automatically)
        a = a * np.exp(-lr * g)

        # re‑normalise to stay on the simplex
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            # guard against numerical failure – fall back to uniform
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional tiny random jitter to help escape plateaus
        if it % 500 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        # safe‑guard: if we have gone for a while without improvement,
        # re‑initialise from the best warm‑start (helps avoid being stuck)
        if time.time() - last_improve > 30.0:
            # pick a random warm‑start and continue from there
            a = rng.choice(warm)
            last_improve = time.time()

    # --------------------------------------------------------------
    # 3) Return the best vector found
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
