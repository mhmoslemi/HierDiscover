def adam_smooth_max_descent(a0, start_time, deadline, best_seq, best_score):
   a = a0.copy()
   n = len(a)
   L_fft = _next_pow2(2*n - 1)
   L_grad = _next_pow2(3*n - 2)
   a_rev = a[::-1]
   τ = 5.0
   τ_factor = 1.3
   α = 0.02
   β1 = 0.9
   β2 = 0.999
   eps = 1e-8
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   t = 0
   while time.time() < deadline:
        t += 1
        # full convolution and raw score
        conv = _conv_self(a, L_fft)
        raw_score = 2.0 * n * conv.max()
        if raw_score < best_score:
             best_score = raw_score
             best_seq = a.copy()
             print(f"[{time.time()-start_time:.2f}s] Adam iteration {t:,} score={raw_score:.6f}")

        # compute softmax weights
        max_c = conv.max()
        w = np.exp(τ * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(conv)
            w[ np.argmax(conv) ] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # gradient:
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev) # length 3n-2
        grad = 2.0 * conv_w[n-1: n-1+n]

        # Adam updates
        m = β1 * m + (1 - β1) * grad
        v = β2 * v + (1 - β2) * (grad ** 2)
        m_hat = m / (1 - β1 ** t)
        v_hat = v / (1 - β2 ** t)
        step = α * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # enforce non-negative using projection onto simplex
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # increase temperature gradually
        if t % 2000 == 0:
            τ = min(τ * τ_factor, 1e9)

   return best_seq, best_score

"""
Improved optimiser for the “step‑function” challenge.

The objective (scale‑invariant) is
    2·n·max(conv(a,a)) / (sum(a)²) .
We therefore normalise the vector to sum to 1 and minimise
    2·n·max(conv(a,a)).

The optimiser works in four stages (adjusted from the reference run).

1️⃣  *Rich parametric sweep* – geometric, power, beta families and an
    additional sinusoidal family, scanned on a fine grid (α,β up to 8).

2️⃣  *Diverse Dirichlet starts* – draws with many concentration parameters.

3️⃣  *Adam‑accelerated smooth‑max descent* – smooth approximation of the
    max‑convolution (soft‑max over shifts) with an Adam optimiser and a
    geometrically increasing temperature τ.  After each step the vector is
    projected onto the probability simplex.

4️⃣  *Fine‑grained hill‑climb* – moves a tiny amount of mass from the index
    with the most positive sub‑gradient to the one with the most negative,
    accepting only improving moves.

The routine respects a 1000 s budget and finally returns a plain list of
Python floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full self‑convolution a*a."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helper (scale‑invariant, fast)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (used for hill‑climb)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta, sinusoid)
# ----------------------------------------------------------------------
def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # ---- geometric ----------------------------------------------------
    for eps in np.logspace(-5, -0.5, 250):            # more values, finer near 0
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        # blend with uniform – the blend factor is scanned coarsely
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    for p in np.linspace(-4.0, 4.0, 161):
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.linspace(0.05, 8.0, 101)   # extended range, finer grid
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))  # stabilise
            s = seq.sum()
            if not np.isfinite(s) or s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- sinusoid family ------------------------------------------------
    for power in np.linspace(0.5, 6.0, 12):
        if time.time() >= deadline:
            break
        sinus = np.sin(np.pi * (idx + 0.5) / n) ** power
        s = sinus.sum()
        if s == 0:
            continue
        sinus /= s
        sc = _objective_score(sinus)
        if sc < best_score:
            best_score, best_seq = sc, sinus.copy()
            print(f"[{time.time() - start_time:.2f}s] Sin power={power:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 80
    for _ in range(trials):
        if time.time() >= deadline:
            break
        conc = rng.uniform(0.05, 10.0)           # wider concentration range
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet α={conc:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam‑accelerated smooth‑max descent
# ----------------------------------------------------------------------
def _adam_smooth_max_descent(a0: np.ndarray,
                             start_time: float,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of the max‑convolution.
    Uses Adam optimisation with a temperature τ that is increased
    geometrically, turning the smooth proxy into the true max.
    """
    a = a0.copy()
    n = a.shape[0]

    L_fft = _next_pow2(2 * n - 1)          # for the self‑convolution
    L_grad = _next_pow2(3 * n - 2)         # for gradient convolution

    # Adam hyper‑parameters
    alpha = 0.025
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # temperature schedule
    tau = 5.0
    tau_factor = 1.3
    tau_inc_every = 2500   # increase τ every this many iterations

    while time.time() < deadline:
        t += 1

        # ----- current objective (real, not smooth) -----
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} score={cur_score:.6f}")

        # ----- smooth weights (soft‑max) -----
        max_c = conv.max()
        # avoid overflow by shifting
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient of the smooth max -----
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)                # length = 3n‑2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]       # slice of length n

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # ----- temperature increase -----
        if t % tau_inc_every == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective_score(a)

    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source: index with largest *positive* sub‑gradient among those that have mass
        src_cands = np.where(a > 1e-15)[0]
        if src_cands.size == 0:
            break
        src = src_cands[np.argmax(grad[src_cands])]

        # destination: index with most *negative* sub‑gradient
        dst = np.argmin(grad)

        if src == dst:
            # fall back to a tiny random shuffle
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        # transfer a small fraction of the source mass
        frac = rng.random() * 0.006 + 1e-5   # between 1e‑5 and ~0.006
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # re‑project to the simplex to keep sum = 1 and non‑negativity
        a_prop = _project_simplex(a_prop)

        new_score = _objective_score(a_prop)
        if new_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0,

    """

    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem size – use a larger n when we have enough time.

    if budget_s >= 900:

    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1 if present)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:

            if starter.size != n:
                x_old = np.linspace(0, 1, starter.size)
                x_new = np.linspace(0, 1, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0  # reserve ~250 s for later phases
    if time.time() < phase_deadline:

        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet starts

    # --------------------------------------------------------------
    # 3️⃣  Adam‑accelerated smooth‑max descent
    # --------------------------------------------------------------
    hill_time = 120.0
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth‑max descent")
        best_seq, best_score = _adam_smooth_max_descent(
            best_seq, start_time, smooth_deadline, best_seq, best_score

    # --------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, start_time, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
