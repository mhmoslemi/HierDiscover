    L = 1 << ((2*n-2).bit_length())
    fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft*fft, n=L)

def _pgd_optimize(a0, start_time, deadline):
    a = a0.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a, L_self).max()
    it = 0
    lr = 0.1
    beta_start = 5.0
    beta_end = 200.0
    max_iter = 1000000
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute current conv and score
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it} score={cur_score:.6f}")
        # compute beta schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 5000.0)  # ramp over 5000 iterations
        # compute gradient via softmax weights
        max_val = conv.max()
        exps = np.exp(beta * (conv - max_val))
        w = exps / exps.sum()
        # compute gradient
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        conv_rev = np.fft.irfft(Af * Wf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]
        # step size dynamic: lr / sqrt(it) etc.
        step = lr / np.sqrt(it)
        # try update
        a_new = _project_simplex(a - step * grad)
        conv_new = _conv_self(a_new, L_self)
        new_score = 2.0 * n * conv_new.max()
        # accept if improved else reduce step; we can reduce lr adaptively
        if new_score < cur_score:
            a = a_new
        else:
            # reduce lr for next iteration maybe
            lr *= 0.5
            if lr < 1e-6:
                lr = 0.1
        # check for stagnation: if no improvement for many iterations, break maybe
        # not needed
    return best_a, best_score

def _hill_climb(a_start, start_time, deadline):
    a = a_start.copy()
    n = a.shape[0]
    L_self = _next_pow2(2*n - 1)
    best_a = a.copy()
    best_score = 2 * n * _conv_self(a, L_self).max()
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer fraction up to 0.05 of source mass
        frac = rng.random() * 0.05 + 1e-6
        delta = a[src] * frac
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        # project onto simplex? Since sum unchanged and nonnegative okay.
        # Ensure nonneg
        if a_prop[src] < -1e-12:
            continue
        # Evaluate
        score = 2 * n * _conv_self(a_prop, L_self).max()
        if score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = score
            print(f"[{time.time() - start_time:.2f}s] Hill step it={it} score={score:.6f}")
    return best_a, best_score

"""
Propose a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
1. **Coarse parametric sweep (small n)**
   – Search the family  a = λ·uniform + (1‑λ)·geometric(r)
   – Add a random mixture of two geometric sequences.
   – Work on n_small = 1024 for speed, keep the best parameters,
     then up‑sample to the full size n = 4096.

2. **Projected gradient descent with a soft‑max surrogate**
   – The true objective contains a `max`.  Replace it by a smooth
     soft‑max with an annealing temperature β.
   – Gradient of the surrogate can be written as a convolution, so
     each iteration uses two FFT‑based convolutions (O(n log n)).
   – After the gradient step we project back onto the probability
     simplex (Euclidean projection).

3. **Stochastic hill‑climb**
   – Tiny random moves of mass between two positions are accepted
     only if they improve the *real* objective.
   – Larger moves (up to 1 % of the source mass) are allowed.

All three phases respect a total budget of 1000 s; the routine returns
a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑2: projected gradient descent with soft‑max surrogate
# ----------------------------------------------------------------------


def _pgd_optimize(a0: np.ndarray,
                  start_time: float,
                  deadline: float,
                  max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent that minimises a soft‑max approximation of the
    max‑convolution.  Returns the best vector seen and its true objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)          # for a*a
    L_grad = _next_pow2(3 * n - 2)          # for w_rev * a
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a, L_self).max()
    # schedule parameters
    lr0 = 0.15                               # base learning‑rate
    beta_start = 5.0
    beta_end = 200.0
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # current convolution and objective
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        # annealing factor for the soft‑max
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 5_000.0)

        # soft‑max weights over the 2n‑1 convolution entries
        max_val = conv.max()
        w = np.exp(beta * (conv - max_val))
        w /= w.sum()                     # shape (2n‑1,)

        # gradient of the weighted sum = 2 * (w_rev * a)  (convolution)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        conv_rev = np.fft.irfft(Af * Wf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]

        # simple step‑size schedule
        step = lr0 / np.sqrt(it)

        # try the update
        a_candidate = _project_simplex(a - step * grad)
        conv_cand = _conv_self(a_candidate, L_self)
        cand_score = 2.0 * n * conv_cand.max()

        if cand_score < cur_score:
            a = a_candidate
        else:
            # if not improving, shrink the learning‑rate a little
            lr0 *= 0.7
            if lr0 < 1e-6:      # avoid getting stuck at an infinitesimal step
                lr0 = 0.15

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: simple hill‑climb on the true objective
# ----------------------------------------------------------------------


def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: move a tiny random fraction of mass from one index
    to another; keep the move only if it improves the true objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a, L_self).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer up to 1 % of the source mass
        frac = rng.random() * 0.01 + 1e-6
        delta = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < -1e-14:
            continue

        score = 2.0 * n * _conv_self(a_prop, L_self).max()
        if score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    2·n·max(conv(a,a)) / (sum(a)**2).

    The search consists of three stages:
        1️⃣  Fast parametric sweep on a small problem (n_small).
        2️⃣  Projected gradient descent with a soft‑max surrogate.
        3️⃣  Final stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep on a reduced size (n_small)
    # ------------------------------------------------------------------
    n_small = 1024
    idx_small = np.arange(n_small, dtype=np.float64)
    L_small = _next_pow2(2 * n_small - 1)

    uniform_small = np.full(n_small, 1.0 / n_small)
    best_seq_small = uniform_small.copy()
    best_score_small = 2.0 * n_small * _conv_self(best_seq_small, L_small).max()
    print(f"[{time.time() - start_time:.2f}s] Small‑uniform baseline score={best_score_small:.6f}")

    # fine grid over (eps, λ) for a single geometric component
    eps_grid = np.logspace(-5, -2, 120)                # 1e‑5 … 1e‑2
    lam_grid = np.arange(0.0, 1.001, 0.01)            # λ ∈ [0,1]

    phase1_deadline = start_time + 0.35 * budget_s   # ~35 % of total time
    for eps in eps_grid:
        if time.time() >= phase1_deadline:
            break
        r = 1.0 - eps
        geo = r ** idx_small
        geo /= geo.sum()
        for lam in lam_grid:
            seq = lam * uniform_small + (1.0 - lam) * geo
            seq /= seq.sum()
            cur = 2.0 * n_small * _conv_self(seq, L_small).max()
            if cur < best_score_small:
                best_score_small = cur
                best_seq_small = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Grid eps={eps:.1e} λ={lam:.2f} score={cur:.6f}")

    # random mixtures of two geometric families
    n_random = 3000
    for _ in range(n_random):
        if time.time() >= phase1_deadline:
            break
        eps1 = 10 ** np.random.uniform(-5, -2)
        eps2 = 10 ** np.random.uniform(-5, -2)
        r1 = 1.0 - eps1
        r2 = 1.0 - eps2
        geo1 = r1 ** idx_small
        geo2 = r2 ** idx_small
        geo1 /= geo1.sum()
        geo2 /= geo2.sum()
        lam = np.random.rand()
        w1 = np.random.rand() * (1.0 - lam)
        w2 = 1.0 - lam - w1
        seq = lam * uniform_small + w1 * geo1 + w2 * geo2
        seq /= seq.sum()
        cur = 2.0 * n_small * _conv_self(seq, L_small).max()
        if cur < best_score_small:
            best_score_small = cur
            best_seq_small = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Random mix score={cur:.6f}")

    # ------------------------------------------------------------------
    # Upsample the best small solution to the full size n = 4096
    # ------------------------------------------------------------------
    n = 4096
    idx_full = np.arange(n, dtype=np.float64)
    # Linear interpolation (periodic endpoints are not needed)
    best_seq_full = np.interp(
        np.linspace(0, 1, n, endpoint=False),
        np.linspace(0, 1, n_small, endpoint=False),
        best_seq_small,
    )
    best_seq_full = np.maximum(best_seq_full, 0.0)
    best_seq_full /= best_seq_full.sum()

    # optionally start from a globally supplied candidate
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur = 2.0 * n * _conv_self(starter, _next_pow2(2 * n - 1)).max()
                cur_full = 2.0 * n * _conv_self(best_seq_full, _next_pow2(2 * n - 1)).max()
                if cur < cur_full:
                    best_seq_full = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter adopted, score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Projected gradient descent with soft‑max surrogate
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.80 * budget_s   # up to ~80 % of total time
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting PGD optimiser")
        a_opt, sc_opt = _pgd_optimize(best_seq_full,
                                      start_time,
                                      phase2_deadline)
        cur_full = 2.0 * n * _conv_self(best_seq_full,
                                        _next_pow2(2 * n - 1)).max()
        if sc_opt < cur_full:
            best_seq_full = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq_full, start_time, deadline)
        cur_full = 2.0 * n * _conv_self(best_seq_full,
                                        _next_pow2(2 * n - 1)).max()
        if sc_hc < cur_full:
            best_seq_full = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improvement score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq_full = np.maximum(best_seq_full, 0.0)
    total = best_seq_full.sum()
    if total > 0:
        best_seq_full = best_seq_full / total
    else:
        best_seq_full = np.full(n, 1.0 / n)

    final_score = 2.0 * n * _conv_self(best_seq_full,
                                      _next_pow2(2 * n - 1)).max()
    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {final_score:.6f}")

    return [float(x) for x in best_seq_full]
