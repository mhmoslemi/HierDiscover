assistantfinal**Solution Overview**

The program searches for a non‑negative step function `a` (a list of floats) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max\bigl( a*a \bigr)}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the linear convolution of `a` with itself.  
The optimisation consists of three phases

1. **Exploratory parametric sweep** – we try families of *U‑shaped* distributions  

   * `a_i ∝ (i+1)^p·(n‑i)^q` (product of two power laws)  
   * `a_i ∝ exp\bigl(α·(i‑(n‑1)/2)^2\bigr)` (convex Gaussian).  

   Every shape is mixed with the uniform distribution (`λ·uniform + (1‑λ)·shape`) and
   projected onto the probability simplex. The best vector discovered is kept.

2. **Simple sub‑gradient descent** – with the current best vector we minimise the
   *exact* `max(conv)` using the sub‑gradient  
   `∂/∂a_i max(conv) = 2·a_{k*‑i}` where `k*` is the index of the maximal convolution entry.
   After each step we project back onto the simplex.

3. **Targeted hill‑climb** – random mass‑transfer moves and moves that explicitly
   reduce the dominant convolution entry are performed. Only improving moves are kept,
   and the step size is automatically shrunk when a move is rejected.

All phases respect the 1000 s time budget, constantly checking the remaining time.
The best vector found is normalised to sum 1, printed together with the true score
(`evaluate_sequence`) and returned as a plain list of Python floats.

```python
# -*- coding: utf-8 -*-
"""
Propose a non‑negative step function (list of floats) that minimises  

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1. **Parametric sweep** – explore two families of shapes that put more mass at the
   ends of the interval (U‑shaped).  Each shape is mixed with the uniform vector,
   projected onto the simplex and evaluated.

2. **Sub‑gradient descent** – minimise the exact max‑convolution using the
   sub‑gradient ∂max/∂a_i = 2·a_{k*‑i} (k* = index of the maximal convolution term).
   The iterate is projected onto the simplex after each step.

3. **Hill‑climb polish** – random and targeted mass‑transfer moves that only keep
   improvements, with an adaptive step‑size.

The routine stops before the 1000 s deadline, prints intermediate scores and
returns the best sequence found (as a list of Python floats).
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast, exact evaluation (sum≈1) – identical to evaluate_sequence but without
# the security clamping.
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution term
# ----------------------------------------------------------------------


def _subgrad_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on  max(conv(a,a))  with simplex projection.
    The step size decays like η₀ / √t.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    # initial learning‑rate (this is on the original scale, not the normalised one)
    eta0 = 0.5

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # compute convolution and locate the maximal entry
        conv = _conv_self(a, L)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a:
        #   grad_i = 2 * a_{k_max - i}   if 0 ≤ k_max‑i < n, else 0
        grad = np.zeros_like(a)
        i_min = max(0, k_max - (n - 1))
        i_max = min(k_max, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]

        # step size schedule
        lr = eta0 / math.sqrt(t)

        # gradient step + projection onto the simplex
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # current exact score (sum is forced to 1 → denominator = 1)
        cur_score = float(2.0 * n * max_val / (a.sum() ** 2))

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad it={t:,} score={cur_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (random + max‑entry moves)
# ----------------------------------------------------------------------


def _hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves (up to ~15 % of the source mass) plus
    targeted moves that directly reduce the currently dominant convolution entry.
    Only strictly improving moves are kept; step‑size shrinks on rejection.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # step‑size control
    step_factor = 0.15      # up to 15 % of source mass per move
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.6:
            # ---- random transfer -------------------------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # ---- targeted transfer ------------------------------------------------
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose the move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)  # reset after success
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score

    # ----- uniform baseline -------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(
        f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}"
    )

    # ----- optionally start from a globally supplied starter -----------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep – two complementary families
    # --------------------------------------------------------------
    if time.time() < deadline - 300:   # reserve ~5 min for the sweep
        best_seq, best_score = _sweep_product_power(
            n, uniform, L_fft, deadline, start_time, best_seq, best_score
        )
        best_seq, best_score = _sweep_quadratic(
            n, uniform, L_fft, deadline, start_time, best_seq, best_score
        )

    # --------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the exact objective (multiple restarts)
    # --------------------------------------------------------------
    # we run a few short restarts with tiny random perturbations
    subgrad_seeds = [seed, seed + 10, seed + 20]
    for sg_seed in subgrad_seeds:
        if time.time() >= deadline - 120:
            break
        rng = np.random.default_rng(sg_seed)
        a0 = best_seq.copy()
        # tiny random jitter (≤1 % of each entry)
        a0 += 1e-4 * rng.standard_normal(n) * a0
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 15000
        a_opt, sc_opt = _subgrad_descent(
            a0,
            max_iters,
            deadline - 30,
            start_time,
            best_score,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad improvement score={sc_opt:.6f}"
            )

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb polish (random + targeted moves)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}"
            )

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n, dtype=float)

    # sanity check with the official evaluator (optional)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
