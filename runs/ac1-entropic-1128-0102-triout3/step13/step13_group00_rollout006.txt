"""
Step‑function optimiser – search for a non‑negative sequence minimising
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines several complementary heuristics:

1. **Parametric sweep** – families based on β‑distributions, power laws and
   geometric progressions are explored on a fine grid, including extreme
   exponents (α,β down to 0.01) that were not covered by the previous
   implementation.

2. **Random initialisers** – many Dirichlet draws (varying concentration)
   and random β‑mixtures provide diverse starting points.

3. **Smooth‑max descent with Adam** – a differentiable soft‑max approximation
   of the max‑convolution is minimised by Adam‑style projected gradient
   descent, gradually sharpening the approximation (τ ↑).

4. **Projected sub‑gradient descent** – directly uses the sub‑gradient of
   the true max‑convolution and a diminishing step‑size, staying on the
   probability simplex.

5. **Directed hill‑climb with line‑search** – repeatedly moves a small
   fraction of mass from the index with the largest positive sub‑gradient
   to the index with the most negative gradient, trying several step sizes
   and accepting only improving moves.

All stages respect the global time budget (default 1000 s) and keep the
best sequence found.  The final sequence is normalised to sum to 1 and
returned as a plain list of floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (scale‑invariant – sum is normalised to 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    Averaged over all active shifts attaining the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(a_best: np.ndarray, best_score: float,
                     n: int, deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-6, -1, 400)       # finer near zero
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)                # decreasing geometric
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, a_best = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return a_best, best_score


def _sweep_power(a_best: np.ndarray, best_score: float,
                 n: int, deadline: float) -> tuple[np.ndarray, float]:

    p_grid = np.linspace(-3.0, 3.0, 301)   # extended range

            best_score, a_best = sc, pow_seq.copy()

def _sweep_beta(a_best: np.ndarray, best_score: float,
                n: int, deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    # use log‑space grid for α,β: 0.01 … 10
    alphas = np.geomspace(0.01, 10.0, 80)
    betas = alphas.copy()
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            # stabilise exponentiation
            log_seq -= np.max(log_seq)
            seq = np.exp(log_seq)
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, a_best = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.3f} β={b:.3f} score={sc:.6f}")
    return a_best, best_score

def _random_beta_mixtures(a_best: np.ndarray, best_score: float,

    return a_best, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max descent (Adam optimiser)
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the smooth approximation
    S_τ(a) = (1/τ)·log Σ exp(τ·conv(a,a)).  τ grows geometrically.
    """
    a = a0.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    tau = 200.0
    tau_factor = 1.3
    lr0 = 0.08                # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # self‑convolution
        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} τ={tau:.1f} score={cur_score:.6f}")

        # soft‑max weights over shifts
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        # safety clipping
        w = np.clip(w, 0.0, 1e300)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2 * (w * rev(a))
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)                     # length 3n-2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]            # central slice

        # Adam update (projected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new

        # increase temperature occasionally
        if it % 1500 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score

def _subgrad_descent(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on max(conv).  Step size η_t = η0 / sqrt(t).
    """
    a = a0.copy()
    eta0 = 0.12
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new
    return best_seq, best_score

# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray,
                n: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained hill‑climb: move a tiny fraction of mass from the index with
    the largest positive sub‑gradient to the one with the most negative.
    Several candidate step sizes are tried; only improving moves are kept.
    """
    a = a_start.copy()
    L_self = _next_pow2(2 * n - 1)
    it = 0
    rng = np.random.default_rng()
    # candidate fractions of the source mass to move
    fracs = np.array([0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001])
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        grad = _subgradient_maxconv(a, conv)

        # pick source (largest positive grad) among entries that still have mass
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)  # most negative (or smallest) gradient

        if src == dst:
            # fall back to a random pair
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        improved = False
        best_local = best_score
        best_move = None

        for f in fracs:
            eps = a[src] * f
            if eps <= 0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            # maintain feasibility
            if a_prop[src] < -1e-12:
                continue
            cur_score = 2.0 * n * _conv_self_fft(a_prop).max()
            if cur_score < best_local:
                best_local = cur_score
                best_move = a_prop
                improved = True
                # keep searching for even better fraction
        if improved and best_move is not None:
            a = best_move
            best_score = best_local
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} score={best_score:.6f}")
        else:
            # occasional random jitter to escape flat plateaus
            if it % 500 == 0:
                jitter = rng.normal(scale=0.002, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                cur_score = 2.0 * n * _conv_self_fft(a_jit).max()
                if cur_score < best_score:
                    a = a_jit
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Jitter it={it:,} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns a plain Python list of floats (the sequence).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    # problem size – keep a power of two for fast FFTs
    n = 4096

    # ------------------------------------------------------------------
    # Initialise with uniform (baseline) and optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        # ``height_sequence_1`` may be provided by the environment
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep
    # ------------------------------------------------------------------
    phase_deadline = deadline - 250.0   # reserve time for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(best_seq, best_score, n, phase_deadline)
        best_seq, best_score = _sweep_power(best_seq, best_score, n, phase_deadline)
        best_seq, best_score = _sweep_beta(best_seq, best_score, n, phase_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – random Dirichlet and β‑mixture initialisers
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet(best_seq, best_score, n, phase_deadline)
        best_seq, best_score = _random_beta_mixtures(best_seq, best_score, n, phase_deadline)

    # ------------------------------------------------------------------
    # Phase 3 – smooth‑max descent (Adam)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 120.0   # keep last ~2 min for fine tuning
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam‑based smooth‑max descent")
        best_seq, best_score = _smooth_max_adam(best_seq, n, smooth_deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent on true max
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 60.0
    if time.time() < subgrad_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting projected sub‑gradient descent")
        best_seq, best_score = _subgrad_descent(best_seq, n, subgrad_deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Phase 5 – final directed hill‑climb (with occasional jitter)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, n, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
