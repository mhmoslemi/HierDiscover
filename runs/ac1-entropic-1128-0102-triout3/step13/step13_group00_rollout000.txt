"""
Optimiser for the “step‑function” challenge – version 2.

The algorithm runs four phases inside the 1000 s budget:

1️⃣  Broad parametric sweep – mixes uniform with geometric, power,
    and beta‑type families.  Instead of exhaustive grids, a
    random sampling of the (α,β)‑space is used for speed.

2️⃣  Random Dirichlet initialisers – many draws with varied
    concentration to explore sparse / dense shapes.

3️⃣  Smooth‑max Adam descent – minimises the smooth log‑sum‑exp
    approximation of the max‑convolution.  The temperature τ is
    increased geometrically, the step uses the Adam update rule,
    and every iterate is projected onto the probability simplex.

4️⃣  Simulated‑annealing + directed hill‑climb – tiny mass‑transfer
    moves are performed with a Metropolis acceptance rule; after
    the annealing schedule finishes a final deterministic hill‑climb
    (moving mass from the most positive to the most negative
    sub‑gradient) cleans up the remaining noise.

All convolutions are performed with FFT (real‑valued) for speed,
and every phase respects the remaining time budget.  The best
sequence found is normalised to sum = 1 and returned as a plain
list of Python floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution via FFT (full length) using a fixed length L."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: x.shape[0] + y.shape[0] - 1]


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a (full length) using pre‑computed length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helper (scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray, L_self: int) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq, L_self)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the non‑smooth max‑convolution (used for hill‑climb)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# 1️⃣  Parametric families – random sampling of geometric / power / beta

def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    L_self: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)

    uniform = np.full(n, 1.0 / n)

    # ---- geometric mix ------------------------------------------------
    for eps in np.logspace(-5, -1, 200):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0, 1, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq, L_self)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power law ----------------------------------------------------
    for p in np.linspace(-3.0, 3.0, 151):
        if time.time() >= deadline:
            break
        pow_seq = (idx + 1.0) ** p
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq, L_self)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type (random sampling) ----------------------------------
    for _ in range(300):
        if time.time() >= deadline:
            break
        a = rng.uniform(0.05, 4.0)
        b = rng.uniform(0.05, 4.0)
        # log‑space evaluation to avoid overflow
        log_seq = (a - 1.0) * np.log(idx + 1.0) + (b - 1.0) * np.log(np.maximum(n - idx, 1))
        seq = np.exp(log_seq - np.max(log_seq))
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq, L_self)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Beta a={a:.2f} b={b:.2f} score={sc:.6f}")

    return best_seq, best_score

def _random_dirichlet_trials(n: int,

# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max Adam descent (log‑sum‑exp proxy)
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     L_self: int,
                     L_grad: int,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style minimisation of the smooth max‑convolution.
    Temperature τ starts low and is multiplied by τ_factor every
    τ_update_iter iterations.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam parameters
    lr = 0.015
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    tau = 2.0          # start relatively smooth
    tau_factor = 1.35
    tau_update_iter = 3000

    it = 0
    while time.time() < deadline:
        it += 1

        # self‑convolution and current score
        conv = _conv_self(a, L_self)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} tau={tau:.1f} score={cur_score:.6f}")

        # smooth soft‑max weights over the convolution entries
        max_c = conv.max()
        # use (conv - max_c) to avoid overflow
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2 * (w ⋆ a_rev) restricted to n entries
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev, L_grad)                       # length = 3n-2
        grad = 2.0 * conv_w[n - 1 : n - 1 + n]                     # slice of length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = _project_simplex(a_new)  # enforce non‑negativity & sum=1
        a = a_new

        # temperature increase
        if it % tau_update_iter == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Simulated annealing + final hill‑climb
# ----------------------------------------------------------------------
def _anneal_and_hill_climb(start_a: np.ndarray,
                           start_time: float,
                           deadline: float,
                           L_self: int) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with a geometrically decaying temperature,
    followed by a deterministic hill‑climb that moves mass from the
    most positive sub‑gradient to the most negative.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_self).max())
    rng = np.random.default_rng()

    # ------- annealing parameters -------
    T_start = 1e-2   # initial acceptance temperature (in objective units)
    T_end   = 1e-7
    anneal_time = (deadline - time.time()) * 0.6  # 60 % of remaining time
    t0 = time.time()
    # geometric decay: T = T_start * (T_end/T_start)^{elapsed/anneal_time}
    while time.time() < t0 + anneal_time:
        # propose a random tiny transfer
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer fraction in (0, 0.005)
        frac = rng.random() * 0.005 + 1e-6
        eps = a[src] * frac
        if eps <= 0.0:
            continue
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        # projection is not needed – total mass stays 1 and non‑negativity holds
        cur_score = float(2.0 * n * _conv_self(prop, L_self).max())
        delta = cur_score - best_score
        elapsed = time.time() - t0
        T = T_start * (T_end / T_start) ** (elapsed / anneal_time)
        accept = delta < 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12))
        if accept:
            a = prop
            if cur_score < best_score:
                best_score, best_a = cur_score, prop.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal score={best_score:.6f}")

    # ------- deterministic hill‑climb -------
    # Continue until time expires or no improvement in a full pass
    while time.time() < deadline:
        conv = _conv_self(a, L_self)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            # random fallback
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        # move a tiny fraction (up to 0.3 % of source)
        eps = a[src] * (rng.random() * 0.003 + 1e-6)
        if eps <= 0.0:
            continue

        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        cur_score = float(2.0 * n * _conv_self(prop, L_self).max())
        if cur_score < best_score:
            a = prop
            best_a = prop.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.  The routine follows the four
    phases described above and respects the time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gave a good trade‑off in the previous run
    # --------------------------------------------------------------
    n = 4096
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initial uniform baseline
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, L_self)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric, power, beta)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0  # reserve time for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, L_self, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, L_self, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam descent
    # --------------------------------------------------------------
    # Reserve ≈ 30 % of remaining time for the annealing/hill‑climb stage
    remaining = deadline - time.time()
    hill_time = remaining * 0.35
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth‑max descent")
        best_seq, best_score = _smooth_max_adam(
            best_seq,
            start_time,
            smooth_deadline,
            L_self,
            L_grad,
            best_seq,
            best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Simulated annealing + final hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing + hill‑climb")
        best_seq, best_score = _anneal_and_hill_climb(
            best_seq, start_time, deadline, L_self
        )
        print(f"[{time.time() - start_time:.2f}s] Final score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
