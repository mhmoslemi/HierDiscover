"""
search_step_function.py

A timed optimiser for the “step‑function” challenge.

Key ideas
---------
*   Use a larger pool of random Dirichlet starts (≈200) and keep the
    best few to launch local optimisation.
*   Local optimisation = Adam‑accelerated gradient descent on a *smooth*
    approximation of the max‑convolution (soft‑max with a temperature τ
    that starts low and is increased gradually).  The iterate is always
    projected onto the probability simplex.
*   After the Adam phase a cheap directed hill‑climb (mass transfer
    guided by the sub‑gradient of the true max) performs the final fine‑tuning.
*   Additional parametric families (geometric, power, beta, cosine)
    are scanned on a finer grid than in the reference solution.
*   The whole routine respects a 1000 s budget and prints periodic progress.

The optimiser works entirely with normalised vectors
(`sum(a)=1`).  The returned list can be used directly with
``evaluate_sequence`` because the function is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Simplex projection (O(n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (used for hill‑climb)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Parametric families
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray, best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Explore several analytic families on a fine grid."""
    idx = np.arange(n, dtype=np.float64)

    # uniform baseline (already evaluated elsewhere, kept for completeness)
    uniform = np.full(n, 1.0 / n)

    # ----- geometric ----------------------------------------------------
    for eps in np.logspace(-5, -1, 300):
        if time.time() >= deadline:
            return best_seq, best_score
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        # blend with uniform – the blend fraction is a cheap way to keep
        # the search space connected.
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} "
                      f"lam={lam:.2f} score={sc:.6f}")

    # ----- power (a_i ∝ (i+1)^p) ---------------------------------------
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            return best_seq, best_score
        pow_seq = np.power(idx + 1.0, p)
        pow_seq /= pow_seq.sum()
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ----- beta‑type (a_i ∝ i^{α-1} (n‑i)^{β-1}) -----------------------
    alphas = np.linspace(0.05, 6.0, 111)
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            return best_seq, best_score
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                return best_seq, best_score
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            seq_sum = seq.sum()
            if not np.isfinite(seq_sum) or seq_sum == 0:
                continue
            seq /= seq_sum
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ----- cosine bump (a_i ∝ 1+cos(π i/(n‑1))) -------------------------
    cos_seq = 1.0 + np.cos(np.pi * idx / (n - 1))
    cos_seq /= cos_seq.sum()
    sc = _objective_score(cos_seq)
    if sc < best_score:
        best_score, best_seq = sc, cos_seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Cosine bump score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float,
                             trials: int = 200) -> tuple[np.ndarray, float]:
    """Draw many Dirichlet vectors with varied concentration and keep the best."""
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        conc = rng.uniform(0.01, 8.0)          # wider range than before
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={conc:.3f} "
                  f"score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam‑accelerated smooth‑max descent
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth log‑sum‑exp approximation of max(conv).
    Temperature τ starts tiny (smooth average) and is multiplied by τ_factor
    every τ_step iterations, driving the surrogate towards the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07                      # base learning rate
    it = 0
    # Temperature schedule
    tau = 0.3
    tau_factor = 1.12
    tau_step = 1200               # increase τ every this many iterations

    best_local_seq = a.copy()
    best_local_score = best_score

    while time.time() < deadline:
        it += 1
        # ---------- forward pass ----------
        conv = _conv_self(a)                     # length 2n‑1
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                      f"score={cur_score:.6f}")

        # ---------- smooth max (soft‑max weights) ----------
        # stability trick: subtract max before exp
        w = np.exp(tau * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:               # numerical fallback
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ---------- gradient ----------
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)              # length 3n‑2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]       # slice of length n

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # ---------- temperature schedule ----------
        if it % tau_step == 0:
            tau = min(tau * tau_factor, 1e6)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (mass‑transfer)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """Fine‑grained hill‑climb guided by the exact sub‑gradient."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # source = index with most positive gradient (where increasing a hurts)
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)  # most negative gradient

        if src == dst:
            # fallback: random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # Transfer a modest fraction of the source mass
        eps = a[src] * (rng.random() * 0.01 + 1e-6)
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence``.  The routine follows four stages inside the
    allotted time budget:

    1. parametric sweep (geometric, power, beta, cosine);
    2. many random Dirichlet starts;
    3. Adam‑accelerated smooth‑max descent (temperature annealing);
    4. directed hill‑climb fine‑tuning.

    Returns a plain ``list[float]`` ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096                       # keep the size used by the reference run
    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initialise with uniform (baseline)
    # --------------------------------------------------------------
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (provided by the platform)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0          # reserve ~250 s for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After sweep score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, trials=200
        )
        print(f"[{time.time() - start_time:.2f}s] After Dirichlet trials score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Adam smooth‑max descent
    # --------------------------------------------------------------
    # reserve ~150 s for final hill‑climb
    hill_time = 150.0
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam‐smooth‑max descent")
        best_seq, best_score = _adam_smooth_max(
            best_seq, start_time, smooth_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, start_time, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
