"""
Improved search for a step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The idea is to start from a richer pool of analytic shapes (including
inverse power ramps, inverted Hann windows and mixtures of Gaussians),
run a fast Adam‑based projected sub‑gradient optimiser that uses the
exact active set of the current maximum convolution entries, and finish
with a long simulated‑annealing polish that moves relatively large
chunks of probability mass.  The optimiser works on a power‑of‑two
length (default 2**15) which keeps the FFT cheap and leaves most of the
budget for the annealing phase – the part that gave the biggest gains
in the previous implementation.  The code respects the 1000 s time
budget and returns a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length power‑of‑two)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2).  No safety checks."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # should never happen for a probability vector
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric shape built from two power ramps joining a flat plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_inverse_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Flip a power‑shape upside‑down (high at the edges, low in the centre).
    """
    base = _make_power_shape(n, p, alpha)
    inv = base.max() - base          # reflect around the maximum
    total = inv.sum()
    if total > 0.0:
        inv /= total
    else:
        inv = np.full(n, 1.0 / n)
    return inv


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_hann_inverse_shape(n: int) -> np.ndarray:
    """
    Inverted Hann window – maximum at the two ends, zero in the centre.
    w[i] = 0.5 + 0.5*cos(2π i/(n‑1)).
    """
    i = np.arange(n, dtype=float)
    shape = 0.5 + 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    # clip tiny negative values caused by floating‑point noise
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_mixture(n: int,
                           means: tuple[float, float],
                           sigmas: tuple[float, float],
                           weights: tuple[float, float] = (0.5, 0.5)) -> np.ndarray:
    """
    Two Gaussian bumps (centred at `means` – given in index units) with
    standard deviations `sigmas`.  The mixture is normalised to sum = 1.
    """
    xs = np.arange(n, dtype=float)
    g1 = np.exp(-0.5 * ((xs - means[0]) / sigmas[0]) ** 2)
    g2 = np.exp(-0.5 * ((xs - means[1]) / sigmas[1]) ** 2)
    shape = weights[0] * g1 + weights[1] * g2
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent
# ----------------------------------------------------------------------
def _adam_subgradient(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      top_k: int = 30) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the max‑convolution objective.
    Only the `top_k` largest convolution entries are used to build the
    (sub‑)gradient – this mimics the “active‑set” idea of the reference
    implementation while benefitting from Adam's adaptive step sizes.
    Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    best_a = a.copy()
    best_score = _objective(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,}  best={cur_score:.6f}")

        # active indices – exact top‑k (or all if very few)
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient for the selected indices
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # Adam update (projected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.12 / (np.sqrt(v_hat) + eps)   # a modest base learning rate
        a -= step * m_hat
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (more aggressive than the reference)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Move relatively large packets of probability mass between random
    positions.  Temperature decays linearly (fast at first, slowly at
    the end) and the acceptance rule follows the Metropolis criterion.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # schedule parameters
    total_steps = int(8e6 * (deadline - time.time()))  # roughly 8 M steps per full budget
    total_steps = max(200_000, total_steps)           # guarantee a minimum

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # linear cooling: T starts at 0.05 and falls to 1e‑4
        T = 0.05 * (1.0 - it / total_steps) + 1e-4

        src = rng.choice(n, p=a)          # more likely to move from a heavy bin
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move a *significant* fraction (up to 80 %) of the source mass
        eps = a[src] * rng.random() * 0.8
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    The routine builds a diverse pool of analytic shapes (including
    inverse ramps and Gaussian mixtures), refines the best one with an
    Adam‑based sub‑gradient method, and finishes with a long simulated‑
    annealing polish.  The returned list contains non‑negative floats and
    respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1) Choose a power‑of‑two length – 2**15 gives a good trade‑off
    # --------------------------------------------------------------
    n = 32768   # 2**15

    # --------------------------------------------------------------
    # 2) Build an initial pool of promising candidates
    # --------------------------------------------------------------
    pool = []                # list of (score, vector)
    pool_size = 12
    best_seq = None
    best_score = float("inf")

    # ---- uniform (baseline) -----------------------------------------
    a_uni = np.full(n, 1.0 / n)
    sc = _objective(a_uni)
    best_seq, best_score = a_uni.copy(), sc
    pool.append((sc, a_uni.copy()))

    # ---- power‑ramp family -------------------------------------------
    for p in np.linspace(0.2, 3.0, 11):
        for alpha in np.linspace(0.0, 0.5, 7):
            a = _make_power_shape(n, p, alpha)
            sc = _objective(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
            if len(pool) < pool_size:
                pool.append((sc, a.copy()))
                pool.sort(key=lambda x: x[0])
            elif sc < pool[-1][0]:
                pool[-1] = (sc, a.copy())
                pool.sort(key=lambda x: x[0])

    # ---- inverse power‑ramp ------------------------------------------
    for p in np.linspace(0.5, 4.0, 9):
        for alpha in np.linspace(0.0, 0.5, 5):
            a = _make_inverse_power_shape(n, p, alpha)
            sc = _objective(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
            if len(pool) < pool_size:
                pool.append((sc, a.copy()))
                pool.sort(key=lambda x: x[0])
            elif sc < pool[-1][0]:
                pool[-1] = (sc, a.copy())
                pool.sort(key=lambda x: x[0])

    # ---- beta‑distribution family ------------------------------------
    a_grid = np.linspace(0.3, 4.0, 9)
    b_grid = np.linspace(0.3, 4.0, 9)
    for a_par in a_grid:
        for b_par in b_grid:
            a = _make_beta_shape(n, a_par, b_par)
            sc = _objective(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
            if len(pool) < pool_size:
                pool.append((sc, a.copy()))
                pool.sort(key=lambda x: x[0])
            elif sc < pool[-1][0]:
                pool[-1] = (sc, a.copy())
                pool.sort(key=lambda x: x[0])

    # ---- inverted Hann window -----------------------------------------
    a = _make_hann_inverse_shape(n)
    sc = _objective(a)
    if sc < best_score:
        best_seq, best_score = a.copy(), sc
    if len(pool) < pool_size:
        pool.append((sc, a.copy()))
        pool.sort(key=lambda x: x[0])
    elif sc < pool[-1][0]:
        pool[-1] = (sc, a.copy())
        pool.sort(key=lambda x: x[0])

    # ---- Gaussian mixtures -------------------------------------------
    rng = np.random.default_rng()
    for _ in range(80):
        mu1 = rng.integers(int(0.1 * n), int(0.4 * n))
        mu2 = rng.integers(int(0.6 * n), int(0.9 * n))
        sigma1 = rng.uniform(0.02, 0.07) * n
        sigma2 = rng.uniform(0.02, 0.07) * n
        w1 = rng.random()
        w2 = 1.0 - w1
        a = _make_gaussian_mixture(n,
                                    means=(mu1, mu2),
                                    sigmas=(sigma1, sigma2),
                                    weights=(w1, w2))
        sc = _objective(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
        if len(pool) < pool_size:
            pool.append((sc, a.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, a.copy())
            pool.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Try to incorporate a globally supplied starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                if starter.size != n:
                    # Resample / interpolate onto our working length
                    x_old = np.linspace(0, 1, starter.size)
                    x_new = np.linspace(0, 1, n)
                    starter = np.interp(x_new, x_old, starter)
                starter = _project_simplex(starter)   # safety
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                if len(pool) < pool_size:
                    pool.append((sc, starter.copy()))
                    pool.sort(key=lambda x: x[0])
                elif sc < pool[-1][0]:
                    pool[-1] = (sc, starter.copy())
                    pool.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4) Local refinement – Adam‑based projected sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    now = time.time()
    remaining = deadline - now
    if remaining > 60.0:               # reserve at least a minute for annealing
        sub_deadline = now + 0.45 * remaining
        refined_seq, refined_score = _adam_subgradient(best_seq,
                                                       sub_deadline,
                                                       start_time,
                                                       top_k=30)
        if refined_score < best_score:
            best_seq, best_score = refined_seq.copy(), refined_score
            print(f"[{time.time() - start_time:.2f}s] Refined (Adam) score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (larger budget than before)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_seq, best_score = anneal_seq.copy(), anneal_score
            print(f"[{time.time() - start_time:.2f}s] Anneal improvement = {anneal_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalisation & conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
