def project_to_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function that (approximately) minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

using a multi‑start, targeted hill‑climbing search.
The algorithm

1.  Generates a diverse pool of analytic and random shapes and keeps the
    best few (after exact evaluation) as warm‑starts.
2.  For each warm‑start runs a fast O(n) “mass‑transfer” hill‑climber.
    The source of the transferred mass is chosen proportionally to its
    contribution to the current maximum of the convolution, i.e. indices
    where a[i]·a[max_idx‑i] is large.  Destination is chosen uniformly.
    Moves that do not increase the objective are kept; the step size shrinks
    as the time budget is exhausted.
3.  Occasionally recomputes the full convolution (FFT) to avoid drift.
4.  Returns the best vector found, normalised to sum = 1 (the objective is
    scale‑invariant).

The routine respects the 1000 s budget, prints progress and works for any
vector length (default 4096, or the length of the previously discovered
`height_sequence_1` if it exists).
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector (no assumption on its sum)."""
    n = a.shape[0]
    total = a.sum()
    if total <= 0.0:
        return math.inf
    conv = _conv_fft(a)
    return 2.0 * n * conv.max() / (total * total)


def _apply_move(a: np.ndarray, conv: np.ndarray,
                src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst` (in‑place update).
    The convolution `conv` (full length 2n‑1) is updated exactly.
    """
    n = a.shape[0]

    # linear part 2·a·Δ   (Δ = eps·(e_dst – e_src))
    conv[src:src + n] -= 2.0 * eps * a
    conv[dst:dst + n] += 2.0 * eps * a

    # quadratic part Δ·Δ
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # update the vector itself
    a[src] -= eps
    a[dst] += eps


def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """
    Produce a collection of analytic and random shapes (unnormalised).
    The caller should normalise them before evaluation.
    """
    rng = np.random.default_rng(12345)
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Symmetric exponentials
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Symmetric power‑law from the edges
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Half‑cosine
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussians with different σ
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet draws (sparse, medium, uniform‑like)
    for alpha in (0.2, 0.5, 1.0):
        for _ in range(6):
            rnd = rng.dirichlet(np.full(n, alpha))
            candidates.append(rnd)

    # 8) A few plain uniform random vectors
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # Normalise and discard the zero vectors
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)

    return normalised


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

        best_vec = np.full(n, 1.0 / n)

    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0.0:
        best_vec /= total
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
