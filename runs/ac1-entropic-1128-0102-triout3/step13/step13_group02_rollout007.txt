def subgradient_adam(a0, n, deadline, start_time, best_score_initial):
    L_fft = _next_pow2(2*n-1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    lr0 = 0.5
    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))
        # subgradient g
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]
        # Adam
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)

        a_candidate = _project_simplex(a_candidate)
        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score

                best_a = a.copy()

"""
Propose a step‑function (non‑negative list) that minimises

        2 * n * max(convolve(a, a)) / (sum(a)**2)

by a multi‑phase search:

1️⃣  Fast parametric sweep over several analytical families
    (geometric, power, mixed, Gaussian, triangular) to obtain a strong
    seed.

2️⃣  A handful of random restarts, each followed by an Adam‑style projected
    descent on a soft‑max approximation of the max‑convolution.  The
    temperature (β) is increased slowly so the optimisation first sees a
    smooth landscape and later focuses on the true objective.

3️⃣  Projected Adam on the *exact* sub‑gradient (the maximal shift of the
    convolution).  This refines the candidate on the true objective.

4️⃣  A simulated‑annealing polishing stage that repeatedly transfers a tiny
    amount of mass between two random coordinates, accepting worsening moves
    with a decreasing temperature.

All phases respect the 1000 s budget, keep the vector on the probability
simplex, and finally return the best sequence found (as a plain Python list
of floats).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, sum w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max Adam descent
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, best_score_initial: float):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0                # start temperature for soft‑max
    beta_mul = 1.02
    beta_max = 500.0

    while time.time() < deadline:
        t += 1

        # --- soft‑max of the convolution ---------------------------------
        conv = _conv_self_fft(a, L_conv)            # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                      # stabilise exponentials
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()               # soft‑max weights

        # gradient of the smooth max (correlation of w with reversed a)
        a_rev = a[::-1]
        W = rfft(w, n=L_corr)
        A = rfft(a_rev, n=L_corr)
        corr = irfft(W * A, n=L_corr)               # linear convolution
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # monitor exact objective
        cur_score = _eval_fast(a, L_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} "
                  f"score={cur_score:.6f}")

        # --- Adam update on the simplex -----------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _subgradient_adam(a0: np.ndarray, n: int, deadline: float,
                      start_time: float, best_score_initial: float):

    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    v = np.zeros_like(a)
    t = 0

        t += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))

        i_vals = np.arange(i_start, i_end + 1)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] sub‑grad Adam it={t:,} "
                  f"score={cur_score:.6f}")
        else:
            # occasional uphill move (simulated annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(t) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter every few thousand iterations
        if t % 2500 == 0:
            jitter = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – simulated‑annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_time: float, best_score_initial: float):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    temp = 0.02               # initial temperature
    decay = 0.9999

    while time.time() < deadline:
        # pick a source with a non‑negligible amount of mass
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 0.2 % of the source mass
        delta = a[src] * rng.random() * 0.002
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] SA improvement score={new_score:.6f}")
        else:
            # accept with probability exp(-(Δ)/temp)
            if rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
                a = a_prop

        # cool down
        temp = max(temp * decay, 1e-12)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine combines a parametric sweep, several random
    restarts with smooth‑max Adam descent, an Adam‑based sub‑gradient
    optimisation on the exact max‑convolution, and a final simulated‑annealing
    polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # safety margin
    n = 4096                                    # problem size
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0b️⃣  optional global starter (if supplied)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  fast parametric sweep over several families
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based index

    # --- geometric + power family (already used before) ---
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=28)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 250:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 250:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.3, 0.6, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} "
                          f"p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # --- Gaussian‑shaped families ---------------------------------------
    for sigma in np.linspace(0.05, 0.30, 12):
        if time.time() >= deadline - 250:
            break
        mu = (n - 1) / 2.0
        gaussian = np.exp(-0.5 * ((idx - 1 - mu) / (sigma * n)) ** 2)
        gaussian /= gaussian.sum()
        for lam in (0.0, 0.5, 1.0):
            cand = lam * uniform + (1.0 - lam) * gaussian
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] gauss σ={sigma:.3f} λ={lam:.2f} "
                      f"score={sc:.6f}")

    # --- triangular (linear ramp) ----------------------------------------
    tr = np.linspace(0.0, 2.0, n)
    tr = np.minimum(tr, 2.0 - tr)         # creates a tent shape
    tr /= tr.sum()
    for lam in (0.0, 0.5, 1.0):
        cand = lam * uniform + (1.0 - lam) * tr
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] tent λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  random restarts + smooth‑max Adam descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # keep roughly half of the remaining budget for this stage
    time_for_smooth = min(remaining * 0.45, 450.0)
    n_restarts = 6
    per_restart = time_for_smooth / max(n_restarts, 1)

    for i in range(n_restarts):
        if time.time() >= deadline - 200:
            break
        rng = np.random.default_rng(seed + i + 1234)

        # two different ways to initialise a restart
        if rng.random() < 0.6:
            # perturb current best
            a0 = best_seq.copy()
            noise = rng.normal(scale=0.015, size=n) * a0
            a0 = np.maximum(a0 + noise, 0.0)
        else:
            # fresh Dirichlet draw with random concentration
            alpha = rng.uniform(0.05, 2.0)
            a0 = rng.gamma(shape=alpha, scale=1.0, size=n)
        a0 = _project_simplex(a0)

        sub_deadline = time.time() + per_restart
        a_smooth, sc_smooth = _smooth_descent(
            a0, n, sub_deadline, start_time, best_score
        )
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {i+1}/{n_restarts} "
                  f"improved to {sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam on the exact sub‑gradient (max shift)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        sub_deadline = deadline - 150
        a_sub, sc_sub = _subgradient_adam(
            best_seq, n, sub_deadline, start_time, best_score
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient Adam score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _simulated_annealing(
            best_seq, deadline, start_time, best_score
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final SA score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
