"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm mixes several ideas that improve on the previous state‑of‑the‑art:

1️⃣ A richer parametric sweep (geometric‑power and beta‑like families) to get a
   strong starting vector.

2️⃣ An Adam‑style projected descent on a *smooth* approximation of the
   “max – min” of the autocorrelation.  By pulling the maximum down and the
   minimum up we flatten the whole convolution, which is more effective than
   minimising only the max.

3️⃣ A sub‑gradient descent on the exact max‑convolution (as before) to polish
   the result.

4️⃣ A final hill‑climb that moves larger chunks of mass and uses a simple
   simulated‑annealing acceptance rule.

All steps work on the probability simplex (non‑negative, sum = 1) and respect a
1000 s time budget.  The search dimension is 8192 (≈ 8 k) which gives a finer
approximation of the continuous optimum while still being fast enough with FFT‑based
convolutions.

The function returns the best sequence found as a plain Python ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    w ≥ 0, sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # fallback (should not happen)
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a, a)) for a vector with sum≈1.
    Returns ``inf`` if sum is too small.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Phase 1 – parametric families
# ----------------------------------------------------------------------


def _parametric_sweep(n: int, L_fft: int, deadline: float, start_time: float,
                      best_seq: np.ndarray, best_score: float):
    """
    Scan a handful of families (geometric‑power and beta‑like) and keep the best.
    """
    uniform = np.full(n, 1.0 / n)

    idx = np.arange(n, dtype=np.float64)           # 0 … n‑1
    ii = idx + 1.0                                 # 1 … n
    rev = n - idx                                   # n … 1

    # geometric‑power family:   a_i ∝ r^i * i^p
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=18)
    p_grid = np.linspace(-3.0, 3.0, 13)
    lam_grid = [0.0, 0.25, 0.5, 0.75, 1.0]

    for r in r_grid:
        if time.time() >= deadline - 180:
            break
        geo = np.power(r, idx)
        for p in p_grid:
            shape = geo * np.power(ii, p)
            s = shape.sum()
            if s == 0:
                continue
            shape /= s
            for lam in lam_grid:
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep geo p={p:.2f} r={r:.4f} λ={lam:.2f} → {sc:.6f}")

    # beta‑like family: a_i ∝ (i+1)^α * (n‑i)^β * r^i
    alpha_grid = np.linspace(-3.0, 3.0, 13)
    beta_grid = np.linspace(-3.0, 3.0, 13)
    r_grid2 = np.logspace(np.log10(0.90), np.log10(0.9999), num=10)

    for r in r_grid2:
        if time.time() >= deadline - 120:
            break
        geo = np.power(r, idx)
        for alpha in alpha_grid:
            term1 = np.power(ii, alpha)
            for beta in beta_grid:
                term2 = np.power(rev, beta)
                shape = geo * term1 * term2
                s = shape.sum()
                if s <= 0:
                    continue
                shape /= s
                for lam in lam_grid:
                    cand = lam * uniform + (1.0 - lam) * shape
                    cand /= cand.sum()
                    sc = _eval_fast(cand, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                        print(f"[{time.time() - start_time:.2f}s] sweep beta α={alpha:.2f} β={beta:.2f} r={r:.4f} λ={lam:.2f} → {sc:.6f}")

    return best_seq, best_score

# Phase 2 – smooth “max‑minus‑min” Adam descent
# ----------------------------------------------------------------------


def _smooth_maxmin_descent(a0: np.ndarray, n: int, deadline: float,
                           start_time: float, best_score_initial: float):
    """
    Projected Adam descent on a smooth surrogate of (max(conv)-min(conv)).
    By pushing the minimum up while pulling the maximum down we flatten the
    entire autocorrelation, which usually yields a lower final max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # temperature (soft‑max sharpness) schedule
    beta_temp = 1.0          # start very smooth
    beta_mul = 1.03
    beta_max = 400.0

    while time.time() < deadline:
        t += 1

        # --- convolution -------------------------------------------------
        conv = _conv_self_fft(a, L_conv)              # length 2n‑1
        max_c = conv.max()
        min_c = conv.min()

        # soft‑max for the max
        exp_max = np.exp(beta_temp * (conv - max_c))
        w_max = exp_max / exp_max.sum()

        # soft‑min (via negative exponent)
        exp_min = np.exp(beta_temp * (min_c - conv))
        w_min = exp_min / exp_min.sum()

        w_diff = w_max - w_min                         # weights for gradient

        # gradient of (soft‑max − soft‑min) w.r.t. a
        a_rev = a[::-1]
        W = np.fft.rfft(w_diff, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        grad = 2.0 * corr[n - 1:n - 1 + n]

        # exact objective for monitoring
        cur_score = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta_temp:.1f} score={cur_score:.6f}")

        # --- Adam update (projected onto simplex) -----------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.15 / (np.sqrt(v_hat) + eps)          # a bit larger steps
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule (sharpen)
        beta_temp = min(beta_temp * beta_mul, beta_max)

        # occasional jitter to avoid plateaus
        if t % 3500 == 0:
            jitter = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """
    Classic projected sub‑gradient descent directly on the true max‑convolution.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.10
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # Sub‑gradient for the maximising shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / np.sqrt(it)
        a_cand = a - step * g
        a_cand = _project_simplex(a_cand)

        cur_score = _eval_fast(a_cand, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cand.copy()
            a = a_cand
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cur_score:.6f}")
        else:
            # small acceptance prob to escape plateaus
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_cand

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _final_hill_climb(a0: np.ndarray, deadline: float, start_time: float):
    """
    Tiny stochastic hill‑climb that moves a variable fraction of mass.
    The move size slowly shrinks with time; occasionally a worsening move
    is accepted with a temperature that decays.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()

    # annealing schedule
    temp0 = 0.15
    decay = 0.9995

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a fraction (up to a few percent) of src mass to dst
        eps_frac = rng.random() * 0.04   # ≤ 4 %
        eps = a[src] * eps_frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)

        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            # simulated‑annealing acceptance
            temp = max(temp0 * decay ** (time.time() - start_time), 1e-6)
            prob = np.exp(-(new_score - best_score) / temp)
            if rng.random() < prob:
                a = a_prop

    return best_a, best_score
