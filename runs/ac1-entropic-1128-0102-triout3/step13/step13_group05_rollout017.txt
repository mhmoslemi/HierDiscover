"""
Propose a step‑function (non‑negative heights) that dramatically lowers the
evaluation score.

Idea
----
The objective for a sequence a (with sum(a)=1) is

    score = 2·n·max_k (a * a)[k]          (1)

where “*” is the linear convolution.  For a monotone polynomial shape

    a_i ∝ (i+1)^p ,   i = 0,…,n‑1                (2)

the convolution is maximised at the centre k = n‑1 and

    max_k (a*a)[k]  =  B(p+1,p+1)·(p+1)² / n      (3)

with B the Beta‑function.  Formula (3) shows that the score decreases
approximately like 2·B(p+1,p+1)·(p+1)², i.e. it drops quickly when p grows.
Even a modest p=5 yields a score ≈ 0.03, far below the target 1.503.

Implementation

* Scan a small grid of exponents `p` (1‑12).  For each `p` build the

  evaluate the score with a fast FFT‑based convolution.
* Keep the best exponent, then perform a few iterations of projected
  sub‑gradient descent on the exact max‑convolution (3) to polish the
  solution.  The sub‑gradient of the maximal convolution entry is

* Finally normalise the vector, clip to the allowed range and return it

# FFT utilities (same as in the starter code)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/ (sum(seq)²).  Assumes seq is non‑negative."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Build a monotone power‑law shape a_i ∝ (i+1)^p  (log‑space for stability)
# ----------------------------------------------------------------------
def _make_power_shape(N: int, p: float) -> np.ndarray:
    """Return a monotone increasing shape (i+1)^p, normalised to sum 1."""
    i = np.arange(1, N + 1, dtype=np.float64)          # 1 … N
    # work in log‑space to avoid overflow for large p
    log_w = p * np.log(i)
    log_w -= log_w.max()                               # shift for stability
    w = np.exp(log_w)
    total = w.sum()
    if total > 0.0:
        w /= total
    else:
        w = np.full(N, 1.0 / N)
    return w

    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    # Adam‑style parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.05
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        k_star = int(np.argmax(conv))

        # sub‑gradient of conv[k_star] w.r.t. a
        grad = np.zeros_like(a)
        idx = k_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a monotone power‑law step‑function that gives a very low
    evaluation score.  The routine scans a few exponents, picks the best
    and (time permitting) performs a short sub‑gradient polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # ------------------------------------------------------------------
    # Decide on the problem size (larger N → finer discretisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # Scan a small set of exponents p and keep the best one
    # ------------------------------------------------------------------
    exponent_grid = [1, 2, 3, 4, 5, 6, 8, 10, 12]   # a few candidates
    best_seq = None
    best_score = np.inf

    for p in exponent_grid:
        a = _make_power_shape(N, float(p))
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] p={p:<2} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    print(f"[{time.time() - start_time:.2f}s] Initial best exponent score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional short sub‑gradient polishing (uses the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:               # need a few seconds for a useful polish
        sub_deadline = time.time() + remaining
        polished_seq, polished_score = _subgradient_descent(best_seq,
                                                            sub_deadline,
                                                            start_time)
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Polished best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    # Return as a plain list of floats (the harness expects this)
    return [float(v) for v in best_seq]
