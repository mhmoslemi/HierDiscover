def _softmax_weights(conv, alpha):
    # conv: np.ndarray length 2n-1.
    # Use shift for numerical stability
    max_c = conv.max()
    exp = np.exp(alpha * (conv - max_c)) # scaled
    sum_exp = exp.sum()
    if sum_exp == 0:
        return np.full_like(conv, 1.0/conv.size)
    return exp / sum_exp

def _grad_softmax(a, conv, alpha):
    n = a.shape[0]
    w = _softmax_weights(conv, alpha)
    # compute correlation a ⋆ w via FFT: correlation.
    L2 = _next_pow2(n + (2*n-1) - 1)  # = next pow2 of 3n-2
    A = np.fft.rfft(a, n=L2)
    W = np.fft.rfft(w, n=L2)
    # cross-correlation via conj: np.fft.irfft(A * np.conj(W), n=L2) obtains length L2.
    corr = np.fft.irfft(A * np.conj(W), n=L2)
    # correlation indexes 0..n-1 correspond to sum_{t} a_t w_{t + i}
    grad = 2.0 * corr[:n]
    return grad

    n = a.shape[0]

    m = np.zeros_like(a); v = np.zeros_like(a)
    it = 0

    phases = [(50000, 10), (50000, 40), (50000, 160), (50000, 640), (50000, 2560), (50000, 10240)]

    while time.time() < deadline:
        it+=1; phase_iter+=1
        # compute A
        A = np.fft.rfft(a, n=L_corr)

"""
Propose a step‑function (non‑negative list) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

by a smooth‑gradient Adam optimiser on the simplex.
A soft‑max approximation of the max‑convolution gives a dense,
differentiable surrogate; the Adam update follows the
corresponding sub‑gradient and is projected onto the probability
simplex after each step.  A final stochastic hill‑climb refines the
solution.  The whole search respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w ≥ 0, Σ w = 1 }.
    Returns the projected vector w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a⋆a (length 2·len(a)‑1) via FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max approximation of the max‑convolution and its gradient
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max of the convolution values with temperature 1/α."""
    # stability shift
    shift = conv.max()
    exp = np.exp(alpha * (conv - shift))
    sumexp = exp.sum()
    if sumexp == 0:
        # fallback to uniform (should never happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp / sumexp


def _grad_from_softmax(a: np.ndarray,
                       conv: np.ndarray,
                       alpha: float,
                       L_corr: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution:
        ∇_a  ( 1/α log Σ exp(α·(a⋆a)) )
    which equals 2·(a ⋆ w) where w are the soft‑max weights.
    """
    w = _softmax_weights(conv, alpha)

    # correlation a⋆w (linear, not circular) via FFT
    A = np.fft.rfft(a, n=L_corr)
    W = np.fft.rfft(w, n=L_corr)
    corr = np.fft.irfft(A * np.conj(W), n=L_corr)[: a.shape[0]]
    return 2.0 * corr


# ----------------------------------------------------------------------
# Adam optimiser that works on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize_smooth(a0: np.ndarray,
                          start_time: float,
                          deadline: float,
                          best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using the soft‑max
    surrogate of the max‑convolution.
    Returns the best sequence seen and its true objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    # length for linear correlation a(∗)w : n + (2n‑1) - 1 = 3n‑2
    L_corr = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    # schedule of temperature (α) – start smooth, end sharp
    phases = [
        (30_000, 10.0),
        (30_000, 40.0),
        (30_000, 160.0),
        (30_000, 640.0),
        (30_000, 2560.0),
        (30_000, 10240.0),
    ]
    phase_idx = 0
    phase_counter = 0
    cur_alpha = phases[0][1]
    phase_limit = phases[0][0]

    while time.time() < deadline:
        it += 1
        phase_counter += 1

        # FFT of current a
        A = np.fft.rfft(a, n=L_corr)

        # linear self‑convolution (needed for the soft‑max)
        conv_full = np.fft.irfft(A * A, n=L_corr)[: 2 * n - 1]

        # gradient of the smoothed max‑convolution
        grad = _grad_from_softmax(a, conv_full, cur_alpha, L_corr)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project back onto the simplex
        a = _project_simplex(a)

        # true objective (no smoothing)
        cur_max = conv_full.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} α={cur_alpha:.0f} score={cur_score:.6f}")

        # phase switch
        if phase_counter >= phase_limit:
            phase_idx += 1
            if phase_idx < len(phases):
                cur_alpha = phases[phase_idx][1]
                phase_limit = phases[phase_idx][0]
                phase_counter = 0
                print(f"[{time.time() - start_time:.2f}s] Switching α to {cur_alpha}")

        # occasional jitter if stuck too long
        if it - last_improve > 20_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Tiny random mass transfers between two coordinates.
    Only improvements (according to the true objective) are kept.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 1 % of the source mass
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The method consists of:
        * a simplex‑projected Adam optimiser on a soft‑max
          surrogate of the max‑convolution;
        * a short stochastic hill‑climb.
    The routine returns a plain Python list ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a little margin for the final hill‑climb
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # problem size (feel free to change – 4096 works well)
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # initialise – try to reuse a global starter if available
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size < n:
                a0 = np.concatenate([starter, np.zeros(n - starter.size)])
            else:
                a0 = starter[:n]
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            raise ValueError
    except Exception:
        a0 = np.full(n, 1.0 / n)

    # baseline true score
    conv0 = _conv_self(a0)
    best_score = float(2.0 * n * conv0.max())
    best_seq = a0.copy()
    print(f"[{time.time() - start_time:.2f}s] Initial baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Adam optimiser on the smoothed objective
    # --------------------------------------------------------------
    hill_time = 150.0                # seconds reserved for hill‑climb
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser (smooth surrogate)")
        a_opt, sc_opt = _adam_optimize_smooth(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement: score={sc_opt:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Final hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb improvement: score={sc_hc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
