"""
Propose a low‑objective step‑function.
-------------------------------------

The algorithm consists of four stages:

1. **Warm‑start generation** – a rich set of analytic shapes
   (tent, cosine, circle, Gaussian, power‑tents, exponentials) and a
   handful of random Dirichlet draws.  The candidate with the smallest
   exact objective 2·n·max(conv) (sum is forced to 1) is kept.

2. **Projected Adam** – an adaptive first‑order method on the probability
   simplex using the exact sub‑gradient of the max‑convolution entry.
   Occasionally a tiny Gaussian jitter is added to escape plateaus.

3. **Simulated Annealing** – a Metropolis walk that moves a random
   packet of mass from a source index (chosen proportionally to its
   current mass) to a destination index.  Moves that increase the
   objective are accepted with probability exp(‑Δ/T) where the temperature
   T is linearly cooled to almost zero.  This allows the search to leave
   local minima found by Adam.

4. **Final polishing** – a short greedy hill‑climb that accepts only
   strictly improving moves.

All heavy work uses NumPy’s real FFT for fast self‑convolution.
The routine respects the 1000 s budget – when the time limit is reached
the current best vector is returned.  The returned list is normalised to
sum = 1 and contains only non‑negative floats, ready to be passed to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns the first (2·n‑1) samples of the linear convolution.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    (The optimiser always works with sum = 1.)
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic warm‑starts
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Produce a diverse collection of normalised candidate vectors."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]
    cand = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    cand.append(tent)

    # 2) half‑cosine
    cosine = np.sin(np.pi * x)
    cand.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    cand.append(circ)

    # 4) Gaussian
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    cand.append(gauss)

    # 5) power‑tents
    for p in np.linspace(0.2, 2.0, 9):
        cand.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponentials (forward and reversed)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        cand.append(exp_fwd)
        cand.append(exp_rev)

    # 7) constant (uniform)
    cand.append(np.ones_like(x))

    # Normalise
    res = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            res.append(c / s)
    return res

def _random_dirichlet_candidates(n: int, rng: np.random.Generator,
                                 alphas: list[float], per_alpha: int = 5) -> list[np.ndarray]:
    """Generate a few random points on the simplex using Dirichlet draws."""
    out = []
    for a in alphas:
        for _ in range(per_alpha):
            vec = rng.gamma(a, 1.0, size=n)
            s = vec.sum()
            if s > 0.0:
                out.append(vec / s)
    return out


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (ignoring the outer 2·n factor).
    """
    n = a.shape[0]
    k = int(conv.argmax())               # location of the maximal entry
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]          # g_i = 2·a_{k‑i}
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the simplex with occasional Gaussian jitter."""
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1

        # Full self‑convolution
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # Sub‑gradient of max entry
        grad = _max_subgradient(a, conv)

        # Scale for the true objective 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # Project back onto the simplex
        a = _project_simplex(a)

        # Occasionally inject tiny noise to help escape flat regions
        if it % 25000 == 0:
            noise = rng.normal(scale=0.005, size=n)
            a = _project_simplex(np.maximum(a + noise, 0.0))

        cur_score = 2.0 * n * conv.max()      # same as _objective(a) but cheaper
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # Learning‑rate decay
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (Metropolis walk with temperature decay)
# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis walk that moves a random packet of mass.
    Moves that worsen the objective are accepted with probability exp(-Δ/T).
    Temperature T decays linearly from T0 to ~0.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_time = deadline - start_time
    # Initial temperature set as a small fraction of the current score
    T0 = max(1e-4, cur_score * 0.01)

    it = 0
    while time.time() < deadline:
        it += 1

        # Linear temperature schedule
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T0 * (1.0 - frac)       # cool to zero

        # Propose a move
        src = rng.choice(n, p=a)          # source ∝ current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        # Move a random fraction (up to 40 %) of the source mass
        eps = a[src] * rng.random() * 0.40
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        elif T > 0:
            prob = math.exp(-delta / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Short greedy hill‑climb for final polishing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """Greedy uphill‑walk that only accepts strictly improving moves."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={new_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns a plain list of floats (sum is forced to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – power of two for fast FFTs.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Warm‑starts: analytic families + random Dirichlet draws.
    # --------------------------------------------------------------
    warm = _make_warm_starts(n)
    rng = np.random.default_rng(seed + 1)
    warm += _random_dirichlet_candidates(n, rng,
                                          alphas=[0.01, 0.05, 0.2, 1.0],
                                          per_alpha=4)

    # If a global seed vector is supplied, resample it to length n.
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            if np.all(h1_resampled >= 0):
                s = h1_resampled.sum()
                if s > 0:
                    warm.append(h1_resampled / s)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for vec in warm:
        sc = _objective(vec)
        if sc < best_score:
            best_score = sc
            best_a = vec.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Projected Adam (≈ 50 % of remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:
        adam_deadline = time.time() + remaining * 0.5
        a_opt, score_opt = _adam_optimize(a0, adam_deadline, start_time, lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated annealing (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _anneal_optimize(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] SA best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 4) Final greedy hill‑climb polish (a few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = time.time() + min(5.0, deadline - time.time())
        a_opt, score_opt = _hill_climb(a0, polish_deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 5) Normalise and return
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
