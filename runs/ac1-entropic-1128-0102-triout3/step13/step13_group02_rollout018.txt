"""
step_search.py – An improved optimisation pipeline for the step‑function challenge.

The programme builds on the previous multi‑phase approach but introduces
a *soft‑max surrogate* (log‑sum‑exp) of the max‑convolution together with
`scipy.optimize.fmin_l_bfgs_b`.  The variables are represented in the
soft‑max (log‑probability) space, guaranteeing non‑negativity and unit
sum without an explicit projection step.  A temperature schedule
(beta = 0.5 → 500) slowly turns the smooth surrogate into the true
max‑convolution, allowing the optimiser to first see a smooth landscape
and then focus on the exact objective.  After the L‑BFGS phase a cheap
hill‑climb that swaps tiny masses between coordinates is applied.

The whole routine respects a 1000 s wall‑clock budget, records the best
sequence found, and finally returns it as a plain ``list[float]``.
"""

import time
import numpy as np

# optional – may be missing in a minimal environment
try:
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Utility primitives
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)
    with a already non‑negative and (approximately) normalised.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    shift = np.max(z)
    e = np.exp(z - shift)
    total = e.sum()
    if total == 0.0:
        # fall back to uniform – should never happen
        return np.full_like(z, 1.0 / z.shape[0])
    return e / total


# ----------------------------------------------------------------------
# Surrogate objective (smooth max) used by L‑BFGS
# ----------------------------------------------------------------------


class _SurrogateMax:
    """
    Callable implementing the smooth‑max surrogate of max(conv) together
    with its gradient w.r.t. the soft‑max parameters `z`.
    """

    def __init__(self, n: int, beta: float,
                 L_conv: int, L_corr: int):
        self.n = n
        self.beta = beta
        self.L_conv = L_conv          # for a*a
        self.L_corr = L_corr          # for w * a_rev
        # pre‑allocated reusable buffers (optional, for speed)
        self._buf_a_rev = None

    def __call__(self, z: np.ndarray):
        """
        Returns (value, gradient) where value is the smooth surrogate
        and gradient is ∂value/∂z (same shape as `z`).
        """
        a = _softmax(z)                     # a ≥0, Σa = 1
        # --- self‑convolution -------------------------------------------------
        conv = _conv_self_fft(a, self.L_conv)     # length 2n‑1
        max_c = conv.max()
        # smooth‑max (log‑sum‑exp) and its normalised weights
        shifted = conv - max_c               # stabilise exponentials
        exp_vals = np.exp(self.beta * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max weights, length 2n‑1
        # surrogate value = max_c + (1/β)·log Σ exp(β·(conv−max_c))
        surrogate = max_c + np.log(exp_vals.sum()) / self.beta

        # --- gradient w.r.t. a ------------------------------------------------
        # we need g_i = 2·∑_k w_k·a_{k‑i}
        a_rev = a[::-1]
        # convolution w * a_rev (length 3n‑2)
        prod = np.fft.rfft(w, n=self.L_corr) * np.fft.rfft(a_rev, n=self.L_corr)
        conv_wa = np.fft.irfft(prod, n=self.L_corr)[: 3 * self.n - 2]
        # pick the slice (n‑1 … 0) and reverse to obtain g_i
        grad_a = 2.0 * conv_wa[self.n - 1 :: -1]  # length n

        # --- chain rule through the soft‑max transformation -------------------
        # ∂a_i/∂z_j = a_i·(δ_{ij} − a_j)
        dot = np.dot(grad_a, a)              # scalar
        grad_z = a * (grad_a - dot)          # shape (n,)

        return surrogate, grad_z


# ----------------------------------------------------------------------
# Final microscopic hill‑climb (pure objective)
# ----------------------------------------------------------------------


def _final_hill_climb(a: np.ndarray, deadline: float,
                      start_time: float, L_fft: int):
    """
    Very local random mass transfers accepted only if they improve the
    exact objective.  Returns the best vector found.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(best_a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if best_a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 0.1 % of the source mass
        eps = best_a[src] * rng.random() * 1e-3
        cand = best_a.copy()
        cand[src] -= eps
        cand[dst] += eps
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score

    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if the notebook defines `height_sequence_1`)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (geometric‑power families)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based index
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 150:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 150:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS optimisation (annealed β)
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and time.time() < deadline - 250:
        # initialise the log‑probability vector `z`
        eps = 1e-12
        z0 = np.log(np.maximum(best_seq, eps))

        # temperature schedule – starts smooth, finishes sharp
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]

        for beta in beta_schedule:
            if time.time() >= deadline - 250:
                break
            obj = _SurrogateMax(n=n, beta=beta,
                                L_conv=_next_pow2(2 * n - 1),
                                L_corr=_next_pow2(3 * n - 2))

            maxfun = 250  # very cheap – each call is cheap thanks to FFT
            try:
                z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                    func=obj,
                    x0=z0,
                    maxfun=maxfun,
                    disp=False,
                    factr=1e7,    # tolerance (loose – we rely on the schedule)
                )
            except Exception as e:  # pragma: no cover
                print(f"  L‑BFGS failed at β={beta}: {e}")
                break

            # obtain the candidate vector and its true score
            cand = _softmax(z_opt)
            real_score = _eval_fast(cand, L_fft)
            if real_score < best_score:
                best_score = real_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} → improved score={real_score:.6f}")

            # use the solution as the warm‑start for the next temperature
            z0 = z_opt

    # ------------------------------------------------------------------
    # 3️⃣  Random restarts (smooth‑max, short L‑BFGS runs)
    # ------------------------------------------------------------------
    restarts = 2
    for r in range(restarts):
        if time.time() >= deadline - 200:
            break
        rng = np.random.default_rng(seed + r + 12345)
        noisy = best_seq + rng.normal(scale=0.01, size=n)   # tiny jitter
        noisy = np.maximum(noisy, 0.0)
        noisy /= noisy.sum()
        z0 = np.log(np.maximum(noisy, 1e-12))
        beta = 5.0
        obj = _SurrogateMax(n=n, beta=beta,
                            L_conv=_next_pow2(2 * n - 1),
                            L_corr=_next_pow2(3 * n - 2))
        # very short optimisation – just enough to escape shallow traps
        try:
            z_opt, f_val, info = scipy.optimize.fmin_l_bfgs_b(
                func=obj,
                x0=z0,
                maxfun=120,
                disp=False,
                factr=1e7,
            )
        except Exception:  # pragma: no cover
            continue
        cand = _softmax(z_opt)
        real_score = _eval_fast(cand, L_fft)
        if real_score < best_score:
            best_score = real_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {r+1}/{restarts} improved score={real_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb polishing (exact objective)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Post‑process: ensure non‑negative, normalise to unit sum and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
