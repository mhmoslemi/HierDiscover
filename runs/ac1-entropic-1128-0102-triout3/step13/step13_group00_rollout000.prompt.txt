def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    If several shifts attain the maximum the gradients are averaged.

def _hill_climb_directed(start_a: np.ndarray,
                         start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained stochastic hill‑climb.
    At each iteration we move a tiny fraction of mass from the index with
    the largest positive sub‑gradient to the index with the most
    negative sub‑gradient.  Only improving moves are kept.
    """

    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()

    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # choose source (mass to give away) and destination (mass to receive)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)             # most negative gradient

        if src == dst:
            # fallback to a random move if gradients are degenerate
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        # amount of mass to transfer (up to 0.5 % of source)
        frac = rng.random() * 0.005 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score
