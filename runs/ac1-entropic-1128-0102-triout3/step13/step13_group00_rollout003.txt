        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)

        a = np.maximum(a, 0.0)
        # project onto simplex
        a = project_simplex(a)

        # increase temperature gradually
        if t % 2000 == 0:
            tau = min(tau * tau_factor, 1e8)

"""
Propose a step‑function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is scale‑invariant, so we enforce sum(a)=1 and minimise
2·n·max(conv).  The optimiser works in three phases inside the time
budget (default 1000 s):

1️⃣  **Initialisation** – try the supplied global starter, a uniform
    vector and a handful of random Dirichlet draws; keep the best.

2️⃣  **Smooth‑max Adam descent** – minimise a differentiable
    approximation of the max, Sτ(a)= (1/τ)·log Σ exp(τ·conv), using an
    Adam optimiser and a temperature τ that is slowly increased.  Each
    iterate is projected onto the probability simplex.

3️⃣  **Simulated‑annealing fine‑tuning** – repeatedly move a small
    amount of mass from a source index to a destination index; moves
    that improve the true objective are always accepted, otherwise they
    are accepted with probability exp(−Δ/T) where the temperature T
    cools over time.

Both phases use FFT based full convolutions (O(n log n)) and all
vectors are kept on the probability simplex, guaranteeing that the
returned list is a valid candidate for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)−1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution via FFT (pre‑allocated length L)."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – initial candidates
# ----------------------------------------------------------------------
def _initial_candidate(n: int, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Return the best of uniform, global starter, and a few Dirichlet draws."""
    # uniform
    best = np.full(n, 1.0 / n)
    best_score = _objective_score(best)

    # global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best, best_score = starter, sc
    except Exception:
        pass

    # random Dirichlet draws (concentration ∈ [0.1,5])
    rng = np.random.default_rng()
    trials = 25
    for _ in range(trials):
        if time.time() >= deadline:
            break
        c = rng.uniform(0.1, 5.0)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best, best_score = a, sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet(c={c:.3f}) score={sc:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max Adam descent
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    # FFT lengths
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.12                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    tau = 5.0                     # start fairly smooth
    tau_factor = 1.15
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution and objective
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max weights over shifts
        max_c = cur_max
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient of the smooth‑max (2·(w * a_rev) restricted to length n)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev, L_grad)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]           # slice of length n

        # Adam update (minimise → move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # temperature annealing
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves tiny masses between bins."""
    a = a_start.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    # temperature schedule (starts relatively warm)
    T0 = 0.02
    iteration = 0
    rng = np.random.default_rng()

    # keep a separate best (global) record
    best_local = a.copy()
    best_local_score = cur_score

    while time.time() < deadline:
        iteration += 1
        # linear cooling
        T = T0 * (0.99995 ** iteration)

        # pick source (must have some mass) and destination
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[rng.integers(src_candidates.size)]
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer fraction – larger early, smaller later
        if iteration < 20000:
            max_frac = 0.07
        elif iteration < 100000:
            max_frac = 0.02
        else:
            max_frac = 0.005
        eps = a[src] * (rng.random() * max_frac + 1e-8)
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        new_score = _objective_score(a_prop)
        delta = new_score - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            # probabilistic acceptance
            if T > 0.0 and np.exp(-delta / T) > rng.random():
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            # update global best
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best={best_score:.6f}")
            # update local best (helps to keep a good reference)
            if new_score < best_local_score:
                best_local_score, best_local = new_score, a.copy()

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs three phases (initialisation, Adam descent,
    simulated‑annealing) and respects the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.30          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep a moderately large vector to have enough
    # degrees of freedom while staying fast.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Phase 1 – initialise a good starting point
    # ------------------------------------------------------------------
    init_deadline = deadline - 400.0   # reserve ~400 s for later phases
    best_seq, best_score = _initial_candidate(n, init_deadline, start_time)
    print(f"[{time.time() - start_time:.2f}s] Initial best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam descent on smooth‑max
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 200.0   # keep ~200 s for final SA
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max Adam descent")
        best_seq, best_score = _smooth_max_adam(
            best_seq, start_time, smooth_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
