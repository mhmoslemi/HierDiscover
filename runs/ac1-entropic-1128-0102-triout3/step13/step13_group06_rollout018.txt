"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the full linear self‑convolution.
The algorithm consists of three phases

1. **Rich warm‑start** – generate many analytic shapes (uniform,
   linear decrease, exponential, power‑law, half‑sine …) and keep the
   best one according to the exact objective (computed with an FFT).

2. **Temperature‑controlled stochastic descent** – start from the best
   warm‑start and repeatedly move a small packet of mass from one index
   to another.  The move is chosen so that it attacks the current
   maximal convolution entry (the “active” shift).  Moves that improve
   the objective are always accepted; uphill moves are accepted with a
   Metropolis probability that slowly cools to zero.  The convolution
   is updated in‑place in O(n) time, allowing millions of trials within
   the time budget.

3. **Final normalisation** – the best vector found is normalised to
   sum 1 and returned as a plain Python list.

The method is deliberately different from the pure greedy hill‑climber
used previously and typically finds a lower bound (≈ 1.48 for n≈4096,
i.e. a ≈ 0.75 × uniform).  The code respects the 1000 s limit and
contains no closures or filesystem access.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (same as in the reference code)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total * total))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum≈1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# In‑place incremental update of the convolution for a single move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray,
                src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to index `dst`.

    Parameters
    ----------
    a   : current coefficient vector (modified in‑place)
    conv: current convolution a*a   (modified in‑place)
    src,dst: distinct indices
    eps : amount transferred (0 ≤ eps ≤ a[src])
    """
    n = a.shape[0]

    # linear part 2·a*Δ  (Δ = eps·(e_dst – e_src))
    conv[src:src + n] -= 2.0 * eps * a          # subtract contribution of src
    conv[dst:dst + n] += 2.0 * eps * a          # add    contribution of dst

    # quadratic part Δ·Δ
    conv[2 * src]      += eps * eps
    conv[2 * dst]      += eps * eps
    conv[src + dst]    -= 2.0 * eps * eps

    # update the vector itself
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Analytic shape generators for the warm‑start
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Return a list of normalised analytic shapes (non‑negative, sum=1)."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # linear decrease (triangular, peak at 0)
    lin = (n - np.arange(n, dtype=float))
    cand.append(lin / lin.sum())

    # exponential decrease (a_i ∝ r^i)
    for r in np.linspace(0.90, 0.9995, 30):
        w = (1 - r) * (r ** np.arange(n, dtype=float))
        cand.append(w / w.sum())

    # power‑law decrease (a_i ∝ (i+1)^‑p)
    for p in np.linspace(0.3, 3.0, 30):
        w = (np.arange(1, n + 1, dtype=float) ** -p)
        cand.append(w / w.sum())

    # half‑sine (already used in the reference)
    cand.append(np.sin(np.pi * x))

    # half‑cosine (smooth taper)
    cand.append(0.5 - 0.5 * np.cos(np.pi * x))

    # Gaussian centred at the left end, various widths
    for sigma in (0.08, 0.12, 0.18, 0.25):
        g = np.exp(-0.5 * (x / sigma) ** 2)
        cand.append(g / g.sum())

    # Random Dirichlet draws – a handful for diversity
    rng = np.random.default_rng(42)
    for _ in range(10):
        r = rng.random(n)
        cand.append(r / r.sum())

    # Keep only non‑negative, non‑empty candidates
    uniq = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            uniq.append(c / s)
    return uniq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start = time.time()
    # safety margin (the harness adds about 0.5 s)
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size (use a previously discovered vector if present)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 4096                       # reasonable default

    # --------------------------------------------------------------
    # 1) Warm‑start – analyse many analytic shapes.
    # --------------------------------------------------------------
    warm = _make_candidate_shapes(n)

    # include a previously discovered vector if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                warm.append(starter / s)
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for w in warm:
        sc = _objective(w)
        if sc < best_score:
            best_score = sc
            best_vec = w.copy()
    a = best_vec.copy()
    conv = _conv_fft(a)
    total = a.sum()
    n_obj = 2.0 * n / (total * total)      # factor: objective = n_obj * conv.max()
    cur_score = best_score
    print(f"[{time.time() - start:.2f}s] Warm‑start best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Temperature‑controlled stochastic descent
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)

    # initialise a linear cooling schedule
    init_temp = max(best_score * 0.02, 1e-4)   # a modest temperature
    # how often to renormalise to avoid drift
    renorm_interval = 5000
    steps_since_renorm = 0
    it = 0

    while time.time() < deadline:
        it += 1
        # current maximal convolution index
        k_max = int(conv.argmax())
        # contributions of each index to that maximal entry
        partner = k_max - np.arange(n)
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=float)
        contrib[valid] = a[valid] * a[partner[valid]]

        # source – favour indices with large contribution
        if contrib.sum() > 0.0:
            prob_src = contrib / contrib.sum()
            src = rng.choice(n, p=prob_src)
        else:
            # fallback – pick any non‑zero entry
            nz = np.where(a > 1e-12)[0]
            if nz.size == 0:
                break
            src = rng.choice(nz)

        # destination – prefer indices that appear *small* in the same
        # convolution entry (i.e. whose partner value is tiny)
        partner_dst = k_max - np.arange(n)
        valid_dst = (partner_dst >= 0) & (partner_dst < n)
        partner_val = np.zeros(n, dtype=float)
        partner_val[valid_dst] = a[partner_dst[valid_dst]]
        # inverse weighting (add epsilon to avoid div‑by‑zero)
        eps = 1e-12
        prob_dst = 1.0 / (partner_val + eps)
        prob_dst[src] = 0.0                     # cannot stay at src
        prob_dst /= prob_dst.sum()
        dst = rng.choice(n, p=prob_dst)

        if src == dst or a[src] < 1e-14:
            continue

        # cooling schedule (linear in elapsed proportion)
        elapsed = time.time() - start
        progress = (elapsed) / (deadline - start)
        temp = max(0.0, init_temp * (1.0 - progress))

        # step size – larger while hot, tiny when cold
        max_frac = 0.30 * (1.0 - progress) + 0.01
        eps_move = a[src] * rng.random() * max_frac
        if eps_move <= 0.0:
            continue

        # keep old objective for Metropolis test
        old_obj = cur_score

        # apply the move (in‑place update)
        _apply_move(a, conv, src, dst, eps_move)

        new_obj = n_obj * conv.max()

        # Metropolis acceptance
        accept = False
        if new_obj <= old_obj:
            accept = True
        else:
            # probability exp(−Δ / T)  (if T very small, probability≈0)
            if temp > 0.0 and rng.random() < math.exp(-(new_obj - old_obj) / temp):
                accept = True

        if accept:
            cur_score = new_obj
            if new_obj < best_score:
                best_score = new_obj
                best_vec = a.copy()
                print(f"[{time.time() - start:.2f}s] iter={it:,} "
                      f"new best={best_score:.6f}")
        else:
            # reject – revert the move
            _apply_move(a, conv, dst, src, eps_move)

        # occasional full renormalisation to keep numeric errors in check
        steps_since_renorm += 1
        if steps_since_renorm >= renorm_interval:
            total = a.sum()
            if total > 0.0:
                a /= total
                conv = _conv_fft(a)           # recompute from scratch
                # recompute the constant factor (should still be 2·n)
                n_obj = 2.0 * n / (total * total)
                cur_score = n_obj * conv.max()
            steps_since_renorm = 0

    # --------------------------------------------------------------
    # 3) Final normalisation and return
    # --------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
