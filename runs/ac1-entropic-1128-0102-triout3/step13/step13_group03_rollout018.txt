"""
A fast, multi‑stage optimiser for the step‑function objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works inside the 1000 s budget:

1️⃣  Very cheap parametric sweeps (geometric, exponential, mixed with uniform,
    log‑normal) to obtain a few good seeds.
2️⃣  A simplex‑projected *mirror‑descent* that uses the exact sub‑gradient
    of the (non‑smooth) max‑convolution.  The update is
        a ← a * exp(‑η·g)  ,  g_i = 2·a_{k‑i}   (k = argmax conv)
    which directly reduces the worst convolution entry.
3️⃣  A short stochastic hill‑climb (random + targeted moves) for polishing.
All steps respect the time budget and keep the vector non‑negative with unit
sum.  The routine returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Quick exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective for a vector whose L1‑norm is ≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# 1️⃣  Parametric sweeps – cheap generators of promising seeds
# ----------------------------------------------------------------------
def _parametric_sweep(

    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:

    Explore several families of analytic shapes (geometric, exponential,
    mixed with uniform, log‑normal) and keep the best vector seen.
    The sweep stops early if the remaining time is low.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    i = np.arange(n, dtype=np.float64)                  # i
    # --------------------------------------------------------------
    # family 1 – geometric * power law, mixed with uniform
    # --------------------------------------------------------------
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=45)
    p_grid = np.linspace(-4.0, 3.0, 41)

    for r in r_grid:

            s = shape.sum()
            if s == 0.0:

                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time()-start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )
    # --------------------------------------------------------------
    # family 2 – exponential * power law (different decay rates)
    # --------------------------------------------------------------
    alpha_grid = np.linspace(0.001, 0.020, 20)
    for alpha in alpha_grid:
        if time.time() >= deadline - 90:
            break
        exp_part = np.exp(-alpha * i)
        for p in p_grid:
            if time.time() >= deadline - 90:
                break
            shape = exp_part * np.power(idx, p)
            if np.any(~np.isfinite(shape)):
                continue
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time()-start_time:.2f}s] sweep α={alpha:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )
    # --------------------------------------------------------------
    # family 3 – random log‑normal vectors (high variance)
    # --------------------------------------------------------------
    for _ in range(12):

        seq = 0.5 * uniform + 0.5 * shape

        sc = _eval_fast(seq, L_fft)
        if sc < best_score:

            best_seq = seq.copy()
            print(

# ----------------------------------------------------------------------
# 2️⃣  Mirror‑descent on the exact max‑convolution (sub‑gradient method)
# ----------------------------------------------------------------------
def _mirror_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Simple mirror‑descent on the non‑smooth objective.
    We use the exact sub‑gradient given by the worst convolution entry.
    The update is multiplicative (a ← a·exp(‑η·g)) followed by renormalisation.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = best_score_initial
    # step‑size schedule – start fairly aggressive and decay slowly
    eta = 200.0
    decay = 0.9997

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        k = int(np.argmax(conv))
        maxc = conv[k]
        cur_score = 2.0 * n * maxc  # sum(a) == 1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time()-start_time:.2f}s] mirror it={it:,} score={cur_score:.6f}"
            )

        # sub‑gradient of max‑conv at index k:
        #   g_i = 2·a_{k‑i}   if 0 ≤ k‑i < n else 0
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[k - idx_range]

        # mirror‑descent multiplicative update
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            # fallback to uniform if numerical trouble
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # decay eta slowly
        eta *= decay

        # occasional tiny jitter to avoid stagnation
        if (it + 1) % 2500 == 0:
            jitter = 1e-4 * np.random.default_rng().standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final stochastic hill‑climb (random + targeted moves)
# ----------------------------------------------------------------------
def _targeted_hill_climb(
    start_a: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves together with targeted moves that try to
    reduce the current worst convolution entry.  Only improving moves are
    kept (no uphill steps) – the method works well as a final polish.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    step_factor = 0.13               # up to 13 % of source mass per move
    min_factor = 1e-5
    decay = 0.993

    while time.time() < deadline:
        if rng.random() < 0.55:
            # ---- random transfer -----------------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # ---- targeted transfer (focus on current max convolution) ----
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            # choose a pair (i,j) that contributes to k_max
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.13)   # reset after success
            print(
                f"[{time.time()-start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score

        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time()-start_time:.2f}s] mirror descent #{idx_start} improvement score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, L_fft, deadline, start_time)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time()-start_time:.2f}s] final hill‑climb score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise once more and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check with the official evaluator (optional)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time()-start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
