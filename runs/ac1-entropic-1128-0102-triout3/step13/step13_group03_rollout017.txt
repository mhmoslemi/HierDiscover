def subgradient_descent(a0, deadline, start_time, best_score):
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    t = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    n = a.shape[0]
    L_fft = _next_pow2(2*n-1)
    while time.time() < deadline:
        t+=1
        conv = _conv_self(a, L_fft)
        k = int(np.argmax(conv))
        # compute gradient for max entry
        grad = a.copy()
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi+1)
        grad[i] += a[k - i]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        lr = 0.5 / (1 + 1e-4*t)   # may adapt
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)
        # evaluate
        conv2 = _conv_self(a, L_fft)
        sc = 2.0 * n * conv2.max() / (a.sum()**2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] subgrad it={t} score={sc:.6f}")
    return best_a, best_sc

"""
search for a non‑negative step function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

1.  Initialise with the uniform distribution and, if available, the global
    starter `height_sequence_1`.
2.  Perform a **two‑stage parametric sweep**:
    – a coarse grid over the family
          a_i ∝ r**i · (i+1)**p   (mixed with a uniform component),
      keeping the best candidate.
    – a finer grid centred on the best parameters + a short random
      search over mixtures of two geometric laws.
3.  Run a **sub‑gradient descent** that uses the exact gradient of the
    current maximal convolution entry (instead of a smooth‑max surrogate).
    The iterate is projected onto the probability simplex after each step.
4.  Polish the result with a **targeted hill‑climb** that moves a small
    fraction of mass from indices that contribute most to the maximal
    convolution entry to random destinations.
5.  Return the best sequence found (normalised to sum 1).  The total
    runtime never exceeds the supplied budget (default 1000 s).
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Simple linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w | w≥0, Σw=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector whose sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_geometric_shape(r: float, p: float, n: int) -> np.ndarray:
    """Return normalized shape a_i ∝ r**i·(i+1)**p (i=0..n‑1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    geo = r ** np.arange(n, dtype=np.float64)           # r**i
    shape = geo * (idx ** p)
    s = shape.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return shape / s


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return λ·uniform + (1‑λ)·shape, re‑normalised."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n)
    seq = lam * uniform + (1.0 - lam) * shape
    seq /= seq.sum()
    return seq


def _make_mixture_two_exp(r1: float, r2: float, alpha: float, n: int) -> np.ndarray:
    """
    Mixture α·Geom(r1) + (1‑α)·Geom(r2) truncated to length n and renormalised.
    Each geometric law is the standard infinite law (1‑r)·r**i.
    """
    i = np.arange(n, dtype=np.float64)
    component1 = (1.0 - r1) * (r1 ** i)
    component2 = (1.0 - r2) * (r2 ** i)
    a = alpha * component1 + (1.0 - alpha) * component2

    # truncated sums of the two infinite geometrics
    sum1 = 1.0 - r1 ** n
    sum2 = 1.0 - r2 ** n
    total = alpha * sum1 + (1.0 - alpha) * sum2
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return a / total


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution term
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Projected (Adam‑style) descent using the exact gradient of
    max(conv(a,a)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        t += 1

        conv = _conv_self(a, L_fft)
        k_max = int(np.argmax(conv))

        # exact sub‑gradient of the active entry
        grad = a.copy()
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        grad[i] += a[k_max - i]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = 0.6 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # exact score after the projection
        conv2 = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv2.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={t:,} score={cur_score:.6f}"
            )

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            rng = np.random.default_rng()
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (random + max‑entry attacks)
# ----------------------------------------------------------------------


def _targeted_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers (≤15 % of source mass) plus
    targeted moves that directly reduce the current dominant convolution entry.
    Only improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    step_factor = 0.15      # max fraction of source mass moved per trial
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        if rng.random() < 0.6:   # --- random transfer ---------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:                    # --- targeted transfer -------------------------------
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose the move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)     # reset after a success
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Coarse‑to‑fine parametric sweep (including two‑exp random search)
# ----------------------------------------------------------------------


def _parametric_sweep_fine(
    n: int,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Exhaustive grid over (r, p, λ) with a coarse stage,
    a fine stage around the best parameters and a random two‑exponential
    search.  Returns the best vector discovered.
    """
    uniform = np.full(n, 1.0 / n)
    idx = np.arange(1, n + 1, dtype=np.float64)

    best_r = best_p = best_lam = None

    # ---------------------- coarse grid ----------------------
    r_grid = np.logspace(np.log10(0.95), np.log10(0.9999), 60)
    p_grid = np.linspace(-2.0, 3.0, 101)
    lam_grid = np.linspace(0.0, 1.0, 11)

    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        geo = r ** np.arange(n, dtype=np.float64)
        for p in p_grid:
            if time.time() >= deadline - 120:
                break
            shape = geo * (idx ** p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    best_r, best_p, best_lam = r, p, lam
                    print(
                        f"[{time.time() - start_time:.2f}s] coarse r={r:.6f} p={p:.3f} λ={lam:.2f} → {sc:.6f}"
                    )

    # ---------------------- fine grid around the best ----------------------
    if best_r is not None and time.time() < deadline - 60:
        r_low = max(best_r * 0.98, 0.90)
        r_high = min(best_r * 1.02, 0.999999)
        p_low = best_p - 0.5
        p_high = best_p + 0.5
        lam_low = max(best_lam - 0.2, 0.0)
        lam_high = min(best_lam + 0.2, 1.0)

        r_grid2 = np.linspace(r_low, r_high, 30)
        p_grid2 = np.linspace(p_low, p_high, 41)
        lam_grid2 = np.linspace(lam_low, lam_high, 11)

        for r in r_grid2:
            if time.time() >= deadline - 60:
                break
            geo = r ** np.arange(n, dtype=np.float64)
            for p in p_grid2:
                if time.time() >= deadline - 60:
                    break
                shape = geo * (idx ** p)
                s = shape.sum()
                if s == 0.0:
                    continue
                shape /= s
                for lam in lam_grid2:
                    seq = lam * uniform + (1.0 - lam) * shape
                    seq /= seq.sum()
                    sc = _eval_fast(seq, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print(
                            f"[{time.time() - start_time:.2f}s] fine   r={r:.6f} p={p:.3f} λ={lam:.2f} → {sc:.6f}"
                        )

    # ---------------------- random mixture of two geometries ----------------------
    rng = np.random.default_rng()
    while time.time() < deadline - 30:
        # sample r1, r2 in (0.90, 0.9999) on a log‑scale
        r1 = np.exp(rng.uniform(np.log(0.90), np.log(0.9999)))
        r2 = np.exp(rng.uniform(np.log(0.90), np.log(0.9999)))
        alpha = rng.random()
        seq = _make_mixture_two_exp(r1, r2, alpha, n)
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] two‑exp r1={r1:.6f} r2={r2:.6f} α={alpha:.3f} → {sc:.6f}"
            )

    return best_seq, best_score
