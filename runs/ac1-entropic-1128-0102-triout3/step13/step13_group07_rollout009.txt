"""
Propose a step‑function (non‑negative sequence) that minimizes the objective

    2·n·max(conv(a,a))   with   Σ a = 1 .

The search blends three ideas that go beyond the reference implementation:

1️⃣  Warm‑starts with a rich palette of analytic shapes **and** a few random
    Dirichlet vectors.

2️⃣  A two‑stage Adam optimiser.
   •  Stage A – classic projected Adam on the exact sub‑gradient of the max
      convolution entry (identical to the reference, but we keep it as a
      quick “coarse” phase).
   •  Stage B – an *annealed* Adam that minimizes a **smoothed** version of
      the max‑convolution using the soft‑max (log‑sum‑exp) trick.
      The temperature (α) is increased stepwise (0.5 → … → 40) which lets the
      optimiser first explore a gentle landscape and then focus on the true
      peak.  The gradient of the smooth objective is obtained analytically as
      a cross‑correlation and implemented with a single FFT per iteration.

3️⃣  A cheap stochastic hill‑climb that moves a random packet of mass between
    two indices and accepts only strict improvements.  It runs on whatever
    time is left after the Adam phases.

All iterates are projected onto the probability simplex
({w ≥ 0, Σw = 1}) after each update, guaranteeing that the final sequence can
be fed directly to ``evaluate_sequence``.  The routine respects the 1000 s
budget – if the time runs out the current best sequence is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns the (2·n‑1)‑length “valid” part.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(a∗a)   (a already normalised to sum = 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families (plus a few random vectors)
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) flat (uniform) – baseline
    candidates.append(np.ones_like(x))

    # normalise and keep only non‑negative ones
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    conv must be the full linear self‑convolution.
    """
    n = a.shape[0]
    k = int(conv.argmax())            # index attaining the maximum
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the simplex using the *exact* max‑convolution sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        grad = _max_subgradient(a, conv)          # sub‑gradient of max(conv)
        grad_obj = 2.0 * n * grad                  # gradient of the true objective

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, alpha: float,
                         L_full: int) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_objective, gradient) for the current a.
    The smooth objective is
          (1/α)·log Σ_k exp(α·conv_k)
    whose gradient is 2·(a ⋆ w) where w is the soft‑max over conv.
    """
    n = a.shape[0]

    # ---- self‑convolution (full length 2·n‑1) -------------------------
    Af = np.fft.rfft(a, n=L_full)
    conv_full = np.fft.irfft(Af * Af, n=L_full)
    conv = conv_full[: 2 * n - 1]                     # length 2·n‑1

    # ---- soft‑max weights -------------------------------------------
    mx = conv.max()
    shifted = np.exp(alpha * (conv - mx))            # safe: max shifted to 0
    sum_shifted = shifted.sum()
    w = shifted / sum_shifted                        # w_k ≥ 0, Σ w_k = 1

    # ---- gradient = 2·cross‑correlation(a, w) -----------------------
    # we need the slice of the full correlation a * rev(w)
    w_rev = w[::-1]
    corr_full = np.fft.irfft(Af * np.fft.rfft(w_rev, n=L_full), n=L_full)

    start = n - 1
    segment = corr_full[start:start + n]             # length n
    grad = 2.0 * segment[::-1]                       # see derivation in analysis

    # ---- smooth objective value (optional, used only for logging) ---
    smooth_val = mx + np.log(sum_shifted) / alpha
    return smooth_val, grad


def _adam_optimize_smooth(a0: np.ndarray,
                          alphas: list[float],
                          iters_per_alpha: int,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.02,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Annealed Adam that minimises the smoothed max‑convolution.
    For each α in *alphas* we run *iters_per_alpha* Adam steps.
    """
    a = a0.copy()
    n = a.shape[0]
    L_full = _next_pow2(3 * n - 2)          # size for a⋆rev(w) (full length 3n‑2)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it_total = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it_total += 1

            # smooth objective & gradient
            smooth_val, grad = _smooth_obj_and_grad(a, alpha, L_full)

            # Adam update (the smooth objective is already the scalar we minimise)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it_total)
            v_hat = v / (1.0 - beta2 ** it_total)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a -= step
            a = _project_simplex(a)

            # occasional exact evaluation (every few hundred steps)
            if it_total % 200 == 0:
                cur_score = _objective(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"SmoothAdam α={alpha:.2f} it={it_total:,} "
                          f"score={cur_score:.6f}")

        lr *= 0.9        # gentle decay after each temperature level

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """Greedy Metropolis walk that moves a random packet of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                     # source weighted by mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30          # move at most 30 % of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:                    # accept only improvements
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"HillClimb it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The returned list is already normalised (Σ = 1) and can be passed
    directly to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # --------------------------------------------------------------
    n = 4096            # works comfortably inside the time budget

    # --------------------------------------------------------------
    # 1) Warm‑starts – analytic shapes + a few random points.
    # --------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # a handful of random Dirichlet vectors (sparse supports are useful)
    for _ in range(5):
        rnd = np.random.exponential(scale=1.0, size=n)
        rnd /= rnd.sum()
        warm_starts.append(rnd)

    # if a global construction is supplied, resample it to length n
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            h1_resampled = np.maximum(h1_resampled, 0.0)
            s = h1_resampled.sum()
            if s > 0:
                warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase‑A: classic projected Adam on the *exact* max‑convolution.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        phase_a_deadline = time.time() + remaining * 0.30   # ~30 % of remaining
        a_opt, score_opt = _adam_optimize(a0, phase_a_deadline, start_time, lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Phase‑A Adam best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Phase‑B: annealed Adam on the *smoothed* objective.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        # temperature schedule – gradually tighten around the true max
        alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
        iters_per_alpha = 3000                     # ≈21 k iterations total
        phase_b_deadline = time.time() + remaining * 0.60   # allocate ~60 %
        a_opt, score_opt = _adam_optimize_smooth(
            a0, alphas, iters_per_alpha, phase_b_deadline, start_time, lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Phase‑B smooth‑Adam best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 4) Random‑restart stage – cheap smoothed Adam runs from fresh starts.
    # --------------------------------------------------------------
    while True:
        remaining = deadline - time.time()
        if remaining < 6.0:            # need a few seconds for a meaningful run
            break

        # generate a sparse random start (Dirichlet with small concentration)
        rnd = np.random.exponential(scale=1.0, size=n)
        rnd /= rnd.sum()

        # give each restart a modest budget (~4 s)
        restart_deadline = time.time() + min(4.0, remaining * 0.20)

        a_opt, score_opt = _adam_optimize_smooth(
            rnd,
            alphas=[1.0, 5.0, 10.0, 20.0, 40.0],
            iters_per_alpha=800,
            deadline=restart_deadline,
            start_time=start_time,
            lr=0.02)

        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart best score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 5) Light hill‑climb polish (uses the exact objective).
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={score_opt:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation & conversion.
    # --------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
