```
def _exponential_gradient_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # determine active set: indices with values close to max
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # step size schedule
        eta = step0 / np.sqrt(it)
        # exponentiated update (multiplicative)
        a = a * np.exp(-eta * grad)
        # renorm to simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0/n)
        # record best
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it} score={cur_score:.6f}")

    return best_a, best_score
```

assistantfinal**Solution Explanation**

The objective to minimise is  

\[
J(a)=\frac{2\,n\,\max\_{k}(a*a)[k]}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the denominator is homogeneous, we may always normalise the vector to a
probability simplex (`∑a = 1`).  After normalisation  

\[
J(a)=2\,n·\max\_k (a*a)[k] .
\]

The maximum of the self‑convolution is a convex, piece‑wise linear function.
A sub‑gradient at a point `a` is obtained from the convolution index (or a
small set of indices) where the maximum is attained:

\[
\partial_k J(a)=2\,a_{k-i}\qquad\text{for every }i\text{ with }0\le i,k-i<n .
\]

The classic sub‑gradient method used in the reference code averages a fixed
number of the largest convolution entries (`top_k`).  In practice this
produces very small steps and the search stalls around the score
≈ 1.508.

The new algorithm replaces the additive update by a *multiplicative*
(Exponentiated‑Gradient) update on the simplex.  For a convex objective
the EG update

\[
a_i \leftarrow a_i\;e^{-\eta\,g_i}
\]

automatically keeps the variables non‑negative; a final normalisation puts
the vector back on the simplex.  Because the update is multiplicative it
pushes mass *away* from the indices that contribute most to the current
maximum and concentrates it where the contribution is small – exactly what
is needed to lower the peak of the convolution.

The optimisation proceeds in three stages

1. **Initial pool** – a few hundred analytic shapes (beta‑distributions,
   random mixtures, the uniform vector and the optionally supplied global
   starter) are generated and the ten best are kept.

2. **Exponentiated‑gradient descent** – the currently best vector is refined
   using the EG rule.  In each iteration  
   * the full self‑convolution is computed with an FFT,  
   * all indices whose value is within a tiny tolerance of the maximum are
     collected (active set),  
   * the sub‑gradient is the average of the gradients of those active
     indices,  
   * a diminishing step size `η = η0/√t` is used, and  
   * the vector is projected back onto the simplex by renormalisation.  
   The best iterate seen so far is stored.

3. **Simulated‑annealing polish** – a cheap Metropolis walk that swaps tiny
   packets of mass between two positions.  The walk accepts worsening moves
   with a fast exponential cooling schedule, allowing the algorithm to
   escape tiny plateaus left by the EG phase.

All three stages respect the global time budget (default 1000 s).  The
result is returned as a plain Python list of non‑negative floats.

```python
"""
Propose a low‑score non‑negative step function.

The algorithm consists of three phases:

1.  Build a small pool of analytic candidate shapes (beta‑distributions,
    random mixtures, uniform).  Keep the best few.

2.  Refine the current best vector with an Exponentiated‑Gradient descent
    on the simplex.  At each iteration the full self‑convolution is
    computed with a real FFT, the active set of indices attaining
    (approximately) the maximum is built, and a sub‑gradient is formed as
    the average of the gradients of those indices.  The update is
    multiplicative (a_i ← a_i·exp(–η·g_i)) followed by renormalisation.
    The step size η decays as η0/√t.

3.  A short simulated‑annealing polish swaps tiny packets of mass between
    random positions and accepts moves according to a fast exponential
    cooling schedule.

The whole routine stops before the supplied time budget (default 1000 s)
and returns the best sequence found as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)  (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    The continuous beta pdf ∝ x^{a‑1} (1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Multiplicative (EG) descent on the probability simplex.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    # tiny floor to avoid zeros that would get stuck forever
    eps_floor = 1e-14

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active set: all indices within a very small relative tolerance
        tol = max_val * 1e-12
        ks = np.where(conv >= max_val - tol)[0]

        # build sub‑gradient (average over the active set)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # step size schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update
        a = a * np.exp(-eta * grad)

        # enforce a tiny floor, then renormalise to the simplex
        a = np.maximum(a, eps_floor)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions and accept
    according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # cooling: T goes from 0.02 to 1e-6
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)               # mass‑biased source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5      # move up to 50 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score
```
