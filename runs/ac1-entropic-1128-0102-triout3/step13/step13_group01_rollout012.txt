"""
propose_candidate –  multi‑scale smoothed‑max gradient descent + annealing

The algorithm works in three stages:

1.  Build a modest‑size “coarse” problem (n_small = 8192) where each
    iteration is cheap.  For a given shape a the objective
        J(a)=2·n·max(a*a)
    is approximated by a smooth log‑sum‑exp of the convolution.  The
    gradient of the smooth surrogate is computed analytically via FFTs,
    a momentum step is taken, and the result is projected onto the
    probability simplex.

2.  Run the above optimiser from several analytic seeds (uniform,
    beta‑distributions, power ramps) and from a number of random
    Dirichlet initialisations.  The best coarse shape found is kept.

3.  Upsample the coarse shape to the full resolution (n=65536) by
    replication, renormalise, and perform a short refinement with the
    same smoothed‑max optimiser followed by a light simulated‑annealing
    polish.

All steps respect the 1000 s budget, report progress, and finally
return a plain list of non‑negative heights (already normalised to sum = 1).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1} (Wang‑Carreira‑Perpiñán)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (return full length)."""
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)
    return conv[: len(a) + len(b) - 1]


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic prototype families (used as seeds)
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with a flat plateau, normalised to sum = 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised Beta(a,b) on {0,…,n‑1}, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (on a given length)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    beta_start: float = 0.8,
                    beta_max: float = 30.0,
                    beta_inc: float = 0.001,
                    step0: float = 0.08,
                    momentum_coef: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed max of the autocorrelation.
    Uses a single momentum term and projects onto the simplex after each step.
    Returns the best iterate (according to the exact objective) and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Pre‑compute FFT length (covers both self‑convolution and a*w_rev)
    L = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = beta_start
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- FFT of a (shared for self‑conv and gradient) -----
        Af = np.fft.rfft(a, n=L)

        # ----- self‑convolution (to obtain weights) -----
        conv_self = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # ----- soft‑max weights for the smooth max -----
        max_c = conv_self.max()
        # stable exponentiation
        exps = np.exp(beta * (conv_self - max_c))
        w = exps / exps.sum()                     # length 2n‑1, Σw = 1

        # ----- gradient of the smooth max -----
        w_rev = w[::-1]
        W_rev_f = np.fft.rfft(w_rev, n=L)
        conv_full = np.fft.irfft(Af * W_rev_f, n=L)[: 3 * n - 2]
        # Extract the n‑dimensional gradient component:
        # g_k = 2 * Σ_j a_j * w_{j+k}  → 2 * conv_full[(2n‑2)‑k]
        grad_inner = 2.0 * conv_full[(2 * n - 2) - np.arange(n)]

        # Full gradient of J = 2·n·S
        grad = 2.0 * n * grad_inner

        # ----- momentum step & projection -----
        momentum = momentum_coef * momentum + (1.0 - momentum_coef) * grad
        step = step0 / math.sqrt(it)               # diminishing step‑size
        a -= step * momentum
        a = _project_simplex(a)

        # ----- record the best exact score -----
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑descent it={it:,}  score={cur_score:.6f}")

        # ----- slowly raise the soft‑max temperature (beta) -----
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same as in the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Metropolis annealing that swaps tiny mass packets between coordinates."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # fast exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # more mass → higher move probability
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function within the given time budget.
    The routine works on a coarse grid (n = 8192) using a smoothed‑max
    gradient descent, then upsamples to 65536 points and finishes with a short
    annealing polish.  The returned list contains non‑negative floats and
    respects the 1000 s limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Problem sizes
    # --------------------------------------------------------------
    n_large = 65536
    factor = 8                      # n_large = factor * n_small
    n_small = n_large // factor     # 8192

    # --------------------------------------------------------------
    # 2) Build a pool of seed vectors (length n_small)
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # uniform
    seeds.append(np.full(n_small, 1.0 / n_small))

    # beta‑distribution grid (coarse)
    for a_par in [0.4, 0.7, 1.0, 2.0, 3.0]:
        for b_par in [0.4, 0.7, 1.0, 2.0, 3.0]:
            seeds.append(_make_beta_shape(n_small, a_par, b_par))

    # power‑ramp family
    for p in [0.5, 1.0, 1.5, 2.0]:
        for alpha in [0.0, 0.1, 0.2, 0.3]:
            seeds.append(_make_power_shape(n_small, p, alpha))

    # global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size >= n_large:
            # trim/pad to exact large size
            if starter.size > n_large:
                starter = starter[:n_large]
            elif starter.size < n_large:
                # pad with zeros (very unlikely)
                starter = np.concatenate([starter, np.zeros(n_large - starter.size)])
            # down‑sample by averaging over blocks of length `factor`
            a_small = starter.reshape(factor, n_small).mean(axis=0)
            a_small = np.maximum(a_small, 0.0)
            total = a_small.sum()
            if total > 0.0:
                a_small /= total
                seeds.append(a_small)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) Coarse optimisation on n_small
    # --------------------------------------------------------------
    best_small = None
    best_small_score = np.inf

    # allocate a modest slice of the budget for deterministic seeds
    per_seed_time = min(30.0, (deadline - time.time()) / max(1, len(seeds)))
    for idx, a0 in enumerate(seeds):
        if time.time() >= deadline:
            break
        sub_deadline = min(deadline, time.time() + per_seed_time)
        a_opt, sc = _smooth_descent(a0, sub_deadline, start_time)
        if sc < best_small_score:
            best_small_score = sc
            best_small = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Seed {idx} improved best to {sc:.6f}")

    # Random Dirichlet restarts for the remaining time
    rng = np.random.default_rng()
    while time.time() < deadline - 5.0:          # keep ~5 s for final stages
        a0 = rng.dirichlet(np.full(n_small, 0.4))   # sparse‐ish draws
        sub_deadline = min(deadline, time.time() + 12.0)   # ~12 s per restart
        a_opt, sc = _smooth_descent(a0, sub_deadline, start_time)
        if sc < best_small_score:
            best_small_score = sc
            best_small = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Random restart improved to {sc:.6f}")

    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)
        best_small_score = _objective(best_small)

    # --------------------------------------------------------------
    # 4) Upsample to full resolution
    # --------------------------------------------------------------
    a_large = np.repeat(best_small, factor)
    a_large = np.maximum(a_large, 0.0)
    total = a_large.sum()
    if total > 0.0:
        a_large /= total
    else:
        a_large = np.full(n_large, 1.0 / n_large)

    # --------------------------------------------------------------
    # 5) Fine‑grained refinement on the full size (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        refine_deadline = deadline - 5.0          # leave a few seconds for anneal
        a_large, best_large_score = _smooth_descent(a_large, refine_deadline, start_time)
        best_small_score = best_large_score      # reuse variable for final reporting

    # --------------------------------------------------------------
    # 6) Light annealing polish (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_large, final_score = _anneal(a_large, deadline, start_time)
        best_small_score = final_score

    # --------------------------------------------------------------
    # 7) Normalise and return as plain Python list
    # --------------------------------------------------------------
    a_large = np.maximum(a_large, 0.0)
    total = a_large.sum()
    if total > 0.0:
        a_large /= total
    else:
        a_large = np.full_like(a_large, 1.0 / a_large.shape[0])

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_small_score:.6f}")
    return [float(v) for v in a_large]
