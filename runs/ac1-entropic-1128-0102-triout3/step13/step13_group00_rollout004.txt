def _adam_smooth_max(a0, start_time, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.1  # base step size
    tau = 5.0
    tau_factor = 1.3
    iteration = 0
    while time.time() < deadline:
        iteration+=1
        # compute conv via FFT
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={iteration} score={cur_score:.6f}")
        # softmax weights
        maxc = conv.max()
        w = np.exp(tau * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum
        # gradient
        grad = 2.0 * _conv_fft(w, a[::-1])[n-1:n-1+n]
        # Adam updates
        m = beta1*m + (1-beta1) * grad
        v = beta2*v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        a_new = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new

        # schedule tau increase every 2000 its
        if iteration % 2000 == 0:
            tau *= tau_factor
            if tau > 1e6:
                tau = 1e6

    return a, best_seq, best_score
```

def _hill_climb_sa(a0, start_time, deadline):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2*n-1)
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    T0 = 0.01  # starting temperature (relative to best_score)
    # schedule: T = T0 * (1 - (elapsed / total))
    while time.time() < deadline:
        elapsed = time.time() - start_time

"""
Step‑function optimiser – 1000 s budget

The algorithm is a multi‑stage search for a probability vector a (sum = 1,
non‑negative) that minimises

    2·n·max(conv(a,a))

where conv is the linear self‑convolution.  The value equals the score
returned by ``evaluate_sequence`` because the objective is scale‑invariant.

Stages
------
1️⃣  Broad parametric sweep (geometric, power, beta, Gaussian) on the
    simplex.
2️⃣  Random Dirichlet initialisers with a wide range of concentration.
3️⃣  Smooth‑max descent using the Adam optimiser.  The smooth objective is
    Sτ(a)= (1/τ)·log∑exp(τ·conv) whose gradient is
        ∇Sτ = 2·(softmax(conv) * a_rev)
    The temperature τ is increased geometrically; each iterate is projected
    onto the probability simplex.
    Several restarts are performed with different seeds.
4️⃣  Simulated‑annealing hill‑climb.  Random mass transfers between two
    indices are accepted if they improve the true objective, otherwise
    accepted with a temperature‑controlled probability, allowing the
    search to escape shallow local minima.
5️⃣  The best vector found is returned as a plain list of Python floats.

All convolutions use FFT for O(n log n) speed, the simplex projection is
the classic O(n) algorithm, and every loop checks the global deadline
to respect the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming sum(a)=1.
    This equals the score returned by evaluate_sequence().
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

def _sweep_families(n: int, best_seq: np.ndarray, best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Search simple analytic families (geometric, power, beta, Gaussian)."""
    idx = np.arange(n, dtype=np.float64)

    # Uniform (already considered as baseline – kept for completeness)
    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Geometric series a_i ∝ r^i
    # --------------------------------------------------------------
    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        # blend with uniform for additional flexibility
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # Power law a_i ∝ (i+1)^p
    # --------------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.5, 140)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pw = np.power(idx + 1.0, p)
        s = pw.sum()
        if s == 0:
            continue
        pw /= s
        sc = _objective_score(pw)
        if sc < best_score:
            best_score, best_seq = sc, pw.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # Beta‑type a_i ∝ i^{α‑1} (n‑i)^{β‑1}
    # --------------------------------------------------------------
    alphas = np.linspace(0.05, 8.0, 81)
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))   # stabilise exp
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # Discrete Gaussian centred in the middle
    # --------------------------------------------------------------
    mu = (n - 1) / 2.0
    sigma_grid = np.logspace(np.log10(0.5), np.log10(n / 2.0), 40)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        gauss = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        s = gauss.sum()
        if s == 0:
            continue
        gauss /= s
        sc = _objective_score(gauss)
        if sc < best_score:
            best_score, best_seq = sc, gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma:.2f} score={sc:.6f}")

    return best_seq, best_score

def _random_dirichlet_trials(n: int, best_seq: np.ndarray, best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:

    rng = np.random.default_rng()

# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max descent with Adam optimiser
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam optimisation of the smooth max Sτ(a) = (1/τ)·log∑exp(τ·conv).
    The temperature τ is increased geometrically; after each step the
    vector is projected onto the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam's internal states
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08           # base learning rate
    tau = 5.0                # start with a smooth objective
    tau_factor = 1.2
    it = 0
    # pre‑allocation for FFT lengths
    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_grad = _next_pow2(3 * n - 2)          # for w * a_rev

    while time.time() < deadline:
        it += 1

        # --- true objective (for bookkeeping) --------------------
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # --- smooth max weights ----------------------------------
        conv = _conv_self(a, L_conv)
        max_c = conv.max()
        # shift to avoid overflow
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # --- gradient of Sτ --------------------------------------
        # grad = 2 * (w * a_rev) restricted to length n
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)               # length = 3n‑2
        grad = 2.0 * conv_w[n - 1: n - 1 + n]

        # --- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new

        # --- temperature schedule --------------------------------
        if it % 1800 == 0:
            tau *= tau_factor
            tau = min(tau, 1e7)

    return a, best_seq, best_score

# 4️⃣  Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _hill_climb_sa(start_a: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with occasional uphill moves.
    Randomly transfers a small amount of mass between two indices.
    Acceptance follows a simple annealing schedule.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = _objective_score(a)

    # annealing schedule parameters
    total_sa_time = deadline - time.time()
    if total_sa_time <= 0:
        return best_seq, best_score
    T0 = 0.02 * cur_score          # initial temperature (relative)
    it = 0

    while time.time() < deadline:
        it += 1
        # linearly decreasing temperature
        elapsed = time.time() - start_time
        T = T0 * max(0.0, 1.0 - (elapsed / (deadline - start_time)))

        # propose a move: take ε from i and give to j
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] < 1e-12:
            continue
        # amount is a modest fraction of source mass
        frac = rng.random() * 0.04 + 0.001   # 0.1 % – 4 %
        eps = min(a[i] * frac, a[i])
        if eps <= 0:
            continue
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps

        # evaluate
        prop_score = _objective_score(a_prop)

        # acceptance test
        if prop_score < cur_score:
            a, cur_score = a_prop, prop_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA better score={best_score:.6f}")
        else:
            # uphill move with probability exp(-(Δ)/T)
            if T > 0 and rng.random() < math.exp(-(prop_score - cur_score) / T):
                a, cur_score = a_prop, prop_score

        # occasional re‑projection to mitigate drift
        if it % 2000 == 0:
            a = _project_simplex(a)

    return best_seq, best_score

                c = rng.uniform(0.05, 8.0)
                init = rng.gamma(shape=c, scale=1.0, size=n)
                init /= init.sum()
            best_seq, best_seq, best_score = _adam_smooth_max(
                init, start_time, time.time() + per_restart,
                best_seq, best_score
            )
        print(f"[{time.time() - start_time:.2f}s] After Adam runs best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final stochastic hill‑climb (simulated annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final SA hill‑climb")
        best_seq, best_score = _hill_climb_sa(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final SA score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
