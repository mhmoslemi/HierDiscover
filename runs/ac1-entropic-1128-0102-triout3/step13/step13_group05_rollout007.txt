def _subgrad_descent(a0, deadline, start_time):
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a)
    it =0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_c = conv.max()
        # find indices where conv close to max; tolerance relative e.g., 1e-12? Use equality.
        mask = (conv == max_c)
        # if too many? fine.
        w = mask.astype(float)
        if w.sum() == 0:
            break
        w /= w.sum()
        # gradient using w
        # conv_weight_rev = conv(w, a_rev) via FFT
        a_rev = a[::-1]
        L = _next_pow2(3 * len(a) - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3*len(a) -2]
        grad = 2.0 * corr[len(a)-1 : len(a)-1+len(a)]

        # compute step size, maybe diminishing: step = step0 / sqrt(it)
        step = step0 / math.sqrt(it)

        a_new = a - step * grad
        # projection
        a_new = _project_simplex(a_new)

        # evaluate objective
        new_score = _objective(a_new)
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()

        # optionally accept if better, else maybe still move for exploration; but we can just keep a=a_new for next iteration always.
        a = a_new
    return best, best_score

```
"""
Multi‑scale projected sub‑gradient descent for the max‑convolution
objective.  The algorithm builds a few analytic candidates, refines them
by deterministic sub‑gradient steps on the exact max‑convolution (no
soft‑max), upsamples the best candidate to a larger discretisation and
repeats.  A short simulated‑annealing finish is performed if time
remains.  The result is returned as a plain Python list of non‑negative
floats.
"""
import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp shape with optional plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the sorted top‑k pool (ascending scores)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int = None) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient (Adam) descent on the *exact* max‑convolution.
    The vector is kept on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.2  # base learning rate, will be scaled down automatically

    it = 0
    while time.time() < deadline:
        it += 1
        if max_iters is not None and it > max_iters:
            break

        # current convolution and its maximum
        conv = _conv_self(a)
        max_c = conv.max()

        # mask of maximising indices (allow a few ties for numerical stability)
        mask = (conv >= max_c - 1e-15)
        w = mask.astype(float)
        nz = w.sum()
        if nz == 0:
            break
        w /= nz  # uniform weights over the active set

        # gradient of the weighted max‑convolution = 2 * (w ⋆ a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update (step size slowly decays)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a_new = _project_simplex(a_new)

        cur_score = _objective(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

def _upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = vec.shape[0]
    if old_n == new_n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full(new_n, 1.0 / new_n)
    return up

        best_vec /= s
    else:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.shape[0])
```

The program builds a step‑function that minimises  

```
2·n·max(conv(a,a)) / (sum(a)**2)
```  

subject to `a_i ≥ 0`.  
The search is performed in three phases

1. **Multi‑scale candidates** – a small pool of analytic shapes
   (Beta‑densities, power‑ramps, random Dirichlet draws) is generated.
   The best few vectors are kept.  If a global starter (`height_sequence_1`)
   is present it is up‑sampled and added to the pool.

2. **Deterministic sub‑gradient descent** – starting from every pooled
   vector a projected Adam‑style descent is executed on the *exact*
   max‑convolution (no soft‑max).  The gradient of the max‑convolution is
   `2·(w ⋆ a_rev)` where `w` is the uniform weight over the indices that
   achieve the current maximum.  After each step the vector is projected
   onto the probability simplex.  The best result of the level is kept.

3. **Final simulated‑annealing polish** – if time is left a short SA
   run moves tiny amounts of mass between entries, accepting worse
   states with a decreasing temperature.  This helps to escape the few
   remaining local minima.

The algorithm works on a hierarchy of sizes (`N = 4 k → 16 k → 64 k` by
default).  Optimisation on the small grid is cheap, allowing many
gradient steps; the best vector is then interpolated to the next resolution
and refined again.  The whole routine respects the 1000 s budget and
returns a plain Python list of non‑negative numbers.

```python
"""
Multi‑scale projected sub‑gradient descent for the max‑convolution
objective.  Analytic seeds are refined with deterministic Adam‑style
updates on the exact max‑convolution, up‑sampled to finer grids and
finally polished by a short simulated‑annealing run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the sorted top‑k pool (ascending scores)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


def _upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = vec.shape[0]
    if old_n == new_n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full(new_n, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Adam‑style descent on the true max‑convolution.
    The vector stays on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.2

    it = 0
    while time.time() < deadline:
        it += 1
        if max_iters is not None and it > max_iters:
            break

        # current convolution and its maximum
        conv = _conv_self(a)
        max_c = conv.max()

        # active set (all indices attaining the maximum, tolerance for fp noise)
        mask = (conv >= max_c - 1e-15)
        nz = mask.sum()
        if nz == 0:
            break
        w = mask.astype(float) / nz         # uniform weight on the active set

        # gradient = 2·(w ⋆ a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update (step size decays as 1/√t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a_new = _project_simplex(a_new)

        cur_score = _objective(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑GD iter={it:,} best={cur_score:.6f}")

        a = a_new

    return best_a, best_score

        best_vec = np.full(N_final, 1.0 / N_final)

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0:
        best_vec /= s
    else:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.shape[0])

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished, final objective ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
```
