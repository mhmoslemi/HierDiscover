Propose a low‑score step function.

The algorithm works in three stages:

1. Initialise a probability vector `a` (non‑negative, sum = 1).  
   If the global variable ``height_sequence_1`` is present it is
   interpolated to the working length (2**16) and used, otherwise a
   uniform distribution is taken.

2. Refine the vector with a *smooth* optimisation of the
   objective  2·n·max(a∗a).  Instead of the non‑smooth max we minimise
   the log‑sum‑exp surrogate
        Fτ(a) = 2·n·(1/τ)·log Σ_i exp(τ·(a∗a)_i) .
   The gradient of `Fτ` is obtained analytically and the variables are
   optimised in the *soft‑max* parametrisation `a = softmax(z)`,
   guaranteeing the simplex constraints automatically.
   The optimiser is Adam with a slowly increasing smoothing
   parameter τ (τ starts at 5 and is multiplied by 1.001 each iteration
   up to a maximum of 200).  The loop runs until the time budget is
   almost exhausted.

3. A short simulated‑annealing polish tries tiny random mass moves
   and accepts them with a Metropolis rule.  The annealer is lightweight
   but often yields a small extra improvement.

The whole routine respects the 1000 s budget, prints periodic progress
messages and finally returns a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth gradient of the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, conv: np.ndarray, tau: float,
                 n: int, Lc: int) -> np.ndarray:
    """
    Gradient of 2·n·(1/τ)·log Σ exp(τ·conv) with respect to a.
    Returns a vector of shape (n,).
    """
    # soft‑max over convolution entries
    maxc = conv.max()
    w = np.exp(tau * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # correlation w ⋆ a (full linear correlation)
    W = np.fft.fft(w, n=Lc)
    A = np.fft.fft(a, n=Lc)
    corr = np.fft.ifft(W * np.conj(A)).real
    # keep the linear part (length = (2n‑1)+(n‑1) = 3n‑2)
    corr = corr[: (2 * n - 1) + n - 1]

    # gradient w.r.t. a: 2·n·g,   where g_i = 2·Σ_m w_m a_{m‑i}
    idx = (n - 1) - np.arange(n)          # map i → required slice
    g = 2.0 * corr[idx]                   # the inner “g”
    return 2.0 * n * g                    # factor 2·n from the smoothed objective

    beta2 = 0.999

    lr = 0.20                           # base learning rate in log‑space

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    tau = 5.0
    tau_max = 200.0
    tau_factor = 1.001

    best_a = None
    best_score = float("inf")

    last_print = start_time

        # ----- a = softmax(z) -----
        a = _softmax(z)

        # ----- convolution and true objective -----
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] it={it:,}  score={cur_score:.6f}  τ={tau:.2f}")

        # ----- gradient w.r.t. a (smooth) -----
        grad_a = _smooth_grad(a, conv, tau, n, Lc)

        # ----- chain rule through soft‑max -----
        a_dot_g = np.dot(a, grad_a)
        grad_z = a * (grad_a - a_dot_g)          # ∂F/∂z

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z -= step

        # ----- increase τ (smooth → exact) -----
        if tau < tau_max:
            tau = min(tau * tau_factor, tau_max)

        # optional learning‑rate decay
        if it % 1000 == 0:
            lr *= 0.99

        # occasional status line (every ~10 s)
        if time.time() - last_print > 10.0:
            last_print = time.time()

    return best_a if best_a is not None else a0, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny Metropolis annealer that moves small packets of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # total number of attempted moves is set by remaining time
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # choose a source proportionally to its mass, random destination
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 50 %) of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low value of
    2·n·max(conv(a,a)) / (sum(a)**2).

    The routine builds an initial distribution (re‑using the global
    `height_sequence_1` when present), then refines it with a
    smooth Adam optimiser on the simplex, and finishes with a short
    simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1) Choose a power‑of‑two length (FFT‑friendly)
    # --------------------------------------------------------------
    n = 65536                     # 2**16, good trade‑off between resolution and speed

    # --------------------------------------------------------------
    # 2) Initialise a probability vector
    # --------------------------------------------------------------
    try:
        # use the globally supplied construction if it exists
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # interpolate / up‑sample to the working length
            if starter.size != n:
                x_old = np.linspace(0.0, 1.0, starter.size)
                x_new = np.linspace(0.0, 1.0, n)
                starter = np.interp(x_new, x_old, starter)
            a0 = np.maximum(starter, 0.0)
            if a0.sum() > 0.0:
                a0 /= a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            raise ValueError
    except Exception:
        # fallback – uniform distribution
        a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 3) Smooth Adam optimisation on the simplex (soft‑max)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:          # keep a few seconds for the final polish
        opt_deadline = time.time() + 0.92 * remaining
        best_a, best_score = _smooth_optimize(a0, opt_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] smooth‑opt score ≈ {best_score:.6f}")
    else:
        best_a = a0
        best_score = _objective(best_a)

    # --------------------------------------------------------------
    # 4) Light simulated‑annealing polish (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _anneal(best_a, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] final score ≈ {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0.0:
        best_a /= s
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished, objective ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
