def _smooth_obj_and_grad(u, beta, n, L_fft, L_corr):

    # Compute a via softmax
    max_u = np.max(u)  # for numerical stability
    exp_u = np.exp(u - max_u)  # shape n
    a = exp_u / np.sum(exp_u)

    # compute conv = a * a
    conv = _conv_self(a, L_fft)  # length 2n-1

    # smoothing
    max_c = np.max(conv)
    shifted = conv - max_c
    exp_vals = np.exp(beta * shifted)  # as stable
    s = np.sum(exp_vals)
    smooth_max = max_c + np.log(s) / beta

    obj = 2.0 * n * smooth_max  # denominator sum^2 = 1 because sum a = 1

    # gradient of smooth_max w.r.t. a (grad_a)
    w = exp_vals / s  # shape 2n-1

    # correlation of w with a_rev to get grad_a. This is same as earlier.
    a_rev = a[::-1]
    # Using FFT
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)
    start = n - 1
    grad_a = 2.0 * corr[start:start+n]

    # chain rule to compute grad_u: grad_u = (grad_a - (a·grad_a) * a)
    dot = np.dot(a, grad_a)
    grad_u = grad_a - dot * a

    # gradient of obj = 2*n * grad_u
    grad = 2.0 * n * grad_u
    return obj, grad

    # compute a via softmax
    # shift for numerical stability
    max_u = u.max()

"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

* The sequence is forced onto the probability simplex by representing it as a
  soft‑max (a_i = exp(u_i)/∑exp(u)).
* A smooth approximation of the “max” entry of the self‑convolution is used:
  smooth_maxβ(conv) = (1/β)·log∑_k exp(β·conv_k).  As β→∞ the value tends to the
  true maximum.
* For a given β we minimise the smooth objective
      fβ(u) = 2·n·smooth_maxβ(conv(a(u)))
  using SciPy’s L‑BFGS‑B (gradient is computed analytically via FFT‑based
  correlation).  The optimisation is warm‑started for a sequence of increasing
  β values (annealing), which progressively sharpens the approximation.
* Several restarts are performed (starting from the best known construction,
  the global starter if present, and from noisy perturbations).  The best exact
  score (computed with the original evaluator) is kept.
* A short random hill‑climb finishes the search, moving a small amount of mass
  from indices that contribute to the current maximal convolution entry to
  random destinations, always keeping the vector on the simplex.
* The final vector is normalised (sum = 1) and returned as a plain list of
  Python floats.

The routine respects the 1000 s budget, prints intermediate scores and returns
the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders (used by the smooth‑max objective)
# ----------------------------------------------------------------------
_beta_current = None          # smoothing parameter β
_n_current = None            # problem dimension n
_L_fft_current = None        # FFT length for convolution
_L_corr_current = None       # FFT length for correlation

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact evaluation of the objective for a vector with sum≈1.
    Returns 2·n·max(conv) (the denominator is 1 because sum(a)=1).
    """
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (w.r.t. the logits u)
# ----------------------------------------------------------------------
def _smooth_obj_grad(u: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) for the current β.
    The vector a = softmax(u) lives on the probability simplex.
    """
    global _beta_current, _n_current, _L_fft_current, _L_corr_current

    # a = softmax(u)
    max_u = u.max()
    exp_u = np.exp(u - max_u)
    a = exp_u / exp_u.sum()

    # self‑convolution of a
    conv = _conv_self(a, _L_fft_current)               # length 2n‑1

    # smooth maximum via log‑sum‑exp
    max_c = conv.max()
    shifted = conv - max_c
    exp_vals = np.exp(_beta_current * shifted)
    sum_exp = exp_vals.sum()
    smooth_max = max_c + np.log(sum_exp) / _beta_current

    # objective (the denominator is 1 because ∑a = 1)
    obj = 2.0 * _n_current * smooth_max

    # gradient of smooth_max w.r.t. a (see derivation in the analysis)
    w = exp_vals / sum_exp                              # soft‑max weights, length 2n‑1
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=_L_corr_current)
    A = np.fft.rfft(a_rev, n=_L_corr_current)
    corr = np.fft.irfft(W * A, n=_L_corr_current)
    start = _n_current - 1
    grad_a = 2.0 * corr[start:start + _n_current]       # ∂smooth_max/∂a

    # chain rule: ∂smooth_max/∂u = (∂smooth_max/∂a) · (∂a/∂u)
    dot = np.dot(a, grad_a)                             # scalar
    grad_u = grad_a - dot * a                           # Jacobian of soft‑max
    grad = 2.0 * _n_current * grad_u                     # chain rule for the whole objective
    return obj, grad


# ----------------------------------------------------------------------
# Simple hill‑climb that moves a small amount of mass away from the dominant
# convolution entry.
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, L_fft: int, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Random + targeted mass‑transfer moves.  Only improving moves are kept.
    Returns the best vector found and its exact score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    step_factor = 0.08      # up to 8 % of source mass per move
    min_factor = 1e-6
    decay = 0.985

    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.6:
            # random transfer
            src = rng.integers(n)
            if a[src] < 1e-18:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # targeted transfer (remove from an index that participates in the current max)
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-18:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.08)      # reset step size after success
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improved score={best_score:.6f}")
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score

    and a final hill‑climb polish.

    start_time = time.time()
    deadline = start_time + budget_s - 0.2            # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressivity
    # and speed under the 1000 s budget.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best sequence (uniform, possibly overridden by a global
    # starter if present).
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # Try to use a globally supplied starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                # adjust length if necessary
                if starter.shape[0] != n:
                    if starter.shape[0] > n:
                        starter = starter[:n]
                    else:
                        tmp = np.zeros(n)
                        tmp[: starter.shape[0]] = starter
                        starter = tmp
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Prepare globals used by the smooth objective
    # --------------------------------------------------------------
    global _beta_current, _n_current, _L_fft_current, _L_corr_current
    _n_current = n
    _L_fft_current = L_fft
    _L_corr_current = L_corr

    # Annealing schedule for the smoothing parameter β
    beta_schedule = [5, 20, 80, 300, 800, 2000]

    # --------------------------------------------------------------
    # Multiple restarts of L‑BFGS‑B (each with the full annealing schedule)
    # --------------------------------------------------------------
    max_restarts = 5
    for restart in range(max_restarts):
        if time.time() >= deadline - 5:
            break

        # Start from the current best sequence plus a modest random perturbation
        eps = 1e-14
        u0 = np.log(np.maximum(best_seq, eps))
        rng = np.random.default_rng(seed + restart + 12345)
        u0 = u0 + 0.3 * rng.normal(size=n)

        for beta in beta_schedule:
            if time.time() >= deadline - 5:
                break
            _beta_current = float(beta)

            # L‑BFGS‑B options – we rely on the internal convergence test,
            # but cap the number of iterations to keep the overall run time bounded.
            opts = {
                "maxiter": 2500,
                "disp": False,
                "gtol": 1e-12,
                "ftol": 1e-12,
            }

            # Run the optimisation
            try:
                res = minimize(_smooth_obj_grad,
                               u0,
                               method="L-BFGS-B",
                               jac=True,
                               options=opts)
            except Exception as exc:
                print(f"[{time.time() - start_time:.2f}s] L‑BFGS failed (β={beta}): {exc}")
                break

            u_opt = res.x
            u0 = u_opt               # warm‑start the next β‑stage

            # Convert back to a probability vector
            max_u = u_opt.max()
            exp_u = np.exp(u_opt - max_u)
            a_candidate = exp_u / exp_u.sum()

            # Exact evaluation (no smoothing)
            sc = _eval_fast(a_candidate, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start_time:.2f}s] restart {restart} β={beta} improved score={sc:.6f}")

    # --------------------------------------------------------------
    # Final hill‑climb polish (very short, uses remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 1:
        a_opt, sc_opt = _hill_climb(best_seq, L_fft, deadline, start_time)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}")

    # --------------------------------------------------------------
    # Normalise (just in case) and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Optional sanity check with the official evaluator
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
