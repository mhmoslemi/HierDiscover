def _subgrad_descent(a0, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    iteration = 0
    rng = np.random.default_rng()
    step = 0.5
    while time.time() < deadline and iteration < max_iters:
        iteration += 1
        # compute conv and max index
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())
        # compute gradient g
        g = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        # g[i] = 2*a[k_max - i] for i in [i_low,i_high]
        # using vectorized assignment
        idx = np.arange(i_low, i_high + 1)
        g[idx] = 2 * a[k_max - idx]
        # take subgradient step with step size
        a_new = a - step * g
        a_new = _project_simplex(a_new)

        new_score = _eval_fast(a_new, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad iter={iteration} step={step:.4f} score={new_score:.6f}")
            # maybe increase step a bit after improvement
            step = min(step * 1.1, 1.0)
        else:
            # reduce step size
            step *= 0.9
        a = a_new
    return best_a, best_score
```

"""
Hybrid optimizer for the “max‑convolution” step‑function problem.

The strategy is a multi‑stage pipeline that respects the 1000 s budget:

1️⃣  Coarse parametric sweep – explores families a_i ∝ r^i·(i+1)^p (mixed with a
    uniform vector) and keeps the best candidate.

2️⃣  Several short Adam‑style projected‑gradient descents on a smooth‑max
    (log‑sum‑exp) surrogate of max(conv(a,a)).  The temperature β is raised
    gradually up to 800, giving an increasingly accurate approximation of the
    true max.

3️⃣  Targeted stochastic hill‑climb – random mass‑transfer moves enriched by
    “max‑entry attacks’’ that move mass away from the pair of indices that
    currently realise the maximal convolution entry.

4️⃣  Fine‑grained sub‑gradient descent – the true objective is  max(conv).  The
    sub‑gradient for a given peak index k is g_i = 2·a_{k‑i} (zero when the
    index is out of range).  We perform a projected sub‑gradient step with an
    adaptive step‑size and line‑search‑style back‑off, which is very cheap
    (one FFT per iteration) and capable of shaving a few‑thousandths off the
    final score.

All iterates are projected onto the probability simplex to keep the sum
exactly one and enforce non‑negativity.  The routine prints intermediate
scores and returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max surrogate utilities (used by Adam descent)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    conv = _conv_self(a, L=L_conv)          # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                 # shift for numerical stability
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()          # soft‑max weights, shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)

    start = a.shape[0] - 1                 # index aligning convolution centre
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 800.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type projected descent on the smooth‑max surrogate.
    β grows from a small value to beta_max, turning the surrogate gradually
    into the true max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0               # start fairly smooth
    beta_mul = 1.02          # geometric increase per iteration

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} score={cur_score:.6f}")

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.5 / (1.0 + 1e-4 * t)      # slowly decaying learning‑rate
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (random + max‑entry attacks)
# ----------------------------------------------------------------------
def _targeted_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers (up to ~15 % of the source mass) plus
    targeted moves that directly reduce the currently dominant convolution entry.
    Only strictly improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    step_factor = 0.15
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        if rng.random() < 0.6:
            # random transfer
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # targeted transfer
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)  # reset after success
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score

# Coarse‑to‑fine parametric sweep
# ----------------------------------------------------------------------
def _parametric_sweep(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore the family a_i ∝ r^i·(i+1)^p (mixed with uniform) and keep the
    best vector seen.  The sweep stops early if the remaining time is low.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)      # i+1
    # A slightly broader grid than before – still cheap enough.
    r_grid = np.logspace(np.log10(0.85), np.log10(0.9999), num=35)
    p_grid = np.linspace(-4.0, 3.0, 51)

    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 120:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

def _subgradient_descent(

    """

    docstring and respects the 1000 s (≈16 min) runtime budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # Problem size – 4096 offers a good trade‑off between quality and speed.
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0.5️⃣  Global starter (if the competition provides one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep (reserve ~300 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        best_seq, best_score = _parametric_sweep(
            n, uniform, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style smooth‑max descent (two diverse seeds)
    # ------------------------------------------------------------------
    descent_seeds = [seed, seed + 1]
    for ds in descent_seeds:
        if time.time() >= deadline - 150:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        a0 += 1e-4 * rng.standard_normal(n) * a0   # tiny random perturbation
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 15000
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,          # leave a bit of slack for later stages
            start_time,
            best_score,
            beta_max=800.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth descent improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Targeted hill‑climb (random + max‑entry moves)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fine sub‑gradient refinement (last ~60 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        a_opt, sc_opt = _subgradient_descent(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient refinement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and sanity check
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # optional verification with the official evaluator
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
