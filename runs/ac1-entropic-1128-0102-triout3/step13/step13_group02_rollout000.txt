    a = a0.copy()
    cur_score = best_score
    best_a = a.copy()

    # SA parameters

    it = 0

    while time.time() < deadline - 0.1:
        it += 1
        # temperature schedule: exponential cooling
        t_frac = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1 - t_frac) + Tf * t_frac # linear cooling
        step_frac = step_max * (1 - t_frac) + 0.001 * t_frac
        # pick src + target
        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1
        # epsilon:
        eps = a[src] * rng.random() * step_frac
        a_candidate = a.copy()
        a_candidate[src] -= eps
        a_candidate[dst] += eps
        # maintain simplex (sum unchanged)
        # compute objective
        score_candidate = _objective(a_candidate, L_fft)
        delta = score_candidate - cur_score

                best_a = a.copy()

"""
Propose a low‑score step function.

The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a)**2)

Because the expression is homogeneous we may (and do) normalise the
vector to sum = 1, i.e. we minimise `2·n·max(conv)`.

The search consists of the following stages (all respecting the 1000 s
budget).

1️⃣  **Rich parametric sweep** – families of monotone, symmetric and
    mixed shapes (geometric‑power, pure power, symmetric Beta‑type).
    The best candidate of this cheap sweep seeds the later phases.

2️⃣  **Projected smooth‑max Adam descent** – gradient descent on a
    soft‑max approximation of the max‑convolution, with an exponential
    temperature schedule.  Several random restarts improve robustness.

3️⃣  **Exact sub‑gradient descent** – descent on the true max‑convolution
    using the analytic sub‑gradient.  A diminishing step size and a tiny
    simulated‑annealing acceptance rule help to escape shallow plateaus.

4️⃣  **Simulated annealing (global move search)** – random mass transfers
    between two coordinates, accepting worsening moves with a temperature
    that cools linearly.  This phase can break out of the basin found by
    the local methods.

5️⃣  **Final greedy hill‑climb** – tiny deterministic transfers that only
    accept improvements, polishing the solution.

All helper functions are top‑level, there are no closures or lambdas,
and the code works with a single CPU.  The returned list is a plain
Python list of floats; the external `evaluate_sequence` will be used to
score the answer.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0, sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective for a vector with sum≈1:
        2·n·max(conv(a,a))
    """
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * a.shape[0] * conv.max())

def _parametric_sweep(n: int, deadline: float, start_time: float,
                      best_seq: np.ndarray, best_score: float):
    """
    Scan several one‑parameter families (geometric‑power, power,
    symmetric Beta‑type) possibly mixed with the uniform distribution.
    Returns an improved (seq, score) pair if any improvement is found.
    """
    L_fft = _next_pow2(2 * n - 1)
    uniform = np.full(n, 1.0 / n)

    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based index
    rev_idx = n - np.arange(n) + 1                       # reversed

    # --- geometric * power family (already present in the baseline) ---
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 200:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 200:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _objective(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep geo r={r:.4f} "
                          f"p={p:.2f} λ={lam:.2f} → {sc:.6f}")

    # --- pure power families (monotone) -----------------------------------
    for p in np.linspace(-4.0, 4.0, 61):
        if time.time() >= deadline - 200:
            break
        shape = np.power(idx, p)
        shape /= shape.sum()
        for lam in (0.0, 0.5, 1.0):
            cand = lam * uniform + (1.0 - lam) * shape
            cand /= cand.sum()
            sc = _objective(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep power p={p:.2f} "
                      f"λ={lam:.2f} → {sc:.6f}")

    # --- symmetric Beta‑type families (x^{α‑1}·(1‑x)^{α‑1}) ---------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # centred points in (0,1)
    for alpha in np.linspace(0.2, 5.0, 46):
        if time.time() >= deadline - 200:
            break
        shape = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
        shape_sum = shape.sum()
        if shape_sum == 0.0:
            continue
        shape /= shape_sum
        for lam in (0.0, 0.5, 1.0):
            cand = lam * uniform + (1.0 - lam) * shape
            cand /= cand.sum()
            sc = _objective(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep beta α={alpha:.2f} "
                      f"λ={lam:.2f} → {sc:.6f}")

    return best_seq, best_score

def _smooth_descent(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, best_score_initial: float):
    """
    Adam‑style projected descent on the soft‑max surrogate of max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0                # start temperature (soft‑max sharpness)
    beta_mul = 1.02
    beta_max = 500.0

    while time.time() < deadline:
        t += 1

        # soft‑max of convolution values
        conv = _conv_self_fft(a, L_conv)
        max_c = conv.max()
        shifted = conv - max_c                # stabilise exponentials
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()         # soft‑max weights, length 2n‑1

        # gradient of the soft‑max term (via correlation)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)  # length ≥ 3n‑2
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # monitor exact objective
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} "
                  f"score={cur_score:.6f}")

        # Adam update + Euclidean projection onto simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.12
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for shift k (O(n))
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

        # diminishing step
        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _objective(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional probabilistic acceptance
            prob = np.exp(-(cur_score - best_score) * 15.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter every few thousand iterations
        if it % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _simulated_annealing(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """
    Classical SA: move a random ε‑fraction of mass from one coordinate to
    another, accept worsening moves with a linearly cooled temperature.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    cur_score = best_score_initial
    best_a = a.copy()
    best = cur_score
    rng = np.random.default_rng()

    # temperature and step schedules (linear in wall‑clock progress)
    T_start = 0.02 * cur_score          # relative to objective magnitude
    T_end = 0.0002 * cur_score
    step_start = 0.20                   # up to 20 % of source mass
    step_end = 0.001                    # final granularity

    while time.time() < deadline - 0.1:
        # progress fraction (0 → 1)
        prog = (time.time() - start_time) / (deadline - start_time)
        T = T_start * (1.0 - prog) + T_end * prog
        step_frac = step_start * (1.0 - prog) + step_end * prog

        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        eps = a[src] * rng.random() * step_frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cand_score = _objective(a_prop, L_fft)
        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_prop
            cur_score = cand_score
            if cur_score < best:
                best = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={int(prog*1e6):,} "
                      f"score={best:.6f}")

    return best_a, best
