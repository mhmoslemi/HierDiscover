def _simulated_annealing(a, best_score, deadline):
    n = len(a)
    rng = np.random.default_rng()
    cur_a = a.copy()
    cur_score = best_score
    T0 = 0.01 # initial temperature relative to score
    # schedule: T = T0 * (1 - (t/t_max)) 
    t_start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - t_start
        T = max(1e-8, T0 * (1 - elapsed / (deadline - t_start)))
        # random move
        src = rng.integers(n)
        if cur_a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst: continue
        # choose epsilon as fraction of src mass, maybe random uniform upto 0.01
        frac = rng.random() * 0.01 + 1e-6
        eps = cur_a[src] * frac
        if eps <= 0: continue
        a_new = cur_a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # compute conv max for new a
        new_score = _objective_score(a_new)
        # accept if improvement or with prob
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / (T + 1e-12)):
            cur_a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time()-start_time:.2f}s] Anneal best score={best_score:.6f}")
        # else reject
    return best_seq, best_score

"""
Optimiser for the “step‑function” challenge (alternative to the reference).

Approach
--------
1️⃣  **Extended Beta sweep** – explore a much larger (α,β) grid (up to 12,
    logarithmic spacing) to obtain a strong initial candidate.

2️⃣  **Block‑wise evolutionary search** – treat the vector as 256 constant
    blocks.  A small population of block‑vectors is evolved with log‑normal
    mutations (a simple CMA‑ES‑style scheme).  After each mutation the
    blocks are expanded to full length and the objective
    2·n·max(conv) is evaluated via FFT.

3️⃣  **Simulated annealing on the full sequence** – starting from the best
    block‑solution we perform mass‑transfer moves (source→destination) that
    are accepted either if they improve the score or with a Metropolis
    probability that slowly cools down.  This helps escape shallow local
    minima.

4️⃣  **Directed hill‑climb** – a final fine‑grained sub‑gradient move
    (identical to the reference) to polish the result.

All stages respect the global time budget (default 1000 s) and use fast
FFT‑based convolution.  The final sequence is normalised to sum = 1 and
returned as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helper (fast, scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used in final hill‑climb)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Stage 1 – extended beta sweep
# ----------------------------------------------------------------------
def _beta_sweep(n: int, best_seq: np.ndarray, best_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))

    # α,β grids (log‑linear for small values, linear for larger)
    alphas = np.concatenate((
        np.linspace(0.02, 0.5, 30, endpoint=False),
        np.linspace(0.5, 4.0, 40, endpoint=False),
        np.linspace(4.0, 12.0, 30)
    ))
    betas = alphas.copy()

    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            # stabilize
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.3f} β={b:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – block‑wise evolutionary search (log‑normal mutations)
# ----------------------------------------------------------------------
def _expand_blocks(blocks: np.ndarray, n: int) -> np.ndarray:
    """Repeat each block value to obtain a full‑length sequence."""
    m = blocks.shape[0]
    block_len = n // m
    seq = np.repeat(blocks, block_len)
    if seq.shape[0] < n:
        # pad the remainder with zeros (tiny mass will be redistributed later)
        seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
    return seq[:n]


def _mutate_log_normal(w: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal mutation preserving the simplex."""
    pert = np.exp(sigma * rng.normal(size=w.shape))
    w_new = w * pert
    w_new = np.maximum(w_new, 0.0)
    w_new /= w_new.sum()
    return w_new


def _evolutionary_search(n: int, m: int,
                         init_seq: np.ndarray, init_score: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple CMA‑ES‑like evolution on block weights.
    n – full length, m – number of blocks (e.g. 256).
    Returns the best full‑length sequence found.
    """
    block_len = n // m
    # initialise block weights from the current best full‑vector
    block_weights = np.array(
        [init_seq[i * block_len:(i + 1) * block_len].mean()
         for i in range(m)], dtype=np.float64
    )
    block_weights = np.maximum(block_weights, 1e-12)
    block_weights /= block_weights.sum()

    # population parameters
    pop_size = 30
    elite = 5
    sigma = 0.12               # mutation strength (log‑normal)
    rng = np.random.default_rng()

    # create initial population (block vectors)
    pop = [block_weights]
    for _ in range(pop_size - 1):
        w = rng.random(m)
        w = np.maximum(w, 1e-12)
        w /= w.sum()
        pop.append(w)

    best_seq = init_seq.copy()
    best_score = init_score

    while time.time() < deadline:
        # evaluate
        scores = [_objective_score(_expand_blocks(w, n)) for w in pop]
        # keep elite
        elite_idx = np.argsort(scores)[:elite]
        elite_pop = [pop[i] for i in elite_idx]
        elite_scores = [scores[i] for i in elite_idx]

        # update global best
        if elite_scores[0] < best_score:
            best_score = elite_scores[0]
            best_seq = _expand_blocks(elite_pop[0], n).copy()
            print(f"[{time.time() - start_time:.2f}s] Evol best score={best_score:.6f}")

        # generate new population
        new_pop = elite_pop.copy()
        while len(new_pop) < pop_size:
            parent = elite_pop[rng.integers(elite)]
            child = _mutate_log_normal(parent, sigma, rng)
            new_pop.append(child)
        pop = new_pop

        # slowly decay mutation strength
        sigma *= 0.9995
        # safety: ensure we still have time for next generation
        if time.time() + 0.05 * pop_size > deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – simulated annealing on the full vector
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_score: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing (Metropolis acceptance)."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = cur_score

    # Annealing schedule – linear cooling
    t0 = time.time()
    total = max(1e-12, deadline - t0)
    T0 = 1e-2 * cur_score          # initial temperature (1 % of current score)

    while time.time() < deadline:
        elapsed = time.time() - t0
        T = max(1e-12, T0 * (1.0 - elapsed / total))

        # propose a move: transfer a fraction from src to dst
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        frac = rng.random() * 0.008 + 1e-6      # up‑to ~0.8 % of source mass
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        prop_score = _objective_score(a_prop)

        # Metropolis acceptance
        accept = prop_score < cur_score or rng.random() < np.exp((cur_score - prop_score) / T)
        if accept:
            a = a_prop
            cur_score = prop_score
            if prop_score < best_score:
                best_score = prop_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal best score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 4 – final directed hill‑climb (sub‑gradient based)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, start_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb moving mass from positive to negative sub‑gradient."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = start_score
    L = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a, L)
        grad = _subgradient_maxconv(a, conv)

        # source = index with largest positive gradient (and positive mass)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)                     # most negative gradient

        # tiny mass transfer
        frac = rng.random() * 0.005 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        prop_score = _objective_score(a_prop)
        if prop_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = prop_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The optimiser runs four stages described above
    and respects the allocated wall‑clock budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096                         # same length used by the reference run
    uniform = np.full(n, 1.0 / n)    # baseline
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Stage 1 – extended beta sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 600.0    # leave ~10 min for later stages
    if time.time() < phase_deadline:
        best_seq, best_score = _beta_sweep(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # Stage 2 – block‑wise evolutionary search (256 blocks)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0    # reserve ~5 min for SA + hill‑climb
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting block‑wise evolution")
        best_seq, best_score = _evolutionary_search(
            n, 256, best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After evolution score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – simulated annealing on the full vector
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0    # keep ~2 min for final hill‑climb
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After annealing score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – final directed hill‑climb (sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
This implementation explores a broader family of candidate shapes, adds a low‑dimensional evolutionary optimiser, and finishes with a simulated‑annealing‑plus‑hill‑climb refinement. It respects the 1000 s time limit, works entirely on the simplex, and is designed to push the bound below the target 1.503.
