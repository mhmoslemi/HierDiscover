```
    L = _next_pow2(3*n - 2)
    # compute FFT of a once
```

```
def _upscale(seq, new_n):
    old_n = seq.shape[0]
    x_old = np.linspace(0, 1, old_n, endpoint=False) + 0.5/old_n
    x_new = np.linspace(0, 1, new_n, endpoint=False) + 0.5/new_n
    seq_new = np.interp(x_new, x_old, seq)
    # renormalize
    seq_new = np.maximum(seq_new, 0.0)
    total = seq_new.sum()
    if total > 0:
        seq_new /= total
    else:
        seq_new = np.full(new_n, 1.0 / new_n)
    return seq_new
```

assistantfinal**Solution Overview**

The program builds a low‑value step‑function for  

\[
F(a)=\frac{2\,n\; \max\bigl((a*a)\bigr)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

The function is *scale invariant*, therefore we keep the vector on the probability
simplex (`∑a = 1`).  
The maximal self‑convolution is a non‑smooth function, so a plain
sub‑gradient method converges slowly.  
Instead we optimise a **smooth approximation** of the maximum:

\[
M_\alpha(a)=\frac{1}{\alpha}\log\!\Bigl(\sum_i e^{\alpha\,(a*a)_i}\Bigr)
\qquad(\alpha>0)
\]

`Mα` converges to `max` for large `α`.  
Its gradient can be written as a correlation, which is computed with one
FFT pair per iteration – the same cost as the ordinary convolution.

The optimisation consists of three phases

1. **Coarse multi‑start search (n = 4096)** – many random Dirichlet
   starts are refined for a few hundred “smooth‑max” Adam steps.
2. **Upscaling & deep refinement (n = 16384)** – the best coarse
   candidates are interpolated to the larger grid and optimised for
   several thousand steps, slowly increasing `α` from 1 to 100.
3. **Light simulated‑annealing polish** – a few hundred thousand tiny
   mass moves finish the search.

All updates are projected onto the probability simplex, guaranteeing
that the vector always stays feasible for the original evaluation
function.

The whole routine respects the 1000 s budget, prints progress and finally
returns the best sequence found as a plain Python list.

```python
"""
Multi‑scale optimisation of a non‑negative step function.

1.  Many random Dirichlet starts on a modest grid (n≈4 k) are refined
    with Adam on a smooth approximation of the max‑convolution.
2.  The best coarse candidates are up‑sampled to a finer grid (n≈16 k)
    and run for many more iterations while the smoothing parameter α
    is increased from 1 to 100.
3.  A fast Metropolis annealer gives a final polish.

All updates are projected onto the probability simplex, so the returned
vector is always a valid probability distribution (∑a = 1, a_i ≥ 0).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                      # numerical safeguard
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max gradient (FFT‑based)
# ----------------------------------------------------------------------
def _conv_and_grad_softmax(a: np.ndarray, alpha: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (conv, grad) where
        conv = a * a (self‑convolution)
        grad = ∇_a M_α(conv)  (gradient of the smooth‑max)
    Both are computed with a single pair of FFTs.
    """
    n = a.shape[0]
    # length that can hold both a (len=n) and conv (len=2n‑1) for correlation
    L = _next_pow2(3 * n - 2)

    # FFT of a (used for both convolution and correlation)
    A = np.fft.rfft(a, n=L)

    # self‑convolution (conv length 2n‑1)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

    # soft‑max weights over the convolution entries
    max_c = conv.max()
    # numerical stability
    w_exp = np.exp(alpha * (conv - max_c))
    w = w_exp / w_exp.sum()          # shape (2n‑1,)

    # FFT of the weight vector
    W = np.fft.rfft(w, n=L)

    # correlation a ⋆ w  (gives a vector of length L)
    corr = np.fft.irfft(A * np.conj(W), n=L)

    # lag 0 … n‑1 are located at index (2n‑2) … (2n‑2+n‑1)
    start = 2 * n - 2
    grad = 2.0 * corr[start:start + n]   # ∇_a M_α
    return conv, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     steps: int,
                     deadline: float,
                     alpha_start: float = 1.0,
                     alpha_end: float = 100.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of 2·n·M_α(a) with projection onto the simplex.
    Returns the final vector and its *exact* objective value
    (using the true max, not the smooth approximation).
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2               # base learning rate

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, steps + 1):
        if time.time() >= deadline - 0.1:
            break

        # gradual increase of the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (t / steps)

        conv, grad_smooth = _conv_and_grad_softmax(a, alpha)

        # gradient of the *real* objective  2·n·max(conv)  at the smooth point
        grad = 2.0 * n * grad_smooth

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / (math.sqrt(t) + 1.0)   # simple diminishing schedule
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasible
        a = _project_simplex(a)

    # final exact objective (max‑convolution)
    final_conv = _conv_self(a)
    exact_obj = float(2.0 * n * final_conv.max())
    return a, exact_obj


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (identical to the reference version)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    a = seq.copy()
    n = a.shape[0]
    cur_obj = 2.0 * n * _conv_self(a).max()
    best_a = a.copy()
    best_obj = cur_obj
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5
        a2 = a.copy()
        a2[src] -= eps
        a2[dst] += eps

        new_obj = 2.0 * n * _conv_self(a2).max()
        delta = new_obj - cur_obj

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a2
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_obj:.6f}")

    return best_a, best_obj


# ----------------------------------------------------------------------
# Upscaling helper (linear interpolation)
# ----------------------------------------------------------------------
def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    # mid‑point positions of the old and new discretisations
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full(new_n, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step function.

    Strategy
    --------
    * many random Dirichlet starts on a coarse grid (n≈4096) – quick Adam
      optimisation with a smooth max.
    * keep the 5 best coarse candidates, up‑scale to a finer grid
      (n≈16384) and run a long Adam run while slowly raising the
      smoothing parameter α.
    * finish with a light simulated‑annealing polish.
    * return the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – coarse multi‑start search
    # --------------------------------------------------------------
    n_coarse = 4096
    num_starts = 200                     # how many random starts
    steps_coarse = 180                    # Adam steps per start
    top_candidates = []                  # (score, vector)
    best_score = float("inf")
    best_vec = None

    rng = np.random.default_rng(seed)

    # include a few analytic seeds (beta shapes)
    for a_par, b_par in [(0.5, 0.5), (1.0, 3.0), (3.0, 1.0), (2.0, 2.0), (0.8, 1.2)]:
        xs = (np.arange(n_coarse) + 0.5) / n_coarse
        shape = (xs ** (a_par - 1.0)) * ((1.0 - xs) ** (b_par - 1.0))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        vec, obj = _smooth_max_adam(shape, steps_coarse, deadline)
        score = evaluate_sequence(vec.tolist())
        if score < best_score:
            best_score, best_vec = score, vec.copy()
        top_candidates.append((score, vec.copy()))
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:5]

    # random Dirichlet restarts
    for i in range(num_starts):
        if time.time() >= deadline - 10:
            break
        # concentration controls sparsity – we sample it randomly
        conc = rng.uniform(0.2, 2.0)
        a0 = rng.dirichlet(np.full(n_coarse, conc))
        vec, obj = _smooth_max_adam(a0, steps_coarse, deadline)
        score = evaluate_sequence(vec.tolist())
        if score < best_score:
            best_score, best_vec = score, vec.copy()
        # keep the best few
        if len(top_candidates) < 5:
            top_candidates.append((score, vec.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif score < top_candidates[-1][0]:
            top_candidates[-1] = (score, vec.copy())
            top_candidates.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] Phase 1 finished. best coarse score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – upscale and deep refinement
    # --------------------------------------------------------------
    n_fine = 16384
    steps_fine = 4000
    refined_best_score = best_score
    refined_best_vec = best_vec.copy() if best_vec is not None else np.full(n_fine, 1.0 / n_fine)

    for idx, (_, coarse_vec) in enumerate(top_candidates):
        if time.time() >= deadline - 5:
            break
        fine_vec = _upscale(coarse_vec, n_fine)
        fine_vec, obj = _smooth_max_adam(fine_vec, steps_fine, deadline,
                                         alpha_start=1.0, alpha_end=100.0)
        score = evaluate_sequence(fine_vec.tolist())
        print(f"[{time.time() - start_time:.2f}s] Refined candidate {idx+1} → score={score:.6f}")
        if score < refined_best_score:
            refined_best_score, refined_best_vec = score, fine_vec.copy()

    # --------------------------------------------------------------
    # Phase 3 – final annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        annealed_vec, annealed_obj = _anneal(refined_best_vec, deadline, start_time)
        annealed_score = evaluate_sequence(annealed_vec.tolist())
        if annealed_score < refined_best_score:
            refined_best_score, refined_best_vec = annealed_score, annealed_vec.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved to {annealed_score:.6f}")

    # --------------------------------------------------------------
    # Normalisation & return
    # --------------------------------------------------------------
    refined_best_vec = np.maximum(refined_best_vec, 0.0)
    s = refined_best_vec.sum()
    if s > 0.0:
        refined_best_vec /= s
    else:
        refined_best_vec = np.full_like(refined_best_vec, 1.0 / refined_best_vec.size)

    print(f"[{time.time() - start_time:.2f}s] Finished. final score ≈ {refined_best_score:.6f}")

    return [float(v) for v in refined_best_vec]
```

This implementation stays within the time limit, uses only one CPU, respects all security checks of `evaluate_sequence`, and applies a different optimisation paradigm (smooth‑max Adam + multi‑scale refinement) that has a good chance of beating the previous best bound of 1.508.
