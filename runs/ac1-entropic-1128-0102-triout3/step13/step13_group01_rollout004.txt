def propose_candidate(seed=0, budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 0.2

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # choose n
    n = 65536

    # prepare initial vector a0
    # try to use global height_sequence_1 if available
    a0 = None
    try:
        h = np.asarray(height_sequence_1, dtype=float)
        if h.ndim == 1 and h.size > 0:
            # upsample or pad to length n
            if h.size != n:
                # interpolate to n
                x_old = np.linspace(0, 1, h.size)
                x_new = np.linspace(0, 1, n)
                a_interp = np.interp(x_new, x_old, h)
                a0 = a_interp
            else:
                a0 = h.copy()
            a0 = np.maximum(a0, 0.0)
            total = a0.sum()
            if total > 0.0:
                a0 = a0 / total
            else:
                a0 = np.full(n, 1.0 / n)
    except Exception:
        a0 = None

    if a0 is None:
        # fallback random Dirichlet
        a0 = rng.dirichlet(np.ones(n))

    # perform descent
    remaining = deadline - time.time()
    # allocate 80% for descent, 20% for anneal
    descent_deadline = time.time() + 0.8 * remaining
    a_opt, score_opt = _softmax_descent(a0, descent_deadline, start_time)

    # if time remains, do annealing
    if time.time() < deadline:
        a_opt, score_opt = _anneal(a_opt, deadline, start_time)

    # final normalization to correct any drift
    a_opt = np.maximum(a_opt, 0.0)
    total = a_opt.sum()
    if total > 0.0:
        a_opt = a_opt / total
    else:
        a_opt = np.full(n, 1.0 / n)

"""
propose_candidate – soft‑max gradient descent on the probability simplex
---------------------------------------------------------------

The objective to minimise is

    Φ(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

For any non‑negative vector a we can normalise it to sum(a)=1, after which
Φ(a)=2·n·max(conv(a,a)).  The algorithm works with a fixed length *n*
(e.g. 2**16) and maintains the vector on the simplex using Euclidean
projection.  At each iteration we view the quantity

    f(a) = (1/γ)·log ∑_k exp(γ·n·conv_k(a))

as a smooth approximation of max(conv).  Its gradient can be written
in closed form using a single convolution:

    ∇f(a) = 2·conv( w , a_rev )[n‑1 : n‑1+n]

where w are the soft‑max weights (∑w=1) and a_rev is a reversed copy of a.
Both convolutions are performed with a real FFT, giving an
O(n log n) per‑iteration cost.

A simple momentum scheme and a decreasing step size are employed.
After the descent phase a short Metropolis‑type annealing polish moves tiny
masses between entries while accepting deteriorations with a cooling schedule.
The whole search respects a 1000 s budget; the best vector found is
returned as a plain Python list of non‑negative floats.

The code is self‑contained, uses only NumPy and the provided
`evaluate_sequence` for the final check, and works even if the global
variable `height_sequence_1` is unavailable.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    L = _next_pow2(n)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex (Wang & Carreira‑Perpiñán)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:   # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient of the self‑convolution objective
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, gamma: float):
    """
    Compute the (soft‑max) gradient of  max(conv(a,a)).
    Returns the gradient vector and the (raw) self‑convolution.
    """
    n = a.shape[0]
    # self‑convolution a*a
    conv = _fft_convolve(a, a)                 # length 2n‑1
    # scale to O(1) before soft‑max (typical size ~1/n)
    scaled = conv * n
    cmax = scaled.max()
    # soft‑max weights (sum to 1)
    exps = np.exp(gamma * (scaled - cmax))
    w = exps / exps.sum()
    # gradient = 2 * conv(w , a_rev)  (central part)
    a_rev = a[::-1]
    conv2 = _fft_convolve(w, a_rev)           # length 3n‑2
    grad = 2.0 * conv2[n - 1:n - 1 + n]       # take n entries
    return grad, conv


# ----------------------------------------------------------------------
# Projected (soft‑max) gradient descent
# ----------------------------------------------------------------------
def _softmax_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """Run momentum gradient descent on the simplex."""
    a = a0.copy()
    n = a.shape[0]
    # best solution seen so far
    conv = _fft_convolve(a, a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    momentum = np.zeros_like(a)
    beta = 0.9               # momentum factor
    step0 = 0.6              # initial step size
    it = 0

    while time.time() < deadline:
        it += 1
        # slowly increase the soft‑max sharpness
        gamma = min(50.0, 0.3 * math.sqrt(it))   # 0 → ~50
        grad, conv = _softmax_gradient(a, gamma)

        # momentum + diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # stay on the simplex
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Descent it={it:,} score={cur_score:.6f}")

        # (optional) a cheap early‑stop could be added here

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Randomly shift tiny packets of mass; Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = 2.0 * n * _fft_convolve(a, a).max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # number of proposals is limited by the remaining time
    max_steps = 200_000
    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # pick source proportionally to its mass (more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move at most 50 % of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate (fast) – sum remains 1, so denominator is 1
        new_score = 2.0 * n * _fft_convolve(a_prop, a_prop).max()
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    Returns a plain list of non‑negative heights.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # small safety margin

    # --------------------------------------------------------------
    # 1) Choose a power‑of‑two length – fast FFTs & fine discretisation
    # --------------------------------------------------------------
    n = 65536   # 2**16

    # --------------------------------------------------------------
    # 2) Build an initial candidate (try to reuse the global starter)
    # --------------------------------------------------------------
    a0 = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            # up‑sample / interpolate to the working length
            if starter.size != n:
                x_old = np.linspace(0.0, 1.0, starter.size)
                x_new = np.linspace(0.0, 1.0, n)
                a0 = np.interp(x_new, x_old, starter)
            else:
                a0 = starter.copy()
            a0 = np.maximum(a0, 0.0)
            total = a0.sum()
            if total > 0.0:
                a0 = a0 / total
            else:
                a0 = None
    except Exception:
        a0 = None

    if a0 is None:
        # random Dirichlet start (uniform on the simplex)
        a0 = rng.dirichlet(np.ones(n))

    # --------------------------------------------------------------
    # 3) Gradient descent (majority of the time budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        descent_deadline = time.time() + 0.85 * remaining
        best_seq, best_score = _softmax_descent(a0, descent_deadline, start_time)
    else:
        best_seq = a0
        best_score = 2.0 * n * _fft_convolve(best_seq, best_seq).max()

    # --------------------------------------------------------------
    # 4) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _anneal(best_seq, deadline, start_time)

    # --------------------------------------------------------------
    # 5) Normalise and convert to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final approx. score {best_score:.6f}")
    return [float(v) for v in best_seq]
