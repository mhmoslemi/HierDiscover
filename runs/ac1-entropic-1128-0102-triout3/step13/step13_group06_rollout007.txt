### Projection onto simplex

Define function project_onto_simplex(v) returning projection onto {x>=0, sum=1}. Standard algorithm:

```
def project_simplex(v):
    # v is array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑objective non‑negative step‑function.

Algorithm
---------

1.  **Problem size** – Use the length of the previously found vector
    (`height_sequence_1`) if present, otherwise fall back to a larger
    default (`n = 8192`).  Larger n gives a lower theoretical bound
    (≈ 1) and still fits easily into the time budget.

2.  **Warm‑start** – Generate a collection of analytic shapes
    (power‑tents, exponentials, power‑law, Gaussian, U‑shaped, etc.)
    together with a few random Dirichlet draws.  All candidates are
    normalised to sum = 1 and the best one (lowest exact objective) is
    kept.

3.  **Biased hill‑climbing** – Repeatedly move a small packet of mass
    from a *source* index to a *destination* index, updating the
    self‑convolution in‑place (`_apply_move`).  The source is drawn from
    a distribution proportional to its contribution to the current
    maximal convolution entry, i.e. `a[i]·a[k_max‑i]`.  This directly
    attacks the current bottleneck.  A simple simulated‑annealing
    acceptance rule is added (temperature linearly decreasing) so that
    occasional uphill moves are possible early on.

    The move amount `eps` is a random fraction of the source mass,
    shrinking from ≈ 30 % to ≈ 2 % as the deadline approaches.

4.  **Record the best vector** – Whenever a move yields a strictly
    better objective the current vector is saved.

5.  **Return** – The best vector is re‑normalised (sum = 1), clipped to
    non‑negative values and returned as a plain Python list ready for
    `evaluate_sequence`.

All heavy work is done with NumPy vectorised operations; the
convolution is maintained incrementally, so each iteration costs only
`O(n)` work and many millions of moves fit comfortably into the 1000 s
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum≈1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    max_b = conv.max()
    # sum(a) == 1 after normalisation
    return float(2.0 * n * max_b)


def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`.

    The convolution `conv = a*a` (full length) is updated in‑place.
    """
    n = a.shape[0]

    # linear part: 2 * eps * a shifted by src (subtraction) and dst (addition)
    conv[src: src + n] -= 2.0 * eps * a
    conv[dst: dst + n] += 2.0 * eps * a

    # quadratic part: eps^2 contributions
    conv[2 * src] += eps * eps
    conv[2 * dst] += eps * eps
    conv[src + dst] -= 2.0 * eps * eps

    # finally move the mass in the coefficient vector
    a[src] -= eps
    a[dst] += eps


def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of analytic (unnormalised) shapes."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]
    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Symmetric exponential decay from centre
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped, symmetric)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Half‑cosine (already used as a warm‑start)
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, several sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet like draws (few of them)
    rng = np.random.default_rng(0)
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # Normalise, keep only vectors with positive total mass
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, sum=1}. """
    if v.ndim != 1:
        raise ValueError("Input must be a one‑dimensional array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

        n = 8192                               # default – good balance of quality / speed

    # --------------------------------------------------------------
    # 1) Warm‑start – a collection of analytic shapes + a few random draws.
    # --------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # include the global starter if it exists and has the right length
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm_starts.append(starter / starter.sum())
    except Exception:
        pass

    # normalise all candidates (they should already be normalised but be safe)
    warm_vectors = []
    for w in warm_starts:
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s > 0.0:
            warm_vectors.append(w / s)

    # --------------------------------------------------------------
    # 2) Pick the best warm‑start according to the exact objective.
    # --------------------------------------------------------------
    best_vec = None
    best_score = math.inf
    for ws in warm_vectors:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    if best_vec is None:
        # fall back to the uniform distribution
        best_vec = np.full(n, 1.0 / n)

    a = best_vec.copy()
    conv = _conv_fft(a)                     # full self‑convolution
    cur_score = best_score
    n_obj_factor = 2.0 * n                  # because sum(a) == 1 throughout

    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Biased hill‑climbing with occasional uphill acceptance.
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective (fast: just use cached max)
        cur_obj = n_obj_factor * conv.max()
        k_max = int(np.argmax(conv))

        # ------------------------------------------------------------------
        # Choose a source index biasing towards the terms that form the
        # current maximal convolution entry.
        # ------------------------------------------------------------------
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        # contributions a[i] * a[k_max-i] for i in [lo, hi]
        idx = np.arange(lo, hi + 1)
        contrib = np.zeros(n, dtype=float)
        contrib[idx] = a[idx] * a[k_max - idx]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            src = rng.choice(n, p=contrib / total_contrib)
        else:
            # fallback – pick proportionally to the mass itself
            src = rng.choice(n, p=a)

        # Destination – uniform but must differ from src
        dst = rng.integers(0, n)
        if dst == src:
            continue
        if a[src] < 1e-15:
            continue

        # --------------------------------------------------------------
        # Amount to move – a shrinking fraction of the source mass.
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        total_time = deadline - start_time
        # early on we allow larger jumps; near the end we shrink them
        frac_max = 0.30 * (1.0 - elapsed / total_time) + 0.02
        eps = a[src] * rng.random() * frac_max
        if eps <= 0.0:
            continue

        before_obj = cur_obj

        # ------------------------------------------------------------------
        # Apply the tentative move (updates a and conv in‑place).
        # ------------------------------------------------------------------
        _apply_move(a, conv, src, dst, eps)

        new_obj = n_obj_factor * conv.max()

        # ------------------------------------------------------------------
        # Acceptance test: greedy + simulated annealing.
        # ------------------------------------------------------------------
        accept = False
        if new_obj <= before_obj:
            accept = True
        else:
            # temperature linearly decays to zero
            remaining = deadline - time.time()
            T0 = best_score * 0.05               # scale temperature to the problem
            T = T0 * max(remaining / total_time, 0.01)
            if np.random.random() < math.exp(-(new_obj - before_obj) / T):
                accept = True

        if accept:
            cur_obj = new_obj
            if new_obj < best_score:
                best_score = new_obj
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # revert the move
            _apply_move(a, conv, dst, src, eps)

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain Python list.
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score = {best_score:.6f}")
    return [float(v) for v in best_vec]
