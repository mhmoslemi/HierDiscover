            dst = rng.integers(n)

                continue
            eps = a[src] * rng.random() * 0.5
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
        else:  # mirror move
            src = rng.choice(n, p=a)
            dst = n-1 - src
            if src==dst or a[src] < 1e-15:

            eps = a[src] * rng.random() * 0.5
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time()-start_time:.2f}s] anneal iter={it:,} best={best_score:.6f}")

    return best, best_score

"""
Small‑scale “Simulated‑annealing + smooth‑gradient” optimizer.
The algorithm builds a diverse pool of analytic shapes (power ramps,
beta‑type densities, monotone polynomial ramps), picks the best
candidate and refines it with a smooth‑max gradient descent that works
on the probability simplex (projected FFT‑based gradient).  Finally a
light Metropolis‑type annealer with both random and mirror moves is
run.  The whole search respects the 1000 s budget and returns a plain
list of non‑negative heights.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT helpers (real FFT, power‑of‑two length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·n‑1) using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) without safety checks."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the Wang‑Carreira‑Perpiñán algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # very unlikely numerically
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric power‑ramp with a flat centre.
    n      – length
    p      – exponent of the ramp (≥0)
    alpha  – fraction of the length that stays flat (0 … 0.5)
    """
    alpha = float(np.clip(alpha, 0.0, 0.5))
    plateau_len = int(round(alpha * n))
    # ensure even plateau (helps the symmetry)
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:                      # degenerate → uniform
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # Trim / pad to exact length n
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    The continuous beta pdf ∝ x^{a‑1} (1‑x)^{b‑1}.
    """
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    if shape.sum() == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _make_monotone_ramp(n: int, p: float) -> np.ndarray:
    """
    Strictly decreasing power‑law ramp (no symmetry).
    a_i ∝ (1 - i/(n-1))^{p}.
    """
    i = np.arange(n, dtype=float)
    ramp = (1.0 - i / (n - 1)) ** p
    ramp[0] = 1.0                     # avoid 0⁰ if p=0
    ramp = np.maximum(ramp, 0.0)
    ramp /= ramp.sum()
    return ramp


# ----------------------------------------------------------------------
# Gradient from weighted convolution (FFT version)
# ----------------------------------------------------------------------
def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] w.r.t. a.
    Using the identity  ∇_a Σ_k w_k·(a*a)[k] = 2·corr(a, w)
    where corr is the (non‑reversed) cross‑correlation.
    Both a (length n) and w (length 2n‑1) are real.
    """
    n = a.shape[0]
    Lw = w.shape[0]                # = 2·n‑1
    # reverse w once – correlation becomes a convolution with w[::-1]
    w_rev = w[::-1]
    L = _next_pow2(n + Lw - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    conv_full = np.fft.irfft(Af * Wf, n=L)[: n + Lw - 1]
    # slice the part that corresponds to indices 0…n‑1 (see analysis)
    start = Lw - n                 # = n‑1
    grad = 2.0 * conv_full[start:start + n][::-1]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> Tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth approximation
          f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k])
    λ is increased gradually (annealed) so that the method converges
    to the true max‑objective.
    Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # control parameters
    lam = 5.0               # start with a mild smoothing
    lam_max = 150.0
    step0 = 0.6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # smooth weights (soft‑max) – shift by max to avoid overflow
        w_raw = np.exp(lam * (conv - max_conv))
        w = w_raw / w_raw.sum()               # Σ w = 1

        # gradient of the smooth max
        grad = _grad_from_weights(a, w)

        # diminishing step‑size
        step = step0 / math.sqrt(it)
        a -= step * grad

        # back to the simplex
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑GD iter={it:,}  score={cur_score:.6f}")

        # anneal λ slowly (log‑linear schedule)
        if it % 500 == 0 and lam < lam_max:
            lam = min(lam_max, lam * 1.15)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (with mirror moves)
# ----------------------------------------------------------------------
def _anneal_improved(seq: np.ndarray,
                     deadline: float,
                     start_time: float) -> Tuple[np.ndarray, float]:
    """
    Randomly move packets of mass; occasionally move mass to the mirrored
    position (i ↔ n‑1‑i) – this directly attacks the central autocorrelation.
    Accepts with a fast exponential cooling schedule.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 2_000_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # decide move type
        if rng.random() < 0.8:                     # standard random move
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
        else:                                      # mirror move
            src = rng.choice(n, p=a)
            dst = n - 1 - src

        if src == dst or a[src] < 1e-15:
            continue

        # move a random fraction (up to 40 %) of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,}  best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a low‑score step function.
    The routine builds a varied pool of analytic shapes, picks the best,
    refines it with a smooth‑max gradient descent on the probability
    simplex and finishes with a mirror‑aware simulated annealing polish.
    The returned list contains non‑negative floats and respects the 1000 s
    time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # 1) Length – power of two for very fast FFTs
    # --------------------------------------------------------------
    n = 65536  # 2¹⁶

    # --------------------------------------------------------------
    # 2) Build an initial pool of analytic candidates
    # --------------------------------------------------------------
    pool: List[Tuple[float, np.ndarray]] = []
    pool_size = 12
    best_seq = None
    best_score = float("inf")

    def _maybe_add(seq: np.ndarray, score: float):
        nonlocal best_seq, best_score, pool
        if score < best_score:
            best_score, best_seq = score, seq.copy()
        if len(pool) < pool_size:
            pool.append((score, seq.copy()))
            pool.sort(key=lambda x: x[0])
        elif score < pool[-1][0]:
            pool[-1] = (score, seq.copy())
            pool.sort(key=lambda x: x[0])

    # ---- uniform baseline (useful for sanity) --------------------
    uniform = np.full(n, 1.0 / n)
    _maybe_add(uniform, _objective(uniform))

    # ---- power‑ramp family ---------------------------------------
    p_grid = np.linspace(0.1, 5.0, 41)          # 0.1 … 5.0
    alpha_grid = np.linspace(0.0, 0.5, 21)      # 0 … 0.5
    for p in p_grid:
        for alpha in alpha_grid:
            seq = _make_power_shape(n, p, alpha)
            sc = _objective(seq)
            _maybe_add(seq, sc)
        if (time.time() - start_time) % 25 < 0.2:
            print(f"[{time.time() - start_time:.2f}s] power‑grid progress best={best_score:.6f}")

    # ---- monotone decreasing power ramps (new family) ----------
    for p in np.linspace(0.1, 5.0, 31):
        seq = _make_monotone_ramp(n, p)
        sc = _objective(seq)
        _maybe_add(seq, sc)

    # ---- beta‑distribution family (extended ranges) ------------
    a_vals = np.concatenate((
        np.geomspace(0.02, 0.5, 12),
        np.linspace(0.5, 5.0, 12)
    ))
    b_vals = a_vals.copy()
    for a_par in a_vals:
        for b_par in b_vals:
            seq = _make_beta_shape(n, a_par, b_par)
            sc = _objective(seq)
            _maybe_add(seq, sc)

    # ---- random mixtures of two beta shapes (more diversity) ---
    rng = np.random.default_rng()
    for _ in range(800):
        if time.time() >= deadline:
            break
        a1, b1 = rng.uniform(0.02, 6.0, 2)
        a2, b2 = rng.uniform(0.02, 6.0, 2)
        w = rng.random()
        s1 = _make_beta_shape(n, a1, b1)
        s2 = _make_beta_shape(n, a2, b2)
        seq = w * s1 + (1.0 - w) * s2
        seq /= seq.sum()
        sc = _objective(seq)
        _maybe_add(seq, sc)

    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Use a globally supplied starter if available
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter /= starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                _maybe_add(starter, sc)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4) Pick the current best candidate for local refinement
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 5) Smooth‑max gradient descent (projected) on the simplex
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 70.0:
        gd_deadline = time.time() + 0.70 * remaining
        refined_seq, refined_score = _smooth_gradient_descent(best_seq,
                                                              gd_deadline,
                                                              start_time)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] After smooth‑GD score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 6) Light annealing polish (mirror‑aware)
    # --------------------------------------------------------------
    if time.time() < deadline:
        annealed_seq, annealed_score = _anneal_improved(best_seq, deadline, start_time)
        if annealed_score < best_score:
            best_score, best_seq = annealed_score, annealed_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] After anneal score = {annealed_score:.6f}")

    # --------------------------------------------------------------
    # 7) Normalisation & conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
