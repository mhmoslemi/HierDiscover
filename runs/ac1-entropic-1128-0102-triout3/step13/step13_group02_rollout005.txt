"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  **Coarse parametric sweep** – we search over a simple one‑parameter
    family a_i ∝ r^i * (i+1)^p (geometric × power) and mix it with the
    uniform distribution.  The best triple (r,p,λ) is kept as a warm start.

2️⃣  **Fine optimisation of the three parameters** – a greedy “hill‑climb”
    (simulated‑annealing‑style) perturbs (r,p,λ) with decreasing step sizes
    while always accepting improving moves.  This refines the shape far beyond
    the coarse grid.

3️⃣  **Exact sub‑gradient descent** – starting from the refined shape we run
    projected sub‑gradient descent on the true objective (the maximal entry of
    the self‑convolution).  The sub‑gradient for the currently maximal shift
    is computed in O(n) after locating the shift with an FFT, and each iterate
    is projected onto the probability simplex.

All stages respect the 1000 s budget.  The final vector is normalised to sum
to 1 and returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric × power) and mixture with uniform
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """
    Returns a normalised vector of length n with components
    proportional to r**i * (i+1)**p (i = 0…n‑1).
    Computation is performed in log‑space for stability.
    """
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # log‑space to avoid overflow/underflow
    log_shape = i * np.log(r) + p * np.log(idx)
    # shift to keep exponentials moderate
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Returns lam * uniform + (1‑lam) * shape (both already sum to 1).
    The result automatically sums to 1.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    # Numerical noise may break the exact sum‑to‑1 property;
    # re‑normalise just in case.
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the true objective.
    The sub‑gradient corresponds to a single maximal shift of the convolution.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16                     # a slightly larger step than the baseline
    it = 0

    while time.time() < deadline:
        it += 1

        # locate the (first) maximal convolution entry with an FFT
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # --- sub‑gradient for shift k (O(n)) ---------------------------------
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # for i≠j also add the symmetric contribution a_i to ∂/∂a_j
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        # --- step with diminishing size ------------------------------------
        step = step0 / (it ** 0.6)          # slower decay than 1/√it
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional probabilistic uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # small jitter every few thousand iterations to escape plateaus
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_lam *= 0.9995

        # occasional jitter to avoid stagnation
        if rng.random() < 0.005:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    # --------------------------------------------------------------
    # 3️⃣  Sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        sub_deadline = deadline - 80.0
        a_sub, sc_sub = _subgradient_descent(best_seq, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final score={sc_sub:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb (mass transfer between random indices)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        L_fft = _next_pow2(2 * n - 1)
        best_a = a.copy()
        best_sc = _eval_fast(a, L_fft)
        while time.time() < deadline:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            # move up to 0.2 % of the source mass
            eps = a[src] * rng.random() * 2e-3
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_sc = _eval_fast(a_prop, L_fft)
            if new_sc < best_sc:
                a = a_prop
                best_sc = new_sc
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={best_sc:.6f}")

        best_seq = best_a
        best_score = best_sc

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
