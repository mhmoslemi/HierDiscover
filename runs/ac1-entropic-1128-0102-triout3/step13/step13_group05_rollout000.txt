def _subgradient_descent(a0, deadline, start_time):
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps=1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        score = 2 * a.shape[0] * max_val / (a.sum()**2)  # but sum=1 always; compute via _objective maybe cheaper
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={t} best={score:.6f}")
        # find max index
        kmax = int(np.argmax(conv))
        # compute gradient
        N = a.shape[0]
        grad = np.zeros(N)
        i_start = max(0, kmax - (N-1))
        i_end = min(N-1, kmax)
        idx = np.arange(i_start, i_end+1, dtype=int)
        grad[idx] = 2.0 * a[kmax - idx]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.01 / np.sqrt(t)  # adjust step size
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        # Project onto simplex
        a = _project_simplex(a)
    return best, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three phases and respects the 1000 s time budget.

1. **Diverse initial pool** – creates a handful of analytic shapes
   (power ramps with optional plateau, symmetric Beta‑densities, random
   Dirichlet vectors and the optional global starter).  The best
   `TOP_K` candidates are kept.

2. **Projected sub‑gradient descent** – starting from the current best
   candidate we repeatedly
   * compute the full self‑convolution (FFT),
   * locate the index where the convolution attains its maximum,
   * build the sub‑gradient `g_i = 2·a_{k_max‑i}` (only the overlapping
     part is non‑zero),
   * take an Adam‑style step and project back onto the probability
     simplex (non‑negative, sum = 1).

   The descent runs until the remaining budget is exhausted.  The best
   vector seen during the whole run is retained.

3. **Optional perturbation passes** – if a few seconds are left we
   briefly restart the descent from noisy perturbations of the current
   best solution to try to escape possible local minima.

The final vector is normalised to sum = 1 and returned as a plain list
of floats.  All heavy work is performed with NumPy and FFTs, which
allows tens of thousands of iterations for the chosen problem size
(`N` up to 65536) within the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant, sum is assumed to be 1 after projection)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp shape with optional plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simple top‑K pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score, vec) into the top‑k pool (kept sorted by score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    N = a.shape[0]
    best_a = a.copy()
    # since a will stay on the simplex, sum(a) == 1
    best_score = 2.0 * N * _conv_self(a).max()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        score = 2.0 * N * max_val          # denominator is 1 after projection
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} "
                  f"best={score:.6f}")

        # index attaining the maximum (first occurrence)
        kmax = int(np.argmax(conv))

        # build sub‑gradient g_i = 2·a_{kmax‑i} on the overlapping region
        grad = np.zeros_like(a)
        i_start = max(0, kmax - (N - 1))
        i_end = min(N - 1, kmax)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[kmax - idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size (decreases like 1/√t)
        step = 0.02 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score
