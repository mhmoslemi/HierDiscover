    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n-2)

    # initial sequence
    try: use height_sequence_1 if available
    except: uniform = np.full(n, 1/n)

    a = init_seq.copy()
    best_a = a.copy()
    best_score = evaluate(a) maybe compute via evaluate_sequence?

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.2

    # schedule for beta (smoothness) and learning rate
    total_iters = 0
    # We'll schedule: increase beta linearly or exponentially across all iterations. Use time based schedule: compute fraction = (time_elapsed / (deadline - start_time)).
    # At each iteration compute beta = beta_min * (beta_max/beta_min) ** fraction.

    while time.time() < deadline:
        # compute conv = self convolution (size 2n-1)
        conv = _conv_self_fft(a, L_self)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv  # since sum(a)=1
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] New best score {best_score:.6f}")

        # compute beta
        t_frac = (time.time()-start_time) / (deadline-start_time)
        beta = 2.0 * (200.0 / 2.0) ** t_frac # exponential from 2 to 200

        # compute softmax weights p
        # For numerical stability subtract max_conv first
        # Since exponent largest term exp(beta * max), but subtractions OK.
        # compute expvals = np.exp(beta * (conv - max_conv))  # may cause overflow small but safe
        expvals = np.exp(beta * (conv - max_conv))
        p = expvals / expvals.sum()

        # compute gradient
        a_rev = a[::-1]
        # compute convolution a_rev * p (full)
        # Use FFT length L_grad.
        A_rev = np.fft.rfft(a_rev, n=L_grad)
        P = np.fft.rfft(p, n=L_grad)
        gp_full = np.fft.irfft(A_rev * P, n=L_grad)
        # slice out indices n-1 to n-1 + n
        grad = 2.0 * gp_full[n-1:n-1+n]

        # Adam update
        total_iters += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**total_iters)
        v_hat = v / (1 - beta2**total_iters)
        # compute learning rate schedule: decreasing
        lr = base_lr * (1.0 / (1.0 + 0.0005 * total_iters))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_simplex(a)

        # occasionally add noise to escape plateau
        if total_iters % 2000 == 0:

            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    # final hill-climb (micro moves) same as earlier but maybe more aggressive? Could scrap for time.

    best_a = np.maximum(best_a, 0.0)

"""
Propose a step‑function (non‑negative list) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  A *fine* parametric sweep over a *symmetric* geometric family
    (mixing a left‑right exponential decay with the uniform vector).

2️⃣  A long projected Adam optimisation that minimises a *smooth*
    approximation of the max‑convolution (soft‑max with an annealed
    temperature).  The gradient is obtained by a single FFT‑based
    convolution, and the iterate is projected onto the probability
    simplex after each update.

3️⃣  A stochastic hill‑climb that performs larger mass transfers
    (up to a few % of the source) and accepts only improving moves.
    This stage cleans the remaining tiny irregularities.

All phases respect the overall time budget (default 1000 s) and use
FFT‑based convolutions (O(n log n)).  The final returned sequence is a
plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a NumPy 1‑d array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    return conv_full[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical safety – fall back to the uniform vector
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Score utilities
# ----------------------------------------------------------------------


def _raw_score(a: np.ndarray, L_self: int) -> float:
    """
    Fast (exact) computation of 2·n·max(conv(a,a)) assuming sum(a)=1.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a, L_self)
    return float(2.0 * n * conv.max())

# Adam optimiser on the smooth‑max objective
# ----------------------------------------------------------------------

    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_obj = best_score

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.12

    # temperature schedule (smooth‑max)
    beta_low = 2.0          # start almost linear
    beta_high = 4000.0      # end approximates true max very tightly

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- convolution of the current iterate ----
        conv = _conv_self_fft(a, L_self)          # length 2n‑1
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)      # exact objective (sum(a)=1)

        if cur_score < best_obj:
            best_obj = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # ---- smooth‑max softmax weights (p) ----
        t = (time.time() - start_time) / (deadline - start_time)
        beta = beta_low * (beta_high / beta_low) ** t   # exponential anneal
        # numerical stabilisation
        shifted = conv - conv.max()
        exp_vals = np.exp(beta * shifted)
        p = exp_vals / exp_vals.sum()                  # shape (2n‑1,)

        # ---- gradient of smooth‑max w.r.t. a ----
        a_rev = a[::-1]                                # reversed a
        A_rev = np.fft.rfft(a_rev, n=L_grad)
        P = np.fft.rfft(p, n=L_grad)
        grad_full = np.fft.irfft(A_rev * P, n=L_grad)
        # slice the part that corresponds to indices 0…n‑1
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple decay of the learning rate
        lr = base_lr / (1.0 + 0.0001 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (keeps sum(a)=1 and a≥0)
        a = _project_simplex(a)

        # occasional tiny random shake to escape shallow plateaux
        if it % 7000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    return best_a, best_obj


# ----------------------------------------------------------------------
# Local stochastic hill‑climb (larger mass moves)
# ----------------------------------------------------------------------


def _hill_climb(a_start: np.ndarray,
                start_time: float,
                deadline: float,
                L_self: int) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that repeatedly transfers a random
    fraction (up to 5 %) of mass from one index to another.
    Only improving moves are kept.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _raw_score(a, L_self)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # pick a fraction between 0 and 5 % of the source mass
        frac = rng.random() * 0.05 + 1e-6
        delta = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # projection is not needed – the move preserves total mass
        # and preserves non‑negativity (by construction)
        cur_score = _raw_score(a_prop, L_self)

        if cur_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    metric.  The routine follows the three‑phase strategy described above
    and returns a plain Python list ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final hill‑climb
    deadline = start_time + budget_s - 0.3

    # problem size -------------------------------------------------------
    n = 4096
    L_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # FFT length for gradient convolution

    # -------------------------------------------------------------------
    # initialise with a global starter if present, otherwise uniform
    # -------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
                best_seq = starter.copy()
                best_score = _raw_score(best_seq, L_self)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={best_score:.6f}")
        else:
            raise ValueError
    except Exception:
        # fall back to the uniform vector (sum = 1)
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = _raw_score(best_seq, L_self)
        print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # -------------------------------------------------------------------
    # 1️⃣  Symmetric geometric + uniform sweep
    # -------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-4.5, -2.5, 350)          # decay ε = 1‑r
    lam_grid = np.arange(0.0, 1.001, 0.01)          # mixing weight λ

    for eps in eps_grid:
        if time.time() >= deadline - 30.0:
            break
        r = 1.0 - eps
        geo_left = np.power(r, idx)                 # r^i
        geo_right = np.power(r, n - 1 - idx)        # r^{n‑1‑i}
        geo_sym = (geo_left + geo_right) * 0.5
        geo_norm = geo_sym / geo_sym.sum()

        for lam in lam_grid:
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo_norm
            seq = seq / seq.sum()                    # safety normalisation
            cur_score = _raw_score(seq, L_self)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Grid eps={eps:.1e} λ={lam:.2f} "
                      f"score={cur_score:.6f}")

    # -------------------------------------------------------------------
    # 2️⃣  Adam optimiser on the smooth‑max objective
    # -------------------------------------------------------------------
    adam_deadline = deadline - 150.0          # reserve time for hill‑climb
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        a_opt, sc_opt = _adam_smooth_opt(best_seq, start_time,
                                         adam_deadline, best_score,
                                         L_self, L_grad)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # -------------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climb (larger moves)
    # -------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline, L_self)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # -------------------------------------------------------------------
    # Normalise (just in case) and return a plain Python list
    # -------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
