```
def _anneal(seq, deadline, start_time):
    a = seq.copy()
    n=len(a)
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng=np.random.default_rng()
    max_steps=int(5e5)
    for it in range(max_steps):
        if time.time()>=deadline: break
        # Temperature schedule: start high, exponentially decay
        T = 0.1 * (1e-5/0.1) ** (it / max_steps)
        src = rng.choice(n, p=a)  # choose index weighted by mass
        dst = rng.integers(0, n)
        if src==dst or a[src] < 1e-14: continue
        # move fraction up to 0.5
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0 or np.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")
```

assistantfinal**Solution Overview**

The evaluation function  

\[
E(a)=\frac{2n\;\max\bigl((a*a)\bigr)}{(\sum a)^2}
\]

is *scale‑invariant*: multiplying the whole vector does not change its value.  
Hence we can safely work on the simplex  

\[
\Delta=\{a\in\mathbb R_{\ge0}^n\mid\sum a=1\}
\]

and minimise  

\[
F(a)=\max\bigl(a*a\bigr) .
\]

The objective is a pointwise maximum of many quadratic forms, therefore it is non‑smooth.  
Our optimiser consists of two complementary phases

1. **Projected Adam on a smooth approximation of the max**  
   We replace `max(conv)` by the soft‑max  
   \[
   m_\theta(a)=\frac{1}{\theta}\log\!\Bigl(\sum_{k}e^{\theta\,\text{conv}_k}\Bigr)
   \]
   whose gradient can be written analytically and evaluated with a *single* extra FFT.  
   Adam’s adaptive steps together with an increasing temperature θ (from 0.5 up to 50) give a fast descent on the simplex.

2. **Light simulated‑annealing polish**  
   After the gradient phase we perform a Metropolis walk that moves small packets of mass between positions.  
   The temperature decays exponentially, allowing the algorithm to escape shallow basins and to settle into a high‑quality local minimum.

Both phases respect the 1000 s budget, keep the vector non‑negative, and repeatedly project onto the simplex (the standard \(O(n\log n)\) algorithm).  
The code works for any length; if the global variable `height_sequence_1` is present we start from it (normally 4096 entries), otherwise we fall back to a random β‑distribution shape.

The final sequence is normalised to sum = 1 and returned as a plain Python list.

```python
"""
Propose a low‑score non‑negative step function.

The algorithm works in two stages:
1. Projected Adam optimisation on a smooth approximation of
   max(conv(a,a)).  The smooth max is realised by a soft‑max with a
   temperature that grows from 0.5 to 50, allowing a well‑behaved
   gradient that can be computed with three FFTs per iteration.
   After each Adam update the vector is projected onto the probability
   simplex (non‑negative, sum = 1).

2. A final simulated‑annealing polish that moves tiny packets of mass
   between positions, accepting worsening moves with a decreasing
   temperature schedule.

Both stages respect the 1000 s wall‑clock limit and return a plain
list of floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant, sum‑free version)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx=1, x≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smooth max and its gradient (soft‑max with temperature θ)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, theta: float) -> np.ndarray:
    """
    Return the soft‑max weights w_k = exp(θ·conv_k)/∑exp(θ·conv)
    (numerically stable).
    """
    max_c = conv.max()
    shifted = conv - max_c  # avoid overflow
    exps = np.exp(theta * shifted)
    w = exps / exps.sum()
    return w


def _grad_smooth_max(a: np.ndarray, conv: np.ndarray, theta: float) -> np.ndarray:
    """
    Gradient of the smoothed max m_θ(a) = (1/θ)·log Σ e^{θ·conv}
    w.r.t. the vector a (the factor 2 from the bilinear form is included).
    """
    n = a.shape[0]
    w = _softmax_weights(conv, theta)               # shape (2n‑1,)
    a_rev = a[::-1]                                 # reversed a
    L = _next_pow2(a_rev.shape[0] + w.shape[0] - 1)

    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(Af * Wf, n=L)

    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]          # length n
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex with a growing temperature θ.
    Returns the best iterate seen and its (true) objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # full self‑convolution
        conv = _conv_self(a)

        # temperature schedule (linear increase)
        theta = 0.5 + (50.0 - 0.5) * (it / max_iters)
        if theta > 50.0:
            theta = 50.0

        # gradient of the smooth max
        grad = _grad_smooth_max(a, conv, theta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # optional symmetry enforcement (helps convergence)
        a = (a + a[::-1]) * 0.5

        # projection onto the simplex
        a = _project_simplex(a)

        # occasional true‑objective evaluation
        if it % 500 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,}  best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (packet moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move tiny packets of mass between positions.
    Accept according to a Metropolis rule with an exponentially decaying temperature.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric cooling from 0.1 to 1e-6
        T = 0.1 * (1e-6 / 0.1) ** (it / max_steps)

        # pick a source index proportionally to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(0, n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 50 % of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine combines a projected Adam optimisation (smooth max) with a
    modest simulated‑annealing polish, all respecting a 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1) Choose problem size and an initial candidate
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError("global height_sequence_1 is malformed")
        n = starter.size
        a0 = np.maximum(starter, 0.0)
        total = a0.sum()
        if total > 0.0:
            a0 /= total
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        # fallback: moderately sized symmetric beta shape
        n = 16384  # power‑of‑two for fast FFTs
        a = (np.arange(n) + 0.5) / n
        a0 = (a ** (2.0 - 1)) * ((1.0 - a) ** (2.0 - 1))
        a0 /= a0.sum()

    print(f"[{time.time() - start_time:.2f}s] Problem size n={n}")

    # --------------------------------------------------------------
    # 2) Quick sanity check of the starter
    # --------------------------------------------------------------
    cur_score = _objective(a0)
    best_a = a0.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] Starter score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase‑1: projected Adam on the smooth max
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + 0.85 * remaining
        a_adam, score_adam = _adam_optimize(best_a, adam_deadline, start_time)
        if score_adam < best_score:
            best_score, best_a = score_adam, a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam phase improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase‑2: simulated‑annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_ann, score_ann = _anneal(best_a, deadline, start_time)
        if score_ann < best_score:
            best_score, best_a = score_ann, a_ann.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal phase improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to a plain list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
```
