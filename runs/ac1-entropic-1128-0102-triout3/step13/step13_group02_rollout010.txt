"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The program follows a multi‑phase strategy that stays inside the 1000 s
budget and works completely on the probability simplex (non‑negative,
sum = 1):

1️⃣  A fast parametric sweep over geometric‑power families (mixed with the
    uniform distribution) to obtain a good initial candidate.

2️⃣  Several random restarts of an Adam‑style projected descent on the
    smooth‑max surrogate of the max‑convolution.  The temperature β is
    driven up to a large value (2000) so the surrogate becomes almost the
    true max.

3️⃣  A sub‑gradient descent that uses *all* active shifts whose
    convolution value is within a small tolerance of the current maximum
    (instead of a single shift).  This gives a richer descent direction.

4️⃣  A simulated‑annealing refinement that repeatedly moves a random
    fraction of mass from one coordinate to another; moves that increase
    the objective are occasionally accepted, allowing the search to jump
    out of shallow local minima.

5️⃣  A final microscopic hill‑climb that transfers tiny amounts of
    mass between random pairs and only keeps improving moves.

The length of the vector is taken to be 8192 – larger than the previous
run – which gives the optimiser many more degrees of freedom while still
being cheap enough for FFT‑based convolutions.  All phases respect the
overall time limit and the best sequence found is returned as a plain
Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT (length 2·len(a)‑1).
    Returns a real ndarray of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (w ≥ 0, sum(w) = 1).  Implements the O(n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assuming sum≈1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1.
    The function is deliberately cheap (single FFT + a few scalars).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

        if time.time() >= deadline - 150:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 150:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                cand = lam * np.full(n, 1.0 / n) + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _objective(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} "
                          f"p={p:.2f} λ={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max Adam descent
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, best_score_initial: float):
    """
    Adam‑style projected descent on the soft‑max of the convolution.
    The temperature β starts at 5 and is multiplied by 1.03 each step,
    capped at 2000 (much larger than the previous 500).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0
    beta_mul = 1.03
    beta_max = 2000.0

    while time.time() < deadline:
        t += 1

        # ----- soft‑max gradient -----
        conv = _conv_self_fft(a, L_conv)
        max_c = conv.max()
        shifted = conv - max_c               # for numerical stability
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max weights, shape (2n‑1,)

        # correlation w * a_rev  → gradient of the smooth max
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        cur_score = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} "
                  f"score={cur_score:.6f}")

        # ----- Adam update with projection onto the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # ----- temperature schedule -----
        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to escape plateaus
        if t % 3000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – sub‑gradient descent with active set
# ----------------------------------------------------------------------


def _active_set_gradient(a: np.ndarray, conv: np.ndarray,
                         max_val: float, rel_tol: float = 1e-3) -> np.ndarray:
    """
    Compute the average sub‑gradient of all shifts whose convolution value
    lies within rel_tol·max_val of the current maximum.
    """
    n = a.shape[0]
    # indices that are “active”
    active = np.where(conv >= max_val * (1.0 - rel_tol))[0]
    if active.size == 0:
        return np.zeros_like(a)

    g = np.zeros_like(a, dtype=np.float64)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals

        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]

        # for i≠j also add a_i to ∂/∂a_j (the symmetric term)
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

    # average over the active set to keep the step size moderate
    g /= float(active.size)
    return g


def _subgradient_descent_active(a0: np.ndarray, n: int,
                                deadline: float, start_time: float,
                                best_score_initial: float):
    """
    Projected sub‑gradient descent using the *active set* of near‑maximum
    convolution entries.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.15               # initial step magnitude
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()

        # compute richer sub‑gradient
        g = _active_set_gradient(a, conv, max_val, rel_tol=1e-3)

        # diminishing step size
        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _objective(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # occasional simulated‑annealing style acceptance
            prob = np.exp(-(cur_score - best_score) * 10.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # small jitter every few thousand iterations
        if it % 1500 == 0:
            jitter = 1e-4 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         best_score_initial: float):
    """
    Classic SA that moves a random fraction of mass from a source index
    to a destination index.  The temperature is lowered linearly with
    time.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    cur_score = _objective(a, L_fft)
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    start_sa = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_sa
        # linear cooling: T goes from 0.02*best_score to 0.0005*best_score
        T = (0.02 - 0.0195 * (elapsed / total_time)) * best_score
        T = max(T, 1e-4)

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a random fraction of src to dst (up to 20 % of src)
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop, L_fft)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA score={best_score:.6f}")

        # occasionally inject a small random perturbation around the best point
        if rng.random() < 0.001:
            a = best_a.copy()
            a += 0.001 * rng.standard_normal(n) * a
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            cur_score = _objective(a, L_fft)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 5 – final microscopic hill‑climb
# ----------------------------------------------------------------------


def _final_hill_climb(a0: np.ndarray, deadline: float, start_time: float):
    """
    Very small transfers of mass between random coordinates; only keeps
    improving moves.  Acts as a polishing step.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _objective(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3      # ≤0.1 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine follows the five phases described
    above, respects the 1000 s time budget and finally returns the best
    sequence as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin – we stop a bit before the absolute deadline
    deadline = start_time + budget_s - 0.2

    # ---------- problem size ----------
    n = 8192                           # more degrees of freedom than before
    L_fft = _next_pow2(2 * n - 1)

    # ---------- baseline (uniform) ----------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ---------- optional global starter ----------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _objective(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ---------- Phase 1 : parametric sweep ----------
    if time.time() < deadline - 150:
        best_seq, best_score = _parametric_sweep(
            n, L_fft, deadline, start_time, best_seq, best_score)

    # ---------- Phase 2 : smooth‑max Adam descent (several restarts) ----------
    time_for_smooth = max(0.0, min(deadline - time.time() - 250, 400))
    n_restarts = 4
    per_restart = time_for_smooth / max(1, n_restarts)

    for i in range(n_restarts):
        if time.time() >= deadline - 250:
            break
        rng = np.random.default_rng(seed + i + 1000)
        a0 = best_seq.copy()
        noise = rng.normal(scale=0.025, size=n)   # 2.5 % relative noise
        a0 = np.maximum(a0 + noise, 0.0)
        a0 = _project_simplex(a0)

        sub_deadline = time.time() + per_restart
        a_smooth, sc_smooth = _smooth_descent(
            a0, n, sub_deadline, start_time, best_score)
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {i+1}/{n_restarts} "
                  f"improvement score={sc_smooth:.6f}")

    # ---------- Phase 3 : active‑set sub‑gradient descent ----------
    if time.time() < deadline - 200:
        sub_deadline = deadline - 200
        a_sub, sc_sub = _subgradient_descent_active(
            best_seq, n, sub_deadline, start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] active sub‑grad score={sc_sub:.6f}")

    # ---------- Phase 4 : simulated‑annealing refinement ----------
    if time.time() < deadline - 70:
        sa_deadline = deadline - 70
        a_sa, sc_sa = _simulated_annealing(
            best_seq, n, sa_deadline, start_time, best_score)
        if sc_sa < best_score:
            best_score = sc_sa
            best_seq = a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] SA refinement score={sc_sa:.6f}")

    # ---------- Phase 5 : final hill‑climb ----------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ---------- final normalisation ----------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
