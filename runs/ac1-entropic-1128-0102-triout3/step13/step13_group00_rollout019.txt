#!/usr/bin/env python
"""
search_step_function.py

Search for a sequence of non‑negative heights that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is scale‑invariant, therefore we always keep the
vector normalised to sum = 1 and minimise 2·n·max(conv).  The
algorithm consists of four phases (all respecting the 1000 s wall‑clock
budget):

1. **Broad parametric sweep** – geometric families (including a
   mixture of a left‑hand and a right‑hand geometric decay) and an
   extended beta family (α∈[0.05,4], β∈[0.05,10]).
2. **Random Dirichlet starts** – many draws with concentrations drawn
   from a log‑uniform distribution.
3. **Simulated‑annealing refinement** – a fast projected‑simplex
   random‑walk that accepts uphill moves with a temperature schedule.
4. **Deterministic hill‑climb** – sub‑gradient moves that only keep
   improving candidates.

All convolutions use an FFT based linear convolution; the simplex
projection is O(n).  The routine returns a plain list of Python floats
suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helper (fast, scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric families
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray, best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Explore a wide range of geometric, mixed‑geometric and beta families."""
    idx = np.arange(n, dtype=np.float64)

    uniform = np.full(n, 1.0 / n)

    # ---- geometric + reverse mixture ------------------------------------
    eps_grid = np.logspace(-6, -0.5, 600)           # finer near 1/n
    lam_grid = np.linspace(0.0, 1.0, 21)            # mixing weight left↔right
    lam_u_grid = np.array([0.0, 0.05, 0.10, 0.15]) # optional tiny uniform mix

    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0.0:
            continue
        left = geo / geo_sum                       # sum(left)=1
        right = left[::-1]                         # reversed copy, also sum=1

        for lam_u in lam_u_grid:
            for lam in lam_grid:                    # weight of left Geo
                if lam_u + lam > 1.0:
                    continue
                seq = lam_u * uniform + lam * left + (1.0 - lam_u - lam) * right
                # seq already sums to 1 because each component sums to 1
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} "
                          f"lam_u={lam_u:.2f} lam={lam:.2f} score={sc:.6f}")

    # ---- beta‑type (more extensive than the previous run) --------------
    alphas = np.linspace(0.05, 4.0, 81)      # same resolution for α
    betas = np.linspace(0.05, 10.0, 96)      # extend β up to 10
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))

    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            # Log‑density of Beta(α,β) on the discrete grid
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0.0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} "
                      f"score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet starts
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 250                      # many more draws than before
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration drawn log‑uniformly in [0.05, 20]
        c = np.exp(rng.uniform(np.log(0.05), np.log(20.0)))
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with Metropolis acceptance.
    Moves consist of transferring a small fraction of mass from a source
    index to a destination index.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)

    rng = np.random.default_rng()
    # initial temperature – a fraction of the current score
    T = max(best_score * 0.02, 5e-4)

    it = 0
    while time.time() < deadline:
        it += 1
        # pick source (must have some mass) and destination
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[rng.integers(src_candidates.size)]
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a random fraction of src's mass
        frac = rng.random() * 0.01 + 1e-6      # up to ~1 % of src
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # sum unchanged, keep non‑negativity
        if a_prop[src] < -1e-16:
            continue

        cur_score = _objective_score(a_prop)

        # Metropolis acceptance
        if cur_score < best_score or rng.random() < np.exp((best_score - cur_score) / T):
            a = a_prop
            if cur_score < best_score:
                best_a, best_score = a.copy(), cur_score
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} "
                      f"new best score={cur_score:.6f}")
        # geometric cooling
        T = max(T * 0.99997, 1e-6)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (sub‑gradient moves)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """Fine‑grained deterministic hill climb."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # indices with most positive / most negative sub‑gradient
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

        # transfer a tiny fraction (up to 0.4 % of source)
        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a, best_score = a.copy(), cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter /= s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric, mixed‑geometric, beta)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0            # reserve ~300 s for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After param sweep best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After Dirichlet trials best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing refinement
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0    # keep ~2 min for final hill‑climb
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, start_time, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After SA best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final deterministic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, start_time, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
