"""
Propose a low‑objective step‑function by combining:

* a rich warm‑start (analytic shapes + random Dirichlet draws);
* a simulated‑annealing / hill‑climbing loop that updates the linear
  self‑convolution incrementally (O(n) per move);
* periodic gradient‑based bias: we compute how much each coefficient
  contributes to the current maximum of the convolution and preferentially
  move mass away from high‑contributing indices;
* an annealing schedule that sometimes accepts uphill moves, helping the
  search escape local minima.

The algorithm works for any length `n` (default 4096) and respects the
security checks of ``evaluate_sequence`` – all entries stay non‑negative
and the sum stays constant (≈1).  The best vector seen during the run is
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector (sum≈1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


def _apply_move(a: np.ndarray, conv: np.ndarray,
                src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from `src` to `dst` and update the convolution.

    Parameters
    ----------
    a : np.ndarray (n,)
        Coefficient vector (modified in‑place).
    conv : np.ndarray (2n‑1,)
        Full linear self‑convolution (modified in‑place).
    src, dst : int
        Distinct indices.
    eps : float
        Amount moved, 0 ≤ eps ≤ a[src].
    """
    n = a.shape[0]

    # linear part:  -2*eps*a shifted by src, +2*eps*a shifted by dst
    conv[src:src + n] -= 2.0 * eps * a
    conv[dst:dst + n] += 2.0 * eps * a

    # quadratic part (the pure eps*eps contribution)
    conv[2 * src] += eps * eps
    conv[2 * dst] += eps * eps
    conv[src + dst] -= 2.0 * eps * eps

    # update vector
    a[src] -= eps
    a[dst] += eps


def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """
    Produce a diverse collection of analytic (unnormalised) shapes.
    All returned arrays have length n.
    """
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tent (|x‑0.5|)^p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Symmetric exponential decay
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Edge‑dominant power‑law
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Simple sine/tent
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, few sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet draws (a few)
    rng = np.random.default_rng(0)
    for _ in range(12):
        rnd = rng.random(n)
        candidates.append(rnd)

    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)

    return normalised


def _weight_from_max(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    For each index i compute its contribution to the current maximum
    convolution value.  If the maximum is attained at multiple positions
    the contributions are summed.

    Returns an array w of length len(a) where
        w[i] = sum_{k:conv[k]==max} a[k-i]   (only terms with 0≤k-i<n count)
    """
    n = a.shape[0]
    max_val = conv.max()
    idxs = np.where(conv == max_val)[0]
    w = np.zeros_like(a)

    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(n, k + 1)
        i = np.arange(lo, hi)
        w[i] += a[k - i]

    return w

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with low value of
        2·n·max(conv(a,a)) / (sum(a)**2)

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the harness will subtract a small safety
        margin; we stop a bit earlier).

    Returns
    -------
    list[float]
        A non‑negative vector ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Determine length n
    # ------------------------------------------------------------------

    # if the global starter exists, add it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:

    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    a = best_vec.copy()
    conv = _conv_fft(a)
    cur_score = best_score
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing / gradient‑biased hill‑climbing
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)

    total = a.sum()
    n_obj = 2.0 * n / (total * total)          # constant factor: obj = n_obj * conv.max()

    # annealing parameters
    T0 = 0.02                                   # initial temperature (objective units)
    grad_update_interval = 4000                # recompute weight every this many moves
    next_grad_update = 0
    weight = _weight_from_max(conv, a)          # initial contribution weights

    it = 0
    while time.time() < deadline:
        it += 1

        # periodically recompute the contribution weight
        if it >= next_grad_update:
            weight = _weight_from_max(conv, a)
            next_grad_update = it + grad_update_interval

        # choose move mode: 0‑50 % gradient‑biased, else random
        if rng.random() < 0.5 and weight.max() > 0.0:
            # gradient‑biased source (high contribution) and destination (low)
            src_candidates = np.where((a > 1e-12) & (weight > 0.0))[0]
            if src_candidates.size == 0:
                src = rng.choice(np.where(a > 1e-12)[0])
            else:
                # probability ∝ a[i] * weight[i]
                probs = a[src_candidates] * weight[src_candidates]
                probs_sum = probs.sum()
                if probs_sum > 0:
                    src = rng.choice(src_candidates, p=probs / probs_sum)
                else:
                    src = rng.choice(src_candidates)

            # destination: pick index with minimal weight (break ties randomly)
            # ensure dst ≠ src
            min_w = weight.min()
            dst_candidates = np.where(weight == min_w)[0]
            if dst_candidates.size == 0:
                dst = rng.integers(n)
            else:
                dst = rng.choice(dst_candidates)
            if dst == src:
                dst = (dst + 1) % n
        else:
            # pure random source/destination
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        # compute epsilon (fraction of source mass)
        elapsed = time.time() - start_time
        remaining = deadline - time.time()
        time_frac = max(0.0, min(1.0, remaining / (deadline - start_time)))
        eps_frac = 0.30 * time_frac + 0.02          # large early, tiny late
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        cur_obj = n_obj * conv.max()

        new_obj = n_obj * conv.max()
        delta = new_obj - cur_obj

        # decide acceptance
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            # temperature linearly decays to near zero
            T = max(1e-9, T0 * (1.0 - (elapsed / (budget_s - 0.5))))
            prob = math.exp(-delta / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # keep the move; possibly update best

                best_score = new_obj
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # reject – revert
            _apply_move(a, conv, dst, src, eps)

        # occasional progress report (every ~5 % of time)
        if it % 500000 == 0:
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} cur={n_obj*conv.max():.6f} best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final normalisation and conversion
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score = {best_score:.6f}")
    return [float(v) for v in best_vec]
