def _annealing_refine(a, deadline, start_time):
    n = len(a)
    conv = _conv_self_fft(a)
    best_a = a.copy()
    best_score = _eval_fast(a, _next_pow2(2*n-1))
    best_conv = conv.copy()
    # after each accepted move, update a and conv.
    iter = 0
    while time.time() < deadline:
        iter += 1
        # temperature schedule:
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = max(1e-6, 0.01 * (1 - elapsed / total))  # linear cooling
        # pick src, dst
        src = rng.integers(n)
        if a[src] <= 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # step fraction
        max_frac = max(0.5, 0.1) # maybe variable
        # for exploration, choose fraction scale:
        frac = rng.uniform(0.0, 0.5)  # up to 50% of src mass
        delta = a[src] * frac
        # ensure delta > tiny
        if delta < 1e-12:
            continue
        # compute new a and conv
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # update conv via slice
        conv_new[src:src+n] -= delta * a
        conv_new[dst:dst+n] += delta * a
        # extra adjustments
        conv_new[2*src] -= delta * a[src]
        conv_new[2*dst] -= delta * a[dst]
        conv_new[src+dst] += delta * (a[src] - a[dst])

        new_score = 2.0 * n * conv_new.max()  # sum=1

        # accept?
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / T):
            a = a_new
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                best_conv = conv.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={iter} score={best_score:.6f}")
    return best_a, best_score

"""
Propose a low‑score step‑function via a multi‑stage search.

Stage 1 – cheap baselines (uniform, the supplied global starter,
parameteric geometric‑power families) give a decent starting point.

Stage 2 – a large number of very cheap simulated‑annealing moves.
The self‑convolution of the current vector is kept incrementally,
so each trial move costs only O(n) vector operations (no FFT).  The
temperature schedule and the step‑size slowly shrink, allowing the
search to first explore widely and then polish the best solution.
All vectors are kept on the probability simplex (non‑negative,
summing to 1).  The best vector encountered during the whole run is
returned.

The algorithm respects the 1000 s budget, works completely with
NumPy (no external solvers) and returns a plain Python list of
floats suitable for ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0, sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (incremental convolution)
# ----------------------------------------------------------------------


def _annealing_refine(a0: np.ndarray,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing on the simplex.
    Keeps the full convolution array and updates it in O(n) per trial.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)          # exact convolution
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = 2.0 * n * conv.max()    # sum(a)=1

    rng = np.random.default_rng()
    it = 0

    # annealing schedule parameters (tuned empirically)
    T_start = best_score * 5e-3         # initial temperature
    T_end = best_score * 1e-7           # final temperature
    max_frac_start = 0.5                # up‑to 50 % of source mass
    max_frac_end = 1e-4                 # final fine‑grained moves

    while time.time() < deadline:
        it += 1
        # linear schedules
        elapsed = time.time() - start_time
        total = deadline - start_time
        t_frac = min(1.0, elapsed / total)
        T = T_start * (1.0 - t_frac) + T_end * t_frac
        max_frac = max_frac_start * (1.0 - t_frac) + max_frac_end * t_frac

        # pick a source with non‑negligible mass
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # propose a transfer amount
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # keep a copy of the *old* a for the convolution update
        a_old = a.copy()

        # ----- incremental convolution update (using the old a) -----
        conv_new = conv.copy()
        # shift‑i contribution (subtract)
        conv_new[src:src + n] -= delta * a_old
        # shift‑j contribution (add)
        conv_new[dst:dst + n] += delta * a_old

        # extra diagonal adjustments
        conv_new[2 * src] -= delta * a_old[src]
        conv_new[2 * dst] -= delta * a_old[dst]
        conv_new[src + dst] += delta * (a_old[src] - a_old[dst])

        # ----- apply the mass move -----
        a[src] = a_old[src] - delta
        a[dst] = a_old[dst] + delta

        # ----- evaluate proposal -----
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # ----- Metropolis acceptance -----
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # occasional uphill moves (temperature may be very small)
            prob = np.exp(-(new_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            conv = conv_new
            # update global best if improved
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                best_conv = conv.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} "
                      f"score={best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine performs a fast parametric sweep, a handful of
    random Dirichlet restarts and finishes with an intensive simulated‑
    annealing refinement that updates the convolution incrementally.
    It respects the 1000 s time budget and returns the best sequence
    as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    n = 4096                                 # problem size
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # truncate / pad to the required length
            if starter.size > n:
                starter = starter[:n]
            elif starter.size < n:
                pad = np.zeros(n - starter.size)
                starter = np.concatenate([starter, pad])
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # parametric sweep (geometric‑power families)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based indices
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-3.0, 2.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 200:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 200:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape = shape / s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                cand = lam * uniform + (1.0 - lam) * shape
                cand = cand / cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} "
                          f"p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # a few random Dirichlet restarts (very cheap)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    for _ in range(12):
        if time.time() >= deadline - 150:
            break
        # concentration <1 encourages sparsity, >1 encourages smoothness
        alpha = rng.uniform(0.2, 2.0)
        cand = rng.dirichlet(np.full(n, alpha))
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] dirichlet α={alpha:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # intensive simulated‑annealing refinement (incremental conv)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:          # keep a few seconds for final tidy‑up
        a_refined, sc_refined = _annealing_refine(best_seq, deadline, start_time)
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] annealing final score={sc_refined:.6f}")

    # ------------------------------------------------------------------
    # final normalisation (sum → 1) and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
