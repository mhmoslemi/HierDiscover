"""
Minimum‑max‑convolution step‑function search (≈ 1000 s budget).

Algorithm
---------
1. **Warm‑starts** – a large pool of analytic and random shapes
   (tent, cosine, circle, Gaussian, power‑tents, exponentials,
   Beta‑pdfs, Dirichlet draws).  The best one (according to the exact
   objective 2·n·max(conv)) is kept as a seed.

2. **Smoothed‑max Adam** – minimise a smooth approximation of the
   Chebyshev norm:
        Sβ(a)= (1/β)·log Σ exp(β·(a*a)[k]).
   Its gradient is a weighted convolution and can be evaluated by FFT.
   β is increased slowly (β≈1 → 200) so the optimiser first sees a
   smooth landscape and then focuses on the true peak.  Adam with a
   simplex projection updates the probabilities.

3. **Exact‑max Adam** – after the smooth phase a short Adam run on the
   true sub‑gradient of the maximum entry (as in the reference code)
   refines the solution.

4. **Greedy hill‑climbing** – cheap Metropolis moves that shift a
   random packet of mass between two indices, keeping any strict
   improvement.  This polishing often yields a final gain of several
   10⁻⁴.

All heavy work uses NumPy’s real FFT, the optimisation respects the
1 000 s time limit and returns a list that can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families + random families
# ----------------------------------------------------------------------
def _make_warm_starts(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # ----- Analytic families -----
    # 1) linear tent
    candidates.append(np.minimum(x, 1.0 - x))

    # 2) half‑cosine (raised cosine)
    candidates.append(np.sin(np.pi * x))

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    candidates.append(np.exp(-0.5 * ((x - 0.5) / sigma) ** 2))

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        candidates.append(exp_fwd)
        candidates.append(exp_fwd[::-1])

    # 7) constant (uniform) – baseline
    candidates.append(np.ones_like(x))

    # 8) Beta‑pdf families (α,β in a small grid)
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    betas = [0.2, 0.5, 1.0, 2.0, 5.0]
    eps = 1e-12
    for a in alphas:
        for b in betas:
            pdf = (x + eps) ** (a - 1) * (1.0 - x + eps) ** (b - 1)
            candidates.append(pdf)

    # 9) Random Dirichlet draws (few, cheap)
    for _ in range(8):
        conc = rng.uniform(0.1, 1.0)          # encourage sparsity
        cand = rng.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # Normalise each candidate to sum = 1 and keep only the non‑negative ones.
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max(conv) w.r.t. a.
    conv is the full linear self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())            # index where the max is attained
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]      # g_i = 2·a_{k‑i} on the overlap
    return grad

def _adam_optimize(a0: np.ndarray,

                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of the max‑convolution.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # self‑convolution
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # sub‑gradient of the current max entry
        grad = _max_subgradient(a, conv)

        # scale because the true objective is 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

        # current objective (already known from conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cur_score:.6f}")

        # learning‑rate decay
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed‑max Adam (softmax approximation of the peak)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.02,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          beta_init: float = 0.5,
                          beta_target: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth proxy   Sβ(a) = (1/β)·log Σ exp(β·(a*a)[k]).
    The schedule slowly raises β, turning the proxy into the true max.
    Returns the best iterate (according to the exact objective).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = beta_init

    while time.time() < deadline:
        it += 1

        # full convolution (length 2·n‑1)
        L = _next_pow2(2 * n - 1)
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # soft‑max weights, stabilized
        max_c = conv.max()
        exp_scaled = np.exp(beta * (conv - max_c))
        w = exp_scaled / exp_scaled.sum()          # Σ w = 1, w ≥ 0

        # gradient of the smoothed objective:
        #   g = 4·n· Σ_k w_k·a_{k-i}
        # implement via convolution a * w_rev
        w_rev = w[::-1]                             # reversed weights
        L2 = _next_pow2(n + len(w_rev) - 1)
        grad_full = np.fft.irfft(
            np.fft.rfft(a, n=L2) * np.fft.rfft(w_rev, n=L2), n=L2
        )
        offset = len(w_rev) - 1                     # = 2·n‑2
        grad_weight = grad_full[offset - np.arange(n)]
        grad = 4.0 * n * grad_weight                 # full gradient of 2·n·Sβ

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        # exact objective for bookkeeping
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth‑Adam it={it:,} "
                  f"β={beta:.1f} score={cur_score:.6f}")

        # schedule: slowly increase β, decay lr a little
        beta = min(beta_target, beta * 1.02)
        if it % 3000 == 0:
            lr *= 0.95

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepting only improvements.  Very cheap and works
    well as a final polish.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        # source chosen proportionally to its mass → faster convergence
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:                     # greedy acceptance
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain list of floats (sum = 1, all entries ≥ 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid hard timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – power of two for FFT speed.
    # ------------------------------------------------------------------
    n = 4096                     # 4 k works well within the time budget

    # ------------------------------------------------------------------
    # Warm‑starts – analytic + random families.
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n, rng)

    # optionally incorporate a globally supplied construction
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            h1_resampled = np.maximum(h1_resampled, 0.0)
            s = h1_resampled.sum()
            if s > 0:
                warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Smoothed‑max Adam (50 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        smooth_deadline = time.time() + remaining * 0.50
        a_opt, score_opt = _smooth_adam_optimize(
            a0, smooth_deadline, start_time,
            lr=0.02, beta_init=0.5, beta_target=200.0
        )
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] After smooth‑Adam best={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 2) Exact‑max Adam (30 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + remaining * 0.30
        a_opt, score_opt = _adam_optimize(a0, adam_deadline, start_time, lr=0.015)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] After exact‑Adam best={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Light hill‑climb polish (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective={best_score:.6f}, n={n}")
    return [float(v) for v in a0]
