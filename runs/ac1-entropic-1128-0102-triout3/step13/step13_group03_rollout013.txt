"""
Improved search for a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines several complementary phases:

1️⃣  A richer parametric sweep (geometric × power‑law shapes, mixed with
    the uniform vector) to obtain a strong seed.
2️⃣  Long Adam‑style projected gradient descent on a smooth‑max (log‑sum‑exp)
    surrogate with a higher temperature ceiling (β up to 2000) and a
    slower learning‑rate decay.
3️⃣  Projected sub‑gradient descent on the exact max‑convolution (the true
    objective) – a clean way to refine the shape once the soft‑approximation
    is already close to a local optimum.
4️⃣  A final stochastic hill‑climb (random + targeted mass transfers) for
    polishing.
All phases respect the 1000 s wall‑time budget, printing intermediate scores.
The returned sequence is normalised to sum 1 (any scaling leaves the
objective invariant).
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
#  Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
#  Fast exact evaluation (used only for bookkeeping – sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) for a vector (sum≈1)."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
#  Smooth‑max surrogate utilities (log‑sum‑exp)
# ----------------------------------------------------------------------


def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a * a   (full linear convolution)
        grad = ∂/∂a ( (1/β)·log Σ_k exp(β·conv_k) )
    Gradient is obtained via correlation (FFT).
    """
    conv = _conv_self(a, L_conv)               # length 2n‑1
    max_c = conv.max()

    # Stable soft‑max weights
    shifted = conv - max_c                     # max becomes 0
    exp_vals = np.exp(beta * shifted)          # very small values under‑flow to 0
    w = exp_vals / exp_vals.sum()              # shape (2n‑1,)

    # Correlation w ⋆ a_rev  → gradient
    a_rev = a[::-1]                            # reversed a
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)       # linear convolution w * a_rev

    start = a.shape[0] - 1                     # index that aligns the centre
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected descent on the smooth‑max surrogate.
    β grows geometrically up to beta_max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0               # start smooth
    beta_mul = 1.015         # slow increase

    while time.time() < deadline and t < max_iters:
        t += 1

        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # exact score (sum≈1 ⇒ denominator≈1)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update (projected onto simplex)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.5 / (1.0 + 1e-4 * t)          # slowly decaying learning rate
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
#  Projected sub‑gradient descent on the true max‑convolution term
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    Gradient for the active index k is   g_i = 2·a_{k‑i}   (zero outside range).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    base_step = 0.5

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={t:,} score={cur_score:.6f}"
            )

        k = int(np.argmax(conv))

        # build the sub‑gradient (vectorised)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        grad = np.zeros_like(a)
        grad[i_range] = 2.0 * a[j_range]

        # diminishing step‑size
        eta = base_step / np.sqrt(t)
        a_new = a - eta * grad
        a = _project_simplex(a_new)

    return best_a, best_score


# ----------------------------------------------------------------------
#  Stochastic hill‑climb (random + targeted mass transfers)
# ----------------------------------------------------------------------


def _hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves mixed with targeted moves that
    directly reduce the currently dominant convolution entry.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # step‑size control
    step_factor = 0.15      # up to 15 % of source mass per move
    min_factor = 1e-6
    decay = 0.995

    while time.time() < deadline:
        if rng.random() < 0.6:
            # ----- random transfer -----
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # ----- targeted transfer -----
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))

            # choose a pair (i,j) on the active anti‑diagonal
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            # decide which of the two contributes more to the peak
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose the move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)   # reset step size after success
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
#  Coarse‑to‑fine parametric sweep
# ----------------------------------------------------------------------


def _parametric_sweep(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore the family   a_i ∝ r**i · (i+1)**p   (mixed with uniform)
    on a finer grid and keep the best vector.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    r_grid = np.logspace(np.log10(0.85), np.log10(0.99999), num=40)
    p_grid = np.linspace(-6.0, 4.0, num=81)

    L_fft = _next_pow2(2 * n - 1)

    for r in r_grid:
        if time.time() >= deadline - 180:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 180:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq_sum = seq.sum()
                if seq_sum == 0.0:
                    continue
                seq /= seq_sum
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )
    return best_seq, best_score
