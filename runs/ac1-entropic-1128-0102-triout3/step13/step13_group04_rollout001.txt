assistantfinal**Solution Overview**

The helper `evaluate_sequence` is the only function that scores a candidate step‑function.
We must return a list of non‑negative numbers (the “heights”) whose sum is positive.
The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the ordinary linear convolution of `a` with itself.
After normalising `a` to a probability vector (sum = 1) the objective reduces to  

\[
F(a)=2\,n\;\max (a*a).
\]

The program follows three phases

1. **Parametric sweep** – very fast exploration of a large family of shapes  
   *geometric* (`a_i∝r^i`) and *power‑law* (`a_i∝(i+1)^{-p}`) – each optionally mixed
   with the uniform distribution.  The best vector from this step seeds the later
   optimisation.

2. **Smooth‑gradient Adam optimisation** – the non‑smooth “max” is replaced by a
   smooth Log‑Sum‑Exp approximation  

   \[
   \operatorname{lse}_\beta(c)=\tfrac1\beta\log\sum_k e^{\beta c_k},
   \]

   whose gradient is a weighted sum of the gradients of all shifts.
   The weights are obtained by a soft‑max with temperature `β`.  
   The gradient can be evaluated by a single FFT‑based correlation,
   keeping the per‑iteration cost comparable to the baseline code.
   Adam’s learning‑rate is gradually reduced while `β` is slowly increased,
   giving a coarse‑to‑fine descent.

3. **Stochastic hill‑climbing** – tiny (up to 5 %) mass transfers between two
   positions are accepted only if they improve the true (non‑approximated) 
   objective.  This clean‑up stage removes any remaining rounding artefacts.

All vectors are projected onto the probability simplex after every update,
ensuring feasibility (`a_i≥0`, `∑a_i=1`).  The whole routine respects the
1000 s time limit; the budget is split roughly 15 % sweep, 65 % Adam, 20 % hill‑climb.

```python
"""
Propose a non‑negative step‑function (list of floats) that minimises
2·n·max(conv(a,a)) / (sum(a)**2).

The algorithm consists of:
  1️⃣  A dense parametric sweep over geometric and power‑law families,
      optionally mixed with the uniform distribution.
  2️⃣  A projected Adam optimiser that uses a *smooth* approximation of the
      max‑convolution (log‑sum‑exp).  The gradient is obtained by a single FFT
      correlation, giving a cheap but much more informative direction than the
      previous “single‑active‑shift” sub‑gradient.
  3️⃣  A final stochastic hill‑climb that moves a small random fraction of
      mass between two indices and keeps only improving moves.

All operations respect the 1000 s budget and the simplex constraints
( non‑negative, sum = 1 ).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0 , Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) sub‑gradient of max(conv(a,a))
# ----------------------------------------------------------------------
def _smooth_subgrad(a: np.ndarray,
                    conv: np.ndarray,
                    beta: float = 30.0) -> np.ndarray:
    """
    Compute the gradient of the smoothed max (log‑sum‑exp) of the convolution.
    The gradient is
        grad = 2 * corr(weight, a_rev)
    where weight_k ∝ exp(beta * conv_k) and corr denotes cross‑correlation.
    """
    # avoid overflow – conv values are tiny, so simple exponent is safe
    max_val = conv.max()
    # soft‑max weights (sum to 1)
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        # fallback to the single dominant shift
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
        w_sum = 1.0
    w = w / w_sum

    # correlation between w and a (a reversed)
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(W * A_rev, n=L)

    start = len(a) - 1          # index where correlation aligns with grad[0]
    grad = 2.0 * corr[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser that uses the smooth sub‑gradient
# ----------------------------------------------------------------------
def _adam_optimize_smooth(a0: np.ndarray,
                          start_time: float,
                          deadline: float,
                          best_score_initial: float,
                          max_iter: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex using the smooth gradient.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.25                         # initial learning rate

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    # temperature schedule for the smooth max
    beta_start = 15.0
    beta_end = 200.0
    beta_inc = (beta_end - beta_start) / max_iter

    while time.time() < deadline and it < max_iter:
        it += 1

        # current convolution and objective (sum(a) is 1)
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth gradient
        beta = beta_start + it * beta_inc
        grad = _smooth_subgrad(a, conv, beta=beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple learning‑rate decay
        lr = lr0 / (1.0 + it * 1e-7)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex
        a = _project_simplex(a)

        # occasional random shake to escape plateaus
        if it - last_improve > 20_000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (tiny random mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float,
                max_move_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: move a random fraction (≤max_move_frac) of mass
    from one index to another; keep the move only if it improves the true
    objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction up to max_move_frac
        frac = rng.random() * max_move_frac + 1e-7
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
