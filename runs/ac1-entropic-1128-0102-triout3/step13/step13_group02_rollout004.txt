"""
Improved search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  **Rich parametric sweep** – explores several symmetric families
    (exponential, power‑product and double‑exponential) and mixes each
    with the uniform distribution.  This quickly discovers a very strong
    starting point that is often better than the previous “geometric‑power”
    sweep.

2️⃣  **Projected Adam on a smooth‑max surrogate** – starting from the best
    candidate we run a few restarts of Adam‑style descent on the
    soft‑max approximation of the maximum convolution entry.
    The temperature β is increased aggressively (up to 2000) so the
    optimisation focuses on the true objective in the later phase.
    An adaptive learning‑rate and occasional jitter help escape plateaus.

3️⃣  **Sub‑gradient descent + annealed hill‑climb** – a classic sub‑gradient
    step on the exact max‑convolution is performed with a diminishing step
    size.  Afterwards a small‑temperature simulated‑annealing hill‑climb
    transfers tiny masses between random coordinates, accepting worsening
    moves with a temperature that cools to zero.  This polishing often
    reduces the objective a few 1e‑4 further.

All phases respect the 1000 s budget, keep the vector on the probability
simplex (non‑negative, sum = 1), and finally return the best sequence found
as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0, Σ w = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------


def _family_exponential(n: int, r: float) -> np.ndarray:
    """Symmetric exponential: a_i ∝ r^{min(i, n-1-i)}."""
    idx = np.arange(n, dtype=np.float64)
    d = np.minimum(idx, n - 1 - idx)
    shape = r ** d
    return shape


def _family_power_product(n: int, p: float) -> np.ndarray:
    """Power‑product: a_i ∝ ((i+1)*(n-i))^{p}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    prod = i * (n + 1 - i)
    shape = prod ** p
    return shape


def _family_double_exponential(n: int, r1: float, r2: float) -> np.ndarray:
    """
    Double‑exponential: a_i ∝ r1^{i} + r2^{n-1-i}.
    Allows asymmetric decay on the two ends.
    """
    i = np.arange(n, dtype=np.float64)
    shape = r1 ** i + r2 ** (n - 1 - i)
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _run_parametric_sweep(n: int, L_fft: int, deadline: float, start_time: float):
    """Explore the families defined above and return the best candidate."""
    best_seq = np.full(n, 1.0 / n)
    best_score = _eval_fast(best_seq, L_fft)

    # grid specifications (chosen to be fast yet thorough)
    r_grid = np.logspace(np.log10(0.55), np.log10(0.9999), num=24)
    p_grid = np.linspace(-4.0, 3.0, 41)
    mix_grid = (0.0, 0.2, 0.5, 0.8, 1.0)

    # 1) symmetric exponential
    for r in r_grid:
        if time.time() >= deadline - 180:
            break
        shape = _family_exponential(n, r)
        shape /= shape.sum()
        for lam in mix_grid:
            cand = _mix_with_uniform(shape, lam)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time()-start_time:.2f}s] exp r={r:.4f} λ={lam:.2f} → {sc:.6f}")

    # 2) power‑product
    for p in p_grid:
        if time.time() >= deadline - 180:
            break
        shape = _family_power_product(n, p)
        if shape.sum() == 0.0:
            continue
        shape /= shape.sum()
        for lam in mix_grid:
            cand = _mix_with_uniform(shape, lam)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time()-start_time:.2f}s] prod p={p:.2f} λ={lam:.2f} → {sc:.6f}")

    # 3) double exponential (asymmetric)

def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_score_initial: float):
    """
    Projected Adam on the soft‑max surrogate of max(conv) with an aggressive
    temperature schedule (β up to 2000) and occasional jitter.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 2.0                # start temperature (soft)
    beta_mul = 1.03
    beta_max = 2000.0

    while time.time() < deadline:
        t += 1

        # --- soft‑max gradient -------------------------------------------------
        conv = _conv_self_fft(a, L_conv)            # linear conv, length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                      # stabilise exponentials
        # to avoid overflow when beta large, clip exponentials
        exp_vals = np.exp(np.clip(beta * shifted, -700, 700))
        w = exp_vals / exp_vals.sum()               # soft‑max weights, shape (2n‑1,)

        # correlation w * rev(a)  → gradient of soft‑max
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)        # linear convolution
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # exact objective for monitoring
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] smooth it={t:,} β={beta:.0f} score={cur_score:.6f}")

        # --- Adam update on the simplex ---------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.25 / (np.sqrt(v_hat) + eps)        # slightly larger step
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # jitter every few thousand iterations
        if t % 3000 == 0:
            jitter = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float):
    """
    Classic sub‑gradient descent on the true max‑convolution.
    Uses a diminishing step size and occasional probabilistic acceptance.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.15
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for shift k (O(n))
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

        # diminishing step
        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time()-start_time:.2f}s] subgrad it={it:,} step={step:.4e} score={cur_score:.6f}")
        else:
            # simulated‑annealing style acceptance
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter occasionally
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


def _annealed_hill_climb(a0: np.ndarray,
                         deadline: float,
                         start_time: float):
    """
    Microscopic hill‑climb with a temperature schedule.
    Small mass moves are accepted probabilistically according to a
    decreasing temperature, allowing the search to escape shallow local minima.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    rng = np.random.default_rng()
    it = 0

    # temperature schedule (linear decay)
    T0 = 1e-3
    T_min = 1e-7
    total_iters_est = max(1, int((deadline - time.time()) / 0.0004))

    while time.time() < deadline:
        it += 1
        # linear cooling
        T = max(T_min, T0 * (1.0 - it / total_iters_est))

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 5e-4   # up to 0.05 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        Δ = new_score - best_score
        if Δ < 0 or rng.random() < np.exp(-Δ / T):
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] anneal hill it={it:,} T={T:.2e} score={best_score:.6f}")

    return best_a, best_score

            if sc_smooth < best_score:
                best_score, best_seq = sc_smooth, a_smooth.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam restart {i+1}/{n_restarts} improved to {sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        sub_deadline = deadline - 120
        a_sub, sc_sub = _subgradient_descent(
            best_seq, n, sub_deadline, start_time, best_score)
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient refined score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final annealed hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _annealed_hill_climb(best_seq, deadline, start_time)
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final annealed hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
