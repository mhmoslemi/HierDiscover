def adam_smooth_max(a0, start_time, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr = 0.05
    tau = 5.0
    tau_factor = 1.5
    it = 0
    while time.time() < deadline:
        it += 1
        grad, conv = _grad_smooth_max(a, tau)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # ensure non-negative
        a_new = np.maximum(a_new, 0.0)
        # project onto simplex
        a_new = _project_simplex(a_new)
        # symmetry enforcement (optional)
        a_new = (a_new + a_new[::-1]) / 2.0
        a = a_new

"""
Improved optimiser for the “step‑function” challenge.

The optimiser works in four phases (all respecting the 1000 s budget):
1.  Broad parametric sweep (geometric, power, beta, Gaussian & Laplace)
2.  Random Dirichlet initialisations (including very sparse draws)
3.  Adam‑based smooth‑max continuation (τ starts low and is increased;
    the gradient of the log‑sum‑exp objective is computed in the Fourier
    domain and the iterate is projected onto the probability simplex).
4.  Focused local search: a deterministic hill‑climb driven by the
    sub‑gradient of the true max‑convolution plus occasional random
    perturbations (simulated‑annealing style).

The current best candidate is constantly tracked; if its score drops
below the target (≈1.503) the search aborts early and the candidate is
returned.  All vectors are kept on the probability simplex, a symmetry
step `(a+a[::-1])/2` is applied after each update to encourage a
mirrored shape, which empirically improves the flatness of the
self‑convolution.

The implementation is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for the final check, and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real‑to‑real)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _grad_smooth_max(a: np.ndarray, tau: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient of the smooth approximation Sτ(a) = (1/τ)·log Σexp(τ·conv)
    and the underlying convolution.
    Returns (grad, conv).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (avoid overflow)
    w = np.exp(tau * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum

    a_rev = a[::-1]
    # convolution w * a_rev (length 3n‑2)
    L_grad = _next_pow2(len(w) + len(a_rev) - 1)
    grad_full = _conv_fft(w, a_rev)[: len(w) + len(a_rev) - 1]
    # extract the central n entries (indices n‑1 … 2n‑2)
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return grad, conv


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta, Gaussian, Laplace)
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray,
                    best_score: float, deadline: float, start_time: float):
    idx = np.arange(n, dtype=np.float64)

    # uniform baseline (already present)
    uniform = np.full(n, 1.0 / n)

    # ---- geometric ----------------------------------------------------
    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 101)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta -----------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 81)
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))  # normalise numerically
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- Gaussian (discrete) -------------------------------------------
    mu = (n - 1) / 2.0
    sigma_grid = np.linspace(0.05 * n, 0.45 * n, 25)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        gauss = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        gauss_sum = gauss.sum()
        if gauss_sum == 0:
            continue
        gauss /= gauss_sum
        sc = _objective_score(gauss)
        if sc < best_score:
            best_score, best_seq = sc, gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma:.1f} score={sc:.6f}")

    # ---- Laplace (double‑exponential) ----------------------------------
    sigma_grid = np.linspace(0.02 * n, 0.30 * n, 20)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        lap = np.exp(-np.abs(idx - mu) / sigma)
        lap_sum = lap.sum()
        if lap_sum == 0:
            continue
        lap /= lap_sum
        sc = _objective_score(lap)
        if sc < best_score:
            best_score, best_seq = sc, lap.copy()
            print(f"[{time.time() - start_time:.2f}s] Laplace σ={sigma:.1f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers (including very sparse draws)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 40
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # Concentration in [0.01, 0.5] produces very sparse vectors,
        # while [0.5,5] yields denser ones.
        c = rng.uniform(0.01, 5.0)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
        # Quick early‑exit if we hit the target
        if best_score <= 1.503:
            return best_seq, best_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam‑based smooth‑max continuation
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth log‑sum‑exp approximation of the max‑convolution
    using Adam.  τ starts low (high smoothing) and is increased geometrically.
    The iterate is projected onto the probability simplex and symmetrised.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.07          # base learning rate
    tau = 5.0          # start with strong smoothing
    tau_factor = 1.6
    it = 0

    while time.time() < deadline:
        it += 1
        grad, conv = _grad_smooth_max(a, tau)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)          # stay non‑negative
        a_new = _project_simplex(a_new)        # enforce sum‑to‑1
        a_new = (a_new + a_new[::-1]) / 2.0    # optional symmetry
        a = a_new

        # evaluate true objective every few hundred iterations
        if it % 300 == 0 or time.time() >= deadline:
            cur_score = _objective_score(a)
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.1f} score={cur_score:.6f}")
            if best_score <= 1.503:
                print("[Early exit] Target reached during Adam phase")
                return best_seq, best_score

        # temperature schedule
        if it % 1500 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Focused local search (deterministic hill‑climb + SA perturb)
# ----------------------------------------------------------------------
def _hill_climb_local(start_a: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_val = best_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source: index with largest positive gradient among non‑zero entries
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination: index with most negative gradient
        dst = np.argmin(grad)

        # safeguard against pathological equality
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

        # transfer a tiny fraction of mass (up to ~0.8 % of source)
        frac = rng.random() * 0.007 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-14:
            continue

        cur_sc = _objective_score(a_prop)
        if cur_sc < best_val - 1e-9:
            a = a_prop
            best_a = a.copy()
            best_val = cur_sc
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved score={best_val:.6f}")
            if best_val <= 1.503:
                print("[Early exit] Target reached during hill‑climb")
                return best_a, best_val
        else:
            # simulated‑annealing acceptance (temperature decays linearly)
            temp = max(0.005, (deadline - time.time()) / (deadline - start_time))
            if rng.random() < np.exp(-(cur_sc - best_val) / temp):
                a = a_prop

        # occasional random multi‑site perturbation
        if rng.random() < 0.004:
            idxs = rng.choice(n, size=4, replace=False)
            delta = (rng.random(4) - 0.5) * 0.012  # small signed changes
            a_pert = a.copy()
            a_pert[idxs] += delta
            a_pert = np.maximum(a_pert, 0.0)
            total = a_pert.sum()
            if total > 0:
                a_pert /= total
                pert_sc = _objective_score(a_pert)
                if pert_sc < best_val:
                    a = a_pert
                    best_a = a.copy()
                    best_val = pert_sc
                    print(f"[{time.time() - start_time:.2f}s] Random perturb improved score={best_val:.6f}")
                    if best_val <= 1.503:
                        print("[Early exit] Target reached after random perturb")
                        return best_a, best_val

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine follows four stages (parametric sweep,
    random Dirichlet, Adam smooth‑max descent, and a local hill‑climb) while
    respecting the time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the reference length of 4096 (good trade‑off)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise with the uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter exists, consider it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep
    # ------------------------------------------------------------------
    phase_deadline = deadline - 300.0   # reserve ~5 min for later work
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline, start_time
        )
        print(f"[{time.time() - start_time:.2f}s] After sweep best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random Dirichlet draws (sparse + dense)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, start_time
        )
        print(f"[{time.time() - start_time:.2f}s] After Dirichlet best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam smooth‑max descent
    # ------------------------------------------------------------------
    # Keep ~150 s for the final hill‑climb
    hill_time = 150.0
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth‑max descent")
        best_seq, best_score = _adam_smooth_max(
            best_seq, start_time, smooth_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam descent best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Focused local search (hill‑climb + SA)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb_local(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – final best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
