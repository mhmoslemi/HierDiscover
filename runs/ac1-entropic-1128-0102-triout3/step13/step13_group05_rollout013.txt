"""propose_candidate
A multi–stage optimizer for the step‑function bound.

* Stage 1 – a diverse initial pool (uniform, power‑ramps, Beta‑densities,
  many Dirichlet draws).
* Stage 2 – refined descent on the true L∞‑objective via a sub‑gradient
  method (Adam on an active‑set approximation of the max‑convolution).
  The best vectors from the pool are each polished for a slice of the
  remaining budget.
* Stage 3 – many short random restarts, each followed by the same
  sub‑gradient descent, to explore distant basins.
* Stage 4 – a final polish of the overall best candidate.

All updates keep the vector on the probability simplex (non‑negative,
unit sum).  The resulting sequence is up‑sampled to 65536 entries
(the length used by the reference solution) and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {a ≥ 0, Σa = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes used for the initial pool
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau, normalised to sum 1."""
    # clamp alpha to a reasonable range for the plateau proportion
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # pad / truncate to exact length
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1.
    α < 1 gives a U‑shaped distribution.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Small helper to keep a fixed‑size pool of the best candidates
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient descent on the L∞‑objective (Adam optimizer)
# ----------------------------------------------------------------------
def _max_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style descent on the true max‑convolution objective.
    The gradient is computed from an active set (the largest convolution
    entries).  The iterate is kept on the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} "
                      f"best={cur_score:.6f}")

        # ----- active set -------------------------------------------------
        # pick a small set of the largest convolution entries
        k_active = max(1, int(0.001 * conv.size))          # ≈0.1 % of positions
        idx_top = np.argpartition(-conv, k_active - 1)[:k_active]
        w = np.zeros_like(conv)
        w[idx_top] = 1.0 / k_active                         # normalised weights

        # ----- gradient via correlation (w * a_rev) ----------------------
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)                           # length for corr
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # ----- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.12 / math.sqrt(it)                        # base learning rate
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -------------------------------
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 123456)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Choose an internal resolution – we adapt to the remaining time.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # 1) Build a diverse initial pool of candidates
    # ------------------------------------------------------------------
    TOP_K = 7
    top = []                     # list of (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(top, _objective(uniform), uniform, TOP_K)

    # power‑ramps (wide grid for p and plateau proportion)
    p_vals = np.linspace(0.1, 3.0, 20)                # 20 values
    plat_vals = np.linspace(0.0, 0.5, 11)             # 11 values
    for p in p_vals:
        for alpha in plat_vals:
            a = _make_shape(N, p, alpha)
            _push_top(top, _objective(a), a, TOP_K)

    # Beta‑shapes – include very small alphas (strong U‑shape)
    beta_alphas = np.concatenate([
        np.linspace(0.02, 0.5, 25),      # fine near 0
        np.linspace(0.55, 2.0, 15)      # hump region for completeness
    ])
    for alpha in beta_alphas:
        a = _make_beta_shape(N, alpha)
        _push_top(top, _objective(a), a, TOP_K)

    # random Dirichlet samples (concentration < 1 → sparse)
    for _ in range(60):
        if time.time() >= deadline:
            break
        a = rng.gamma(0.3, 1.0, size=N)
        a /= a.sum()
        _push_top(top, _objective(a), a, TOP_K)

    # ------------------------------------------------------------------
    # Keep the current best
    # ------------------------------------------------------------------
    best_score, best_vec = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Refine the top candidates using the max‑gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0:
        # allocate roughly half of the remaining budget to this stage
        refine_budget = remaining * 0.5
        per_candidate = refine_budget / len(top)
        for idx, (_, vec) in enumerate(top):
            sub_deadline = time.time() + per_candidate
            cand_vec, cand_score = _max_gradient_descent(vec, sub_deadline, start)
            print(f"[{time.time() - start:.2f}s] Refine top {idx} → {cand_score:.6f}")
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand_vec.copy()

    # ------------------------------------------------------------------
    # 3) many short random restarts + quick descent
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:          # keep a few seconds margin
        a0 = rng.gamma(0.5, 1.0, size=N)
        a0 /= a0.sum()
        sub_deadline = time.time() + 5.0
        cand_vec, cand_score = _max_gradient_descent(a0, sub_deadline, start)
        if cand_score < best_score:
            best_score, best_vec = cand_score, cand_vec.copy()
            print(f"[{time.time() - start:.2f}s] New global best = {cand_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final polish of the overall best candidate (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_vec, final_score = _max_gradient_descent(best_vec, deadline, start)
        if final_score < best_score:
            best_score, best_vec = final_score, final_vec.copy()
            print(f"[{time.time() - start:.2f}s] Final polish best = {final_score:.6f}")

    # ------------------------------------------------------------------
    # Upsample to the canonical length 65536 (optional – keeps shape)
    # ------------------------------------------------------------------
    TARGET_N = 65536
    if best_vec.shape[0] != TARGET_N:
        x_old = np.linspace(0, 1, best_vec.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, TARGET_N, endpoint=False)
        best_vec = np.interp(x_new, x_old, best_vec)

    # ------------------------------------------------------------------
    # Normalise again and return a plain Python list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(TARGET_N, 1.0 / TARGET_N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return best_vec.tolist()
