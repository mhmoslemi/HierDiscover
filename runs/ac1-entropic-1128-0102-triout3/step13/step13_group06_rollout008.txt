import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv_full


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective for a vector with sum=1."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate shape generators
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of analytic shapes (unnormalised)."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]
    cand = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        cand.append(np.minimum(x, 1.0 - x) ** p)

    # 2) Exponential decay from the centre (symmetric)
    for lam in np.logspace(-4, -0.5, 12):
        cand.append(np.exp(-lam * np.abs(x - 0.5)))

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        cand.append((ix ** -p) + ((n + 1 - ix) ** -p))

    # 4) Inverse‑sqrt (U‑shaped)
    inv = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    cand.append(inv + inv[::-1])

    # 5) Half‑cosine (sine window)
    cand.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        cand.append(np.exp(-0.5 * ((x - 0.5) / sigma) ** 2))

    # 7) Classical cosine windows
    cand.append(0.5 - 0.5 * np.cos(2 * np.pi * x))                     # Hann
    cand.append(0.54 - 0.46 * np.cos(2 * np.pi * x))                 # Hamming
    cand.append(0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))  # Blackman

    # 8) Kaiser windows (beta from 0 to 12)
    try:
        from scipy.special import i0
        for beta in np.linspace(0.0, 12.0, 13):
            t = (np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)
            w = i0(beta * np.sqrt(1.0 - t * t)) / i0(beta)
            cand.append(w)
    except Exception:
        pass

    # 9) DPSS (Slepian) windows
    try:
        from scipy.signal import windows as sigw
        for NW in np.linspace(2.0, 6.0, 9):
            dpss = sigw.dpss(n, NW, Kmax=1, return_ratios=False)[0]
            cand.append(np.abs(dpss))
    except Exception:
        pass

    # 10) Random Dirichlet‑like draws
    rng = np.random.default_rng(0)
    for _ in range(8):
        rnd = rng.random(n)
        cand.append(rnd)

    # Normalise and keep only those with positive mass
    normals = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Gradient‑descent utilities
# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax producing a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _grad_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of 2·n·max(conv) (smooth version) with respect to `a`.
    We have ∂conv_k/∂a_j = 2·a_{k‑j}.  Using the soft‑max weights `w`
    (∑_k w_k = 1) the gradient becomes:
        g_j = 2 * Σ_k w_k * a_{k‑j}.
    This is a cross‑correlation and is computed via FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(a_rev) + len(w) - 1)
    Af = np.fft.rfft(a_rev, n=L)
    Bf = np.fft.rfft(w, n=L)
    cw = np.fft.irfft(Af * Bf, n=L)[: len(a_rev) + len(w) - 1]
    # slice the central n entries (indices n‑1 … n‑1+n‑1)
    g = 2.0 * cw[n - 1 : n - 1 + n]
    return g

    # Adam optimiser state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # Schedules
    lr0 = 0.12                     # initial learning rate
    tau0 = 0.10                    # initial soft‑max temperature
    tau_min = 0.001                # final temperature
    total_budget = budget_s - 0.5

    iteration = 0
    rng = np.random.default_rng(seed + 12345)

    while time.time() < deadline:
        iteration += 1

        # ------- current probability vector & its convolution ----------
        a = _softmax(y)                     # always non‑negative, sums to 1
        conv = _conv_fft(a)                 # full length 2n‑1
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max

        # record best ever seen
        if cur_obj < best_score:
            best_score = cur_obj
            best_vec = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={iteration:,} new best={best_score:.6f}")

        # ------- temperature and learning‑rate schedules ---------------
        elapsed = time.time() - start_time
        progress = elapsed / total_budget            # in [0,1]
        # exponential decay
        tau = tau0 * (tau_min / tau0) ** progress
        lr = lr0 * (0.1 ** progress)                 # final lr is 0.01*lr0

        # ------- soft‑max weights approximating the max ----------------
        # stability: subtract the actual maximum before exponentiation
        shift = conv - cur_max
        w = np.exp(shift / max(tau, 1e-12))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:

        grad_y = a * (g_a - a_dot_g)                # shape (n,)

        # ------- Adam update -----------------------------------------

        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        y -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional tiny random perturbation to avoid stagnation
        if iteration % 5000 == 0:
            noise_scale = 0.02 * (1.0 - progress)
            y += noise_scale * rng.standard_normal(n)

    # --------------------------------------------------------------
    # 3) Final sanitisation
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0.0:
        best_vec /= total
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
