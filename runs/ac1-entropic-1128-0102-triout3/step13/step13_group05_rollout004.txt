def _make_exponential_shape(N, c):
    # shape = exp(-c* i) (or linear) normalized.
    i = np.arange(N, dtype=float)
    shape = np.exp(-c * i)
    total = shape.sum()
    if total>0:

    else:

    shape = i ** (-p)
    total = shape.sum()
    if total>0: shape/=total

"""
Hybrid simulated‑annealing + Adam‑style smooth‑max descent.

1.  Build a diverse initial pool (uniform, power‑ramps,
    Beta‑shapes, exponential, power‑law and sparse Dirichlet draws).
2.  Run a short, relatively hot simulated‑annealing phase to jump
    out of poor basins.
3.  Polish the best vector with a long smooth‑max (soft‑max) gradient
    descent that uses Adam updates and projects onto the probability
    simplex after each step.
4.  Return the normalised non‑negative sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential_shape(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i), normalised."""
    i = np.arange(n, dtype=float)
    shape = np.exp(-c * i)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_powerlaw_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    i = np.arange(n, dtype=float) + 1.0
    shape = i ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Top‑K pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k list (ascending scores)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Smooth‑max descent (Adam style)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam descent on a smooth surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.08          # base learning rate
    T0 = 0.02
    T_end = 1e-6

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} score={cur_score:.6f}")

        # temperature decays linearly over the whole polishing phase
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient = 2 * (w * a_rev) correlation
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1: n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best_a, best_score

# Main entry point for the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,

    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ---- problem size -------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # ---- initial pool -------------------------------------------------
    TOP_K = 7
    top = []   # (score, vector)

    uniform = np.full(N, 1.0 / N)

            _push_top(top, _objective(a), a, TOP_K)

    for alpha in np.linspace(0.1, 1.2, 12):

    # power‑law family
    for p in np.linspace(0.5, 2.5, 11):
        a = _make_powerlaw_shape(N, p)

    # sparse Dirichlet draws (different concentrations)

            s = a.sum()
            if s > 0:
                a /= s

    # previous global starter, if present

                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            tot = starter_up.sum()
            if tot > 0:
                starter_up /= tot
                _push_top(top, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # best of the pool becomes the starting point
    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ---- simulated annealing (exploratory) ---------------------------
    # allocate roughly 25 % of the total budget to SA
    total_time = deadline - start_time
    sa_deadline = start_time + total_time * 0.25

    cur_seq = best_seq.copy()
    cur_score = best_score
    best_global_seq = cur_seq.copy()
    best_global_score = cur_score

    max_move_factor = 0.15          # up to 15 % of source mass per move
    T_start = 0.3
    T_end = 1e-7

    it = 0
    while time.time() < sa_deadline:
        it += 1

        # linear cooling (more gentle than exponential)
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / total_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # source index biased by current weight
        src = rng.choice(N, p=cur_seq)
        if cur_seq[src] < 1e-15:
            continue

        else:
            dst = rng.integers(N)
        if src == dst:
            continue

        if delta <= 0.0:
            continue

        a_prop[src] -= delta

        delta_score = new_score - cur_score

        if delta_score < 0.0 or rng.random() < math.exp(-delta_score / T):
            cur_seq = a_prop
            cur_score = new_score
            if new_score < best_global_score:
                best_global_score = new_score
                best_global_seq = cur_seq.copy()

            sigma = 0.002
            noise = rng.normal(0.0, sigma, size=N)
            a_noisy = cur_seq + noise
            a_noisy = np.maximum(a_noisy, 0.0)
            s = a_noisy.sum()
            if s > 0:
                a_noisy /= s
                cur_seq = a_noisy
                cur_score = _objective(cur_seq)

    # ---- smooth‑max polishing (majority of the budget) ---------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_deadline = time.time() + remaining
        refined_seq, refined_score = _smooth_gradient_descent(best_global_seq,
                                                             sub_deadline,
                                                             start_time)
        if refined_score < best_global_score:
            best_global_score = refined_score
            best_global_seq = refined_seq
            print(f"[{time.time() - start_time:.2f}s] Polished best = {best_global_score:.6f}")

    best_global_seq = np.maximum(best_global_seq, 0.0)
    total = best_global_seq.sum()
    if total > 0:

    else:
        best_global_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_global_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_global_seq]
