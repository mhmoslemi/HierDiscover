"""
A multi‑stage optimizer for the step‑function evaluation problem.

The algorithm works on a fixed length `n` (here 8192) and tries to
minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

subject to a_i ≥ 0.  The search consists of

1️⃣  a fast geometric‑power sweep on a moderate size (4096) to obtain a
    promising shape,
2️⃣  up‑sampling the best sweep candidate to the target length,
3️⃣  a smooth‑max Adam descent (soft‑max surrogate for the max‑convolution)
    with an annealing temperature,
4️⃣  an exponentiated‑gradient sub‑gradient descent that works on the
    exact maximal convolution entry,
5️⃣  a final simulated‑annealing polishing phase that moves tiny amounts of
    mass between coordinates.

All phases respect the global time budget (default 1000 s) and keep the
vector on the probability simplex.  The best sequence found is returned as
a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w≥0, sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    `a` is assumed to be non‑negative; it need not be normalised.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2: smooth‑max Adam descent (soft‑max surrogate)
# ----------------------------------------------------------------------


def _smooth_max_adam(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected descent on the smooth‑max surrogate of the
    maximal-self‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0                # starting temperature
    beta_mul = 1.02
    beta_max = 2500.0         # a larger cap than before

    while time.time() < deadline:
        t += 1

        # ----- soft‑max of the convolution -----
        conv = _conv_self_fft(a, L_conv)          # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                    # for numerical stability
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()             # soft‑max weights

        # ----- gradient of the smooth‑max -----
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)      # linear convolution
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # ----- objective monitoring -----
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth‑Adam it={t:,} β={beta:.1f} "
                f"score={cur_score:.6f}"
            )

        # ----- Adam update on the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # ----- temperature schedule -----
        beta = min(beta * beta_mul, beta_max)

        # ----- occasional jitter -----
        if t % 4000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: exponentiated‑gradient sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def _subgrad_exponential(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent on the exact maximal convolution
    entry (the true objective).  The update is multiplicative, which
    automatically keeps the vector non‑negative; a simplex projection restores
    the unit sum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.5                # initial step magnitude (tuned empirically)

    it = 0
    L_fft = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1

        # ----- locate maximal convolution entry -----
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # ----- sub‑gradient for shift k (vectorised) -----
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals

        g = np.zeros_like(a)
        g[i_vals] += a[j_vals]
        g[j_vals] += a[i_vals]          # double contribution when i==j → 2*a_i

        # ----- exponentiated‑gradient step -----
        eta = step0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        a = a / a.sum()                  # simplex normalisation (projected)

        # ----- occasional jitter (helps escape plateaus) -----
        if it % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

        cur_score = _objective(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑subgrad it={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4: simulated‑annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Tiny‑mass transfer moves with a simple temperature schedule.
    Improves the best solution found by occasional uphill steps.
    """
    a_cur = a0.copy()
    cur_score = _objective(a_cur, _next_pow2(2 * n - 1))
    best_a = a_cur.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()

    # initial temperature: a fraction of the current objective
    T0 = max(1e-5, best_score * 0.002)

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)
        T = max(1e-7, T0 * (1.0 - progress))

        # propose a tiny transfer
        src = rng.integers(n)
        if a_cur[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # up to 0.5 % of the source mass
        eps = a_cur[src] * rng.random() * 5e-3
        a_prop = a_cur.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # re‑normalise (sum is already preserved, but numerical errors may appear)
        a_prop = a_prop / a_prop.sum()

        new_score = _objective(a_prop, _next_pow2(2 * n - 1))
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a_cur = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Miscellaneous helpers
# ----------------------------------------------------------------------


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D sequence to a new length.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.astype(float).copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    return np.interp(x_new, x_old, seq).astype(float)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine mixes a parametric sweep, smooth‑max Adam
    optimisation, exponentiated‑gradient sub‑gradient descent and a final
    simulated‑annealing polishing phase.  It respects the 1000 s budget
    and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – larger than the previous 4096, still cheap for FFTs
    # ------------------------------------------------------------------
    n_target = 8192
    L_fft_target = _next_pow2(2 * n_target - 1)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n_target, 1.0 / n_target)
    best_seq = uniform.copy()
    best_score = _objective(best_seq, L_fft_target)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (if provided)
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            starter = _interpolate(starter, n_target)
            cur = _objective(starter, L_fft_target)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass  # no global starter available

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep on a smaller n (4096) to get a good shape
    # ------------------------------------------------------------------
    n_sweep = 4096
    L_fft_sweep = _next_pow2(2 * n_sweep - 1)
    idx = np.arange(1, n_sweep + 1, dtype=np.float64)

    sweep_best_seq = None
    sweep_best_score = float("inf")

    r_grid = np.logspace(np.log10(0.90), np.log10(0.99999), num=35)
    p_grid = np.linspace(-3.5, 2.5, 45)
    lam_vals = [0.0, 0.2, 0.5, 0.8, 1.0]
    cut_fracs = [1.0, 0.9, 0.8, 0.7]

    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        geo = r ** np.arange(n_sweep, dtype=np.float64)          # r^i
        for p in p_grid:
            if time.time() >= deadline - 120:
                break
            shape = geo * (idx ** p)
            for cf in cut_fracs:
                m = int(cf * n_sweep)
                shape_cut = shape.copy()
                if m < n_sweep:
                    shape_cut[m:] = 0.0
                for lam in lam_vals:
                    cand = lam * np.full(n_sweep, 1.0 / n_sweep) + (1.0 - lam) * shape_cut
                    s = cand.sum()
                    if s == 0.0:
                        continue
                    cand /= s
                    sc = _objective(cand, L_fft_sweep)
                    if sc < sweep_best_score:
                        sweep_best_score = sc
                        sweep_best_seq = cand.copy()
                        print(
                            f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} "
                            f"cut={cf:.1f} λ={lam:.2f} score={sc:.6f}"
                        )

    if sweep_best_seq is not None:
        upscale = _interpolate(sweep_best_seq, n_target)
        sc_up = _objective(upscale, L_fft_target)
        if sc_up < best_score:
            best_seq = upscale.copy()
            best_score = sc_up
            print(f"[{time.time() - start_time:.2f}s] up‑scaled sweep best score={sc_up:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam descent (multiple restarts)
    # ------------------------------------------------------------------
    time_for_smooth = max(0.0, deadline - time.time() - 200)
    if time_for_smooth > 30:
        n_restarts = 2
        per_restart = time_for_smooth / n_restarts
        for i in range(n_restarts):
            if time.time() >= deadline - 200:
                break
            rng = np.random.default_rng(seed + i + 101)
            a0 = best_seq.copy()
            noise = rng.normal(scale=0.015, size=n_target)
            a0 = np.maximum(a0 + noise, 0.0)
            a0 = _project_simplex(a0)

            sub_deadline = time.time() + per_restart
            a_smooth, sc_smooth = _smooth_max_adam(
                a0, n_target, sub_deadline, start_time, best_score
            )
            if sc_smooth < best_score:
                best_seq = a_smooth.copy()
                best_score = sc_smooth
                print(
                    f"[{time.time() - start_time:.2f}s] smooth restart {i+1} improved "
                    f"score={sc_smooth:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 100:
        sub_deadline = deadline - 100
        a_sub, sc_sub = _subgrad_exponential(
            best_seq, n_target, sub_deadline, start_time, best_score
        )
        if sc_sub < best_score:
            best_seq = a_sub.copy()
            best_score = sc_sub
            print(
                f"[{time.time() - start_time:.2f}s] exp‑subgradient improved score={sc_sub:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _simulated_annealing(
            best_seq, n_target, deadline, start_time, best_score
        )
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(
                f"[{time.time() - start_time:.2f}s] simulated‑annealing final score={sc_final:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n_target, 1.0 / n_target)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
