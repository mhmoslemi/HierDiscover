     two non‑zero entries,
   * accept the move if it lowers the peak; otherwise accept with a
     Metropolis probability based on an exponential cooling schedule.
   * keep the best ever seen vector.
   The update uses only NumPy slice operations – no FFT inside the
   inner loop – allowing many millions of moves within the 1000 s
   budget.
4. **Polish** – if time remains, run the soft‑max Adam descent from
   the best vector (the routine from the previous solution).
5. **Return** – normalise the final vector to sum 1 and return it as a
   plain Python list.

The method dramatically increases the number of cheap Monte‑Carlo
steps while directing moves towards coordinates that actually affect
the current worst convolution entry, which yields better scores than
the plain SA in the starter code.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (used only for the initial evaluation of a few candidates)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective_fft(seq: np.ndarray) -> float:
    """Exact objective using FFT (fast enough for a few dozen vectors)."""
    n = seq.shape[0]
    conv = _conv_self_fft(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

# ----------------------------------------------------------------------
# Fast O(N) update of the convolution after moving δ mass
# ----------------------------------------------------------------------
def _conv_update_move(a: np.ndarray,
                      conv: np.ndarray,
                      src: int,
                      dst: int,
                      delta: float) -> np.ndarray:
    """
    Return a *new* convolution after moving `delta` from `src` to `dst`.
    `a` is the old vector (unchanged), `conv` is the old convolution.
    The update follows
        conv' = conv + 2·δ·(shift(dst) – shift(src)) + Δ*Δ,
    where shift(i)[k] = a[k‑i] (zero outside the support) and
    Δ*Δ contributes at indices 2·src, 2·dst and src+dst.
    """
    n = a.shape[0]
    m = conv.shape[0]                # = 2·n‑1
    new_conv = conv.copy()
    # linear part: 2·δ·(shift(dst) – shift(src))
    new_conv[src:src + n] -= 2.0 * delta * a
    new_conv[dst:dst + n] += 2.0 * delta * a

    # quadratic part Δ*Δ
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    d2 = delta * delta
    if idx2src < m:
        new_conv[idx2src] += d2
    if idx2dst < m:
        new_conv[idx2dst] += d2
    if idxsum < m:
        new_conv[idxsum] -= 2.0 * d2
    return new_conv

# ----------------------------------------------------------------------
# Analytic shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape.fill(1.0 / n)
    return shape


def _make_power_ramp(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “tent‑with‑plateau’’ shape.
    `p` is the ramp exponent, `alpha` controls the relative plateau width.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape.fill(1.0 / n)
    return shape

# ----------------------------------------------------------------------
# Top‑K pool helper
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending scores)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# ----------------------------------------------------------------------
# Smooth‑max polishing (same implementation as the starter code)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a soft‑max surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_fft(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.05
    T0 = 0.02
    T_end = 1e-4
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature schedule (linear freeze)
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights over the convolution vector
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient = correlation of w with reversed a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_simplex(a)

    return best_a, best_score


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1) Choose problem size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384
    M = 2 * N - 1
    factor = 2.0 * N                     # denominator will be sum^2 = 1

    # --------------------------------------------------------------
    # 2) Build an initial pool of candidates
    # --------------------------------------------------------------
    TOP_K = 7
    top = []   # (score, vector)

    # uniform
    uniform = np.full(N, 1.0 / N)
    _push_top(top, _objective_fft(uniform), uniform, TOP_K)

    # power‑ramps
    for p in np.linspace(0.3, 2.5, 7):
        for a in np.linspace(0.0, 0.5, 5):
            vec = _make_power_ramp(N, p, a)
            _push_top(top, _objective_fft(vec), vec, TOP_K)

    # beta‑U‑shapes (α<1)
    for alpha in np.linspace(0.05, 0.5, 10):
        vec = _make_beta_shape(N, alpha)
        _push_top(top, _objective_fft(vec), vec, TOP_K)

    # random Dirichlet (sparse / medium)
    rng = np.random.default_rng(seed + 123456)
    for conc in (0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            _push_top(top, _objective_fft(a), a, TOP_K)

    # use the global starter if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size:
            # up‑sample / interpolate to N
            if starter.size != N:
                if N % starter.size == 0:
                    factor_up = N // starter.size
                    starter_up = np.repeat(starter, factor_up)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _push_top(top, _objective_fft(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # current best
    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Gradient‑guided simulated annealing
    # --------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_conv = _conv_self_fft(cur_seq)
    cur_max = cur_conv.max()
    cur_score = factor * cur_max / (cur_seq.sum() ** 2)   # sum is 1 → factor*cur_max
    best_global_seq = cur_seq.copy()
    best_global_score = cur_score

    # annealing parameters
    T_start = 0.3
    T_end = 1e-7
    max_move_factor = 0.25            # up to 25 % of source mass per move
    it = 0
    last_improve_iter = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * (T_end / T_start) ** frac

        # ----------------------------------------------------------
        # Identify the peak and compute its contribution vector g
        # ----------------------------------------------------------
        k = int(np.argmax(cur_conv))          # index of the current peak
        # g_i = 2 * a_{k-i} when the index is valid, else 0
        g = np.zeros(N, dtype=float)
        j_start = max(0, k - (N - 1))
        j_end = min(N - 1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            g[idx] = 2.0 * cur_seq[k - idx]

        # ----------------------------------------------------------
        # Choose a source (large contribution) and a destination
        # ----------------------------------------------------------
        # source: weighted by g (add epsilon to avoid empty)
        eps = 1e-15
        src_weights = g + eps
        src_weights_sum = src_weights.sum()
        if src_weights_sum == 0.0:
            src = rng.integers(N)
        else:
            src_probs = src_weights / src_weights_sum
            src = rng.choice(N, p=src_probs)

        # destination: inverse weighting (small contribution)
        inv_weights = 1.0 / (g + eps)
        inv_weights_sum = inv_weights.sum()
        dst_probs = inv_weights / inv_weights_sum
        dst = rng.choice(N, p=dst_probs)

        if src == dst or cur_seq[src] < 1e-18:
            continue

        # ----------------------------------------------------------
        # Propose a move
        # ----------------------------------------------------------
        delta = cur_seq[src] * max_move_factor * rng.random()
        if delta <= 0.0:
            continue

        # compute tentative convolution
        new_conv = _conv_update_move(cur_seq, cur_conv, src, dst, delta)
        new_max = new_conv.max()
        new_score = factor * new_max / (cur_seq.sum() ** 2)   # sum unchanged = 1

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            # apply the move
            cur_seq[src] -= delta
            cur_seq[dst] += delta
            cur_conv = new_conv
            cur_max = new_max
            cur_score = new_score

            if new_score < best_global_score:
                best_global_score = new_score
                best_global_seq = cur_seq.copy()
                last_improve_iter = it
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

        # ----------------------------------------------------------
        # Occasional diversification
        # ----------------------------------------------------------
        # Restart if we have stagnated for a long time
        if it % 200000 == 0:
            if it - last_improve_iter > 500000:
                # pick a random top‑k candidate as new start
                choice = rng.integers(len(top))
                cur_seq = top[choice][1].copy()
                cur_conv = _conv_self_fft(cur_seq)
                cur_max = cur_conv.max()
                cur_score = factor * cur_max
                print(f"[{time.time() - start_time:.2f}s] Restarted from top‑candidate")
            else:
                # small Gaussian jitter to shake out of tiny traps
                sigma = 0.003
                noise = rng.normal(0.0, sigma, size=N)
                a_noisy = cur_seq + noise
                a_noisy = np.maximum(a_noisy, 0.0)
                s = a_noisy.sum()
                if s > 0:
                    a_noisy /= s
                    cur_seq = a_noisy
                    cur_conv = _conv_self_fft(cur_seq)
                    cur_max = cur_conv.max()
                    cur_score = factor * cur_max

    # --------------------------------------------------------------
    # 4) Optional polishing with smooth‑max descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining
        refined_seq, refined_score = _smooth_gradient_descent(best_global_seq,
                                                             sub_deadline,
                                                             start_time)
        if refined_score < best_global_score:
            best_global_score = refined_score
            best_global_seq = refined_seq
            print(f"[{time.time() - start_time:.2f}s] Polished score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion
    # --------------------------------------------------------------
    best_global_seq = np.maximum(best_global_seq, 0.0)
    total = best_global_seq.sum()
    if total > 0:
        best_global_seq /= total
    else:
        best_global_seq = np.full(N, 1.0 / N)

    final_score = factor * np.max(_conv_self_fft(best_global_seq))
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_global_seq]
