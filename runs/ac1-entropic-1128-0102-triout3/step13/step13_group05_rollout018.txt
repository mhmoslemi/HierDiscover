The evaluation function is scale‑invariant, therefore we can restrict the search to sequences that sum to 1.
For a fixed length `n` the objective is  

```
score = 2·n·max(conv(a,a))            ( a ≥ 0 , Σa = 1 )
```

where `conv(a,a)` is the ordinary linear convolution.
The problem is non‑convex, but the *smooth max*  

```
S_T(a) = T·log Σ exp( conv(a,a) / T )
```

is a differentiable upper bound of `max(conv)`.  
When the temperature `T` is large the landscape is smooth, when `T → 0` it
converges to the true objective.  Using this surrogate we can apply a
projected Adam optimiser on the simplex (non‑negative, unit‑sum) – a
well‑known, fast method for large‑scale problems.

The algorithm consists of three phases

1. **Initial pool** – a handful of analytic shapes (uniform, power‑ramps,
   Beta‑densities, random Dirichlet samples).  The best few candidates are
   kept.
2. **Long Adam optimisation** – starting from the best candidate we minimise
   the smooth‑max surrogate.  The temperature is decreased linearly from
   `T₀ = 0.2` to `Tₑ = 1e‑6`.  Adam‑steps are projected onto the probability
   simplex after every iteration.  Every `10 000` steps a tiny Gaussian
   perturbation is added to avoid stagnation.
3. **Final normalisation** – the best vector found is renormalised to a
   proper probability distribution and returned as a plain Python list.

All heavy operations are performed with real FFTs (`np.fft.rfft`), giving an
`O(n log n)` cost per iteration.  With `n = 65536` a full iteration takes
well under a millisecond on a modern CPU, allowing several hundred thousand
updates inside the 1000 s budget.

The implementation follows the required interface (`propose_candidate`) and
contains no closures, file IO or external dependencies other than `numpy`.

"""
A fast simplex‑projected Adam optimiser that minimises the smooth‑max
upper bound of the convolution peak.  The method starts from a small
hand‑crafted pool of analytic shapes (uniform, power ramps, Beta densities,
random Dirichlet vectors) and then runs a long projected Adam descent
with a temperature schedule.  The result is a non‑negative step‑function
with a very low evaluation score.
"""

import time
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection  (Euclidean projection onto {x≥0, Σx=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes (used for the initial pool)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Helper for keeping the K best candidates
# ----------------------------------------------------------------------
def _push_top(pool: List[Tuple[float, np.ndarray]],
              score: float,
              vec: np.ndarray,
              k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Objective (scale‑invariant) – we keep the sum = 1, so the denominator is 1
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a unit‑sum sequence."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float) -> Tuple[np.ndarray, float]:
    """Projected Adam on the smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]

    # best result so far
    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.05          # base learning rate
    T0 = 0.2              # initial temperature
    T_end = 1e-6          # final temperature
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # temperature schedule (linear decay)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = max(T_end, T0 * (1.0 - frac))

        # --------------------------------------------------------------
        # compute convolution and soft‑max weights
        # --------------------------------------------------------------
        conv = _conv_self(a)
        max_c = conv.max()
        # stable soft‑max
        w_raw = np.exp((conv - max_c) / T)
        w = w_raw / w_raw.sum()

        # --------------------------------------------------------------
        # gradient of the smooth‑max (2·corr(w, a_rev))
        # --------------------------------------------------------------
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)           # length for w * a_rev
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1: n - 1 + n]   # slice of length n

        # --------------------------------------------------------------
        # Adam update
        # --------------------------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # occasional tiny Gaussian noise (helps escaping shallow minima)
        # --------------------------------------------------------------
        if it % 10000 == 0:
            sigma = 0.002
            noise = np.random.default_rng().normal(0.0, sigma, size=n)
            a += noise
        # --------------------------------------------------------------
        # projection onto the simplex
        # --------------------------------------------------------------
        a = _project_simplex(a)

        # --------------------------------------------------------------
        # keep the best ever seen
        # --------------------------------------------------------------
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger N → finer discretisation)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # Phase 1 – build a small pool of analytical / random starts
    # --------------------------------------------------------------
    TOP_K = 5
    pool: List[Tuple[float, np.ndarray]] = []

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    p_grid = np.linspace(0.2, 2.0, 9)          # 0.2 … 2.0
    alpha_grid = np.linspace(0.0, 0.5, 6)      # 0 … 0.5
    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_shape(N, p, alpha)
            _push_top(pool, _objective(a), a, TOP_K)

    # Beta‑shapes (U‑shaped for α<1)
    for alpha in np.linspace(0.1, 0.9, 9):
        a = _make_beta_shape(N, alpha)
        _push_top(pool, _objective(a), a, TOP_K)

    # random Dirichlet vectors (α=0.3 → sparsity)
    rng = np.random.default_rng(seed + 12345)
    for _ in range(30):
        a = rng.gamma(0.3, 1.0, size=N)
        a /= a.sum()
        _push_top(pool, _objective(a), a, TOP_K)

    # incorporate the global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Choose the best start from the pool
    # --------------------------------------------------------------
    best_start_score, best_start_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_start_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long projected Adam descent on the smooth‑max surrogate
    # --------------------------------------------------------------
    refined_seq, refined_score = _adam_descent(best_start_seq, deadline, start_time)

    if refined_score < best_start_score:
        best_seq = refined_seq
        best_score = refined_score
    else:
        best_seq = best_start_seq
        best_score = best_start_score

    # --------------------------------------------------------------
    # Final sanity checks & conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
