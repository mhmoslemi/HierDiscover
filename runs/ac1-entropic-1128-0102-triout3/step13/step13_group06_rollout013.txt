"""
Hybrid search for a low‑objective non‑negative step function.

The algorithm consists of three stages:

1. **Warm‑start** – a collection of analytic shapes (tent, exponential,
   power‑law, Gaussian, etc.) is generated, normalised to unit sum and the
   best (lowest exact objective) is kept.

2. **Simulated‑annealing hill‑climbing** – the current vector and its
   linear self‑convolution are kept in memory.  In each step a random
   amount of mass is transferred from a source index to a destination
   index.  Moves that lower the maximal convolution value are always
   accepted; moves that raise it are accepted with probability
   ``exp(-Δ/T)`` where ``Δ`` is the increase of the maximal convolution and
   ``T`` is a temperature that linearly decays to zero.  This simple
   stochastic descent allows the search to escape the deterministic local
   minima observed by the previous implementation.

3. **Return the best vector** – the best vector seen during the search
   (unit‑sum, non‑negative) is converted to a plain Python list.

All heavy work (convolution updates and objective evaluation) is performed
with NumPy and incremental O(n) updates, enabling millions of iterations
within the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector (no assumption on its sum)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current vector (will be modified in‑place).
    conv : np.ndarray, shape (2n‑1,)
        Current convolution a*a (will be updated in‑place).
    src, dst : int
        Indices with src ≠ dst.
    eps : float
        Amount moved from src to dst (eps ≥ 0, eps ≤ a[src]).
    """
    n = a.shape[0]

    # linear term 2·a*Δ   (Δ = eps*(e_dst – e_src))
    # subtract the contribution of src
    conv[src : src + n] -= 2.0 * eps * a
    # add the contribution of dst
    conv[dst : dst + n] += 2.0 * eps * a

    # quadratic term Δ*Δ
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # finally update the vector
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Candidate analytic shapes (all produce a non‑negative vector of length n)
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Return a list of diverse analytically generated, unit‑sum vectors."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]
    rng = np.random.default_rng(42)

    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Symmetric exponential decay
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Symmetric power‑law from the edges
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Simple half‑cosine (already known to be decent)
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, various σ
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet‑like draws
    for _ in range(10):
        rnd = rng.random(n)
        candidates.append(rnd)

    # Normalise and discard the all‑zero case
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the harness gives us ~1000 s).

    Returns
    -------
    list[float]
        A non‑negative vector ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin (the harness adds its own 0.5 s)
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Determine vector length
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name‑defined]
    except Exception:
        n = 4096                                   # default size used before

    # --------------------------------------------------------------
    # 1) Warm‑start – pick the best analytic shape
    # --------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # if a previous construction is available, add it to the pool
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name‑defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm_starts.append(starter / starter.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    if best_vec is None:                     # fallback – uniform
        best_vec = np.full(n, 1.0 / n)

    a = best_vec.copy()                      # current working vector (unit sum)
    conv = _conv_fft(a)                      # its self‑convolution
    cur_max = conv.max()
    best_max = cur_max
    best_vec = a.copy()
    best_score = 2.0 * n * best_max           # sum = 1 ⇒ denominator = 1

    print(f"[{time.time() - start_time:.2f}s] Warm‑start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Simulated‑annealing hill climbing on the convolution
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)

    total_iter = 0
    # temperature schedule – start a fraction of the current excess over the average
    avg_conv = 1.0 / (2 * n - 1)              # average value of the convolution when sum=1
    excess = max(cur_max - avg_conv, 0.0)
    T0 = excess if excess > 0 else 1e-5      # fallback tiny temperature

    while time.time() < deadline:
        total_iter += 1

        # linear temperature decay (zero at the end)
        remaining = deadline - time.time()
        frac_time = max(remaining / (deadline - start_time), 0.0)
        T = T0 * frac_time

        # probability of picking a source proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # early‑phase allows larger moves, later phases shrink them
        # max fraction of the source mass to relocate
        max_frac = 0.40 * frac_time + 0.02          # from 0.42 initially down to 0.02
        eps = a[src] * rng.random() * max_frac
        if eps <= 0.0:
            continue

        old_max = cur_max

        # apply the move
        _apply_move(a, conv, src, dst, eps)
        new_max = conv.max()
        delta = new_max - old_max

        # acceptance test
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            # Metropolis criterion – avoid overflow for huge delta/T
            if T > 0.0:
                prob = math.exp(-delta / T) if delta / T < 700 else 0.0
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_max = new_max
            # record a new best if we've improved the objective
            if new_max < best_max:
                best_max = new_max
                best_vec = a.copy()
                best_score = 2.0 * n * best_max
                print(f"[{time.time() - start_time:.2f}s] iter={total_iter:,} "
                      f"new best score={best_score:.6f}")
        else:
            # reject – undo the move (swap src/dst)
            _apply_move(a, conv, dst, src, eps)
            # cur_max stays unchanged

    # --------------------------------------------------------------
    # 3) Final normalisation and conversion
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
