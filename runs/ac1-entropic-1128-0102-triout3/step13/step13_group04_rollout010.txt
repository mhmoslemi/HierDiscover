```
def _adam_optimize_flat(a0, start_time, deadline, best_score, target_val, max_iter=10_000_000):
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    n = a.shape[0]
    L_self = _next_pow2(2*n-1)
    L_full = _next_pow2(3*n-2)
    # Adam params
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr=0.2  # maybe lower?
    m=np.zeros_like(a)
    v=np.zeros_like(a)
    it=0; last_improve=0
    rng = np.random.default_rng()
    while time.time()<deadline and it<max_iter:
        it+=1
        # compute gradient and conv
        grad, conv = _grad_flatten(a, target_val, L_full)
        cur_max = conv.max()
        cur_sc = float(2.0*n*cur_max)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            last_improve=it
            print(f"[{time.time()-start_time:.2f}s] Flatten Adam it={it:,} score={cur_sc:.6f}")
        # Adam updates
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat)+eps)
        a = a - step
        # projection
        a = _project_simplex(a)
        # occasional noise for plateau
        if it-last_improve>5000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0)
            a = _project_simplex(a)
            last_improve=it
    return best_a, best_sc
```

```
        a_flat, sc_flat = _adam_optimize_flat(best_seq, start_time, flatten_deadline, best_score, target_val, L_self, L_full)
        if sc_flat < best_score:
            best_score = sc_flat
            best_seq = a_flat.copy()
            print(f"[{time.time()-start_time:.2f}s] Flatten phase improvement score={sc_flat:.6f}")

    # Phase 2: max-Adam phase (single active)
    max_deadline = start_time + budget_s * 0.85
    if time.time() < max_deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting max-Adam phase")
        a_max, sc_max = _adam_optimize_max(best_seq, start_time, max_deadline, best_score)
        if sc_max < best_score:
            best_score = sc_max
            best_seq = a_max.copy()
            print(f"[{time.time()-start_time:.2f}s] Max-Adam improvement score={sc_max:.6f}")

    # Phase 3: hill climbing
    if time.time() < deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting hill-climb phase")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time()-start_time:.2f}s] Hill-climb improvement score={sc_hc:.6f}")

    # final normalization
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)
    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```

The search works on the probability simplex (the sequence is normalised to sum = 1 – the objective is scale‑invariant).  
Three phases are performed inside the 1000 s budget:

1. **Flattening phase** – minimise the squared deviation of the full autocorrelation from the uniform value `1/(2n‑1)`.  
   The gradient of this smooth objective can be written as a convolution and is obtained with two FFTs per iteration.  
   Adam with projection onto the simplex drives the autocorrelation towards a flat shape, which is known to lower the maximal entry.

2. **Peak‑reduction phase** – identical to the reference Adam optimiser but using the sub‑gradient of the *single* active shift (the current maximum of the autocorrelation).  
   After flattening the remaining peaks are small, so this short phase removes the last excess.

3. **Fine stochastic hill‑climb** – repeatedly moves a tiny random fraction of mass from one index to another and keeps the move only if the true objective improves.

All three phases respect the time limit, use FFT‑based convolutions (`O(n log n)`) and keep the sequence non‑negative and normalised.

```python
"""
Propose a non‑negative step‑function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1.  Flattening stage – minimise the L2 distance between the full
    autocorrelation and the uniform value 1/(2n‑1) using Adam on the
    exact gradient (computed by two FFT‑convolutions).

2.  Peak‑reduction stage – Adam optimiser that follows the sub‑gradient
    of the currently maximal autocorrelation entry (single active shift).

3.  Hill‑climb – tiny random mass transfers that are accepted only when
    the true objective improves.

All updates are projected onto the probability simplex, the total
runtime is limited to the supplied budget (default 1000 s) and the
function returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a⋆a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of x and y using the supplied FFT length L."""
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)[: len(x) + len(y) - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σ w = 1 }.
    Returns w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (single active shift)
# ----------------------------------------------------------------------
def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal autocorrelation entry."""
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for the *flattening* objective
# ----------------------------------------------------------------------
def _adam_optimize_flat(a0: np.ndarray,
                        start_time: float,
                        deadline: float,
                        best_score_initial: float,
                        target_val: float,
                        L_self: int,
                        L_full: int,
                        max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises  Σ (conv(a,a) - target)^2.
    Returns the best sequence seen and its true objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12  # a bit smaller than the max‑phase learning rate

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        # self‑convolution and its deviation from the uniform target
        conv = _conv_self(a, L_self)
        diff = conv - target_val

        # gradient = 4 * (diff * a)  (full convolution, slice the centre)
        G = _conv_full(diff, a, L_full)               # length 3n‑2
        grad = 4.0 * G[n - 1: n - 1 + n]               # length n

        # true objective (max‑peak based)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Flatten Adam it={it:,} score={cur_score:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex
        a = _project_simplex(a)

        # occasional jitter when stuck
        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser for the *max‑peak* objective (single active shift)
# ----------------------------------------------------------------------
def _adam_optimize_max(a0: np.ndarray,
                       start_time: float,
                       deadline: float,
                       best_score_initial: float,
                       max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that directly reduces the maximal autocorrelation entry.
    Returns the best sequence seen and its true objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.20

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Max‑Adam it={it:,} score={cur_score:.6f}")

        grad = _subgrad_one_active(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very small random mass transfers; keep only improving moves.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer up to 1 % of the source mass
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises
    2·n·max(conv(a,a))/(sum(a)²).

    The search runs three phases (flattening → max‑peak Adam → hill‑climb)
    within the given time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4096
    L_self = _next_pow2(2 * n - 1)          # for a⋆a
    L_full = _next_pow2(3 * n - 2)          # for the flattening gradient
    target_val = 1.0 / (2 * n - 1)          # uniform autocorrelation value

    # --------------------------------------------------------------
    # initialise (use the global starter if present)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)

    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        starter = np.maximum(starter, 0.0)
        if starter.ndim == 1 and starter.size > 0:
            s = starter.sum()
            if s > 0:
                starter = starter / s
                conv0 = _conv_self(starter, L_self)
                cur_score = float(2.0 * n * conv0.max())
                best_seq = starter.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur_score:.6f}")
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        best_seq = uniform.copy()
        conv0 = _conv_self(best_seq, L_self)
        best_score = float(2.0 * n * conv0.max())
        print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Flattening phase (Adam on L2 autocorrelation error)
    # --------------------------------------------------------------
    flatten_deadline = start_time + budget_s * 0.55
    if time.time() < flatten_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting flattening Adam phase")
        a_flat, sc_flat = _adam_optimize_flat(best_seq,
                                               start_time,
                                               flatten_deadline,
                                               best_score,
                                               target_val,
                                               L_self,
                                               L_full)
        if sc_flat < best_score:
            best_score = sc_flat
            best_seq = a_flat.copy()
            print(f"[{time.time() - start_time:.2f}s] Flattening improvement score={sc_flat:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Peak‑reduction phase (Adam on the max‑peak)
    # --------------------------------------------------------------
    max_deadline = start_time + budget_s * 0.85
    if time.time() < max_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting max‑peak Adam phase")
        a_max, sc_max = _adam_optimize_max(best_seq,
                                            start_time,
                                            max_deadline,
                                            best_score)
        if sc_max < best_score:
            best_score = sc_max
            best_seq = a_max.copy()
            print(f"[{time.time() - start_time:.2f}s] Max‑Adam improvement score={sc_max:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improvement score={sc_hc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
