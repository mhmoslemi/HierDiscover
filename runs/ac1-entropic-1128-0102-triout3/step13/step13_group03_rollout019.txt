def _smooth_descent(a0, max_iters, deadline, start_time, best_score_initial, beta_max=2000.0):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    n = a.shape[0]
    L_conv = _next_pow2(2*n-1)
    L_corr = _next_pow2(3*n-2)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    beta = 2.0
    beta_mul = 1.02

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # compute exact score using sum (since a sum may not be exactly 1)
        s = a.sum()
        if s <= 0:
            cur_score = float('inf')
        else:
            cur_score = 2.0 * n * conv.max() / (s * s)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] smooth iter={t:,} β={beta:.1f} score={cur_score:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.5 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            jitter = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

            break
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        s = a.sum()
        cur_score = 2.0 * n * max_val / (s * s) if s > 0 else float('inf')
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] subgrad iter={t} score={cur_score:.6f}")
        # subgradient for max
        # choose all indices within tolerance
        tol = max_val * 1e-6
        active = np.where(conv >= max_val - tol)[0]
        # compute gradient as mean of individual gradients
        grad = np.zeros_like(a)
        for k in active:
            i_low = max(0, k-(n-1))
            i_high = min(k, n-1)
            idx = np.arange(i_low, i_high+1)
            grad[idx] += 2 * a[k - idx]
        grad /= len(active)
        # step size schedule: decreasing
        step = 0.2 / np.sqrt(t)
        a = a - step * grad
        # projection to simplex
        a = _project_simplex(a)
    return best_a, best_score

"""
Search for a non‑negative step function (probability vector) that
minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine combines several complementary phases:

1️⃣  A dense parametric sweep over exponential‑polynomial shapes
    a_i ∝ r^i·(i+1)^p (mixed with the uniform vector) to obtain a strong
    initial seed.

2️⃣  Adam‑style projected descent on a smooth‑max surrogate of the
    “max‑convolution’’ term with a temperature schedule that reaches a
    very sharp soft‑max (β up to 2000).

3️⃣  Projected sub‑gradient descent that uses the exact max‑entry
    (averaged over a tiny epsilon‑active set) – this reduces the true
    objective directly.

4️⃣  A stochastic hill‑climb that repeatedly moves a small amount of
    mass from the indices that dominate the current maximal convolution
    entry to random low‑mass positions; only improving moves are kept.

All phases respect the 1000 s budget, print intermediate scores and finally
return the best sequence as a plain list of Python floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a                     (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    conv = _conv_self(a, L=L_conv)          # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (stable)
    shifted = conv - max_c                 # max becomes 0
    exp_vals = np.exp(beta * shifted)      # under‑flow for large negative values is harmless
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]                        # length n
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)   # linear convolution w * a_rev

    start = a.shape[0] - 1                 # n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the smooth‑max surrogate.
    β starts small (smooth landscape) and is increased up to `beta_max`.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # Adam  hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    beta = 2.0               # start fairly smooth
    beta_mul = 1.02          # geometric increase per iteration

    while time.time() < deadline and t < max_iters:
        t += 1

        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        s = a.sum()
        cur_score = float("inf") if s <= 0.0 else 2.0 * n * conv.max() / (s * s)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth iter={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Decaying base learning‑rate
        lr = 0.5 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional tiny jitter to escape tiny plateaus
        if t % 5000 == 0:
            jitter = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution term
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,

    """
    Sub‑gradient descent on the exact objective.
    Uses the gradient of the (possibly multiple) maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]

    for t in range(1, max_iters + 1):

        tol = max_val * 1e-6
        active = np.where(conv >= max_val - tol)[0]

        # average of the individual gradients 2·a shifted by each active index
        grad = np.zeros_like(a)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            idx = np.arange(i_low, i_high + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= max(1, len(active))

        # diminishing step size
        step = 0.15 / np.sqrt(t)
        a = a - step * grad
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (random + max‑entry attacks)
# ----------------------------------------------------------------------
def _targeted_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves combined with greedy reductions of the
    currently dominant convolution entry.
    Only improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # step‑size control
    step_factor = 0.15      # up to 15 % of source mass per move
    min_factor = 1e-6
    decay = 0.995

    while time.time() < deadline:
        # choose move type
        if rng.random() < 0.6:
            # ---- random transfer -------------------------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # ---- targeted transfer ------------------------------------------------
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            # pick a random pair (i,j) that contributes to this entry
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose the move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)  # reset after success
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Coarse‑to‑fine parametric sweep
# ----------------------------------------------------------------------
def _parametric_sweep_ext(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Exhaustive exploration of shapes a_i ∝ r^i·(i+1)^p mixed with the uniform vector.
    The grid is intentionally dense to discover very close‑to‑optimal seeds.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)   # i+1
    r_grid = np.logspace(np.log10(0.95), np.log10(0.99999), num=70)
    p_grid = np.linspace(-5.0, 5.0, num=81)
    lam_grid = np.linspace(0.0, 1.0, num=11)

    L_fft = _next_pow2(2 * n - 1)

    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 120:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * shape
                seq_sum = seq.sum()
                if seq_sum == 0.0:
                    continue
                seq /= seq_sum
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises the evaluation function.
    The routine respects the 1000 s budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Problem size – use the length of the global starter if it exists,
    # otherwise fall back to 4096 (a good trade‑off between expressivity
    # and speed).
    # ------------------------------------------------------------------
    try:
        starter_global = np.asarray(height_sequence_1, dtype=float)
        n = int(starter_global.shape[0])
        if n <= 0:
            n = 4096
    except Exception:
        n = 4096

    # Avoid pathological huge vectors (the search would be too slow)
    n = min(n, 8192)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    L_fft = _next_pow2(2 * n - 1)

    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter exists, normalise it and use it as a seed
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (coarse → fine)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:
        best_seq, best_score = _parametric_sweep_ext(
            n, uniform, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style smooth‑max descent runs (several seeds)
    # ------------------------------------------------------------------
    descent_seeds = [seed, seed + 1, seed + 2, seed + 3]
    for ds in descent_seeds:
        if time.time() >= deadline - 180:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        # tiny Gaussian perturbation to escape possible plateaus
        a0 += 5e-4 * rng.standard_normal(n) * a0
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 60000
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,
            start_time,
            best_score,
            beta_max=2000.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth descent improvement score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        for run in range(2):
            if time.time() >= deadline - 60:
                break
            a0 = best_seq.copy()
            # add a very small random jitter
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            a0 = np.maximum(a0 + jitter, 0.0)
            a0 = _project_simplex(a0)

            max_iters = 40000
            a_opt, sc_opt = _subgradient_descent(
                a0,
                max_iters,
                deadline - 15,
                start_time,
                best_score,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] subgradient improvement score={sc_opt:.6f}"
                )

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb(
            best_seq, deadline, start_time, L_fft
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Optional sanity check with the official evaluator
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
