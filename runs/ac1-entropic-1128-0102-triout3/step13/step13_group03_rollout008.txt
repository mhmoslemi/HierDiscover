"""
Improved search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of several cheap global explorations followed by a
refined projected Adam optimisation on a smooth‑max surrogate of the
maximum convolution entry and a final targeted hill‑climb.  The main new
ideas compared with the reference implementation are:

* a symmetric parametric sweep (exponential‑/power‑shaped “U” profiles
  that put more mass at the ends of the interval);
* a few random Dirichlet samples to diversify the search space;
* a higher temperature schedule (β up to 2000) for a tighter smooth‑max
  approximation;
* a slightly larger Adam learning‑rate and more iterations;
* a hill‑climb that explicitly removes mass from the pairs that are
  responsible for the current maximal convolution entry.

The routine respects a 1000 s budget, prints intermediate scores and
returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
from typing import List

# ------------------------------------------------------------
# Basic linear‑algebra utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a * a (full length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ------------------------------------------------------------
# Fast exact evaluation (used internally – sum≈1)
# ------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ------------------------------------------------------------
# Smooth‑max surrogate utilities
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float, L_conv: int, L_corr: int):
    """
    Returns (grad, conv) where
        conv = a*a
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    conv = _conv_self(a, L=L_conv)          # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                 # stabilise exponentials
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()          # soft‑max weights, shape (2n‑1,)

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)

    start = a.shape[0] - 1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type projected descent on the smooth‑max surrogate.
    The temperature β is increased geometrically up to beta_max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 1.0                # start very smooth
    beta_mul = 1.03           # exponential increase
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] descent it={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.6 / (1.0 + 2e-4 * t)          # a bit larger initial step
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional tiny jitter to escape plateaus
        if t % 5000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

# Targeted hill‑climb (random + max‑entry attacks)
# ------------------------------------------------------------
def _targeted_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,

    """
    Random mass‑transfer moves (≈15 % of source mass) together with
    moves that directly diminish the currently dominant convolution entry.
    Only strictly improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    step_factor = 0.15
    min_factor = 1e-5
    decay = 0.985

    while time.time() < deadline:
        # choose move type
        if rng.random() < 0.55:
            # -------------------- random transfer --------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # -------------------- targeted transfer --------------------
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))

            # pick a pair (i,j) with i+j=k_max that contributes most
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            # compute contributions a[i] * a[k_max-i]
            idxs = np.arange(i_low, i_high + 1)
            contrib = a[idxs] * a[k_max - idxs]
            if contrib.max() < 1e-15:
                continue
            i = idxs[np.argmax(contrib)]
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)   # reset after success
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ------------------------------------------------------------
# Symmetric parametric sweep
# ------------------------------------------------------------
def _symmetric_parametric_sweep(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore two families of symmetric “U‑shaped” sequences:
      1) exponential × power law (controlled by r and p);
      2) pure product power   ( (i+1)*(n-i) )^p .
    Each candidate is optionally mixed with the uniform distribution.
    """
    half = n // 2
    L_fft = _next_pow2(2 * n - 1)

    # grids
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=35)
    p_grid = np.linspace(-3.5, 2.5, 55)
    lam_grid = (0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0)

    # --------- 1) exponential × power ----------
    for r in r_grid:
        if time.time() >= deadline - 180:
            break
        # exponential decay on the left half (0 … half‑1)
        base = np.power(r, np.arange(half, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 180:
                break
            left = base * np.power(np.arange(1, half + 1, dtype=np.float64), p)
            # reflect to obtain full symmetric vector
            if n % 2 == 0:
                seq = np.concatenate([left, left[::-1]])
            else:
                seq = np.concatenate([left, left[-2::-1]])
            if seq.sum() == 0.0:
                continue
            seq /= seq.sum()
            for lam in lam_grid:
                cand = lam * uniform + (1.0 - lam) * seq
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep exp r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )

    # --------- 2) product power -----------------
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = (idx * (n - idx + 1.0))  # (i+1)*(n-i)
    for p in p_grid:
        if time.time() >= deadline - 180:
            break
        seq = np.power(prod, p)
        if seq.sum() == 0.0:
            continue
        seq /= seq.sum()
        for lam in lam_grid:
            cand = lam * uniform + (1.0 - lam) * seq
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep prod p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                )
    return best_seq, best_score


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence that minimises
        2 * n * max(conv(a,a)) / (sum(a)**2)
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 gives a good trade‑off between expressivity and speed
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise: uniform baseline (sum ≈ 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter exists, try it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Global exploration – symmetric parametric sweep
    # ------------------------------------------------------------------
    if time.time() < deadline - 240:   # reserve a few minutes for later stages
        best_seq, best_score = _symmetric_parametric_sweep(
            n, uniform, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet sampling (adds diversity)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200:
        rng = np.random.default_rng(seed + 42)
        for _ in range(800):
            if time.time() >= deadline - 200:
                break
            # occasionally use a low concentration to get spiky candidates
            conc = rng.choice([0.2, 0.5, 1.0, 2.0])
            cand = rng.dirichlet(np.full(n, conc))
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] dirichlet conc={conc:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam descent runs (from several seeds)
    # ------------------------------------------------------------------
    descent_seeds = [seed, seed + 5, seed + 11]
    for ds in descent_seeds:
        if time.time() >= deadline - 120:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        # small random perturbation before descent
        a0 += 1e-4 * rng.standard_normal(n) * a0
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 40000
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,          # keep a tiny margin for final polish
            start_time,
            best_score,
            beta_max=2000.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] descent seed={ds} improved score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb polish (random + targeted moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Normalise (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check with the official evaluator (optional)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
