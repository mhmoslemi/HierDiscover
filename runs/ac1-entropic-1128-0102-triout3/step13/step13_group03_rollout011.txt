import time, math, numpy as np
# helpers ...

def propose_candidate(seed=0, budget_s=1000):
    # set random seed
    np.random.seed(seed)
    start_time=time.time()
    deadline = start_time + budget_s - 0.2

    n = 4096
    uniform = np.full(n, 1.0/n)
    L_fft = _next_pow2(2*n-1)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"Baseline uniform score={best_score}")

    # param sweep families
    best_seq, best_score = _parametric_sweep_extended(n, uniform, deadline, start_time, best_seq, best_score)

    # gradient descent runs
    for run in range(5):
        if time.time() >= deadline - 200:
            break
        rng = np.random.default_rng(seed+run)
        a0 = best_seq.copy()
        a0 += 1e-5 * rng.standard_normal(n) * a0
        a0 = np.maximum(a0, 0)
        a0 = _project_simplex(a0)
        a_opt, sc_opt = _smooth_descent(a0, max_iters=80000, deadline=deadline-20, start_time=start_time, best_score_initial=best_score, beta_max=2000.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"Smooth descent improvement {sc_opt}")

    # final hill climb polish
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"Final hill climb improvement {sc_opt}")

    # Normalize and finalize
    best_seq = np.maximum(best_seq,0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq/total

    # optional final evaluation via evaluate_sequence
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"True score {true_score}")
    except Exception:
        pass

    print(f"Finished best score approx {best_score}")
    return best_seq.tolist()

def _parametric_sweep_extended(n, uniform, deadline, start_time, best_seq, best_score):
   idx = np.arange(1, n+1, dtype=float)
   inv_idx = np.arange(n,0,-1,dtype=float) # n-i
   L_fft = _next_pow2(2*n-1)
   lam_grid = np.linspace(0,1,9) # 0,0.125,...1

   # Family A
   r_grid_A = np.logspace(np.log10(0.92), np.log10(0.9999), 30)
   p_grid_A = np.linspace(-3.5,2.5,61) # step ~0.09
   for r in r_grid_A:
        if time.time()>=deadline - 30: break
        geo = np.power(r, np.arange(n,dtype=float))
        for p in p_grid_A:
            if time.time()>=deadline - 30: break
            shape = geo * np.power(idx, p)
            norm = shape.sum()
            if norm == 0: continue
            shape /= norm
            for lam in lam_grid:
                seq = lam*uniform + (1-lam)*shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"New best A r={r:.4f} p={p:.2f} lam={lam:.2f} score={sc:.6f}")

   # Family B (symmetric tails)
   r_grid_B = np.logspace(np.log10(0.92), np.log10(0.9999), 30)
   for r in r_grid_B:
        if time.time() >= deadline - 30: break
        shape = np.power(r, np.arange(n,dtype=float)) + np.power(r, np.arange(n-1,-1,-1,dtype=float))
        shape_sum = shape.sum()
        if shape_sum == 0: continue
        shape /= shape_sum
        for lam in lam_grid:
            seq = lam*uniform + (1-lam)*shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"New best B r={r:.4f} lam={lam:.2f} score={sc:.6f}")

   # Family C (beta-like)
   p_grid_C = np.linspace(-2.5, 2.5, 21)
   q_grid_C = np.linspace(-2.5, 2.5, 21)
   for p in p_grid_C:
         if time.time() >= deadline - 30: break
         for q in q_grid_C:
              if time.time() >= deadline - 30: break
              shape = np.power(idx, p) * np.power(inv_idx, q)  # inv_idx = n-i

                      best_seq = seq.copy()
                      print(f"New best C p={p:.2f} q={q:.2f} lam={lam:.2f} score={sc:.6f}")

   return best_seq, best_score

"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Author: ChatGPT

Algorithm Overview
------------------
1. **Problem size** – 4096 coefficients (good trade‑off).
2. **Coarse parametric sweep** – three families of monotone shapes
   (geometric‑power, symmetric tails, beta‑like).  For each family a dense
   grid of parameters is explored; the best candidate is kept.
3. **Smooth‑max Adam descent** – starting from the best sweep vector,
   a long projected Adam optimisation on the soft‑max surrogate of the
   max‑convolution is performed.  The temperature β grows up to 2000
   (tight approximation) and 80 000 iterations are used.
4. **Final stochastic hill‑climb** – random mass‑transfer moves together
   with targeted moves that directly reduce the current dominant
   convolution entry.  Only improving moves are accepted; the step size
   adapts on the fly.
5. The best vector (normalised to sum 1) is returned as a plain Python list.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective   2·n·max(conv(a,a)) / (sum(a)**2)
    for a vector whose sum≈1.
    """
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a                     (full linear convolution)
        grad = ∂/∂a ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    conv = _conv_self(a, L=L_conv)          # length 2n‑1
    max_c = conv.max()

    # stable soft‑max weights
    shifted = conv - max_c                 # max becomes 0
    exp_vals = np.exp(beta * shifted)      # under‑flow for large negative values is harmless
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]                        # length n
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)   # linear convolution w * a_rev

    start = a.shape[0] - 1                 # n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the smooth‑max surrogate.
    β starts small (smooth landscape) and is slowly increased up to beta_max.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0               # initial temperature (very smooth)
    beta_mul = 1.015         # slower increase ⇒ finer approximation

    while time.time() < deadline and t < max_iters:
        t += 1

        # gradient of the smooth surrogate
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # exact score (sum of a is 1 → denominator = 1)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # base learning‑rate decays slowly
        lr = 0.5 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional tiny jitter to escape tiny plateaus
        if t % 4000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (random + max‑entry attacks)
# ----------------------------------------------------------------------
def _targeted_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers (up to ~15 % of the source mass) plus
    targeted moves that remove mass from the pair of indices contributing
    to the current maximum convolution entry.
    Only improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # step‑size control
    step_factor = 0.15      # up to 15 % of source mass per move
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.6:
            # ---- random transfer -------------------------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # ---- targeted transfer ------------------------------------------------
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose the move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)  # reset after success
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
        else:
            # shrink step size when a move is rejected
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Extended parametric sweep – three families of monotone shapes
# ----------------------------------------------------------------------
def _parametric_sweep_extended(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore three families:
        A – a_i ∝ r^i·(i+1)^p
        B – symmetric exponential tails  a_i ∝ r^i + r^{n-1-i}
        C – beta‑like a_i ∝ (i+1)^p·(n-i)^q
    Mixed with the uniform vector (parameter λ) and keep the best.
    """
    L_fft = _next_pow2(2 * n - 1)
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    rev_idx = np.arange(n, 0, dtype=np.float64)         # n-i
    lam_grid = np.linspace(0.0, 1.0, 9)                  # 0,0.125,…,1

    # --------------------------------------------------------------
    # Family A : r^i * (i+1)^p
    # --------------------------------------------------------------
    r_grid_A = np.logspace(np.log10(0.92), np.log10(0.9999), num=30)
    p_grid_A = np.linspace(-3.5, 2.5, 61)                # finer than before
    for r in r_grid_A:
        if time.time() >= deadline - 30:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid_A:
            if time.time() >= deadline - 30:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep A r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )

    # --------------------------------------------------------------
    # Family B : symmetric exponential tails
    # --------------------------------------------------------------
    r_grid_B = np.logspace(np.log10(0.92), np.log10(0.9999), num=30)
    for r in r_grid_B:
        if time.time() >= deadline - 30:
            break
        left = np.power(r, np.arange(n, dtype=np.float64))
        right = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
        shape = left + right
        s = shape.sum()
        if s == 0.0:
            continue
        shape /= s
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep B r={r:.4f} λ={lam:.2f} score={sc:.6f}"
                )

    # --------------------------------------------------------------
    # Family C : beta‑like (i+1)^p·(n-i)^q
    # --------------------------------------------------------------
    p_grid_C = np.linspace(-2.5, 2.5, 21)
    q_grid_C = np.linspace(-2.5, 2.5, 21)
    for p in p_grid_C:
        if time.time() >= deadline - 30:
            break
        for q in q_grid_C:
            if time.time() >= deadline - 30:
                break
            # note: both factors are positive for all i, even for negative exponents
            shape = np.power(idx, p) * np.power(rev_idx, q)
            # Numerical safety: replace NaNs / infs
            shape = np.where(np.isfinite(shape), shape, 0.0)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep C p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}"
                    )
    return best_seq, best_score

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 coefficients
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}")

    # --------------------------------------------------------------
    # Use global starter if available
    # --------------------------------------------------------------
    try:

                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse‑to‑fine parametric sweep (families A, B, C)
    # --------------------------------------------------------------
    if time.time() < deadline - 300:   # keep ~5 min for later phases
        best_seq, best_score = _parametric_sweep_extended(
            n, uniform, deadline, start_time, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] after sweep best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Multiple smooth‑max Adam descent runs (different seeds)

    descent_seeds = [seed, seed + 10, seed + 20, seed + 30, seed + 40]

    if time.time() < deadline:

        )
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
