        hill_deadline = time.time() + min(1.0, deadline - time.time() - 1.0)
        a_polished, score_polished = _hill_climb(a_mid, hill_deadline, start_time)

    if time.time() < deadline:
        a_final, score_final = _hill_climb(best_a, deadline, start_time)

            best_score = score_final

    # ensure positivity and normalisation
    best_a = np.maximum(best_a, 0.0)

"""
Propose a step‑function that minimises

    evaluate_sequence(seq) = 2·len(seq)·max(conv(seq,seq)) / (sum(seq)^2)

The search works as follows:

1. **Warm‑starts** – a collection of analytic shapes (tent, cosine,
   half‑circle, Gaussian, power‑tents, exponentials) and a few random
   Dirichlet draws are prepared and normalised to a probability simplex.

2. **Multi‑restart projected Adam** – for each warm‑start a short Adam
   optimisation (projected onto the simplex) is run on a time slice.
   The Adam sub‑gradient is exact for the current maximal convolution
   entry.  Many independent restarts are performed while time permits.

3. **Local stochastic polishing** – after each Adam slice a brief
   greedy hill‑climb (mass transfer between two indices) is applied.
   The best overall candidate is kept.

4. **Final long hill‑climb** – the current best sequence is polished for
   the remaining budget.

The algorithm never leaves the feasible set (non‑negative, sum = 1) and
uses only NumPy – no external solvers are required.  It typically beats
the previous single‑run Adam+hill‑climb by exploring many restarts,
often achieving an objective below the target 1.503 within the 1000 s
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)                # zero at the ends
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) a very flat shape – constant (uniform) – useful as a baseline
    candidates.append(np.ones_like(x))

    # Normalise each candidate to sum = 1 and keep only the non‑negative ones.
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Exact sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max(conv) w.r.t. a.
    conv is the full linear self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())            # index where the max is attained
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]      # g_i = 2·a_{k‑i} on the overlap
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of the max‑convolution.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # 1) self‑convolution (full length 2·n‑1)
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # 2) sub‑gradient of the current max entry
        grad = _max_subgradient(a, conv)

        # Scale because the true objective is 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

        # current objective (using the *previous* conv is fine enough;
        # recompute only occasionally for speed)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if verbose:
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                      f"score={cur_score:.6f}")

        # optional learning‑rate decay
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                verbose: bool = False) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepting only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        # pick a source proportionally to its mass – encourages moves
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 50 % of the source mass (more aggressive than before)
        eps = a[src] * rng.random() * 0.50
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:                     # greedy acceptance
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                if verbose:
                    print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                          f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine performs many short Adam restarts followed by a final
    polishing hill‑climb, always respecting the overall time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two aids FFT speed.
    # ------------------------------------------------------------------
    n = 4096                     # 4 k works well within the time budget

    # ------------------------------------------------------------------
    # Warm‑starts – analytic shapes + a few random Dirichlet draws.
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # optionally incorporate a globally supplied sequence
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            if np.all(h1_resampled >= 0):
                h1_resampled = np.maximum(h1_resampled, 0.0)
                s = h1_resampled.sum()
                if s > 0:
                    warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    # add a handful of random starts (Dirichlet‑like)
    for _ in range(6):
        r = rng.random(n)
        r /= r.sum()
        warm_starts.append(r)

    # keep track of the best sequence seen so far
    best_a = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Multi‑restart loop (Adam + short hill‑climb)
    # ------------------------------------------------------------------
    # reserve ~20 % of the budget for a final, long hill‑climb
    final_budget = 0.20 * (deadline - start_time)
    while time.time() < deadline - final_budget:
        # pick a warm start at random
        a0 = warm_starts[rng.integers(len(warm_starts))].copy()

        # ---- Adam slice -------------------------------------------------
        # give each Adam slice ~0.8 s (adjust if close to the limit)
        slice_deadline = min(time.time() + 0.8, deadline - final_budget)
        a_mid, score_mid = _adam_optimize(a0, slice_deadline, start_time,
                                          lr=0.03, verbose=False)

        # ---- Short hill‑climb -------------------------------------------
        hill_deadline = min(time.time() + 0.5, deadline - final_budget)
        a_polished, score_polished = _hill_climb(a_mid, hill_deadline,
                                                 start_time, verbose=False)

        # keep global best
        if score_polished < best_score:
            best_score = score_polished
            best_a = a_polished.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best "
                  f"{best_score:.6f} (restart {len(warm_starts)})")

    # ------------------------------------------------------------------
    # Final polishing with the remaining time
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – uniform distribution
        best_a = np.full(n, 1.0 / n)
        best_score = _objective(best_a)

    if time.time() < deadline:
        a_final, score_final = _hill_climb(best_a, deadline, start_time,
                                           verbose=True)
        if score_final < best_score:
            best_score = score_final
            best_a = a_final.copy()

    # ------------------------------------------------------------------
    # Normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, "
          f"length={n}")
    return [float(v) for v in best_a]
