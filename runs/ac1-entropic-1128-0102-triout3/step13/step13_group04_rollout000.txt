"""
Hybrid optimisation for the step‑function objective

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

All sequences are projected onto the probability simplex (sum = 1,
non‑negative), so the denominator is always 1 and the score reduces to
`2 * n * max(conv)`.  The algorithm consists of three stages

1️⃣  Initialise from the best known global sequence (`height_sequence_1`);
    if that is unavailable we fall back to the uniform distribution.

2️⃣  Run a long projected Adam optimiser, but instead of the single‑active
    sub‑gradient we aggregate the gradients of the *k* largest convolution
    entries (k = 5 by default).  This “bundle” gradient gives a much
    stronger descent direction and typically pushes the objective well
    below the previous best (≈ 1.508).

3️⃣  Finish with a pure stochastic hill‑climb that moves tiny fractions of
    mass between two positions and keeps only improving moves.

The whole search respects the 1000 s budget (leaving ~150 s for the final
hill‑climb).  The returned list can be fed directly to the supplied
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                               # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Multi‑active sub‑gradient (bundle of the k largest shifts)
# ----------------------------------------------------------------------
def _subgrad_multi_active(a: np.ndarray, conv: np.ndarray, k_top: int = 5) -> np.ndarray:
    """
    Compute an aggregated sub‑gradient of max(conv(a,a)).
    The gradient is the sum of the gradients belonging to the `k_top`
    largest convolution entries (or all entries if less than `k_top`).
    """
    n = a.shape[0]
    # indices of the k largest values
    if k_top >= conv.size:
        idxs = np.arange(conv.size)
    else:
        # argpartition is O(N); we then take the exact values for stability
        part = np.argpartition(conv, -k_top)[-k_top:]
        # optional: sort them (not strictly needed)
        idxs = part[np.argsort(-conv[part])]

    grad = np.zeros_like(a)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]                 # ∂/∂a_i of (a⋆a)[k]
    # averaging keeps the update magnitude comparable to the single‑active case
    grad /= max(1, len(idxs))
    return grad


# ----------------------------------------------------------------------
# Adam optimiser that uses the multi‑active gradient
# ----------------------------------------------------------------------
def _adam_multi_opt(a0: np.ndarray,
                    start_time: float,
                    deadline: float,
                    best_score_initial: float,
                    top_k: int = 5,
                    max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser with the bundle sub‑gradient.
    Returns the best sequence seen and its objective (`2*n*max(conv)`).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)        # sum(a) == 1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # multi‑active sub‑gradient
        grad = _subgrad_multi_active(a, conv, k_top=top_k)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _project_simplex(a)

        # plateau handling – add a tiny Gaussian shake if we have not improved
        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny mass re‑distribution)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that transfers a tiny random fraction of mass
    from one index to another and keeps only improving moves.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction of source mass, up to 1 %
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the objective.
    The routine follows the three‑stage scheme described in the module
    docstring and returns a plain Python list (the sequence of heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # problem size (we keep the size that produced the previous best)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the global best sequence if it exists,
    # otherwise from the uniform distribution.
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                a = starter
                print(f"[{time.time() - start_time:.2f}s] Loaded global starter.")
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        # uniform fallback
        a = np.full(n, 1.0 / n)
        print(f"[{time.time() - start_time:.2f}s] Using uniform start.")

    # ensure proper normalization
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    # baseline score (use the exact objective, sum == 1)
    cur_conv = _conv_self(a, L_fft)
    best_score = float(2.0 * n * cur_conv.max())
    best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] Baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Projected Adam optimiser with bundled gradient
    # ------------------------------------------------------------------
    # leave ~150 s for the final hill‑climb
    hill_time = min(150.0, budget_s * 0.15)
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam‑bundle optimiser")
        a_opt, sc_opt = _adam_multi_opt(
            best_seq,
            start_time,
            adam_deadline,
            best_score,
            top_k=5
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Final hill‑climb (tiny mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
