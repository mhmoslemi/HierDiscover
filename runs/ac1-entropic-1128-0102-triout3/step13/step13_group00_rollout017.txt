"""
Improved optimiser for the “step‑function” challenge.

The objective is 2·n·max(conv(a,a)) with ∑a_i = 1, a_i ≥ 0.
The routine works in five stages (all respecting the 1000 s budget):

1.  Initialise from a supplied global starter (or a uniform vector) and
    normalise to the probability simplex.

2.  Broad parametric sweep – geometric, power, beta families *plus*
    a stretched‑exponential family (a_i ∝ exp(‑c·i^p)).  Very cheap
    FFT‑based evaluation is used and the best sequence is kept.

3.  Random Dirichlet draws – many sparse/dense samples are generated
    (Gamma concentration 0.01…5) to diversify the search.

4.  Smooth‑max Adam descent – minimise a differentiable proxy
    S_τ(a)= (1/τ)·log∑exp(τ·conv(a,a)).  τ starts small (≈1) and is
    increased geometrically.  Adam‑style adaptive steps are taken and the
    iterate is projected onto the simplex after each step.

5.  Simulated‑annealing fine‑tuning – repeatedly moves a random amount
    of mass from one index to another.  Improvements are always kept;
    uphill moves are accepted with a temperature that decays to zero.
    This phase is able to escape shallow local minima left by the
    gradient phase.

All convolutions use FFT (full linear convolution) for speed.
The best sequence found is finally re‑normalised and returned as a plain
list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution via FFT (full length, with pre‑allocated length L)."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Self‑convolution a*a (full length) using the supplied FFT size."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (scale‑invariant version)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(seq, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (used for guidance only)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (average over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta, stretched‑exp)
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray, best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # uniform baseline (already considered, but keep for completeness)
    uniform = np.full(n, 1.0 / n)

    # -------------------------------------------------- geometric --------------------------------------------------
    eps_grid = np.logspace(-6, -1, 300)  # finer near 0
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        # also allow mixing with uniform
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.2f}s] Geo eps={eps:.1e} λ={lam:.2f} score={sc:.6f}")

    # -------------------------------------------------- power -------------------------------------------------------
    p_grid = np.linspace(-4.0, 4.0, 161)   # more aggressive negatives
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time():.2f}s] Power p={p:.3f} score={sc:.6f}")

    # -------------------------------------------------- beta --------------------------------------------------------
    alphas = np.linspace(0.05, 8.0, 81)   # extended range
    betas  = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))   # protect overflow
            s = seq.sum()
            if not np.isfinite(s) or s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # -------------------------------------------------- stretched exponential ---------------------------------------
    # a_i ∝ exp( -c * i^p ),  c∈[0.5e-5,5], p∈[0.2,3]
    c_grid = np.logspace(-5, 0.7, 25)   # 10^-5 … 5
    p_grid = np.linspace(0.2, 3.0, 15)
    for c in c_grid:
        if time.time() >= deadline:
            break
        for p in p_grid:
            if time.time() >= deadline:
                break
            # i^p may overflow for large i; use float64 safely here
            power = np.power(idx, p, dtype=np.float64)
            seq = np.exp(-c * power)
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.2f}s] StretchExp c={c:.1e} p={p:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers (more tries & wider concentration)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trial = 0
    while time.time() < deadline and trial < 200:
        trial += 1
        # concentration drawn log‑uniformly between 0.01 and 10
        log_c = rng.uniform(np.log(0.01), np.log(10.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.2f}s] Dirichlet trial={trial} c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max Adam descent (temperature continuation)
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth max convolution proxy."""
    a = a0.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)          # FFT length for self‑conv
    L_grad = _next_pow2(3 * n - 2)          # FFT length for w * a_rev

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.08                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = 1.0                      # start temperature (very smooth)
    tau_factor = 1.25
    it = 0
    while time.time() < deadline:
        it += 1
        # current objective (exact max, not the smooth one)
        conv = _conv_self(a, L_self)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max weights over all shifts
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2·(w * a_rev) for the part of length n
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev, L_grad)          # length = 3n‑2
        grad = 2.0 * conv_w[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        eta = lr0 / np.sqrt(it)                       # diminishing step size
        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new

        # temperature schedule – increase every few thousand iterations
        if it % 3000 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Simulated‑annealing fine tuning (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Simple SA that moves a random chunk of mass between two indices."""
    rng = np.random.default_rng()
    a = a0.copy()
    cur_score = _objective_score(a)
    best_local = a.copy()
    best_local_score = cur_score

    # initial temperature – a small fraction of the current score
    T0 = max(1e-4, 0.015 * cur_score)
    # total SA steps will be limited only by the remaining time budget
    while time.time() < deadline:
        # linear cooling schedule (square‑root decay)
        remaining = deadline - time.time()
        total = deadline - start_time
        frac = remaining / total
        T = T0 * (frac ** 0.5) + 1e-12

        # pick a source with some mass and a (different) destination
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(len(a))
        if src == dst:
            continue

        # transfer a random fraction of the source mass (up to ~3 %)
        eps_frac = rng.random() * 0.03 + 1e-6
        eps = min(a[src] * eps_frac, a[src])
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        prop_score = _objective_score(a_prop)
        delta = prop_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            a = a_prop
            cur_score = prop_score
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_local_score:.6f}")

    # update global best if needed
    if best_local_score < best_score:
        best_score, best_seq = best_local_score, best_local.copy()
    return best_seq, best_score

                x_old = np.linspace(0.0, 1.0, starter.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total < 1e-12:
                starter = np.full(n, 1.0 / n)
            else:
                starter = starter / total
            best_seq = starter
        else:
            raise ValueError
    except Exception:
        best_seq = np.full(n, 1.0 / n)

    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initialise score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric, power, beta, stretched‑exp)
    # --------------------------------------------------------------
    phase_deadline = deadline - 600.0   # reserve ~10 min for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] Post‑sweep best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet starts
    # --------------------------------------------------------------
    phase_deadline = deadline - 400.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )
        print(f"[{time.time() - start_time:.2f}s] Post‑Dirichlet best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam descent (multiple restarts if time permits)
    # --------------------------------------------------------------
    # allocate ~150 s for the Adam optimisation
    adam_deadline = deadline - 250.0
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam descent")
        # first run from the current best
        best_seq, best_score = _smooth_max_adam(
            best_seq, start_time, adam_deadline, best_seq, best_score
        )
        # second random restart (if still time)
        if time.time() < adam_deadline:
            rng = np.random.default_rng()
            rand_start = rng.gamma(shape=0.5, scale=1.0, size=n)
            rand_start /= rand_start.sum()
            best_seq, best_score = _smooth_max_adam(
                rand_start, start_time, adam_deadline, best_seq, best_score
            )
        print(f"[{time.time() - start_time:.2f}s] After Adam best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Simulated‑annealing fine‑tuning (uses all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final SA best={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
