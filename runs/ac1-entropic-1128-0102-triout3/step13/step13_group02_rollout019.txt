"""
Propose a step‑function (non‑negative vector) with a low value of

        2 * n * max(conv(a,a)) / (sum(a)**2)

The search mixes a *large parametric sweep* (exponential, power‑law,
piece‑wise mixtures) with a *sub‑gradient descent* that works on the
exact max‑convolution, followed by a short *simulated‑annealing hill‑climb*.
All vectors are kept on the probability simplex (non‑negative,
sum = 1) – the evaluation function rescales them internally, but using the
simplex makes the optimisation well‑behaved.

The algorithm is deliberately lightweight (single‑core, pure NumPy) and
fits comfortably into the 1000 s budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0 , Σ w = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (vector already on the simplex)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helper: generate a family of candidate vectors
# ----------------------------------------------------------------------


def _make_exponential(n: int, r: float, p: float = 0.0) -> np.ndarray:
    """a_i ∝ r**i * (i+1)**p ."""
    i = np.arange(n, dtype=np.float64)
    shape = np.power(r, i) * np.power(i + 1.0, p)
    return shape


def _make_powerlaw(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-beta}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = np.power(i, -beta)
    return shape


def _make_piecewise(n: int, r: float, k_frac: float) -> np.ndarray:
    """
    Exponential on the first k_frac·n entries, tiny constant afterwards.
    The constant tail is chosen so that the vector still sums to 1 after
    normalisation.
    """
    k = max(1, int(k_frac * n))
    head = np.power(r, np.arange(k, dtype=np.float64))
    tail_len = n - k
    # a tiny constant (10‑⁶) – essentially zero but keeps support full
    tail = np.full(tail_len, 1e-9, dtype=np.float64)
    shape = np.concatenate([head, tail])
    return shape


def _normalise(v: np.ndarray) -> np.ndarray:
    """Scale to the probability simplex (protect against zero sum)."""
    v = np.maximum(v, 0.0)
    tot = v.sum()
    if tot <= 0.0:
        raise ValueError("zero vector")
    return v / tot


# ----------------------------------------------------------------------
# Sub‑gradient descent (averaging over all maximal convolution indices)
# ----------------------------------------------------------------------


def _subgrad_step(a: np.ndarray, L_fft: int, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact max‑convolution.
    The sub‑gradient is the average of the gradients of *all* maximal
    convolution entries.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a, L_fft)
    max_val = conv.max()
    # indices where the max is attained (within a tolerance)
    active = np.where(np.abs(conv - max_val) < 1e-12)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        # contribution a_j to ∂/∂a_i
        g[i_vals] += a[j_vals]
        # symmetric term for i≠j
        mask = i_vals != j_vals
        g[j_vals[mask]] += a[i_vals[mask]]

    # average over the active set – keeps step sizes reasonable
    if active.size > 1:
        g /= float(active.size)

    a_new = a - step * g
    a_new = _project_simplex(a_new)
    return a_new


def _refine_subgradient(a0: np.ndarray, deadline: float, start_time: float,
                        best_score: float) -> tuple[np.ndarray, float]:
    """Run sub‑gradient descent until the deadline."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    rng = np.random.default_rng()
    step0 = 0.15
    it = 0

    while time.time() < deadline:
        it += 1
        step = step0 / np.sqrt(it)
        a_candidate = _subgrad_step(a, L_fft, step)
        cur_sc = _eval_fast(a_candidate, L_fft)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} "
                  f"score={cur_sc:.6f}")
        else:
            # occasional simulated‑annealing acceptance
            prob = np.exp(-(cur_sc - best_sc) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # small jitter every few thousand iterations – helps escape plateaus
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def _anneal_hill_climb(a0: np.ndarray, deadline: float,
                       start_time: float, best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score
    rng = np.random.default_rng()

    # temperature schedule (fast decay)
    T_start = 2e-2
    T_end = 1e-6

    while time.time() < deadline:
        # linear decay in the remaining time
        t_frac = (time.time() - start_time) / (deadline - start_time)
        T = T_start * (T_end / T_start) ** t_frac

        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move a tiny fraction (up to 0.2 % of src mass)
        eps = a[src] * rng.random() * 2e-3
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        prop = _project_simplex(prop)

        new_sc = _eval_fast(prop, L_fft)
        if new_sc < best_sc or rng.random() < np.exp(-(new_sc - best_sc) / T):
            a = prop
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = prop.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal score={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising the evaluation
    function.  The routine performs
        1. a broad parametric sweep (exponential / power‑law / piecewise)
        2. sub‑gradient refinement from the few best seeds
        3. a final simulated‑annealing hill‑climb.
    The best vector is returned as a plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # problem size – a moderate length that still gives a good lower bound
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optionally start from the globally supplied starter (if it exists)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).flatten()
        if starter.size > 0:
            starter = _normalise(starter)
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Aggressive parametric sweep
    # ------------------------------------------------------------------
    # grids – deliberately richer than the previous version
    r_grid = np.logspace(np.log10(0.95), np.log10(0.999999), num=40)
    p_grid = np.linspace(-5.0, 5.0, 61)          # include steeper decays / rises
    beta_grid = np.linspace(0.0, 4.0, 41)       # power‑law exponents
    lam_grid = np.linspace(0.0, 1.0, 11)        # mixing weight with uniform
    k_frac_grid = np.linspace(0.02, 0.5, 9)     # size of exponential block in piecewise

    candidates = []

    # exponential * power‑law family
    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        for p in p_grid:
            shape = _make_exponential(n, r, p)
            if shape.sum() == 0.0:
                continue
            shape = _normalise(shape)
            for lam in lam_grid:
                cand = lam * uniform + (1.0 - lam) * shape
                cand = _normalise(cand)
                sc = _eval_fast(cand, L_fft)
                candidates.append((sc, cand.copy()))
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep exp r={r:.6f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # pure power‑law family
    for beta in beta_grid:
        if time.time() >= deadline - 120:
            break
        shape = _make_powerlaw(n, beta)
        shape = _normalise(shape)
        for lam in lam_grid:
            cand = lam * uniform + (1.0 - lam) * shape
            cand = _normalise(cand)
            sc = _eval_fast(cand, L_fft)
            candidates.append((sc, cand.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep pow β={beta:.2f} λ={lam:.2f} score={sc:.6f}")

    # piecewise exponential / tiny tail family
    for r in r_grid[::4]:               # coarse sampling – cheap
        for k_frac in k_frac_grid:
            shape = _make_piecewise(n, r, k_frac)
            shape = _normalise(shape)
            cand = 0.7 * uniform + 0.3 * shape   # a fixed mixing that worked well
            cand = _normalise(cand)
            sc = _eval_fast(cand, L_fft)
            candidates.append((sc, cand.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] piecewise r={r:.6f} k={k_frac:.2f} score={sc:.6f}")

    # keep only the K best seeds for the refinement phase
    candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in candidates[:8]]

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient refinement (exact max‑convolution)
    # ------------------------------------------------------------------
    # allocate the majority of the remaining budget to this phase
    remaining = deadline - time.time()
    if remaining > 200:
        per_seed = remaining / max(len(top_seeds), 1)
        refined = []
        for idx, seed_vec in enumerate(top_seeds):
            sub_deadline = time.time() + per_seed
            a_ref, sc_ref = _refine_subgradient(seed_vec, sub_deadline, start_time, best_score)
            refined.append((sc_ref, a_ref))
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.copy()
                print(f"[{time.time() - start_time:.2f}s] refined seed {idx+1}/{len(top_seeds)} "
                      f"score={sc_ref:.6f}")

        # keep the best of the refined vectors for the final polishing
        refined.sort(key=lambda x: x[0])
        if refined:
            best_seq = refined[0][1].copy()
            best_score = refined[0][0]

    # ------------------------------------------------------------------
    # 3️⃣  Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        a_hc, sc_hc = _anneal_hill_climb(best_seq, deadline, start_time, best_score)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (guard against tiny negative drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
