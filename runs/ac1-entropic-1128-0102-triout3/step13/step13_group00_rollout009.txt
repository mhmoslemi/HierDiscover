"""
Optimiser for the “step‑function” challenge (v2).

The objective is
    2 · n · max(conv(a, a)) / (sum(a)²)

which is scale‑invariant; we therefore normalise the vector to sum to 1 and
minimise 2·n·max(conv(a,a)).  The optimiser runs four phases inside a 1000 s
budget:

1️⃣  Very cheap parametric sweep (geometric, power, beta families) to obtain
    a decent baseline.

2️⃣  A large number of random Dirichlet starts; each start is followed by a
    short Adam‑based smooth‑max descent (soft‑max approximation of the
    “max”), which quickly improves the candidate.

3️⃣  A long Adam‑based smooth‑max continuation starting from the best
    candidate observed so far.  The temperature τ of the soft‑max starts
    close to 1 (very smooth) and is increased geometrically, while the
    Adam step‑size follows the usual schedule.  After each step the vector
    is projected onto the probability simplex (Euclidean projection).

4️⃣  A final directed hill‑climb that repeatedly transfers a tiny amount
    of mass from the index with the most positive sub‑gradient to the one
    with the most negative, accepting only improving moves.

All convolutions are performed with FFT (using a power‑of‑two length for speed)
and the code respects the time budget.  The returned sequence is a plain
list of Python floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta)
# ----------------------------------------------------------------------
def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    uniform = np.full(n, 1.0 / n)

    # ---- geometric ----------------------------------------------------
    eps_grid = np.logspace(-10, -1, 500)          # finer near 0
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 241)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.linspace(0.05, 6.0, 101)   # extended range
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers + short Adam descent
# ----------------------------------------------------------------------
def _short_adam_from_random(a0: np.ndarray,
                            deadline: float,
                            cur_best_seq: np.ndarray,
                            cur_best_score: float,
                            max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """Run a brief Adam‑based smooth‑max descent from a0."""
    n = a0.shape[0]
    L_self = _next_pow2(2 * n - 1)
    # Adam state
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2
    tau = 1.0
    tau_factor = 1.3
    a = a0.copy()
    best_seq, best_score = cur_best_seq, cur_best_score
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_self)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Short‑Adam it={it} score={cur_score:.6f}")

        # soft‑max weights
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2 * (w * a_rev) on length n
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)                     # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        if it % 500 == 0:
            tau = min(tau * tau_factor, 1e6)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Long Adam‑based smooth‑max continuation
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     deadline: float,
                     cur_best_seq: np.ndarray,
                     cur_best_score: float,
                     max_iter: int | None = None) -> tuple[np.ndarray, float]:
    """Full‑scale Adam descent with increasing temperature."""
    n = a0.shape[0]
    L_self = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2

    # temperature schedule
    tau = 1.0
    tau_factor = 1.3

    a = a0.copy()
    best_seq, best_score = cur_best_seq, cur_best_score
    it = 0
    while time.time() < deadline and (max_iter is None or it < max_iter):
        it += 1

        conv = _conv_self(a, L_self)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max of the convolution
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        if it % 2000 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb_directed(start_a: np.ndarray,
                         start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Fine‑grained hill‑climb moving a tiny fraction of mass."""
    a = start_a.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_self).max())

    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self(a, L_self)
        grad = _subgradient_maxconv(a, conv)

        # source = index with largest positive gradient among those that have mass
        src_cand = np.where(a > 1e-15)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)  # most negative gradient

        if src == dst:
            # fall back to a small random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        # transfer a tiny fraction of source mass
        eps = a[src] * (rng.random() * 0.005 + 1e-6)  # ≤0.5 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop, L_self).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The procedure follows the four‑phase strategy
    described above and respects the allotted time budget.
    """
    np.random.seed(seed)

    global start_time
    start_time = time.time()
    # a small safety margin to avoid overshooting the budget
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem size (4096 proved successful in earlier runs)
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep (if time permits)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0  # reserve ~5 min for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet starts + short Adam descent
    # --------------------------------------------------------------
    phase_deadline = deadline - 180.0
    rng = np.random.default_rng()
    while time.time() < phase_deadline:
        # concentration draw (favoring both sparse and dense vectors)
        conc = rng.uniform(0.05, 5.0)
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        # a brief Adam run (≈2000 iterations or time‑limited)
        best_seq, best_score = _short_adam_from_random(
            a, phase_deadline, best_seq, best_score, max_iter=2500
        )

    # --------------------------------------------------------------
    # 3️⃣  Long Adam‑based smooth‑max continuation
    # --------------------------------------------------------------
    # keep ~120 s for final hill‑climb
    hill_time = 120.0
    smooth_deadline = deadline - hill_time
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting long Adam descent")
        best_seq, best_score = _smooth_max_adam(
            best_seq, smooth_deadline, best_seq, best_score, max_iter=None
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb_directed(best_seq, start_time, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
