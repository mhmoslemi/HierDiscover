"""
Step‑function optimiser – 1000 s budget

The objective to minimise is

    2·n·max(conv(a,a))          (a is normalised so that sum(a)=1)

which is scale‑invariant.  The optimiser works in five stages:

1️⃣  Warm‑up: generate a diverse pool of candidates via uniform,
    geometric / power / beta families and random Dirichlet draws.
    Each candidate is projected onto the probability simplex and its
    true score (2·n·max‑conv) is recorded.  The best K≈10 candidates
    are kept for later mixing.

2️⃣  Mixing: repeatedly form convex combinations of the kept candidates
    (random weights sampled from a Dirichlet) and keep any improvement.
    This simple “averaging” often flattens the autocorrelation.

3️⃣  Adam‑descent on a smooth‑max proxy.
    The smooth‐max of the convolution is
        Sτ(a)= (1/τ)·log Σ exp(τ·conv(a,a))
    whose gradient is 2·(w ∗ a_rev) where w is the soft‑max of the
    convolution.  The temperature τ starts small (smooth landscape) and
    is multiplied by a factor each few thousand steps, ending at a large
    value so that Sτ≈max.  The iterates are projected onto the simplex
    after each Adam update.

4️⃣  Aggressive hill‑climb.
    At every iteration a tiny (or moderate) amount of mass is moved
    from the index with the most positive sub‑gradient of max(conv) to
    the one with the most negative.  The move is accepted only when it
    improves the true objective.  After a long plateau a random “kick”
    (redistribute a small fraction of the mass) helps to escape local
    minima.

5️⃣  Return the best sequence found (as plain Python floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    if L is None:
        L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Full self‑convolution a*a."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Candidate pool utilities
# ----------------------------------------------------------------------
def _keep_top(pool: list, seq: np.ndarray, score: float, max_len: int = 10):
    """Insert (seq,score) into the sorted pool, keep at most max_len entries."""
    if len(pool) < max_len:
        pool.append((seq.copy(), score))
        pool.sort(key=lambda x: x[1])
    else:
        # pool is sorted low→high; replace worst if better
        if score < pool[-1][1]:
            pool[-1] = (seq.copy(), score)
            pool.sort(key=lambda x: x[1])
    return pool


# ----------------------------------------------------------------------
# Stage 1 – diverse initialisation
# ----------------------------------------------------------------------
def _initial_pool(n: int, deadline: float) -> list:
    """Generate a pool of promising candidates."""
    pool = []
    uniform = np.full(n, 1.0 / n)
    best_score = _objective_score(uniform)
    pool = _keep_top(pool, uniform, best_score)

    idx = np.arange(n, dtype=np.float64)

    # geometric families
    eps_grid = np.logspace(-5, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline:
            return pool
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        score = _objective_score(geo)
        pool = _keep_top(pool, geo, score)

    # power families
    p_grid = np.linspace(-2.0, 2.0, 81)
    for p in p_grid:
        if time.time() >= deadline:
            return pool
        pw = np.power(idx + 1.0, p)
        pw /= pw.sum()
        score = _objective_score(pw)
        pool = _keep_top(pool, pw, score)

    # beta families (deterministic grid)
    alphas = np.linspace(0.05, 4.0, 61)
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            return pool
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                return pool
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            seq /= seq.sum()
            score = _objective_score(seq)
            pool = _keep_top(pool, seq, score)

    # random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(60):
        if time.time() >= deadline:
            break
        conc = rng.uniform(0.1, 5.0)
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        a /= a.sum()
        score = _objective_score(a)
        pool = _keep_top(pool, a, score)

    # random mixtures of two beta shapes
    for _ in range(60):
        if time.time() >= deadline:
            break
        a1, b1 = rng.uniform(0.05, 4.0, size=2)
        a2, b2 = rng.uniform(0.05, 4.0, size=2)
        lam = rng.random()
        # beta‑type 1
        log_seq1 = (a1 - 1.0) * log_idx + (b1 - 1.0) * log_rev
        seq1 = np.exp(log_seq1 - np.max(log_seq1))
        # beta‑type 2
        log_seq2 = (a2 - 1.0) * log_idx + (b2 - 1.0) * log_rev
        seq2 = np.exp(log_seq2 - np.max(log_seq2))
        seq = lam * seq1 + (1.0 - lam) * seq2
        seq /= seq.sum()
        score = _objective_score(seq)
        pool = _keep_top(pool, seq, score)

    return pool


# ----------------------------------------------------------------------
# Stage 2 – convex mixing of the pool
# ----------------------------------------------------------------------
def _mix_pool(pool: list, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Random convex combinations of top candidates."""
    best_seq, best_score = pool[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick 2 or 3 distinct candidates
        k = rng.integers(2, 4)          # 2 or 3
        indices = rng.choice(len(pool), size=k, replace=False)
        weights = rng.dirichlet(np.ones(k))
        mix = np.zeros(n)
        for w, idx in zip(weights, indices):
            mix += w * pool[idx][0]
        # ensure normalisation (numerical drift)
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        sc = _objective_score(mix)
        if sc < best_score:
            best_score, best_seq = sc, mix.copy()
            pool = _keep_top(pool, mix, sc)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – Adam optimiser on smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_smoothmax(a0: np.ndarray,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max proxy.
    Temperature τ starts at 0.5 and grows geometrically to ~500.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    tau = 0.5
    tau_target = 500.0
    tau_factor = 1.0015  # very slow increase

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- current true score (for bookkeeping) ----
        conv_full = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv_full.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            # keep pool updated later if desired
        # ---- smooth‑max gradient ----
        max_c = conv_full.max()
        w = np.exp(tau * (conv_full - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv_full)
            w[np.argmax(conv_full)] = 1.0
            w_sum = 1.0
        w /= w_sum

        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev, L_grad)
        grad = 2.0 * conv_w[n - 1: n - 1 + n]   # length n

        # ---- Adam update (minimisation) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature
        tau = min(tau * tau_factor, tau_target)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 4 – aggressive hill‑climb with occasional kicks
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Move mass from indices with positive sub‑gradient to those with
    negative sub‑gradient.  The step size (fraction of source mass) is
    sampled from a wide range to allow both fine and coarse changes.
    After a long plateau a random perturbation (‘kick’) spreads a small
    amount of mass uniformly.
    """
    a = a_start.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    no_imp = 0
    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source – biggest positive gradient among positions with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)  # most negative gradient (might be same as src)

        if src == dst:
            # fallback random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # fraction of source mass to move – uniform in [0.5 %, 20 %]
        frac = rng.random() * 0.195 + 0.005
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # exact evaluation
        conv_new = _conv_self(a_prop, L_fft)
        new_score = float(2.0 * n * conv_new.max())

        if new_score < best_score:
            a = a_prop
            best_score, best_seq = new_score, a.copy()
            no_imp = 0
            # optional: keep best pool if you wish
        else:
            no_imp += 1

        # ----- occasional “kick” to escape plateaus -----
        if no_imp >= 4000:
            # redistribute a small random chunk of the total mass
            kick_frac = 0.02               # 2 % of total mass
            total_mass = a.sum()
            kick_mass = total_mass * kick_frac
            # take mass uniformly from a random subset
            idxs = rng.choice(n, size=int(0.1 * n), replace=False)
            taken = np.minimum(a[idxs], kick_mass / idxs.size)
            a[idxs] -= taken
            # re‑deposit uniformly on all positions
            a += kick_mass / n
            a = _project_simplex(a)       # re‑normalise
            no_imp = 0
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative floats (step‑function heights) that
    minimises ``evaluate_sequence``.  The optimiser follows the five‑stage
    strategy described in the module docstring.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # small safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initialise pool (including optional global starter)
    # --------------------------------------------------------------
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    pool = [(best_seq, best_score)]

    # optional global starter from the previous run
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _objective_score(starter)
            if sc < best_score:
                best_score, best_seq = sc, starter.copy()
            pool = _keep_top(pool, starter, sc)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Warm‑up pool generation
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0   # keep ~5 min for later phases
    if time.time() < phase_deadline:
        pool = _initial_pool(n, phase_deadline)
        best_seq, best_score = pool[0]   # pool is sorted low→high

    # --------------------------------------------------------------
    # 2️⃣  Mixing of top candidates
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _mix_pool(pool, n, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Adam descent on smooth‑max surrogate
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        best_seq, best_score = _adam_smoothmax(best_seq,
                                               phase_deadline,
                                               best_seq,
                                               best_score)

    # --------------------------------------------------------------
    # 4️⃣  Aggressive hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq,
                                           deadline,
                                           best_seq,
                                           best_score)

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
