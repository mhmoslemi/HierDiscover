def _flatten_descent(a0, n, deadline, start_time, best_score_initial, weight=1.0):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a, L_conv)
        # weight for gradient: using conv (maybe after scaling)
        w = conv.copy()
        # Optionally add small uniform regularization to avoid zero gradient
        # compute correlation:
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        start = n - 1
        grad = 2.0 * corr[start:start + n]
        #Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # evaluate objective (max based)
        cur_score = _eval_fast(a, L_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] flatten it={t:,} score={cur_score:.6f}")
        # optional jitter
        if t % 5000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
    return best_a, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search combines several complementary stages within the 1000 s budget:

1️⃣ **Parametric sweep** – fast evaluation of several families
   (geometric, power, beta‑like, sinusoidal, Hann window) possibly mixed
   with the uniform distribution.

2️⃣ **Flattening descent** – Adam‑style projected descent on the L2‑norm
   of the self‑convolution (i.e. minimise Σ conv²).  This flattens the
   convolution and strongly reduces its maximum.

3️⃣ **Soft‑max (smooth‑max) descent** – Adam optimisation of a
   temperature‑controlled soft‑max surrogate of the max‑convolution,
   as in the reference implementation.

4️⃣ **Exact sub‑gradient descent** – projected sub‑gradient steps on the
   true max‑convolution (the same sub‑gradient used previously).

5️⃣ **Microscopic hill‑climb** – random mass transfers that accept only
   improving moves (with a tiny probability of uphill steps).

All vectors are kept on the probability simplex (non‑negative,
sum = 1).  The best sequence seen during the run is returned as a plain
Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility routines
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    a : 1‑D real array (length n)
    L : FFT length (optional). If not given, the next power of two
        ≥ 2·n‑1 is used.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        w ≥ 0, sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective (2·n·max(conv)/sum²) for a vector whose sum is ≈1.
    Uses FFT for the convolution to keep the evaluation cheap.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 2 – flattening descent (minimise Σ conv²)
# ----------------------------------------------------------------------


def _flatten_descent(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     best_score_initial: float):
    """
    Adam‑style projected descent on the L2‑norm of the self‑convolution.
    The gradient of Σ(conv_k)² is 2·correlation(conv, a_rev).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)          # length for convolution w * a_rev
    start = n - 1

    while time.time() < deadline:
        t += 1

        conv = _conv_self_fft(a, L_conv)            # length 2n‑1
        # weight = conv (gradient of Σ conv²)
        w = conv.copy()

        # correlation w * a_rev -> gradient of Σ conv²
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)        # length L_corr
        grad = 2.0 * corr[start:start + n]

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] flatten it={t:,} "
                  f"score={cur_score:.6f}")

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            eps_noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + eps_noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

# Phase 1 – smooth‑max (soft‑max) Adam descent
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_score_initial: float):
    """
    Adam‑style descent on a soft‑max surrogate of max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)
    start = n - 1

    beta = 5.0                 # initial temperature
    beta_mul = 1.02
    beta_max = 800.0           # push a bit further than the reference

    while time.time() < deadline:
        t += 1

        conv = _conv_self_fft(a, L_conv)            # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                      # numerical stabiliser
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()               # soft‑max weights

        # gradient of Σ w_k·conv_k  (as in the reference)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        grad = 2.0 * corr[start:start + n]

        cur_score = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} "
                  f"β={beta:.1f} score={cur_score:.6f}")

        # Adam update + simplex projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # jitter every few thousand iterations
        if t % 4000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    The sub‑gradient is taken from a single maximal shift.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.14          # slightly larger initial step
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for shift k (O(n))
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            # small uphill acceptance (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

def _final_hill_climb(a0: np.ndarray,
                      deadline: float,
                      start_time: float):
    """
    Tiny mass transfers between random coordinates, accepting only
    improving moves (with a tiny probability of uphill steps).
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3      # up to 0.1 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            # occasional uphill move to shake out of tiny plateaus
            if rng.random() < 1e-5:
                a = a_prop

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the evaluation function.  The method combines a parametric sweep,
    a flattening‑descent phase, a smooth‑max Adam phase, a sub‑gradient
    refinement and a final hill‑climb.  The best sequence seen is returned.
    """
    # ------------------------------------------------------------------
    # set up time budget and random seed
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15          # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # baseline: uniform distribution
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64) + 1.0

    p_grid = np.linspace(-3.0, 2.0, 31)

    alpha_grid = np.linspace(-2.0, 2.0, 9)
    beta_grid = np.linspace(-2.0, 2.0, 9)

    # sinusoidal family: a_i ∝ sin(π·(i+0.5)/n)^p  (p≥0)
    p_sin_grid = np.linspace(0.5, 4.0, 8)

    # Hann‑window family (cosine bell)
    hann = 0.5 * (1 - np.cos(2 * np.pi * (np.arange(n) + 0.5) / n))

    for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
        # geometric‑power sweep
        for r in r_grid:
            if time.time() >= deadline - 200:
                break
            geo = np.power(r, np.arange(n, dtype=np.float64))
            for p in p_grid:
                shape = geo * np.power(idx, p)
                s = shape.sum()
                if s == 0.0:

                shape = shape / s
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep geo r={r:.4f} "

        # beta‑like sweep
        for α in alpha_grid:
            if time.time() >= deadline - 200:
                break
            term1 = np.power(idx, α)
            for β in beta_grid:
                term2 = np.power(n - np.arange(n, dtype=np.float64) + 1.0, β)
                shape = term1 * term2
                s = shape.sum()
                if s == 0.0:

        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient final "
                  f"score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  final hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb "
                  f"score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
