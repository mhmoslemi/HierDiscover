"""
Propose a step‐function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1️⃣  **Initialisation** – start from the uniform vector, optionally from a
    supplied global starter, and generate many random Dirichlet samples
    (different concentration parameters).  The best of these candidates is
    kept.

2️⃣  **Simulated annealing with O(n) updates** – the convolution of a vector
    with itself is maintained explicitly.  A move consists of transferring a
    tiny amount of mass from one index to another; the resulting change of the
    convolution can be written in closed form and applied with two vectorised
    slice updates (no FFT needed).  The move is accepted if it improves the
    objective or with a Metropolis probability that slowly cools down.
    The temperature and the size of the transferred mass decay over time.
    Periodically the full convolution is recomputed from scratch to avoid
    drift.

3️⃣  **Final normalisation** – the best vector found is projected back onto the
    probability simplex (non‑negative, sum = 1) and returned as a plain list of
    Python floats.

The whole search respects the 1000 s budget, prints progress messages and
never returns an empty or invalid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using a padded FFT.
    a must be a 1‑D float array.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective for a vector whose entries sum to 1."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (O(n) update of the convolution)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a: np.ndarray,
    n: int,
    cur_score: float,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:

    Runs a temperature‑controlled random walk that transfers tiny amounts
    of mass between two coordinates.  The convolution is updated in O(n) by
    a closed‑form formula, avoiding the O(n log n) cost of an FFT on each step.
    """
    # maintain current convolution
    conv = _conv_self_fft(a)

    best_a = a.copy()
    best_score = cur_score

    # annealing parameters
    T = cur_score * 0.02           # starting temperature
    T_min = 1e-6
    cooling = 0.99995              # geometric cooling per iteration
    step_scale = 0.25              # max fraction of src mass to move
    step_decay = 0.9999            # slowly shrink the step size
    reproj_every = 5000           # periodic re‑projection & full reconvolution

    it = 0
    while time.time() < deadline:
        it += 1

        # pick a source with non‑zero mass
        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # amount to transfer (up to step_scale·a[src])
        eps = a[src] * rng.random() * step_scale
        if eps <= 0.0:
            continue

        ds = -eps          # delta for source
        dd = eps           # delta for destination

        # ----- build candidate convolution (copy + O(n) slice updates) -----
        conv_cand = conv.copy()
        # each unordered pair appears twice in the linear convolution
        conv_cand[src : src + n] += 2.0 * ds * a
        conv_cand[dst : dst + n] += 2.0 * dd * a
        conv_cand[2 * src] += ds * ds
        conv_cand[2 * dst] += dd * dd
        conv_cand[src + dst] += 2.0 * ds * dd

        # evaluate the candidate
        cand_score = _score_from_conv(conv_cand, n)
        delta = cand_score - cur_score

        # Metropolis acceptance
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if T > T_min and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            # commit the move
            a[src] = a[src] + ds
            a[dst] = a[dst] + dd
            conv = conv_cand
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] SA iteration {it:,} "
                    f"improved score={best_score:.6f}"
                )

        # cooling schedule
        T = max(T * cooling, T_min)
        step_scale = max(step_scale * step_decay, 1e-4)

        # occasional re‑projection onto the simplex and full reconvolution
        if it % reproj_every == 0:
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            conv = _conv_self_fft(a)
            cur_score = _score_from_conv(conv, n)
            # keep best copy up‑to‑date in case the re‑projection improved it
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] SA re‑projected "
                    f"new best score={best_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the given evaluation
    function.  The routine creates a random pool of Dirichlet samples,
    refines the best one with a fast simulated‑annealing walk that updates the
    self‑convolution in O(n) per move, and finally returns the best sequence
    found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.2

    n = 4096                         # dimension (moderate, works fast)
    # ------------------------------------------------------------------
    # Baseline – uniform distribution
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    conv = _conv_self_fft(a)
    best_score = _score_from_conv(conv, n)
    best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter is available, consider it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=np.float64)
        starter = np.maximum(starter, 0.0)
        if starter.sum() > 0.0:
            # adjust length to n (truncate or pad with zeros)
            if starter.size > n:
                starter = starter[:n]
            elif starter.size < n:
                starter = np.pad(starter, (0, n - starter.size))
            starter /= starter.sum()
            conv = _conv_self_fft(starter)
            sc = _score_from_conv(conv, n)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Random Dirichlet pool (different concentration parameters)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:          # reserve ~30 s for SA
        # draw a concentration parameter uniformly in a log‑range
        log_alpha = rng.uniform(-2.0, 2.0)       # α ∈ [0.01, 100]
        alpha = 10 ** log_alpha
        x = rng.gamma(alpha, size=n)
        s = x.sum()
        if s == 0.0:
            continue
        x /= s
        conv = _conv_self_fft(x)
        sc = _score_from_conv(conv, n)
        if sc < best_score:
            best_score = sc
            best_seq = x.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet α={alpha:.3f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a = best_seq.copy()
        a, best_score = _simulated_annealing(
            a, n, best_score, deadline, start_time, rng
        )
        best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] SA finished – best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity projection onto the probability simplex
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final best ≈ {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]
