"""
Step‑function optimiser (improved).

The optimiser builds a candidate vector `a` (sum = 1, a_i ≥ 0) that
minimises   2·n·max(conv(a,a)) .  It works in four stages:

1. **Diverse random starts** – many Dirichlet draws with different
   concentration parameters; keep the best.
2. **Smooth‑max ADAM descent** – minimise a differentiable
   approximation of the max‑convolution using the Adam optimiser,
   gradually increasing the “temperature’’ τ so that the smooth
   approximation converges to the true max.
3. **Active‑set LP refinement** – given the current shape `f`,
   solve a small linear programme (maximise Σg_j subject to
   conv(f,g) ≤ M, where M = max(conv(f,f))) using only the active
   constraints (shifts attaining the current maximum).  Replace `f`
   by the normalised LP solution if it improves the objective.  The
   active‑set is expanded if the LP solution violates any previously
   ignored shift.
4. **Targeted hill‑climb** – move a tiny fraction of mass from the
   index with the most positive sub‑gradient to the one with the most
   negative, accepting only improving moves.  The step‑size is
   increased adaptively (up to 5 % of the source mass) to escape flat
   plateaus.

All stages respect the global time budget (default 1000 s).  The final
vector is returned as a plain list of Python floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear convolution via FFT (length = next pow‑2)."""
    if L is None:
        L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative, sum = 1).  O(n) algorithm (see Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) under the assumption sum(a)=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

    trials = 120  # more trials than the previous script

            print(f"[{time.time() - start_time:.1f}s] Dirichlet conc={conc:.3f} → score={sc:.6f}")
    return best_a, best_score

# Stage 2 – smooth‑max Adam descent
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray, deadline: float, start_time: float,
                     best_score: float, best_vec: np.ndarray) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    alpha = 0.08          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the smooth‑max
    tau = 15.0
    tau_factor = 1.03   # very gentle increase

    it = 0
    while time.time() < deadline:
        it += 1
        # self‑convolution and current objective
        conv = _conv_self(a, L_self)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_vec = cur_score, a.copy()
            print(f"[{time.time() - start_time:.1f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth‑max weights (soft‑max over all shifts)
        max_c = conv.max()
        # subtract max_c to avoid overflow; exponents are ≤0
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:               # numerical safety

        grad = 2.0 * conv_w[n - 1: n - 1 + n]   # slice of length n

        # Adam bias‑corrected update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # temperature schedule (make the surrogate sharper)
        if it % 2000 == 0:
            tau = min(tau * tau_factor, 1e7)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Stage 3 – active‑set LP refinement
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, deadline: float, start_time: float,
               best_score: float, best_vec: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a tiny linear programme:
        maximise   Σ g_j
        subject to g ≥ 0  and  conv(a,g) ≤ M
    only for the currently active shifts (those attaining M).
    If the LP solution reduces the objective, replace a by the
    normalised LP solution and repeat.
    """
    import scipy.optimize as opt

    n = a.shape[0]
    # keep a copy that we will improve
    cur = a.copy()
    cur_score = best_score

    # pre‑allocate for speed
    max_lp_iters = 30

    while time.time() < deadline and max_lp_iters > 0:
        max_lp_iters -= 1
        conv_self = _conv_self(cur)          # length 2n‑1
        M = conv_self.max()
        # indices where the max is attained (allow small tolerance)
        active = np.where(conv_self >= M - 1e-12)[0].tolist()
        # ensure we have at least one constraint (should always be true)
        if not active:
            break

        # ------------------------------------------------------------------
        # Build the constraint matrix for the current active set
        # ------------------------------------------------------------------
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1, dtype=int)
            is_ = k - js
            A[row, js] = cur[is_]

        c = -np.ones(n)                     # maximise Σg_j  → minimise −Σg_j
        bounds = [(0.0, None)] * n
        b = np.full(len(active), M, dtype=np.float64)

        # ------------------------------------------------------------------
        # Solve the LP (highs is fast for our tiny problems)
        # ------------------------------------------------------------------
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 5.0})
        if not res.success:
            # If the LP failed, abort the LP stage.
            print(f"[{time.time() - start_time:.1f}s] LP failed (status={res.message})")
            break

        g = res.x
        # normalise to sum = 1
        if g.sum() == 0:
            # degenerate solution – abort LP refinement
            break
        g_norm = g / g.sum()

        # ------------------------------------------------------------------
        # Verify feasibility w.r.t. *all* shifts (not only the active ones)
        # ------------------------------------------------------------------
        conv_ag = _conv_fft(cur, g_norm)
        viol = np.where(conv_ag > M + 1e-12)[0]
        if viol.size == 0:
            # feasible – evaluate the new objective
            new_score = _objective_score(g_norm)
            if new_score < cur_score - 1e-9:
                cur, cur_score = g_norm, new_score
                if new_score < best_score:
                    best_score, best_vec = new_score, g_norm.copy()
                print(f"[{time.time() - start_time:.1f}s] LP refinement improved score={new_score:.6f}")
                # continue with the new vector
                continue
            else:
                # No improvement – stop LP refinement
                break
        else:
            # Add the violating shifts to the active set and retry
            new_active = set(active).union(set(viol.tolist()))
            active = list(new_active)
            # Loop again with a larger active set
            continue

    return best_vec, best_score


# ----------------------------------------------------------------------
# Stage 4 – directed hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, start_time: float,
                best_score: float, best_vec: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Transfer a small (or moderate) amount of mass from the index with the
    most positive sub‑gradient to the index with the most negative.
    Accept only moves that strictly improve the objective.
    """
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)
    cur = a.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    no_improve_cnt = 0

    while time.time() < deadline:
        conv = _conv_self(cur, L_self)
        grad = _subgradient_maxconv(cur, conv)

        # source = index where increasing a would raise the max (positive grad)
        src_cand = np.where(cur > 1e-15)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)   # most negative gradient

        if src == dst:
            # fallback random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or cur[src] < 1e-15:
                continue

        # amount to move – try a relatively bold step (up to 5 % of source)
        max_frac = min(0.05, cur[src] / (cur[src] + 1e-12))
        frac = rng.random() * max_frac + 1e-6
        eps = cur[src] * frac
        if eps <= 0:
            continue

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        # maintain feasibility (non‑negative)
        if cand[src] < -1e-12:
            continue
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < cur_score - 1e-12:
            cur, cur_score = cand, cand_score
            no_improve_cnt = 0
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.1f}s] Hill‑climb improved score={cand_score:.6f}")
        else:
            no_improve_cnt += 1
            if no_improve_cnt > 3000:
                # stagnation – exit early
                break

    return best_vec, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence``.  The routine follows the four‑stage
    strategy described above and respects the given time budget.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (the reference run used 4096 – we keep it)
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Initial vector – try the supplied global starter first
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a)

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_a, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.1f}s] Global starter improves score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Stage 1 – random Dirichlet initialisers
    # ------------------------------------------------------------------
    phase_deadline = deadline - 600.0   # reserve ~10 min for later stages
    if time.time() < phase_deadline:
        best_a, best_score = _random_dirichlet_initialisers(
            n, phase_deadline, rng
        )
        best_vec = best_a.copy()
        print(f"[{time.time() - start_time:.1f}s] After Dirichlet stage score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max Adam descent
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 250.0   # keep ~250 s for LP + hill‑climb
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.1f}s] Starting Adam smooth‑max descent")
        best_vec, best_score = _adam_smooth_max(
            best_a, smooth_deadline, start_time, best_score, best_a
        )
        print(f"[{time.time() - start_time:.1f}s] After Adam descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – LP refinement (active‑set)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 100.0   # leave ~100 s for final hill‑climb
    if time.time() < lp_deadline:
        print(f"[{time.time() - start_time:.1f}s] Starting LP refinement")
        best_vec, best_score = _lp_refine(
            best_vec, lp_deadline, start_time, best_score, best_vec
        )
        print(f"[{time.time() - start_time:.1f}s] After LP refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final directed hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.1f}s] Starting final hill‑climb")
        best_vec, best_score = _hill_climb(
            best_vec, deadline, start_time, best_score, best_vec
        )
        print(f"[{time.time() - start_time:.1f}s] Final hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (just in case) and return plain Python list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec = best_vec / total
    else:
        best_vec = uniform.copy()

    print(f"[{time.time() - start_time:.1f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_vec]
