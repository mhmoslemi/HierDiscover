"""
Improved search for a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines several complementary phases:

1. **Problem size** – uses n = 8192 when time allows (more freedom).
2. **Parametric sweep** – explores families a_i ∝ r^i·(i+1)^p together with a
   truncated Gaussian family.  The best vector from this sweep seeds the next
   phases.
3. **Smooth‑max Adam descent** – minimises a soft‑max surrogate of max(conv)
   with an increasing temperature β.  Several random seeds are tried.
4. **Mirror‑descent on the exact max‑convolution** – a sub‑gradient method on
   the true max‑term using a multiplicative (entropy) update and projection onto
   the probability simplex.
5. **Targeted stochastic hill‑climb** – random mass‑transfer moves plus
   specialised moves that directly reduce the dominant convolution entry.
   A simple simulated‑annealing acceptance rule helps escape tiny plateaus.
6. The best vector is normalised to sum 1 and returned as a plain list of
   Python floats.

All heavy work is guarded by time checks so the routine always finishes
inside the 1000 s budget.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (w≥0, sum(w)=1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max surrogate utilities (used by Adam descent)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    conv = _conv_self(a, L=L_conv)          # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (stable)
    shifted = conv - max_c                 # max becomes 0
    exp_vals = np.exp(beta * shifted)      # under‑flow for large negative values is harmless
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]                        # length n
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)   # linear convolution w * a_rev

    start = a.shape[0] - 1                 # n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 800.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the smooth‑max surrogate.
    β starts small (smooth landscape) and is slowly increased up to beta_max.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0               # start fairly smooth
    beta_mul = 1.02          # geometric increase per iteration

    while time.time() < deadline and t < max_iters:
        t += 1

        # gradient of the smooth surrogate
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        # exact score (sum of a is 1 → denominator = 1)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] smooth it={t:,} β={beta:.1f} score={cur_score:.6f}")

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # base learning‑rate decays slowly with the number of iterations
        lr = 0.5 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to help escape plateaus
        if t % 3000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (sub‑gradient method)
# ----------------------------------------------------------------------
def _mirror_descent_maxconv(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true objective using an entropy‑mirror (multiplicative)
    update.  At each iteration we pick the currently maximal convolution index k,
    compute the sub‑gradient 2·a_{k‑i} and perform a multiplicative step followed
    by projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        M = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient of the max entry (2·a_{k‑i} where defined)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k - idx]

        # diminishing step size
        eta = 0.25 / math.sqrt(t)

        # mirror (entropy) update: a_i ← a_i * exp(-η·grad_i)
        a = a * np.exp(-eta * grad)

        # projection back onto the simplex (keeps sum = 1)
        a = _project_simplex(a)

        cur_score = 2.0 * n * M / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] mirror it={t:,} score={cur_score:.6f}")

    return best_a, best_score

    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # step‑size policy
    step_factor = 0.15          # up to 15 % of source mass per proposal
    min_factor = 1e-6
    decay = 0.995

    # simulated‑annealing temperature schedule
    temp0 = 1e-3
    temp_decay = 0.9995

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # decide move type
        if rng.random() < 0.6:
            # ----- random transfer -----------------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # ----- targeted transfer (focus on maximal convolution) -----
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)

        # acceptance rule (allow occasional uphill moves)
        temperature = temp0 * (temp_decay ** iter_cnt)
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_score - best_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept and new_score <= best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)   # reset after a success
            print(f"[{time.time()-start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            # shrink step size on rejection
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Coarse‑to‑fine parametric sweep (geometric‑power & Gaussian families)
# ----------------------------------------------------------------------
def _parametric_sweep(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore families:
        a_i ∝ r^i·(i+1)^p                (geometric × power‑law)
        a_i ∝ exp(-((i-μ)^2)/(2σ^2))    (truncated Gaussian)
    and convex combinations with the uniform vector.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)        # i+1
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=28)
    p_grid = np.linspace(-4.0, 2.0, 41)

    L_fft = _next_pow2(2 * n - 1)

    # ----- geometric × power‑law -----
    for r in r_grid:
        if time.time() >= deadline - 140:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 140:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time()-start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # ----- truncated Gaussian family -----
    sigma_factors = (0.10, 0.15, 0.20, 0.25)
    mu_factors = (0.30, 0.40, 0.50, 0.60, 0.70)
    for sigma_fac in sigma_factors:
        if time.time() >= deadline - 140:
            break
        sigma = sigma_fac * n
        for mu_fac in mu_factors:
            if time.time() >= deadline - 140:
                break
            mu = int(mu_fac * n)
            diff = np.arange(n) - mu
            shape = np.exp(-0.5 * (diff / sigma) ** 2)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in (0.0, 0.3, 0.6, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time()-start_time:.2f}s] gauss μ={mu} σ={sigma:.0f} λ={lam:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects the 1000 s budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Small safety margin to avoid hitting the wall
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # Choose a problem size: 8192 when we have enough time, otherwise 4096
    # ------------------------------------------------------------------
    if budget_s >= 800:
        n = 8192
    else:
        n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline: uniform vector
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time()-start_time:.2f}s] baseline uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Use a globally provided starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time()-start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric‑power + Gaussian families)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:   # keep ~5 min for later phases
        best_seq, best_score = _parametric_sweep(
            n, uniform, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam descent (multiple seeds)
    # ------------------------------------------------------------------
    # Decide how many seeds we can afford based on remaining time
    remaining = deadline - time.time()
    # Rough estimate: 0.03 s per iteration -> 20000 iterations ≈ 600 s for 3 seeds,
    # so we adaptively scale.
    descent_seeds = [seed, seed + 1, seed + 2, seed + 3, seed + 4]
    for ds in descent_seeds:
        if time.time() >= deadline - 150:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        a0 += 1e-4 * rng.standard_normal(n) * a0   # tiny jitter
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        # allocate iterations proportionally to remaining time
        time_left = deadline - time.time()
        max_iters = int(min(40000, max(5000, time_left * 40)))   # very rough
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,          # leave a little room for polishing
            start_time,
            best_score,
            beta_max=1200.0,        # allow a higher temperature for finer max approximation
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time()-start_time:.2f}s] smooth descent improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Mirror‑descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        # allocate a few thousand iterations (the update is cheap)
        iters_md = int(min(15000, (deadline - time.time()) * 30))
        a_opt, sc_opt = _mirror_descent_maxconv(
            best_seq,
            iters_md,
            deadline - 30,
            start_time,
            best_score,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time()-start_time:.2f}s] mirror descent improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time()-start_time:.2f}s] final hill‑climb score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check with the official evaluator (optional, may be omitted)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time()-start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
