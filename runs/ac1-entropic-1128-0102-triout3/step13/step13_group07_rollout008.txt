"""
Smooth‑max Adam optimiser for the step‑function bound.

The original bound is 2·n·max(conv(a,a)) with a_i ≥ 0, Σa_i = 1.
We minimise the smooth approximation
        g(a) = (1/α)·log Σ_k exp(α·conv_k(a))
and use its exact gradient:
        ∇g_i = 2·n·Σ_k (softmax_k)·a_{k-i}
The softmax weights are w_k = exp(α·(conv_k–max(conv))) / Σ exp(·).
A projected Adam step keeps a on the probability simplex.
After a long Adam phase a cheap greedy hill‑climb is applied
to polish the solution.

The routine works for any length n that fits into the 1000 s budget.
It automatically warm‑starts from several analytic shapes,
a (optional) previously supplied global `height_sequence_1`,
and a few random Dirichlet draws.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families + random Dirichlet draws
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)                # zero at the ends
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) constant (uniform)
    candidates.append(np.ones_like(x))

    # normalise each candidate to the simplex
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)

    # ------------------------------------------------------------------
    # random Dirichlet draws – give a stochastic spread of start points
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(6):
        rand_vec = rng.dirichlet(np.ones(n))
        normals.append(rand_vec)

    # ------------------------------------------------------------------
    # optionally incorporate a globally supplied construction
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            h1_resampled = np.maximum(h1_resampled, 0.0)
            s = h1_resampled.sum()
            if s > 0:
                normals.append(h1_resampled / s)
    except Exception:
        pass

    return normals


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (exact weighted gradient)
# ----------------------------------------------------------------------
def _adam_smooth_optimize(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.05,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          alpha_start: float = 2.0,
                          alpha_inc: float = 0.02,
                          alpha_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam on the probability simplex using the gradient of a
    soft‑max (log‑sum‑exp) approximation of max(conv).
    Returns the best iterate seen and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT lengths (static for the whole run)
    L_self = _next_pow2(2 * n - 1)          # for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # for a * w_rev convolution

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    alpha = alpha_start

    while time.time() < deadline:
        it += 1

        # ----- self‑convolution (needed for the soft‑max weights) -----
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

        # ----- soft‑max weights over all shifts -----
        conv_max = conv.max()
        # shift by the maximum to keep the exponent range moderate
        w_raw = np.exp(alpha * (conv - conv_max))
        w_sum = w_raw.sum()
        if w_sum == 0:
            w_norm = np.zeros_like(w_raw)
        else:
            w_norm = w_raw / w_sum

        # ----- weighted gradient of the smoothed objective -----
        # gradient_i = 2·n· Σ_k w_k·a_{k‑i}
        # which is a convolution of a with the reversed weight vector.
        w_rev = w_norm[::-1]                       # reversed weights
        # forward FFTs (different lengths)
        Af_big = np.fft.rfft(a, n=L_grad)
        Bf = np.fft.rfft(w_rev, n=L_grad)
        conv_rev = np.fft.irfft(Af_big * Bf, n=L_grad)
        len_w = 2 * n - 1
        # extract the needed slice: indices len_w-1‑i (i=0..n‑1)
        grad = 2.0 * n * conv_rev[len_w - 1 - np.arange(n)]

        # ----- Adam update on the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

        # ----- evaluate the true objective -----
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth it={it:,} "
                  f"α={alpha:.2f} score={cur_score:.6f}")

        # increase the soft‑max sharpness slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # occasional learning‑rate decay (helps in the long run)
        if it % 3000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy stochastic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepting only improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        # pick source proportionally to current mass – encourages moves
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # amount to move: up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:                     # greedy acceptance
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    The function returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the evaluator
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two makes the FFT fast.
    # ------------------------------------------------------------------
    n = 4096  # works comfortably within the time budget

    # ------------------------------------------------------------------
    # Warm‑start – pick the analytically and randomly generated best.
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Smoothed‑max Adam optimisation (majority of the budget).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate roughly 70 % of the remaining time to Adam
        adam_deadline = time.time() + remaining * 0.70
        a_opt, score_opt = _adam_smooth_optimize(
            a0, adam_deadline, start_time,
            lr=0.05, alpha_start=2.0, alpha_inc=0.02, alpha_max=30.0
        )
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑smooth best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 2) Light hill‑climb polish (use whatever time remains).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Final normalisation & conversion to a plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)                # safety
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
