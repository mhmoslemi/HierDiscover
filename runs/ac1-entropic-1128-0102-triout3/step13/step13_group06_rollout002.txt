"""
Simple simulated‑annealing optimiser for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm:
1. Build a diverse collection of analytic “warm‑start” shapes and keep the
   best one (exact objective via FFT).
2. Starting from this vector, repeatedly transfer a small amount of mass from
   a source index to a destination index.  The move is accepted either if it
   improves the objective or with a probability exp(‑Δ/temperature) (simulated
   annealing).  The convolution is updated incrementally in O(n) per move.
3. After the time budget the best vector is optionally symmetrised and smoothed
   (local refinement) and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for an arbitrary non‑negative vector."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`.

    Updates `a` and the linear self‑convolution `conv` in‑place.
    """
    n = a.shape[0]

    # 1) linear term 2·a*Δ   (Δ = eps·(e_dst – e_src))
    #    subtract contribution of source
    conv[src : src + n] -= 2.0 * eps * a
    #    add contribution of destination
    conv[dst : dst + n] += 2.0 * eps * a

    # 2) quadratic term Δ·Δ
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # finally modify the probability vector
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Candidate analytic shapes (all produce a non‑negative vector of length n)
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of analytical shapes (unnormalised)."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Symmetric exponentials
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]                     # symmetric
    candidates.append(shape)

    # 5) Half‑cosine (already used as warm‑start but keep)
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet draws – a few of them
    rng = np.random.default_rng(0)
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # Normalise and filter
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective non‑negative step function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the harness will stop the function after
        this many seconds; we stop a bit earlier to be safe).

    Returns
    -------
    list[float]
        The best sequence found, ready for ``evaluate_sequence``.
    """
    # --------------------------------------------------------------
    # Setup
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)
    start_time = time.time()
    # Safety margin for the harness (≈0.5 s)
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 4096                               # default size

    # --------------------------------------------------------------
    # Warm‑start: generate a collection of analytic shapes
    # --------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # include the previous global vector if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm_starts.append(starter / starter.sum())
    except Exception:
        pass

    # a few extra random Dirichlet draws (helps diversification)
    for _ in range(12):
        rnd = rng.random(n)
        if rnd.sum() > 0.0:
            warm_starts.append(rnd / rnd.sum())

    # --------------------------------------------------------------
    # Pick the best warm‑start according to the exact objective
    # --------------------------------------------------------------
    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()

    a = best_vec.copy()                # current working vector (sum == 1)
    conv = _conv_fft(a)                # current convolution
    total = a.sum()
    n_obj = 2.0 * n / (total * total)  # factor: objective = n_obj * conv.max()
    cur_score = best_score

    print(f"[{time.time() - start_time:.2f}s] Warm‑start: n={n}, best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing hill‑climbing
    # --------------------------------------------------------------
    # Annealing schedule parameters
    T_start = 0.008                     # initial temperature (objective units)
    T_end = 1e-6                        # final temperature
    frac_start = 0.30                   # max fraction of source mass moved initially
    frac_end = 0.02                     # min fraction near the end
    max_iter_without_improve = 500_000
    iters_without_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)       # 0 … 1
        # linear schedules
        temp = T_start * (1.0 - progress) + T_end * progress
        frac_max = frac_start * (1.0 - progress) + frac_end * progress

        # pick source proportional to its mass (helps moving larger packets)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # random fraction of the source mass
        eps = a[src] * rng.random() * frac_max
        if eps <= 0.0:
            continue

        # current objective
        cur_obj = n_obj * conv.max()

        # apply the move
        _apply_move(a, conv, src, dst, eps)

        new_obj = n_obj * conv.max()

        # acceptance test
        if new_obj <= cur_obj or rng.random() < math.exp(-(new_obj - cur_obj) / temp):
            # accept
            if new_obj < best_score:
                best_score = new_obj
                best_vec = a.copy()
                iters_without_improve = 0
                print(f"[{time.time() - start_time:.2f}s] iter={it:,}  new best={best_score:.6f}")
            else:
                iters_without_improve += 1
        else:
            # reject – revert
            _apply_move(a, conv, dst, src, eps)
            iters_without_improve += 1

        # occasional progress report
        if it % 500_000 == 0:
            print(f"[{time.time() - start_time:.2f}s] iter={it:,}  cur={cur_obj:.6f}  best={best_score:.6f}")

        # if we have been stuck for a long time, perform a small random shake
        if iters_without_improve >= max_iter_without_improve:
            # shake: pick a random source with non‑negligible mass and move a tiny packet
            src2 = rng.choice(n, p=a)
            dst2 = rng.integers(n)
            if src2 != dst2 and a[src2] > 1e-12:
                eps2 = a[src2] * 0.005 * rng.random()
                _apply_move(a, conv, src2, dst2, eps2)
                # we keep the shake unconditionally (it may be uphill)
                iters_without_improve = 0

    # --------------------------------------------------------------
    # Local refinement of the best vector (symmetrisation + smoothing)
    # --------------------------------------------------------------
    def _refine(vec: np.ndarray) -> np.ndarray:
        """Try a few simple deterministic refinements."""
        best_local = vec.copy()
        best_val = _objective(best_local)

        # 1) enforce exact symmetry
        sym = (best_local + best_local[::-1]) / 2.0
        sym /= sym.sum()
        val = _objective(sym)
        if val < best_val:
            best_local, best_val = sym, val

        # 2) simple moving‑average smoothing (window size 3)
        for _ in range(3):
            smooth = np.convolve(best_local, np.ones(3) / 3.0, mode='same')
            smooth = np.maximum(smooth, 0.0)
            smooth /= smooth.sum()
            val = _objective(smooth)
            if val < best_val:
                best_local, best_val = smooth, val
        return best_local

    if best_vec is not None:
        best_vec = _refine(best_vec)

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    final_obj = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished – best objective={final_obj:.6f}")

    return [float(v) for v in best_vec]
