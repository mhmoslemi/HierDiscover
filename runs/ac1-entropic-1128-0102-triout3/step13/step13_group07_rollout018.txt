"""
Extended step‑function optimiser.
--------------------------------
* Generates a large, diverse set of analytic families (exponential,
  power‑law, beta‑distribution, random Dirichlet etc.).
* Picks the best analytic shape w.r.t. the exact objective
  `2·n·max(conv(a,a))` (sum of the vector is normalised to 1).
* Refines the best candidate with a projected Adam optimiser (exact
  sub‑gradient of the current maximal convolution entry).
* Finishes with a simple simulated‑annealing hill‑climber that can
  escape tiny local minima.
* All work respects the 1000 s budget – if time runs out the current
  best vector is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numerical helpers (FFT based convolution, simplex projection, …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns the valid part (length 2·n‑1).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) under the constraint sum(a)=1.
    (The function evaluate_sequence is scale‑invariant, therefore we
    normalise to sum = 1 and use the closed‑form expression.)
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                         # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑start generation (large, diverse analytic families)
# ----------------------------------------------------------------------
def _make_rich_warm_starts(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Create many candidate vectors (all normalised to sum=1).
    Includes deterministic families plus a sweep over several
    parameterised shapes and a handful of random draws.
    """
    x = np.linspace(0.0, 1.0, n)               # domain ∈[0,1]

    candidates = []

    # ---- deterministic families (same as the reference code) ----
    tent = np.minimum(x, 1.0 - x)                     # symmetric tent
    candidates.append(tent)

    cosine = np.sin(np.pi * x)                       # half‑cosine, zero at ends
    candidates.append(cosine)

    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))  # half‑circle
    candidates.append(circ)

    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)  # Gaussian centre
    candidates.append(gauss)

    for p in np.linspace(0.2, 2.0, 9):               # power‑tent
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # ------------------------------------------------------------------
    # Parameter sweeps over useful families
    # ------------------------------------------------------------------
    # Exponential decays (forward and reversed)
    for scale in np.linspace(0.3, 5.0, 20):
        exp_fwd = np.exp(-scale * x)
        candidates.append(exp_fwd)
        candidates.append(exp_fwd[::-1])

    # Monotone power‑law decreasing: (1‑x)^p
    for p in np.linspace(0.2, 5.0, 12):
        dec = (1.0 - x) ** p
        candidates.append(dec)
        candidates.append(dec[::-1])                 # the increasing counterpart

    # Pure power‑law: x^p
    for p in np.linspace(0.2, 5.0, 12):
        inc = x ** p
        candidates.append(inc)
        candidates.append(inc[::-1])

    # Beta‑distribution shapes (asymmetric, flexible)
    for _ in range(30):
        a = rng.uniform(0.3, 5.0)
        b = rng.uniform(0.3, 5.0)
        beta = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
        candidates.append(beta)

    # Random Dirichlet draws (different concentration parameters)
    for _ in range(30):
        alpha = rng.uniform(0.3, 3.0)
        vec = rng.gamma(alpha, size=n)
        candidates.append(vec)

    # ------------------------------------------------------------------
    # Normalise, clip negative numbers and discard zero‑vectors
    # ------------------------------------------------------------------
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Exact sub‑gradient of the maximal convolution entry (used by Adam)
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    conv = a * a (full linear convolution, length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())               # index where the maximum occurs
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]          # g_i = 2·a_{k‑i}
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of the current
    maximal convolution entry.  Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # full self‑convolution
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # sub‑gradient of the current maximum entry
        grad = _max_subgradient(a, conv)

        # scale to the true objective 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

        # current objective (no extra FFT needed)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cur_score:.6f}")

        # optional learning‑rate decay (helps the long run)
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (accepts uphill moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis walk with a slowly cooling temperature.
    Only cheap moves (transfer a fraction of mass from one index to another)
    are considered.  The best ever seen vector is returned.
    """
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    rng = np.random.default_rng()
    # start temperature – a few percent of the current score
    T = 0.02 * cur_score
    T_min = 1e-9 * cur_score
    decay = 0.99996                      # exponential cooling per iteration

    it = 0
    while time.time() < deadline:
        it += 1

        # pick source proportionally to its mass – encourages meaningful moves
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        new_score = _objective(a_new)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} "
                      f"new best={new_score:.6f}")

        # temperature schedule
        T = max(T * decay, T_min)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine follows the three‑stage plan
    described in the module docstring and returns a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Basic initialisation
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin (ensure we return before the external timeout)
    deadline = start_time + budget_s - 0.5

    # problem size – a power of two works well with the FFT backend
    n = 4096

    # ------------------------------------------------------------------
    # 1) Rich warm‑starts – pick the best analytic / random shape
    # ------------------------------------------------------------------
    warm_starts = _make_rich_warm_starts(n, rng)

    # optionally insert a globally supplied construction (if present)
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            # resample to length n (linear interpolation)
            xs = np.linspace(0.0, 1.0, h1.size)
            xq = np.linspace(0.0, 1.0, n)
            h1_resampled = np.interp(xq, xs, h1)
            h1_resampled = np.maximum(h1_resampled, 0.0)
            s = h1_resampled.sum()
            if s > 0.0:
                warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    # evaluate the candidates – abort early if we run out of budget
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
        if time.time() > deadline - 20.0:   # keep a little slack for optimisation
            break

    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Projected Adam optimisation (exact sub‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # a small safety net for the final polish
        adam_deadline = time.time() + remaining * 0.65   # ~65 % of the remaining time
        a_opt, score_opt = _adam_optimize(a0, adam_deadline, start_time,
                                          lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing final polish (uses whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _simulated_annealing(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final SA best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, "
          f"length={n}")
    return [float(v) for v in a0]
