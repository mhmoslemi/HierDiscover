def simulated_annealing(a0, start_time, deadline):
    best = a0.copy()
    best_score = _eval_fast(best, L_fft)
    a = best.copy()
    T0 = 1.0
    Tmin = 1e-5
    total_iter = 0
    while time.time() < deadline:
        # schedule: temperature decays exponentially
        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        T = T0 * (1 - fraction) + Tmin * fraction
        # propose move
        src = rng.integers(n)
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1
        # decide mass transfer amount
        eps = a[src] * rng.random() * 0.1   # up to 10%
        if eps == 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # ensure non-negative
        if a_prop[src] < 0:
            continue
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")
        total_iter += 1
    return best, best_score
```

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear self‑convolution.  The optimiser works in
four stages inside the 1000 s budget:

1️⃣  A dense parametric sweep over families a_i ∝ (i+1)^p · r^i
    (geometric‑power shapes) mixed with the uniform distribution.
2️⃣  Adam‑style projected descent on a smooth‑max surrogate of the
    true objective (temperature β is increased slowly).
3️⃣  A simulated‑annealing phase that repeatedly moves a tiny amount of
    mass between two coordinates; moves are accepted with a
    Metropolis probability that slowly cools.
4️⃣  A final deterministic hill‑climb that transfers mass between
    random pairs only if it strictly improves the score.

All vectors are projected onto the probability simplex (non‑negative,
sum = 1) after each modification, so the evaluation is invariant to
scale.  The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0, sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – exhaustive parametric sweep
# ----------------------------------------------------------------------


def _parametric_sweep(n: int, deadline: float, start_time: float,
                      uniform: np.ndarray, best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Sweep over a_i ∝ (i+1)^p · r^i and mixtures with the uniform vector.
    The grids are deliberately fine – the whole sweep usually finishes
    within a few dozen seconds.
    """
    L_fft = _next_pow2(2 * n - 1)
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based indices

    # geometric ratio grid – denser near 1
    r_grid = np.logspace(np.log10(0.90), np.log10(0.999999), num=200)
    # power‑law exponent grid
    p_grid = np.linspace(-6.0, 6.0, num=121)
    lam_grid = (0.0, 0.2, 0.5, 0.8, 1.0)                # mixing with uniform

    for r in r_grid:
        if time.time() >= deadline - 150:
            break
        log_r = np.log(r)
        for p in p_grid:
            if time.time() >= deadline - 150:
                break
            # shape = (i+1)^p * r^i
            shape = np.exp(p * np.log(idx) + np.arange(n) * log_r)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s                                 # normalise to sum = 1
            for lam in lam_grid:
                cand = lam * uniform + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.6f} "
                          f"p={p: .2f} λ={lam:.1f} → score={sc:.6f}")

    # a few random draws to broaden the search
    rng = np.random.default_rng(12345)
    while time.time() < deadline - 120:
        r = np.exp(rng.uniform(np.log(0.90), np.log(0.999999)))
        p = rng.uniform(-8.0, 8.0)
        lam = rng.choice(lam_grid)
        shape = np.exp(p * np.log(idx) + np.arange(n) * np.log(r))
        s = shape.sum()
        if s == 0.0:
            continue
        shape /= s
        cand = lam * uniform + (1.0 - lam) * shape
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] rand r={r:.6f} p={p:.2f} "
                  f"λ={lam:.1f} → score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – Adam descent on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, best_score_initial: float):
    """
    Adam‑style projected descent on the soft‑max of the convolution.
    The temperature β starts at 5 and is multiplied by 1.03 each step,
    caps at 800.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0
    beta_mul = 1.03
    beta_max = 800.0

    while time.time() < deadline:
        t += 1

        # soft‑max over the 2n‑1 convolution entries
        conv = _conv_self_fft(a, L_conv)
        max_c = conv.max()
        shifted = conv - max_c
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth‑max term
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} β={beta:.1f} "
                  f"score={cur_score:.6f}")

        # Adam update + simplex projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.15 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional tiny jitter to escape plateaus
        if t % 3500 == 0:
            jitter = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated annealing (mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """
    Classical SA: repeatedly move a small fraction of the mass from a
    randomly chosen source index to a randomly chosen destination.
    Acceptance probability is exp(-Δ / T) with a linear cooling schedule.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    T_start = 1.0
    T_end = 1e-5

    total_iters = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = (elapsed) / (deadline - start_time)
        T = T_start * (1 - frac) + T_end * frac

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move up to 5 % of the source mass (larger steps early on)
        max_move = min(a[src], a[src] * rng.random() * 0.05)
        if max_move <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= max_move
        a_prop[dst] += max_move

        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a_prop.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={total_iters:,} "
                      f"T={T:.3e} → score={new_score:.6f}")
        total_iters += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – deterministic hill‑climb (microscopic polishing)
# ----------------------------------------------------------------------


def _final_hill_climb(a0: np.ndarray, deadline: float, start_time: float):
    """Only accept strictly improving tiny mass transfers."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-4      # ≤0.01 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] hill‑climb → score={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine consists of a parametric sweep, Adam descent,
    simulated annealing and a final hill‑climb.  It respects the 1000 s
    budget and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2           # tiny safety margin

    # Problem size – we keep it moderate (4 k) to stay within the budget.
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optional global starter (previous best)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (dense + random)
    # ------------------------------------------------------------------
    best_seq, best_score = _parametric_sweep(
        n, deadline, start_time, uniform, best_seq, best_score
    )

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style smooth‑max descent (few restarts)
    # ------------------------------------------------------------------
    remaining = deadline - time.time() - 200
    if remaining > 30:
        restarts = 2
        per_restart = remaining / restarts
        for i in range(restarts):
            if time.time() >= deadline - 200:
                break
            rng = np.random.default_rng(seed + i + 777)
            a0 = best_seq.copy()
            # inject a modest amount of noise
            a0 += rng.normal(scale=0.01, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            sub_deadline = time.time() + per_restart
            a_smooth, sc_smooth = _smooth_descent(
                a0, n, sub_deadline, start_time, best_score
            )
            if sc_smooth < best_score:
                best_score = sc_smooth
                best_seq = a_smooth.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam restart {i+1}/{restarts} "
                      f"new best score={sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated annealing (global exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100:
        sa_deadline = deadline - 80
        a_sa, sc_sa = _simulated_annealing(
            best_seq, n, sa_deadline, start_time, best_score
        )
        if sc_sa < best_score:
            best_score = sc_sa
            best_seq = a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] SA finished best score={sc_sa:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise (sum → 1) and return as a Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
