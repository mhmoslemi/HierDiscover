def _smooth_optimize(a0, deadline, start_time):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _objective(a)  # actual objective

    iter = 0

        # compute conv with FFT
        L_self = _next_pow2(2*n-1)
        Af = np.fft.rfft(a, n=L_self)

        max_conv = conv.max()
        exp_args = alpha * (conv - max_conv)
        w = np.exp(exp_args)  # stable
        sum_w = w.sum()
        w_norm = w / sum_w
        # compute gradient via convolution a * w_norm_rev
        L_grad = _next_pow2(n + len(w_norm) - 1)
        A_fft = np.fft.rfft(a, n=L_grad)
        W_fft = np.fft.rfft(w_norm[::-1], n=L_grad)
        grad_full = np.fft.irfft(A_fft * W_fft, n=L_grad)
        start_idx = len(w_norm) - 1
        grad = 2.0 * grad_full[start_idx : start_idx + n]  # gradient of smooth max
        # step size schedule: use simple diminishing step size
        lr = 0.05 / (1 + iter * 1e-5)  # small decaying
        a -= lr * grad
        # projection onto simplex
        a = _project_simplex(a)

        if iter % 2000 == 0 and alpha < alpha_max:

"""
Propose a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a))**2

Algorithm
---------
1. **Warm‑starts** – generate a diverse pool of analytic shapes
   (tent, cosine, circle, Gaussian, exponential, powers, constant) and a few
   random Dirichlet vectors.  The best shape (lowest exact objective) is kept.

2. **Projected Adam on the exact sub‑gradient** – quick early descent.
   The sub‑gradient of the max‑convolution entry is
        g_i = 2·a_{k‑i}   (k = argmax conv)
   Adam updates are projected onto the probability simplex.

3. **Smooth‑max gradient descent (continuation)** – replaces the hard max by
   a soft‑max with temperature α.  The objective
        F_α(a) = max(conv) + (1/α)·log Σ exp(α·conv)
   has the gradient
        ∇F_α = 2·a * w_rev   (convolution with the normalised soft‑max weights)
   α is increased slowly (continuation) so that the optimiser first sees a
   smooth landscape and finally focuses on the true max.

4. **Greedy hill‑climbing polish** – cheap Metropolis moves that only accept
   strict improvements.  This often squeezes a few 10⁻⁴ out of the objective.

All heavy work uses real FFTs; all vectors are explicitly projected onto the
simplex (Σ a = 1, a_i ≥ 0).  The routine respects the given time budget – if
the budget expires the current best sequence is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·n‑1) using a real FFT of length L.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families + random Dirichlet samples
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]
    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)                # zero at the ends
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) constant (uniform)
    candidates.append(np.ones_like(x))

    # normalise each candidate to sum = 1 and keep only the non‑negative ones.
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Projected Adam optimiser (sub‑gradient of the hard max)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of max(conv).
    Returns the best iterate seen and its true objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # self‑convolution (full length 2·n‑1)
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # sub‑gradient of the current max entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Scale because the true objective is 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

        # current true objective (no extra FFT needed)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cur_score:.6f}")

        # occasional learning‑rate decay
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max gradient descent with continuation on α
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     alpha_start: float = 1.0,
                     alpha_end: float = 120.0,
                     lr0: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Minimise the true objective using a smooth approximation of the max.
    α is increased geometrically (continuation) to focus on the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    alpha = alpha_start
    it = 0

    # Pre‑allocate length for the large FFT (covers both self‑conv and a*w)
    L = _next_pow2(3 * n - 2)          # ≥ n + (2n‑1) – 1

    while time.time() < deadline:
        it += 1

        # ---- self‑convolution (linear) ----
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af * Af, n=L)
        conv = conv_full[: 2 * n - 1]

        # ---- smooth‑max weights ----
        max_conv = conv.max()
        exp_args = alpha * (conv - max_conv)          # ≤ 0, numerically safe
        w = np.exp(exp_args)
        sum_w = w.sum()
        if sum_w == 0 or np.isnan(sum_w):
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / sum_w

        # ---- gradient of smooth‑max ----
        # convolution a * (w_norm reversed)
        Wr = np.fft.rfft(w_norm[::-1], n=L)
        grad_full = np.fft.irfft(Af * Wr, n=L)
        start_idx = w_norm.size - 1
        grad = 2.0 * grad_full[start_idx : start_idx + n]   # exact gradient

        # ---- simple step‑size schedule ----
        lr = lr0 / (1.0 + it * 1e-4)
        a -= lr * grad
        a = _project_simplex(a)

        # ---- occasional monitoring ----
        if it % 50 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} "
                      f"α={alpha:.1f} score={cur_score:.6f}")

        # ---- continuation on α ----
        if it % 200 == 0 and alpha < alpha_end:
            alpha = min(alpha_end, alpha * 1.25)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepting only improvements.  Very cheap and works
    well as a final polish.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)               # more mass → higher chance to be moved
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30     # move up to 30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:               # strict improvement
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.5

        if score_smooth < best_score:

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, n={n}")
