def _hill_climb(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    it = 0
    # temperature schedule for simulated annealing
    T0 = 1e-3  # small
    while time.time() < deadline:
        it += 1
        # random source and destination
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue
        # amount to move
        amt = a[src] * (rng.random() * 0.01 + 1e-6)  # up to 1% of source
        if amt <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= amt
        a_prop[dst] += amt
        a_prop = _project_simplex(a_prop) # ensure sum=1, non-neg
        cur_score = _objective_score(a_prop)
        delta = cur_score - best_score
        if cur_score < best_score:
            a = a_prop
            best_seq = a_prop.copy()
            best_score = cur_score
        else:
            # accept with probability
            T = T0 / (1 + it * 0.01)  # decreasing
            if np.exp(-delta / max(T, 1e-12)) > rng.random():
                a = a_prop
        # optionally update best_seq if cur_score better but not incorporated yet
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_prop.copy()
    return best_seq, best_score

"""
Optimiser for the “step‑function” challenge.

Goal:
    minimise 2·n·max(conv(a,a)) / (sum(a)²)
    – the expression is scale‑invariant, therefore we keep the vector on the
      probability simplex (sum = 1, a_i ≥ 0).

Strategy (1000 s budget)

1. **Problem sizes** – try several lengths (256 … 8192).  The best
   construction among all lengths is kept.

2. **Initialisation** – uniform vector and a handful of Dirichlet draws
   with randomly chosen concentration (both dense and sparse starts).

3. **Smooth‑max descent** – minimise a differentiable approximation of the
   max–convolution:
        Sτ(a) = (1/τ)·log Σ exp(τ·conv)
   The gradient is 2·(w∗a_rev) where w are soft‑max weights.
   τ is increased geometrically, turning the smooth proxy into the true max.

4. **Hard‑max sub‑gradient descent** – after the smooth phase the exact
   sub‑gradient of max(conv) is used (average over all active shifts).
   A diminishing step size (η∝1/√t) together with projection onto the simplex
   drives the iterate further downhill.

5. **Stochastic hill‑climb / simulated annealing** – repeatedly move a tiny
   fraction of mass from the index with the largest positive sub‑gradient
   to one with the most negative (or a random index).  Moves that improve
   the objective are always accepted; otherwise they are accepted with a
   temperature‑controlled probability, allowing the optimiser to escape
   shallow local minima.

All convolutions are performed with FFTs (full linear convolution),
the simplex projection uses the O(n) algorithm of Duchi et al., and the
whole routine respects the 1000 s time limit, finally returning a plain
list of Python floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    return conv_full[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _smooth_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(τ·conv_k) stabilised by subtracting the max."""
    max_c = conv.max()
    w = np.exp(tau * (conv - max_c))
    s = w.sum()
    if s == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        s = 1.0
    w /= s
    return w


# ----------------------------------------------------------------------
# 1️⃣  Smooth‑max descent (soft proxy of max‑convolution)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    tau = 1.0               # start temperature (very smooth)
    tau_factor = 2.0        # geometric growth
    step0 = 0.08            # basic step size
    it = 0
    best_seq = a.copy()
    best_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        # convolution and current score
        conv = _conv_self(a, L_fft)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[smooth] it={it:,} score={cur_score:.6f}")

        # soft‑max weights and gradient
        w = _smooth_weights(conv, tau)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)           # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # slice of length n

        # diminishing step
        eta = step0 / math.sqrt(it)
        a -= eta * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # increase temperature occasionally
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Hard‑max sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # indices that attain the maximum (within a tiny tolerance)
        active = np.where(conv >= max_val - 1e-12)[0]

        # exact sub‑gradient of max(conv) at a
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        # diminishing step size
        eta = 0.08 / math.sqrt(it)
        a -= eta * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[subgrad] it={it:,} score={cur_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Stochastic hill‑climb / simulated annealing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    it = 0

    # temperature for occasional uphill moves
    T0 = 1e-3

    while time.time() < deadline:
        it += 1
        # source (mass donor) – pick a positive coordinate
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[rng.integers(src_candidates.size)]

        # destination (mass receiver)
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a tiny fraction of the source mass
        frac = rng.random() * 0.01 + 1e-6   # up to ~1 % of src
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        cur_score = _objective_score(a_prop)
        delta = cur_score - best_score

        # always accept improving moves
        if cur_score < best_score:
            a = a_prop
            best_seq = a_prop.copy()
            best_score = cur_score
            print(f"[hill] it={it:,} score={cur_score:.6f}")

        else:
            # occasional uphill moves (simulated annealing)
            T = T0 / (1.0 + it * 0.001)
            if np.exp(-delta / max(T, 1e-12)) > rng.random():
                a = a_prop

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Optimisation for a single problem size
# ----------------------------------------------------------------------
def _optimise_one_n(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Run a full optimisation pipeline for a fixed length n."""
    # baseline – uniform vector (score = 2)
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)

    rng = np.random.default_rng()
    # number of random restarts is limited by the remaining budget
    while time.time() < deadline - 2.0:   # keep a tiny safety margin
        # --------------------------------------------------------------
        #  a) random Dirichlet start (different concentration)
        # --------------------------------------------------------------
        conc = rng.uniform(0.1, 5.0)
        a0 = rng.gamma(shape=conc, scale=1.0, size=n)
        a0 /= a0.sum()

        # --------------------------------------------------------------
        #  b) smooth‑max descent
        # --------------------------------------------------------------
        t1 = time.time()
        phase_deadline = t1 + (deadline - t1) * 0.35
        a_s, sc_s = _smooth_max_descent(a0, phase_deadline)

        # --------------------------------------------------------------
        #  c) hard‑max sub‑gradient descent
        # --------------------------------------------------------------
        t2 = time.time()
        phase_deadline = t2 + (deadline - t2) * 0.35
        a_g, sc_g = _subgradient_descent(a_s, phase_deadline)

        # --------------------------------------------------------------
        #  d) final hill‑climb
        # --------------------------------------------------------------
        a_h, sc_h = _hill_climb(a_g, deadline)

        # keep the best among the three phases
        for seq, sc in [(a_s, sc_s), (a_g, sc_g), (a_h, sc_h)]:
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[n={n}] new best score = {sc:.6f}")

        # safety – break if the deadline is very close
        if time.time() > deadline - 1.0:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence``.  The routine follows a multi‑size, multi‑restart
    scheme and respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # a small safety margin

    # ------------------------------------------------------------------
    # try a handful of problem sizes
    # ------------------------------------------------------------------
    candidate_lengths = [256, 512, 1024, 2048, 4096, 8192]
    best_overall_seq = None
    best_overall_score = np.inf

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget to this length
        remaining = deadline - time.time()
        # give each length about the same share of the remaining time
        sub_deadline = time.time() + remaining * 0.18
        print(f"[{time.time() - start_time:.2f}s] Optimising length n={n}")
        seq_n, score_n = _optimise_one_n(n, sub_deadline)
        if score_n < best_overall_score:
            best_overall_score, best_overall_seq = score_n, seq_n.copy()
            print(f"[global] improved best score = {score_n:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 5️⃣  Try to improve the current best with a final hill‑climb
    # ------------------------------------------------------------------
    if best_overall_seq is not None and time.time() < deadline:
        seq_ref, sc_ref = _hill_climb(best_overall_seq, deadline)
        if sc_ref < best_overall_score:
            best_overall_score, best_overall_seq = sc_ref, seq_ref.copy()
            print(f"[final hill] further improved score = {sc_ref:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Fallback to a global starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_overall_score:
                    best_overall_score, best_overall_seq = sc, starter.copy()
                    print(f"[global starter] score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # extremely unlikely – fall back to a modest uniform vector
        best_overall_seq = np.full(1024, 1.0 / 1024)

    best_overall_seq = np.maximum(best_overall_seq, 0.0)
    total = best_overall_seq.sum()
    if total > 0.0:
        best_overall_seq = best_overall_seq / total
    else:
        best_overall_seq = np.full_like(best_overall_seq, 1.0 / best_overall_seq.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_overall_score:.6f}")
    return [float(x) for x in best_overall_seq]
