def proj_simplex(v):
    # Project onto simplex {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine n:
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 4096

    # Warm start shapes: use previous best plus candidates
    warm_candidates = _make_candidate_shapes(n)
    # include previous global best if available
    try:
        hv = np.asarray(height_sequence_1, float)
        warm_candidates.append(np.maximum(hv, 0))
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    # initial: random or best from warm-start
    for cand in warm_candidates:
        # normalize sum to 1
        if cand.sum() > 0:
            a = cand / cand.sum()
        else:
            continue
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_vec = a.copy()

    # Add random Dirichlet restarts if needed
    # We'll keep a current vector a_cur and best
    a_cur = best_vec.copy()
    conv_cur = _conv_fft(a_cur)
    cur_score = best_score
    print(f"[{time.time()-start:.2f}s] Initial best score: {best_score:.6f}")

    # Parameters for subgradient descent
    # We'll use step size schedule eta_t = eta0 / sqrt(t)
    eta0 = 0.5  # initial step size
    iteration = 0

    # Pre compute mask for active indices each iteration
    while time.time() < deadline:
        iteration += 1
        # compute conv and max
        conv_cur = _conv_fft(a_cur)
        max_conv = conv_cur.max()
        cur_score = 2.0 * n * max_conv  # sum=1 so denominator 1 in our normalization

        # if better record
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a_cur.copy()
            print(f"[{time.time()-start:.2f}s] iter={iteration} new best={best_score:.6f}")

        # Determine active indices where conv is near max (within epsilon)
        eps_active = 1e-12  # tolerance
        active = np.where(conv_cur >= max_conv - eps_active)[0]
        # pick a random active index to compute gradient
        k = np.random.choice(active)
        # compute gradient g_i = 2 * a[k - i] for i with 0 <= k-i < n
        g = np.zeros_like(a_cur)
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        # i runs from lo..hi
        # for i: a_i interacts with a_{k-i}
        # gradient at i = 2 * a_{k-i}
        # So g[i] = 2 * a_cur[k-i]
        # vectorized:
        i_idx = np.arange(lo, hi+1)
        g[i_idx] = 2.0 * a_cur[k - i_idx]

        # optional: if multiple active indices, average gradient
        if len(active) > 1:
            # compute average gradient over all active ks
            g_sum = np.zeros_like(a_cur)
            for kk in active:
                lo2 = max(0, kk - (n-1))
                hi2 = min(kk, n-1)
                i_idx2 = np.arange(lo2, hi2+1)
                g_sum[i_idx2] += 2.0 * a_cur[kk - i_idx2]
            g = g_sum / len(active)

        # step size
        eta = eta0 / np.sqrt(iteration)
        # gradient step towards decreasing objective (subtract)
        a_next = a_cur - eta * g
        # project onto simplex
        a_next = proj_simplex(a_next)

        # enforce small negative tolerance due to rounding
        # compute new objective
        new_score = _objective(a_next)
        # accept if improved else maybe accept if not too worse (simulated annealing)
        if new_score <= cur_score or np.random.rand() < 0.001:
            a_cur = a_next
        # otherwise keep a_cur unchanged (could also do a move but ignoring)

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0/n)

    print(f"[{time.time() - start:.2f}s] Finished – best score={best_score:.6f}")
    return list(map(float, best_vec))

def proj_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑objective non‑negative step function.

The algorithm consists of three stages:

1. **Rich warm‑start** – generate many analytic shapes (power‑tents,
   exponentials, power‑laws, Gaussians, etc.) together with a few random
   Dirichlet draws and the previous best vector (``height_sequence_1`` if
   available).  The best of these is taken as the initial candidate.

2. **Biased simulated‑annealing hill‑climb** – we keep the current vector
   ``a`` and its convolution ``conv`` (updated incrementally).  At each
   iteration we identify the index ``k`` where the convolution attains its
   maximum.  The contribution of each coefficient to this maximal term is
   ``c_i = a_i * a_{k-i}`` (zero when the index is out of range).  A source
   index is sampled proportionally to ``c`` (hence we move mass away from the
   indices that hurt the objective most) and a destination is chosen
   uniformly.  A random fraction of the source mass is transferred; the
   move is accepted if it lowers the objective or with a Metropolis
   probability that decays with a simple linear temperature schedule.
   The incremental update ``_apply_move`` runs in O(n) and allows millions
   of moves within the time budget.

3. **Projected‑gradient refinement** – after the annealing loop we perform
   a few iterations of exponentiated‑gradient descent on the simplex to
   polish the solution.

All vectors are kept normalised to sum = 1, which makes the objective
scale‑invariant and simplifies the computation:
    objective = 2·n·max(conv)   (since sum(a)=1).
The code respects the 1000 s budget and prints intermediate progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum≈1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


def _apply_move(a: np.ndarray, conv: np.ndarray,
                src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass ``eps`` from ``src`` to ``dst`` (in‑place).

    Parameters
    ----------
    a : ndarray (n,)
        Current coefficient vector.
    conv : ndarray (2n‑1,)
        Current self‑convolution (will be updated in‑place).
    src, dst : int
        Indices, src ≠ dst.
    eps : float
        Amount moved (0 ≤ eps ≤ a[src]).
    """
    n = a.shape[0]

    # linear part 2·a*Δ
    conv[src: src + n] -= 2.0 * eps * a
    conv[dst: dst + n] += 2.0 * eps * a

    # quadratic part Δ·Δ
    conv[2 * src] += eps * eps
    conv[2 * dst] += eps * eps
    conv[src + dst] -= 2.0 * eps * eps

    # update the vector
    a[src] -= eps
    a[dst] += eps


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of analytic (unnormalised) shapes."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]
    cand = []

    # 1) Power‑tent   (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 30):
        shape = np.minimum(x, 1.0 - x) ** p
        cand.append(shape)

    # 2) Exponential decay from centre
    for lam in np.logspace(-5, -0.4, 30):
        shape = np.exp(-lam * np.abs(x - 0.5))
        cand.append(shape)

    # 3) Power‑law edges (symmetric)
    for p in np.linspace(0.2, 2.0, 25):
        ix = np.arange(1, n + 1, dtype=float)
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        cand.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    cand.append(shape)

    # 5) Half‑cosine
    cand.append(np.sin(np.pi * x))

    # 6) Gaussians with different σ
    for sigma in (0.10, 0.15, 0.20, 0.25, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        cand.append(gauss)

    # 7) Random Dirichlet draws
    rng = np.random.default_rng(0)
    for _ in range(12):
        cand.append(rng.random(n))

    # normalise and keep only non‑negative vectors with positive mass
    out = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            out.append(c / s)
    return out

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a small evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the harness subtracts a small safety margin).

    Returns
    -------
    list[float]
        Non‑negative coefficient list ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin for the harness

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 4096

    # ------------------------------------------------------------------
    # 1) Warm‑start: gather many analytic candidates and random draws
    # ------------------------------------------------------------------
    warm = _make_candidate_shapes(n)

    # bring the previously discovered vector into the pool if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size == n:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0.0:
                warm.append(prev / s)
    except Exception:
        pass

    best_vec = None
    best_score = math.inf

    for a in warm:
        # All candidates are already normalised to sum = 1
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()

    # Fallback to uniform if everything failed
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 2) Biased annealing hill‑climb with incremental convolution updates
    # ------------------------------------------------------------------
    a_cur = best_vec.copy()
    conv_cur = _conv_fft(a_cur)
    total = a_cur.sum()                     # should be 1.0
    n_obj = 2.0 * n / (total * total)       # factor such that objective = n_obj * conv.max()
    cur_score = n_obj * conv_cur.max()
    best_vec = a_cur.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed + 12345)

    it = 0
    last_improve = 0
    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # Determine the critical convolution index and contribution weights
        # ------------------------------------------------------------------
        max_val = conv_cur.max()
        # pick the first index attaining the maximum
        k_max = int(np.flatnonzero(conv_cur == max_val)[0])

        # contribution of each coefficient to this maximal term:
        # c_i = a_i * a_{k_max-i}   (zero when the partner index is out of bounds)
        contrib = np.zeros_like(a_cur)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            contrib[idx] = a_cur[idx] * a_cur[k_max - idx]

        # If for numerical reasons the contribution sum is zero, fall back to plain mass
        weight_sum = contrib.sum()
        if weight_sum <= 0.0:
            src_probs = a_cur / a_cur.sum()
        else:
            src_probs = contrib / weight_sum

        src = int(rng.choice(n, p=src_probs))
        # destination – uniform (different from src)
        dst = int(rng.integers(n - 1))
        if dst >= src:
            dst += 1

        # --------------------------------------------------------------
        # Choose the amount of mass to move
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        remaining = deadline - time.time()
        # early iterations allow larger jumps, later ones shrink
        frac_max = 0.30 * remaining / (deadline - start_time) + 0.02
        eps = a_cur[src] * rng.random() * frac_max
        if eps <= 0.0 or a_cur[src] < 1e-15:
            continue

        # --------------------------------------------------------------
        # Apply the move and decide on acceptance (simulated annealing)
        # --------------------------------------------------------------
        before_score = cur_score
        _apply_move(a_cur, conv_cur, src, dst, eps)

        new_score = n_obj * conv_cur.max()
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            # linear temperature schedule: start at ~1 % of current score
            T0 = 0.01 * cur_score
            T = T0 * (remaining / (deadline - start_time))
            if T > 0.0:
                prob = math.exp(-(new_score - cur_score) / T)
                accept = rng.random() < prob

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a_cur.copy()
                last_improve = it
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # revert
            _apply_move(a_cur, conv_cur, dst, src, eps)

        # ------------------------------------------------------------------
        # Occasional global perturbation if stagnation is detected
        # ------------------------------------------------------------------
        if it - last_improve > 200_000:
            # mix a little random noise into the current best
            noise = np.random.dirichlet(np.ones(n))
            a_cur = 0.7 * best_vec + 0.3 * noise
            a_cur = proj_simplex(a_cur)
            conv_cur = _conv_fft(a_cur)
            cur_score = n_obj * conv_cur.max()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} perturbation applied")

        # ------------------------------------------------------------------
        # Periodic re‑normalisation to combat numeric drift
        # ------------------------------------------------------------------
        if it % 10_000 == 0:
            a_cur = proj_simplex(a_cur)
            conv_cur = _conv_fft(a_cur)
            cur_score = n_obj * conv_cur.max()

    # ------------------------------------------------------------------
    # 3) Projected‑gradient (exponentiated‑gradient) refinement
    # ------------------------------------------------------------------
    a_ref = best_vec.copy()
    for i in range(3000):
        conv = _conv_fft(a_ref)
        max_v = conv.max()
        ks = np.flatnonzero(conv >= max_v - 1e-12)
        grad = np.zeros_like(a_ref)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a_ref[k - idx]
        # multiplicative update (exponentiated gradient)
        eta = 0.1 / np.sqrt(i + 1)
        a_ref = a_ref * np.exp(-eta * grad)
        a_ref = proj_simplex(a_ref)

        new_score = _objective(a_ref)
        if new_score < best_score:
            best_score = new_score
            best_vec = a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] refine step {i} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
