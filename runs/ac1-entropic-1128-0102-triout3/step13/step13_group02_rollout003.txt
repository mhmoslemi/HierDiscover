"""
Propose a non‑negative step function (probability simplex vector) that
minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm differs from the reference solution in several ways:

  1. **Larger search space** – we use n = 8192 (instead of 4096) which gives
     more degrees of freedom to flatten the self‑convolution.

  2. **Rich parametric seed** – a dense sweep over symmetric “U‑shaped”
     families a_i ∝ ((i+1)*(n‑i))**α (α∈[−0.9, 2.5]) together with mixes with the
     uniform vector provides strong starting candidates.

  3. **Smooth‑max Adam** – similar to the reference, but with a slower
     temperature schedule (β up to 1000) and a slightly larger Adam step
     to explore the landscape more aggressively.

  4. **Multi‑active sub‑gradient descent** – instead of using only the
     single maximal shift, we collect the top‑K convolution entries (K≈5)
     and compute the exact sub‑gradient as the average of the corresponding
     shift vectors via a single FFT.  This follows the true sub‑gradient of
     the max‑convolution more faithfully.

  5. **Simulated‑annealing mass transfer** – random transfers of a
     (up‑to‑2 %) fraction of mass between two coordinates, accepted with a
     Metropolis probability that slowly cools.

  6. **Fine‑grained hill‑climb** – after the annealing phase we perform
     tiny deterministic improvements (≤0.1 % of a source entry) to polish
     the final score.

All phases respect the overall 1000 s budget, keep the vector on the
probability simplex (non‑negative, sum = 1) and finally return a plain
Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    if L is None:
        L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: n + m - 1]


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    return _conv_fft(a, a, L)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0, Σ w = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_simplex(a: np.ndarray, L_fft: int) -> float:
    """
    Fast objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max Adam descent
# ----------------------------------------------------------------------


def _smooth_adam(a0: np.ndarray, n: int, deadline: float,
                 start_time: float, best_score: float):
    """
    Adam projected descent on the soft‑max surrogate of max(conv).
    Uses a temperature β that is slowly increased up to 1000.
    Returns the best vector seen in this run and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    rng = np.random.default_rng()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0            # start temperature
    beta_mul = 1.015
    beta_max = 1000.0

    # Adam step scaling (a bit larger than the reference)
    base_lr = 0.3

    while time.time() < deadline:
        t += 1

        # ---- smooth‑max gradient -----------------------------------
        conv = _conv_self_fft(a, L_conv)          # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                    # for numerical stability
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()             # soft‑max weights

        # correlation w * a_rev  → gradient of soft‑max term
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)      # length 3n‑2
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        # exact score for monitoring (sum(a)≈1)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} "
                  f"score={cur_sc:.6f}")

        # ---- Adam update with simplex projection --------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # ---- temperature schedule ----------------------------------
        beta = min(beta * beta_mul, beta_max)

        # ---- occasional jitter ------------------------------------
        if t % 3000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase 2 – multi‑active exact sub‑gradient descent
# ----------------------------------------------------------------------


def _multi_active_subgrad(a0: np.ndarray, n: int, deadline: float,
                          start_time: float, best_score: float):
    """
    Sub‑gradient descent on the true max‑convolution.
    The sub‑gradient is computed as the average of the shift vectors
    corresponding to the K largest convolution entries (K≈5).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = best_score
    rng = np.random.default_rng()

    L_fft = _next_pow2(2 * n - 1)
    step0 = 0.14   # initial step size (will decay ∝ 1/√t)
    it = 0
    K = 5        # number of active shifts to aggregate

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()

        # ---- select up to K active shifts (largest entries) -----
        # Using argsort on the negative values gives descending order.
        top_inds = np.argpartition(-conv, K - 1)[:K]
        # keep them sorted for reproducibility (not essential)
        active_shifts = np.sort(top_inds)

        # ---- build indicator for the chosen shifts -------------
        ind = np.zeros(2 * n - 1, dtype=np.float64)
        ind[active_shifts] = 1.0 / len(active_shifts)   # average weight

        # ---- sub‑gradient = 2 * (a_rev * ind) ---------------
        a_rev = a[::-1]
        L_corr = _next_pow2(len(a_rev) + len(ind) - 1)
        g_full = _conv_fft(a_rev, ind, L_corr)   # length 3n‑2
        grad = 2.0 * g_full[n - 1:n - 1 + n]      # extract middle segment

        # ---- diminishing step ---------------------------------
        step = step0 / np.sqrt(it)
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        cur_sc = _score_simplex(a_candidate, L_fft)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_sc:.6f}")
        else:
            # occasional simulated‑annealing acceptance
            prob = np.exp(-(cur_sc - best_sc) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter to escape tiny plateaus
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_sc

# Phase 3 – simulated‑annealing mass transfer (fine polishing)
# ----------------------------------------------------------------------


def _annealing_hill_climb(a0: np.ndarray, deadline: float, start_time: float):
    """
    Random mass transfers with a slowly decreasing temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_sc = _score_simplex(a, L_fft)
    best_a = a.copy()
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 2 % of the source mass
        frac = rng.random() * 0.02
        eps = a[src] * frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_sc = _score_simplex(a_prop, L_fft)
        if new_sc < best_sc:
            a = a_prop
            best_sc = new_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} "
                  f"score={new_sc:.6f}")
        else:
            # temperature schedule (log‑linear cooling)
            T = max(0.05, 0.5 / np.log(it + 2))
            prob = np.exp(-(new_sc - best_sc) / T)
            if rng.random() < prob:
                a = a_prop

    return best_a, best_sc

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function that minimises the provided
    evaluation function.  The routine proceeds through four phases:

        0.  a dense parametric sweep over symmetric U‑shaped families;
        1.  several Adam‑style smooth‑max refinements;
        2.  multi‑active exact sub‑gradient descent;
        3.  simulated‑annealing mass‑transfer polishing.

    """

    start_time = time.time()
    # safety margin to avoid edge‑effects
    deadline = start_time + budget_s - 0.15

    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)

    best_score = _score_simplex(best_seq, L_fft)

    # ------------------------------------------------------------------
    # try to use a globally supplied starter if present
    # ------------------------------------------------------------------
    try:

            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                cur = _score_simplex(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 0️⃣  parametric sweep over symmetric U‑shaped families
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    # α controls how much mass is pushed to the edges (negative α → U‑shape)
    alpha_grid = np.concatenate((
        np.linspace(-0.9, -0.2, 12),
        np.linspace(-0.15, 0.2, 12),
        np.linspace(0.25, 2.5, 16)

    mix_grid = np.linspace(0.0, 1.0, 6)  # λ mixing with uniform

        if time.time() >= deadline - 250:

        for lam in mix_grid:
            cand = lam * uniform + (1.0 - lam) * shape
            cand /= cand.sum()
            sc = _score_simplex(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep α={α:.3f} λ={lam:.2f}"

    # ------------------------------------------------------------------
    time_for_adam = min(deadline - time.time() - 250, 420.0)
    n_restarts = 4

        rng = np.random.default_rng(seed + i + 123)
        a0 = best_seq.copy()
        # inject modest relative noise (≈2 %)

        a_smooth, sc_smooth = _smooth_adam(a0, n, sub_dead, start_time, best_score)
        if sc_smooth < best_score:

            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] restart {i+1}/{n_restarts}"
                  f" improvement score={sc_smooth:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  multi‑active exact sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:

        a_sub, sc_sub = _multi_active_subgrad(best_seq, n, sub_dead,
                                              start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] multi‑active subgrad"

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    best_seq = np.maximum(best_seq, 0.0)

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
