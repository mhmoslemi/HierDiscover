"""
Propose a low‑score step function.

The search works as follows

1.  Build a modest pool of analytic shapes (uniform, power ramps,
    skewed beta PDFs and simple exponentials).  All candidates are
    normalised to sum = 1 and the best few are kept.

2.  From the current best candidate we run a *deterministic* projected
    sub‑gradient descent on the *exact* max‑convolution.  At each
    iteration the full gradient of the active (maximal) convolution
    entry is computed analytically (a cheap shift of the vector) and
    the iterate is projected onto the probability simplex.

3.  After the gradient phase a light stochastic local search (tiny
    mass moves between random positions) is performed while time
    permits.

The whole routine respects the 1000 s budget, uses only a single
CPU and returns a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two (for FFT padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (fast, O(n log n))
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple objective (no safety checks) – uses the scale‑invariance
# ----------------------------------------------------------------------
def _objective_fast(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) (sum is 1 after normalisation)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx=1, x≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                 # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution on {0,…,n‑1}."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_ramp(n: int, p: float) -> np.ndarray:
    """Monotone power ramp a_i ∝ (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** p
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """
    Exponential‑type shape a_i ∝ exp(-lam·i/n).
    lam>0 controls the decay; lam≈1 gives a moderate skew.
    """
    i = np.arange(n, dtype=float)
    r = np.exp(-lam * i / n)
    total = r.sum()
    if total > 0.0:
        r /= total
    else:
        r = np.full(n, 1.0 / n)
    return r


# ----------------------------------------------------------------------
# Exact gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, kmax: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a for the active index kmax.
    Returns a vector of the same length as a.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    if kmax < n:
        # overlap is a[0:kmax+1] with reversed a[kmax::-1]
        grad[: kmax + 1] = 2.0 * a[kmax :: -1]
    else:
        t = kmax - (n - 1)                # how far we are past the centre
        # indices i = t … n‑1 overlap with j = kmax‑i = n‑1 … t
        idx_i = np.arange(t, n, dtype=int)
        idx_j = kmax - idx_i                # this runs descending from n‑1 to t
        grad[idx_i] = 2.0 * a[idx_j]
    return grad


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _descent_on_max(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    step0: float = 0.7,
                    max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Deterministic projected sub‑gradient descent using the exact active set.
    Returns the best iterate seen and its (fast) objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective_fast(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a)
        kmax = int(conv.argmax())
        cur_score = 2.0 * n * conv[kmax]          # sum(a)==1 always

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Descent iter={it:,}  score={cur_score:.6f}")

        # exact sub‑gradient for the currently maximal entry
        grad = _gradient_max_conv(a, kmax)

        # diminishing step‑size
        step = step0 / math.sqrt(it)
        a -= step * grad
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light random‑move local search (simulated annealing style)
# ----------------------------------------------------------------------
def _local_perturbations(a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_steps: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Perform tiny random mass transfers between positions.
    Accept improvements, otherwise keep the current state.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective_fast(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (step / max_steps)

        src = rng.choice(n, p=cur)                 # more likely to move from large mass
        dst = rng.integers(n)
        if src == dst or cur[src] < 1e-14:
            continue

        eps = cur[src] * rng.random() * 0.3        # move up to 30 % of the source mass
        prop = cur.copy()
        prop[src] -= eps
        prop[dst] += eps

        new_score = _objective_fast(prop)
        if new_score < cur_score or math.exp(-(new_score - cur_score) / T) > rng.random():
            cur = prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                print(f"[{time.time() - start_time:.2f}s] Perturb step={step:,}  best={new_score:.6f}")

    return best, best_score

        best_seq, best_score = _descent_on_max(best_seq,
                                               sub_deadline,
                                               start_time)
        print(f"[{time.time() - start_time:.2f}s] After descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light random‑move polishing (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _local_perturbations(best_seq,
                                                    deadline,
                                                    start_time)
        print(f"[{time.time() - start_time:.2f}s] Final polished score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – returning sequence of length {n}")
    return [float(v) for v in best_seq]
