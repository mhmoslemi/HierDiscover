def _adam_smooth_max_descent(a0, start_time, deadline, best_seq, best_score):

    a = a0.copy()
    n = a.shape[0]
    # precompute reversed a each iteration.
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    epsilon = 1e-8
    tau = 10.0  # start lower maybe fine.
    tau_factor = 1.5
    step0 = 0.1
    it=0
    while time.time()<deadline:
        it+=1
        conv = _conv_self(a) # L maybe next pow2 not needed else we use next_pow2 for speed
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"...")
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isinf(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum
        # compute gradient
        a_rev = a[::-1]
        # conv w with a_rev length = len(w)+len(a_rev)-1 = (2n-1) + n -1 = 3n-2
        grad_full = _conv_fft(w, a_rev)
        grad = 2.0 * grad_full[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step0 * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = a - step
        # projection
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new

"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)^2)

The objective is scale‑invariant, so we keep the vector on the probability
simplex (sum = 1).  The search consists of four time‑bounded stages:

1.  Broad parametric sweep – geometric, power and both symmetric and
    *asymmetric* Beta families (the latter were not explored previously).
2.  Random Dirichlet draws with a wide range of concentration parameters.
3.  Smooth‑max continuation using an Adam optimiser.  The temperature τ
    (smoothness) is increased geometrically, turning the smooth
    approximation of the max into the true max.
4.  A directed hill‑climb that repeatedly transfers a tiny amount of mass
    from the index with the most positive sub‑gradient to the index with the
    most negative, keeping only improving moves.

All convolutions are performed with FFT (full length).  The simplex
projection uses the O(n) algorithm of Duchi et al.  The routine respects a
1000 s budget; intermediate best scores are printed for monitoring.
The returned list is a plain Python list of floats ready for
``evaluate_sequence``.
"""
import time
import numpy as np

# --------------------------------------------------------------
# Linear‑algebra utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Objective helpers
# --------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

def _adam_smooth_max_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    tau = 10.0               # initial temperature
    tau_factor = 1.5
    it = 0

    # Pre‑allocate FFT length for gradient convolutions
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth soft‑max weights over the convolution entries
        max_c = conv.max()
        w = np.exp(tau * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2 * (w * rev(a)) restricted to the original window
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)           # length = 3n‑2 (full)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase temperature gradually – makes the approximation tighter
        if it % 3000 == 0:
            tau = min(tau * tau_factor, 1e9)

    return best_seq, best_score

def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue

        # transfer a tiny fraction of source mass
        frac = rng.random() * 0.005 + 1e-6   # up to 0.5 % of source
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill climb score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 5000:

                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score

    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth‑max descent")
        best_seq, best_score = _adam_smooth_max_descent(
            best_seq, smooth_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After Adam descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
