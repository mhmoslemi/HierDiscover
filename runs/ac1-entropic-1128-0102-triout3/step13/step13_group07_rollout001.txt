"""
Propose a step‑function (non‑negative vector summing to 1) that minimises
the evaluation function

    2·n·max(conv(a,a))   (i.e. the true objective of `evaluate_sequence`).

The algorithm consists of four phases

1. **Rich warm‑starts** – generate many analytic families (triangular,
   cosine, Gaussian, Beta‑distributions, arcsine, exponentials,
   plus an optional external seed vector).  The best of them with respect
   to the exact objective is kept.

2. **Smooth‑Adam** – minimise a smooth approximation of the max-
   convolution using Adam on the probability simplex.
   The smooth objective is the log‑sum‑exp of the convolution,
   which yields a dense sub‑gradient and lets the optimiser escape the
   local plateaus that the raw sub‑gradient suffers from.

3. **Exact‑Adam** – refined optimisation with the exact sub‑gradient
   (the same optimiser as in the reference solution).

4. **Greedy hill‑climb** – a cheap Metropolis walk that moves a random
   packet of mass between two coordinates and only accepts strict
   improvements.  This polish usually squeezes a few 10⁻⁴ off the final
   bound.

The whole pipeline respects a 1000 s wall‑clock budget; if the time runs
out the best solution seen so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns the valid part of length 2·n‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – a broad portfolio of analytic families
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse set of candidate vectors (each summed to 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]
    cand = []

    # 1) basic shapes already used in the reference implementation
    tent = np.minimum(x, 1.0 - x)
    cand.append(tent)

    cosine = np.sin(np.pi * x)                 # raised cosine
    cand.append(cosine)

    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    cand.append(circ)

    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    cand.append(gauss)

    # power‑tent
    for p in np.linspace(0.2, 2.0, 9):
        cand.append(np.minimum(x, 1.0 - x) ** p)

    # exponentials
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        cand.append(exp_fwd)
        cand.append(exp_fwd[::-1])

    # 2) Beta‑distribution family (covers a lot of shapes)
    alphas = [0.3, 0.5, 0.7, 1.0, 2.0, 4.0]
    betas  = [0.3, 0.5, 0.7, 1.0, 2.0, 4.0]
    for a in alphas:
        for b in betas:
            # pdf ∝ x^{a-1} (1-x)^{b-1}
            beta = (x ** (a - 1)) * ((1.0 - x) ** (b - 1))
            cand.append(beta)

    # 3) arcsine (a=b=0.5) – weight at the ends
    eps = 1e-12
    arcsine = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    cand.append(arcsine)

    # 4) a very flat shape – constant (baseline)
    cand.append(np.ones_like(x))

    # Normalise each candidate, drop degenerate ones.
    res = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            res.append(c / s)
    return res


# ----------------------------------------------------------------------
# Exact sub‑gradient (used by the reference Adam)
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    conv is the full linear self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad

# Adam optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Projected Adam using the exact sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        grad = _max_subgradient(a, conv)                     # sub‑gradient of max
        grad_obj = 2.0 * n * grad                              # gradient of the true objective

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExactAdam it={it:,} "
                  f"score={cur_score:.6f}")

        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score

# Smooth‑Adam (log‑sum‑exp surrogate for max)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.02,
                          lambda_start: float = 5.0,
                          lambda_max: float = 35.0,
                          lambda_multiplier: float = 1.05,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth approximation
          L(a) = (1/λ)·log Σ_i exp(λ·c_i/ max(c)),
    where c = a*a (linear convolution), max(c) is the current maximum.
    The surrogate tends to the true max as λ→∞.
    """
    a = a0.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)          # for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # for w*a convolution in the gradient

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    lam = lambda_start

    while time.time() < deadline:
        it += 1

        # 1) self‑convolution (full length 2·n‑1)
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

        max_val = conv.max()
        if max_val <= 0:
            break

        # 2) build soft weights w_i = exp(λ·(c_i / max - 1))
        conv_norm = conv / max_val                     # now ≤ 1, max = 1
        expo = np.exp(lam * (conv_norm - 1.0))         # max entry → 1, others < 1
        w = expo / expo.sum()                         # normalised weights, Σ w = 1

        # 3) gradient of the smooth objective:
        #    g = Σ_i w_i * ∂c_i/∂a  = 2·valid_convolution(w, a)
        A_fft = np.fft.rfft(a, n=L_grad)
        W_fft = np.fft.rfft(w, n=L_grad)
        conv_w_full = np.fft.irfft(W_fft * A_fft, n=L_grad)
        # valid part has length n and starts at index n‑1
        grad = 2.0 * conv_w_full[n - 1: n - 1 + n]

        # 4) Adam step on the smooth surrogate
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        a = _project_simplex(a)

        # monitor true objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothAdam it={it:,} "
                  f"λ={lam:.2f} score={cur_score:.6f}")

        # schedule λ upwards (more focus on the true max)
        if lam < lambda_max:
            lam = min(lambda_max, lam * lambda_multiplier)

        # occasional decay of the learning rate
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score

# Greedy hill‑climb polish (as in the reference solution)

def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """Greedy Metropolis walk that moves a random packet of mass."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                     # bias toward larger masses
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

                h1_resampled = np.maximum(h1_resampled, 0.0)
                s = h1_resampled.sum()
                if s > 0:
                    warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑Adam optimisation (exploratory phase)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                         # keep a few seconds for polishing
        smooth_deadline = time.time() + 0.45 * remaining    # ~45 % of remaining budget
        a_smooth, score_smooth = _smooth_adam_optimize(
            a0, smooth_deadline, start_time,
            lr=0.025,
            lambda_start=5.0,
            lambda_max=35.0,
            lambda_multiplier=1.07,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8
        )
        if score_smooth < best_score:
            best_score = score_smooth
            a0 = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] After SmoothAdam best score={score_smooth:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑Adam optimisation (fine‑tuning)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        exact_deadline = time.time() + 0.30 * remaining     # ~30 % of the rest
        a_exact, score_exact = _adam_optimize(a0, exact_deadline, start_time, lr=0.018)
        if score_exact < best_score:
            best_score = score_exact
            a0 = a_exact.copy()
            print(f"[{time.time() - start_time:.2f}s] After ExactAdam best score={score_exact:.6f}")

    # ------------------------------------------------------------------
    # 4) Light hill‑climb polish (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _hill_climb(a0, deadline, start_time)
        if score_final < best_score:
            best_score = score_final
            a0 = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={score_final:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
