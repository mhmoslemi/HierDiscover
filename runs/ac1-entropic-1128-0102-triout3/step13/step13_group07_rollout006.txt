def _adam_smoothed(a0, deadline, start_time, lr=0.02, beta_start=0.5, beta_factor=1.05, beta_max=200.0, eps=1e-8, beta_increase_iter=500):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3*n-2) # convolution length
    best_a = a.copy()
    best_score = _objective(a)  # using exact max
    # Adam variables:
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    beta = beta_start
    while time.time() < deadline:
        it += 1
        # compute conv = a*a
        Lc = _next_pow2(2*n-1)
        Af = np.fft.rfft(a, n=Lc)
        conv = np.fft.irfft(Af * Af, n=Lc)[:2*n-1]
        # compute softmax weights p
        max_conv = conv.max()
        # For numerical stable: compute exp(beta*(conv - max_conv))
        exp_beta = np.exp(beta * (conv - max_conv))
        sum_exp = exp_beta.sum()
        p = exp_beta / sum_exp
        # compute gradient via convolution of a and reversed p
        p_rev = p[::-1]
        # use FFT
        Af_full = np.fft.rfft(a, n=L)
        Pf = np.fft.rfft(p_rev, n=L)
        conv_rev = np.fft.irfft(Af_full * Pf, n=L)[:3*n-2]
        # grad_i = 2 * conv_rev[2*n-2 - i]
        idx = 2*n-2 - np.arange(n)
        grad = 2.0 * conv_rev[idx]
        # scale factor for objective: we minimize 2*n*max(conv) but gradient corresponds to derivative of max approx: derivative = 2*n * (approx). Actually the gradient we derived is of f(a) = sum_k p_k conv_k (the smoothed max). So derivative of objective = 2*n*grad. So we multiply by (2 * n).
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1-beta1) * grad_obj
        v = beta2 * v + (1-beta2) * (grad_obj ** 2)
        m_hat = m / (1-beta1**it)
        v_hat = v / (1-beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        # projection
        a = _project_simplex(a)

        # evaluate exact max objective
        cur_score = _objective(a)  # 2*n*max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)

        # increase beta gradually
        if it % beta_increase_iter == 0 and beta < beta_max:
            beta *= beta_factor

        # optional lr decay certain iterations
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score

def _hill_climb_sa(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # temperature schedule: start T = initial_temp (maybe 5e-4? scaled to score difference)
    init_temp = 1e-4 * cur_score  # maybe 0.00015.
    # as time passes, temperature decays exponentially to near 0
    total_time = deadline - start_time
    it = 0
    while time.time() < deadline:
        it+=1
        # pick two distinct indices
        src = rng.choice(n, p=a)  # weighted by mass
        dst = rng.integers(n)
        if src == dst:
            continue
        max_move = a[src] * 0.5
        eps = rng.random() * max_move
        a2 = a.copy()
        a2[src] -= eps
        a2[dst] += eps
        new_score = _objective(a2)
        delta = new_score - cur_score
        # acceptance
        elapsed = time.time() - start_time
        temp = init_temp * max(0.0, 1 - elapsed / total_time)  # linear decay
        accept = delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-12))
        if accept:
            a = a2
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] SA hillclim it={it:,} best={best_score:.6f}")
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)  # for reproducibility for any use.
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # choose n
    if budget_s >= 900:
        n = 8192
    else:
        n = 4096

    # Warm starts
    warm_starts = _make_warm_starts(n)
    # Add random warm starts
    rng = np.random.default_rng(seed+1)
    num_random = 20
    for _ in range(num_random):
        a_rand = rng.random(n)
        a_rand = np.maximum(a_rand, 0)
        a_rand_sum = a_rand.sum()
        if a_rand_sum > 0:
            warm_starts.append(a_rand / a_rand_sum)

    # Add global known height_sequence_1 if present
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            if np.all(h1_resampled >= 0):
                h1_resampled = np.maximum(h1_resampled, 0.0)
                s = h1_resampled.sum()
                if s > 0:
                    warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    # Evaluate warm starts and pick best
    best_score = np.inf
    best_a = None
    for a in warm_starts:
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm-start best score={best_score:.6f}")

    # Determine time allocation
    remaining = deadline - time.time()
    if remaining <= 0:
        # return warm start
        pass

    # Phase durations
    phase1_frac = 0.45
    phase2_frac = 0.35
    phase3_frac = 0.20
    # Ensure each atleast some seconds
    phase1_deadline = time.time() + max(10.0, remaining * phase1_frac)
    phase2_deadline = time.time() + max(10.0, remaining * phase2_frac)
    # The third phase uses the rest
    # Phase 1: smoothed Adam
    a_opt, score_opt = _adam_smoothed(a0, phase1_deadline, start_time,
                                      lr=0.02, beta_start=0.5, beta_factor=1.05,
                                      beta_max=200.0, beta_increase_iter=500)
    if score_opt < best_score:
        best_score = score_opt
        a0 = a_opt.copy()
        print(f"[{time.time() - start_time:.2f}s] Smoothed Adam best score={score_opt:.6f}")

    # Phase 2: exact Adam
    if time.time() < deadline:
        remaining2 = deadline - time.time()
        phase2_deadline_actual = time.time() + max(10.0, remaining2 * 0.7)
        a_opt, score_opt = _adam_optimize(a0, phase2_deadline_actual, start_time, lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Exact Adam best score={score_opt:.6f}")

    # Phase 3: hill-climb with simulated annealing
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb_sa(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] SA hill-climb best score={score_opt:.6f}")

    # Final shape normalisation
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0/n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]

"""
search for a non‑negative step function that minimizes
evaluate_sequence (i.e. 2·n·max(conv(a,a)) with sum(a)=1).

Approach
--------
1. Warm‑starts – many analytic shapes, a few random Dirichlet vectors
   and the optional global variable ``height_sequence_1`` – are normalised
   to the simplex and the best one (exact objective) is kept.

2. Smoothed Adam – minimise a smooth approximation of the max:
      f_β(a) = log‑sum‑exp(β·conv(a,a)) / β .
   The exact gradient of f_β is obtained by a convolution of the
   probability vector p_k ∝ exp(β·conv_k) with a, all through FFTs.
   β starts small (≈0.5) and is multiplied by a factor (≈1.05) every
   few hundred iterations until a prescribed maximum (≈200).  Adam with
   projection onto the probability simplex drives the optimisation.

3. Exact max‑Adam – after the smoothed phase we switch to the true
   sub‑gradient (only the entry attaining the maximum) and run another
   Adam optimisation to fine‑tune the solution.

4. Simulated‑annealing hill‑climb – a cheap Metropolis walk that moves
   a random packet of mass between two indices.  Moves that lower the
   exact objective are always kept; uphill moves are accepted with a
   temperature that linearly decays to zero.

The routine automatically respects the 1000 s budget – if time runs out,
the current best sequence is returned.  The final list is normalised to
sum = 1, contains only non‑negative floats and can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) constant (uniform)
    candidates.append(np.ones_like(x))

    # normalise each candidate, discard degenerate ones
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    conv is the full linear self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser – exact max‑subgradient
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Projected Adam on the exact sub‑gradient of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1

        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        grad = _max_subgradient(a, conv)
        grad_obj = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser – smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_smoothed(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   beta_start: float = 0.5,
                   beta_factor: float = 1.05,
                   beta_max: float = 200.0,
                   incr_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth approximation
        f_β(a) = log‑sum‑exp(β·conv) / β .
    ``beta_start`` is increased by ``beta_factor`` every ``incr_every``
    iterations up to ``beta_max``.
    """
    a = a0.copy()
    n = a.shape[0]
    Lc = _next_pow2(2 * n - 1)          # for self‑convolution
    Lx = _next_pow2(3 * n - 2)          # for gradient convolution

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = beta_start

    while time.time() < deadline:
        it += 1

        # ----- self‑convolution (for objective and soft‑max) -----
        Af_c = np.fft.rfft(a, n=Lc)
        conv = np.fft.irfft(Af_c * Af_c, n=Lc)[: 2 * n - 1]

        # ----- smooth max (softmax weights) -----
        max_conv = conv.max()
        exp_beta = np.exp(beta * (conv - max_conv))           # stable
        p = exp_beta / exp_beta.sum()                         # probabilities

        # ----- gradient of the smooth max -----
        p_rev = p[::-1]                                       # reversed
        Af_x = np.fft.rfft(a, n=Lx)
        Pf = np.fft.rfft(p_rev, n=Lx)
        conv_rev = np.fft.irfft(Af_x * Pf, n=Lx)[: 3 * n - 2]
        idx = 2 * n - 2 - np.arange(n)                        # indices for gradient
        grad = 2.0 * conv_rev[idx]                            # g_i = 2·∑_k p_k a_{k‑i}

        # scale to the true objective (2·n·max)
        grad_obj = 2.0 * n * grad

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        # ----- exact objective for bookkeeping -----
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothedAdam it={it:,} β={beta:.2f} score={cur_score:.6f}")

        # ----- schedule for β -----
        if it % incr_every == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # ----- (optional) learning‑rate decay -----
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb with simulated annealing
# ----------------------------------------------------------------------
def _hill_climb_sa(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """Greedy Metropolis walk with a linearly decaying temperature."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # initial temperature – a small fraction of the current score
    init_temp = cur_score * 1e-4
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        it += 1
        # pick a source index proportionally to its mass (encourages moves)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        elapsed = time.time() - start_time
        temp = max(0.0, init_temp * (1.0 - elapsed / total_time))

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-16)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] SA hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence found.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – choose a power of two that fits the time budget.
    # ------------------------------------------------------------------
    if budget_s >= 900:
        n = 8192
    else:
        n = 4096

    # ------------------------------------------------------------------
    # 1) Warm‑starts – analytic families + a few random Dirichlet vectors.
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # add a handful of random Dirichlet vectors
    for _ in range(20):
        rand_vec = rng.random(n)
        rand_vec = np.maximum(rand_vec, 0.0)
        s = rand_vec.sum()
        if s > 0:
            warm_starts.append(rand_vec / s)

    # optionally inject the globally supplied construction
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            if np.all(h1_resampled >= 0):
                h1_resampled = np.maximum(h1_resampled, 0.0)
                s = h1_resampled.sum()
                if s > 0:
                    warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    # pick the best warm‑start according to the exact objective
    best_score = np.inf
    best_a = None
    for a in warm_starts:
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed Adam phase (majority of the remaining time).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        phase1_end = time.time() + max(5.0, remaining * 0.45)
        a_opt, score_opt = _adam_smoothed(
            a0,
            phase1_end,
            start_time,
            lr=0.025,
            beta_start=0.5,
            beta_factor=1.05,
            beta_max=200.0,
            incr_every=400,
        )
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Smoothed‑Adam best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max Adam refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_end = time.time() + max(5.0, (deadline - time.time()) * 0.35)
        a_opt, score_opt = _adam_optimize(a0, phase2_end, start_time, lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Exact‑Adam best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Final stochastic hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb_sa(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] SA‑hill‑climb best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalise and return as plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
