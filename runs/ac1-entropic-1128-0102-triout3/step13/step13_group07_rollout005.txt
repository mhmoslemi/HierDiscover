"""
Propose a non‑negative step function (list of floats) that minimises
evaluate_sequence(...) .  The algorithm proceeds in four stages:

1. **Coarse warm‑starts** – generate a small library of analytic shapes
   (tent, cosine, circle, Gaussian, power‑tents, exponentials) at a
   modest resolution (n_coarse = 256) and keep the best one w.r.t.
   max(self‑convolution).

2. **Linear‑programming “exchange” phase** – starting from the coarse
   best shape, repeatedly solve a small LP:
       maximise   Σ g_j
       subject to f * g ≤ M   (component‑wise),  g ≥ 0
   where M = max(f * f).  The solution g is rescaled to Σ g = 1 and
   mixed with f (f ← (1‑t) f + t g) with an adaptive step size t.
   This is the exact implementation of the LP‑step described in the
   statement and quickly drives the maximal convolution down.

   The LP is solved with `scipy.optimize.linprog` on a sparse Toeplitz
   matrix; the whole phase lasts only a few seconds.

3. **Fine‑grained optimisation** – up‑sample the coarse vector to the
   target length (n = 4096), symmetrise it and normalise to sum = 1.
   Then run a projected Adam optimiser on the exact sub‑gradient of the
   max‑convolution (as in the previous state‑of‑the‑art code) for the
   bulk of the remaining time.

4. **Polish** – a cheap stochastic hill‑climb (with optional simulated
   annealing) is applied until the time budget expires.

All steps respect the 1000 s budget, keep the sequence non‑negative,
and return a plain list of floats ready for `evaluate_sequence`.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a (sum = 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a vector with sum = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_conv(a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families (any n)
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)  # x ∈ [0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) constant (uniform)
    candidates.append(np.ones_like(x))

    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Linear‑programming exchange step (coarse resolution)
# ----------------------------------------------------------------------
def _lp_exchange(f: np.ndarray, t: float, deadline: float,
                 start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform a single LP exchange step for the current vector f.
    Returns (new_f, new_M) where M = max(f*f).  If the LP is infeasible
    or time runs out, the original f and its M are returned.
    """
    n = f.shape[0]
    M = _max_conv(f)

    # Build sparse convolution matrix C such that (C @ g) == f * g
    rows = []
    cols = []
    data = []
    for i in range(2 * n - 1):
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        for k in range(lo, hi + 1):
            rows.append(i)
            cols.append(k)
            data.append(f[i - k])  # coefficient = f_{i-k}
    C = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M)
    c = -np.ones(n)                      # maximise sum(g)  ⇒ minimise -sum(g)

    # Solve the LP; use a tiny timeout via SciPy's timeout option if needed
    res = linprog(c, A_ub=C, b_ub=b, bounds=(0, None),
                  method='highs', options={'presolve': True})
    if not res.success:
        # fall back to unchanged vector
        return f, M

    g0 = res.x
    s = g0.sum()
    if s <= 0 or not np.isfinite(s):
        return f, M

    # Rescale to meet the normalisation Σg = 1
    g = g0 / s

    # Mix with current f
    f_new = (1.0 - t) * f + t * g
    M_new = _max_conv(f_new)
    return f_new, M_new


def _coarse_lp_phase(initial_f: np.ndarray,
                     deadline: float,
                     start_time: float) -> np.ndarray:
    """
    Repeatedly apply the LP exchange step at coarse resolution.
    An adaptive mixing weight t is used.  The routine stops when time
    runs out or when the mixing weight becomes vanishingly small.
    """
    f = initial_f.copy()
    best_f = f.copy()
    best_M = _max_conv(f)

    t = 0.5                      # start mixing aggressively
    it = 0
    while time.time() < deadline:
        it += 1
        f_new, M_new = _lp_exchange(f, t, deadline, start_time)
        if M_new < best_M - 1e-12:
            best_f = f_new.copy()
            best_M = M_new
            f = f_new
            t = min(0.5, t * 1.1)           # slightly enlarge step if we improve
            print(f"[{time.time() - start_time:.2f}s] LP it={it:,} M={M_new:.6f}")
        else:
            t *= 0.5                         # shrink step size when no improvement
            if t < 1e-4:                     # stop if step too tiny
                break
    return best_f


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    conv is the full linear self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.03,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of the max‑convolution.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        grad = _max_subgradient(a, conv)                 # sub‑gradient of max(conv)
        grad_obj = 2.0 * n * grad                         # gradient of the full objective

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        if it % 4000 == 0:
            lr *= 0.9               # gentle decay

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing polish (with optional SA)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy (with occasional simulated‑annealing acceptance) walk that moves
    a random packet of mass between two indices.  Only strictly improving
    moves are kept, but a tiny probability of uphill steps helps escape
    shallow traps.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                # weighted source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30      # move up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        improve = new_score < cur_score
        if not improve:
            # simulated‑annealing acceptance
            T = max(1e-4, (deadline - time.time()) / (deadline - start_time) * 0.001)
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                improve = True

        if improve:                              # accept
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (sum ≈ 1) that minimises
    `evaluate_sequence`.  The routine respects a 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Coarse warm‑starts (resolution 256)
    # ------------------------------------------------------------------
    n_coarse = 256
    warm = _make_warm_starts(n_coarse)

    # Optional global seed if it exists
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n_coarse)
            h1_res = np.interp(xq, xs, h1)
            h1_res = np.maximum(h1_res, 0.0)
            s = h1_res.sum()
            if s > 0:
                warm.append(h1_res / s)
    except Exception:
        pass

    # Add a few random Dirichlet seeds
    for _ in range(5):
        rand = np.random.default_rng().dirichlet(np.full(n_coarse, 0.5))
        warm.append(rand)

    best_f = None
    best_M = np.inf
    for cand in warm:
        M = _max_conv(cand)
        if M < best_M:
            best_M = M
            best_f = cand.copy()
    print(f"[{time.time() - start_time:.2f}s] Coarse warm‑start best M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 2) LP exchange phase (quick coarse optimisation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        lp_deadline = time.time() + min(120.0, remaining * 0.25)   # at most ~2 min
        best_f = _coarse_lp_phase(best_f, lp_deadline, start_time)
        best_M = _max_conv(best_f)
        print(f"[{time.time() - start_time:.2f}s] After LP phase M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to target resolution (4096) and symmetrise
    # ------------------------------------------------------------------
    n_target = 4096
    xs = np.linspace(0, 1, n_coarse)
    xq = np.linspace(0, 1, n_target)
    up = np.interp(xq, xs, best_f)
    up = np.maximum(up, 0.0)
    up = (up + up[::-1]) / 2.0               # enforce symmetry
    up = _project_simplex(up)                # keep on simplex

    # ------------------------------------------------------------------
    # 4) Projected Adam optimisation (most of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0:
        adam_deadline = time.time() + remaining * 0.70   # ≈70 % of what is left
        a_adam, score_adam = _adam_optimize(up, adam_deadline, start_time,
                                            lr=0.03)
        if score_adam < _objective(up):
            up = a_adam
            print(f"[{time.time() - start_time:.2f}s] Adam best score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # 5) Stochastic hill‑climb polish (use the rest of the time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, score_final = _hill_climb(up, deadline, start_time)
        up = a_final
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={score_final:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total > 0.0:
        up /= total
    else:
        up = np.full(n_target, 1.0 / n_target)

    final_obj = _objective(up)
    print(f"[{time.time() - start_time:.2f}s] Finished – objective={final_obj:.6f}, length={n_target}")

    return [float(v) for v in up]
