def _make_random_smooth_shape(n, rng, decay=1.0):
    # generate random Fourier coefficients with magnitude 1/k^decay for k>0
    # Use real FFT symm.
    # Equivalent: sample random real vector, apply low-pass filter: Fourier transform, zero out high frequencies beyond cutoff.
    # Simpler: start with random normal vector, apply Gaussian smoothing via convolution with kernel.
    # We'll use np.fft.rfft for speed.
    x = rng.normal(size=n)
    X = np.fft.rfft(x)
    # apply decay to high frequencies
    freqs = np.arange(len(X))
    decay_factor = 1.0 / (1.0 + freqs)**decay
    X *= decay_factor
    smooth = np.fft.irfft(X, n=n)
    shape = np.maximum(smooth, 0.0)
    # ensure sum >0
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0/n)
    return shape

"""
Optimises a non‑negative step function (a probability vector) for the
objective

    2·n·max(conv(a,a)) / (sum(a)**2)

using a multi‑start scheme, an enhanced projected sub‑gradient descent
with a dynamic active set and symmetry enforcement, followed by a
robust simulated‑annealing polish.  The routine respects the 1000 s
budget and returns a plain Python list of floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # should never happen
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric shape built from two power ramps joining a flat plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sine_power_shape(n: int, p: float) -> np.ndarray:
    """Shape ∝ sin(π·x)^p on [0,1] (zero at the ends, symmetric)."""
    xs = (np.arange(n) + 0.5) / n
    shape = np.sin(np.pi * xs) ** p
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Gaussian bump centred at 0.5, σ given as a fraction of the interval."""
    xs = (np.arange(n) + 0.5) / n
    centre = 0.5
    shape = np.exp(-0.5 * ((xs - centre) / sigma) ** 2)
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_random_smooth_shape(n: int, rng: np.random.Generator,
                              decay: float = 1.0) -> np.ndarray:
    """
    Random smooth shape generated by low‑pass filtering white noise.
    The decay parameter controls how quickly high‑frequency components die out.
    """
    # start from white noise
    x = rng.normal(size=n)
    # FFT
    X = np.fft.rfft(x)
    freqs = np.arange(X.shape[0])
    # dampen high frequencies
    damp = 1.0 / (1.0 + freqs) ** decay
    X *= damp
    smooth = np.fft.irfft(X, n=n)
    shape = np.maximum(smooth, 0.0)
    # enforce symmetry (helps the convolution structure)
    shape = (shape + shape[::-1]) / 2.0
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Enhanced projected sub‑gradient descent (dynamic active set + symmetry)
# ----------------------------------------------------------------------
def _subgradient_descent_enhanced(a0: np.ndarray,
                                  deadline: float,
                                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex with:
        * dynamic active‑set size,
        * a small tolerance to capture near‑max entries,
        * momentum (beta=0.9),
        * symmetric enforce after every projection.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 1.0
    it = 0

    # schedule parameters
    max_top_k = min(200, max(20, n // 256))   # never exceed a few hundred
    min_top_k = 5
    tol = 5e-3                               # start with 0.5 % tolerance

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # dynamic number of active constraints
        cur_top_k = max(min_top_k,
                        int(max_top_k * (1.0 - 0.6 * min(1.0, it / 2000.0))))
        # indices of the largest conv entries
        ks = np.argpartition(conv, -cur_top_k)[-cur_top_k:]

        # also gather entries within a small tolerance of the maximum
        near = np.where(conv >= max_conv * (1.0 - tol))[0]
        ks = np.unique(np.concatenate([ks, near]))

        # compute sub‑gradient (loop – cheap because |ks| stays modest)
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(ks.size)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a -= step * momentum

        # enforce symmetry (a_i = a_{n-1-i}) – reduces the search space
        a = (a + a[::-1]) / 2.0

        # project back onto the simplex
        a = _project_simplex(a)

        # keep best
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,}  score={cur_score:.6f}")

        # slowly tighten tolerance
        if it % 1000 == 0 and tol > 1e-6:
            tol *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (larger moves + symmetry)
# ----------------------------------------------------------------------
def _anneal_enhanced(seq: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis annealing that moves relatively large packets of mass,
    occasionally picks a source uniformly (not only proportional to its mass)
    and enforces symmetry after each move.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.05 * (1e-6 / 0.05) ** (it / max_steps)

        # pick source: 30 % uniform, 70 % mass‑proportional
        if rng.random() < 0.3:
            src = rng.integers(n)
        else:
            src = rng.choice(n, p=a)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 80 %) of the source mass
        eps = a[src] * rng.random() * 0.8
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # enforce symmetry and project
        a_prop = (a_prop + a_prop[::-1]) / 2.0
        a_prop = _project_simplex(a_prop)

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    The routine builds a pool of diverse analytic and random shapes,
    refines the best candidate with an enhanced projected sub‑gradient
    descent, finishes with a robust simulated‑annealing polish and
    returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose a power‑of‑two length (fast FFTs) – 2¹⁶ = 65536
    # ------------------------------------------------------------------
    n = 65536

    # ------------------------------------------------------------------
    # 2) Build an initial pool of promising candidates
    # ------------------------------------------------------------------
    pool = []            # list of (score, vector)
    pool_size = 12
    best_seq = None
    best_score = float("inf")

    def _maybe_add(vec: np.ndarray):
        nonlocal best_seq, best_score
        sc = _objective(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
        if len(pool) < pool_size:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # ---- power‑ramp family -------------------------------------------------
    p_grid = np.linspace(0.2, 3.0, 15)          # 0.2 … 3.0
    alpha_grid = np.linspace(0.0, 0.5, 11)      # 0 … 0.5
    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_power_shape(n, p, alpha)
            _maybe_add(a)
    print(f"[{time.time() - start_time:.2f}s] after power‑grid best={best_score:.6f}")

    # ---- beta‑distribution family -----------------------------------------
    a_grid = np.linspace(0.2, 5.0, 13)
    b_grid = np.linspace(0.2, 5.0, 13)
    for a_par in a_grid:
        for b_par in b_grid:
            a_vec = _make_beta_shape(n, a_par, b_par)
            _maybe_add(a_vec)
    print(f"[{time.time() - start_time:.2f}s] after beta‑grid best={best_score:.6f}")

    # ---- sine‑power family -------------------------------------------------
    p_grid2 = np.linspace(1.0, 6.0, 12)
    for p in p_grid2:
        a_vec = _make_sine_power_shape(n, p)
        _maybe_add(a_vec)
    print(f"[{time.time() - start_time:.2f}s] after sine‑power best={best_score:.6f}")

    # ---- Gaussian family ---------------------------------------------------
    sigma_grid = np.linspace(0.08, 0.30, 11)   # relative to the interval
    for sigma in sigma_grid:
        a_vec = _make_gaussian_shape(n, sigma)
        _maybe_add(a_vec)
    print(f"[{time.time() - start_time:.2f}s] after Gaussian best={best_score:.6f}")

    # ---- random smooth shapes (low‑pass filtered noise) -------------------
    for _ in range(80):
        if time.time() >= deadline:
            break
        a_vec = _make_random_smooth_shape(n, rng, decay=1.2)
        _maybe_add(a_vec)

    # ---- occasional random mixtures of two beta shapes --------------------
    for _ in range(80):
        if time.time() >= deadline:
            break
        a1, b1 = rng.uniform(0.2, 5.0, 2)
        a2, b2 = rng.uniform(0.2, 5.0, 2)
        w = rng.random()
        shape1 = _make_beta_shape(n, a1, b1)
        shape2 = _make_beta_shape(n, a2, b2)
        a_vec = w * shape1 + (1.0 - w) * shape2
        a_vec = a_vec / a_vec.sum()
        _maybe_add(a_vec)

    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) If a global starter exists (height_sequence_1) incorporate it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                _maybe_add(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter considered")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4) Local refinement – enhanced projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 60.0:                     # give sub‑gradient the bulk of time
        sub_deadline = time.time() + 0.85 * remaining
        refined_seq, refined_score = _subgradient_descent_enhanced(
            best_seq, sub_deadline, start_time)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal_enhanced(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improvement = {anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Normalisation & conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
