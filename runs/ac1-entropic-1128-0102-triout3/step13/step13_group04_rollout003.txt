"""
Propose a non‑negative step function (list of floats) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The optimizer combines several ideas that differ from the reference
implementation:

* a **coarse parametric sweep** over mixtures of a geometric decay and the
  uniform distribution (the same family as before but with a wider log‑grid);
* **many short Adam runs** from diverse random initialisations (Dirichlet or
  random geometric mixes).  Each run uses a projection onto the probability
  simplex and stops after a few thousand iterations, keeping the best local
  optimum found.
* a **long refinement** of the best sequence with the same Adam optimiser
  (single‑active sub‑gradient, as in the reference code) to push the score
  further.
* a **final stochastic hill‑climb** that moves a (larger) random fraction of
  mass between two positions and accepts only improving moves.  The fraction
  is gradually reduced, allowing the search to escape shallow local minima.

All steps respect the overall 1000 s time budget; the routine prints the
current best bound and returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σ w = 1 }.
    Returns w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient from a single active shift (as in the reference code)
# ----------------------------------------------------------------------
def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) obtained from a single active shift,
    the shift with maximal convolution value.
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser (single‑active sub‑gradient) – identical to the reference
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float,
                   max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser using the *single‑active* sub‑gradient.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        grad = _subgrad_one_active(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        a = _project_simplex(a)

        # occasional jitter if no improvement for a while
        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (allows larger transfers early on)
# ----------------------------------------------------------------------
def _hill_climb_general(start_a: np.ndarray,
                        start_time: float,
                        deadline: float,
                        max_frac: float = 0.05,
                        min_frac: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that moves a random fraction of mass between two positions.
    The fraction is drawn uniformly from [min_frac, max_frac].
    Only moves that improve the true objective are kept.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        frac = rng.random() * (max_frac - min_frac) + min_frac
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine mixes a parametric sweep, many short Adam runs
    from random starts, a long refinement, and a final hill‑climb.  The
    best sequence (as a plain Python list) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for finalisation
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # problem size (the same size used by the reference code)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optional global starter (if supplied by the environment)
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur = float(2.0 * n * _conv_self(starter, L_fft).max())
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fine parametric sweep (geometric + uniform)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    # broaden the ε‑grid a little compared to the reference
    eps_grid = np.logspace(-5, -2, 600)   # 1e‑5 … 1e‑2
    lam_grid = np.arange(0.0, 1.001, 0.01)    # 0 % … 100 % uniform

    for eps in eps_grid:
        # leave a small window for the later stages
        if time.time() >= deadline - 40:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo = geo / geo_sum
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq = seq / seq.sum()
            cur = float(2.0 * n * _conv_self(seq, L_fft).max())
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Grid eps={eps:.1e} λ={lam:.2f} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Multiple short Adam runs from diverse random starts
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    # allocate most of the remaining time to random restarts,
    # but keep ~30 s for a long refinement and final hill‑climb
    while time.time() < deadline - 30.0:
        # -------- generate a random seed ----------
        if rng.random() < 0.6:
            # random geometric+uniform mix
            eps = 10 ** rng.uniform(-5, -2)          # log‑uniform ε
            lam = rng.random()
            r = 1.0 - eps
            geo = np.power(r, idx)
            geo = geo / geo.sum()
            seq0 = lam * uniform + (1.0 - lam) * geo
        else:
            # Dirichlet distribution with random concentration
            alpha = rng.uniform(0.1, 1.0)
            seq0 = rng.dirichlet(np.full(n, alpha))
        seq0 = seq0 / seq0.sum()

        # ---- run a short Adam optimisation ----
        local_deadline = min(deadline - 15.0, time.time() + 4.0)  # ~4 s per restart
        seq_opt, sc_opt = _adam_optimize(
            seq0, start_time, local_deadline, best_score, max_iter=3000
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] New best from restart score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long refinement of the current best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        print(f"[{time.time() - start_time:.2f}s] Starting long refinement")
        # guess a reasonable number of iterations based on remaining time
        remaining_secs = deadline - time.time() - 10.0
        max_iter = int(remaining_secs * 2000)  # ~2000 it/s is typical
        seq_ref, sc_ref = _adam_optimize(
            best_seq, start_time, deadline - 10.0, best_score, max_iter=max_iter
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = seq_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] Refinement improvement score={sc_ref:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb (larger moves early, then finer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        # first phase: relatively aggressive transfers
        a_hc, sc_hc = _hill_climb_general(
            best_seq, start_time, deadline - 0.5, max_frac=0.05, min_frac=1e-5
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improvement score={sc_hc:.6f}")

        # second (optional) finer phase if time permits
        if time.time() < deadline - 0.3:
            a_hc2, sc_hc2 = _hill_climb_general(
                best_seq, start_time, deadline - 0.1, max_frac=0.01, min_frac=1e-6
            )
            if sc_hc2 < best_score:
                best_score = sc_hc2
                best_seq = a_hc2.copy()
                print(f"[{time.time() - start_time:.2f}s] Fine hill‑climb improvement score={sc_hc2:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
