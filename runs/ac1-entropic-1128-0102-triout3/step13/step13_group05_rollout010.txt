"""
A fast, gradient‑based optimiser for the step‑function bound.

The algorithm follows three phases:

1.  **Initial pool** – generate a small set of analytic shapes
    (uniform, power‑ramps, Beta‑densities) together with a few
    random Dirichlet vectors and the optional global starter.  The
    best few vectors (lowest objective) are retained.

2.  **Projected Adam on a sub‑gradient of the max‑convolution** –
    the objective 2·n·max(conv(a,a))/sum(a)² is scale‑invariant,
    therefore we keep a on the simplex (non‑negative, sum = 1).  In
    each iteration the index `k*` attaining the maximum of the
    linear self‑convolution is found, the exact sub‑gradient
    `∇max(conv) = 2·a shifted by k*` is computed, and an Adam step
    is taken.  After the step the vector is projected back onto the
    simplex and forced to be symmetric (mirrored about the centre);
    symmetry is known to be optimal for this problem.  The loop runs
    until the time budget is exhausted, constantly remembering the
    best vector seen.

3.  **Final normalisation & conversion** – the best vector is
    re‑normalised to sum 1, clipped to the allowed range and returned
    as a plain Python list.

The whole routine respects the 1000 s budget, never exceeds the
remaining time, and prints progressive diagnostics.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum(a) is assumed > 0)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = float(conv.max())
    s = float(seq.sum())
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # fallback – uniform vector
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (used for the initial pool)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric “ramp‑with‑plateau” shape, normalised to sum 1."""
    # clamp parameters to safe ranges
    p = max(0.0, p)
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # pad / trim to exact length n
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1.
    For α<1 the shape is U‑shaped.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Helper: keep the K best candidates seen so far
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Gradient descent on the exact max‑convolution (sub‑gradient + Adam)
# ----------------------------------------------------------------------
def _subgradient_adam(a0: np.ndarray,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the sub‑gradient of max(conv(a,a)).
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # initial statistics
    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.10            # base learning rate

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and objective
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index achieving the maximum (the active constraint)
        k = int(conv.argmax())

        # exact sub‑gradient of max(conv) : g_i = 2 * a_{k-i}  (zero if out of range)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[idx[mask]] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / math.sqrt(it)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = _project_simplex(a)

        # enforce symmetry (mirroring around the centre)
        a = (a + a[::-1]) * 0.5
        a = _project_simplex(a)          # re‑project – cheap, guarantees sum=1

    return best_a, best_score

            cand_score = _objective(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand
                print(f"[{time.time() - start_time:.2f}s] Polishing iter={it:,} "
                      f"score={cand_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
```
