        max_idx = np.where(conv == max_val)[0]
        # subgradient: sum over each max idx of 2*shift_right(a, idx)
        grad = np.zeros_like(a)
        for k in max_idx:
            # shift right by k: g_i += 2 * a_{k-i}
            idx = np.arange(a.shape[0])
            partner = k - idx
            mask = (partner >= 0) & (partner < a.shape[0])
            grad[mask] += 2 * a[partner[mask]]
        # step size schedule
        step = 0.01 / (1 + it*0.001)  # decreasing
        a = a - step * grad
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / a.shape[0])
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} best={cur_score:.6f}")
    return best_a, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The search proceeds in three stages:

1. **Analytic seeds** – a small pool of hand‑crafted shapes
   (uniform, power‑ramps, beta‑densities, sinusoidal bumps and a few
   random Dirichlet draws).  The best few candidates are kept.

2. **Targeted simulated‑annealing (SA)** – starting from the current best
   candidate we repeatedly move a small amount of mass.  The source index
   is chosen proportionally to its contribution to the *current* maximal
   convolution entry, i.e. we explicitly attack the offending peak.
   Acceptance follows a standard exponential cooling schedule.

3. **Smooth‑max (soft‑max) gradient descent** – a short Adam‑style descent
   on a temperature‑controlled soft‑max surrogate of the max‑convolution.
   The temperature is driven down to ~1e‑6, the step‑size decays, and after
   each update the vector is projected onto the probability simplex.
   This phase polish‑es the SA result.

All stages respect the total 1000 s budget.  The returned sequence is
normalised to sum 1, contains only non‑negative floats and can be of any
length (default 65536 when time permits).  The code is fully self‑contained
and only uses ``numpy`` and the already‑imported ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑invariant) objective
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau (normalised to sum 1)."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    return shape / total if total > 0 else np.full(n, 1.0 / n)


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1} (normalised to sum 1)."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    return shape / total if total > 0 else np.full(n, 1.0 / n)


def _make_sin_shape(n: int, phase: float = 0.0) -> np.ndarray:
    """A smooth sinusoidal bump (non‑negative, normalised)."""
    t = np.linspace(0, 2 * np.pi, n, endpoint=False) + phase
    shape = 0.5 * (1.0 + np.sin(t))
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    return shape / total if total > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Keep a top‑k pool
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the sorted top‑k pool."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Adam‑style smooth‑max descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Minimise a smooth surrogate of max(conv) using Adam."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.05          # base learning rate
    T0 = 0.02             # initial temperature
    T_end = 1e-6          # final temperature (much sharper than before)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature schedule (exponential decay)
        frac = it / (it + 5000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights over convolution entries
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of ∑ w_k (a * a)[k]  w.r.t a :
        #   g = 2 * (w * a_rev)  (convolution with reversed a)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]  # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # slice of length n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted simulated‑annealing
# ----------------------------------------------------------------------
def _targeted_sa(initial_seq: np.ndarray,
                 start_time: float,
                 deadline: float,
                 rng: np.random.Generator,
                 top_pool: list) -> tuple[np.ndarray, float]:
    """SA that moves mass away from the currently most offending convolution peak."""
    n = initial_seq.shape[0]
    cur_seq = initial_seq.copy()
    cur_score = _objective(cur_seq)
    conv_cur = _conv_self(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    max_move_factor = 0.2          # up to 20 % of source mass per accepted move
    T_start = 0.5
    T_end = 1e-8
    it = 0

    # pre‑compute a list of indices for quick access
    idx_arr = np.arange(n)

    while time.time() < deadline:
        it += 1

        # cooling schedule (exponential)
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * (T_end / T_start) ** frac

        # pick the index of the current maximal convolution entry
        max_idx = int(conv_cur.argmax())

        # contribution of each coordinate to that peak: a_i * a_{max_idx-i}
        partner = max_idx - idx_arr
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = cur_seq[idx_arr[mask]] * cur_seq[partner[mask]]
        tot_contrib = contrib.sum()

        if tot_contrib > 0.0:
            src = rng.choice(n, p=contrib / tot_contrib)
        else:
            # fallback: pick proportional to current mass
            src = rng.choice(n, p=cur_seq)

        # pick a destination – we prefer places where the partner term is tiny
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = cur_seq[src] * max_move_factor * rng.random()
        if delta <= 0.0:
            continue

        a_prop = cur_seq.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)

        # Metropolis acceptance on the *objective* (which is monotone with max)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_seq = a_prop
            cur_score = new_score
            conv_cur = _conv_self(cur_seq)

            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

        # occasional restart from another top candidate (helps escape traps)
        if it % 250000 == 0 and len(top_pool) > 1:
            choice = rng.integers(len(top_pool))
            cur_seq = top_pool[choice][1].copy()
            cur_score = _objective(cur_seq)
            conv_cur = _conv_self(cur_seq)

        # occasional tiny Gaussian jitter when temperature is already tiny
        if it % 100000 == 0:
            sigma = 0.003
            noisy = cur_seq + rng.normal(0.0, sigma, size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                noisy /= s
                cur_seq = noisy
                cur_score = _objective(cur_seq)
                conv_cur = _conv_self(cur_seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the harness

    # --------------------------------------------------------------
    # Choose problem size (larger N → finer discretisation)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build an initial pool of analytic candidates
    # --------------------------------------------------------------
    TOP_K = 6
    top = []   # list of (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(top, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    for p in np.linspace(0.2, 2.0, 9):
        for alpha in np.linspace(0.0, 0.5, 6):
            a = _make_shape(N, p, alpha)
            _push_top(top, _objective(a), a, TOP_K)

    # beta‑shapes (U‑shaped for α<1)
    for alpha in np.linspace(0.1, 0.9, 9):
        a = _make_beta_shape(N, alpha)
        _push_top(top, _objective(a), a, TOP_K)

    # sinusoidal bumps (different phases)
    for phase in np.linspace(0, np.pi, 5):
        a = _make_sin_shape(N, phase)
        _push_top(top, _objective(a), a, TOP_K)

    # random Dirichlet draws (α = 0.3 gives sparsity)
    for _ in range(30):
        if time.time() >= deadline:
            break
        a = rng.gamma(0.3, 1.0, size=N)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        _push_top(top, _objective(a), a, TOP_K)

    # incorporate the global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(top, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # best from the initial pool
    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Targeted simulated‑annealing (≈ 55 % of the total budget)
    # --------------------------------------------------------------
    sa_deadline = start_time + 0.55 * (deadline - start_time)
    best_seq, best_score = _targeted_sa(best_seq, start_time, sa_deadline, rng, top)

    # --------------------------------------------------------------
    # 3) Smooth‑max polishing (use the rest of the time)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        gd_deadline = deadline - 0.1   # leave a tiny buffer for finalisation
        refined_seq, refined_score = _smooth_gradient_descent(best_seq,
                                                             gd_deadline,
                                                             start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(f"[{time.time() - start_time:.2f}s] After smooth GD best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
