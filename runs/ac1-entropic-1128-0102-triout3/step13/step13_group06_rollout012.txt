def propose_candidate(seed: int=0, budget_s: int=1000, **kwargs):
    import numpy as np, math, time

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 4096

    # Warm-start shapes
    candidates = _make_candidate_shapes(n)

    # include existing height_sequence_1 if available
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size == n:
            hs = np.maximum(hs, 0)
            if hs.sum() > 0:
                candidates.append(hs / hs.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for cand in candidates:
        # compute convolution via FFT
        conv = _conv_fft(cand)
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
    # ensure sum=1
    a = best_vec.copy()
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0/n)
        total = 1.0
    else:
        a /= total
    conv = _conv_fft(a)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best_vec = a.copy()
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best={best_score:.6f}")

"""
Fast deterministic + targeted hill‑climbing search for a step function that
minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages

1.  Warm‑start – generate many analytic shapes (power‑tents, exponentials,
    power‑laws, Gaussians, random Dirichlet draws), normalise them and keep
    the best according to the exact objective.

2.  Targeted descent – repeatedly locate the index where the convolution is
    maximal, then move a fraction of mass from the index that contributes most
    to that peak to an index that contributes the least.  The move is accepted
    only if the current maximum does not increase, guaranteeing monotone
    improvement.  The transferred amount shrinks linearly with the remaining
    time, allowing coarse moves early and fine tuning later.

3.  Optional symmetrisation – after the time budget the best vector is made
    symmetric (a ↦ (a+a[::-1])/2) and, if this lowers the objective, it is kept.

The whole search respects the 1000 s budget and returns a list of non‑negative
floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns a length‑(2·n‑1) array.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _apply_move(a: np.ndarray,
                conv: np.ndarray,
                src: int,
                dst: int,
                eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Coefficient vector (modified in‑place).
    conv : np.ndarray, shape (2·n‑1,)
        Current linear convolution a*a (modified in‑place).
    src, dst : int
        Source and destination indices (src ≠ dst).
    eps : float
        Amount moved from src to dst (eps ≥ 0, eps ≤ a[src]).
    """
    n = a.shape[0]

    # linear part 2·a*Δ   (Δ = eps·(e_dst – e_src))
    conv[src: src + n] -= 2.0 * eps * a
    conv[dst: dst + n] += 2.0 * eps * a

    # quadratic part Δ·Δ
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # update the probability vector
    a[src] -= eps
    a[dst] += eps


def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """
    Generate a diverse collection of analytic (unnormalised) shapes.
    All returned vectors have length n, are non‑negative and will be
    normalised later.
    """
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Symmetric exponential decay from the centre
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Half‑cosine
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet‑like draws
    rng = np.random.default_rng(0)
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # normalise and discard the all‑zero vectors
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a tiny safety margin for the harness
    deadline = start_time + budget_s - 0.5
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 4096                              # default size

    # --------------------------------------------------------------
    # 1) Warm‑start – pick the best among many simple analytic shapes
    # --------------------------------------------------------------
    candidates = _make_candidate_shapes(n)

    # include the previously discovered vector if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                candidates.append(starter / starter.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for cand in candidates:
        conv = _conv_fft(cand)
        score = 2.0 * n * conv.max()   # exact objective (sum(cand) == 1)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()

    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # make sure the vector is a probability vector (sum == 1)
    a = best_vec.copy()
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    conv = _conv_fft(a)
    cur_max = float(conv.max())
    best_score = 2.0 * n * cur_max
    best_vec = a.copy()
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Targeted deterministic descent on the peak of the convolution
    # --------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # locate current maximal convolution entry
        k = int(np.argmax(conv))
        max_b = float(conv[k])

        # compute contributions a[i]·a[k‑i] for all i that appear in b_k
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        opp = k - idx
        contrib = a[idx] * a[opp]                      # shape (hi‑lo+1,)

        # source = index with largest contribution (reduces the peak most)
        src = int(idx[np.argmax(contrib)])

        # destination = index with smallest contribution (adds little to the peak)
        dst = int(idx[np.argmin(contrib)])

        # avoid degenerate move
        if src == dst or a[src] < 1e-15:
            # fall back to a random admissible source and destination
            nz = np.where(a > 1e-12)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz))
            dst = int(rng.integers(n))
            while dst == src:
                dst = int(rng.integers(n))

        # linearly decreasing maximum fraction of the source mass we are allowed to move
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        max_frac = 0.8 * (1.0 - frac) + 0.01          # from ~80 % early to ~1 % at the end
        eps = a[src] * max_frac * rng.random()
        if eps <= 0.0:
            continue

        # apply the move
        _apply_move(a, conv, src, dst, eps)

        new_max = float(conv.max())

        # accept only if the global maximum does not increase
        if new_max <= max_b + 1e-15:
            # move kept
            cur_max = new_max
            cur_score = 2.0 * n * new_max
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # revert the move
            _apply_move(a, conv, dst, src, eps)

    # --------------------------------------------------------------
    # 3) Optional symmetrisation (often improves the bound)
    # --------------------------------------------------------------
    sym = (best_vec + best_vec[::-1]) / 2.0
    sym_sum = sym.sum()
    if sym_sum > 0.0:
        sym /= sym_sum
        conv_sym = _conv_fft(sym)
        sym_score = 2.0 * n * conv_sym.max()
        if sym_score < best_score:
            best_score = sym_score
            best_vec = sym
            print(f"[{time.time() - start_time:.2f}s] symmetrised best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec = best_vec / s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
