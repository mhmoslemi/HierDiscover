"""
Propose a step‐function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The method combines several complementary phases that respect a 1000 s
budget:

1. **Baseline & starter** – uniform and any global starter are evaluated.
2. **Extended parametric sweep** – a rich family of monotone shapes
   (geometric·power, exponential·power, pure power, sinusoidal) is mixed
   with the uniform distribution; the best vector discovered is kept.
3. **Smooth‑max Adam descent** – a differentiable soft‑max surrogate of the
   max‑convolution is minimised.  The temperature β is raised up to 2000
   and each run performs many Adam updates while staying on the probability
   simplex (with Euclidean projection).  Several random seeds are used.
4. **Exact sub‑gradient descent** – the true max‑convolution entry is
   identified, its exact gradient (2·shifted a) is taken, and a diminishing
   step‑size update is performed on the simplex.  After each step the vector
   is symmetrised (a ← (a+a[::-1])/2) to exploit the inherent symmetry of the
   problem.
5. **Targeted hill–climb polish** – random mass‑transfer moves together with
   moves that explicitly lower the dominant convolution entry are applied.
   Only improving moves are kept (greedy).

All phases are interrupted as soon as the overall deadline (1000 s minus a
tiny safety margin) is reached.  The best vector found is normalised to
sum = 1 and returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective for a vector with sum≈1 (no safety clipping)."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max surrogate & Adam descent
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float, L_conv: int, L_corr: int):
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∂/∂a ( (1/β)·log Σ_k exp(β·conv_k) )
    """
    conv = _conv_self(a, L=L_conv)               # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                     # numerical stability
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()              # soft‑max weights, shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)       # linear convolution w * a_rev
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad, conv


def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type projected descent on the smooth‑max surrogate.
    β starts small (smooth landscape) and is slowly increased up to beta_max.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)
    beta = 5.0
    beta_mul = 1.02

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} score={cur_score:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = 0.5 / (1.0 + 1e-4 * t)       # slowly decaying LR
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional jitter for escaping plateaus
        if t % 3000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

# Exact sub‑gradient descent (max‑entry)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution term.
    Uses the exact gradient of the dominant entry and projects onto the
    simplex after each update.  The vector is kept symmetric (a ← (a+a[::-1])/2)
    which respects the problem’s inherent symmetry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        k_max = int(np.argmax(conv))

        # exact gradient of the max entry: grad_i = 2 * a_{k_max-i} if in range else 0
        grad = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]

        # diminishing step size (classical sub‑gradient schedule)
        eta = 0.5 / math.sqrt(t)
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # enforce symmetry (helps to equilibrate the convolution profile)
        a = (a + a[::-1]) / 2.0

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={t:,} score={cur_score:.6f}")

    return best_a, best_score

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)  # reset after success
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            # shrink step size when move rejected
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Extended parametric sweep
# ----------------------------------------------------------------------
def _extended_parametric_sweep(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore several monotone families:
        - r^i * (i+1)^p               (geometric·power)
        - exp(-λ i) * (i+1)^p         (exponential·power)
        - (i+1)^p                     (pure power)
        - sin(π (i+0.5)/n)^p          (sinusoidal power)
    Mixed with the uniform distribution via λ∈{0,0.2,0.5,0.8,1}.
    The best vector is retained.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)   # i+1
    L_fft = _next_pow2(2 * n - 1)

    # parameter grids
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=30)
    p_grid = np.linspace(-4.0, 2.0, 41)
    lam_grid = np.logspace(np.log10(1e-4), np.log10(0.5), 15)   # for exponential term

    for r in r_grid:
        if time.time() >= deadline - 100:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 100:
                break
            # 1) geometric·power
            shape1 = geo * np.power(idx, p)

            # 2) exponential·power
            for lam in lam_grid:
                shape2 = np.exp(-lam * np.arange(n, dtype=np.float64)) * np.power(idx, p)
                for mix in (0.0, 0.2, 0.5, 0.8, 1.0):
                    seq = mix * uniform + (1.0 - mix) * shape2
                    seq_sum = seq.sum()
                    if seq_sum == 0.0:
                        continue
                    seq /= seq_sum
                    sc = _eval_fast(seq, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print(f"[{time.time() - start_time:.2f}s] sweep exp r={r:.4f} p={p:.2f} λ={lam:.5f} mix={mix:.2f} score={sc:.6f}")

            # 3) pure power (geometric term omitted)
            shape3 = np.power(idx, p)
            for mix in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = mix * uniform + (1.0 - mix) * shape3
                seq_sum = seq.sum()
                if seq_sum == 0.0:
                    continue
                seq /= seq_sum
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep pow r={r:.4f} p={p:.2f} mix={mix:.2f} score={sc:.6f}")

            # 4) sinusoidal power (always positive)
            sin_vals = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
            # avoid zero by adding a tiny epsilon when p is negative
            eps = 1e-12
            shape4 = np.power(sin_vals + eps, p)
            for mix in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = mix * uniform + (1.0 - mix) * shape4
                seq_sum = seq.sum()
                if seq_sum == 0.0:
                    continue
                seq /= seq_sum
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep sin r={r:.4f} p={p:.2f} mix={mix:.2f} score={sc:.6f}")

            # finally blend geometric·power with uniform (original family)
            shape1 = geo * np.power(idx, p)
            for mix in (0.0, 0.2, 0.5, 0.8, 1.0):
                seq = mix * uniform + (1.0 - mix) * shape1
                seq_sum = seq.sum()
                if seq_sum == 0.0:
                    continue
                seq /= seq_sum
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep geo r={r:.4f} p={p:.2f} mix={mix:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises
    `2 * n * max(conv(a, a)) / (sum(a)**2)`.
    The routine respects the 1000 s budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # problem size – 4096 is a good compromise
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣ baseline & optional global starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}")

    # try to use a globally provided starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ extended parametric sweep (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200:
        best_seq, best_score = _extended_parametric_sweep(
            n, uniform, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 2️⃣ smooth‑max Adam descent runs (multiple seeds)
    # ------------------------------------------------------------------
    descent_seeds = [seed, seed + 1, seed + 2, seed + 10, seed + 11]
    for ds in descent_seeds:
        if time.time() >= deadline - 120:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        a0 += 1e-4 * rng.standard_normal(n) * a0   # tiny perturbation
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        # allocate a generous iteration budget but respect time
        max_iters = 30000
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,          # keep some slack for later phases
            start_time,
            best_score,
            beta_max=2000.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth descent improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ exact sub‑gradient descent (focus on the true max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60:
        max_iters = 25000
        a_opt, sc_opt = _subgradient_descent(
            best_seq,
            max_iters,
            deadline - 30,
            start_time,
            best_score,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ final targeted hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check with the official evaluator (optional)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
