"""
Propose a low‑score step function for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three phases

1.  A cheap random search on a modest size (n≈2¹⁴) using flexible analytic
    families (beta, power‑ramp, mixtures).  The best shape is kept.
2.  The shape is interpolated to a larger size (n=2¹⁶) and refined by a
    *max‑only* projected gradient descent.  The gradient of the
    currently maximal convolution entry is exact and cheap (O(n)).
3.  A short simulated‑annealing polish moves tiny packets of mass between
    positions, accepting worsening moves with a fast exponential cooling.
    The whole routine respects the 1000 s budget and returns a plain
    Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast linear self‑convolution (real FFT, padded to next power‑of‑2)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant, no safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # should never happen for a probability vector
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Wang & Carreira‑Perpiñán projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                               # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with a central plateau, normalised to sum = 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    # force even plateau length to keep symmetry exact
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # adjust length if rounding produced a short/long vector
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_mixed_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent beta PDFs."""
    a1, b1 = rng.uniform(0.2, 5.0, 2)
    a2, b2 = rng.uniform(0.2, 5.0, 2)
    w = rng.random()
    s1 = _make_beta_shape(n, a1, b1)
    s2 = _make_beta_shape(n, a2, b2)
    shape = w * s1 + (1.0 - w) * s2
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Gradient of the *max* convolution entry (exact, O(n))
# ----------------------------------------------------------------------
def _gradient_max_entry(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the convolution entry b_k = (a*a)[k] w.r.t. a.
    For non‑negative a this is 2·a shifted by k (with zero padding).
    """
    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Projected gradient descent that always follows the *current* max entry
# ----------------------------------------------------------------------
def _max_conv_descent(a0: np.ndarray,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Reduce max(conv(a,a)) by projected gradient descent.
    Uses the exact gradient of the maximal entry only.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # simple momentum – improves stability
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = _objective(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,}  score={cur_score:.6f}")

        k_max = int(conv.argmax())
        grad = _gradient_max_entry(a, k_max)

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same idea as in the reference code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Small‑scale Metropolis annealer that moves packets of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000  # will stop much earlier if time runs out

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # source chosen proportionally to its current mass – more likely to move
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 50 %) of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score

# Main entry point required by the judging system
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low objective value.
    The routine runs a quick random‑parameter sweep on a small grid,
    up‑scales the best candidate, refines it with a focused projected
    gradient descent and finishes with a short annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) cheap exploratory search on a modest size
    # ------------------------------------------------------------------
    n_small = 16384                         # 2¹¹⁴ – fast for many evaluations
    rng = np.random.default_rng(seed)

    best_small = np.full(n_small, 1.0 / n_small)
    best_score = _objective(best_small)

    # include a few deterministic seeds (uniform, simple ramps)
    candidates = []

    # uniform
    candidates.append(best_small.copy())

    # power‑ramp grid
    for p in np.linspace(0.1, 2.5, 9):
        for alpha in np.linspace(0.0, 0.5, 6):
            candidates.append(_make_power_shape(n_small, p, alpha))

    # beta grid (coarse)
    for a in np.linspace(0.3, 4.0, 7):
        for b in np.linspace(0.3, 4.0, 7):
            candidates.append(_make_beta_shape(n_small, a, b))

    # random mixtures – 200 samples
    for _ in range(200):
        candidates.append(_make_mixed_beta(n_small, rng))

    # random simple beta – 300 samples
    for _ in range(300):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        candidates.append(_make_beta_shape(n_small, a, b))

    # evaluate all candidates (fast because n_small is modest)
    for seq in candidates:
        if time.time() >= deadline:
            break
        sc = _objective(seq)
        if sc < best_score:
            best_score, best_small = sc, seq.copy()
            # occasional progress report
            print(f"[{time.time() - start_time:.2f}s] New prob‑grid best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) up‑scale to a larger length for finer optimisation
    # ------------------------------------------------------------------
    n_big = 65536                         # 2¹⁶ – the length used in the reference solution
    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_big = np.linspace(0, 1, n_big, endpoint=False)
    # linear interpolation preserves non‑negativity
    best_big = np.interp(x_big, x_small, best_small)
    # re‑normalise (tiny numerical drift may appear)
    best_big = np.maximum(best_big, 0.0)
    total = best_big.sum()
    if total > 0.0:
        best_big /= total
    else:
        best_big = np.full(n_big, 1.0 / n_big)

    print(f"[{time.time() - start_time:.2f}s] Upscaled candidate score ≈ {_objective(best_big):.6f}")

    # ------------------------------------------------------------------
    # 3) refined projected gradient descent (max‑entry only)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep enough time for polishing
        sub_deadline = time.time() + 0.75 * remaining
        refined_seq, refined_score = _max_conv_descent(best_big, sub_deadline, start_time)
        if refined_score < best_score:
            best_score, best_big = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Post‑GD score = {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final annealing polish (if any time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_big, deadline, start_time)
        if anneal_score < best_score:
            best_score, best_big = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improvement = {anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 5) final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    best_big = np.maximum(best_big, 0.0)
    total = best_big.sum()
    if total > 0.0:
        best_big /= total
    else:
        best_big = np.full(n_big, 1.0 / n_big)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(v) for v in best_big]
