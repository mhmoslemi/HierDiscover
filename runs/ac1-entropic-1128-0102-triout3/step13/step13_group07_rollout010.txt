def _smooth_descent(a0, beta0, deadline, start_time):

    n = a.shape[0]
    L = _next_pow2(3*n - 2)
    best_a = a.copy()
    best_score = _objective(a)
    eta0 = 0.5
    beta = beta0
    it = 0
    while time.time() < deadline:
        it += 1
        # compute FFT of a
        Af = np.fft.rfft(a, n=L)
        # convolution a*a
        conv = np.fft.irfft(Af * Af, n=L)[: 2*n - 1]
        maxc = conv.max()
        # softmax
        x = conv - maxc
        exp_x = np.exp(beta * x)
        p = exp_x / exp_x.sum()
        # gradient via correlation
        Pf = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(np.conj(Af) * Pf, n=L)[:n]
        grad = 2.0 * corr
        # step size schedule
        eta = eta0 / np.sqrt(it)
        # update a via gradient descent with projection
        a = a - eta * grad
        a = _project_simplex(a)
        # evaluate objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothDesc it={it:,} score={cur_score:.6f}")
        # increase beta gradually
        if it % 500 == 0:
            beta = min(beta * 1.2, 500.0)   # cap
            # optionally reduce eta0
        # optional break if too many iterations?
    return best_a, best_score

"""
Multi‑scale smooth descent + deterministic polishing.
---------------------------------------------------
1.  A small “coarse” problem (n≈2 k) is solved quickly with a
    projected Adam optimizer – this gives a good basin of attraction.
2.  The coarse solution is interpolated to a larger size (n≈8 k)
    which reduces discretisation error of the continuous optimum.
3.  On the fine grid a smoothed version of the max‑convolution
    (log‑sum‑exp) is minimised by simple projected gradient descent.
    The smoothing parameter β is grown slowly, turning the smooth
    objective into the true max.
4.  Finally a deterministic greedy hill‑climb moves mass from the
    highest‑gradient coordinate to the lowest‑gradient one, accepting
    only strictly improving moves.  This typically shaves the last
    few 10⁻⁴ off the objective.
The whole routine respects the 1000 s budget and returns a
non‑negative list that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families + a few random Dirichlet draws
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diversified collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)                # zero at the ends
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) uniform (flat) – useful as a baseline
    candidates.append(np.ones_like(x))

    # Normalise each candidate to sum = 1 and keep only non‑negative.
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of the max‑convolution.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # sub‑gradient for the (single) active max index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # scale to the true objective 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    beta0: float,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of max(conv) via projected gradient descent.
    The smoothing parameter β is increased gradually.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)          # enough for a * a and a ⋆ p
    best_a = a.copy()
    best_score = _objective(a)

    eta0 = 0.5                         # initial step size
    beta = beta0
    it = 0
    while time.time() < deadline:
        it += 1

        # FFT of the current iterate
        Af = np.fft.rfft(a, n=L)

        # self‑convolution a*a (full length 2n‑1)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # soft‑max weights over the convolution entries
        shifted = conv - maxc
        exp_shifted = np.exp(beta * shifted)
        p = exp_shifted / exp_shifted.sum()          # shape (2n‑1,)

        # gradient of the smooth surrogate: 2 * (a ⋆ p)
        Pf = np.fft.rfft(p, n=L)
        corr = np.fft.irfft(np.conj(Af) * Pf, n=L)[:n]   # correlation a⋆p, length n
        grad = 2.0 * corr                               # gradient of smooth max

        # simple diminishing step size
        eta = eta0 / math.sqrt(it)

        # gradient step followed by projection onto the simplex
        a = a - eta * grad
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} score={cur_score:.6f}")

        # slowly increase the smoothness parameter
        if it % 500 == 0:
            beta = min(beta * 1.2, 500.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _deterministic_hill_climb(a0: np.ndarray,
                              deadline: float,
                              start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always moves mass from the highest‑gradient
    coordinate to the lowest‑gradient one, accepting only improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    it = 0

    while time.time() < deadline:
        it += 1

        # current self‑convolution and the index attaining the maximum
        Lc = _next_pow2(2 * n - 1)
        conv = _conv_self(a, Lc)
        k = int(conv.argmax())

        # sub‑gradient of the active max entry
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        subgrad = np.zeros_like(a)
        subgrad[idx] = 2.0 * a[k - idx]

        # source = coordinate with largest sub‑gradient (most “expensive”)
        src = idx[np.argmax(subgrad[idx])]

        # target = a coordinate with smallest sub‑gradient (ideally zero)
        zero_mask = subgrad == 0.0
        if zero_mask.any():
            tgt = np.flatnonzero(zero_mask)[0]
        else:
            # all positions are active – pick the smallest one
            tgt = idx[np.argmin(subgrad[idx])]

        if src == tgt or a[src] < 1e-14:
            continue

        # try to move a fraction of the source mass; halve the fraction
        # until an improvement is observed or the fraction becomes tiny.
        moved = False
        frac = 0.5
        while frac > 1e-4:
            eps = a[src] * frac
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[tgt] += eps
            new_score = _objective(a_prop)
            if new_score < cur_score:                # greedy acceptance
                a = a_prop
                cur_score = new_score
                moved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={new_score:.6f}")
                break
            frac *= 0.5
        if not moved:
            # fall back to a tiny random move to escape possible plateaus
            # (still greedy – only keep if it improves)
            rng = np.random.default_rng()
            src2 = rng.choice(n, p=a)
            tgt2 = rng.integers(n)
            if src2 != tgt2 and a[src2] > 1e-14:
                eps = a[src2] * 0.05
                a_prop = a.copy()
                a_prop[src2] -= eps
                a_prop[tgt2] += eps
                new_score = _objective(a_prop)
                if new_score < cur_score:
                    a = a_prop
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                        print(f"[{time.time() - start_time:.2f}s] RandomHill it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines:
      * a quick coarse Adam optimisation,
      * up‑sampling,
      * a smoothed gradient descent on a larger grid,
      * a deterministic hill‑climb polish.
    Returns a plain list of floats suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the host runner
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # 1) Coarse problem (n≈2 k) – fast exploration
    # ------------------------------------------------------------------
    n_coarse = 2048
    warm = _make_warm_starts(n_coarse)

    # add a few random Dirichlet‑like starts
    for _ in range(3):
        rnd = np.random.rand(n_coarse)
        warm.append(rnd / rnd.sum())

    # optionally incorporate an externally supplied construction
    try:
        global height_sequence_1
        h_ext = np.asarray(height_sequence_1, dtype=float).ravel()
        if h_ext.size > 0:
            xs = np.linspace(0, 1, h_ext.size)
            xq = np.linspace(0, 1, n_coarse)
            h_res = np.interp(xq, xs, h_ext)
            if h_res.sum() > 0:
                warm.append(np.maximum(h_res, 0.0) / h_res.sum())
    except Exception:
        pass

    # keep the best warm start (exact objective, cheap)
    best_a = None
    best_score = np.inf
    for cand in warm:
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[{time.time() - start_time:.2f}s] Coarse warm‑start best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Coarse Adam optimisation (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    coarse_budget = min(remaining * 0.3, 180.0)      # cap to keep time for fine phase
    if coarse_budget > 5.0:
        coarse_deadline = time.time() + coarse_budget
        a_coarse, sc_coarse = _adam_optimize(best_a, coarse_deadline, start_time, lr=0.02)
        if sc_coarse < best_score:
            best_score = sc_coarse
            best_a = a_coarse.copy()
            print(f"[{time.time() - start_time:.2f}s] Coarse Adam best={sc_coarse:.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to a finer grid (n≈8 k) – improves discretisation
    # ------------------------------------------------------------------
    n_fine = 8192
    xs_c = np.linspace(0, 1, n_coarse)
    xs_f = np.linspace(0, 1, n_fine)
    a_fine = np.interp(xs_f, xs_c, best_a)
    a_fine = np.maximum(a_fine, 0.0)
    a_fine /= a_fine.sum()

    # ------------------------------------------------------------------
    # 4) Smoothed gradient descent on the fine grid (≈ 55 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    fine_budget = remaining * 0.55
    if fine_budget > 5.0:
        fine_deadline = time.time() + fine_budget
        a_fine, sc_fine = _smooth_descent(a_fine, beta0=1.0,
                                          deadline=fine_deadline,
                                          start_time=start_time)
        if sc_fine < best_score:
            best_score = sc_fine
            print(f"[{time.time() - start_time:.2f}s] Fine smooth best={sc_fine:.6f}")

    # ------------------------------------------------------------------
    # 5) Deterministic hill‑climb polish (spend the rest of the time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        hill_deadline = time.time() + remaining
        a_fine, sc_hill = _deterministic_hill_climb(a_fine, hill_deadline, start_time)
        if sc_hill < best_score:
            best_score = sc_hill
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best={sc_hill:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    a_fine = np.maximum(a_fine, 0.0)
    total = a_fine.sum()
    if total > 0.0:
        a_fine /= total
    else:
        a_fine = np.full(n_fine, 1.0 / n_fine)

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective={best_score:.6f}, length={n_fine}")
    return [float(v) for v in a_fine]
