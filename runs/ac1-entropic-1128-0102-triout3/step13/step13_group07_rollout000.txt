The new search routine combines three ideas that were not used together before:

1. **Warm‑starts from a rich family of analytic shapes** – the same set as in the reference code, plus a few extra exponentials.
2. **Projected Adam on a *smoothed* max‑convolution** – instead of the non‑differentiable `max`, we minimise  
   \[
      M_\alpha(a)=\frac1\alpha\log\Bigl(\sum_{k}\exp(\alpha\,(a*a)_k)\Bigr)
   \]  
   whose gradient is a weighted sum of the usual sub‑gradients.  A small α gives a smooth landscape, a large α approximates the true objective.  We increase α in stages (5 → 15 → 50 → 150) which lets the optimiser escape shallow local minima and then focus on the true max.
3. **Greedy hill‑climbing with occasional uphill moves** – a cheap Metropolis walk that accepts improving moves and also a tiny fraction of worsening moves (simulated‑annealing) to get out of plateaus.

All iterates are kept on the probability simplex (`Σa_i = 1, a_i ≥ 0`) via Euclidean projection.  
The code respects the 1000 s budget, automatically splits the time between the phases and prints progress.

```python
"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.
Algorithm
---------
1. Warm‑start from many analytic shapes (tent, cosine, circle, Gaussian,
   power‑tents, exponentials and the optional global variable
   ``height_sequence_1``).  The best shape w.r.t. the exact objective
   2·n·max(conv) is kept.

2. Projected Adam on the *smoothed* max‑convolution.
   The smoothed objective is
       M_α(a) = (1/α)·log( Σ exp(α·(a*a)_k) )
   whose gradient is
       ∇M_α = 4·n· ( w ⋆ a_rev )[n‑1 : n‑1+n],
   where w_k = exp(α·(a*a)_k) / Σ exp(α·(a*a)_k) (a soft‑max).
   α is increased in four stages (5, 15, 50, 150) so that the method
   first sees a smooth landscape and then refines the true max.

3. Light hill‑climbing polish with a Metropolis walk.
   Moves that improve the exact objective are always accepted;
   with probability `temp` (which decays) a worsening move is also
   accepted – a cheap simulated‑annealing variant.

All iterates are projected onto the probability simplex after each
update.  The routine stops automatically when the 1000 s budget is
exhausted and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns length 2·n‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Warm‑starts – analytic families
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (Σ=1, ≥0)."""
    x = np.arange(n, dtype=float) / (n - 1)  # x∈[0,1]
    cand = []

    # 1) linear tent
    cand.append(np.minimum(x, 1.0 - x))

    # 2) half‑cosine
    cand.append(np.sin(np.pi * x))

    # 3) half‑circle
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    cand.append(circ)

    # 4) Gaussian centred
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    cand.append(gauss)

    # 5) power‑tents
    for p in np.linspace(0.2, 2.0, 9):
        cand.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponentials (forward & reverse)
    for scale in (3.0, 5.0, 8.0):
        e = np.exp(-scale * x)
        cand.append(e)
        cand.append(e[::-1])

    # 7) constant (uniform)
    cand.append(np.ones_like(x))

    # Normalise and keep only non‑negative
    res = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            res.append(c / s)
    return res


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective & gradient
# ----------------------------------------------------------------------
def _smooth_obj_grad(a: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Compute the smoothed objective M_α(a) = (1/α)·log Σ exp(α·conv)
    multiplied by 2·n, together with its gradient w.r.t. a.
    """
    n = a.shape[0]
    # self‑convolution a*a (full length 2·n‑1)
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)                     # shape (2·n‑1,)

    # soft‑max weights w_k = exp(α·conv_k) / Σexp(α·conv)
    weighted = alpha * conv
    max_w = weighted.max()
    exp_shift = np.exp(weighted - max_w)        # avoid overflow
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp

    # smoothed objective (exact objective = 2·n·max(conv))
    M = (np.log(sum_exp) + max_w) / alpha
    obj = 2.0 * n * M

    # gradient: 4·n· Σ_k w_k·a_{k-i}  (i‑wise)
    a_rev = a[::-1]                             # reversed a
    L2 = _next_pow2(len(w) + n - 1)
    Wf = np.fft.rfft(w, n=L2)
    Af = np.fft.rfft(a_rev, n=L2)
    conv_wa = np.fft.irfft(Wf * Af, n=L2)[: len(w) + n - 1]

    # slice that corresponds to Σ_k w_k·a_{k-i}
    grad_vec = conv_wa[(n - 1):(n - 1) + n]     # length n
    grad = 4.0 * n * grad_vec

    return obj, grad


# ----------------------------------------------------------------------
# Projected Adam on the *exact* max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_exact(a0: np.ndarray,
                deadline: float,
                start_time: float,
                lr: float = 0.02,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of max(conv).
    Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # self‑convolution
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # index of current maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)

        # sub‑gradient of max(conv)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # g_i = 2·a_{k‑i}
        grad_obj = 2.0 * n * grad             # chain rule for full objective

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        # current exact objective
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamExact it={it:,} score={cur_score:.6f}")

        # decay learning rate occasionally
        if it % 4000 == 0:
            lr *= 0.9

    return best, best_score


# ----------------------------------------------------------------------
# Projected Adam on the *smoothed* max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 alphas: list[float],
                 iters_per_stage: int = 2000,
                 lr: float = 0.03,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed objective.  The list `alphas` defines the
    schedule (e.g. [5, 15, 50, 150]).  Each stage runs at most
    `iters_per_stage` iterations or until the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    total_it = 0
    for alpha in alphas:
        it = 0
        while it < iters_per_stage and time.time() < deadline:
            it += 1
            total_it += 1

            obj, grad = _smooth_obj_grad(a, alpha)

            # Adam step on the smoothed objective
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** total_it)
            v_hat = v / (1.0 - beta2 ** total_it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a -= step
            a = _project_simplex(a)

            # Evaluate the *exact* objective to monitor progress
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] AdamSmooth α={alpha:.0f} it={total_it:,} score={cur_score:.6f}")

            # optional decay of lr within a stage
            if it % 4000 == 0:
                lr *= 0.95

        if time.time() >= deadline:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing polish (with occasional uphill moves)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepts improvements always and accepts a worsening
    move with a small temperature that decays over time.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    # exponential cooling schedule for the uphill‑accept probability
    while time.time() < deadline:
        it += 1
        temp = max(0.0, 0.5 * (deadline - time.time()) / (deadline - start_time))

        # source proportional to current mass, destination uniform
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30    # move up to 30 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)

        improve = new_score < cur_score
        accept = improve or (rng.random() < temp and new_score <= cur_score * 1.02)

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={new_score:.6f}")

    return best, best_score

        if h1.size > 0:

            xq = np.linspace(0, 1, n)

            if np.all(h1_res >= 0):
                h1_res = np.maximum(h1_res, 0.0)
                s = h1_res.sum()
                if s > 0:

    except Exception:

    best_a = None
    best_score = np.inf
    for w in warm:
        sc = _objective(w)
        if sc < best_score:
            best_score = sc
            best_a = w.copy()
    a0 = best_a
    print(f"[{time.time() - start:.2f}s] Warm‑start best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Projected Adam on the exact max‑convolution (short phase)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()

        if sc_opt < best_score:

    # ------------------------------------------------------------------
    remaining = deadline - time.time()

        # α schedule – gradually focuses on the true max
        alpha_sched = [5.0, 15.0, 50.0, 150.0]
        a_opt, sc_opt = _adam_smooth(a0, smooth_dead, start,
                                      alphas=alpha_sched,

                                      lr=0.03)
        if sc_opt < best_score:
            best_score = sc_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start:.2f}s] After AdamSmooth best = {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Light hill‑climbing polish (use leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _hill_climb(a0, deadline, start)
        if sc_opt < best_score:

            a0 = a_opt.copy()
            print(f"[{time.time() - start:.2f}s] Final hill‑climb best = {sc_opt:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()

    else:

    print(f"[{time.time() - start:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
```
