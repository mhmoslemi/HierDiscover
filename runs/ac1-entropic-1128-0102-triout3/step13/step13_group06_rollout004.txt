"""
Propose a low‑objective step‑function by:

1.   Generating many analytic “warm‑start” shapes (tents, exponentials,
     power‑laws, Gaussians, random Dirichlet draws, the previous best etc.).
2.   Picking the best warm‑start according to the exact objective
     (FFT convolution, no approximations).
3.   Performing a simulated‑annealing hill‑climber that moves a small packet
     of mass from one index to another while updating the self‑convolution
     incrementally (O(n) per move).  Moves that increase the objective are
     occasionally accepted with a temperature that cools linearly to zero.
     Every 500 k moves the convolution is recomputed from scratch to
     suppress numerical drift.
4.   Periodically restarting from a fresh warm‑start to escape deep minima.
5.   Returning the best vector found, normalised to sum‑one.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a coefficient vector (sum≈1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


def _apply_move(a: np.ndarray, conv: np.ndarray,
                src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current coefficient vector (modified in‑place).
    conv : np.ndarray, shape (2n‑1,)
        Current convolution a*a (modified in‑place).
    src, dst : int
        Indices with src ≠ dst.
    eps : float
        Amount moved from src to dst (eps ≥ 0, eps ≤ a[src]).
    """
    n = a.shape[0]

    # linear term: 2*a*Δ    (Δ = eps*(e_dst – e_src))
    conv[src:src + n] -= 2.0 * eps * a
    conv[dst:dst + n] += 2.0 * eps * a

    # quadratic term Δ·Δ
    conv[2 * src] += eps * eps
    conv[2 * dst] += eps * eps
    conv[src + dst] -= 2.0 * eps * eps

    # update the coefficient vector
    a[src] -= eps
    a[dst] += eps


def _make_candidate_shapes(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a diverse set of analytic (unnormalised) shapes of length n.
    The caller should normalise each candidate to sum‑one before use.
    """
    x = np.arange(n, dtype=float) / (n - 1)  # scaled to [0,1]

    cand = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        cand.append(shape)

    # 2) Symmetric exponential decay from the centre
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        cand.append(shape)

    # 3) Symmetric power‑law from the edges
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        cand.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    cand.append(shape + shape[::-1])

    # 5) Half‑cosine (already used as warm‑start)
    cand.append(np.sin(np.pi * x))

    # 6) Gaussians with different σ
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        cand.append(gauss)

    # 7) Random Dirichlet‑like draws
    for _ in range(8):
        rnd = rng.random(n)
        cand.append(rnd)

    # Normalise and keep only those with positive total mass
    normalised = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    The algorithm uses a warm‑start from analytic shapes, then performs a
    simulated‑annealing hill‑climber with incremental convolution updates.
    It returns the best vector (list of non‑negative floats) discovered before
    the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # a small safety margin – the harness may add a further 0.5 s
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size (length of the vector)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore
    except Exception:
        n = 4096                     # default size
    # If we have plenty of time we can try a larger n for a second phase
    if deadline - start > 300 and n < 8192:
        n = max(n, 8192)

    # ------------------------------------------------------------------
    # Warm‑start: generate many analytic shapes and keep the best
    # ------------------------------------------------------------------
    warm = _make_candidate_shapes(n, rng)

    # include the previously found vector if it exists and matches length
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size == n:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0.0:
                warm.append(prev / prev.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for ws in warm:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    # safety fallback
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Initialise the search state
    # ------------------------------------------------------------------
    a = best_vec.copy()
    conv = _conv_fft(a)
    cur_obj = best_score
    total = a.sum()
    n_obj = 2.0 * n / (total * total)   # = 2*n because total≈1

    # Simulated‑annealing schedule
    total_time = deadline - start
    T0 = 0.07 * best_score               # initial temperature (≈7 % of the score)
    # keep a moving maximum of conv for fast reporting
    it = 0
    last_report = start

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        now = time.time()
        elapsed = now - start
        remaining = deadline - now

        # linear cooling
        T = T0 * max(0.0, remaining / total_time)

        # ------------------------------------------------------------------
        # Decide which type of move to perform
        # ------------------------------------------------------------------
        # 10 % of the time we bias the source towards indices that
        # contribute heavily to the current maximal convolution entry.
        if rng.random() < 0.10:
            # targeted move
            kmax = int(np.argmax(conv))
            lo = max(0, kmax - (n - 1))
            hi = min(n - 1, kmax)
            idxs = np.arange(lo, hi + 1, dtype=int)
            partners = kmax - idxs
            contrib = a[idxs] * a[partners]
            sum_contrib = contrib.sum()
            if sum_contrib > 0:
                src = rng.choice(idxs, p=contrib / sum_contrib)
            else:
                src = rng.choice(n, p=a)
        else:
            # plain source proportional to mass
            src = rng.choice(n, p=a)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # Step size: early in the run we allow larger packets,
        # later we shrink them.
        frac = 0.30 * (remaining / total_time) + 0.02
        eps = a[src] * rng.random() * frac
        if eps <= 0.0:
            continue

        # ------------------------------------------------------------------
        # Apply the move and evaluate the new objective
        # ------------------------------------------------------------------
        before_obj = cur_obj
        _apply_move(a, conv, src, dst, eps)
        new_max = conv.max()
        new_obj = n_obj * new_max
        delta = new_obj - before_obj

        # Acceptance rule (simulated annealing)
        if delta <= 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            cur_obj = new_obj
            # record best ever seen
            if new_obj < best_score:
                best_score = new_obj
                best_vec = a.copy()
                print(f"[{now - start:.2f}s] iter={it:,}  new best={best_score:.6f}")
        else:
            # reject – revert the move
            _apply_move(a, conv, dst, src, eps)

        # ------------------------------------------------------------------
        # Periodic maintenance
        # ------------------------------------------------------------------
        # recompute convolution from scratch every 500 k moves to avoid drift
        if it % 500_000 == 0:
            conv = _conv_fft(a)
            cur_obj = n_obj * conv.max()

        # occasional restart from a fresh warm‑start (once per ~1 M moves)
        if it % 1_000_000 == 0 and remaining > 30.0:
            cand = rng.choice(warm)
            a[:] = cand
            conv = _conv_fft(a)
            cur_obj = n_obj * conv.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_vec = a.copy()
                print(f"[{time.time() - start:.2f}s] restart → new best={best_score:.6f}")

        # light progress report (every ~2 seconds)
        if now - last_report > 2.0:
            print(f"[{now - start:.2f}s] iter={it:,}  cur={cur_obj:.6f}  best={best_score:.6f}")
            last_report = now

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec = best_vec / s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
