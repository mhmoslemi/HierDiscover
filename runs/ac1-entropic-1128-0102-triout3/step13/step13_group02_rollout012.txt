"""
Improved search for a step function minimizing

    2 * n * max(conv(a,a)) / (sum(a)**2)

The strategy combines:

1️⃣  A richer parametric sweep (geometric‑power, symmetric beta‑like,
    sine‑based families) to obtain a strong initial candidate.

2️⃣  A linear‑programming refinement: for the current vector f compute M=max(f*f)
    and solve  max Σg_i  subject to f*g ≤ M, g≥0.
    If Σg_i>1 we can increase the total mass without raising the bound,
    normalise the solution and keep the improvement (Frank‑Wolfe style).

3️⃣  A smooth‑max Adam descent (soft‑max surrogate of max(conv)) with a
    temperature schedule, starting from the LP‑improved vector.

4️⃣  Exact sub‑gradient descent on the true max‑convolution.

5️⃣  A final microscopic hill‑climb that transfers tiny amounts of mass
    between random coordinates.

All phases respect the 1000 s budget, keep the vector on the probability
simplex and finally return the best sequence found as a plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w≥0, sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑2: smooth‑max Adam descent
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray, n: int, deadline: float,
                    start_time: float, best_score_initial: float):
    """Adam‑style projected descent on the soft‑max surrogate of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0                # start temperature
    beta_mul = 1.02
    beta_max = 500.0

    while time.time() < deadline:
        t += 1

        # ---- soft‑max gradient -----------------------------------------
        conv = _conv_self_fft(a, L_conv)            # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                      # stabilise exponentials
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()               # soft‑max weights

        # correlation of w with reversed a → gradient of soft‑max
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)        # length ≥ 3n‑2
        start = n - 1
        grad = 2.0 * corr[start:start + n]

        cur_score = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={t:,} "
                  f"β={beta:.1f} score={cur_score:.6f}")

        # ---- Adam update on the simplex --------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # occasional jitter
        if t % 4000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3: sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """Projected sub‑gradient descent directly on the true max‑convolution."""
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.12   # initial step magnitude
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for shift k (O(n))
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            prob = np.exp(-(cur_score - best_score) * 15.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4: final microscopic hill‑climb
# ----------------------------------------------------------------------


def _final_hill_climb(a0: np.ndarray, deadline: float, start_time: float):
    """Very fine local search that moves ε‑mass between random coordinates."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 1e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, M: float, time_left: float):
    """Return a better vector using the LP  max Σg_i  s.t. f*g ≤ M."""
    n = f.shape[0]
    # Build the (2n‑1)×n convolution matrix with rows = shifts, cols = g‑index
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)                     # each column repeats f
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)

    c = -np.ones(n)                          # maximise Σg_i  ⇔ minimise −Σg_i
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast for sparse problems)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'time_limit': time_left})
    if not res.success:
        print("LP failed:", res.message)
        return None, None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 1.0001:
        return None, None
    g_norm = g / sum_g
    return g_norm, sum_g

        lp_iter += 1

    # ------------------------------------------------------------------
    # 3️⃣  smooth‑max Adam descent (warm‑starting from LP result)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        smooth_deadline = time.time() + min(300.0, deadline - time.time() - 150.0)
        best_seq, best_score = _smooth_descent(best_seq.copy(), n, smooth_deadline,
                                               start_time, best_score)

    # ------------------------------------------------------------------
    # 4️⃣  sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 100:
        sub_deadline = deadline - 100.0
        best_seq, best_score = _subgradient_descent(best_seq.copy(), n,
                                                    sub_deadline, start_time,
                                                    best_score)

    # ------------------------------------------------------------------
    # 5️⃣  final hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq.copy(), deadline, start_time)
        if sc_hc < best_score:
            best_seq = a_hc.copy()
            best_score = sc_hc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and output a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
