"""
Propose a non‑negative step function (probability distribution) that
minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  **Large parametric sweep** – a family
        a_i ∝ r^i·(i+1)^p   (mixed with the uniform vector)
    is explored on a fine grid (r, p, λ).  The best parameters give a very
    good seed for the later optimisation.

2️⃣  **Long Adam‑style projected descent on a soft‑max surrogate** of the
    “max‑convolution’’ term.  β (the soft‑max temperature) starts small and is
    increased up to 2000, turning the surrogate into the true max.  The
    iterate is always projected onto the probability simplex.

3️⃣  **Targeted stochastic hill‑climb** – random mass transfers together
    with moves that explicitly reduce the currently dominant convolution
    entry polish the solution after the gradient phase.

All stages respect the 1000 s time limit, constantly checking the deadline.
The final vector is normalised to sum 1 and returned as a plain list of
Python floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, simplex projection, fast eval)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation from three parameters (r, p, λ)
# ----------------------------------------------------------------------
def _make_sequence(r: float, p: float, lam: float,
                   uniform: np.ndarray, n: int) -> np.ndarray:
    """
    Build a candidate distribution:
        a = λ·uniform + (1‑λ)·shape,
    where shape_i ∝ r**i · (i+1)**p.
    The result is renormalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    # log‑space computation prevents under‑/overflow for extreme r
    log_r = math.log(r) if r > 0 else -np.inf
    shape = np.exp(i * log_r) * np.power(i + 1.0, p)
    s = shape.sum()
    if s == 0.0:
        shape = uniform.copy()
    else:
        shape /= s
    seq = lam * uniform + (1.0 - lam) * shape
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        seq = uniform.copy()
    else:
        seq /= seq_sum
    return seq


# ----------------------------------------------------------------------
# Large parametric sweep (fine grid)
# ----------------------------------------------------------------------
def _parametric_grid_sweep(n: int, uniform: np.ndarray,
                          deadline: float, start_time: float,
                          best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Explore a fine (r, p, λ) grid and keep the best vector."""
    # grid specifications – a bit finer than the previous implementation
    r_grid = np.logspace(math.log10(0.90), math.log10(0.99999), num=80)
    p_grid = np.linspace(-5.0, 5.0, 81)      # step 0.125
    lam_grid = np.array([0.0, 0.25, 0.5, 0.75, 1.0])

    L_fft = _next_pow2(2 * n - 1)

    for r in r_grid:
        if time.time() > deadline - 120:
            break
        for p in p_grid:
            if time.time() > deadline - 120:
                break
            for lam in lam_grid:
                seq = _make_sequence(r, p, lam, uniform, n)
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"grid r={r:.5f} p={p:.3f} λ={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Random parameter search (exploratory)
# ----------------------------------------------------------------------
def _random_param_search(n: int, uniform: np.ndarray,
                         deadline: float, start_time: float,
                         best_seq: np.ndarray, best_score: float,
                         rng: np.random.Generator,
                         max_trials: int = 3000) -> tuple[np.ndarray, float]:
    """Sample random (r, p, λ) triples – accept only improvements."""
    L_fft = _next_pow2(2 * n - 1)
    for _ in range(max_trials):
        if time.time() > deadline - 120:
            break
        # log‑uniform sampling for r, wide uniform for p, λ∈[0,1]
        r = np.exp(rng.uniform(np.log(0.90), np.log(0.99999)))
        p = rng.uniform(-5.0, 5.0)
        lam = rng.random()
        seq = _make_sequence(r, p, lam, uniform, n)
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] rand r={r:.5f} p={p:.3f} λ={lam:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float,
                     L_conv: int, L_corr: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∂/∂a ( (1/β)·log Σ_k exp(β·conv_k) )
    Gradient is computed via FFT‑based correlation.
    """
    conv = _conv_self(a, L=L_conv)               # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (stable)
    shifted = conv - max_c                       # max becomes 0
    exp_vals = np.exp(beta * shifted)            # may under‑flow → 0
    w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)         # linear convolution w * a_rev

    start = a.shape[0] - 1                       # index of a_rev start inside corr
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray, max_iters: int,
                    deadline: float, start_time: float,
                    best_score_initial: float,
                    beta_max: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the smooth‑max surrogate.
    β is increased geometrically up to *beta_max*.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0                # start with a very smooth landscape
    beta_mul = 1.02           # increase temperature slowly

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] descent it={t:,} β={beta:.1f} score={cur_score:.6f}")

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.5 / (1.0 + 1e-4 * t)          # slowly decaying learning‑rate
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to escape tiny plateaus
        if t % 3000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (random + max‑entry attacks)
# ----------------------------------------------------------------------
def _targeted_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves together with moves that explicitly remove
    mass from the pair of indices contributing to the current maximal
    convolution entry.
    Only improving moves are kept.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    step_factor = 0.15       # up to 15 % of source mass per move
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        # --- decide move type ------------------------------------------------
        if rng.random() < 0.6:
            # random transfer
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # targeted transfer based on the dominant convolution entry
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose the move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)      # reset step size after success
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function (probability distribution) that
    minimises the evaluation function.  The routine respects the 1000 s
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2     # tiny safety margin

    # problem size – larger n gives more flexibility but costs a bit more CPU
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1️⃣  Uniform baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1.5  Global starter (if the competition provides one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score, best_seq = cur, starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Fine parametric sweep
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:          # reserve ~5 min for later phases
        best_seq, best_score = _parametric_grid_sweep(
            n, uniform, deadline, start_time, best_seq, best_score)

    # ------------------------------------------------------------------
    # 2.5️⃣  Random sampling of parameters (quick exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200:
        rng = np.random.default_rng(seed + 12345)
        best_seq, best_score = _random_param_search(
            n, uniform, deadline, start_time, best_seq, best_score, rng,
            max_trials=4000)

    # ------------------------------------------------------------------
    # 3️⃣  Long Adam‑style projected descent on smooth‑max surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 100:
        # several independent runs with tiny random perturbations
        runs = 4
        for run in range(runs):
            if time.time() >= deadline - 30:
                break
            rng = np.random.default_rng(seed + run + 100)
            a0 = best_seq.copy()
            a0 += 1e-5 * rng.standard_normal(n) * a0   # very small noise
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            # allocate a generous number of iterations – the loop stops early on deadline
            max_iters = 200_000
            a_opt, sc_opt = _smooth_descent(
                a0, max_iters, deadline, start_time, best_score, beta_max=2000.0
            )
            if sc_opt < best_score:
                best_score, best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] descent run {run} improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb polish (random + targeted moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = uniform.copy()

    # optional sanity check via the official evaluator
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
