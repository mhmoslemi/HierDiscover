    if base.sum()>0:

    a = lam*uniform + lam_sym*sym + (1-lam - lam_sym)*base
    a = np.maximum(a, 0)
    total = a.sum()
    if total>0:
        a /= total
    return a

        conv = _conv_self_fft(a, L_fft)
        max_cov = conv.max()
        # compute objective (sum is 1)

        best_a = a.copy()

        # initialize Adam

                print(f"[{time.time()-start_time:.2f}s] smoothAdam iter {t} score {cur_sc:.6f}")
            # increase temperature
            beta_temp = min(beta_temp*beta_mul, beta_max)
        return best_a, best_sc
    # allocate time for smooth adam

            max_val = conv.max()

            # compute subgradient for shift k

            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1)

            g[i_vals] += a[j_vals]

                g[j_vals[mask]] += a[i_vals[mask]]
            step = step0 / np.sqrt(iter_count)
            a_candidate = a - step * g
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s>0:

            cur_sc = eval_fast(a_candidate)
            if cur_sc < best_sc:
                best_sc = cur_sc; best_a = a_candidate.copy(); a = a_candidate
                print(f"[{time.time()-start_time:.2f}s] subgrad it {iter_count} score {cur_sc:.6f}")
            else:

                if rng.random() < prob:
                    a = a_candidate
            # jitter occasionally
            if iter_count % 3000 == 0:
                eps_j = 1e-4 * rng.random(n) * a
                a = np.maximum(a + eps_j, 0.0)
                s = a.sum()
                if s>0: a = a/s
        return best_a, best_sc
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining*0.5
        a_sub, sc_sub = subgrad_descent(best_seq, sub_deadline)
        if sc_sub < best_score:
            best_score = sc_sub; best_seq = a_sub.copy()
            print(f"Subgrad final improved to {best_score:.6f}")
    # Phase 4: final hill climb
    def hill_climb(a0, time_limit):
        a = a0.copy()
        best_a = a.copy()
        best_sc = eval_fast(a)
        rng = np.random.default_rng(seed+54321)
        while time.time() < time_limit:
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            eps = a[src] * rng.random() * 5e-4  # smaller 0.05%
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            cur_sc = eval_fast(a_prop)
            if cur_sc < best_sc:
                a = a_prop
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] hill climb score {best_sc:.6f}")
        return best_a, best_sc
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_hc, sc_hc = hill_climb(best_seq, deadline)
        if sc_hc < best_score:
            best_score = sc_hc; best_seq = a_hc.copy()
            print(f"Hill climb final improved to {best_score:.6f}")
    # final normalize
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total>0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0/n)
    print(f"Finished best score {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Efficient optimizer for the step‑function bound.
- Length n is taken from a supplied starter or set to 2048 (fast enough).
- Phase 1 : a small‑dimensional differential‑evolution search over
  geometric‑/power‑law mixtures (including a symmetric edge‑decay component).
- Phase 2 : Adam‑style projected descent on a soft‑max approximation of
  the maximal self‑convolution, using exponentiated‑gradient updates
  (they stay on the probability simplex automatically).
- Phase 3 : classic sub‑gradient descent on the exact max‑convolution.
- Phase 4 : microscopic hill‑climb that moves tiny mass chunks between
  coordinates.
All phases respect the 1000 s budget and keep the vector non‑negative with
unit sum. The best sequence found is returned as a plain Python list.
"""
import time
import numpy as np
import scipy.optimize

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective for a vector with sum≈1 (no scaling)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising
    2·n·max(conv(a,a))/(sum(a)²).  Returns a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size n (use supplied starter if possible)
    # ------------------------------------------------------------------
    n = 2048                                      # default moderate size
    try:
        starter_global = np.asarray(height_sequence_1, dtype=float)
        if starter_global.ndim == 1 and starter_global.size > 0:
            n = starter_global.size
    except Exception:
        starter_global = None

    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Possibly improve with a supplied global starter
    # ------------------------------------------------------------------
    if starter_global is not None:
        s = np.maximum(starter_global, 0.0)
        if s.sum() > 0:
            s = s / s.sum()
            sc = _eval_fast(s, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = s.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – parametric differential‑evolution search
    # ------------------------------------------------------------------
    # parametric family: mix of (i) uniform, (ii) left‑hand geometric·power,
    # (iii) symmetric edge‑decay.
    def build_sequence(params):
        r1, p1, lam, lam_sym_raw, r2 = params
        # clamp / normalise mixing weights
        lam = np.clip(lam, 0.0, 1.0)
        lam_sym_raw = np.clip(lam_sym_raw, 0.0, 1.0)
        lam_sym = lam_sym_raw * (1.0 - lam)          # ensure lam+lam_sym≤1
        # component 1 – geometric * power law (left‑hand)
        idx = np.arange(n, dtype=np.float64) + 1.0
        base = np.power(r1, np.arange(n, dtype=np.float64)) * np.power(idx, p1)
        base = np.maximum(base, 0.0)
        sb = base.sum()
        if sb > 0:
            base = base / sb
        else:
            base = uniform.copy()
        # component 2 – symmetric edge‑decay
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n, dtype=np.float64) - mid)
        sym = np.power(r2, dist)
        sym = np.maximum(sym, 0.0)
        ss = sym.sum()
        if ss > 0:
            sym = sym / ss
        else:
            sym = uniform.copy()
        # final mixture
        a = lam * uniform + lam_sym * sym + (1.0 - lam - lam_sym) * base
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = uniform.copy()
        return a

    def objective_de(params):
        a = build_sequence(params)
        # fast evaluation (sum is 1 → denominator =1)
        conv = _conv_self_fft(a, L_fft)
        return float(2.0 * n * conv.max())

    # bounds for the five parameters

        (0.0, 1.0),          # lam (uniform weight)
        (0.0, 1.0),          # lam_sym_raw (edge‑decay weight before scaling)
        (0.90, 0.9999)       # r2 (symmetric decay)
    ]

        de_res = scipy.optimize.differential_evolution(
            objective_de,
            bounds,
            seed=seed,
            maxiter=max_iter_de,
            popsize=15,
            polish=False,
            disp=False,
            updating='deferred',
            workers=1,

        sc_de = _eval_fast(a_de, L_fft)
        if sc_de < best_score:
            best_score = sc_de
            best_seq = a_de.copy()
            print(f"[{time.time() - start_time:.2f}s] DE improvement: score={best_score:.6f}")

    # Phase 2 – smooth‑max Adam descent (exponential‑gradient version)
    # ------------------------------------------------------------------
    def smooth_adam(a0, limit):

        best_a = a.copy()
        best_sc = _eval_fast(a, L_fft)

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t = 0
        beta_temp = 5.0           # start temperature for soft‑max
        beta_mul = 1.03
        beta_max = 2000.0

        L_conv = _next_pow2(2 * n - 1)
        L_corr = _next_pow2(3 * n - 2)

        while time.time() < limit:
            t += 1
            # ---- soft‑max of the convolution ----
            Af = np.fft.rfft(a, n=L_conv)
            conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]
            max_c = conv.max()
            shifted = conv - max_c
            exp_vals = np.exp(beta_temp * shifted)

            a_rev = a[::-1]
            grad_corr = np.fft.irfft(

                n=L_corr
            )

            grad = 2.0 * grad_corr[start:start + n]

            m = beta1 * m + (1.0 - beta1) * grad

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.3 / (np.sqrt(v_hat) + eps)

            a = a * np.exp(-step * m_hat)          # multiplicative update
            tot = a.sum()
            if tot > 0:
                a = a / tot
            else:
                a = np.full_like(a, 1.0 / n)

            # keep the best seen
            cur_sc = _eval_fast(a, L_fft)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] smooth‑Adam it={t:,} score={cur_sc:.6f}")

            # temperature schedule
            beta_temp = min(beta_temp * beta_mul, beta_max)

        return best_a, best_sc

    remaining = deadline - time.time()
    if remaining > 120.0:
        sub_dead = time.time() + remaining * 0.55
        a_smooth, sc_smooth = smooth_adam(best_seq, sub_dead)
        if sc_smooth < best_score:
            best_score = sc_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑Adam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    def subgradient_descent(a0, limit):
        a = a0.copy()
        best_a = a.copy()
        best_sc = _eval_fast(a, L_fft)
        rng_local = np.random.default_rng(seed + 999)
        step0 = 0.15
        it = 0
        while time.time() < limit:
            it += 1
            conv = _conv_self_fft(a, L_fft)
            max_val = conv.max()
            k = int(np.argmax(conv))

            # sub‑gradient for the shift k
            g = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1)
            j_vals = k - i_vals
            g[i_vals] += a[j_vals]
            mask = i_vals != j_vals
            if np.any(mask):
                g[j_vals[mask]] += a[i_vals[mask]]

            step = step0 / np.sqrt(it)
            a_candidate = a - step * g
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s > 0:
                a_candidate = a_candidate / s
            else:
                a_candidate = np.full_like(a_candidate, 1.0 / n)

            cur_sc = _eval_fast(a_candidate, L_fft)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_candidate.copy()
                a = a_candidate
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} score={cur_sc:.6f}")
            else:
                # occasional uphill move (simulated annealing flavour)
                prob = np.exp(-(cur_sc - best_sc) * 12.0 / (np.sqrt(it) + 1.0))
                if rng_local.random() < prob:

            # jitter every few thousand iterations
            if it % 3000 == 0:

        return best_a, best_sc

    remaining = deadline - time.time()
    if remaining > 40.0:
        sub_dead = time.time() + remaining * 0.45
        a_sub, sc_sub = subgradient_descent(best_seq, sub_dead)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()

    # ------------------------------------------------------------------
    # Phase 4 – final hill‑climb (tiny mass transfers)
    # ------------------------------------------------------------------
    def hill_climb(a0, limit):
        a = a0.copy()
        best_a = a.copy()

            if src == dst:

            eps = a[src] * rng_local.random() * 5e-4   # ≤0.05 % of source mass
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            cur_sc = _eval_fast(a_prop, L_fft)
            if cur_sc < best_sc:
                a = a_prop
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_sc:.6f}")
        return best_a, best_sc

    remaining = deadline - time.time()
    if remaining > 5.0:
        a_hc, sc_hc = hill_climb(best_seq, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Efficient optimizer for the step‑function bound.

The algorithm proceeds in four independent phases, each implemented as a
top‑level helper (no closures).  The search works on a probability simplex
(non‑negative entries that sum to one).

1️⃣ **Parametric DE** – a small‑dimensional differential‑evolution search
   over mixtures of (a) the uniform distribution, (b) a left‑hand
   geometric·power law, and (c) a symmetric edge‑decay shape.  The best
   candidate from this cheap global search provides a strong start.

2️⃣ **Smooth‑max Adam** – Adam optimisation on a soft‑max approximation of
   the maximal self‑convolution, using a multiplicative (exponential‑gradient)
   update that stays on the simplex automatically.  The temperature of the
   soft‑max is increased gradually.

3️⃣ **Exact sub‑gradient descent** – classic projected sub‑gradient steps on
   the true max‑convolution term (the maximal shift is identified with an FFT).

4️⃣ **Microscopic hill‑climb** – tiny mass transfers between random indices
   that accept only improvements (occasionally a tiny uphill move).

All phases respect a global 1000 s budget, keep the vector non‑negative with
unit sum, and retain the best sequence encountered.  The final sequence is
returned as a plain Python list of floats.
"""

import time
import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
# Basic utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric family used by the DE stage
# ----------------------------------------------------------------------
def _build_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """
    Construct a candidate sequence from a five‑dimensional parameter vector:
    params = [r1, p1, lam, lam_sym_raw, r2]
    """
    r1, p1, lam, lam_sym_raw, r2 = params
    lam = np.clip(lam, 0.0, 1.0)
    lam_sym_raw = np.clip(lam_sym_raw, 0.0, 1.0)
    # enforce lam + lam_sym ≤ 1
    lam_sym = lam_sym_raw * (1.0 - lam)

    # uniform component
    uniform = np.full(n, 1.0 / n)

    # left‑hand geometric·power law
    idx = np.arange(n, dtype=np.float64) + 1.0
    base = np.power(r1, np.arange(n, dtype=np.float64)) * np.power(idx, p1)
    base = np.maximum(base, 0.0)
    sb = base.sum()
    if sb > 0:
        base = base / sb
    else:
        base = uniform.copy()

    # symmetric edge‑decay component
    mid = (n - 1) / 2.0
    dist = np.abs(np.arange(n, dtype=np.float64) - mid)
    sym = np.power(r2, dist)
    sym = np.maximum(sym, 0.0)
    ss = sym.sum()
    if ss > 0:
        sym = sym / ss
    else:
        sym = uniform.copy()

    # final mixture (weights sum to 1)
    a = lam * uniform + lam_sym * sym + (1.0 - lam - lam_sym) * base
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot > 0:
        a = a / tot
    else:
        a = uniform.copy()
    return a


def _objective_de(params: np.ndarray, n: int, L_fft: int) -> float:
    """Objective used by differential evolution (softmax not needed)."""
    a = _build_sequence(params, n)
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max Adam descent (exponential‑gradient version)
# ----------------------------------------------------------------------
def _smooth_adam(a0: np.ndarray, n: int, L_fft: int,
                 seed: int, start_time: float, deadline: float):
    """Adam on a soft‑max surrogate, multiplicative updates keep a on simplex."""
    rng = np.random.default_rng(seed)
    a = a0.copy()
    best_a = a.copy()
    best_sc = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    beta_temp = 5.0          # start temperature for soft‑max
    beta_mul = 1.03
    beta_max = 2000.0

    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        t += 1

        # ----- soft‑max of the convolution -----
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]
        max_c = conv.max()
        shifted = conv - max_c               # stabilise exponentials
        exp_vals = np.exp(beta_temp * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max weights, shape (2n‑1,)

        # ----- gradient of the soft‑max term -----
        a_rev = a[::-1]
        grad_corr = np.fft.irfft(
            np.fft.rfft(w, n=L_corr) * np.fft.rfft(a_rev, n=L_corr),
            n=L_corr
        )
        start = n - 1
        grad = 2.0 * grad_corr[start:start + n]

        # ----- Adam on log‑space (exponential update) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.3 / (np.sqrt(v_hat) + eps)

        a = a * np.exp(-step * m_hat)   # multiplicative (exponential‑gradient) update
        tot = a.sum()
        if tot > 0:
            a = a / tot
        else:
            a = np.full_like(a, 1.0 / n)

        cur_sc = _eval_fast(a, L_fft)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑Adam it={t:,} score={cur_sc:.6f}")

        # temperature schedule (focuses on the true max later)
        beta_temp = min(beta_temp * beta_mul, beta_max)

    return best_a, best_sc

# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, n: int, L_fft: int,
                         seed: int, start_time: float, deadline: float):
    """Projected sub‑gradient descent using the maximal shift."""
    rng = np.random.default_rng(seed)
    a = a0.copy()
    best_a = a.copy()
    best_sc = _eval_fast(a, L_fft)

    step0 = 0.15
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))

        # ----- sub‑gradient for shift k -----
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0:
            a_candidate = a_candidate / s
        else:
            a_candidate = np.full_like(a_candidate, 1.0 / n)

        cur_sc = _eval_fast(a_candidate, L_fft)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} score={cur_sc:.6f}")
        else:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cur_sc - best_sc) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter to escape plateaus
        if it % 3000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            tot = a.sum()
            if tot > 0:
                a = a / tot

    return best_a, best_sc

# Final hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, n: int, L_fft: int,
                seed: int, start_time: float, deadline: float):
    """Microscopic hill‑climb that moves ≤0.05 % of a source entry."""
    rng = np.random.default_rng(seed)
    a = a0.copy()
    best_a = a.copy()
    best_sc = _eval_fast(a, L_fft)

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 5e-4   # up to 0.05 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        cur_sc = _eval_fast(a_prop, L_fft)
        if cur_sc < best_sc:
            a = a_prop
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_sc:.6f}")

    return best_a, best_sc
