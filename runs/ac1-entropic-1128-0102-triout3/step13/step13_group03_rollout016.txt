"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm (≈ 1000 s)

1️⃣  Global seed → uniform vector.
2️⃣  Fast randomised parametric sweep (geometric + power families) to obtain a good
    starting shape.
3️⃣  Several Adam‑style projected‑gradient descents on the smooth‑max surrogate
    (β up to 2000) – this gives a refined “flat‑autocorrelation” vector.
4️⃣  LP refinement: for the current a compute M = max(a*a).  Solve the linear program
    maximise 1ᵀb  s.t.  a∗b ≤ M, b ≥ 0 .  If the LP yields ∑b > 1 we can move a towards
    the normalised b, strictly decreasing the objective.  Re‑repeat while time permits.
5️⃣  Final simulated‑annealing hill‑climb (random + targeted moves) to polish the
    best vector.
6️⃣  Return the best sequence (sum normalised to 1) as a plain Python list.

All steps respect the 1000 s budget, print intermediate scores and use only
numpy / scipy (single‑core).  The code works for any length; we use n=2048 which
offers a good speed/quality trade‑off.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear (a * a) convolution via FFT.
    Returns length 2·len(a)‑1 array.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex Σv_i=1, v_i≥0.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a (sum≈1).
    Returns ``inf`` if the sum is too small.
    """
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max surrogate utilities (soft‑max of the convolution entries)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a * a (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is obtained via an FFT‑based correlation.
    """
    conv = _conv_self(a, L=L_conv)                # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                        # stabilize soft‑max
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()                # soft‑max weights, shape (2n‑1,)

    # correlation w * a_rev  → gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)         # length L_corr
    start = a.shape[0] - 1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max (β growing to ``beta_max``).
    Each iterate is projected onto the simplex.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0
    beta_mul = 1.025  # accelerate temperature growth

    while time.time() < deadline and t < max_iters:
        t += 1

        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update (Euclidean)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.6 / (1.0 + 1e-4 * t)          # slowly decaying learning‑rate
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional tiny jitter to avoid exact plateaus
        if t % 2500 == 0:
            jitter = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (max‑convolution reduction)
# ----------------------------------------------------------------------
def _lp_refine(
    a_start: np.ndarray,
    deadline: float,
    start_time: float,
    max_outer_iters: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve

        maximise   1ᵀb
        subject to  a * b ≤ M          (linear convolution)
                    b ≥ 0

    where M = max(a*a).  If the LP returns ∑b > 1 we move a towards the
    normalised b (convex combination).  Each successful move provably reduces
    the objective.  The routine stops when time runs out or when no improvement
    can be found.
    """
    a = a_start.copy()
    n = a.shape[0]
    L = 2 * n - 1
    L_fft = _next_pow2(L)
    best_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    t = 0
    # Mixing factor – fixed to 0.5 (aggressive) but can be reduced if needed
    mix = 0.5

    while time.time() < deadline - 1.0 and t < max_outer_iters:
        t += 1
        conv = _conv_self(a, L_fft)
        M = conv.max()

        # Build Toeplitz matrix A_ub with a on the diagonals (offset i)
        A_ub = sp.diags(a, offsets=np.arange(n), shape=(L, n), format="csc")
        b_ub = np.full(L, M, dtype=np.float64)

        # Solve LP: maximise sum(b)  <=> minimise -1ᵀb
        c = -np.ones(n, dtype=np.float64)
        res = opt.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"presolve": True, "time_limit": max(0, deadline - time.time() - 0.5)},
        )

        if res.status != 0:
            # LP failed – give up the refinement stage
            print(f"[{time.time() - start_time:.2f}s] LP failed (status {res.status})")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-9:
            # No room for improvement – stop
            break

        # Normalise b to sum 1 and blend
        b_norm = b_opt / sum_b
        a_new = (1.0 - mix) * a + mix * b_norm
        a_new = _project_simplex(a_new)  # numerical safety

        new_score = _eval_fast(a_new, L_fft)
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            print(
                f"[{time.time() - start_time:.2f}s] LP iter={t} score={new_score:.6f}"
            )
        else:
            # No improvement – stop early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (random + targeted moves)
# ----------------------------------------------------------------------
def _hill_climb_anneal(
    a_start: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts worsening moves with a temperature
    schedule.  Each move transfers a fraction of mass from one index to another.
    Targeted moves focus on a pair (i,j) contributing heavily to the current
    maximum convolution entry.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # temperature schedule (log‑linear)
    T0 = 0.02 * best_score            # start a little below the current score
    Tf = 1e-6
    total_iters = 200_000
    decay = (Tf / T0) ** (1.0 / total_iters)

    T = T0
    it = 0
    while time.time() < deadline - 0.2:
        it += 1
        if rng.random() < 0.55:
            # --- random mass transfer ------------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * 0.12   # up to 12 % of source mass
        else:
            # --- targeted transfer (focus on the current max conv entry) ----------
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            # the interval of i that can appear in the pair (i, j = k_max-i)
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            # move mass from the larger of the two to a random different slot
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * 0.12

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        dE = new_score - best_score

        accept = False
        if dE < 0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-dE / max(T, 1e-12)):
                accept = True

        if accept:
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it} score={best_score:.6f}")

        # temperature decay
        T = max(T * decay, Tf)

    return best_a, best_score


# ----------------------------------------------------------------------
# Randomised parametric family (geometric + power, and two‑exponential mixtures)
# ----------------------------------------------------------------------
def _random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative shape and normalise it to sum = 1.
    The shape is a mixture of three simple families:
      * rⁱ·(i+1)ᵖ                      (geometric × power)
      * λ·r₁ⁱ + (1‑λ)·r₂ⁱ              (mixture of two exponentials)
      * uniform                        (occasionally mixed in)
    Parameters are drawn uniformly from sensible ranges.
    """
    i = np.arange(n, dtype=np.float64)

    # Family 1 – geometric × power
    r = rng.uniform(0.90, 0.9999)
    p = rng.uniform(-5.0, 3.0)
    shape1 = np.power(r, i) * np.power(i + 1.0, p)

    # Family 2 – mixture of two exponentials
    r1 = rng.uniform(0.85, 0.999)
    r2 = rng.uniform(0.85, 0.999)
    lam = rng.random()
    shape2 = lam * np.power(r1, i) + (1.0 - lam) * np.power(r2, i)

    # Random convex combination of the two families (+ optional uniform)
    w = rng.random(3)
    w /= w.sum()
    mixed = w[0] * shape1 + w[1] * shape2 + w[2] * np.ones(n, dtype=np.float64)

    # Ensure non‑negativity and normalise
    mixed = np.maximum(mixed, 0.0)
    if mixed.sum() == 0.0:
        mixed = np.full(n, 1.0 / n)
    else:
        mixed /= mixed.sum()
    return mixed


def _parametric_sweep_random(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Randomised sweep over the above families.  Stops early if time runs low.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    for _ in range(max_iters):
        if time.time() >= deadline - 120:
            break
        cand = _random_shape(n, rng)
        # occasionally blend with the uniform vector
        if rng.random() < 0.3:
            lam = rng.random()
            cand = lam * uniform + (1.0 - lam) * cand
            cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] random sweep score={sc:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function minimising the given evaluation
    function.  The routine respects the 1000 s budget and returns the best
    sequence discovered (as plain Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # safety margin

    # problem size – 2048 provides a fast yet expressive search space
    n = 2048
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣  Baseline (uniform)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(
        f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}"
    )

    # --------------------------------------------------------------
    # 1️⃣  Randomised parametric sweep (fast exploration)
    # --------------------------------------------------------------
    if time.time() < deadline - 300:
        best_seq, best_score = _parametric_sweep_random(
            n,
            uniform,
            deadline,
            start_time,
            best_seq,
            best_score,
            max_iters=4000,
        )

    # --------------------------------------------------------------
    # 2️⃣  Multiple Adam‑style projected gradient descents
    # --------------------------------------------------------------
    descent_seeds = [seed, seed + 10, seed + 20]
    for ds in descent_seeds:
        if time.time() >= deadline - 180:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        # tiny random perturbation to diversify starts
        a0 += 1e-5 * rng.standard_normal(n) * a0
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        a_opt, sc_opt = _adam_descent(
            a0,
            max_iters=50000,
            deadline=deadline - 30,   # keep a few seconds for later steps
            start_time=start_time,
            best_score_initial=best_score,
            beta_max=2000.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}"
            )

    # --------------------------------------------------------------
    # 3️⃣  Linear‑programming refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 120:
        a_opt, sc_opt = _lp_refine(best_seq, deadline, start_time, max_outer_iters=25)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] LP refinement score={sc_opt:.6f}"
            )

    # --------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _hill_climb_anneal(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}"
            )

    # --------------------------------------------------------------
    # 5️⃣  Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check with the official evaluator (optional)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(
        f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}"
    )
    return [float(x) for x in best_seq]
