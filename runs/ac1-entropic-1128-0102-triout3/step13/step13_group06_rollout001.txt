"""
Propose a low‑objective non‑negative step function.

Algorithm
---------

* **Warm‑start** – generate a large portfolio of analytic shapes
  (power‑tents, exponentials, power‑law, Gaussian, etc.) together with
  a few random Dirichlet draws and the previously discovered best vector.
  All candidates are normalised to sum = 1 and the best by the exact
  objective (computed with FFT) is retained.

* **Gradient‑guided simulated‑annealing hill‑climb** – keep the current
  vector `a` and its full convolution `conv`.  In each iteration
  we locate the index `k_max` where `conv` attains its maximum.
  The contribution of coefficient `a[i]` to that maximum is
  `c[i] = a[i] * a[k_max‑i]` (zero when the pair is out of range).  We
  choose a source index proportional to `c[i]` (so we move mass from
  entries that most affect the current peak) and a destination index
  uniformly at random.  A fraction of the source mass is transferred;
  the convolution is updated incrementally in O(n) using the analytical
  update `_apply_move`.  The new objective is evaluated from the
  updated convolution.  The move is always accepted if it does not
  increase the objective; otherwise it is accepted with a Metropolis
  probability that decays linearly with the remaining time
  (simulated‑annealing).  The temperature schedule starts at
  `0.05 * best_score` and ends at zero.

* **Periodic jitter** – if no improvement has been observed for a
  while, a small random perturbation is added to the current best
  vector (followed by renormalisation) to escape possible local minima.

* **Return** – the best vector found (normalised to sum = 1) is returned
  as a plain Python list of floats, ready for `evaluate_sequence`.

The search runs for the full time budget (default 1000 s, with a
0.5 s safety margin).  It uses only NumPy and the provided
`_apply_move` routine, keeping memory usage modest and staying within a
single CPU core.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective (sum of a assumed to be 1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`, updating both the
    coefficient vector `a` and its convolution `conv` in‑place.
    """
    n = a.shape[0]

    # linear term: 2*eps*a shifted by src (subtract) and dst (add)
    conv[src : src + n] -= 2.0 * eps * a
    conv[dst : dst + n] += 2.0 * eps * a

    # quadratic term
    conv[2 * src] += eps * eps
    conv[2 * dst] += eps * eps
    conv[src + dst] -= 2.0 * eps * eps

    # update coefficients
    a[src] -= eps
    a[dst] += eps


def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a rich collection of analytic (unnormalised) shapes."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents (min(x,1‑x))**p
    for p in np.linspace(0.05, 3.0, 30):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Symmetric exponential decays from centre
    for lam in np.logspace(-5, -0.3, 20):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Symmetric power‑law from the edges
    for p in np.linspace(0.1, 3.0, 20):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt U‑shape (more mass at edges)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Half‑cosine / sine bump
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, several sigmas
    for sigma in (0.08, 0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet‑like draws
    rng = np.random.default_rng(12345)
    for _ in range(30):
        rnd = rng.random(n)
        candidates.append(rnd)

    # Normalise all candidates (sum to 1) and discard the zero ones
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with minimal evaluation score.
    Returns a list of non‑negative floats (sum ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.5
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # Determine problem size.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 4096                               # default size

    # --------------------------------------------------------------
    # 1) Warm‑start – generate many analytic shapes + random draws.
    # --------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # include the previously discovered best if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm_starts.append(starter / starter.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    # sanity fallback
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=float)

    # --------------------------------------------------------------
    # Initialise state for local search.
    # --------------------------------------------------------------
    a = best_vec.copy()                    # coefficients (sum = 1)
    conv = _conv_fft(a)                    # full convolution
    n_obj_factor = 2.0 * n                 # because sum(a) == 1
    cur_score = n_obj_factor * conv.max()

    # Global best (might be improved later)
    global_best_vec = a.copy()
    global_best_score = cur_score

    # Random generator for the search
    rng = np.random.default_rng(seed + 42)

    # Parameters for the annealing / move size schedule
    frac_start = 0.30                      # early move fraction
    frac_end = 0.02                        # final move fraction
    temp_start = max(global_best_score * 0.05, 1e-6)   # initial temperature
    jitter_interval = 30.0                # seconds without improvement before jitter

    last_improve_time = start_time

    it = 0
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # Compute the current peak of the convolution and the gradient.
        # ----------------------------------------------------------
        k_max = int(np.argmax(conv))
        # g[i] = a[k_max - i] when the index is inside [0, n)
        g = np.zeros(n, dtype=float)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = a[k_max - idx]

        # Contribution of each coefficient to the current max
        contrib = a * g
        src_weights = contrib
        src_sum = src_weights.sum()
        if src_sum <= 0.0 or np.isnan(src_sum):
            # fallback: choose source proportional to mass
            src = rng.choice(n, p=a)
        else:
            src = rng.choice(n, p=src_weights / src_sum)

        # Destination – uniform (avoid trivial self‑move)
        dst = rng.integers(n)
        if dst == src:
            continue

        # ----------------------------------------------------------
        # Choose move magnitude (fraction of source mass).
        # ----------------------------------------------------------
        now = time.time()
        remaining = deadline - now
        if remaining <= 0:
            break
        # linear decay of the fraction
        frac = frac_end + (frac_start - frac_end) * (remaining / total_time)
        eps = a[src] * rng.random() * frac
        if eps <= 0.0:
            continue

        # ----------------------------------------------------------
        # Apply move and evaluate.
        # ----------------------------------------------------------
        before_obj = n_obj_factor * conv.max()
        _apply_move(a, conv, src, dst, eps)
        after_obj = n_obj_factor * conv.max()

        # ----------------------------------------------------------
        # Acceptance test (deterministic descent + simulated annealing).
        # ----------------------------------------------------------
        accept = False
        if after_obj <= before_obj:
            accept = True
        else:
            # temperature linearly decreasing to zero
            temp = temp_start * (remaining / total_time)
            if temp > 0.0:
                prob = math.exp(-(after_obj - before_obj) / temp)
                if rng.random() < prob:
                    accept = True

        if not accept:
            # revert the move
            _apply_move(a, conv, dst, src, eps)
            after_obj = before_obj   # not used further
        else:
            # successful move – maybe a new global best
            if after_obj < global_best_score:
                global_best_score = after_obj
                global_best_vec = a.copy()
                last_improve_time = now
                print(f"[{now - start_time:.2f}s] iter={it:,} new best={global_best_score:.6f}")

        # ----------------------------------------------------------
        # Periodic jitter to escape stagnation.
        # ----------------------------------------------------------
        if now - last_improve_time > jitter_interval and (deadline - now) > 5.0:
            # perturb the best vector slightly and restart local search
            a = global_best_vec.copy()
            noise = rng.normal(scale=0.005, size=n)
            a += noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s <= 0.0:
                a = np.full(n, 1.0 / n, dtype=float)
            else:
                a /= s
            conv = _conv_fft(a)
            last_improve_time = now
            # print a short notice
            print(f"[{now - start_time:.2f}s] jitter applied after {jitter_interval}s without improvement")

    # --------------------------------------------------------------
    # Return the best vector found (normalised to sum = 1).
    # --------------------------------------------------------------
    best = global_best_vec
    best = np.maximum(best, 0.0)
    s = best.sum()
    if s > 0.0:
        best /= s
    else:
        best = np.full(n, 1.0 / n, dtype=float)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score={global_best_score:.6f}")
    return [float(v) for v in best]
