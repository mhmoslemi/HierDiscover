def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, math, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    try: n = len(height_sequence_1)
    except NameError: n = 4096
    # Warm start: generate many candidate analytic shapes (including new param grid)
    candidates = generate_candidates(n)
    best_vec, best_score = None, float('inf')
    # evaluate each candidate via FFT and objective
    for cand in candidates:
        score = _objective(cand)  # where _objective expects normalized sum 1?
        if score < best_score:
            best_score, best_vec = score, cand.copy()
    a = best_vec.copy()
    conv = _conv_fft(a)
    cur_score = best_score
    n_obj = 2.0 * n  # sum is 1 
    # hill climbing / annealing
    rng = np.random.default_rng(seed+1)
    it = 0
    temperature_start = 1e-2 # maybe objective diff scale
    temperature_end = 1e-6
    while time.time() < deadline:
        it += 1
        # compute temperature linearly decreasing
        elapsed = time.time() - start
        t_frac = (deadline - start - elapsed) / (deadline - start)  # fraction remaining
        temperature = temperature_end + (temperature_start - temperature_end) * t_frac
        # decide whether to do targeted move
        if rng.random() < 0.8:
            # targeted
            k = conv.argmax()
            # choose src weighted by a
            src = rng.choice(n, p=a)
            if a[src] < 1e-15:
                continue
            # partner value at src for max index
            if 0 <= k - src < n:
                partner_src = a[k - src]
            else:
                partner_src = 0.0
            # choose dst that has lower partner value
            # sample up to attempts
            for attempt in range(5):
                dst = rng.integers(n)
                if dst == src:
                    continue
                if 0 <= k - dst < n:
                    partner_dst = a[k - dst]
                else:
                    partner_dst = 0.0
                if partner_dst < partner_src:
                    break
            else:
                # fallback random dst
                dst = rng.integers(n)
                if dst == src:
                    continue
        else:
            # random move
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst:
                continue
        # epsilon
        # choose epsilon fraction decreasing with time
        # early large moves, later small
        frac_max = 0.2 * (1 + t_frac)  # at start approx 0.4, at end ~0.2
        eps = a[src] * rng.random() * frac_max
        if eps <= 0.0:
            continue
        # move
        before_max = conv.max()
        before_obj = n_obj * before_max

    # final normalise best_vec to sum 1

    best_vec = np.clip(best_vec, 0.0, 1000.0)

"""
Propose a low‑objective step‑function.

The search consists of three stages:

1. **Rich warm‑start** – generate a large family of analytic
   candidates (power‑law edges, power‑tents, exponentials, Gaussians,
   plus two‑parameter power‑law families).  All candidates are
   normalised to sum‑to‑one and the best according to the exact objective
   `2·n·max(conv(a,a))/sum(a)^2` is selected.

2. **Guided stochastic descent** – starting from the best warm‑start,
   mass is repeatedly transferred between indices.
   * 80 % of the moves are *targeted*: the current index `k*` at which the
     convolution attains its maximum is identified; a source index is
     chosen proportionally to its current mass, and a destination index
     is accepted only if it reduces the contribution to `k*`.
   * 20 % of the moves are completely random to keep the search
     ergodic.
   * A simple simulated‑annealing scheme occasionally accepts
     uphill moves; the temperature and the allowed step size shrink
     linearly with the remaining time budget.
   * The convolution is updated in‑place in O(n) using the same
     incremental formula as the reference implementation, so millions of
     moves can be tried within the time limit.

3. **Return the best vector found** – the best vector is normalised,
   clipped to the bounds enforced by `evaluate_sequence`, and returned
   as a plain Python list.

The algorithm works for any length `n`; if a previous solution
`height_sequence_1` exists its length is reused, otherwise a default
size of 4096 is used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (assumed non‑negative, sum>0)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from `src` to `dst` and update the convolution.
    In‑place modification of `a` and `conv`.
    """
    n = a.shape[0]

    # linear part: -2*eps*a shifted by src, +2*eps*a shifted by dst
    conv[src : src + n] -= 2.0 * eps * a
    conv[dst : dst + n] += 2.0 * eps * a

    # quadratic part:
    conv[2 * src] += eps * eps
    conv[2 * dst] += eps * eps
    conv[src + dst] -= 2.0 * eps * eps

    # finally move the mass
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Analytic candidate generator (much richer than the previous version)
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """
    Generate a diverse set of analytic, non‑negative, unnormalised shapes.
    The returned list contains vectors that are **already normalised** to
    sum‑to‑one.
    """
    x = np.arange(n, dtype=float) / (n - 1)  # scaled to [0,1]
    candidates = []

    # ------------------------------------------------------------------
    # 1) Power‑tents  (min(x,1‑x))**p
    # ------------------------------------------------------------------
    for p in np.linspace(0.05, 3.0, 30):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # ------------------------------------------------------------------
    # 2) Exponential decay from centre (symmetric)
    # ------------------------------------------------------------------
    for lam in np.logspace(-5, -0.3, 20):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # ------------------------------------------------------------------
    # 3) Power‑law from the edges (symmetric)
    # ------------------------------------------------------------------
    ix = np.arange(1, n + 1, dtype=float)

    # simple power‑law
    for p in np.linspace(0.1, 6.0, 30):
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # two‑parameter family with offset delta
    for delta in (0.0, 0.5, 1.0, 5.0):
        for p in (0.5, 1.0, 2.0, 4.0):
            ix_shift = ix + delta
            shape = (ix_shift ** -p) + ((n + 1 - ix_shift) ** -p)
            candidates.append(shape)

    # ------------------------------------------------------------------
    # 4) Inverse‑sqrt (U‑shaped)
    # ------------------------------------------------------------------
    shape = 1.0 / np.sqrt(ix) + 1.0 / np.sqrt(n + 1 - ix)
    candidates.append(shape)

    # ------------------------------------------------------------------
    # 5) Half‑cosine (already used as warm‑start but keep for completeness)
    # ------------------------------------------------------------------
    candidates.append(np.sin(np.pi * x))

    # ------------------------------------------------------------------
    # 6) Gaussian centred, different sigmas
    # ------------------------------------------------------------------
    for sigma in (0.08, 0.12, 0.20, 0.30, 0.45):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # ------------------------------------------------------------------
    # 7) Random Dirichlet draws – a few of them
    # ------------------------------------------------------------------
    rng = np.random.default_rng(12345)
    for _ in range(12):
        rnd = rng.random(n)
        candidates.append(rnd)

    # ------------------------------------------------------------------
    # Normalise all candidates to sum = 1 and keep only those with positive mass
    # ------------------------------------------------------------------
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)

    return normalised

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin – the harness adds another ~0.5 s
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Determine the problem size (reuse previous solution if possible)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        n = 4096                               # default size

    # ------------------------------------------------------------------
    # 1) Warm‑start – evaluate a very rich collection of analytic shapes
    # ------------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # If a previous solution exists, inject it (normalised)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm_starts.append(starter / starter.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    # sanity fallback
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    a = best_vec.copy()
    conv = _conv_fft(a)
    cur_score = best_score
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Guided stochastic descent with annealing
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    n_obj = 2.0 * n                     # sum(a) stays = 1 => denominator = 1
    it = 0
    # annealing schedule parameters
    temp_start = 0.02    # initial temperature (objective units)
    temp_end = 1e-6     # final temperature
    # step‑size schedule (fraction of source mass)
    eps_start = 0.5
    eps_end = 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break
        it += 1

        # remaining fraction (1 = just started, 0 = time's up)
        frac = max(0.0, 1.0 - (now - start_time) / total_time)

        # linear temperature & epsilon schedules
        temperature = temp_end + (temp_start - temp_end) * frac
        eps_frac = eps_end + (eps_start - eps_end) * frac

        # ------------------------------------------------------------------
        # Choose move type
        # ------------------------------------------------------------------
        if rng.random() < 0.80:
            # ---- Targeted move -------------------------------------------------
            k = int(conv.argmax())                 # index of current max
            # source proportional to its mass (as before)
            src = rng.choice(n, p=a)
            if a[src] < 1e-15:
                continue
            # partner contribution for the max index
            if 0 <= k - src < n:
                partner_src = a[k - src]
            else:
                partner_src = 0.0

            # try up to a few times to find a destination that reduces the partner
            dst = src
            for _ in range(6):
                cand = rng.integers(n)
                if cand == src:
                    continue
                if 0 <= k - cand < n:
                    partner_cand = a[k - cand]
                else:
                    partner_cand = 0.0
                if partner_cand < partner_src:
                    dst = cand
                    break
            if dst == src:
                # fall back to a pure random destination
                dst = rng.integers(n)
                if dst == src:
                    continue
        else:
            # ---- Pure random move ---------------------------------------------
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst:
                continue

        # ------------------------------------------------------------------
        # Amount of mass to transfer
        # ------------------------------------------------------------------
        eps = a[src] * rng.random() * eps_frac
        if eps <= 0.0:
            continue

        # ------------------------------------------------------------------
        # Apply move and decide acceptance
        # ------------------------------------------------------------------
        before_max = conv.max()
        before_obj = n_obj * before_max

        _apply_move(a, conv, src, dst, eps)

        new_max = conv.max()
        new_obj = n_obj * new_max

        # accept if improvement, otherwise with Metropolis probability
        accept = False
        if new_obj <= before_obj:
            accept = True
        else:
            # uphill acceptance
            prob = math.exp(-(new_obj - before_obj) / temperature) if temperature > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            # keep the move – maybe a new record
            if new_obj < best_score:
                best_score = new_obj
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # revert the move
            _apply_move(a, conv, dst, src, eps)

    # ------------------------------------------------------------------
    # 3) Final normalisation and conversion to Python list
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    # Clip to the hard bounds enforced by evaluate_sequence (just for safety)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score = {best_score:.6f}")
    return [float(v) for v in best_vec]
