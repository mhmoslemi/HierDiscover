def _active_subgradient_descent(a0, deadline, start_time, tolerance=1e-5):
    a = a0.copy()
    n = len(a)

    # momentum if we want; for now use Adam
    m = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)  # compute self conv
        maxc = conv.max()
        # if improved
        cur_score = 2 * n * maxc / (a.sum()**2)  # our objective (same as _objective)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        eps_act = maxc * tolerance

"""
Propose a low‑score step function.

The algorithm proceeds in three stages:

1. **Initial pool** – generate a handful of analytic shapes
   (power ramps, beta distributions, mixtures) and keep the best
   candidates after normalising them to the probability simplex.

2. **Active‑set Adam optimisation** – minimise the
   smooth approximation of  max(conv(a,a)) while keeping a on the
   simplex.  At each iteration the set of (near‑maximal) convolution
   entries is selected; the gradient is the sum of the sub‑gradients
   of those entries and is computed by a single FFT‑based convolution.
   An Adam‑style update with projection yields a fast, robust descent.

3. **Metropolis annealing polish** – a light simulated‑annealing
   phase moves tiny packets of mass between positions and accepts
   deteriorations with a temperature schedule, refining the solution.

The whole routine respects a 1000 s budget and returns a plain
Python list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT based linear convolution (real FFT, generic lengths)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using a real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    L = _next_pow2(n)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
#  Exact objective (scale‑free, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) .  (No safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
#  Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x≥0, Σx=1} via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
#  Analytic families of shapes
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape built from two power ramps joining a flat plateau.
    Normalised to sum = 1.
    """
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised beta distribution on {0,…,n‑1}, normalised to sum = 1.
    """
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
#  Active‑set Adam optimiser (smooth max of the convolution)
# ----------------------------------------------------------------------
def _active_adam_opt(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     tol_frac: float = 0.01,
                     base_lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a))/(sum(a)**2) using an Adam optimiser.
    The gradient is the sum of sub‑gradients of all convolution entries
    whose value lies within `tol_frac` of the current maximum.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        maxc = conv.max()
        cur_score = 2.0 * n * maxc / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,}  score={cur_score:.6f}")

        # ---- active set (indices within a fraction of the maximum) ----
        eps_act = maxc * tol_frac
        act = (conv >= maxc - eps_act).astype(np.float64)

        # gradient = sum_{k in active} ∂ conv_k / ∂ a
        # using FFT: grad = 2 * (a * reverse(act))[ (2n‑2)‑i ]
        w_rev = act[::-1]
        full = _conv_fft(a, w_rev)               # length = n + (2n‑1)‑1 = 3n‑2
        slice_ = full[n - 1:2 * n - 1]           # length n, corresponds to indices n‑1 … 2n‑2
        grad = 2.0 * slice_[::-1]                # reverse to obtain the proper ordering

        # ---- Adam update (projected) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

    return best_a, best_score

#  Light simulated‑annealing polish (same as the original code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Metropolis‑type annealer moving tiny packets of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score
