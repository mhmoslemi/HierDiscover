    best_score = _objective(a)

    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5  # base step
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # compute gradient
        g = np.zeros_like(a)
        low = max(0, max_idx - n + 1)
        high = min(n - 1, max_idx)
        idx_i = np.arange(low, high+1)
        idx_j = max_idx - idx_i
        # base gradient
        g[idx_i] = 2.0 * a[idx_j]
        # diagonal correction
        eq = (idx_i == idx_j)
        if np.any(eq):
            g[idx_i[eq]] = a[idx_i[eq]]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        a = _project_simplex(a)

        if score < best_score:
            best_score = score
            best = a.copy()
            # optionally print progress:
            print(f"[{time.time() - start_time:.2f}s] GD it={it} score={score:.6f}")

"""
Propose a step‑function whose self‑convolution has a low ℓ∞‑norm.
The algorithm

1. builds a small pool of analytic‑ and random‑shapes,
2. keeps the few best candidates (by the exact objective),
3. starting from the best one runs a long projected‑Adam optimisation on the
   *hard* maximum of the convolution (sub‑gradient of the max),
   4. occasionally perturbs the current iterate to escape plateaus,
5. returns the best vector found (projected onto the probability simplex).

All heavy work is done with real‑FFT convolution, which is O(N log N) and fast
enough to spend the whole 1000 s budget on optimisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (FFT size)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (exact, no circular aliasing)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return C


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes -------------------------------------------------------
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau, normalised to sum 1."""
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sine_shape(n: int) -> np.ndarray:
    """Half‑sine hill, normalised to sum 1."""
    shape = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Keep the K best candidates ------------------------------------------------
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Sub‑gradient descent on the hard max (Adam on the sub‑gradient)
# ----------------------------------------------------------------------
def _max_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of
        f(a) = max_k (a * a)[k]
    using the exact sub‑gradient of the maximal lag.
    """
    a = a0.copy()
    n = a.shape[0]

    # best ever seen
    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    step0 = 0.4        # base learning rate – tuned empirically
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # ---- sub‑gradient of the maximal lag -------------------------
        g = np.zeros_like(a)
        low = max(0, max_idx - n + 1)
        high = min(n - 1, max_idx)
        idx_i = np.arange(low, high + 1)
        idx_j = max_idx - idx_i
        # generic contribution 2 * a_j
        g[idx_i] = 2.0 * a[idx_j]
        # diagonal terms appear only once
        diag = (idx_i == idx_j)
        if np.any(diag):
            g[idx_i[diag]] = a[idx_i[diag]]

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (np.sqrt(it) + 1.0)    # slowly decreasing step

        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # ---- keep the best -------------------------------------------
        cur_score = _objective(a)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} "
                  f"score={cur_score:.6f}")
        else:
            no_improve += 1

        # ---- occasional random jitter to escape stalls -------------
        if no_improve >= 5000:
            sigma = 0.002
            noise = np.random.default_rng().normal(0.0, sigma, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Choose an appropriate discretisation size
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # --------------------------------------------------------------
    # 1) Build a modest pool of analytic / random candidates
    # --------------------------------------------------------------
    TOP_K = 5
    pool = []          # (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    for p in np.linspace(0.2, 2.0, 9):
        for a in np.linspace(0.0, 0.5, 6):
            vec = _make_shape(N, p, a)
            _push_top(pool, _objective(vec), vec, TOP_K)

    # beta‑shapes (U‑shaped for α<1)
    for a in np.linspace(0.1, 0.9, 9):
        vec = _make_beta_shape(N, a)
        _push_top(pool, _objective(vec), vec, TOP_K)

    # sine hill – another smooth analytic shape
    sine = _make_sine_shape(N)
    _push_top(pool, _objective(sine), sine, TOP_K)

    # random Dirichlet vectors (various concentration parameters)
    rng = np.random.default_rng(seed + 9999)
    for conc in (0.05, 0.15, 0.3, 0.7):
        for _ in range(12):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            _push_top(pool, _objective(a), a, TOP_K)

    # incorporate the global starter if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    old_x = np.linspace(0, 1, starter.size, endpoint=False)
                    new_x = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(new_x, old_x, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # current best
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Long projected Adam descent on the hard max
    # --------------------------------------------------------------
    # give the gradient stage almost the whole remaining budget
    remaining = deadline - time.time()
    if remaining > 5.0:      # need at least a few seconds for a meaningful run
        refined_seq, refined_score = _max_gradient_descent(
            best_seq, deadline, start
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(f"[{time.time() - start:.2f}s] After GD best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
