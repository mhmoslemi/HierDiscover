"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm (≈1000 s budget)

1️⃣  Choose a moderately large problem size (n = 8192).  A larger n gives
    more freedom for “flattening’’ the self‑convolution while still being
    fast thanks to FFT‑based convolutions.

2️⃣  Build a pool of diverse seeds:
        – the uniform vector,
        – a fine parametric sweep a_i ∝ r^i·(i+1)^p (including several
          mixtures with the uniform vector),
        – a symmetric family a_i ∝ ((i+1)(n‑i))^p  (heavy mass near the
          ends, light in the centre),
        – a sinusoidal shape sin(π·(i+½)/n),
        – a few random sparse vectors (≈2·√n non‑zero entries).

3️⃣  For each seed run a **projected sub‑gradient descent** on the exact
    objective.  At every iteration we compute the full self‑convolution
    (FFT), locate the index k where the maximum is attained and use the
    gradient of that single entry (g_i = 2·a_{k‑i}).  The step size is a
    Polyak‑type rule  (f‑f★)/‖g‖², capped to keep updates stable.
    After the update we project back onto the probability simplex.
    The cheap gradient makes it possible to perform many thousands of
    iterations within the time limit.

4️⃣  Polish the best vector with a **targeted hill‑climb** that mixes
    random mass‑transfer moves with “max‑entry attacks’’: move a fraction
    of the mass from the pair (i,j) that contributes most to the current
    maximal convolution entry to a random empty (or low) position.

5️⃣  Return the best sequence (normalised to sum = 1) as a plain list of
    Python floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Length of the result is 2·len(a)‑1, returned as a plain 1‑D array.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective value 2·n·max(conv) for a vector (sum normalised)."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑conv objective
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    max_iters: int = 200_000,
    step_cap: float = 0.2,
    min_step: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses the gradient of the currently dominant convolution entry.
    The iterate is always projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and max entry -----
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        cur_score = 2.0 * n * max_val  # denominator = 1 after projection
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={t:,} score={cur_score:.6f}")

        # ----- gradient of the active entry (g_i = 2·a_{k_max-i}) -----
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # ----- Polyak step (f‑f★)/||g||², capped -----
        norm2 = np.dot(g, g) + 1e-20
        step = (cur_score - best_score) / norm2
        if step <= 0.0 or not np.isfinite(step):
            step = step_cap * 0.5
        step = max(min(step, step_cap), min_step)

        # ----- update & projection -----
        a = a - step * g
        a = _project_simplex(a)

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (random + max‑entry attacks)
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves mixed with targeted reductions of the
    currently dominant convolution entry.
    """
    a = a_start.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()

    # step‑size parameters
    step_factor = 0.15
    min_factor = 1e-6
    decay = 0.995

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Decide move type (random vs targeted)
        # --------------------------------------------------------------
        if rng.random() < 0.6:
            # ----- random transfer -------------------------------------------------
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # ----- targeted transfer ------------------------------------------------
            conv = _conv_self(a, L_fft)
            k_max = int(conv.argmax())

            # Compute which pair (i,j) among the contributors to k_max has the largest product
            idx = np.arange(n)
            j = k_max - idx
            mask = (j >= 0) & (j < n)
            prod = np.zeros_like(a)
            prod[mask] = a[idx[mask]] * a[j[mask]]
            # pick the index with maximal contribution (any of the two)
            i_max = np.argmax(prod)
            j_max = k_max - i_max

            # move mass from the larger of the two indices
            src = i_max if a[i_max] >= a[j_max] else j_max
            if a[src] < 1e-15:
                continue
            # pick a destination that is not part of the current pair
            dst = rng.integers(n)
            if dst == i_max or dst == j_max:
                continue
            delta = a[src] * rng.random() * step_factor

        # ----- propose move --------------------------------------------------
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)   # reset after a success
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Coarse‑to‑fine parametric sweep (geometric + power, symmetric, sinusoid)
# ----------------------------------------------------------------------
def _parametric_sweep(
    n: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Explore several analytic families and keep the best vector."""
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    rev_idx = np.arange(n, 0, -1, dtype=np.float64)    # n-i
    L_fft = _next_pow2(2 * n - 1)

    # geometric‑power family
    r_grid = np.logspace(np.log10(0.92), np.log10(0.9995), num=25)
    p_grid = np.linspace(-2.0, 1.5, 31)

    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            # mix with uniform
            uniform = np.full(n, 1.0 / n)
            for lam in (0.0, 0.15, 0.3, 0.6, 1.0):
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # symmetric power family (heavy at the ends)
    for p in np.linspace(-2.0, -0.2, 21):
        shape = np.power(idx * rev_idx, p)
        s = shape.sum()
        shape /= s
        for lam in (0.0, 0.4, 0.8, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * shape
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] sym‑power p={p:.2f} λ={lam:.2f} score={sc:.6f}")

    # sinusoidal shape (half‑sine)
    sine = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    sine /= sine.sum()
    sc = _eval_fast(sine, L_fft)
    if sc < best_score:
        best_score = sc
        best_seq = sine.copy()
        print(f"[{time.time() - start_time:.2f}s] sinusoid score={sc:.6f}")

    return best_seq, best_score
