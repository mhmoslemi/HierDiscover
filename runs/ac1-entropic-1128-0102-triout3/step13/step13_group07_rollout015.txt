def _adam_optimize_softmax(a0, deadline, start_time, lr=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2*n -1)
    Af = np.fft.rfft(a, n=L)   # compute once maybe? But a changes, need recompute each iteration
    best_a = a.copy()
    best_score = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution a*a via FFT
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
        max_conv = conv.max()
        # compute weight vector (softmax) over conv
        # progress = (time.time() - start_time) / (deadline - start_time)
        # lambda schedule linearly increase
        progress = (time.time() - start_time) / (deadline - start_time)
        lam = 0.5 + 99.5 * progress  # goes from 0.5 to 100.0
        # to avoid overflow, subtract max before exp
        w_raw = np.exp(lam * (conv - max_conv))
        w_sum = w_raw.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w_raw / w_sum
        # compute gradient of weighted max:
        rev_a = a[::-1]
        Wf = np.fft.rfft(w, n=L)
        Afr = np.fft.rfft(rev_a, n=L)
        grad_temp = np.fft.irfft(Wf * Afr, n=L)[:2*n-1]
        grad_max = 2.0 * grad_temp[n-1: n-1+n]  # slice central n entries
        grad_obj = 2.0 * n * grad_max
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        # projection onto simplex
        a = _project_simplex(a)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam-soft it={it} score={cur_score:.6f} lam={lam:.1f}")
        # decay lr occasionally
        if it % 3000 == 0:
            lr *= 0.9
    return best_a, best_score

"""
Improved minimiser for the max‑self‑convolution objective.

The algorithm works in three stages:

1. **Warm‑starts** – generate a set of analytic candidates (tent,
   cosine, exponential, Gaussian, …).  The best of these (in terms of the
   exact objective 2·n·max(conv)) is taken as the initial point.

2. **Smoothed‑max Adam optimisation** – the non‑smooth objective
   `max(conv)` is replaced by a soft‑max with a temperature λ that is
   increased linearly during the run.  This gives a dense sub‑gradient
   (a weighted sum of the individual convolution‑gradients) which helps
   the optimiser escape local plateaus.  The Adam update is performed on
   the probability simplex (Euclidean projection onto Σa_i=1, a_i≥0).

3. **Greedy hill‑climb polish** – a cheap Metropolis walk that moves a
   random packet of mass between two indices, accepting only strictly
   improving moves.  This final polish frequently yields a few 10⁻⁴
   improvements.

All heavy work uses NumPy’s real FFT, which makes each iteration
≈O(n log n).  The routine automatically respects the 1000 s limit: it
uses the first ~70 % of the remaining budget for Adam, the rest for the
hill‑climb, and always returns the best sequence seen so far.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w

    cand.append(np.minimum(x, 1.0 - x))

    cand.append(np.sin(np.pi * x))

    # 3) half‑circle
    cand.append(np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0)))

    # 4) Gaussian centred
    sigma = 0.2
    cand.append(np.exp(-0.5 * ((x - 0.5) / sigma) ** 2))

    # 5) power‑tent
    for p in np.linspace(0.2, 2.0, 9):
        cand.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decays (forward and reversed)
    for scale in (3.0, 5.0, 8.0):
        e = np.exp(-scale * x)
        cand.append(e)
        cand.append(e[::-1])

    # 7) uniform
    cand.append(np.ones_like(x))

    # normalise each candidate
    res = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            res.append(c / s)
    return res


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser
# ----------------------------------------------------------------------
def _adam_optimize_softmax(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           lr: float = 0.02,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex with a smooth approximation of the max‑convolution.
    The temperature λ grows linearly from λ_min to λ_max.
    Returns the best iterate and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L1 = _next_pow2(2 * n - 1)          # for a*a
    L2 = _next_pow2(3 * n - 2)          # for w * rev(a)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    # annealing parameters for the soft‑max
    lam_min, lam_max = 0.5, 100.0

    while time.time() < deadline:
        it += 1

        # ----- self‑convolution (exact) ---------------------------------
        Af = np.fft.rfft(a, n=L1)
        conv = np.fft.irfft(Af * Af, n=L1)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- soft‑max weights -----------------------------------------
        progress = (time.time() - start_time) / (deadline - start_time)
        lam = lam_min + progress * (lam_max - lam_min)   # linear schedule
        # stable exponentiation
        w_raw = np.exp(lam * (conv - max_conv))
        w_sum = w_raw.sum()
        if w_sum == 0.0:               # should not happen, but guard against it
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_raw / w_sum

        # ----- gradient of the smoothed max -------------------------------
        rev_a = a[::-1]
        Wf = np.fft.rfft(w, n=L2)
        Afr = np.fft.rfft(rev_a, n=L2)
        tmp = np.fft.irfft(Wf * Afr, n=L2)[: 3 * n - 2]   # linear convolution

        # slice the central n entries (indices n‑1 … n‑1+n‑1)
        grad_max = 2.0 * tmp[n - 1:(n - 1) + n]          # g_i = 2·∑_k w_k·a_{k‑i}
        grad_obj = 2.0 * n * grad_max                    # derivative of 2·n·max_conv

        # ----- Adam step -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ---- project back onto the simplex ----
        a = _project_simplex(a)

        cur_score = 2.0 * n * max_conv       # denominator = 1 after projection
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑soft it={it:,} "
                  f"score={cur_score:.6f} λ={lam:.1f}")

        # occasional learning‑rate decay
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy stochastic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """Move a random packet of mass between two indices, accepting only improvements."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)               # prefer moving from heavier spots
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps_move = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps_move
        a_prop[dst] += eps_move

        new_score = _objective(a_prop)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget,
    uses warm‑starts, smoothed‑max Adam optimisation and a final
    hill‑climbing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two works nicely with the FFT.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Warm‑start – analytic candidates + a few random Dirichlet draws.
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # include a globally supplied construction if present
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_res = np.interp(xq, xs, h1)
            h1_res = np.maximum(h1_res, 0.0)
            s = h1_res.sum()
            if s > 0:
                warm_starts.append(h1_res / s)
    except Exception:
        pass

    # a few random Dirichlet initialisations for extra diversity
    rng = np.random.default_rng(seed + 12345)
    for _ in range(3):
        rand = rng.random(n)
        rand = np.maximum(rand, 0.0)
        rand /= rand.sum()
        warm_starts.append(rand)

    # pick the best warm‑start according to the exact objective
    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed‑max Adam optimisation (≈ 70 % of the remaining budget).
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + remaining * 0.70
        a_opt, score_opt = _adam_optimize_softmax(a0, adam_deadline, start_time,
                                                  lr=0.02)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑soft best score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Light hill‑climb polish for the rest of the budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final HillClimb score={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Final sanity checks and conversion to plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, n={n}")
    return [float(v) for v in a0]
