"""
Improved optimiser for the “step‑function” challenge.

The objective is 
    2 * n * max(conv(a,a)) / (sum(a)**2) 

which is scale‑invariant, so we normalise the vector to sum to 1 and
 minimise   2·n·max(conv(a,a)).  

The optimiser proceeds in four phases (all respecting the 1000 s budget)

1️⃣ **Broad parametric sweep** – geometric, power and beta families are
   examined on fine grids.  The best few candidates are kept.

2️⃣ **Random Dirichlet launches** – a larger number of random sparse /
   dense draws (Gamma concentration in [0.1,5]) to diversify the pool.

3️⃣ **Smooth‑max Adam descent** – we minimise the smooth approximation

        Sτ(a)= (1/τ)·log Σ exp(τ·conv(a,a))

  by Adam‑style optimisation.  τ starts very small (≈0.5) and is
  increased geometrically, turning the smooth average into the true
  max.  After each step the iterate is projected onto the probability
  simplex.

   Several restarts (uniform, best sweep, best Dirichlet, optional
   global starter) are run in round‑robin until the allotted time is
   exhausted.

4️⃣ **Aggressive hill‑climb** – we repeatedly move a tiny fraction of
   mass from the index with the largest positive sub‑gradient to the
   index with the most negative one.  Occasionally a larger random
   perturbation is tried and non‑improving moves are accepted with a
   very low probability (simulated‑annealing style) to escape shallow
   local minima.

All convolutions use FFT (full length) for speed, the simplex projection
is the standard O(n) algorithm and the code is written with only
numpy, scipy and the provided ``evaluate_sequence`` in mind.

The routine returns a plain list of Python floats that can be fed to
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog  # not used directly – kept for possible extensions

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT based for speed)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real → real)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Self‑convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, a_i ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                           # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, tau: float, L_self: int, L_grad: int) -> np.ndarray:
    """
    Compute ∇ Sτ(a) where Sτ is the smooth approximation of max(conv).
    Returns a vector of length n.
    """
    n = a.shape[0]

    # self‑convolution and soft‑max weights
    conv = _conv_self(a, L_self)
    max_c = conv.max()
    w = np.exp(tau * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:            # extreme numerical case – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum

    # gradient = 2 * (w * a_rev) restricted to the central n entries
    a_rev = a[::-1]
    conv_w = _conv_fft(w, a_rev)               # length = 3n‑2
    grad = 2.0 * conv_w[n - 1: n - 1 + n]       # slice of length n
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max objective
# ----------------------------------------------------------------------
def _adam_descent(start_a: np.ndarray,
                  start_time: float,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  label: str = "adam") -> tuple[np.ndarray, float]:
    """
    Adam‑style descent on the smooth‑max objective.
    τ starts small (≈0.5) and is increased geometrically.
    The iterate is projected onto the probability simplex after each step.
    """
    a = start_a.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.04                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # τ schedule
    tau = 0.5
    tau_factor = 1.03
    tau_max = 1e6
    tau_update_interval = 1000

    # FFT lengths (pre‑computed for speed)
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    it = 0
    last_impr_it = 0
    while time.time() < deadline:
        it += 1

        # Current (non‑smooth) objective – only for bookkeeping
        conv = _conv_self(a, L_self)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_impr_it = it
            print(f"[{time.time() - start_time:.2f}s] {label} it={it:,} score={cur_score:.6f}")

        # Gradient of the smooth‑max surrogate
        grad = _smooth_max_gradient(a, tau, L_self, L_grad)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # τ update
        if it % tau_update_interval == 0:
            tau = min(tau * tau_factor, tau_max)

        # simple early‑stop if no improvement for a long while
        if it - last_impr_it > 20000 and tau > 1e4:
            # we have essentially converged to the true max; exit early
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Directed hill‑climb with occasional stochastic jumps
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained hill‑climb:
    - move a tiny fraction of mass from a source (largest positive sub‑gradient)
      to a destination (most negative sub‑gradient);
    - occasionally perform a larger random redistribution;
    - accept non‑improving moves with a vanishing probability.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    no_impr_counter = 0

    while time.time() < deadline:
        conv = _conv_self(a, L_self)
        grad = _subgradient_maxconv(a, conv)

        # deterministic tiny transfer
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            # fallback to a random pair
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # transfer amount (up to ~0.5 % of source mass)
        frac = rng.random() * 0.004 + 0.0005
        eps = a[src] * frac
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        prop_score = float(2.0 * n * _conv_self(a_prop, L_self).max())

        if prop_score < best_score:
            a = a_prop
            best_seq, best_score = a.copy(), prop_score
            no_impr_counter = 0
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved score={best_score:.6f}")
            continue

        # occasional large random jump (probability decays with time)
        if rng.random() < 0.001:
            # pick a random subset of indices and re‑distribute a tiny chunk
            k = rng.integers(2, 6)            # size of the subset
            idxs = rng.choice(n, size=k, replace=False)
            amounts = a[idxs] * rng.random(k) * 0.02
            total_move = amounts.sum()
            a_rand = a.copy()
            a_rand[idxs] -= amounts
            # add the mass uniformly to the rest of the vector
            a_rand += total_move / n
            a_rand = np.maximum(a_rand, 0.0)
            a_rand = _project_simplex(a_rand)
            rand_score = float(2.0 * n * _conv_self(a_rand, L_self).max())
            if rand_score < best_score:
                a = a_rand
                best_seq, best_score = a.copy(), rand_score
                no_impr_counter = 0
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb random jump score={best_score:.6f}")
                continue

        # occasionally accept a non‑improving move (simulated annealing spirit)
        temperature = max(0.0, 0.01 * (deadline - time.time()))  # vanishes near the end
        if temperature > 0.0 and rng.random() < np.exp(-(prop_score - best_score) / temperature):
            a = a_prop
            no_impr_counter += 1
        else:
            no_impr_counter += 1

        # early stop if stuck for a long period
        if no_impr_counter > 20000:
            break

    return best_seq, best_score

    print(f"[{time.time() - start_time:.2f}s] Starting optimiser, n={n}")

    # --------------------------------------------------------------
    # Helper to keep the current globally best candidate
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter is provided (height_sequence_1) use it
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float).ravel()
        if starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / s
            sc = _objective_score(starter)
            if sc < best_score:
                best_seq, best_score = starter.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric families sweep (geometric, power, beta)
    # --------------------------------------------------------------
    phase_deadline = deadline - 350.0      # reserve ~350 s for later phases
    if time.time() < phase_deadline:
        idx = np.arange(n, dtype=np.float64)

        # ---- geometric ----------------------------------------------------
        eps_grid = np.logspace(-5, -1, 400)
        for eps in eps_grid:
            if time.time() >= phase_deadline:
                break
            r = 1.0 - eps
            geo = np.power(r, idx)
            if geo.sum() == 0:
                continue
            geo /= geo.sum()
            for lam in np.linspace(0.0, 1.0, 11):
                seq = lam * uniform + (1.0 - lam) * geo
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

        # ---- power ---------------------------------------------------------
        p_grid = np.linspace(-2.0, 2.0, 121)
        for p in p_grid:
            if time.time() >= phase_deadline:
                break
            pow_seq = np.power(idx + 1.0, p)
            s = pow_seq.sum()
            if s == 0.0:
                continue
            pow_seq /= s
            sc = _objective_score(pow_seq)
            if sc < best_score:
                best_score, best_seq = sc, pow_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

        # ---- beta‑type ------------------------------------------------------
        alphas = np.linspace(0.05, 4.0, 81)
        betas = alphas.copy()
        log_idx = np.log(idx + 1.0)
        log_rev = np.log(np.maximum(n - idx, 1))
        for a in alphas:
            if time.time() >= phase_deadline:
                break
            log_a = (a - 1.0) * log_idx
            for b in betas:
                if time.time() >= phase_deadline:
                    break
                log_seq = log_a + (b - 1.0) * log_rev
                # safe exponentiation
                seq = np.exp(log_seq - np.max(log_seq))
                s = seq.sum()
                if s == 0.0 or not np.isfinite(s):
                    continue
                seq /= s
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (more extensive than before)
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        rng = np.random.default_rng()
        dirichlet_trials = 120                     # increased number of trials
        for _ in range(dirichlet_trials):
            if time.time() >= phase_deadline:
                break
            conc = rng.uniform(0.1, 5.0)
            a = rng.gamma(shape=conc, scale=1.0, size=n)
            a /= a.sum()
            sc = _objective_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Dirichlet conc={conc:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # Gather a few promising starting points for the Adam phase
    # --------------------------------------------------------------
    start_candidates = [uniform, best_seq]
    try:
        if height_sequence_1 is not None:
            cand = np.asarray(height_sequence_1, dtype=float).ravel()
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand = cand / cand_sum
                start_candidates.append(cand)
    except Exception:
        pass

    # Remove duplicates (within tolerance)
    uniq_starts = []
    for s in start_candidates:
        duplicate = any(np.allclose(s, u, atol=1e-12) for u in uniq_starts)
        if not duplicate:
            uniq_starts.append(s)

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam descent (multiple restarts)
    # --------------------------------------------------------------
    # Reserve around 200 s for the final hill‑climb
    hill_deadline = deadline - 200.0
    if time.time() < hill_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam descents ({len(uniq_starts)} restarts)")
        # split remaining time equally among restarts
        time_per_restart = (hill_deadline - time.time()) / len(uniq_starts)
        for idx_start, a0 in enumerate(uniq_starts):
            sub_deadline = time.time() + time_per_restart
            best_seq, best_score = _adam_descent(
                a0, start_time, sub_deadline, best_seq, best_score,
                label=f"Adam‑{idx_start}"
            )
            print(f"[{time.time() - start_time:.2f}s] After Adam‑{idx_start} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final aggressive hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, start_time, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
