"""
Propose a step‑function (non‑negative heights) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

1. **Rich initial pool** – generate a large variety of analytic shapes:
   * Uniform vector
   * Beta‑density shapes (including very small α)
   * Edge‑concentrated exponential and polynomial shapes
   * Random sparse Dirichlet vectors
   The best few (TOP_K) are kept.

2. **Targeted hill‑climbing** – starting from the current best candidate,
   repeatedly identify the lag `k_max` at which the auto‑convolution is
   maximal.  For this lag the contribution of each coordinate is
   `c_i = a_i * a_{k_max-i}`.  A move transfers a small amount of mass from
   an index with large contribution to one with a small contribution.
   The move is accepted if it improves the objective, otherwise with a
   temperature‑controlled Metropolis rule.  This focuses the search on the
   most critical part of the objective rather than on random coordinates.

3. **Smooth‑max polishing** – the final candidate is refined by Adam‑style
   gradient descent on a soft‑max surrogate of `max(conv)`.  The temperature
   of the soft‑max is slowly annealed to the true max.

All stages respect the overall time budget (`budget_s`).  The result is
projected onto the probability simplex and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exp_shape(n: int, beta: float) -> np.ndarray:
    """
    Edge‑concentrated shape: a_i ∝ exp(-β·i) + exp(-β·(n‑1‑i)), normalised.
    """
    idx = np.arange(n, dtype=float)
    shape = np.exp(-beta * idx) + np.exp(-beta * (n - 1 - idx))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_edge_power_shape(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i/(n‑1))**p + ((n‑1‑i)/(n‑1))**p .  Symmetric, edges heavy for p>0.
    """
    i = np.arange(n, dtype=float)
    norm = (n - 1)
    left = (i / norm) ** p
    right = ((n - 1 - i) / norm) ** p
    shape = left + right
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_ramp(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric ramp with optional plateau (used in earlier code).
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Keeping the K best candidates
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert a (score,vec) pair into a sorted top‑k pool."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Partner array for a given convolution shift (used in targeted HC)
# ----------------------------------------------------------------------
def _partner_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """Return the partner vector such that partner[i] = a[k-i] (0 if out of range)."""
    n = a.shape[0]
    partner = np.zeros_like(a)
    if k < n:
        # indices 0 … k overlap
        partner[: k + 1] = a[k::-1]
    else:
        # indices k‑n+1 … n‑1 overlap
        start = k - n + 1
        # reversed slice from the end down to start
        partner[start:] = a[n - 1 : start - 1 : -1]
    return partner


# ----------------------------------------------------------------------
# Targeted hill‑climbing (mass transfers guided by the maximal lag)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy SA that always attacks the lag where the objective is tight."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    cur_score = best_score

    # temperature schedule (exponential)
    T_start = 0.2
    T_end = 1e-9
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * (T_end / T_start) ** frac

        conv = _conv_self(a)
        k_max = int(np.argmax(conv))

        # compute contributions c_i = a_i * a_{k_max-i}
        partner = _partner_for_shift(a, k_max)
        contrib = a * partner
        # source: index with largest contribution (must have a_i>0)
        src_candidates = np.flatnonzero(contrib > 0.0)
        if src_candidates.size == 0:
            # fall back to a random positive index
            src_candidates = np.flatnonzero(a > 1e-12)
            if src_candidates.size == 0:
                break
        src = src_candidates[rng.integers(src_candidates.size)]

        # destination: index with smallest partner value (fewest overlaps at k_max)
        # sample a few random targets and keep the best
        dst = src
        best_part = partner[src]
        for _ in range(5):
            cand = rng.integers(n)
            if cand == src:
                continue
            if partner[cand] < best_part:
                dst = cand
                best_part = partner[cand]

        if src == dst:
            continue

        # propose moving a small chunk of mass
        delta = a[src] * 0.12 * rng.random()   # at most ~12 % of source mass
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # small numerical fix
        if a_prop[src] < -1e-12:
            continue
        a_prop = np.maximum(a_prop, 0.0)   # protect against tiny negatives

        # re‑normalise to keep the sum exactly the same (sum unchanged by construction)
        new_score = _objective(a_prop)
        delta_score = new_score - cur_score

        if delta_score < 0.0 or rng.random() < math.exp(-delta_score / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HC iter {it:,} new best {new_score:.6f}")

        # occasional random perturbation to avoid dead‑ends
        if it % 50000 == 0:
            sigma = 0.004
            noisy = a + rng.normal(0.0, sigma, size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                noisy /= s
                a = noisy
                cur_score = _objective(a)

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (as in the previous code, but runs as long as time permits)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth surrogate of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    step0 = 0.06          # learning‑rate base
    T0 = 0.03             # initial soft‑max temperature
    T_end = 1e-5

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter {it:,} score {cur_score:.6f}")

        # temperature schedule (linear freeze)
        frac = it / (it + 1500.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights over the convolution vector
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of the smooth‑max w.r.t. a (derived from correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # slice of length n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex to keep a feasible
        a = _project_simplex(a)

    return best, best_score

            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0.0:
                starter_up /= starter_up.sum()
                _push_top(top, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Targeted hill‑climbing on the current best vector
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 200.0:
        hc_deadline = time.time() + remaining * 0.55
        best_seq, best_score = _targeted_hill_climb(
            best_seq, hc_deadline, start_time, rng
        )
        print(f"[{time.time() - start_time:.2f}s] After HC best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Smooth‑max polishing (Adam) with the remaining budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        gd_deadline = time.time() + remaining
        refined_seq, refined_score = _smooth_gradient_descent(
            best_seq, gd_deadline, start_time
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(f"[{time.time() - start_time:.2f}s] After SmoothGD best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
