"""
Propose a low‑objective step‑function.

The algorithm builds on the previous hill‑climber but adds two
independent improvements that help to escape local minima and
focus the search on the most harmful coefficients:

* **Gradient‑directed moves** – the current maximal convolution index
  `k*` is identified; the derivative of the objective w.r.t. each
  coefficient is `g_i = 2·a_{k*‑i}` (zero outside the support).  Mass is
  preferentially taken from indices with a large product `a_i·g_i`
  (they contribute a lot to the peak) and moved to indices where the
  gradient is small.

* **Simulated‑annealing acceptance** – uphill moves are occasionally
  accepted with probability `exp(-(Δ)/T)`, where the temperature `T`
  cools quadratically to zero over the allotted time.  This lets the
  search cross small barriers and find better basins.

All moves are realised with a *linear* O(n) update of the full
self‑convolution, exactly as in the reference implementation, so
millions of iterations still fit into the 1000‑second budget.

The routine works for any length `n`; if the global variable
`height_sequence_1` exists it is used as a starting point, otherwise a
default length of 4096 is chosen.  The returned list is normalised so
that `sum(a)=1`, which makes the objective equal to `2·n·max(conv)`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length) via a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (sum may be any positive number)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer `eps` mass from index `src` to `dst`.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Coefficient vector – modified in place.
    conv : np.ndarray, shape (2n‑1,)
        Current self‑convolution – modified in place.
    src, dst : int
        Source and destination indices (src ≠ dst).
    eps : float
        Amount to move (0 ≤ eps ≤ a[src]).
    """
    n = a.shape[0]

    # linear term: 2·a*Δ, Δ = eps*(e_dst – e_src)
    conv[src: src + n] -= 2.0 * eps * a
    conv[dst: dst + n] += 2.0 * eps * a

    # quadratic term Δ·Δ
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # update the probability vector
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Candidate analytic shapes (all produce a non‑negative vector of length n)
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of analytic shapes (unnormalised)."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]
    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Exponential decay from the centre (symmetric)
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Half‑cosine
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussians of different widths
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet‑like draws
    rng = np.random.default_rng(0)
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # Normalise and keep only those with positive mass
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised
