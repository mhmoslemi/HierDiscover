"""
Propose a step‑function (non‑negative list) that minimises  

        2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
1️⃣  Initialise on the best of several cheap parametric families
    (geometric+uniform, power‑law / Beta‑like shapes) – a dense grid
    search on the simplex (all candidates are normalised to sum=1).

2️⃣  Refine the best candidate with a **continuation Adam optimiser**:
    * The objective  max(conv) is smoothed by a soft‑max with a
      temperature β that is increased linearly during the run.
    * The gradient of the smoothed objective is a *weighted cross‑correlation*
      which is computed efficiently with FFTs.
    * After each Adam step we project back onto the probability simplex.
      A tiny Gaussian shake is added when progress stalls.

3️⃣  Finish with a stochastic hill‑climb that transfers a tiny random
    amount of mass between two indices and keeps improving moves.

The whole search respects the 1000 s budget, uses only one CPU and
relies on NumPy (no external optimisation libraries).  The returned
list is a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    a is assumed to be a real vector.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Scoring helper (objective = 2·n·max(conv)  when sum(a)=1)
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, L_fft: int) -> float:
    """Return the quantity 2·n·max(conv(a,a)) – a is assumed normalized."""
    n = a.shape[0]
    conv = _conv_self(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric sweep over a few families
# ----------------------------------------------------------------------
def _grid_search(n: int,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float,
                 uniform: np.ndarray,
                 L_fft: int) -> tuple[np.ndarray, float]:
    """
    Explore a collection of low‑dimensional families:
        * geometric + uniform (already used in the reference)
        * power‑law / Beta‑like shapes (a_i ∝ (i+1)^p (n-i)^q)
    All candidates are normalised to sum‑1.
    """
    idx = np.arange(n, dtype=np.float64)

    # ----- 1️⃣  geometric + uniform -------------------------------------------------
    eps_grid = np.logspace(-5.5, -2.0, 500)      # 3·10⁻⁶ … 1·10⁻²
    lam_grid = np.arange(0.0, 1.001, 0.02)       # mixing weight of the uniform part

    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0.0:
            continue
        geo /= geo_sum                               # normalised geometric
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            cur = _score_of(seq, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] geo‑grid eps={eps:.1e} λ={lam:.2f} score={cur:.6f}")

    # ----- 2️⃣  power‑law / beta‑like shapes ---------------------------------------
    # a_i ∝ (i+1)^p * (n-i)^q   with p,q possibly negative (U‑shaped)
    p_grid = np.arange(-2.0, 2.01, 0.25)   # 17 values
    q_grid = np.arange(-2.0, 2.01, 0.25)   # 17 values
    lam_grid2 = np.arange(0.0, 1.001, 0.05)  # mix with uniform

    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        ipow = np.power(idx + 1.0, p)                # (i+1)^p
        for q in q_grid:
            shape = ipow * np.power(n - idx, q)      # (i+1)^p * (n-i)^q
            shape_sum = shape.sum()
            if shape_sum == 0.0:
                continue
            shape /= shape_sum
            for lam in lam_grid2:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                cur = _score_of(seq, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] pow‑shape p={p:.2f} q={q:.2f} λ={lam:.2f} score={cur:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (soft‑max on the convolution)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          start_time: float,
                          deadline: float,
                          best_score_initial: float,
                          L_fft: int,
                          beta_start: float = 1.0,
                          beta_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a *soft‑max* approximation of max(conv).
    The temperature β grows linearly from beta_start to beta_end during the run.
    Returns the best sequence seen and its (hard) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]

    # FFT size for the convolution (conv length = 2n-1)
    L_conv = _next_pow2(2 * n - 1)
    # FFT size for the correlation (len(w)+len(a)-1 = (2n-1)+n-1 = 3n-2)
    L_corr = _next_pow2(3 * n - 2)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15                         # a bit more gentle than the reference
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # ---- current convolution and hard score ---------------------------------
        conv = _conv_self(a, L_conv)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # ---- soft‑max weights ----------------------------------------------------
        progress = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + (beta_end - beta_start) * progress
        # stability: subtract max before exponentiation
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum                     # now Σ w = 1

        # ---- gradient = 2 * (w ⋆ a)  (cross‑correlation) ------------------------
        # compute linear cross‑correlation via FFTs
        W_fft = np.fft.rfft(w, n=L_corr)
        A_fft = np.fft.rfft(a, n=L_corr)
        corr_full = np.fft.irfft(W_fft * np.conj(A_fft), n=L_corr)
        # linear correlation length = len(w) + len(a) - 1 = 3n-2
        corr_lin = corr_full[: (2 * n - 1) + n - 1]   # ≡ 3n-2
        # grad[i] = 2 * corr_lin[(n-1) - i]   for i=0..n-1
        grad = 2.0 * corr_lin[(n - 1) - np.arange(n)]

        # ---- Adam update ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- projection onto the simplex -----------------------------------------
        a = _project_simplex(a)

        # ---- plateau handling: small random shake --------------------------------
        if it - last_improve > 8000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: move a tiny random fraction of mass
    from one index to another; keep the move only if it improves the *hard*
    objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _score_of(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 1 % of the source mass
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = _score_of(a_prop, L_fft)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The search consists of:
        * a fast grid sweep over several parametric families,
        * a continuation Adam optimiser with a soft‑max surrogate,
        * a final stochastic hill‑climb.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # leave a small safety margin for the final I/O
    deadline = start_time + budget_s - 0.4

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise (try to reuse the best sequence from the previous run)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score_of(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _score_of(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Grid search over several families
    # ------------------------------------------------------------------
    best_seq, best_score = _grid_search(
        n=n,
        deadline=deadline,
        best_seq=best_seq,
        best_score=best_score,
        uniform=uniform,
        L_fft=L_fft,
    )

    # ------------------------------------------------------------------
    # 2️⃣  Continuation Adam optimiser (soft‑max surrogate)
    # ------------------------------------------------------------------
    # allocate most of the remaining budget to Adam; keep ~120 s for hill‑climb
    hill_time = 120.0
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed Adam optimiser")
        a_opt, sc_opt = _smooth_adam_optimize(
            a0=best_seq,
            start_time=start_time,
            deadline=adam_deadline,
            best_score_initial=best_score,
            L_fft=L_fft,
            beta_start=1.0,
            beta_end=250.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Smoothed Adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
