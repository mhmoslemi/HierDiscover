"""
Propose a low‑objective step‑function.

The algorithm consists of three phases

1️⃣ Warm‑start – a collection of analytic shapes (tent, exponential,
   power‑law, inverse‑sqrt, Gaussian, random Dirichlet draws) is generated.
   All candidates are normalised to sum = 1 and the best one (smallest exact
   objective) is kept as the initial vector.

2️⃣ Gradient‑guided hill‑climbing with simulated‑annealing.
   *   At each iteration the index `k` where the self‑convolution `a*a`
       attains its maximum is identified.
   *   For that `k` we form the “mirror’’ vector `w_i = a_{k‑i}` (zero outside
       the valid range) and the contribution vector `p_i = a_i * w_i`.
       Large values of `p_i` are the pairs that contribute most to the peak.
   *   A source index `src` is chosen among the indices with the largest
       `p_i` (and positive mass) and a destination `dst` is chosen where the
       mirror value `w_i` is smallest (preferably zero).  Moving a small
       packet of mass `eps` from `src` to `dst` reduces the contribution to the
       current peak.
   *   The move is applied using an O(n) incremental update of the
       convolution.  The move is accepted if it does not increase the
       objective, otherwise it may be accepted with a temperature–dependent
       probability (simulated annealing).  The temperature decreases linearly
       with the remaining time.
   *   Whenever a globally better vector is found it is stored.

3️⃣ After the time budget (≈ 1000 s) the best vector is normalised once more
   and returned as a plain Python list ready for ``evaluate_sequence``.

This strategy improves over the previous random‑source hill‑climber by
directing mass transfers from the pairs that dominate the current maximal
convolution value, and by allowing occasional uphill moves to escape local
minima.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum≈1)."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`
    (in–place updates of both the vector and its convolution).
    """
    n = a.shape[0]

    # linear part:  conv += 2*eps*(a after the move) shifted
    conv[src : src + n] -= 2.0 * eps * a   # subtract contribution of src
    conv[dst : dst + n] += 2.0 * eps * a   # add contribution of dst

    # quadratic correction for the three indices that are affected twice
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # finally move the mass
    a[src] -= eps
    a[dst] += eps

# Candidate shape generators (all produce a non‑negative vector of length n)
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Generate a diverse set of analytic (and random) shapes."""
    rnd = np.random.default_rng(0)
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Symmetric exponentials
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Half‑cosine (already used as a warm‑start)
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Random Dirichlet draws (moderate concentration)
    for _ in range(8):
        cand = rnd.random(n)
        candidates.append(cand)

    # 8) Sparse Dirichlet draws (small concentration → spikier vectors)
    for _ in range(5):
        alpha = rnd.uniform(0.05, 0.3)
        cand = rnd.gamma(alpha, 1.0, size=n)
        candidates.append(cand)

    # Normalise and keep only non‑empty vectors
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)

    return normalised


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a probability vector that minimises
    2·n·max(conv(a,a)) / (sum(a)²).

    Returns a plain list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the external harness
    deadline = start_time + budget_s - 0.5
    rng = np.random.default_rng(seed + 1)

    # --------------------------------------------------------------
    # Determine problem size (use the previously discovered vector if present)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)                     # type: ignore[name-defined]
    except Exception:
        n = 4096

    # --------------------------------------------------------------
    # 1) Warm‑start – pick the best among many analytic shapes.
    # --------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # include the publicly supplied starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm_starts.append(starter / starter.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    # sanity fallback
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Prepare for hill‑climbing
    # --------------------------------------------------------------
    a = best_vec.copy()                # current working vector
    conv = _conv_fft(a)                # its convolution (full length)
    total = a.sum()
    assert total > 0.0
    factor = 2.0 * n / (total * total)   # constant s.t. objective = factor * conv.max()
    cur_score = factor * conv.max()

    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score = {best_score:.6f}")

    it = 0
    last_improve_it = 0

    # temperature settings for simulated annealing
    T0 = 0.01                       # initial temperature (objective scale ≈ 1–2)
    total_budget = deadline - start_time

    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # locate a (possibly non‑unique) maximal index of the convolution
        # ----------------------------------------------------------
        max_val = conv.max()
        max_idxs = np.flatnonzero(conv == max_val)
        k = int(rng.choice(max_idxs))          # pick one of the ties at random

        # ----------------------------------------------------------
        # Build the mirror vector w_i = a_{k-i} (zero outside range)
        # ----------------------------------------------------------
        w = np.zeros(n, dtype=float)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        w[idx] = a[k - idx]

        # ----------------------------------------------------------
        # Contribution to the current peak: p_i = a_i * w_i
        # ----------------------------------------------------------
        p = np.zeros(n, dtype=float)
        p[idx] = a[idx] * w[idx]

        # ----------------------------------------------------------
        # Choose a source index – one that really contributes to the peak
        # ----------------------------------------------------------
        src_mask = (a > 1e-14) & (p > 0.0)
        if src_mask.any():
            src_candidates = np.where(src_mask)[0]
            src = int(src_candidates[np.argmax(p[src_candidates])])
        else:
            # fallback: random source weighted by current mass
            src = int(rng.choice(n, p=a / a.sum()))

        # ----------------------------------------------------------
        # Choose a destination – preferably where w is *very* small
        # ----------------------------------------------------------
        zero_mask = (w == 0.0)
        if zero_mask.any():
            dst_candidates = np.where(zero_mask)[0]
            # avoid picking the source itself
            if src in dst_candidates:
                dst_candidates = dst_candidates[dst_candidates != src]
            if dst_candidates.size == 0:
                # extremely unlikely, pick a random different index
                dst = src
                while dst == src:
                    dst = rng.integers(n)
            else:
                dst = int(rng.choice(dst_candidates))
        else:
            # all w positive – pick an index with the smallest w
            inv = (w.max() - w) + 1e-12   # larger weight for smaller w
            inv[src] = 0.0                # never move mass back to the source itself
            dst = int(rng.choice(n, p=inv / inv.sum()))

        if src == dst or a[src] < 1e-15:
            continue

        # ----------------------------------------------------------
        # Determine a step size (fraction of the source mass)
        # ----------------------------------------------------------
        remaining = deadline - time.time()
        # large steps early, shrinking to a few percent near the end
        frac_max = 0.35 * (remaining / total_budget) + 0.02
        eps = a[src] * rng.random() * frac_max
        if eps <= 0.0:
            continue

        # ----------------------------------------------------------
        # Evaluate the move
        # ----------------------------------------------------------
        cur_obj = factor * conv.max()          # current objective

        _apply_move(a, conv, src, dst, eps)    # tentative move
        new_obj = factor * conv.max()          # objective after the move

        # simulated‑annealing acceptance
        accept = False
        if new_obj <= cur_obj:
            accept = True
        else:
            # temperature linearly decreasing
            T = T0 * (remaining / total_budget)
            prob = math.exp(-(new_obj - cur_obj) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # keep the move – update best record if we improve globally
            if new_obj < best_score - 1e-12:
                best_score = new_obj
                best_vec = a.copy()
                last_improve_it = it
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # revert the change
            _apply_move(a, conv, dst, src, eps)

        # occasional cheap restart if we have been stuck for a long while
        if it - last_improve_it > 200_000:
            # perturb the best vector a little and continue from there
            perturb = rng.normal(scale=0.01, size=n)
            a = best_vec + perturb
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s
            conv = _conv_fft(a)
            total = a.sum()
            factor = 2.0 * n / (total * total)
            last_improve_it = it

    # --------------------------------------------------------------
    # 3) Final normalisation and conversion to plain list.
    # --------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score = {best_score:.6f}")
    return [float(v) for v in best_vec]
