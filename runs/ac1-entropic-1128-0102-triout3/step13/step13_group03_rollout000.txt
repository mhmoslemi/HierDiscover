"""
Propose a non‑negative step function (vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm (≈1000 s budget)

1. **Coarse symmetric sweep (n₀≈1024).**  
   Explore families that are symmetric around the centre:
      – exponential decay  a_i ∝ exp(-c·|i‑mid|)  
      – polynomial decay   a_i ∝ ((i+1)(n‑i))^p  
   For each shape also try a convex mixture with the uniform vector.
   Keep the best shape.

2. **Upsample.**  
   Interpolate the best small shape to a larger size (n≈4096) – this gives a
   good “soft‑starter” for the fine optimisation.

3. **Smooth‑max Adam descent** (projected onto the simplex).  
   Optimise a differentiable surrogate
        S_β(a)= (1/β)·log Σ_k exp(β·(a*a)_k)
   with a geometric schedule β↗800.  Several restarts with tiny random
   perturbations improve robustness.

4. **Sub‑gradient descent on the true max‑conv.**  
   Using the exact max‑convolution index we compute the (sub‑)gradient
   g_i = 2·a_{k‑i} (if the pair (i,k‑i) exists) and perform a
   diminishing‑step projected gradient descent.  This often reduces the
   max‑entry further after the smooth phase.

5. **Targeted stochastic hill‑climb.**  
   Random mass‑transfer moves combined with “attacks” that remove mass
   from the two indices that realise the current maximal convolution entry.
   Occasionally accept a worsening move according to a simple annealing
   schedule to escape local minima.

6. **Return the best vector (normalised to sum = 1) as a plain list.**
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {w≥0, Σw=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_obj(a: np.ndarray, L_fft: int = None) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    Assumes a is non‑negative.
    """
    n = a.shape[0]
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max surrogate utilities (used inside Adam descent)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    """
    conv = _conv_self(a, L=L_conv)        # length 2n‑1
    max_c = conv.max()
    shifted = conv - max_c                # make max = 0 for stability
    exp_vals = np.exp(beta * shifted)     # exp may underflow for large negative, ok
    w = exp_vals / exp_vals.sum()         # soft‑max weights, shape (2n‑1,)

    # Correlation w * a_rev → gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)  # linear convolution w * a_rev
    start = a.shape[0] - 1                # offset = n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 800.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected descent on the smooth‑max surrogate.
    β is increased geometrically up to beta_max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0
    beta_mul = 1.02

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1

        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth descent it={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.5 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # occasional jitter to avoid plateaus
        if t % 3000 == 0:
            jitter = 1e-4 * np.random.default_rng().standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (no smoothing)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on
        f(a) = max_k (a*a)_k   (denominator constant = sum(a)^2)
    Uses decreasing step size η_t = η0 / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    eta0 = 0.5  # initial step magnitude

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L=L_fft)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of the max entry
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # indices i where j = k - i is valid
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        j_range = k - i_range
        grad[i_range] = 2.0 * a[j_range]

        # step size schedule
        eta = eta0 / math.sqrt(t)

        a_next = a - eta * grad
        a_next = _project_simplex(a_next)

        # evaluate
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_next.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑grad it={t:,} score={cur_score:.6f}"
            )

        a = a_next

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb with simulated annealing
# ----------------------------------------------------------------------
def _hill_climb(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves (+ targeted attacks) with a simple annealing
    acceptance rule.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_obj(a, L_fft)
    n = a.shape[0]
    rng = np.random.default_rng()
    step_factor = 0.15               # up to 15 % of source mass per move
    min_factor = 1e-6
    decay = 0.995

    # temperature schedule: start high, decay to near 0
    T0 = 1e-2
    T_min = 1e-7
    total_seconds = deadline - start_time
    while time.time() < deadline:
        # linear temperature decay
        elapsed = time.time() - start_time
        T = max(T_min, T0 * (1 - elapsed / total_seconds))

        # decide move type
        if rng.random() < 0.6:
            # random transfer
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # targeted transfer (remove from the pair that realises max conv)
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_obj(a_prop, L_fft)
        improve = new_score < best_score
        if improve or rng.random() < math.exp(-(new_score - best_score) / max(T, 1e-12)):
            a = a_prop
            if improve:
                best_a = a.copy()
                best_score = new_score
                step_factor = max(step_factor, 0.15)   # reset step size after success
                print(
                    f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
                )
            else:
                # on rejected (but accepted by annealing) moves shrink step a bit
                step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric sweep (symmetric families)
# ----------------------------------------------------------------------
def _parametric_sweep_small(
    n_small: int,
    deadline: float,
    start_time: float,
) -> np.ndarray:
    """
    Search the space of symmetric shapes for n_small.
    Returns the best vector (sum normalised to 1).
    """
    best_seq = np.full(n_small, 1.0 / n_small)
    best_score = _eval_obj(best_seq)

    idx = np.arange(n_small, dtype=np.float64)
    mid = (n_small - 1) / 2.0
    abs_dist = np.abs(idx - mid)

    L_fft = _next_pow2(2 * n_small - 1)

    # exponential decay grid
    c_grid = np.logspace(-3, 0.7, num=40)   # exp(-c*|i-mid|)
    # polynomial exponent grid
    p_grid = np.linspace(-2.0, 2.0, num=41)

    lam_grid = (0.0, 0.2, 0.5, 0.8, 1.0)   # mix with uniform

    for c in c_grid:
        if time.time() >= deadline - 120:
            break
        exp_shape = np.exp(-c * abs_dist)
        exp_shape /= exp_shape.sum()
        for lam in lam_grid:
            seq = lam * best_seq + (1.0 - lam) * exp_shape
            seq /= seq.sum()
            sc = _eval_obj(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep exp c={c:.4f} λ={lam:.2f} score={sc:.6f}"
                )

    for p in p_grid:
        if time.time() >= deadline - 120:
            break
        # symmetric power law: ((i+1)*(n-i))^p
        poly_shape = np.power((idx + 1) * (n_small - idx), p)
        poly_sum = poly_shape.sum()
        if poly_sum == 0.0:
            continue
        poly_shape /= poly_sum
        for lam in lam_grid:
            seq = lam * best_seq + (1.0 - lam) * poly_shape
            seq /= seq.sum()
            sc = _eval_obj(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep poly p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                )

    return best_seq


def _upsample_vector(v_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation from len(v_small) to n_large and renormalisation.
    """
    x_small = np.linspace(0.0, 1.0, v_small.shape[0])
    x_large = np.linspace(0.0, 1.0, n_large)
    v_large = np.interp(x_large, x_small, v_small)
    v_large = np.maximum(v_large, 0.0)
    total = v_large.sum()
    if total > 0.0:
        v_large /= total
    else:
        v_large = np.full(n_large, 1.0 / n_large)
    return v_large


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises the given evaluation
    function.  Returns a plain Python list of floats (the heights).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Small symmetric sweep (n≈1024)
    # ------------------------------------------------------------------
    n_small = 1024
    print(f"[{time.time() - start_time:.2f}s] start small sweep n={n_small}")
    best_small = _parametric_sweep_small(n_small, deadline, start_time)
    best_small_score = _eval_obj(best_small)
    print(
        f"[{time.time() - start_time:.2f}s] best small score={best_small_score:.6f}"
    )

    # ------------------------------------------------------------------
    # 2️⃣  Upsample to a larger problem (n≈4096)
    # ------------------------------------------------------------------
    n_large = 4096
    best_large = _upsample_vector(best_small, n_large)
    best_large_score = _eval_obj(best_large)
    print(
        f"[{time.time() - start_time:.2f}s] after upsample n={n_large} score={best_large_score:.6f}"
    )

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam descent (multiple restarts)
    # ------------------------------------------------------------------
    descent_seeds = [seed, seed + 100, seed + 200]
    for ds in descent_seeds:
        if time.time() >= deadline - 120:
            break
        a0 = best_large.copy()
        # tiny random perturbation of magnitude 1e-4 relative
        a0 += 1e-4 * rng.standard_normal(n_large) * a0
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 30000
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,
            start_time,
            best_large_score,
            beta_max=800.0,
        )
        if sc_opt < best_large_score:
            best_large_score = sc_opt
            best_large = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth descent seed={ds} new best score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  Sub‑gradient descent on the exact max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        a_opt, sc_opt = _subgradient_descent(
            best_large,
            max_iters=20000,
            deadline=deadline - 30,
            start_time=start_time,
            best_score_initial=best_large_score,
        )
        if sc_opt < best_large_score:
            best_large_score = sc_opt
            best_large = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑gradient improvement score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # 5️⃣  Final stochastic hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        L_fft = _next_pow2(2 * n_large - 1)
        a_opt, sc_opt = _hill_climb(best_large, deadline, start_time, L_fft)
        if sc_opt < best_large_score:
            best_large_score = sc_opt
            best_large = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_large = np.maximum(best_large, 0.0)
    total = best_large.sum()
    if total > 0.0:
        best_large /= total
    else:
        best_large = np.full(n_large, 1.0 / n_large)

    # sanity check with the official evaluator (optional)
    try:
        true_score = evaluate_sequence(best_large.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(
        f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_large_score:.6f}"
    )
    # Return as plain Python list
    return [float(x) for x in best_large]
