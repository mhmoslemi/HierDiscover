"""
Improved search for a step‑function that minimises

        2 · n · max(conv(a,a)) / (sum(a)**2)

The algorithm mixes several ideas:

1️⃣  A fast parametric sweep over three families of non‑negative
    sequences – geometric, linear (triangular) and power‑law – each
    mixed with the uniform distribution.  The sweep uses a dense
    logarithmic grid for the geometric decay and a fine grid for the
    mixing weight λ.

2️⃣  A refinement stage that zooms into the neighbourhood of the
    best parameters found in the sweep (smaller step sizes and a
    tighter ε‑range).

3️⃣  Several short Adam‑style projected optimisation runs started
    from the current best sequence and from random Dirichlet
    initialisations.  The optimiser uses the exact sub‑gradient of
    the *active* convolution shift and Euclidean projection onto the
    probability simplex.

4️⃣  A final stochastic hill‑climb that moves up to 10 % of the mass
    from one index to another and accepts any improvement.  The
    climb is run until the time budget is exhausted.

All stages respect a global deadline (default 1000 s) and print
progress reports.  The function returns a plain Python list ready
for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σ w = 1 }.
    Returns w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient for the active shift
# ----------------------------------------------------------------------
def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) obtained from the single shift that
    attains the maximum.
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Helper to evaluate the objective (sum(a) is assumed to be 1)
# ----------------------------------------------------------------------
def _score_from_seq(seq: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(seq,seq))."""
    n = seq.shape[0]
    return float(2.0 * n * _conv_self(seq, L_fft).max())


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------
def _geom_shape(n: int, eps: float) -> np.ndarray:
    """Geometric shape a_i ∝ (1‑eps)^i, normalised to sum = 1."""
    r = 1.0 - eps
    geo = np.power(r, np.arange(n, dtype=np.float64))
    s = geo.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return geo / s


def _lin_shape(n: int) -> np.ndarray:
    """Linearly decreasing shape (triangular) that sums to 1."""
    # a_i = 2·(n‑i) / (n·(n+1))
    i = np.arange(n, dtype=np.float64)
    lin = 2.0 * (n - i) / (n * (n + 1))
    return lin


def _pow_shape(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^{‑p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    pw = 1.0 / np.power(idx, p)
    s = pw.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return pw / s


# ----------------------------------------------------------------------
# Adam optimiser (projected onto the simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score: float,
                   max_iter: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that uses the single‑active sub‑gradient.
    Returns the best sequence found and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = best_score
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.15

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # sub‑gradient of the active shift
        grad = _subgrad_one_active(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # occasional tiny random shake to escape plateaus
        if it - last_improve > 4_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climb (larger moves, greedy acceptance)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that transfers a random fraction (up to 10 %)
    of mass from one index to another and keeps any move that improves
    the true objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer up to 10 % of the source mass
        frac = rng.random() * 0.10 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that
    minimises 2·n·max(conv(a,a)) / (sum(a)**2).
    The routine combines a parametric sweep, a refinement pass,
    several Adam runs and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final hill‑climb
    deadline = start_time + budget_s - 0.4

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # start from uniform; keep best so far
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score_from_seq(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optional global starter (if the host supplied one)
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0:
                starter = starter / total
                cur = _score_from_seq(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep over three families
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    lam_grid = np.arange(0.0, 1.001, 0.01)

    # --- geometric family ------------------------------------------------
    eps_grid = np.logspace(-5, -2, 200)          # 10⁻⁵ … 10⁻²
    for eps in eps_grid:
        # early exit if we run out of time (reserve a few seconds)
        if time.time() >= deadline - 30:
            break
        geo = _geom_shape(n, eps)
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq = seq / seq.sum()
            cur = _score_from_seq(seq, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                best_eps, best_lam = eps, lam
                print(f"[{time.time() - start_time:.2f}s] Geom eps={eps:.1e} λ={lam:.2f} score={cur:.6f}")

    # --- linear (triangular) family --------------------------------------
    lin = _lin_shape(n)
    for lam in lam_grid:
        if time.time() >= deadline - 30:
            break
        seq = lam * uniform + (1.0 - lam) * lin
        seq = seq / seq.sum()
        cur = _score_from_seq(seq, L_fft)
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()
            best_shape = 'lin'
            best_lam = lam
            print(f"[{time.time() - start_time:.2f}s] Linear λ={lam:.2f} score={cur:.6f}")

    # --- power‑law family -----------------------------------------------
    p_grid = np.arange(0.2, 5.01, 0.2)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pw = _pow_shape(n, p)
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * pw
            seq = seq / seq.sum()
            cur = _score_from_seq(seq, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                best_shape = 'pow'
                best_p = p
                best_lam = lam
                print(f"[{time.time() - start_time:.2f}s] Pow p={p:.2f} λ={lam:.2f} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Local refinement around the best geometric or power‑law params
    # ------------------------------------------------------------------
    # The refinement only runs if we still have a comfortable margin.
    refine_deadline = deadline - 150.0   # keep ~150 s for later stages
    if time.time() < refine_deadline:
        try:
            shape_type = best_shape
        except NameError:
            shape_type = 'geom'   # fall‑back

        if shape_type == 'geom':
            # zoom into eps (±30 %) and λ (±0.05) with fine steps
            eps_center = best_eps
            eps_low = max(eps_center * 0.7, 1e-7)
            eps_high = eps_center * 1.3
            eps_fine = np.logspace(np.log10(eps_low), np.log10(eps_high), 200)
            lam_fine = np.arange(max(best_lam - 0.05, 0.0),
                                 min(best_lam + 0.05, 1.0) + 1e-12,
                                 0.001)
            for eps in eps_fine:
                if time.time() >= deadline - 30:
                    break
                geo = _geom_shape(n, eps)
                for lam in lam_fine:
                    seq = lam * uniform + (1.0 - lam) * geo
                    seq = seq / seq.sum()
                    cur = _score_from_seq(seq, L_fft)
                    if cur < best_score:
                        best_score = cur
                        best_seq = seq.copy()
                        best_eps, best_lam = eps, lam
                        print(f"[{time.time() - start_time:.2f}s] Refine geom eps={eps:.1e} λ={lam:.3f} score={cur:.6f}")

        elif shape_type == 'pow':
            # refine p (±20 %) and λ
            p_center = best_p
            p_low = max(p_center * 0.8, 0.05)
            p_high = p_center * 1.2
            p_fine = np.arange(p_low, p_high + 1e-12, 0.02)
            lam_fine = np.arange(max(best_lam - 0.05, 0.0),
                                 min(best_lam + 0.05, 1.0) + 1e-12,
                                 0.001)
            for p in p_fine:
                if time.time() >= deadline - 30:
                    break
                pw = _pow_shape(n, p)
                for lam in lam_fine:
                    seq = lam * uniform + (1.0 - lam) * pw
                    seq = seq / seq.sum()
                    cur = _score_from_seq(seq, L_fft)
                    if cur < best_score:
                        best_score = cur
                        best_seq = seq.copy()
                        best_p, best_lam = p, lam
                        print(f"[{time.time() - start_time:.2f}s] Refine pow p={p:.3f} λ={lam:.3f} score={cur:.6f}")

        # linear shape has only λ to refine – already covered by the coarse sweep

    # ------------------------------------------------------------------
    # 3️⃣  Multiple short Adam runs (including random restarts)
    # ------------------------------------------------------------------
    # Reserve about 150 s for the final hill‑climb
    adam_deadline = deadline - 150.0
    if time.time() < adam_deadline:
        # First run: start from the best sequence found so far
        print(f"[{time.time() - start_time:.2f}s] Adam start from best")
        a_opt, sc_opt = _adam_optimize(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam (base) improved score={sc_opt:.6f}")

        # Additional random‑initialised runs (up to the remaining time)
        rng = np.random.default_rng()
        extra_runs = 5
        for run in range(extra_runs):
            if time.time() >= adam_deadline:
                break
            # Dirichlet(1) gives a random point on the simplex
            a0 = rng.dirichlet(np.ones(n))
            print(f"[{time.time() - start_time:.2f}s] Adam random restart {run+1}")
            a_opt, sc_opt = _adam_optimize(a0, start_time, adam_deadline, best_score,
                                           max_iter=200_000)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam (restart {run+1}) improved score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb (large moves, greedy)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb improved score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
