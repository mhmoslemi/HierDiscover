"""
Propose a step‑function (non‑negative list) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  **Fine parametric sweep** – a dense golden‑section search of the
    one‑parameter geometric family `a_i ∝ (1‑ε)^i` mixed with the uniform
    distribution.  The sweep works on the true objective (sum = 1) and
    discovers the best `(ε, λ)` pair far more precisely than a plain grid.

2️⃣  **Randomised multi‑component search** – many random mixtures of two
    geometric components plus uniform are sampled (Dirichlet weights and
    log‑uniform decay rates).  This adds extra flexibility to the shape and
    often finds a tiny improvement over the single‑component optimum.

3️⃣  **Projected Adam optimiser & hill‑climb** – the best sequence from the
    previous steps is refined with an Adam optimiser (projected onto the
    probability simplex) and a final stochastic hill‑climb that transfers
    tiny fractions of mass between two positions.

All convolutions are performed with FFT‐based linear convolution
(`np.fft.rfft`) and the whole search respects a 1000 s budget.
The returned list is ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σ w = 1 }.
    Returns w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Helper to build geometric sequences
# ----------------------------------------------------------------------
def _geo_seq(eps: float, n: int) -> np.ndarray:
    """Return a length‑n geometric distribution with decay ε (r = 1‑ε) normalised to sum 1."""
    r = 1.0 - eps
    idx = np.arange(n, dtype=np.float64)
    geo = np.power(r, idx)
    s = geo.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return geo / s


def _mix_seq(lam: float, eps: float, n: int, uniform: np.ndarray) -> np.ndarray:
    """Mix uniform with a geometric component."""
    if lam <= 0.0:
        return _geo_seq(eps, n)
    if lam >= 1.0:
        return uniform.copy()
    geo = _geo_seq(eps, n)
    return lam * uniform + (1.0 - lam) * geo


# ----------------------------------------------------------------------
# Scoring utilities (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _score(seq: np.ndarray, n: int, L_fft: int) -> float:
    """Return the objective value for a sequence whose entries sum to 1."""
    conv = _conv_self(seq, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser (single‑active sub‑gradient)
# ----------------------------------------------------------------------
def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) obtained from **one** active shift:
    the shift with maximal convolution value.
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float,
                   max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Adam optimiser using the single‑active sub‑gradient."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        grad = _subgrad_one_active(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # occasional shake to escape plateaus
        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (micro‑mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float,
                n: int,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb that moves a tiny random fraction of mass."""
    a = start_a.copy()
    best_a = a.copy()
    conv = _conv_self(a, L_fft)
    best_score = float(2.0 * n * conv.max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # up to 1 % of source mass
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The procedure:
        1. Fine golden‑section search of a geometric + uniform family.
        2. Randomised mixtures of two geometric components + uniform.
        3. Projected Adam optimiser.
        4. Final hill‑climb.
    Returns a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final hill‑climb
    deadline = start_time + budget_s - 0.3

    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    uniform = np.full(n, 1.0 / n)

    # start from uniform
    best_seq = uniform.copy()
    best_score = _score(best_seq, n, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # incorporate a global starter if present
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur = _score(starter, n, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fine golden‑section sweep of (ε, λ) for geometric+uniform
    # ------------------------------------------------------------------
    eps_low = 1e-5          # smallest decay examined
    eps_high = 3e-3         # largest decay examined
    phi = (1.0 + np.sqrt(5.0)) / 2.0

    # lambda steps – a coarse grid, refined later around the best λ
    lam_grid = np.arange(0.0, 1.001, 0.02)

    for lam in lam_grid:
        # golden‑section search on ε for the current λ
        a = eps_low
        b = eps_high
        # initial interior points
        c = b - (b - a) / phi
        d = a + (b - a) / phi
        f_c = _score(_mix_seq(lam, c, n, uniform), n, L_fft)
        f_d = _score(_mix_seq(lam, d, n, uniform), n, L_fft)

        while time.time() < deadline and (b - a) > 1e-9:
            if f_c < f_d:
                b = d
                d = c
                f_d = f_c
                c = b - (b - a) / phi
                f_c = _score(_mix_seq(lam, c, n, uniform), n, L_fft)
            else:
                a = c
                c = d
                f_c = f_d
                d = a + (b - a) / phi
                f_d = _score(_mix_seq(lam, d, n, uniform), n, L_fft)

        eps_opt = (a + b) / 2.0
        # evaluate both interior points to pick the better one
        if f_c < f_d:
            cur_score = f_c
            cur_seq = _mix_seq(lam, c, n, uniform)
        else:
            cur_score = f_d
            cur_seq = _mix_seq(lam, d, n, uniform)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] λ={lam:.2f} eps={eps_opt:.2e} score={cur_score:.6f}")

        # stop early if we are out of time
        if time.time() >= deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Random search over mixtures of two geometric components + uniform
    # ------------------------------------------------------------------
    # allocate the remaining time between random search and Adam
    random_deadline = deadline - 200.0  # leave ~200 s for Adam & hill‑climb
    while time.time() < random_deadline:
        # sample decay rates (log‑uniform)
        eps1 = 10 ** rng.uniform(-5, -2)   # ≈ 1e‑5 … 1e‑2
        eps2 = 10 ** rng.uniform(-5, -2)
        # Dirichlet weights for [geo1, geo2, uniform]

        seq = (w1 * _geo_seq(eps1, n) +
               w2 * _geo_seq(eps2, n) +
               w_u * uniform)
        cur_score = _score(seq, n, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] rand eps1={eps1:.2e} eps2={eps2:.2e} "
                  f"w1={w1:.2f} w2={w2:.2f} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected Adam optimiser (single‑active sub‑gradient)
    # ------------------------------------------------------------------
    adam_deadline = deadline - 150.0
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        a_opt, sc_opt = _adam_optimize(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline, n, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
