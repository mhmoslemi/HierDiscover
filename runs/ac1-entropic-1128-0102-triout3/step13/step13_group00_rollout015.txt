"""
Optimiser for the step‑function challenge using Adam on a smooth‑max
approximation, extensive parametric sweeps, random Dirichlet starts,
and a fast deterministic hill‑climb that updates the convolution
incrementally.  The algorithm works on a fixed length (default 4096)
and respects the given time budget.
"""

import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2*n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n - 1]
    return conv

def _conv_fft(x, y):
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    z = np.fft.irfft(X * Y, n=L)[:len(x) + len(y) - 1]
    return z

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective_score(seq):
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _smoothmax_grad(a, tau):
    """
    Compute gradient of the smooth‑max surrogate
    Sτ(a) = (1/τ)·log Σ exp(τ·conv_k) .
    Returns gradient (size n) and the full convolution vector.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_c = conv.max()
    # use stable softmax
    w = np.exp(tau * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum
    a_rev = a[::-1]
    # conv of w with a_rev, length = 3n-2 -> extract middle n entries
    L = _next_pow2(3 * n - 2)
    grad_full = _conv_fft(w, a_rev)
    grad = 2.0 * grad_full[n-1:n-1+n]
    return grad, conv

def _subgradient_maxconv(a, conv):
    """
    Sub‑gradient of max(conv) at a (non‑smooth case).
    Returns a vector of size n.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _apply_mass_transfer(a, conv, src, dst, eps):
    """
    Return new (a, conv) after moving eps mass from src to dst.
    Both a and conv are assumed to be consistent before the move.
    """
    n = a.shape[0]
    a_new = a.copy()
    a_new[src] -= eps
    a_new[dst] += eps
    # update convolution incrementally
    conv_new = conv.copy()
    # linear part: 2*eps*(shift(dst) - shift(src))
    conv_new[src:src + n] -= 2.0 * eps * a
    conv_new[dst:dst + n] += 2.0 * eps * a
    # quadratic correction
    conv_new[0] += 2.0 * eps * eps
    idx = src + dst
    max_idx = 2 * n - 2
    if idx <= max_idx:
        conv_new[idx] -= 2.0 * eps * eps
    return a_new, conv_new

    The optimiser performs a parametric sweep, random Dirichlet starts,
    Adam‑based smooth‑max descent and a fast deterministic hill‑climb.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    # initialise with uniform distribution
    best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # 1️⃣  Parametric families (geometric, power, beta)
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:

        # Geometric family

            if time.time() >= phase_deadline:

            r = 1.0 - eps
            geo = np.power(r, idx)
            geo_sum = geo.sum()
            if geo_sum == 0:

            geo /= geo_sum
            for lam in np.linspace(0.0, 1.0, 9):
                seq = lam * best_seq + (1.0 - lam) * geo
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:

                    print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

        for p in np.linspace(-4.0, 3.0, 141):
            if time.time() >= phase_deadline:

            s = pow_seq.sum()
            if s == 0:

            pow_seq /= s
            sc = _objective_score(pow_seq)
            if sc < best_score:

        # Beta family (includes arcsine etc.)

                    conv = _conv_self(a)   # recompute to stay consistent

                        conv = conv_prop

            best_score = sym_score

    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

"""
Optimiser for the step‑function challenge.

The algorithm proceeds through four phases:

1️⃣  **Parametric sweep**      – geometric, power and beta families with
                               finer grids than the reference code.
2️⃣  **Random Dirichlet starts** – many draws with concentrations in
                               [0.05, 5] to diversify the initial point.
3️⃣  **Smooth‑max Adam descent** – minimise the smooth (log‑sum‑exp) proxy
    for the max‑convolution using an Adam optimiser.  The temperature τ
    starts very small (τ≈1) and is increased slowly, allowing the optimiser
    to explore the landscape before focussing on the true max.
4️⃣  **Fast deterministic hill‑climb** – repeatedly transfers a tiny
    amount of mass from the index with the largest sub‑gradient to the one
    with the smallest, updating the self‑convolution *exactly* by an
    O(n) formula.  The move is accepted only if the objective improves,
    otherwise the step size is halved a few times.  The convolution is
    periodically recomputed from scratch to avoid drift.

A final symmetricisation (averaging with the reversed vector) is tried;
if it improves the bound the symmetric sequence is kept.

All operations respect the simplex constraint (non‑negative, sum = 1) and
the overall wall‑clock budget (default 1000 s).  The function returns a
plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    z = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return z


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _smoothmax_grad(a: np.ndarray, tau: float):
    """
    Gradient of the smooth‑max surrogate
        Sτ(a) = (1/τ)·log Σ exp(τ·conv_k)
    and the full convolution vector.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_c = conv.max()
    # stable soft‑max over the full convolution
    w = np.exp(tau * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum

    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)
    grad_full = _conv_fft(w, a_rev)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    Averaged over all active (max‑attaining) shifts.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _apply_mass_transfer(a: np.ndarray, conv: np.ndarray,
                         src: int, dst: int, eps: float):
    """
    Return new (a, conv) after moving `eps` mass from `src` to `dst`.
    Both a and conv are assumed to be consistent before the move.
    The update is exact (up to floating‑point error) and O(n).
    """
    n = a.shape[0]
    a_new = a.copy()
    a_new[src] -= eps
    a_new[dst] += eps

    conv_new = conv.copy()
    # linear part: 2·eps·(shift(dst) – shift(src))
    conv_new[src : src + n] -= 2.0 * eps * a
    conv_new[dst : dst + n] += 2.0 * eps * a
    # quadratic correction
    conv_new[0] += 2.0 * eps * eps
    idx = src + dst
    max_idx = 2 * n - 2
    if idx <= max_idx:
        conv_new[idx] -= 2.0 * eps * eps

    return a_new, conv_new

        print(f"[{time.time() - start_time:.2f}s] Hill‑climb finished; best score={best_score:.6f}")

    # ---------- 5️⃣  Symmetrisation (optional) ----------
    if time.time() < deadline:
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = np.maximum(a_sym, 0.0)
        a_sym /= a_sym.sum()
        sym_score = _objective_score(a_sym)
        if sym_score < best_score:
            best_seq = a_sym
            best_score = sym_score
            print(f"[{time.time() - start_time:.2f}s] Symmetrisation improved score to {sym_score:.6f}")

    # final safety normalisation
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
