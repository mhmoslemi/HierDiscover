"""
Propose a step‑function (non‑negative list) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1. **Parameter sweep** – Build a library of simple basis‐functions
   (uniform, geometric decays and power‑law decays).  Randomly mix
   three of them (weights drawn from a Dirichlet) and keep the best
   mixture after a few seconds.  This quickly finds a good starting
   shape that beats the previous 1.508‑score.

2. **Projected Adam** – Starting from the best mixture run a projected
   Adam optimiser on the simplex.  The gradient is a *multi‑active*
   sub‑gradient: we take the average over the top‑k (default 5) shifts
   that attain the current maximum of the convolution.  Adam’s
   momentum and adaptive step‑size accelerate convergence.

3. **Enhanced hill‑climb** – After Adam, perform a stochastic local
   search that transfers a *larger* random fraction (up to 10 %) of
   mass between two indices and occasionally performs a three‑point
   redistribution.  Only improving moves are kept.

All operations respect the 1000 s budget, use FFT for convolutions,
and project onto the probability simplex after each update.
The routine returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σ w = 1 }.
    Returns w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient helpers
# ----------------------------------------------------------------------
def _subgrad_multi_active(a: np.ndarray, conv: np.ndarray, top_k: int = 5) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) obtained as the average of the
    gradients of the ``top_k`` shifts with largest convolution values.
    """
    n = a.shape[0]
    # indices of the k largest entries (ties broken arbitrarily)
    idx = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for k in idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices i where both a[i] and a[k-i] are defined
        seq_idx = np.arange(lo, hi + 1, dtype=int)
        grad[seq_idx] += 2.0 * a[k - seq_idx]      # 2·a_{k-i}
    grad /= float(top_k)
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (multi‑active gradient)
# ----------------------------------------------------------------------
def _adam_optimize_multi(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_score_initial: float,
                         top_k: int = 5,
                         max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser using the *average* gradient over the ``top_k`` active shifts.
    Returns the best sequence seen and its objective value
    (2·n·max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        # convolution and current score (sum(a)=1 after projection)
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # multi‑active sub‑gradient
        grad = _subgrad_multi_active(a, conv, top_k=top_k)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # optional learning‑rate decay (helps at the end)
        lr = lr0 / (1.0 + it * 1e-6)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex
        a = _project_simplex(a)

        # plateau handling – add a modest random shake after long stagnation
        if it - last_improve > 10_000:
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (larger moves)
# ----------------------------------------------------------------------
def _hill_climb_enhanced(start_a: np.ndarray,
                         start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a larger random fraction of mass
    (up to 10 %) and occasionally performs a three‑point redistribution.
    Only strictly improving moves are kept.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        # decide whether to do a 2‑point or a 3‑point move
        if rng.random() < 0.7:          # two‑point move (majority)
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            # move up to 10 % of the source mass
            frac = rng.random() * 0.10 + 1e-6
            eps = a[src] * frac

            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

        else:                           # three‑point move
            src1 = rng.integers(n)
            src2 = rng.integers(n)
            if src1 == src2 or a[src1] < 1e-15 or a[src2] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst in (src1, src2):
                continue
            # split a fraction of the combined source mass
            total_src = a[src1] + a[src2]
            frac = rng.random() * 0.08 + 1e-6
            eps = total_src * frac
            # proportionally take from both sources
            eps1 = eps * a[src1] / total_src
            eps2 = eps - eps1

            a_prop = a.copy()
            a_prop[src1] -= eps1
            a_prop[src2] -= eps2
            a_prop[dst] += eps

        # enforce non‑negativity (tiny numerical drift)
        if np.min(a_prop) < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The search consists of:
        * building a library of simple basis functions and random mixing;
        * a long projected Adam optimiser using a multi‑active sub‑gradient;
        * a final stochastic hill‑climb with larger mass transfers.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the evaluation harness
    deadline = start_time + budget_s - 0.4

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # baseline (uniform)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Build a library of basis‑functions
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric family (r = 1‑ε)
    eps_grid = np.logspace(-4.5, -2.0, 200)             # ε from ~3e‑5 to 1e‑2
    r_grid = 1.0 - eps_grid
    geos = []
    for r in r_grid:
        geo = np.power(r, idx)
        geo = geo / geo.sum()
        geos.append(geo)
    geos = np.stack(geos)                               # shape (G, n)

    # power‑law family a_i ∝ (i+1)^{-α}
    alpha_grid = np.logspace(-0.3, 1.2, 120)            # α from ~0.5 to ~15
    pls = []
    for a in alpha_grid:
        pl = 1.0 / np.power(idx + 1.0, a)
        pl = pl / pl.sum()
        pls.append(pl)
    pls = np.stack(pls)                                 # shape (P, n)

    # --------------------------------------------------------------
    # 2️⃣  Random mixture of up to three components (uniform, geo, power‑law)
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    max_samples = 12000
    sample = 0
    while time.time() < deadline - 200.0 and sample < max_samples:
        sample += 1

        # pick random geometric and power‑law prototypes
        g_idx = rng.integers(geos.shape[0])
        p_idx = rng.integers(pls.shape[0])
        geo = geos[g_idx]
        pl = pls[p_idx]

        # random mixing weights that sum to one (uniform included)
        lam = rng.dirichlet([1.0, 1.0, 1.0])   # [λ_geo, λ_pl, λ_uni]
        seq = lam[0] * geo + lam[1] * pl + lam[2] * uniform

        # safeguard numeric sum
        seq = seq / seq.sum()

        cur_score = float(2.0 * n * _conv_self(seq, L_fft).max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Random mix #{sample} score={cur_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Long Adam optimiser with sharper multi‑active gradient
    # --------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining > 120.0:
        print(f"[{time.time() - start_time:.2f}s] Starting multi‑active Adam optimiser")
        adam_deadline = deadline - 80.0            # reserve time for hill‑climb
        a_opt, sc_opt = _adam_optimize_multi(best_seq,
                                             start_time,
                                             adam_deadline,
                                             best_score,
                                             top_k=5)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final stochastic hill‑climb with larger moves
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb_enhanced(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:

    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
