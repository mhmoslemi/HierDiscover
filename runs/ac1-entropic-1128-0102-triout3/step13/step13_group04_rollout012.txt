def _adam_smooth_optimize(a0, start_time, deadline, best_score, tau, max_iter):
    a = a0.copy()
    best_a = a.copy()
    best_score_local = best_score
    n = len(a)
    
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2   # base
    it = 0
    last_improve = 0
    rng = np.random.default_rng()
    
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute convolution of a with itself
        conv = _conv_self(a)
        # compute the true objective score
        cur_score = 2.0 * n * conv.max()  # sum=1 after projection
        if cur_score < best_score_local:
            best_score_local = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time()-start_time:.2f}s] tau={tau:.3e} Adam it={it:,} score={cur_score:.6f}")
        # compute softmax weights
        shift = conv.max()
        w = np.exp((conv - shift) / tau)   # stable
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        
        # compute weighted conv and corr
        conv_weighted_full = np.convolve(a, w, mode='full')
        corr_weighted_full = np.correlate(a, w, mode='full')
        offset = n - 1
        conv_weighted = conv_weighted_full[offset: offset + n]
        corr_weighted = corr_weighted_full[offset: offset + n]
        grad = 2.0 * n * (conv_weighted + corr_weighted)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_simplex(a)
        # plateau detection: if no improvement for some iters, add noise
        if it - last_improve > 5000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a += noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it
    return best_a, best_score_local

"""
Propose a step‑function (non‑negative list) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
1. Initialise a probability vector (sum = 1) – uniform or the global starter
   if one is provided.
2. Run a *smooth* Adam optimiser that replaces the non‑smooth max by a
   soft‑max:
        fτ(a) = 2·n·τ·log Σi exp(conv_i(a)/τ)
   The gradient of fτ is obtained analytically (a weighted sum of a
   convolution and a correlation).  The soft‑max temperature τ is
   gradually annealed (0.3 → … → 0.001) so the optimisation first sees a
   smooth landscape and then focuses on the true max.
3. After the annealing phase a short Adam run with the *hard* max
   sub‑gradient (as in the reference solution) is performed to polish
   the result.
4. Finally a stochastic hill‑climb (tiny mass transfers) is applied.
   All updates are projected onto the probability simplex.

All heavy work (convolutions, correlations, FFTs) is performed with
FFT‑based O(N log N) operations, allowing millions of iterations well
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a∗a (length 2·n‑1) using FFT of length L."""
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)
    # keep only the linear part (first 2·n‑1 entries)
    return conv_full[: 2 * a.shape[0] - 1]


def _grad_softmax(a: np.ndarray, conv_self: np.ndarray,
                  w: np.ndarray, L: int, offset: int, n: int) -> np.ndarray:
    """
    Compute ∇fτ(a) = 2·n·(T1 + T2) where
        T1_j = Σ_i w_i a_{i+j}   (correlation)
        T2_j = Σ_i w_i a_{j-i}   (convolution)
    using FFTs of length L.
    """
    # FFT of a and of the weight vector w (length 2·n‑1)
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)

    # linear convolution a * w   -> T2
    conv_full = np.fft.irfft(A * W, n=L)
    # linear correlation a ⋆ w -> T1
    corr_full = np.fft.irfft(A * np.conj(W), n=L)

    # only the central n entries are needed
    conv_weighted = conv_full[offset: offset + n]
    corr_weighted = corr_full[offset: offset + n]

    # gradient of the smoothed objective
    grad = 2.0 * n * (conv_weighted + corr_weighted)
    return grad


# ----------------------------------------------------------------------
# Adam optimizer with soft‑max smoothing
# ----------------------------------------------------------------------
def _adam_smooth_optimize(a0: np.ndarray,
                          start_time: float,
                          deadline: float,
                          best_score: float,
                          tau: float,
                          max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smoothed objective
        fτ(a) = 2·n·τ·log Σ exp(conv_i(a)/τ)
    The gradient is computed analytically via a single weighted
    convolution + correlation (FFT based).  After each step the vector is
    projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = best_score
    n = a.shape[0]

    # FFT lengths (fixed for the whole run)
    L_self = _next_pow2(2 * n - 1)            # for a∗a
    L_weight = _next_pow2(3 * n - 2)          # for a∗w and a⋆w
    offset = n - 1

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        # self‑convolution (used both for the objective and the weights)
        conv_self = _conv_self_fft(a, L_self)          # length 2·n‑1
        max_conv = conv_self.max()
        cur_score = 2.0 * n * max_conv                 # true objective (sum(a)=1)

        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] τ={tau:.3e}  "
                  f"Adam it={it:,}  score={cur_score:.6f}")

        # soft‑max weights (stable by subtracting max)
        w = np.exp((conv_self - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient via weighted conv + correlation
        grad = _grad_softmax(a, conv_self, w, L_weight, offset, n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (keeps sum = 1, a_i ≥ 0)
        a = _project_simplex(a)

        # occasional tiny perturbation if we are stuck
        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_val

# Hard‑max Adam (single active shift) – same as the reference code
# ----------------------------------------------------------------------
def _adam_hard_optimize(a0: np.ndarray,
                        start_time: float,
                        deadline: float,
                        best_score: float,
                        max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser using the exact sub‑gradient from a *single* maximising
    shift (the one that attains max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = best_score
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Hard‑Adam it={it:,}"
                  f" score={cur_score:.6f}")

        # sub‑gradient from the (unique) maximising shift
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]  # ∂/∂a_i of (a⋆a)[k]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_val

# Simple hill‑climb (tiny mass transfers)

def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: transfer a tiny random fraction of mass from a
    source index to a destination index; keep the move only if it improves
    the true objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self_fft(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 1 % of the source mass
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self_fft(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score

        # number of allowed iterations is gated by the remaining time
        iters = 200_000
        best_seq, best_score = _adam_smooth_optimize(
            best_seq, start_time, anneal_deadline, best_score,
            tau=tau, max_iter=iters
        )
        # recompute true objective (sum(a)=1 after each projection)
        best_score = float(2.0 * n * _conv_self_fft(best_seq, L_self).max())

    # ------------------------------------------------------------------
    # 2️⃣  Hard‑max Adam refinement (single active shift)
    # ------------------------------------------------------------------
    hard_deadline = deadline - 180.0   # leave ~180 s for hill‑climb
    if time.time() < hard_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hard‑max Adam refinement")
        best_seq, best_score = _adam_hard_optimize(
            best_seq, start_time, hard_deadline, best_score, max_iter=150_000
        )
        best_score = float(2.0 * n * _conv_self_fft(best_seq, L_self).max())
        print(f"[{time.time() - start_time:.2f}s] Hard‑max Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, start_time, deadline)
        best_score = float(2.0 * n * _conv_self_fft(best_seq, L_self).max())
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
