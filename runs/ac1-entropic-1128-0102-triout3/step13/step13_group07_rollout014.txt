def _adam_smooth_optimize(a0, deadline, start_time, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, beta_start=1.0, beta_max=200.0, beta_factor=1.002):
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3*n)  # enough for conv with w_rev
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _objective(a)
    beta = beta_start
    it = 0
    while time.time() < deadline:
        it += 1
        # FFT of a
        Af = np.fft.rfft(a, n=L)
        # self convolution
        conv_fft = Af * Af
        conv = np.fft.irfft(conv_fft, n=L)[:2*n-1]
        # softmax weights
        c = conv - conv.max()
        expc = np.exp(beta * c)  # stable because c <=0
        w = expc / expc.sum()
        # gradient via cross-correlation
        Wf = np.fft.rfft(w[::-1], n=L)
        grad_fft = Af * Wf
        grad_full = np.fft.irfft(grad_fft, n=L)[:3*n-2]
        cross = grad_full[n-1:2*n-1][::-1]
        grad = 4.0 * n * cross
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        # projection onto simplex
        a = _project_simplex(a)
        # compute exact objective (use conv we already have)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothAdam it={it:,} beta={beta:.3f} score={cur_score:.6f}")
        # beta schedule
        beta = beta * beta_factor
        if beta > beta_max:
            beta = beta_max
        # optionally decay lr
        if it % 5000 == 0:
            lr *= 0.9
    return best_a, best_score

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Warm‑starts** – a diverse set of analytic shapes (tent, cosine,
   half‑circle, Gaussian, power‑tents, exponentials, arcsine) and the
   optional global variable ``height_sequence_1`` are normalised to sum = 1.
   The best of them w.r.t. the exact objective
   ``2·n·max(conv(a,a))`` is taken as the initial point.

2. **Smooth‑Adam** – minimise the *soft‑max* surrogate

        f(a) = 2·n·log( Σ_k exp(β·(a∗a)_k) ) / β ,

   whose gradient is
        ∇f = 4·n·cross‑correlation(a , w) ,

   where ``w`` are the soft‑max weights.
   The temperature ``β`` is increased slowly (β←β·1.002, max ≈ 200)
   so the optimisation first sees a smooth landscape and ends
   essentially at the true max‑convolution objective.
   Adam updates are performed on the probability simplex (projection
   onto the Euclidean simplex).

3. **Fine sub‑gradient Adam** – after β has reached its maximum a short
   Adam run with the exact max‑convolution sub‑gradient (as in the
   reference implementation) is applied to polish the result.

4. **Greedy hill‑climbing** – a cheap Metropolis walk moves a random
   packet of mass between two positions and accepts any strictly
   improving move.  This often yields a final tiny improvement.

All heavy work uses NumPy’s real FFT; the routine respects the
``budget_s`` time limit and always returns the best sequence found.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic numerical utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ------------------------------------------------------------
# Warm‑starts – analytic families
# ------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]
    candidates = []

    # 1) linear tent
    candidates.append(np.minimum(x, 1.0 - x))

    # 2) half‑cosine
    candidates.append(np.sin(np.pi * x))

    # 3) half‑circle (Archimedes)
    candidates.append(np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0)))

    # 4) Gaussian centred in the middle
    sigma = 0.2
    candidates.append(np.exp(-0.5 * ((x - 0.5) / sigma) ** 2))

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        fwd = np.exp(-scale * x)
        candidates.append(fwd)
        candidates.append(fwd[::-1])

    # 7) arcsine (U‑shaped) – avoids division by zero at the ends
    eps = 1e-12
    arcsine = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, 1.0))
    candidates.append(arcsine)

    # 8) completely flat (uniform) – useful baseline
    candidates.append(np.ones_like(x))

    # Normalise and keep only non‑negative candidates
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ------------------------------------------------------------
# Exact objective (2·n·max(conv))
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Adam optimiser – smooth surrogate (soft‑max)
# ------------------------------------------------------------
def _adam_smooth_optimize(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          lr: float = 0.05,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          beta_start: float = 1.0,
                          beta_max: float = 200.0,
                          beta_factor: float = 1.002) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max surrogate of max‑convolution.
    Returns the best iterate seen and its exact (non‑smooth) objective.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)                # enough for a∗w_rev (length 3n‑2)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    beta = beta_start
    it = 0

    while time.time() < deadline:
        it += 1

        # FFT of current a (re‑used for conv and gradient)
        Af = np.fft.rfft(a, n=L)

        # ---- self‑convolution (exact values for the true objective) ----
        conv_fft = Af * Af
        conv = np.fft.irfft(conv_fft, n=L)[: 2 * n - 1]

        # ---- soft‑max weights for the surrogate ----
        c = conv - conv.max()                     # stability: max ≤ 0
        expc = np.exp(beta * c)
        w = expc / expc.sum()                     # shape (2n‑1,)

        # ---- gradient of the surrogate ----
        # cross‑correlation a ⋆ w  =  convolution(a, w_rev) with a flip
        Wf = np.fft.rfft(w[::-1], n=L)
        grad_fft = Af * Wf
        grad_full = np.fft.irfft(grad_fft, n=L)[: 3 * n - 2]

        # slice the relevant part and reverse to obtain Σ_j a_j w_{i+j}
        cross = grad_full[n - 1: 2 * n - 1][::-1]   # length n
        grad = 4.0 * n * cross                       # ∇f

        # ---- Adam update on the simplex ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        # ---- exact objective (for bookkeeping) ----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothAdam it={it:,} "
                  f"β={beta:.3f} score={cur_score:.6f}")

        # ---- schedule for β and learning‑rate ----
        beta = min(beta_max, beta * beta_factor)
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ------------------------------------------------------------
# Adam optimiser – exact sub‑gradient of max(conv)
# ------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    conv is the full linear self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]      # g_i = 2·a_{k‑i}
    return grad


def _adam_max_optimize(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       lr: float = 0.02,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the exact sub‑gradient of max(conv).
    Returns the best iterate seen and its exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        grad = _max_subgradient(a, conv)               # sub‑gradient of max
        grad_obj = 2.0 * n * grad                       # ∇(2·n·max)

        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] MaxAdam it={it:,} "
                  f"score={cur_score:.6f}")

        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ------------------------------------------------------------
# Simple hill‑climbing polish
# ------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepting only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # source weighted by current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30    # move up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:           # greedy acceptance
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the list of heights (sum = 1, all ≥ 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------
    # Problem size – a power of two helps the FFT.
    # --------------------------------------------------------
    n = 4096

    # --------------------------------------------------------
    # 1) Warm‑starts – pick the best analytic shape.
    # --------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # incorporate the global variable if supplied
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            if np.all(h1_resampled >= 0):
                h1_resampled = np.maximum(h1_resampled, 0.0)
                s = h1_resampled.sum()
                if s > 0:
                    warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    # a few random Dirichlet warm‑starts (cheap, can improve a bit)
    rng = np.random.default_rng(seed + 12345)
    for _ in range(5):
        rand = rng.dirichlet(np.ones(n))
        warm_starts.append(rand)

    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # --------------------------------------------------------
    # 2) Smooth‑Adam (soft‑max surrogate)
    # --------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        smooth_deadline = time.time() + remaining * 0.60   # 60 % of the rest
        a_opt, score_opt = _adam_smooth_optimize(
            a0, smooth_deadline, start_time,
            lr=0.05, beta_start=1.0, beta_max=200.0, beta_factor=1.002)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothAdam best score={score_opt:.6f}")

    # --------------------------------------------------------
    # 3) Max‑convolution Adam (exact sub‑gradient) – short polish
    # --------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        maxadam_deadline = time.time() + remaining * 0.20   # 20 % of remaining
        a_opt, score_opt = _adam_max_optimize(a0, maxadam_deadline, start_time,
                                              lr=0.015)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] MaxAdam best score={score_opt:.6f}")

    # --------------------------------------------------------
    # 4) Light hill‑climb polish (use whatever time is left)
    # --------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={score_opt:.6f}")

    # --------------------------------------------------------
    # 5) Normalise & return plain list
    # --------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
