"""
Propose a step‑function (non‑negative heights) that minimises  

        2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of three phases:

1. **Broad initial sampling** – generate a diverse set of analytic
   shapes (uniform, power‑ramps, finely‑swept Beta‑distributions) and
   a handful of random Dirichlet vectors with various concentration
   parameters.  The best K candidates are retained.

2. **Smooth‑max Adam optimisation** – from each retained candidate we run
   an Adam‑style descent on a soft‑max surrogate of the objective.
   The temperature starts relatively high (0.3) and is cooled
   exponentially to 1e‑6, allowing the optimiser to escape shallow
   local minima and then to focus on the true max‑convolution.
   The learning‑rate schedule is aggressive (initial step≈0.15) and
   the optimiser is run for as many iterations as the remaining time
   permits.

3. **Fine‑grained simulated‑annealing polish** – a short SA run (mass
   swaps between two indices) starting from the best sequence found in
   phase 2 gives a final nudge.

All phases respect the 1000 s budget, use only NumPy (no external
solvers) and keep the vector on the probability simplex via a Euclidean
projection.  The returned list sums to 1, contains only non‑negative
floats and is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power of two ≥ x (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Linear convolution a*a (real FFT, length 2·len(a)-1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Objective for a probability vector (sum==1)
#   value = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {a ≥ 0, Σa = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Analytic power‑ramp with optional plateau (symmetric)
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# ----------------------------------------------------------------------
# Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# ----------------------------------------------------------------------
# Keep the K best (lowest) candidates (sorted)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_iter: int = 10**7) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on   f_T(a) = T·log Σ exp(conv/T)
    with decreasing temperature.  Returns the best (seq,score)
    seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters
    T_start = 0.30
    T_end = 1e-6
    step0 = 0.15                # initial learning rate

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # forward pass
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature (exponential decay)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = max(T_end, T_start * ((T_end / T_start) ** frac))

        # soft‑max weights (stable)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max = 2 * (w * a[::-1])  (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing (mass swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.05) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        # source index proportional to weight
        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} "
                      f"new best={new_score:.6f}")

    return best_a, best_score
