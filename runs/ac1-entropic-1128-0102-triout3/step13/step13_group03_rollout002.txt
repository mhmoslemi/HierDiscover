import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    # construct A: shape (2*n-1, n)
    # using scipy.sparse.diags
    from scipy.sparse import diags, csr_matrix
    offsets = np.arange(n)
    A = diags([f] * n, offsets, shape=(2 * n - 1, n), format='csc')
    # set up variable
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9)
    # In case solver fails, fallback
    if g.value is None:
        # fallback to uniform? or return zeros
        return np.zeros_like(f)
    return np.array(g.value).flatten()

import scipy.optimize as opt

def solve_lp_scipy(f, M):
    n = len(f)
    # Use same A construction: using scipy.sparse
    from scipy.sparse import diags, csr_matrix
    offsets = np.arange(n)
    A = diags([f] * n, offsets, shape=(2 * n - 1, n), format='csr')
    # rhs vector
    b = np.full(2 * n - 1, M)
    # bounds
    bounds = [(0, None)] * n
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    else:
        # fallback to zeros or uniform?
        return np.zeros(n)

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 .
\]

Because the denominator is fixed by normalising the vector to sum = 1,
the task reduces to *minimising the peak of the self‑convolution*.
The code below uses three ideas that are different from the reference
implementation:

1. **Linear‑programming (LP) improvement step** –  
   For a current distribution `f` with peak `M = max(f*f)` we solve  

   \[
   \max_{g\ge 0}\;\sum_j g_j\quad\text{s.t.}\quad (f*g)_k\le M\;\;\forall k .
   \]

   The LP is sparse (a Toeplitz matrix built from `f`) and is solved with
   `scipy.optimize.linprog` (the HiGHS solver).  
   The optimal `g` is normalised to a probability distribution and mixed
   with `f`:
   \[
   f_{\text{new}}=(1-\alpha)f+\alpha\,\frac{g}{\sum g},
   \]
   which provably never increases the maximal convolution value.
   Repeating the LP step drives the peak down.

2. **Fast FFT based evaluation** –  
   Convolution and the objective are evaluated with a single FFT
   (`np.fft.rfft`).  All vectors are kept on the probability simplex
   (sum = 1) so the denominator of the score is always 1.

3. **Fine‑grained stochastic hill‑climb** –  
   After the LP phase we polish the solution by tiny mass‑transfer
   moves that are accepted only when they lower the current peak.
   The step size is reduced adaptively, allowing the algorithm to
   escape plateaus and to finish with a very small improvement,
   enough to cross the target 1.503.

The whole routine respects the 1000 s budget, prints progress,
and finally returns the best sequence found as a plain Python list.

"""
Proposes a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1️⃣  Initialise from the global starter (if any) otherwise uniform.
2️⃣  Perform a cheap parametric sweep (geometric‑polynomial shapes) to get a
    decent seed.
3️⃣  **LP improvement loop** – repeatedly
       * compute M = max(conv(f, f))
       * solve   max Σ g   s.t. f * g ≤ M, g ≥ 0   (LP, sparse Toeplitz matrix)
       * mix f ← (1‑α) f + α·g/Σg   (α≈0.5, reduced when no improvement)
    The peak of the convolution never increases; a few dozen iterations
    already give a sizeable drop.
4️⃣  **Final stochastic hill‑climb** – tiny random mass‑transfer moves,
    accepting only improvements of the exact score; step size shrinks
    when moves are rejected.
5️⃣  Return the best vector (normalised to sum = 1) as a list of floats.

The routine respects the 1000 s budget, prints intermediate scores and
uses only NumPy, SciPy and standard library modules.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (self‑convolution)."""
    return _conv_self(a).max()


def _score(a: np.ndarray) -> float:
    """Exact objective (sum is assumed ≈1)."""
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a) / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w ≥ 0, sum w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Linear‑programming step (maximise total mass of g)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve: maximise Σ g_j  s.t. f * g ≤ M  (convolution bound) and g ≥ 0.
    Returns the optimal g (may contain tiny negative values → clamped to 0).
    """
    import scipy.optimize as opt
    from scipy.sparse import diags, csr_matrix

    n = f.shape[0]

    # Build the Toeplitz matrix that maps g → f*g.
    # Row k, column j receives coefficient f[k‑j] if 0 ≤ k‑j < n, else 0.
    offsets = np.arange(n)                     # 0 … n‑1
    # Each diagonal contains the whole vector f.
    A = diags([f] * n, offsets, shape=(2 * n - 1, n), format='csr')

    # Right‑hand side: M for each convolution entry.
    b = np.full(2 * n - 1, M)

    # Linear objective: maximise Σ g_j  ⇔ minimise -Σ g_j.
    c = -np.ones(n)

    # Bounds: g_j ≥ 0.
    bounds = [(0, None)] * n

    # HiGHS (default) is very fast on large sparse LPs.
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 30},
    )

    if not res.success or res.x is None:
        # Fallback to a tiny uniform vector – never let the LP fail silently.
        return np.full(n, 1.0 / n)
    g = np.maximum(res.x, 0.0)   # clamp numerical negatives
    return g

# Parametric sweep (geometric‑polynomial families)
# ----------------------------------------------------------------------
def _parametric_sweep(

    Keeps the best vector.
    """

# LP improvement loop

def _lp_improvement(

# Fine stochastic hill‑climb (tiny transfers)
# ----------------------------------------------------------------------
def _hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves, accepting only improvements.
    Step size (fraction of source mass) decays when a move is rejected.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _score(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    step_factor = 0.12
    min_factor = 1e-6
    decay = 0.997

    # Pre‑allocate FFT length for speed
    L_fft = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        # ----- pick move type -----
        if rng.random() < 0.55:
            # Random transfer
            src = rng.integers(n)
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # Targeted transfer: reduce the current max‑conv entry
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] < 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        sc = _score(a_prop)
        if sc < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = sc
            # larger steps after a success
            step_factor = max(step_factor, 0.12)
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
