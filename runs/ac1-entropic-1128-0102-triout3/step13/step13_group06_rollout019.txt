    # after how many iterations to recompute full convolution
    recompute_interval = 10000

    while time.time() < deadline:
        total_iters += 1

        # recompute conv occasionally to avoid drift
        if total_iters % recompute_interval == 0:
            conv = _conv_fft(a)

        # current max and its index
        max_val = conv.max()
        cur_score = n_obj * max_val

        # build weight vector w[i] = a[k_max - i] where defined
        idx = np.arange(n)

        # choose source with large a[i]*w[i] (large contribution to the peak)
        prod = a * w

        # ensure src has some mass
        if a[src] < 1e-12:

            src = int(rng.choice(n, p=a / a.sum()))
        # choose destination where w is minimal (ideally zero)

                dst = int(rng.choice(dst_candidates))
                if dst == src:

        # step size: fraction of src mass, decreasing towards end of run
        elapsed = time.time() - start

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the linear self‑convolution.  The algorithm works in three
stages:

1. **Analytic warm‑start** – a large collection of parametric shapes
   (power‑tents, exponentials, Beta‑type densities, etc.) is generated,
   normalised to sum‑to‑one and the best candidate is kept.

2. **Guided simulated‑annealing** – starting from the warm‑start vector
   we repeatedly transfer a small amount of mass from a “high‑impact”
   index to a “low‑impact” index.  The high‑impact index is chosen by
   the gradient of the currently active convolution peak; the low‑impact
   index by the opposite gradient.  Moves that improve the objective are
   always accepted, otherwise they are accepted with a probability that
   decreases with a linear temperature schedule.

3. **Final deterministic refinement** – a short hill‑climbing phase
   (only improving moves) removes any remaining tiny imperfections.

All operations are performed with an incremental convolution update
(`_apply_move`) so that millions of moves can be tried inside the
~1000 s time limit.  The returned list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total * total))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum ≈ 1)."""
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), a.shape[0])


def _apply_move(a: np.ndarray, conv: np.ndarray,
                src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from `src` to `dst`.  Both ``a`` and ``conv``
    are updated **in‑place**.
    """
    n = a.shape[0]

    # linear contribution 2·a·Δ   (Δ = eps·(e_dst – e_src))
    conv[src:src + n] -= 2.0 * eps * a
    conv[dst:dst + n] += 2.0 * eps * a

    # quadratic contribution Δ·Δ
    i2 = 2 * src
    j2 = 2 * dst
    s = src + dst
    conv[i2] += eps * eps
    conv[j2] += eps * eps
    conv[s] -= 2.0 * eps * eps

    # finally update the probability vector
    a[src] -= eps
    a[dst] += eps

    # ------------------------------------------------------------------
    for p in np.linspace(0.2, 2.0, 15):

        candidates.append(shape)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # ------------------------------------------------------------------

    #    (α,β can be negative – we clamp very small values to avoid div‑0)
    # ------------------------------------------------------------------
    eps = 1e-12

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    # --------------------------------------------------------------
    # Short polishing pass – only accept improving moves.
    refine_iters = 200000
    for i in range(refine_iters):
        if time.time() >= deadline:
            break
        src = int(rng.choice(n, p=a / a.sum()))
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 0.01
        if eps <= 0.0:
            continue
        before_obj = n_obj * conv.max()
        _apply_move(a, conv, src, dst, eps)
        new_obj = n_obj * conv.max()
        if new_obj <= before_obj:
            if new_obj < best_score:
                best_score = new_obj
                best_vec = a.copy()
        else:
            _apply_move(a, conv, dst, src, eps)

    # --------------------------------------------------------------
    # 4) Normalise and return.
    # --------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec = best_vec / s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
