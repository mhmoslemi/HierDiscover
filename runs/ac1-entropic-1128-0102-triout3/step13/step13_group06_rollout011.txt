def propose_candidate(seed=0, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine length n
    n = 4096
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)
        except Exception:
            pass

    # Build candidate shapes
    candidates = _make_candidate_shapes(n)  # from prior but we will extend
    # beta shapes
    x = np.arange(n, dtype=float) / (n - 1)
    for alpha in np.linspace(0.05, 0.9, 18):
        shape = (x ** (alpha-1)) * ((1 - x) ** (alpha-1))
        # handle zero at edges: when alpha<1, power may produce inf at 0,1. Fix using epsilon.
        shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            candidates.append(shape / s)

    # include global sequence
    if 'height_sequence_1' in globals():
        try:
            h = np.asarray(height_sequence_1, dtype=float)
            if h.shape == (n,):
                h = np.maximum(h, 0.0)
                if h.sum() > 0:
                    candidates.append(h / h.sum())
        except Exception:
            pass

    # evaluate candidates
    best_vec = None
    best_score = math.inf
    for cand in candidates:
        sc = _objective(cand)  # conv via FFT
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    print(f"[{time.time() - start_time:.2f}s] Warm start best score={best_score:.6f}")

    # hill climbing with simulated annealing
    a = best_vec.copy()
    conv = _conv_fft(a)
    cur_score = best_score
    total = a.sum()
    n_factor = 2.0 * n / (total * total)  # =2*n since total=1

    # algorithm parameters
    T0 = 5e-3   # initial temperature (tune)
    min_frac = 0.02
    max_frac_initial = 0.9
    it = 0
    last_log = start_time
    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining = deadline - time.time()
        it += 1

        # temperature schedule
        T = T0 * (remaining / (deadline - start_time))

        # find current maximal convolution index
        kmax = int(conv.argmax())
        # compute contributions for that index
        i_inds = np.arange(n)
        j_inds = kmax - i_inds
        mask = (j_inds >= 0) & (j_inds < n)
        contrib = np.zeros_like(a)
        if mask.any():

            jj = j_inds[mask]
            contrib[ii] = a[ii] * a[jj]
            total_contrib = contrib[ii].sum()

            else:
                src = rng.choice(n, p=a/ a.sum())

            src = rng.choice(n, p=a/ a.sum())

        dst = rng.integers(0, n)
        if dst == src or a[src] < 1e-15:
            continue

        frac_max = max_frac_initial * (remaining / (deadline - start_time)) + min_frac
        eps = a[src] * rng.random() * frac_max
        if eps <= 0:
            continue

        # store current objective
        cur_obj = cur_score

        # apply move
        _apply_move(a, conv, src, dst, eps)

        new_max = conv.max()
        new_score = n_factor * new_max

        accept = False

            accept = True

"""
Propose a low‑objective step‑function.

The objective to minimise is

        2·n·max(conv(a,a)) / (sum(a)^2)

where conv(a,a) is the linear self‑convolution.
The function works in three phases

1.  Warm‑start – generate many analytic shapes (uniform, power‑tents,
    exponentials, Beta‑distributions, the previous best vector, etc.)
    and keep the one with the smallest exact objective (computed with an
    FFT‑based convolution).

2.  Simulated‑annealing hill‑climbing – the current vector `a` and its
    convolution `conv` are kept up‑to‑date by an O(n) incremental update.
    In each iteration we focus on the convolution index `k` that attains the
    current maximum and move a random amount of mass from an index that
    contributes heavily to that maximum to a random destination.
    Moves that lower the objective are always kept; uphill moves are accepted
    with a temperature that decays to zero, allowing the search to escape
    local minima.

3.  Return the best vector found (normalised to sum = 1).  All entries are
    non‑negative, so the sequence is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector that sums to 1."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst` and update the
    stored convolution in‑place.  The algorithm follows the exact
    linear/quadratic update described in the reference implementation.
    """
    n = a.shape[0]

    # linear part 2·a·Δ ; Δ = eps·(e_dst – e_src)
    conv[src : src + n] -= 2.0 * eps * a
    conv[dst : dst + n] += 2.0 * eps * a

    # quadratic part Δ·Δ
    conv[2 * src] += eps * eps
    conv[2 * dst] += eps * eps
    conv[src + dst] -= 2.0 * eps * eps

    # finally move the mass in the probability vector
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Analytic candidate generators
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """
    Produce a diverse set of non‑negative vectors (unnormalised).  The caller
    normalises them to sum = 1 before evaluation.
    """
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Exponential decay from the centre (symmetric)
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped, already present in the historic code)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]
    candidates.append(shape)

    # 5) Half‑cosine (already used as a warm‑start)
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) Beta‑type (U‑shaped for α<1) – allows very heavy edges
    for alpha in np.linspace(0.05, 0.9, 18):
        # pdf ∝ x^{α-1} (1‑x)^{α-1}
        shape = (np.where(x == 0, 0.0, x) ** (alpha - 1)) * (np.where(x == 1, 0.0, 1 - x) ** (alpha - 1))
        shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
        candidates.append(shape)

    # 8) Random Dirichlet draws – a few of them
    rng = np.random.default_rng(1)
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # Normalise and keep only vectors with positive total mass
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)
    return normalised


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the objective.
    The function runs for roughly ``budget_s`` seconds and returns the best
    vector (as a plain Python list) it discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – reuse the length of the previously found vector
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)               # type: ignore[name-defined]
        except Exception:
            n = 4096
    else:
        n = 4096

    # --------------------------------------------------------------
    # 1) Warm‑start – generate many analytic shapes and pick the best
    # --------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # include the previous best vector if it exists
    if 'height_sequence_1' in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            if hs.shape == (n,):
                hs = np.maximum(hs, 0.0)
                if hs.sum() > 0.0:
                    warm_starts.append(hs / hs.sum())
        except Exception:
            pass

    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Simulated‑annealing hill‑climbing on the convolution
    # --------------------------------------------------------------
    a = best_vec.copy()                     # current candidate (sum = 1)
    conv = _conv_fft(a)                     # current full convolution
    cur_score = best_score
    n_factor = 2.0 * n / (a.sum() ** 2)     # = 2·n because the sum is 1

    # annealing schedule parameters
    T0 = 5e-3                # initial temperature
    max_frac_initial = 0.9   # start‑up move size (fraction of source mass)
    min_frac = 0.02          # final move size

    it = 0
    last_log = start_time
    total_time = deadline - start_time

    while time.time() < deadline:
        it += 1
        now = time.time()
        elapsed = now - start_time
        remaining = deadline - now

        # linearly decreasing temperature
        T = T0 * (remaining / total_time)

        # ----------------------------------------------------------
        # Focus the move on the index that currently attains max(conv)
        # ----------------------------------------------------------
        kmax = int(conv.argmax())

        # contributions a[i] * a[kmax-i] for all i that are inside the range
        i_vals = np.arange(n, dtype=int)
        j_vals = kmax - i_vals
        mask = (j_vals >= 0) & (j_vals < n)

        if mask.any():
            ii = i_vals[mask]
            jj = j_vals[mask]
            contrib = a[ii] * a[jj]
            total_contrib = contrib.sum()
            if total_contrib > 0.0:
                src = rng.choice(ii, p=contrib / total_contrib)
            else:
                src = rng.choice(n, p=a / a.sum())
        else:
            src = rng.choice(n, p=a / a.sum())

        dst = rng.integers(0, n)
        if dst == src or a[src] < 1e-15:
            continue

        # gradually shrink the allowed transfer size
        frac_max = max_frac_initial * (remaining / total_time) + min_frac
        eps = a[src] * rng.random() * frac_max
        if eps <= 0.0:
            continue

        # ----------------------------------------------------------
        # Apply the move and evaluate the new objective
        # ----------------------------------------------------------
        before_score = cur_score
        _apply_move(a, conv, src, dst, eps)

        new_max = conv.max()
        new_score = n_factor * new_max

        # ----------------------------------------------------------
        # Acceptance test (Metropolis)
        # ----------------------------------------------------------
        accept = False
        if new_score <= before_score:
            accept = True
        else:
            # uphill moves are accepted with probability exp(-Δ/T)
            if T > 0.0 and rng.random() < math.exp(-(new_score - before_score) / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # undo the move
            _apply_move(a, conv, dst, src, eps)

        # occasional progress report (every ~30 s)
        if now - last_log > 30.0:
            print(f"[{now - start_time:.2f}s] iter={it:,} cur={cur_score:.6f} best={best_score:.6f}")
            last_log = now

    # --------------------------------------------------------------
    # 3) Normalise the best vector and return it as a plain list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score = {best_score:.6f}")
    return [float(v) for v in best_vec]
