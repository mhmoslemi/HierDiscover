assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{obj}(a)=\frac{2\,n\;\max_k (a*a)_k}{(\sum_i a_i)^2},
\]

where `a*a` is the *linear* self‑convolution of a non‑negative vector `a`.
The objective is scale‑invariant, therefore we may (and do) normalise the
vector so that its entries sum to 1 – the factor `2·n` is then constant.

The search consists of three stages  

1. **Warm‑start** – a large collection of analytic shapes (tent,
   exponential, power‑law, Gaussian, …) is generated, normalised and the
   best one is kept.  
2. **Simulated‑annealing phase** – random mass transfers are performed.
   The move is accepted if it improves the objective, otherwise it is
   accepted with probability `exp(-(Δ)/T)`.  The temperature `T` is reduced
   linearly from a small fraction of the current objective to near 0.
   This lets the algorithm escape local minima that trapped the pure
   hill‑climber.
3. **Pure descent phase** – after the temperature has vanished we continue
   with a deterministic hill‑climber (identical to the original code) in
   order to polish the final solution.

At the end the best vector and its *symmetrised* counterpart
`(v + reverse(v))/2` are evaluated; the better of the two is returned.
All vectors are finally normalised to sum = 1, clipped to the limits
required by `evaluate_sequence`, and converted to a plain Python list.

The whole routine respects the 1000 s budget, runs on a single core and
uses only `numpy` (no external optimisation packages).

```python
"""
Simulated‑annealing optimiser for the convolution‑max objective.

The algorithm
1. builds a diverse set of analytic candidate shapes and keeps the best;
2. performs a temperature‑controlled random walk that moves small packets
   of mass between indices (accepting uphill moves with probability
   exp(-Δ/T));
3. finishes with a deterministic hill‑climber (only non‑increasing moves).

A final symmetrisation step is tried because the optimal step‑function
is usually symmetric.  The routine returns a list of non‑negative floats
ready for ``evaluate_sequence`` and is designed to run for the full
budget (≈1000 s) on a single CPU.
"""
import time
import math

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (non‑negative) vector."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return _objective_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int,
                eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst`.

    The operation updates both the coefficient vector `a` and its
    convolution `conv` in O(n) time, using the exact incremental formula.
    """
    n = a.shape[0]

    # linear part 2·a*Δ   (Δ = eps·(e_dst – e_src))
    conv[src : src + n] -= 2.0 * eps * a
    conv[dst : dst + n] += 2.0 * eps * a

    # quadratic part Δ·Δ
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum]  -= 2.0 * eps * eps

    # update the probability vector
    a[src] -= eps
    a[dst] += eps


# ----------------------------------------------------------------------
# Analytic shape generators (all produce non‑negative vectors of length n)
# ----------------------------------------------------------------------
def _make_candidate_shapes(n: int) -> list[np.ndarray]:
    """Return a list of normalised analytic shapes (unnormalised)."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Exponential decay from centre (symmetric)
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]                     # symmetric
    candidates.append(shape)

    # 5) Half‑cosine (already used as warm‑start but keep)
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussians centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) A few random Dirichlet draws
    rng = np.random.default_rng(0)
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # normalisation
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)

    return normalised


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the harness subtracts a small safety margin).

    Returns
    -------
    list[float]
        A non‑negative sequence ready for ``evaluate_sequence``.
    """
    # ------------------------------------------------------------------
    # Basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin: the harness adds ~0.5 s
    deadline = start_time + budget_s - 0.5
    total_budget = deadline - start_time

    # problem size ----------------------------------------------------
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        n = 4096                         # default size

    # ------------------------------------------------------------------
    # 1) Warm‑start – pick the best analytic shape
    # ------------------------------------------------------------------
    warm_starts = _make_candidate_shapes(n)

    # include the previous best vector if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if starter.ndim == 1 and starter.size == n:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                warm_starts.append(starter / starter.sum())
    except Exception:
        pass

    best_vec = None
    best_score = math.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_vec = ws.copy()

    # initialise current state -------------------------------------------------
    a = best_vec.copy()
    conv = _conv_fft(a)
    total = a.sum()                # ≡ 1 after normalisation
    n_obj = 2.0 * n / (total * total)   # factor = 2·n when total==1
    cur_score = best_score
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing phase (≈ 80 % of remaining budget)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)

    # temperature starts as a small percentage of the current objective
    T_start = cur_score * 0.05   # 5 % of the objective

    # split budget: SA -> 0.78, deterministic descent -> 0.20
    sa_deadline = start_time + 0.78 * total_budget

    it = 0
    while time.time() < sa_deadline:
        it += 1

        # pick a source index proportionally to its mass
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # fraction of source mass to move (large early, shrink later)
        remaining = deadline - time.time()
        frac_max = 0.30 * (remaining / total_budget) + 0.02
        eps = a[src] * rng.random() * frac_max
        if eps <= 0.0:
            continue

        before_score = cur_score

        # apply move
        _apply_move(a, conv, src, dst, eps)

        new_score = n_obj * conv.max()
        delta = new_score - before_score

        # linear temperature schedule
        T = T_start * (remaining / total_budget)

        # acceptance test
        if delta <= 0.0 or rng.random() < math.exp(-delta / (T + 1e-15)):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # reject – revert the move
            _apply_move(a, conv, dst, src, eps)

    # ------------------------------------------------------------------
    # 3) Pure hill‑climbing (no worsening moves) for the rest of the budget
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        remaining = deadline - time.time()
        frac_max = 0.30 * (remaining / total_budget) + 0.02
        eps = a[src] * rng.random() * frac_max
        if eps <= 0.0:
            continue

        before_score = cur_score

        _apply_move(a, conv, src, dst, eps)

        new_score = n_obj * conv.max()
        if new_score <= before_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time() - start_time:.2f}s] iter={it:,} new best={best_score:.6f}")
        else:
            # undo move
            _apply_move(a, conv, dst, src, eps)

    # ------------------------------------------------------------------
    # 4) Final symmetrisation attempt
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(n, 1.0 / n)

    # try a symmetric version
    sym = (best_vec + best_vec[::-1]) * 0.5
    sym /= sym.sum()
    sym_score = _objective(sym)
    if sym_score < best_score:
        best_vec = sym
        best_score = sym_score
        print(f"[{time.time() - start_time:.2f}s] Symmetrised solution improves to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the result (clipped to the limits required by evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    best_vec /= best_vec.sum()                     # final normalisation
    print(f"[{time.time() - start_time:.2f}s] Finished – best score={best_score:.6f}")
    return [float(v) for v in best_vec]
```
