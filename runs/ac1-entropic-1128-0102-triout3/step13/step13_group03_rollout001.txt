"""
Propose a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

* **Problem size** – we use n = 8192 (≈ twice the previous size) which still
  fits comfortably in memory and allows a richer shape.

* **Initial seeds** – a uniform vector, a parametric family
  a_i ∝ r**i * (i+1)**p (mixed with the uniform vector) and a few random
  perturbations of the current best.

* **Phase 1 – coarse parametric sweep** – explores a wide grid for r∈[0.5,0.999],
  p∈[‑6, 6] and mixtures λ∈{0, 0.25, 0.5, 0.75, 1}.   The best candidate is kept.

* **Phase 2 – smooth‑max Adam descent** – optimises a smoothed version of the
  “max‑convolution” term.  The temperature β grows from 5 up to 2000,
  focusing the gradient onto the true maximum.  Adam with projection onto the
  probability simplex is employed.  Several restarts (different seeds) are run
  until the remaining time is low.

* **Phase 3 – exact sub‑gradient descent** – once β is large we switch to an
  exact sub‑gradient of the max‑convolution (g_i = 2·a_{k∗‑i}) and take
  diminishing steps with projection.  This sharpens the solution near the
  optimum.

* **Phase 4 – stochastic hill‑climb** – random mass‑transfer moves
  (up to 15 % of the source mass) together with targeted moves that
  remove mass from the pair of indices responsible for the current maximal
  convolution entry.  Only improving moves are kept.

All phases respect the global deadline (default 1000 s).  The returned
sequence is normalised to sum = 1 and converted to a plain list of Python
floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, length_fft: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of size `length_fft`.
    The returned vector has length 2·len(a)‑1.
    """
    n = a.shape[0]
    if length_fft is None:
        length_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=length_fft)
    conv = np.fft.irfft(Af * Af, n=length_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _eval_fast(a: np.ndarray, length_fft: int) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)≈1.
    Returns np.inf if the sum is too tiny.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self(a, length_fft)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Smooth‑max (soft‑max) gradient for the “max‑conv” term
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, beta: float, L_conv: int, L_corr: int):
    """
    Returns (grad, conv) where
        conv = a * a  (full linear convolution)
        grad = ∂/∂a ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is obtained via correlation in the frequency domain.
    """
    conv = _conv_self(a, L_conv)                # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (stable)
    shifted = conv - max_c
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()               # shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)

    start = a.shape[0] - 1                     # n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max Adam descent
# ----------------------------------------------------------------------


def _smooth_adam_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    best_score: float,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type projected descent on the smooth‑max surrogate.
    The temperature β starts at 5 and is multiplied by 1.02 each iteration
    until `beta_max` is reached.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = best_score

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    beta = 5.0
    beta_mul = 1.02
    t = 0

    # schedule for the learning‑rate (starts relatively large then decays)
    base_lr = 0.8

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑adam it={t} β={beta:.1f} score={cur_score:.6f}")

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = base_lr / (1.0 + 5e-4 * t)               # slowly decaying LR
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – exact sub‑gradient descent (β already huge)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Performs a simple sub‑gradient descent on the true max‑convolution.
    The step size follows α_t = α₀ / sqrt(t) and the iterate is projected
    onto the simplex after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = best_score

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    alpha0 = 0.5 / n  # heuristic normalisation

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv)
        g = np.zeros_like(a)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        g[mask] = 2.0 * a[partner[mask]]

        # step‑size schedule
        alpha = alpha0 / math.sqrt(t)

        a = a - alpha * g
        a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={t} score={cur_score:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – stochastic hill‑climb (random + targeted moves)
# ----------------------------------------------------------------------


def _hill_climb(
    a0: np.ndarray,
    deadline: float,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves (up to ≈15 % of the source mass) combined with
    targeted moves that try to decrease the currently dominant convolution entry.
    Only moves that improve the exact objective are kept.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = best_score
    n = a.shape[0]

    L_fft = _next_pow2(2 * n - 1)

    rng = np.random.default_rng()
    step_factor = 0.15
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.6:
            # random transfer
            src = rng.integers(n)
            if a[src] <= 1e-15:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            # targeted transfer
            conv = _conv_self(a, L_fft)
            k_max = int(np.argmax(conv))
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            i = rng.integers(i_low, i_high + 1)
            j = k_max - i
            src = i if a[i] > a[j] else j
            if a[src] <= 1e-15:
                continue
            dst = rng.integers(n)
            if dst == i or dst == j:
                continue
            delta = a[src] * rng.random() * step_factor

        # propose move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        cur_val = _eval_fast(a_prop, L_fft)
        if cur_val < best_val:
            a = a_prop
            best_a = a.copy()
            best_val = cur_val
            step_factor = max(step_factor, 0.15)   # reset after success
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_val:.6f}")
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_val


# ----------------------------------------------------------------------
# Coarse parametric sweep (Phase 1)
# ----------------------------------------------------------------------


def _parametric_sweep(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore the family   a_i ∝ r**i · (i+1)**p   possibly mixed
    with the uniform vector.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)   # (i+1)
    r_grid = np.logspace(np.log10(0.5), np.log10(0.9999), num=30)
    p_grid = np.linspace(-6.0, 6.0, 49)
    lam_grid = (0.0, 0.25, 0.5, 0.75, 1.0)

    L_fft = _next_pow2(2 * n - 1)

    for r in r_grid:
        if time.time() >= deadline - 120:
            break
        geo = r ** np.arange(n, dtype=np.float64)
        for p in p_grid:
            if time.time() >= deadline - 120:
                break
            shape = geo * (idx ** p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The algorithm combines a parametric sweep, smooth‑max Adam
    descent, exact sub‑gradient descent and a final hill‑climb.  It respects
    the time budget and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – we use n = 8192 (≈ 2× previous) for a richer search
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Attempt to use a globally supplied starter if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep (coarse → refined)
    # ------------------------------------------------------------------
    if time.time() < deadline - 300:                     # reserve ~5 min
        best_seq, best_score = _parametric_sweep(
            n, uniform, deadline, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Phase 2 – multiple Adam‑style smooth‑max descent runs (different seeds)
    # ------------------------------------------------------------------
    descent_seeds = [seed + i for i in range(8)]        # a few more restarts
    for ds in descent_seeds:
        if time.time() >= deadline - 180:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        # tiny random perturbation (scale 1e-3 of current entry)
        a0 += 1e-3 * rng.standard_normal(n) * a0
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 30000
        a_opt, sc_opt = _smooth_adam_descent(
            a0,
            max_iters,
            deadline - 30,          # keep a small buffer for later phases
            best_score,
            beta_max=2000.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑adam improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent (sharpening)
    # ------------------------------------------------------------------
    if time.time() < deadline - 90:
        max_iters = 4000
        a_opt, sc_opt = _subgradient_descent(
            best_seq,
            max_iters,
            deadline - 30,
            best_score,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑gradient improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final stochastic hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _hill_climb(best_seq, deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Optional sanity check with the official evaluator
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
