The objective to minimise is  

\[
\Phi(a)=\frac{2\,n\,\max (a\ast a)}{(\sum a)^2},
\]

where `a` is a non‑negative vector, `n=len(a)` and `a∗a` is the linear
self‑convolution.  
Because the numerator and denominator are both quadratic in `a`,
any positive scaling of the vector cancels out, therefore we may restrict
the search to the probability simplex  

\[
\Delta=\{a\;|\;a_i\ge 0,\;\sum_i a_i=1\}.
\]

The problem is non‑convex because of the `max` operator.
The previous submission used a projected Adam optimiser on the exact
sub‑gradient of the *single* maximal entry.
Here we improve the search in three ways:

1. **Smoothed sub‑gradient** – instead of using only the unique maximal
   entry, we take a soft‑max weighted combination of the *K* largest
   convolution values.  The smoothing parameter `α` is increased
   linearly during the run, which gives a gentle landscape at the
   beginning (helps escaping shallow local minima) and focuses on the
   true max later on.

2. **Multiple random restarts** – we start the optimiser not only from
   the analytically generated shapes but also from several random
   Dirichlet samples.  Every restart receives an equal share of the
   time budget; the best result is kept.

3. **Final polish** – a cheap greedy hill‑climb (exact same code as in
   the reference) and a short low‑learning‑rate Adam pass are applied to
   the best candidate found by the restarts.

The whole procedure respects the 1000 s budget, reports intermediate
scores and finally returns a plain list of floats that can be fed into
`evaluate_sequence`.

```python
"""
Minimum‑max‑convolution step‑function optimisation.

Algorithm
---------
* Warm‑start with a collection of analytic shapes (tent, cosine, half‑circle,
  Gaussian, power‑tents, exponentials) and with a few random Dirichlet vectors.
* For every start run a projected Adam optimiser on the *smoothed* sub‑gradient
  of max(a∗a).  The smoothing is a soft‑max over the K largest convolution
  entries; the temperature (α) grows linearly so the optimisation starts
  smooth and finishes with the exact sub‑gradient.
* Keep the best iterate seen during the Adam run.
* After all restarts perform a cheap greedy hill‑climb and a short low‑LR
  Adam refinement on the overall best vector.
* Return the normalised vector as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    (The sum constraint is enforced externally.)
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Warm‑starts – analytic families
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)                # zero at the ends
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) uniform (flat) – useful as a baseline
    candidates.append(np.ones_like(x))

    # Normalise each candidate to sum = 1 and keep only the non‑negative ones.
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Smoothed sub‑gradient of max‑convolution
# ----------------------------------------------------------------------
def _smoothed_subgradient(a: np.ndarray,
                          conv: np.ndarray,
                          top_k: int = 5,
                          alpha: float = 10.0) -> np.ndarray:
    """
    Weighted sub‑gradient using a soft‑max over the `top_k` largest
    convolution entries.  For α → ∞ the weights collapse to the true
    sub‑gradient of the max entry.
    """
    n = a.shape[0]
    # Determine the indices of the K largest entries (unsorted)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]

    # Soft‑max weights (more weight on the true maximum)
    max_val = conv.max()
    w = np.exp(alpha * (conv[idxs] - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # Build the weighted gradient: g_i = 2 * Σ_k w_k * a_{k-i}
    grad = np.zeros_like(a)
    for k, wk in zip(idxs, w):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        grad[i] += 2.0 * wk * a[k - i]
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (smoothed sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   top_k: int = 5,
                   alpha_start: float = 5.0,
                   alpha_end: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex with a smoothed sub‑gradient.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution & smoothed gradient -----
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # linearly increase the smoothing parameter α
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        grad = _smoothed_subgradient(a, conv, top_k=top_k, alpha=alpha)
        grad_obj = 2.0 * n * grad

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # ----- projection onto the probability simplex -----
        a = _project_simplex(a)

        # ----- objective tracking -----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cur_score:.6f}")

        # optional learning‑rate decay (helps in the long run)
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing polish (identical to reference)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepting only improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:                     # greedy acceptance
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain ``list[float]``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – power of two for FFT speed.
    # Larger n gives a finer discretisation but a higher per‑iteration cost.
    # 4096 is a good compromise; 8192 is tried if enough time is left.
    # ------------------------------------------------------------------
    remaining_total = deadline - start_time
    n = 4096
    if remaining_total > 600:          # we have plenty of time → try a larger grid
        n = 8192

    # ------------------------------------------------------------------
    # 1) Warm‑starts – analytic shapes + a few random Dirichlet vectors.
    # ------------------------------------------------------------------
    warm_starts = _make_warm_starts(n)

    # add a few random Dirichlet starts (parameter α controls sparsity)
    rng = np.random.default_rng(seed + 1234)
    for _ in range(4):
        # α=0.5 creates somewhat sparse vectors, α=1.0 is uniform‑like
        alpha_dir = 0.6
        w = rng.dirichlet(np.full(n, alpha_dir))
        warm_starts.append(w)

    # include a globally supplied construction if present
    try:
        global height_sequence_1
        h1 = np.asarray(height_sequence_1, dtype=float).ravel()
        if h1.size > 0:
            xs = np.linspace(0, 1, h1.size)
            xq = np.linspace(0, 1, n)
            h1_resampled = np.interp(xq, xs, h1)
            if np.all(h1_resampled >= 0):
                h1_resampled = np.maximum(h1_resampled, 0.0)
                s = h1_resampled.sum()
                if s > 0:
                    warm_starts.append(h1_resampled / s)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Pick the best warm‑start according to the exact objective.
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for ws in warm_starts:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    a0 = best_a
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Multiple Adam runs (random restarts) – each gets an equal time share.
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        # allocate 70 % of the remaining budget to Adam runs,
        # split equally among a handful of restarts.
        adam_budget = time_left * 0.70
        max_restarts = max(1, int(adam_budget // 120.0))   # at least ~2 min per run
        per_restart = adam_budget / max_restarts

        for r in range(max_restarts):
            # choose a start: first run uses the warm‑start, later runs
            # get a fresh random Dirichlet vector.
            if r == 0:
                start_vec = a0.copy()
            else:
                start_vec = np.random.default_rng(seed + r + 999).dirichlet(
                    np.full(n, 0.6))

            rs_deadline = time.time() + per_restart
            a_opt, score_opt = _adam_optimize(start_vec,
                                               rs_deadline,
                                               start_time,
                                               lr=0.02,
                                               top_k=7,
                                               alpha_start=5.0,
                                               alpha_end=25.0)
            if score_opt < best_score:
                best_score = score_opt
                a0 = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] Restart {r+1}/{max_restarts} "
                      f"new best={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Light hill‑climb polish (as much time as left).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = _hill_climb(a0, deadline, start_time)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {score_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Final fine‑tune: short low‑LR Adam pass (no smoothing) to
    #    clean up any remaining small irregularities.
    # ------------------------------------------------------------------
    final_deadline = time.time() + min(5.0, deadline - time.time())
    if final_deadline > time.time():
        a_opt, score_opt = _adam_optimize(a0,
                                          final_deadline,
                                          start_time,
                                          lr=0.005,
                                          top_k=1,               # exact max sub‑gradient
                                          alpha_start=30.0,
                                          alpha_end=30.0)
        if score_opt < best_score:
            best_score = score_opt
            a0 = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final Adam fine‑tune best={score_opt:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalisation & conversion to plain list.
    # ------------------------------------------------------------------
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total > 0.0:
        a0 /= total
    else:
        a0 = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n}")
    return [float(v) for v in a0]
```
