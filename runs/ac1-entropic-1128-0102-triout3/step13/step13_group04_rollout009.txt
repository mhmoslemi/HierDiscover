def _grad_smooth(a, w, L):

"""
Propose a non‑negative step function that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  Initialise from the best known construction (if available) or from a
    uniform distribution.

2️⃣  A long Adam‑style optimisation that minimises a *smooth* approximation
    of the max‑convolution.
    The smooth max is realised by a soft‑max over the convolution vector with a
    temperature (α) that is gradually increased.  The gradient of the smooth
    objective is obtained analytically via an FFT‑based convolution, and the
    iterate is projected onto the probability simplex after each step.

3️⃣  A final stochastic hill‑climb that transfers tiny fractions of mass
    between two positions and keeps improving moves.

All operations respect the 1000 s budget, use FFT for speed and keep the
sequence normalised (∑a = 1) so that the objective reduces to minimising
max(conv(a,a)).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the *real* result of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    prod = Af * Af
    conv = np.fft.irfft(prod, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth max‑convolution
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights that approximate the arg‑max of `conv`.
    `alpha` controls the temperature (larger → sharper).
    The returned array has the same length as `conv` and sums to 1.
    """
    maxc = conv.max()
    # shift by the max to avoid overflow
    exps = np.exp(alpha * (conv - maxc))
    # protect against NaN in degenerate cases
    if np.all(np.isfinite(exps)):
        w = exps / exps.sum()
    else:                     # fallback – uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    return w


def _grad_smooth_max(a: np.ndarray, w: np.ndarray, L2: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)[k]  w.r.t. a.
    Implements 2·(w * a)[:n] using FFT.
    `w` has length 2·n‑1, `a` has length n, `L2` is a power‑of‑two ≥ n+(2n‑1)-1.
    """
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L2)
    Wf = np.fft.rfft(w, n=L2)
    grad_full = np.fft.irfft(Af * Wf, n=L2)
    # Only the first n entries correspond to the gradient w.r.t. a_i
    grad = 2.0 * grad_full[:n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser that minimises a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_smooth_opt(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_score_initial: float):
    """
    Adam optimiser on the simplex with a smooth max objective.
    Returns (best_a, best_score) where `best_score` is the *true*
    evaluation value 2·n·max(conv).
    """
    a = a0.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)          # for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # for gradient conv (w*a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # schedule: (iterations, alpha, learning_rate)
    schedule = [
        (10_000, 0.8, 0.5),
        (20_000, 2.0, 0.3),
        (40_000, 5.0, 0.2),
        (80_000, 10.0, 0.12),
        (120_000, 20.0, 0.07),
    ]

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = best_score_initial
    it = 0
    rng = np.random.default_rng()

    for phase_iters, alpha, lr in schedule:
        target_it = it + phase_iters
        while it < target_it and time.time() < deadline:
            it += 1

            # current objective (true, not smoothed)
            conv = _conv_self(a, L_self)
            cur_max = conv.max()
            cur_score = float(2.0 * n * cur_max)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                      f"α={alpha:.2f} score={cur_score:.6f}")

            # smooth weights and gradient
            w = _smooth_weights(conv, alpha)          # length 2n‑1, sum=1
            grad = _grad_smooth_max(a, w, L_grad)

            # Adam update in the *a–space* (followed by projection)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _project_simplex(a)

            # occasional jitter to escape plateaus
            if it % 5000 == 0:
                jitter = 1e-6 * rng.standard_normal(n) * a
                a = a + jitter
                a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that attempts tiny random mass transfers.
    Returns the best sequence discovered and its true score.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_self).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 1 % of the source mass
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate the true objective
        cur_score = float(2.0 * n * _conv_self(a_prop, L_self).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score
