"""
Optimiser for the step‑function challenge.

The algorithm follows four stages within the allotted 1000 s budget:

1️⃣  **Coarse parametric sweep** – explores geometric, power and beta
    families on a fine grid; keeps the best sequence.

2️⃣  **Random Dirichlet initialisers** – draws a handful of random
    probability vectors (different concentrations) to diversify the
    search.

3️⃣  **Smooth‑max continuation** – minimises a differentiable proxy of
    the max‑convolution using projected gradient descent with an
    increasing temperature τ.  The iterate is projected onto the simplex
    after every step.

4️⃣  **Simulated‑annealing refinement** – starting from the best vector
    obtained so far, a Metropolis Monte‑Carlo scheme repeatedly moves a
    small amount of mass from a “high‑gradient’’ index to a “low‑gradient’’
    one.  The acceptance probability is governed by a linearly cooling
    temperature.  The convolution is updated incrementally (O(n) per
    move) using the exact algebraic formula for a mass‑transfer
    operation, which is much faster than recomputing an FFT each step.
    Periodically the whole convolution is recomputed and a tiny random
    Dirichlet perturbation is mixed in to avoid stagnation.

The routine respects the time budget, prints progress information and
finally returns a plain list of Python floats suitable for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (FFT based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helper (scale‑invariant version)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta)
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray,
                    best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    uniform = np.full(n, 1.0 / n)

    # ---- geometric ----------------------------------------------------
    eps_grid = np.logspace(-5, -1, 400)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 121)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 81)
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    return best_seq, best_score

def _random_dirichlet_trials(n: int, best_seq: np.ndarray,

            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")

# 3️⃣  Smooth‑max continuation (projected gradient descent)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                       start_time: float,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """Minimise a smooth proxy of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)       # for self‑convolution
    τ = 500.0
    τ_factor = 1.25
    η0 = 0.04
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} score={cur_score:.6f}")

        max_c = conv.max()
        w = np.exp(τ * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        a_rev = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(w, n=_next_pow2(3 * n - 2))
                              * np.fft.rfft(a_rev, n=_next_pow2(3 * n - 2)),
                              n=_next_pow2(3 * n - 2))[:n]
        grad = 2.0 * conv_w

        η = η0 / np.sqrt(it)
        a_new = a - η * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        if it % 4000 == 0:
            τ = min(τ * τ_factor, 1e9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Simulated‑annealing refinement (incremental convolution update)
# ----------------------------------------------------------------------
def _inc_conv_update(conv: np.ndarray, pad: np.ndarray,
                     src: int, dst: int, eps: float, n: int) -> np.ndarray:
    """
    Return the convolution after moving mass `eps` from index `src`
    to index `dst`.  `pad` is the zero‑padded version of the current
    sequence (length 3·n‑2) and `conv` is the current self‑convolution.
    The update is performed in O(n) time.
    """
    L = 2 * n - 1
    shift_src = pad[src:src + L]          # a shifted by +src
    shift_dst = pad[dst:dst + L]          # a shifted by +dst
    new_conv = conv + 2.0 * eps * (shift_dst - shift_src)

    # δ*δ term (self‑pair corrections)
    new_conv[2 * src] += eps * eps
    new_conv[2 * dst] += eps * eps
    new_conv[src + dst] -= 2.0 * eps * eps
    return new_conv


def _simulated_annealing(initial_seq: np.ndarray,
                         start_time: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Metropolis annealing that transfers tiny amounts of mass between
    indices.  The convolution is updated incrementally; temperature is
    linearly decreased to zero.  Periodic full recomputation and a tiny
    random Dirichlet mix keep the search from stagnating.
    """
    n = initial_seq.shape[0]
    a = initial_seq.copy()
    conv = _conv_self(a)
    cur_score = float(2.0 * n * conv.max())
    best_a = a.copy()
    best_score = cur_score

    # padded representation of `a` (zeros on both sides)
    pad = np.concatenate((np.zeros(n - 1), a, np.zeros(n - 1)))

    rng = np.random.default_rng()
    total_time = deadline - start_time
    lower_bound = 2.0 * n / (2 * n - 1)    # theoretical limit ≈ 1
    T0 = max(0.05, (cur_score - lower_bound) * 0.5)  # initial temperature
    iteration = 0
    refresh_interval = 5000
    mix_every = 30.0                      # seconds
    next_mix = time.time() + mix_every

    while time.time() < deadline:
        elapsed = time.time() - start_time
        fraction = elapsed / total_time
        T = max(1e-12, T0 * (1.0 - fraction))

        # ---- pick source with non‑negligible mass --------------------
        src = rng.integers(n)
        tries = 0
        while a[src] <= 1e-15 and tries < 10:
            src = rng.integers(n)
            tries += 1
        if a[src] <= 1e-15:
            continue

        # ---- pick destination ----------------------------------------
        dst = rng.integers(n)
        if dst == src:
            continue

        # ---- choose transfer amount ----------------------------------
        max_frac = min(1.0, T * 5.0)        # larger moves early, tiny later
        eps = a[src] * rng.random() * max_frac
        if eps <= 0.0:
            continue

        # ---- incremental convolution update ---------------------------
        new_conv = _inc_conv_update(conv, pad, src, dst, eps, n)
        new_conv = np.maximum(new_conv, 0.0)           # numerical safety
        new_score = float(2.0 * n * new_conv.max())
        delta = new_score - cur_score

        accept = False
        if delta <= 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            # update the sequence
            a[src] -= eps
            a[dst] += eps

            # keep padded version in sync
            pad[n - 1 + src] = a[src]
            pad[n - 1 + dst] = a[dst]

            conv = new_conv
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] SA new best score {best_score:.6f}")

        iteration += 1

        # ---- occasional full recomputation to avoid drift -------------
        if iteration % refresh_interval == 0:
            conv = _conv_self(a)
            cur_score = float(2.0 * n * conv.max())

        # ---- occasional random Dirichlet mixing -----------------------
        if time.time() >= next_mix:
            alpha = 0.02                      # mixing strength
            c = rng.uniform(0.1, 5.0)
            rnd = rng.gamma(shape=c, scale=1.0, size=n)
            rnd /= rnd.sum()
            a = (1.0 - alpha) * a + alpha * rnd
            a = np.maximum(a, 0.0)
            a /= a.sum()
            conv = _conv_self(a)
            cur_score = float(2.0 * n * conv.max())
            pad = np.concatenate((np.zeros(n - 1), a, np.zeros(n - 1)))
            next_mix = time.time() + mix_every

    return best_a, best_score

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
