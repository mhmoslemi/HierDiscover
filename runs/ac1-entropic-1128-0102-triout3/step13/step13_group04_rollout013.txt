"""
Propose a step‑function (non‑negative list) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm improves the previous solution in three ways:

1️⃣  Uses a larger problem size (n≈16384) – the theoretical lower bound
    (≈1.0) is approached faster when more degrees of freedom are available.

2️⃣  Searches a richer family of “U‑shaped’’ distributions derived from the
    symmetric Beta(α,α) density (α∈(0,1)).  The family includes the arcsine
    shape (α=0.5) which concentrates mass at the ends and reduces the
    central autocorrelation peak.  Each shape is mixed with the uniform
    distribution to explore the whole spectrum between flat and highly‑U‑shaped.

3️⃣  Refines the best candidate with a projected Adam optimiser that uses a
    *multi‑active* sub‑gradient (the average of the gradients of all shifts
    whose convolution value is within a tiny tolerance of the maximum).  This
    gives a more accurate descent direction than the single‑active version.
    Finally a stochastic hill‑climb makes tiny mass‑transfer tweaks.

All operations respect the 1000 s budget, use FFT‑based convolutions,
and keep the sequence on the probability simplex (sum = 1, non‑negative).
The function returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σ w = 1 }.
    Returns w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _grad_multi_active(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Average sub‑gradient of max(conv) obtained from *all* active shifts,
    i.e. the shifts whose convolution value is within `eps` (absolute) of the
    maximum.
    """
    n = a.shape[0]
    max_c = conv.max()
    # treat tiny eps relative to max if max is not tiny
    tol = max(eps, max_c * 1e-12)
    active = np.where(conv >= max_c - tol)[0]
    grad = np.zeros_like(a)
    if len(active) == 0:
        return grad
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]       # contribution of this shift
    grad /= len(active)          # average over the active set
    return grad


# ----------------------------------------------------------------------
# Optimiser (projected Adam with multi‑active sub‑gradient)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score_initial: float,
                   max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex, using the averaged sub‑gradient of the
    max‑convolution.  Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        it += 1

        # current convolution and score
        conv = _conv_self(a, L_fft)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a)=1 after projection

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # multi‑active sub‑gradient
        grad = _grad_multi_active(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto simplex
        a = _project_simplex(a)

        # occasional jitter to escape plateaus
        if it - last_improve > 5_000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: move a tiny random fraction of mass from one index
    to another; keep the move only if it improves the true objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction of source mass, up to 1 %
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------
def _beta_u_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on n points (U‑shaped for α<1).
    Returns a non‑negative vector (not normalised).
    """
    # centre points of bins
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid zero/one exactly – α>0 guarantees finiteness
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return shape


def _geometric_shape(n: int, eps: float) -> np.ndarray:
    """
    Pure decreasing geometric shape a_i ∝ (1‑eps)^i.
    eps must be >0.
    """
    r = 1.0 - eps
    idx = np.arange(n, dtype=np.float64)
    geo = np.power(r, idx)
    return geo

            if cur < best_score:

                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.2e} λ={lam:.2f} score={cur:.6f}")
