"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The routine works in three stages:

1️⃣  **Fast parametric sweep** – explores families of geometric‑power shapes
    (a_i ≈ r^i·(i+1)^p) and mixtures with the uniform distribution.
2️⃣  **Simulated‑annealing (SA) on the simplex** – starting from the best
    candidate of the sweep, repeatedly moves a random fraction of mass from
    one coordinate to another.  Moves that increase the objective are accepted
    with probability exp(‑Δ/T); the temperature T is exponentially cooled.
    The move size is tied to the temperature so that early steps are large
    and later steps become microscopic.  A simple “re‑heat” is performed if
    no improvement occurs for a long time.
3️⃣  **Microscopic polish** – a few deterministic sub‑gradient steps that
    transfer a tiny amount of mass from the coordinate that contributes most
    to the current maximal convolution entry to the one that contributes
    least.

All phases respect the 1000 s time budget, keep the vector on the probability
simplex (non‑negative, sum = 1) and finally return the best sequence found as
a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length L (≥ 2·len(a)‑1).
    Returns the full convolution (length 2·len(a)‑1).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray, L: int) -> float:
    """
    Exact objective for a vector that already satisfies sum(a) ≈ 1:
        2 * n * max(conv(a, a))
    """
    n = a.shape[0]
    conv = _conv_self_fft(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (w≥0, sum(w)=1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w
