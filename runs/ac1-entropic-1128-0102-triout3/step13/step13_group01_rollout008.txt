#!/usr/bin/env python
"""
Improved optimisation of a non‑negative step function for

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine builds a small pool of analytical candidates (uniform,
beta‑type, arcsine) and then refines the best one with a **soft‑max
gradient descent** on the probability simplex (Adam with projection) .
A soft‑max (log‑sum‑exp) surrogate of the max‑convolution provides a
smooth gradient that uses *all* convolution entries, not only the few
largest ones.  The gradient is obtained by a single FFT‑based
cross‑correlation, so each iteration costs only a few FFTs on a length
≈3·n vector.  After the main descent a light simulated‑annealing
polish is performed.

The whole search respects the 1000 s budget and returns a plain list of
non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility for FFT sizes
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Return linear convolution a*a (full length 2·n‑1) using FFT of size L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(seq, L)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:                     # should never happen
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:   # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised Beta(a,b) on {0,…,n‑1}, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_arcsine_shape(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – the arcsine distribution – normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n
    # avoid exact 0/1 at the ends
    shape = 1.0 / np.sqrt(xs * (1.0 - xs))
    total = shape.sum()
    shape /= total
    return shape


def _make_uniform_shape(n: int) -> np.ndarray:
    """Trivial uniform shape."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Soft‑max based gradient descent on the simplex (Adam + projection)
# ----------------------------------------------------------------------
def _softmax_adam(a0: np.ndarray,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using a smooth approximation of max(conv) and
    Adam on the simplex.  Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT length that can accommodate both the self‑convolution (2·n‑1)
    # and the correlation with a weight vector (2·n‑1 + n‑1 = 3·n‑2).
    L = _next_pow2(3 * n)

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature (alpha) – start gentle, increase gradually
    alpha = 2.0
    it = 0

    while time.time() < deadline:
        it += 1

        # ------- FFT of the current point ---------------------------------
        A = np.fft.rfft(a, n=L)

        # ------- self‑convolution (needed for the surrogate) -------------
        conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        # ------- soft‑max weights -----------------------------------------
        maxc = conv.max()
        # exponentiate the (conv - max) to keep numbers ≤ 1
        exp_term = np.exp(alpha * (conv - maxc))
        w = exp_term / exp_term.sum()          # shape: (2·n‑1,)

        # ------- gradient of the smooth max -------------------------------
        W = np.fft.rfft(w, n=L)
        # correlation of w with a gives ∂/∂a Σ w_k conv_k
        grad = 2.0 * np.fft.irfft(np.conj(W) * A, n=L)[:n]

        # ------- Adam update ----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple decaying step‑size (the scale cancels because of simplex projection)
        step = 0.08 / (1.0 + it * 1e-4)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ------- projection onto the simplex -------------------------------
        a = _project_simplex(a)

        # ------- occasional exact evaluation -------------------------------
        if it % 25 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,}  best={cur_score:.6f}")

        # ------- anneal the temperature ------------------------------------
        if it % 200 == 0:
            alpha = min(alpha * 1.07, 5000.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same as the previous code)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Randomly shift tiny packets of mass, Metropolis acceptance."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # adapt the number of steps to the remaining time
    max_steps = 800_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.03 * (1e-6 / 0.03) ** (it / max_steps)

        src = rng.choice(n, p=a)           # more mass → more likely to move
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score
