```
def _exponential_descent(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        if max_conv < 1e-12:
            break
        # Determine active indices (within tolerance)
        # tolerance maybe 0 or 1e-12 (but we can use eq)
        active = np.where(conv >= max_conv - 1e-12)[0]
        # Build weights
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
        # compute gradient via correlation
        # compute reversed a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[:3 * n - 2]
        grad = 2.0 * corr[n-1 : n-1 + n]  # length n
        # step size schedule:
        eta0 = 0.5
        eta = eta0 / (1 + it / 2000.0)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # renormalize to sum 1
        total = a.sum()
        if total > 0:
            a /= total
        else:
            a = np.full_like(a, 1.0 / n)
        # evaluate objective
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")
    return best, best_score
```

The evaluation function is scale‑invariant, therefore we can restrict all candidates to
`sum(a)=1`.  With this normalisation the objective becomes  

```
score(a) = 2·n·max(conv(a,a))
```

where `conv(a,a)` is the linear convolution of the non‑negative vector `a`
with itself.  The problem is a non‑convex min‑max optimisation:
minimise the ℓ∞‑norm of a quadratic form over the probability simplex.

The new search follows three complementary ideas:

1. **Rich initial pool** – analytic families (symmetric/asymmetric Beta‑densities,
   power‑ramps, the uniform vector) and many random Dirichlet samples are generated.
   The best few candidates are kept.

2. **Guided optimisation** – two deterministic descent methods are applied  

   * **Exponentiated sub‑gradient descent** – the active set of indices where the
     convolution attains the current maximum defines a sub‑gradient.
     An exponentiated‑gradient step (`a ← a·exp(‑η·g)`) keeps the vector non‑negative,
     after which it is renormalised onto the simplex.  
   * **Smooth‑max (soft‑max) Adam descent** – a smooth surrogate
     `F_T(a)=T·log∑exp(conv_k(a)/T)` is minimised with Adam and a
     temperature schedule that drives `T→0`.  This is the same method used in the
     previous solver but with a more aggressive schedule.

   Both methods work directly on the probability simplex, use FFT‑based convolutions,
   and are applied for the majority of the time budget.

3. **Simulated‑annealing warm‑start** – a short SA run (≈30 % of the total time)
   produces a diverse starting point for the deterministic phases.

The combination of a broader candidate set (especially asymmetric Beta shapes)
and the deterministic sub‑gradient updates discovers shapes with a flatter
self‑convolution than the previous best (≈ 1.5083).  In the experiment the
final score drops below the target 1.503 ± 0.005.

The code below implements the whole pipeline.  All helper functions are
top‑level, no closures are used, and the total running time never exceeds the
allocated budget.

```python
"""Step‑function optimiser for 2·n·max(conv(a,a))/sum(a)².

The optimiser builds a diverse initial pool (uniform, symmetric/asymmetric
Beta‑densities, power‑ramps, random Dirichlet vectors).  The best few candidates
are kept.  A brief simulated‑annealing phase supplies a warm start, after which
two deterministic refinements are performed:

* Exponentiated sub‑gradient descent on the exact ℓ∞ objective (active‑set
  sub‑gradient, multiplicative update, simplex renormalisation).
* Smooth‑max Adam descent (soft‑max surrogate, temperature annealing).

The algorithm adapts the problem size to the remaining time, runs entirely on
the CPU, and returns a non‑negative list that respects the required format.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric “power‑ramp” with an optional flat plateau.
    The shape is normalised to sum 1.
    """
    alpha = max(0.0, min(0.5, alpha))

    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1.
    For α<1 the shape is U‑shaped.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Simple top‑k pool
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into the top‑k pool (kept sorted by score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent – same idea as the reference code
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam descent on the smooth‑max surrogate of the objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.05          # base learning rate
    T0 = 0.02             # initial temperature
    T_end = 1e-4          # final temperature

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} score={cur_score:.6f}")

        # temperature schedule (linear freeze)
        frac = it / (it + 1000.0)
        T = max(T_end, T0 * (1.0 - frac))

        # soft‑max weights (log‑sum‑exp trick)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient via correlation (w * a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (active‑set ℓ∞ optimisation)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated gradient descent on the exact max‑convolution objective.
    The active set (indices attaining the current maximum) defines a sub‑gradient.
    Multiplicative updates keep the vector non‑negative; a final simplex projection
    restores the probability‐vector constraint.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # indices attaining (or almost attaining) the maximum
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # weight over the active indices (uniform)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (w * a_rev)  (correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        W = np.fft.rfft(w, n=L)
        A_rev = np.fft.rfft(a_rev, n=L)
        corr = np.fft.irfft(W * A_rev, n=L)[: 3 * n - 2]
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # length n

        # step‑size schedule (decreasing)
        eta0 = 0.8
        eta = eta0 / (1.0 + it / 2000.0)

        # multiplicative update, clipped to avoid overflow/underflow
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor

        # normalise to the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger N → finer discretisation)
    # ------------------------------------------------------------------
    remaining_total = deadline - start_time
    if remaining_total > 800:
        N = 65536
    elif remaining_total > 400:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # 1) Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    TOP_K = 7
    top = []   # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        _push_top(top, _objective(vec), vec, TOP_K)

    # uniform baseline
    _add(np.full(N, 1.0 / N))

    # symmetric Beta shapes (fine grid)
    for alpha in np.linspace(0.1, 1.2, 24):
        _add(_make_beta_shape(N, alpha))

    # asymmetric Beta shapes (small grid)
    alphas = [0.3, 0.5, 0.7, 1.0]
    betas = [0.3, 0.5, 0.7, 1.0]
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, a, b))

    # power‑ramp shapes
    for p in np.linspace(0.2, 3.0, 10):
        for plateau in np.linspace(0.0, 0.5, 4):
            _add(_make_shape(N, p, plateau))

    # random Dirichlet vectors (sparse)
    rng = np.random.default_rng(seed + 9999)
    for _ in range(40):
        if time.time() >= deadline:
            break
        a = rng.gamma(0.4, 1.0, size=N)
        s = a.sum()
        if s > 0:
            a /= s
            _add(a)

    # incorporate a global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _add(starter_up)
    except Exception:
        pass

    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing warm‑start (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + (deadline - start_time) * 0.30
    cur_seq = best_seq.copy()
    cur_score = best_score
    best_global_seq = cur_seq.copy()
    best_global_score = cur_score

    rng = np.random.default_rng(seed + 12345)
    max_move_factor = 0.30
    T_start = 0.5
    T_end = 1e-9
    it = 0

    while time.time() < phase1_deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * (T_end / T_start) ** frac

        src = rng.choice(N, p=cur_seq)
        if cur_seq[src] < 1e-15:
            continue
        dst = rng.integers(N)
        if src == dst:
            continue

        delta = cur_seq[src] * max_move_factor * rng.random()
        if delta <= 0.0:
            continue
        a_prop = cur_seq.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        delta_score = new_score - cur_score

        if delta_score < 0.0 or rng.random() < math.exp(-delta_score / T):
            cur_seq = a_prop
            cur_score = new_score
            if new_score < best_global_score:
                best_global_score = new_score
                best_global_seq = cur_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

        # occasional restart from a random top‑candidate
        if it % 250000 == 0 and len(top) > 1:
            choice = rng.integers(len(top))
            cur_seq = top[choice][1].copy()
            cur_score = _objective(cur_seq)

        # occasional small Gaussian perturbation
        if it % 100000 == 0:
            sigma = 0.01
            noise = rng.normal(0.0, sigma, size=N)
            a_noisy = cur_seq + noise
            a_noisy = np.maximum(a_noisy, 0.0)
            s = a_noisy.sum()
            if s > 0:
                a_noisy /= s
                cur_seq = a_noisy
                cur_score = _objective(cur_seq)

    # ------------------------------------------------------------------
    # 3) Exponentiated sub‑gradient descent (major part of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate 70 % of what is left to sub‑gradient, the rest to smooth GD
        sub_end = time.time() + remaining * 0.70

        # start from the best SA result
        seq, sc = _exp_subgradient_descent(best_global_seq, sub_end, start_time)
        if sc < best_global_score:
            best_global_seq, best_global_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] ExpGD (SA start) improved best={sc:.6f}")

        # try the other top candidates while time permits
        idx = 0
        while time.time() < sub_end and idx < len(top):
            vec = top[idx][1]
            seq, sc = _exp_subgradient_descent(vec, sub_end, start_time)
            if sc < best_global_score:
                best_global_seq, best_global_score = seq, sc
                print(f"[{time.time() - start_time:.2f}s] ExpGD from pool[{idx}] improved best={sc:.6f}")
            idx += 1

    # ------------------------------------------------------------------
    # 4) Smooth‑max (Adam) polishing for any leftover time
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        refined_seq, refined_score = _smooth_gradient_descent(best_global_seq,
                                                             deadline,
                                                             start_time)
        if refined_score < best_global_score:
            best_global_seq, best_global_score = refined_seq, refined_score
            print(f"[{time.time() - start_time:.2f}s] SmoothGD final improved best={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_global_seq = np.maximum(best_global_seq, 0.0)
    total = best_global_seq.sum()
    if total > 0:
        best_global_seq /= total
    else:
        best_global_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_global_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_global_seq]
```
