"""
Propose a low‑objective step function.

Algorithm Overview
------------------
1. **Warm‑start** – generate a collection of analytic shapes (power‑tents,
   exponentials, power‑law, Gaussian, etc.) together with the previously
   discovered best vector (``height_sequence_1`` if available).  All candidates
   are normalised to sum to 1 and the one with the smallest exact objective
   (computed via FFT) is kept.

2. **Targeted hill‑climbing + simulated annealing** – repeatedly move a
   small amount of mass ``eps`` from a source index to a destination index.
   The source is chosen proportionally to its contribution to the current
   maximal convolution entry (i.e. ``a[i] * a[k_max‑i]``).  The destination is
   biased towards low‑mass indices.  Moves that improve the objective are
   always accepted; worsening moves are accepted with probability
   ``exp(-(Δ)/T)`` where the temperature ``T`` drops linearly from ``0.02`` to
   ``1e‑5``.  This allows escaping local minima.

3. **Periodic drift correction and smoothing** – every 2·10⁵ moves the
   convolution is recomputed from scratch (FFT) to avoid accumulated
   numerical error.  Every 5·10⁵ moves a small Gaussian smoothing step is
   tried; if it yields a better objective the smoothed vector replaces the
   current one.

The search runs until the time budget (``budget_s``) is exhausted or a
pre‑defined target (objective < 1.503) is reached.  The best vector found
is returned as a plain Python list of float.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (total ** 2))


def _apply_move(a: np.ndarray, conv: np.ndarray,
                src: int, dst: int, eps: float) -> None:
    """
    Transfer a mass `eps` from index `src` to `dst` (in‑place).

    Parameters
    ----------
    a   : (n,)   current coefficient vector (will be modified)
    conv: (2n‑1) current self‑convolution (will be updated)
    src,dst : indices, src ≠ dst
    eps : amount to move (0 ≤ eps ≤ a[src])
    """
    n = a.shape[0]

    # linear part: 2*eps*a shifted by src (subtraction) and dst (addition)
    conv[src:src + n] -= 2.0 * eps * a
    conv[dst:dst + n] += 2.0 * eps * a

    # quadratic part (Δ*Δ)
    idx2src = 2 * src
    idx2dst = 2 * dst
    idxsum = src + dst
    conv[idx2src] += eps * eps
    conv[idx2dst] += eps * eps
    conv[idxsum] -= 2.0 * eps * eps

    # update the vector
    a[src] -= eps
    a[dst] += eps


def _make_candidate_shapes(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse collection of analytic (unnormalised) shapes."""
    x = np.arange(n, dtype=float) / (n - 1)          # scaled to [0,1]

    candidates = []

    # 1) Power‑tents  (min(x,1‑x))**p
    for p in np.linspace(0.1, 2.0, 20):
        shape = np.minimum(x, 1.0 - x) ** p
        candidates.append(shape)

    # 2) Exponential decay from the centre (symmetric)
    for lam in np.logspace(-4, -0.5, 12):
        shape = np.exp(-lam * np.abs(x - 0.5))
        candidates.append(shape)

    # 3) Power‑law from the edges (symmetric)
    for p in np.linspace(0.2, 2.0, 15):
        ix = np.arange(n, dtype=float) + 1.0
        shape = (ix ** -p) + ((n + 1 - ix) ** -p)
        candidates.append(shape)

    # 4) Inverse‑sqrt (U‑shaped)
    shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shape = shape + shape[::-1]                     # symmetric
    candidates.append(shape)

    # 5) Half‑cosine
    candidates.append(np.sin(np.pi * x))

    # 6) Gaussian centred, different sigmas
    for sigma in (0.12, 0.20, 0.30):
        gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        candidates.append(gauss)

    # 7) A few random Dirichlet‑like draws
    for _ in range(8):
        rnd = rng.random(n)
        candidates.append(rnd)

    # normalise and keep only those with positive mass
    normalised = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0.0:
            normalised.append(c / s)

    return normalised
