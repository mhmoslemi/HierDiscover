    beta2 = 0.999

    lr = 0.3
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()
    alpha = 0.5
    alpha_inc = (20.0 - 0.5) / 20000 # target increase eventually

        # compute convolution
        conv = _conv_self(a, L_fft_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam-smoothed iter={it} score={cur_score:.6f}")

        # compute weights w
        # stable softmax: subtract max conv
        conv_shift = conv - conv.max()   # negative numbers

"""
Propose a non‑negative step function that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm improves on the previous state‑of‑the‑art by:

1️⃣  A quick parametric sweep over a geometric+uniform family (the same
    idea as before but with a denser log‑grid).

2️⃣  A *soft‑max* Adam optimiser.  Instead of using the sub‑gradient from a
    single maximising shift, we form a smooth approximation of the
    infinity‑norm by a log‑sum‑exp over all convolution values.  The
    resulting weight vector yields a gradient that is a convex combination
    of the gradients from many active shifts.  The smoothing factor α is
    increased gradually, allowing large steps early on and a precise
    descent later.

3️⃣  A final stochastic hill‑climb that transfers tiny (or up to 1 %)
    fractions of mass between two positions and keeps only improving
    moves.

All operations respect the 1000 s budget, use FFT‑based convolutions
(O(n log n)) and project onto the probability simplex after every update.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex { w ≥ 0, Σ w = 1 }.
    Returns w.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient from a weighted combination of convolution shifts
# ----------------------------------------------------------------------
def _gradient_from_weights(weights: np.ndarray, a: np.ndarray,
                           L_grad: int) -> np.ndarray:
    """
    Compute  ∇_a  Σ_k weights_k * (a ⋆ a)[k] .
    The result is 2 * cross‑correlation(weights, a) restricted to n entries.
    """
    n = a.shape[0]
    w_rev = weights[::-1]
    # FFT of w_rev and a on the same length
    Fw = np.fft.rfft(w_rev, n=L_grad)
    Fa = np.fft.rfft(a, n=L_grad)
    # cross‑correlation via multiplication in the frequency domain
    corr = np.fft.irfft(Fw * Fa, n=L_grad)[: 3 * n - 2]
    # pick the slice that corresponds to the gradient (indices 2n-2 … n-1)
    grad = 2.0 * corr[2 * n - 2 : n - 2 : -1]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser that uses a soft‑max over all convolution entries
# ----------------------------------------------------------------------
def _adam_softmax_optimize(a0: np.ndarray,
                           start_time: float,
                           deadline: float,
                           best_score_initial: float,
                           max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a *soft‑max* sub‑gradient.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)          # for self‑convolution
    L_grad = _next_pow2(3 * n)              # for gradient computation

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.25                               # a bit more aggressive than before

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    # smoothing parameter – start small, increase gradually
    alpha = 0.5
    alpha_max = 15.0
    alpha_mul = 1.0004                      # geometric increase per iteration

    while time.time() < deadline and it < max_iter:
        it += 1

        # current convolution and objective
        conv = _conv_self(a, L_conv)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(a)=1 after projection

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Adam‑softmax it={it:,} "
                  f"score={cur_score:.6f}")

        # --- soft‑max weights over all shifts ---------------------------------
        # stable implementation: subtract the max before exponentiation
        shifted = conv - conv.max()
        weights = np.exp(alpha * shifted)
        w_sum = weights.sum()
        if w_sum == 0.0:
            # fall back to a uniform weight (should hardly ever happen)
            weights = np.full_like(conv, 1.0 / conv.size)
        else:
            weights /= w_sum

        # --- gradient ---------------------------------------------------------
        grad = _gradient_from_weights(weights, a, L_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection back onto the simplex
        a = _project_simplex(a)

        # increase the smoothing parameter (but keep it bounded)
        if alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # plateau handling – inject a tiny random shake
        if it - last_improve > 8000:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (micro‑mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                start_time: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: move a tiny random fraction of mass from one
    index to another; keep the move only if it improves the true objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a, L_fft).max())
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # random fraction of source mass, up to 1 %
        frac = rng.random() * 0.01 + 1e-6
        eps = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-15:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative list that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The search consists of:
      * a dense geometric+uniform sweep,
      * an Adam optimiser with a soft‑max sub‑gradient,
      * a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave ~0.5 s for the very final sanity checks
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise with uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = float(2.0 * n * _conv_self(best_seq, L_fft).max())
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # try a global starter if one is present
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0:
                starter = starter / total
                cur = float(2.0 * n * _conv_self(starter, L_fft).max())
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Dense geometric+uniform sweep (log‑grid for ε, linear for λ)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    eps_grid = np.logspace(-4.5, -2.5, 400)   # ≈ 3·10⁻⁵ … 3·10⁻³
    lam_grid = np.arange(0.0, 1.001, 0.01)    # 0 % … 100 %

    for eps in eps_grid:
        if time.time() >= deadline - 30.0:   # leave ~30 s for later phases
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0.0:
            continue
        geo = geo / geo_sum                     # normalised geometric
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq = seq / seq.sum()
            cur = float(2.0 * n * _conv_self(seq, L_fft).max())
            if cur < best_score:
                best_score = cur
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Grid eps={eps:.1e} λ={lam:.2f} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimiser with soft‑max sub‑gradient
    # ------------------------------------------------------------------
    # reserve ~150 s for the final hill‑climb
    hill_time = 150.0
    adam_deadline = deadline - hill_time
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam‑softmax optimiser")
        a_opt, sc_opt = _adam_softmax_optimize(best_seq, start_time,
                                               adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑softmax improvement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a_hc, sc_hc = _hill_climb(best_seq, start_time, deadline)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
