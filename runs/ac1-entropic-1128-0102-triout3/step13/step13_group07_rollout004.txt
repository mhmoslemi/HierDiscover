The goal is to minimise  

\[
\text{score}= \frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a` is a non‑negative step‑function (list of floats) and `*` denotes the linear convolution.
Because the score is invariant under a global scaling of `a`, we may impose the normalisation  

\[
\sum_i a_i = 1 .
\]

Hence the problem is equivalent to **minimising**
\(2\,n\;\max_k (a*a)_k\) under the simplex constraints
\(a_i\ge 0,\;\sum a_i=1\).

The search strategy consists of three phases

1. **Warm‑starts** – a collection of analytic shapes (tent, cosine, Gaussian, …) and a set of random sparse candidates are generated and the best of them is kept.
2. **Multi‑restart Adam optimisation** – the best warm‑start is refined, but the algorithm is also repeatedly re‑started from perturbed copies of the current best solution.  
   A projected Adam optimiser works on the exact sub‑gradient of the `max‑convolution` term.
3. **Final polish** – after the restarts a longer Adam run is performed on the *largest* vector (8192 entries) obtained by linear interpolation of the currently best solution, followed by a cheap stochastic hill‑climb.

All steps respect the time budget (`budget_s` seconds).  The code is completely self‑contained, uses only NumPy, and prints progress information.

```python
"""Step‑function optimiser for minimise 2·n·max(a*a) (sum(a)=1).

The algorithm
--------------
1. Build a diverse set of analytic warm‑starts (tent, cosine, Gaussian…).
   Add a handful of random sparse vectors.
2. Keep the best warm‑start (lowest exact objective).

3. Multi‑restart optimisation:
   * repeatedly perturb the current best vector,
   * run a *short* projected Adam optimiser (exact sub‑gradient of the max‑convolution)
     with a fixed number of iterations,
   * keep the best result.
   This explores many basins of attraction.

4. Upsample the best vector to a larger size (8192) and run a *long* Adam
   optimisation with adaptive learning‑rate decay.
5. Finish with a cheap stochastic hill‑climb (greedy mass moves).

All vectors are projected onto the probability simplex after every update.
The routine returns a plain list of floats that can be passed directly to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1 (the valid part of the convolution).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerical fallback – uniform point
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic warm‑starts
# ----------------------------------------------------------------------
def _make_warm_starts(n: int) -> list[np.ndarray]:
    """Generate a diverse collection of candidate vectors (norm‑sum = 1)."""
    x = np.arange(n, dtype=float) / (n - 1)               # x∈[0,1]

    candidates = []

    # 1) linear tent
    tent = np.minimum(x, 1.0 - x)
    candidates.append(tent)

    # 2) half‑cosine (raised cosine)
    cosine = np.sin(np.pi * x)                # zero at the ends
    candidates.append(cosine)

    # 3) half‑circle (Archimedes)
    circ = np.sqrt(np.clip(1.0 - (2.0 * x - 1.0) ** 2, 0.0, 1.0))
    candidates.append(circ)

    # 4) Gaussian centred in the middle
    sigma = 0.2
    gauss = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    candidates.append(gauss)

    # 5) power‑tent with a range of exponents
    for p in np.linspace(0.2, 2.0, 9):
        candidates.append(np.minimum(x, 1.0 - x) ** p)

    # 6) exponential decay (and its reversal)
    for scale in (3.0, 5.0, 8.0):
        exp_fwd = np.exp(-scale * x)
        exp_rev = exp_fwd[::-1]
        candidates.append(exp_fwd)
        candidates.append(exp_rev)

    # 7) uniform
    candidates.append(np.ones_like(x))

    # Normalise each candidate to sum = 1 and keep only the non‑negative ones.
    normals = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 0:
            normals.append(c / s)
    return normals


# ----------------------------------------------------------------------
# Random / sparse candidates
# ----------------------------------------------------------------------
def _random_sparse(n: int, rng: np.random.Generator,
                   min_k: int = 8, max_k: int | None = None) -> np.ndarray:
    """Create a sparse random probability vector (k non‑zero entries)."""
    if max_k is None:
        max_k = max(min_k, n // 8)
    k = rng.integers(min_k, max_k + 1)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vec = np.zeros(n, dtype=float)
    vec[idx] = vals
    s = vec.sum()
    if s > 0.0:
        vec /= s
    else:
        vec = np.full(n, 1.0 / n)
    return vec


def _perturb(a: np.ndarray,
             rng: np.random.Generator,
             scale: float = 0.1) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    b = a + rng.normal(scale=scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 0.0:
        b /= s
    else:
        b = np.full_like(a, 1.0 / a.shape[0])
    return b


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient of max(conv) w.r.t. a.
    conv is the full linear self‑convolution (length 2·n‑1).
    """
    n = a.shape[0]
    k = int(conv.argmax())            # index where the max is attained
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]      # g_i = 2·a_{k‑i} on the overlap
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact sub‑gradient of the max‑convolution)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   lr: float = 0.02,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex using the exact sub‑gradient of the max‑convolution.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline and (max_iters is None or it < max_iters):
        it += 1

        # 1) self‑convolution (full length 2·n‑1)
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        # 2) sub‑gradient of the current max entry
        grad = _max_subgradient(a, conv)

        # Scale because the true objective is 2·n·max(conv)
        grad_obj = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection onto the probability simplex
        a = _project_simplex(a)

        # current objective (no extra convolution needed)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # learning‑rate decay (helps in the long run)
        if it % 5000 == 0:
            lr *= 0.9

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Greedy Metropolis walk that moves a random packet of mass between
    two indices, accepting only improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline:
        it += 1
        # source is chosen proportionally to its mass – encourages moves
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move at most 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        if new_score < cur_score:                     # greedy acceptance
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain list of floats (already normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin to avoid overrunning the budget
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Warm‑starts (analytic + random sparse)
    # ------------------------------------------------------------------
    n_small = 2048                     # cheap size for many restarts
    warm = _make_warm_starts(n_small)

    # add a few random sparse candidates
    for _ in range(12):
        warm.append(_random_sparse(n_small, rng, min_k=10, max_k=n_small // 8))

    # best of the warm‑starts
    best_a = None
    best_score = np.inf
    for ws in warm:
        sc = _objective(ws)
        if sc < best_score:
            best_score = sc
            best_a = ws.copy()
    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Multi‑restart short Adam runs
    # ------------------------------------------------------------------
    # each restart uses at most 5 s (or fewer if time is tight)
    short_budget = 5.0
    while time.time() + short_budget < deadline:
        # create a perturbed copy of the current best
        cand = _perturb(best_a, rng, scale=0.15)

        # run a brief Adam optimisation (max 2000 iterations)
        cur_deadline = time.time() + short_budget
        a_opt, sc_opt = _adam_optimize(cand,
                                        cur_deadline,
                                        start_time,
                                        lr=0.05,
                                        max_iters=2000)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to a larger size and run a long Adam optimisation
    # ------------------------------------------------------------------
    n_large = 8192
    xs_small = np.linspace(0.0, 1.0, n_small)
    xs_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(xs_large, xs_small, best_a)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()                      # re‑normalise

    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate 70 % of the remaining time to the final Adam run
        adam_deadline = time.time() + 0.70 * remaining
        a_opt, sc_opt = _adam_optimize(a_large,
                                        adam_deadline,
                                        start_time,
                                        lr=0.02)
        if sc_opt < best_score:
            best_score = sc_opt
            a_large = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final Adam best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light hill‑climb polish (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _hill_climb(a_large, deadline, start_time)
        if sc_opt < best_score:
            best_score = sc_opt
            a_large = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    a_large = np.maximum(a_large, 0.0)
    tot = a_large.sum()
    if tot > 0.0:
        a_large /= tot
    else:
        a_large = np.full(n_large, 1.0 / n_large)

    print(f"[{time.time() - start_time:.2f}s] Finished – objective={best_score:.6f}, length={n_large}")
    return [float(v) for v in a_large]
```
