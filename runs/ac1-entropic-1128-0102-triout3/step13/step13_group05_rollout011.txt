        lo = max(0, t_max - (n - 1))

        if new_score < cur_score:
            a = a_new
            conv = conv_new
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

                        print(f"[{time.time()-start:.2f}s] Subgrad iter={it} new best={s_tmp:.6f} factor={factor}")

"""
Propose a low‑score step function.

The search proceeds in three stages:

1. **Seed generation** – a small pool of analytic shapes (uniform,
   power‑ramps, Beta, Hann, half‑sine, random Dirichlet) is built and the
   best K candidates are kept.

2. **Targeted simulated‑annealing** – starting from the best seed we run
   an SA that always moves mass from indices that most contribute to the
   current maximal convolution value towards random locations.  The move
   size shrinks with a classic exponential cooling schedule.

3. **Projected sub‑gradient descent** – using the exact gradient of the
   ``max(conv)`` term (which is simply a reversed copy of the vector at
   the maximising shift) we perform a plain gradient step with a
   decreasing step‑size, followed by Euclidean projection onto the
   probability simplex.  Small back‑tracking is performed if the step
   does not improve the objective.  Random perturbations are injected
   when progress stalls.

All stages respect the overall time budget (default 1000 s).  The final
sequence is projected onto the simplex, clipped to be non‑negative and
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a size‑k sorted pool (ascending scores)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])

# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric power‑ramp with optional plateau (normalised to sum 1)."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        # pad / trim to exact length
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _make_hann_shape(n: int) -> np.ndarray:
    """Hann window, normalised to sum 1."""
    i = np.arange(n, dtype=float)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w

def _make_half_sine_shape(n: int) -> np.ndarray:
    """Half‑sine shape (sinusoid on (0,π))."""
    i = np.arange(n, dtype=float) + 0.5
    w = np.sin(np.pi * i / n)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Gradient of the max‑convolution term
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, conv: np.ndarray):
    """Return gradient of max(conv) w.r.t a and the index t where max occurs."""
    n = a.shape[0]
    t = int(conv.argmax())
    lo = max(0, t - (n - 1))
    hi = min(n - 1, t)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[t - idx]
    return grad, t

# ----------------------------------------------------------------------
# Targeted simulated‑annealing (mass moves that attack the current peak)
# ----------------------------------------------------------------------
def _targeted_sa(a0: np.ndarray, deadline: float, rng: np.random.Generator,
                 start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    it = 0
    max_move_factor = 0.25   # up to 25 % of source mass per move
    T_start = 0.5
    T_end = 1e-8

    while time.time() < deadline:
        it += 1
        # current convolution & max position
        conv = _conv_self(a)
        t = int(conv.argmax())
        # contributions of each index to the current max
        lo = max(0, t - (n - 1))
        hi = min(n - 1, t)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[t - idx]
        if contrib.sum() == 0:
            src = rng.choice(idx)
        else:
            src = rng.choice(idx, p=contrib / contrib.sum())
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_factor * rng.random()
        if delta <= 0.0:
            continue
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        # evaluate
        new_score = _objective(a_prop)
        # temperature schedule (exponential)
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * (T_end / T_start) ** frac
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")
        # occasional tiny Gaussian perturbation for escaping plateaus
        if it % 100000 == 0:
            sigma = 0.005
            noise = rng.normal(0.0, sigma, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_score = _objective(a)
    return best_a, best_score

# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         start_time: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        # gradient at current max
        grad, _ = _max_conv_gradient(a, conv)

        # base step decays like 1/sqrt(it)
        step = 1.0 / math.sqrt(it)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            a_new = a - step * factor * grad
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            new_score = _objective(a_new)

            if new_score < cur_score:
                a, conv, cur_score = a_new, conv_new, new_score
                improved = True
                no_improve = 0
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Subgrad iter={it:,} best={new_score:.6f}")
                break

        if not improved:
            no_improve += 1

        # if stuck for a while, inject a small random perturbation
        if no_improve >= 5000:
            sigma = 0.01
            noise = rng.normal(0.0, sigma, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            conv = _conv_self(a)
            cur_score = _objective(a)
            no_improve = 0

    return best_a, best_score

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3     # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – larger N gives a finer discretisation,
    # but we keep it modest to allow many optimisation steps.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        N = 65536
    elif remaining > 300:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # 1) Build an initial pool of analytic candidates
    # ------------------------------------------------------------------
    TOP_K = 5
    top = []                                     # list of (score, vector)

    # uniform (baseline)
    uniform = np.full(N, 1.0 / N)
    _push_top(top, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    p_grid = np.linspace(0.2, 2.0, 9)            # 0.2 … 2.0
    alpha_grid = np.linspace(0.0, 0.5, 6)        # 0.0 … 0.5
    for p in p_grid:
        for a in alpha_grid:
            vec = _make_shape(N, p, a)
            _push_top(top, _objective(vec), vec, TOP_K)

    # Beta‑shapes
    for a in np.linspace(0.1, 0.9, 9):
        vec = _make_beta_shape(N, a)
        _push_top(top, _objective(vec), vec, TOP_K)

    # Hann and half‑sine windows
    vec = _make_hann_shape(N)
    _push_top(top, _objective(vec), vec, TOP_K)
    vec = _make_half_sine_shape(N)
    _push_top(top, _objective(vec), vec, TOP_K)

    # random sparse Dirichlet samples
    for _ in range(40):
        if time.time() >= deadline:
            break
        a = rng.gamma(0.3, 1.0, size=N)          # gamma≈0.3 → sparsity
        if a.sum() == 0:
            continue
        a /= a.sum()
        _push_top(top, _objective(a), a, TOP_K)

    # incorporate the global starter if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                if N % starter.size == 0:
                    factor = N // starter.size
                    starter_up = np.repeat(starter, factor)
                else:
                    x_old = np.linspace(0, 1, starter.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            if starter_up.sum() > 0:
                starter_up /= starter_up.sum()
                _push_top(top, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # current best candidate
    best_score, best_seq = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Targeted simulated‑annealing (global exploration)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sa_deadline = time.time() + remaining * 0.45
        best_seq, best_score = _targeted_sa(best_seq, sa_deadline, rng, start_time)

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent (fine polishing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        sg_deadline = time.time() + remaining * 0.5
        best_seq, best_score = _subgradient_descent(best_seq, sg_deadline, start_time, rng)

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]
