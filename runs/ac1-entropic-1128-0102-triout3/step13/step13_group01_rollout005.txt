"""
Gradient‑descent with a smooth “log‑sum‑exp” approximation of the
max‑convolution objective.  Starting from a few analytic shapes (the
previous best sequence, some beta‑distribution shapes, and a power‑ramp
shape) we run a continuation method: the smoothness parameter λ is
increased slowly so that early iterations move quickly on a smooth
landscape while later iterations focus on the true max‑value.
Projected gradient steps keep the iterate on the probability simplex.
A light simulated‑annealing polish is applied at the end.  The routine
respects the 1000 s budget and returns a plain Python list of non‑negative
heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper for FFT‑based convolutions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑free) objective
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) with sum(seq) normalised to 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x≥0, Σx=1} via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numeric safeguard
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for seeding)
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric two‑sided power ramp with optional flat plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution on {0,…,n‑1}."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Smooth approximation of max(conv) and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, lam: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient
    of that approximation w.r.t. a.
    The smooth approximation is (log Σ exp(lam·conv))/lam.
    """
    n = a.shape[0]
    conv = _conv_self(a)          # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights
    shifted = conv - maxc                     # ≤ 0
    exp_vals = np.exp(np.clip(lam * shifted, -700, 0))
    Z = exp_vals.sum()
    w = exp_vals / Z                          # softmax over conv entries

    # gradient = Σ_k w_k ∂conv_k/∂a = 2 * (a * w_rev)[2n‑2‑i]
    w_rev = w[::-1]
    conv_full = _conv_fft(a, w_rev)          # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)           # positions of the needed slice
    grad = 2.0 * conv_full[idx]

    smooth_max = maxc + np.log(Z) / lam
    return smooth_max, grad


# ----------------------------------------------------------------------
# Gradient descent with continuation on λ
# ----------------------------------------------------------------------
def _gd_smooth(a0: np.ndarray,
               deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected (momentum) gradient descent on the smooth surrogate.
    λ starts at 5 and is multiplied by 1.5 every `lam_inc` iterations
    up to `lam_max`.  Returns the best iterate (according to the exact
    objective) seen before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.9
    it = 0

    lam = 5.0
    lam_max = 150.0
    lam_inc = 4000          # increase λ every this many iterations

    while time.time() < deadline:
        it += 1
        smooth_val, grad = _smooth_max_and_grad(a, lam)

        # occasional exact evaluation (cheap for n≈4k)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} λ={lam:.1f} score={cur_score:.6f}")

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project back onto the simplex
        a = _project_simplex(a)

        # update λ
        if it % lam_inc == 0 and lam < lam_max:
            lam = min(lam * 1.5, lam_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (identical to the reference version)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Random mass‑transfer Metropolis walk with exponential cooling."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 600_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        src = rng.choice(n, p=a)          # more likely to move from large entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function under the 1000 s budget.
    The algorithm:
      1. Build a small pool of promising seeds (previous best + a few beta shapes).
      2. For each seed run a continuation gradient‑descent on the smooth max.
      3. Keep the best iterate and finally polish it with a light annealer.
    Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Determine a working length (use the length of the supplied
    #    global best if available, otherwise fall back to 4096)
    # --------------------------------------------------------------
    global_seq = globals().get('height_sequence_1')
    if isinstance(global_seq, (list, np.ndarray)):
        base = np.asarray(global_seq, dtype=float)
        base = np.maximum(base, 0.0)
        if base.sum() > 0.0:
            base = base / base.sum()
            n = base.size
        else:
            base = None
            n = 4096
    else:
        base = None
        n = 4096

    # --------------------------------------------------------------
    # 2) Build a tiny pool of diverse starting points
    # --------------------------------------------------------------
    pool = []
    pool_size = 9

    if base is not None:
        pool.append(base.copy())

    rng = np.random.default_rng(seed)

    # a few beta‑distribution seeds
    for _ in range(4):
        a_par = rng.uniform(0.2, 4.0)
        b_par = rng.uniform(0.2, 4.0)
        shape = _make_beta_shape(n, a_par, b_par)
        pool.append(shape)

    # a power‑ramp seed (p≈1, α≈0.2)
    shape = _make_power_shape(n, p=1.0, alpha=0.2)
    pool.append(shape)

    # ensure pool not larger than pool_size
    if len(pool) > pool_size:
        pool = pool[:pool_size]

    best_seq = None
    best_score = float('inf')

    # --------------------------------------------------------------
    # 3) Allocate time for each starting point (≈ equal share)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        raise RuntimeError("No time left for optimisation.")

    per_start = remaining / len(pool)

    for idx, seed_vec in enumerate(pool):
        sub_deadline = time.time() + per_start * 0.85   # keep a margin for final polish
        seq, score = _gd_smooth(seed_vec, sub_deadline, start_time)
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best {score:.6f} from seed #{idx}")

        # optional quick anneal if time permits
        if time.time() < deadline:
            anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
            if anneal_score < best_score:
                best_score, best_seq = anneal_score, anneal_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal improved global best to {anneal_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation & conversion
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
