"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search combines several complementary strategies and runs entirely
within the allotted 1000 s budget:

1️⃣  **Parametric sweep** – explores families a_i ∝ r**i·(i+1)**p and mixtures
    of two exponentials.  This quickly finds a good baseline.

2️⃣  **Adam descent on a smooth‑max surrogate** – repeatedly starts from
    perturbed versions of the current best vector and performs projected Adam
    updates on the log‑sum‑exp approximation of the max‑convolution.

3️⃣  **Projected sub‑gradient descent** – works on the exact max‑convolution
    (gradient derived from the maximal shift) with a diminishing step size.

4️⃣  **Simulated‑annealing hill‑climb** – transfers a tiny amount of mass
    between two random coordinates, accepting worsening moves with a slowly
    decreasing temperature.

All phases keep the vector on the probability simplex (non‑negative,
sum = 1) and keep the best sequence found so far.  The routine returns the
sequence as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using the FFT."""
    n = a.shape[0]
    if L is None:
        L = next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0, sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def eval_fast(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact objective for a vector whose sum is (approximately) 1.
    Returns float('inf') if the sum is too small.
    """
    conv = conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

def sweep_candidates(
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore simple analytic families (geometric, power, mixed exponentials).
    """
    L_fft = next_pow2(2 * n - 1)

    idx = np.arange(1, n + 1, dtype=np.float64)  # i+1

    # ---- single‑parameter families --------------------------------------
    r_grid = np.logspace(np.log10(0.92), np.log10(0.9995), num=35)
    p_grid = np.linspace(-5.0, 3.0, 41)

    for r in r_grid:
        if time.time() >= deadline - 180:
            break
        geo = r ** np.arange(n, dtype=np.float64)
        for p in p_grid:
            if time.time() >= deadline - 180:
                break
            shape = geo * (idx ** p)
            s = shape.sum()
            if s == 0.0:
                continue
            shape /= s
            # mix with uniform to smooth the extremes
            for lam in (0.0, 0.1, 0.3, 0.5, 0.7, 0.9):
                cand = lam * np.full(n, 1.0 / n) + (1.0 - lam) * shape
                cand /= cand.sum()
                sc = eval_fast(cand, n, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} "
                        f"p={p:.2f} λ={lam:.2f} → {sc:.6f}"
                    )

    # ---- two‑exponential mixtures ---------------------------------------
    r1_grid = np.logspace(np.log10(0.90), np.log10(0.999), num=12)
    r2_grid = np.logspace(np.log10(0.90), np.log10(0.999), num=12)
    w_grid = np.linspace(0.0, 1.0, 9)

    for r1 in r1_grid:
        for r2 in r2_grid:
            if time.time() >= deadline - 180:
                break
            base1 = r1 ** np.arange(n, dtype=np.float64)
            base2 = r2 ** np.arange(n, dtype=np.float64)
            for w in w_grid:
                shape = w * base1 + (1.0 - w) * base2
                s = shape.sum()
                if s == 0.0:
                    continue
                shape /= s
                sc = eval_fast(shape, n, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_a = shape.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] mix r1={r1:.5f} "
                        f"r2={r2:.5f} w={w:.2f} → {sc:.6f}"
                    )
    return best_a, best_score

# 2️⃣  Adam descent on a smooth‑max surrogate
# ----------------------------------------------------------------------


def smooth_max_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max approximation of max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score

    rng = np.random.default_rng()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    L_conv = next_pow2(2 * n - 1)
    L_corr = next_pow2(3 * n - 2)

    beta = 6.0              # initial soft‑max temperature
    beta_mul = 1.025
    beta_max = 600.0

    while time.time() < deadline:
        t += 1

        # --- soft‑max gradient -----------------------------------------
        conv = conv_self_fft(a, L_conv)          # length 2n‑1
        max_c = conv.max()
        shifted = conv - max_c                  # stabilise exponentials
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()           # soft‑max weights

        # correlation (w * reversed a) gives gradient of smooth‑max
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A, n=L_corr)
        start = n - 1
        grad = 2.0 * corr[start : start + n]

        # exact objective (for monitoring)
        cur_score = eval_fast(a, n, L_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={t:,} β={beta:.1f} "
                f"score={cur_score:.6f}"
            )

        # --- Adam update (projected) ------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.2 / (np.sqrt(v_hat) + eps)     # slightly larger step
        a = a - step * m_hat
        a = project_simplex(a)

        # temperature schedule (make the surrogate sharper)
        beta = min(beta * beta_mul, beta_max)

        # occasional random jitter to escape plateaus
        if t % 3500 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.maximum(a + noise, 0.0)
            a = project_simplex(a)

    return best_a, best_score

def subgradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score
    rng = np.random.default_rng()
    L_fft = next_pow2(2 * n - 1)

    step0 = 0.14            # initial step magnitude (tuned)
    it = 0

    while time.time() < deadline:
        it += 1

        conv = conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))        # shift achieving the max

        # sub‑gradient for shift k (O(n))
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if np.any(mask):
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = project_simplex(a_candidate)

        cur_score = eval_fast(a_candidate, n, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                f"step={step:.5e} score={cur_score:.6f}"
            )
        else:
            # simulated‑annealing style acceptance
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter now and then
        if it % 1800 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = project_simplex(a)

    return best_a, best_score

def hill_climb_anneal(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    init_score: float,
) -> tuple[np.ndarray, float]:
    """
    Tiny mass transfers accepted with a temperature that slowly cools.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score
    L_fft = next_pow2(2 * n - 1)
    rng = np.random.default_rng()

    T0 = 5e-4          # initial temperature
    T = T0
    decay = 0.99995    # geometric cooling

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer up to 0.3 % of the source mass (larger steps early on)
        eps = a[src] * rng.random() * 3e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = eval_fast(a_prop, n, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb improved → {new_score:.6f}"
            )
        else:
            # probabilistic acceptance
            if rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_prop

        # cool the temperature
        T = max(T * decay, 1e-8)

        # occasional full‑restart jitter if we seem stuck
        if rng.random() < 0.0005:
            jitter = 1e-4 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = project_simplex(a)

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step function within the given time
    budget.  Returns the sequence as a plain Python list of floats.
    The implementation follows the outline described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # problem size (moderate – works well with FFT)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise with the uniform distribution
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)
    best_score = eval_fast(best_seq, n, L_fft)
    print(
        f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # optionally start from the globally supplied construction
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the surrounding environment
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # interpolate / truncate to the required length
                x_src = np.linspace(0.0, 1.0, starter.size)
                x_dst = np.linspace(0.0, 1.0, n)
                starter = np.interp(x_dst, x_src, starter, left=0.0, right=0.0)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter /= starter.sum()
                sc = eval_fast(starter, n, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = starter.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}"
                )
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  fast parametric sweep
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        best_seq, best_score = sweep_candidates(
            n, deadline - 150, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 2️⃣  random restarts with smooth‑max Adam descent
    # ------------------------------------------------------------------
    time_for_adam = min(deadline - time.time() - 120, 400)
    if time_for_adam > 10:
        n_restarts = 5
        per_restart = time_for_adam / n_restarts
        for i in range(n_restarts):
            if time.time() >= deadline - 120:
                break
            rng = np.random.default_rng(seed + i + 12345)
            # start from a perturbed version of the current best
            a0 = best_seq.copy()
            a0 += 0.025 * rng.standard_normal(n) * a0
            a0 = np.maximum(a0, 0.0)
            a0 = project_simplex(a0)

            sub_deadline = time.time() + per_restart
            cand_a, cand_sc = smooth_max_adam(
                a0, n, sub_deadline, start_time, best_score
            )
            if cand_sc < best_score:
                best_score = cand_sc
                best_seq = cand_a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Adam restart {i+1}/{n_restarts} "
                    f"improved to {cand_sc:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  sub‑gradient refinement on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        sub_deadline = deadline - 80
        cand_a, cand_sc = subgradient_descent(
            best_seq, n, sub_deadline, start_time, best_score
        )
        if cand_sc < best_score:
            best_score = cand_sc
            best_seq = cand_a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sub‑gradient final score={cand_sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  final annealed hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cand_a, cand_sc = hill_climb_anneal(
            best_seq, n, deadline, start_time, best_score
        )
        if cand_sc < best_score:
            best_score = cand_sc
            best_seq = cand_a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={cand_sc:.6f}"
            )

    # ------------------------------------------------------------------
    # final normalisation and output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(
        f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}"
    )
    return [float(v) for v in best_seq]
