"""
Improved search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three complementary phases:

1️⃣  A dense parametric sweep over several families (geometric,
    mixed geometric‑uniform, power‑law, and mixtures of two
    exponentials).  The sweep is followed by a tiny local refinement
    on the best parameters.

2️⃣  A long projected Adam optimiser that works directly on the
    simplex.  The optimizer uses the exact sub‑gradient of the
    currently active convolution shift and a linearly decaying learning
    rate, which gives a much sharper descent than the reference code.

3️⃣  A simulated‑annealing hill‑climb that moves a *large* random
    fraction of mass between two positions.  Worse moves are occasionally
    accepted with a temperature that cools down, allowing the search to
    escape shallow local minima.

All phases respect the 1000 s time limit, use FFT‑based convolutions,
and finally return a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real‑valued)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score(a: np.ndarray, L_fft: int) -> float:
    """
    Compute the objective 2·n·max(conv(a,a)) / (sum(a)**2).
    a is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = _conv_self(a, L_fft)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _subgrad_one_active(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Gradient of the active (maximal) convolution shift."""
    n = a.shape[0]
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_score: float,
                   max_iter: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Long projected Adam optimiser using the sub‑gradient of the
    active convolution shift.  Learning rate decays linearly.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    a = a0.copy()
    best_a = a.copy()
    cur_best = best_score

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    init_lr = 0.5

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_iter:
        it += 1
        # 1) evaluate current objective
        conv = _conv_self(a, L_fft)
        cur_score = _score(a, L_fft)

        if cur_score < cur_best:
            cur_best = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # 2) sub‑gradient from the active shift
        grad = _subgrad_one_active(a, conv)

        # 3) Adam update with decaying lr
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = init_lr * max(0.01, 1.0 - it / max_iter)  # linear decay, never below 0.01*init_lr
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # occasional jitter if stuck
        if it % 5000 == 0:
            noise = 1e-5 * rng.standard_normal(n) * a
            a = np.clip(a + noise, 0, None)
            a = _project_simplex(a)

    return best_a, cur_best


def _simulated_annealing(start_a: np.ndarray,
                         start_time: float,
                         deadline: float,
                         init_temp: float = 0.02,
                         cooling: float = 0.99995) -> tuple[np.ndarray, float]:
    """
    Hill‑climb with simulated‑annealing acceptance.
    Moves a random (up to 20 %) fraction of mass from one index to another.
    """
    a = start_a.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    cur_score = _score(a, L_fft)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    temp = init_temp

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # pick a fraction up to 0.2 (20 %) of the source mass
        frac = rng.random() * 0.2 + 1e-6
        delta = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _score(a_prop, L_fft)

        # accept if better, otherwise with probability exp(-(new-old)/temp)
        improve = new_score < cur_score
        if improve or rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")

        # temperature cooling
        temp *= cooling
        if temp < 1e-6:
            temp = 1e-6

    return best_a, best_score

    # --------------------------------------------------------------
    # Problem size – we stay with n = 4096 (the reference size) which
    # gives a good trade‑off between expressive power and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optionally use a global starter (height_sequence_1) if it exists
    # --------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                sc = _score(starter, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # ---- exponential family (geometric) + optional uniform mixing ----
    eps_grid = np.logspace(-5, -2, 300)          # ε = 1‑r
    lam_grid = np.arange(0.0, 1.001, 0.05)      # mixing weight λ

    best_eps, best_lam = None, None
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo = geo / geo_sum                     # normalise to sum=1

        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            # tiny safety normalisation
            seq = seq / seq.sum()
            sc = _score(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                best_eps, best_lam = eps, lam
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} λ={lam:.2f} score={sc:.6f}")

    # ---- power‑law family a_i ∝ (i+β)^(-p) (β≥0) ----
    p_grid = np.linspace(0.2, 4.0, 120)          # exponent
    beta_grid = np.array([0.0, 0.5, 1.0, 2.0])   # shift
    lam_grid2 = np.arange(0.0, 1.001, 0.10)

    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        base = (idx + 1) ** (-p)                # β = 0 version
        for beta in beta_grid:
            if beta > 0:
                base = (idx + beta) ** (-p)
            base_sum = base.sum()
            if base_sum == 0:
                continue
            base = base / base_sum
            for lam in lam_grid2:
                seq = lam * uniform + (1.0 - lam) * base
                seq = seq / seq.sum()
                sc = _score(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} β={beta} λ={lam:.2f} score={sc:.6f}")

    # ---- mixture of two exponentials (more flexibility) ----
    eps1_grid = np.logspace(-5, -2, 30)
    eps2_grid = np.logspace(-5, -2, 30)
    w_grid = np.array([0.3, 0.5, 0.7])
    for eps1 in eps1_grid:
        if time.time() >= deadline - 30:
            break
        r1 = 1.0 - eps1
        geo1 = np.power(r1, idx)
        for eps2 in eps2_grid:
            r2 = 1.0 - eps2
            geo2 = np.power(r2, idx)
            for w in w_grid:
                mixed = w * geo1 + (1.0 - w) * geo2
                mixed_sum = mixed.sum()
                if mixed_sum == 0:
                    continue
                mixed = mixed / mixed_sum
                seq = mixed                           # pure mixture (no uniform)
                sc = _score(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] 2‑exp eps1={eps1:.1e} eps2={eps2:.1e} w={w:.2f} score={sc:.6f}")

    # ---- local refinement around best geometric parameters (if any) ----
    if best_eps is not None:
        # explore a factor‑3 neighbourhood in log‑space and λ±0.1
        eps_fine = np.logspace(np.log10(best_eps / 3), np.log10(best_eps * 3), 200)
        lam_fine = np.linspace(max(0.0, best_lam - 0.1), min(1.0, best_lam + 0.1), 200)
        for eps in eps_fine:
            if time.time() >= deadline - 30:
                break
            r = 1.0 - eps
            geo = np.power(r, idx)
            geo = geo / geo.sum()
            for lam in lam_fine:
                seq = lam * uniform + (1.0 - lam) * geo
                seq = seq / seq.sum()
                sc = _score(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] Refine eps={eps:.1e} λ={lam:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam optimiser – use remaining time (leave ~150 s for SA)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 150.0:
        adam_deadline = time.time() + remaining - 150.0
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        a_opt, sc_opt = _adam_optimize(best_seq, start_time, adam_deadline, best_score)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improvement score={sc_opt:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing hill‑climb (large moves)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated‑annealing hill‑climb")
        a_sa, sc_sa = _simulated_annealing(best_seq, start_time, deadline)
        if sc_sa < best_score:
            best_score = sc_sa
            best_seq = a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] SA final score={sc_sa:.6f}")

    # --------------------------------------------------------------
    # Normalise finally and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
