"""
Improved optimizer for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine builds a diverse pool of analytic shapes (power ramps,
beta‑distributions, sine, Gaussian and raised‑cosine windows), picks the
best few, refines the winner with a **projected Adam** optimiser
(momentum + per‑coordinate step‑size adaptation on the probability
simplex) and finally runs a lightweight simulated‑annealing polish.
The whole search respects the 1000 s budget and returns a plain Python
list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, objective, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective (no safety checks).  Assumes seq.sum() > 0."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # should not happen in practice
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all return a probability vector of length n)
# ----------------------------------------------------------------------
def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric shape: power ramps with an optional flat plateau."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution."""
    xs = (np.arange(n) + 0.5) / n
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sine_shape(n: int, p: float) -> np.ndarray:
    """sin(π·x) raised to a power p (symmetric, zero at the ends)."""
    xs = (np.arange(n) + 0.5) / n  # (0,1)
    shape = np.sin(np.pi * xs) ** p
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma_factor: float) -> np.ndarray:
    """Gaussian window centred in the middle; sigma = sigma_factor·n."""
    sigma = max(1e-6, sigma_factor * n)
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    shape = np.exp(-((idx - centre) / sigma) ** 2)
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_cosine_shape(n: int, width_factor: float) -> np.ndarray:
    """
    Raised‑cosine window.
    Non‑zero for |i−centre| ≤ w where w = width_factor·n/2.
    """
    w = max(1e-6, width_factor * n / 2.0)
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    dist = np.abs(idx - centre)
    mask = dist <= w
    shape = np.zeros(n, dtype=float)
    shape[mask] = 0.5 * (1.0 + np.cos(np.pi * (dist[mask] / w)))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _gradient_from_indices(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (averaged) max‑convolution w.r.t. a.
    For each index k in ks we add 2·a_{k−i} to coordinate i.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    grad /= float(len(ks))
    return grad


# ----------------------------------------------------------------------
# Optimiser: projected Adam (per‑coordinate adaptive steps)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   top_k_start: int = 200,
                   top_k_end: int = 30,
                   lr0: float = 0.6,
                   beta1: float = 0.9,
                   beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the probability simplex.
    The active set size (top_k) is gradually reduced.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # progress‑dependent top_k
        progress = (time.time() - start_time) / (deadline - start_time)
        top_k = int(top_k_start * (1 - progress) + top_k_end * progress)
        top_k = max(1, min(top_k, 2 * n - 1))

        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,}  score={cur_score:.6f}")

        # active indices (largest convolution entries)
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        grad = _gradient_from_indices(a, ks)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # time‑dependent learning rate (linear decay)
        lr = lr0 * (1.0 - progress)

        step = lr / (np.sqrt(v_hat) + 1e-8)
        a = a - step * m_hat

        # re‑project onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (pairwise & split moves)
# ----------------------------------------------------------------------
def _anneal_improved(seq: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly transfers tiny packets of probability mass.
    Two move types are used:
        1) source → single destination (up to 80 % of source)
        2) source → two destinations (splitting the mass)
    The Metropolis rule with an exponential cooling schedule is applied.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # cooling schedule
    T0 = 0.05
    T_end = 1e-7
    max_steps = 400_000  # enough for the remaining budget

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        T = T0 * (T_end / T0) ** (it / max_steps)

        # decide move type
        if rng.random() < 0.7:       # mostly single‑destination moves
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-15:
                continue
            eps = a[src] * rng.random() * 0.8   # up to 80 % of src mass
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
        else:                         # split‑move
            src = rng.choice(n, p=a)
            dst1, dst2 = rng.integers(n, size=2)
            if src == dst1 or src == dst2 or a[src] < 1e-15:
                continue
            eps1 = a[src] * rng.random() * 0.4
            eps2 = a[src] * rng.random() * 0.4
            total = eps1 + eps2
            if total > a[src]:
                scale = a[src] / total
                eps1 *= scale
                eps2 *= scale
                total = a[src]
            a_prop = a.copy()
            a_prop[src] -= total
            a_prop[dst1] += eps1
            a_prop[dst2] += eps2

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,}  best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step function.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1) Problem size – a power of two for fast FFTs
    # ------------------------------------------------------------------
    n = 32768   # 2¹⁵ – half the size of the original script, gives more iterations

    # ------------------------------------------------------------------
    # 2) Build a diverse pool of analytic candidates
    # ------------------------------------------------------------------
    pool = []            # list of (score, vector)
    pool_sz = 10

    def _maybe_add(vec):
        sc = _objective(vec)
        if len(pool) < pool_sz:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # ---- power‑ramp family ------------------------------------------------
    for p in np.linspace(0.2, 2.2, 21):
        for alpha in np.linspace(0.0, 0.5, 13):
            a = _make_power_shape(n, p, alpha)
            _maybe_add(a)

    # ---- beta‑distribution family ----------------------------------------
    for a_par in np.linspace(0.2, 4.0, 13):
        for b_par in np.linspace(0.2, 4.0, 13):
            a = _make_beta_shape(n, a_par, b_par)
            _maybe_add(a)

    # ---- sine family ------------------------------------------------------
    for p in np.linspace(0.2, 4.0, 15):
        a = _make_sine_shape(n, p)
        _maybe_add(a)

    # ---- Gaussian family ---------------------------------------------------
    for sigma_factor in np.linspace(0.05, 0.40, 8):
        a = _make_gaussian_shape(n, sigma_factor)
        _maybe_add(a)

    # ---- Raised‑cosine family ---------------------------------------------
    for width_factor in np.linspace(0.1, 0.5, 9):
        a = _make_cosine_shape(n, width_factor)
        _maybe_add(a)

    # ---- random convex combinations of two families -----------------------
    rng = np.random.default_rng()
    for _ in range(200):
        if time.time() >= deadline:
            break
        # pick two random families and blend them
        shape_fns = [_make_power_shape, _make_beta_shape,
                     _make_sine_shape, _make_gaussian_shape,
                     _make_cosine_shape]
        f1, f2 = rng.choice(shape_fns, size=2, replace=False)
        # random parameters for each
        if f1 is _make_power_shape:
            p1 = rng.uniform(0.2, 2.2)
            al1 = rng.uniform(0.0, 0.5)
            s1 = f1(n, p1, al1)
        elif f1 is _make_beta_shape:
            a1 = rng.uniform(0.2, 4.0)
            b1 = rng.uniform(0.2, 4.0)
            s1 = f1(n, a1, b1)
        elif f1 is _make_sine_shape:
            p1 = rng.uniform(0.2, 4.0)
            s1 = f1(n, p1)
        elif f1 is _make_gaussian_shape:
            sf1 = rng.uniform(0.05, 0.40)
            s1 = f1(n, sf1)
        else:  # cosine
            wf1 = rng.uniform(0.1, 0.5)
            s1 = f1(n, wf1)

        if f2 is _make_power_shape:
            p2 = rng.uniform(0.2, 2.2)
            al2 = rng.uniform(0.0, 0.5)
            s2 = f2(n, p2, al2)
        elif f2 is _make_beta_shape:
            a2 = rng.uniform(0.2, 4.0)
            b2 = rng.uniform(0.2, 4.0)
            s2 = f2(n, a2, b2)
        elif f2 is _make_sine_shape:
            p2 = rng.uniform(0.2, 4.0)
            s2 = f2(n, p2)
        elif f2 is _make_gaussian_shape:
            sf2 = rng.uniform(0.05, 0.40)
            s2 = f2(n, sf2)
        else:
            wf2 = rng.uniform(0.1, 0.5)
            s2 = f2(n, wf2)

        w = rng.random()
        a = w * s1 + (1.0 - w) * s2
        a /= a.sum()
        _maybe_add(a)

    # ------------------------------------------------------------------
    # 3) Incorporate a globally supplied starter if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                # resample / truncate / pad to length n
                if starter.size != n:
                    # simple linear interpolation to required length
                    x_old = np.linspace(0, 1, starter.size)
                    x_new = np.linspace(0, 1, n)
                    starter = np.interp(x_new, x_old, starter)
                starter /= starter.sum()
                _maybe_add(starter)
    except Exception:
        pass

    # best candidate so far
    best_score, best_seq = pool[0]
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Local refinement – projected Adam (majority of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + 0.68 * remaining
        refined_seq, refined_score = _adam_optimize(best_seq,
                                                    adam_deadline,
                                                    start_time,
                                                    top_k_start=200,
                                                    top_k_end=30,
                                                    lr0=0.6)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam refined score = {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Light annealing polish (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal_improved(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improved score = {anneal_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final normalisation & conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
