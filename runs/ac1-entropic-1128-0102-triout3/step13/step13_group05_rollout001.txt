"""
Optimise a non‑negative step function (vector) that minimises  

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three phases under the 1000 s budget:

1. **Fast analytic / random pool** – a few hundred candidate shapes of
   length N₀ (≈ 8192) are generated (uniform, power‑ramps, Beta‑densities,
   edge‑power, random Dirichlet mixes).  The best K are kept.

2. **Projected soft‑max gradient descent** – starting from the current
   best candidate we run a long Adam‑style descent on a smooth
   approximation of the max‑convolution (soft‑max with a temperature
   schedule).  After each step the vector is projected onto the
   probability simplex (non‑negative, sum = 1).  The best ever found is
   retained.

3. **Upsampling & final polishing** – the best length‑N₀ vector is
   interpolated to the final length (65536) and, if a few seconds remain,
   a brief simulated‑annealing sweep fine‑tunes the result.

All heavy computation uses real FFTs for O(N log N) convolution.
The routine returns a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: smallest power of two ≥ x
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via real FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Scale‑invariant objective (sum(a)=1 is assumed)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    # sum(seq) is 1 by construction – avoid division by tiny numbers
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families (all return vectors normalised to sum 1)
# ----------------------------------------------------------------------
def _make_shape(N: int, p: float, alpha: float) -> np.ndarray:
    """Symmetric “ramp‑with‑plateau” shape."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * N))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (N - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(N)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < N:
            shape = np.concatenate([shape, np.zeros(N - shape.size)])
        else:  # shape.size > N (should not happen)
            shape = shape[:N]
    shape /= shape.sum()
    return shape


def _make_beta_shape(N: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,N‑1}."""
    x = (np.arange(N, dtype=float) + 0.5) / N
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_edge_power_shape(N: int, p: float) -> np.ndarray:
    """High weight on the two ends, low in the centre."""
    idx = np.arange(N, dtype=float)
    shape = ((idx + 1) ** p + (N - idx) ** p)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Keep the K best candidates (sorted by score, low is good)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Soft‑max (smooth‑max) gradient descent with Adam
# ----------------------------------------------------------------------
def _softmax_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """Adam‑style descent on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    N = a.shape[0]

    # pre‑allocate FFT length for the correlation step
    L_corr = _next_pow2(3 * N - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.20

    # temperature schedule parameters
    T_start = 2.0
    T_end = 1e-8

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # current convolution and (soft‑)objective
        # --------------------------------------------------------------
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * N * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Descent it={it:,}  score={cur_score:.6f}")

        # --------------------------------------------------------------
        # temperature (linear cooling)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = max(T_end, T_start * (1.0 - frac))

        # --------------------------------------------------------------
        # soft‑max weights over the convolution entries
        # --------------------------------------------------------------
        w_raw = np.exp((conv - max_conv) / T)          # max‑conv is 0 in exponent
        w = w_raw / w_raw.sum()

        # --------------------------------------------------------------
        # gradient of the soft‑max surrogate: 2 * (w ⋆ rev(a))
        # --------------------------------------------------------------
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L_corr)
        A_rev = np.fft.rfft(a_rev, n=L_corr)
        corr = np.fft.irfft(W * A_rev, n=L_corr)[: 3 * N - 2]
        grad = 2.0 * corr[N - 1 : N - 1 + N]   # length N

        # --------------------------------------------------------------
        # Adam update
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / math.sqrt(it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsample a short vector to a larger length (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(vec: np.ndarray, final_len: int) -> np.ndarray:
    n = vec.shape[0]
    if final_len == n:
        return vec.copy()
    # use linear interpolation on the unit interval
    x_old = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / n
    x_new = np.linspace(0.0, 1.0, final_len, endpoint=False) + 0.5 / final_len
    up = np.interp(x_new, x_old, vec)
    # re‑normalise to keep the sum exactly 1
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full(final_len, 1.0 / final_len)
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (sum > 0) that (hopefully)
    minimises 2·n·max(conv(a,a))/(sum(a)**2) within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # a tiny safety margin

    # ------------------------------------------------------------------
    # Choose a modest optimisation size (fast FFT) – 8192 works well.
    # ------------------------------------------------------------------
    N_opt = 8192
    final_N = 65536          # length demanded by the scoring script

    # ------------------------------------------------------------------
    # 1) Build a pool of analytic / random candidates (fast).
    # ------------------------------------------------------------------
    TOP_K = 5
    top = []   # list of (score, vector)

    # uniform baseline
    uniform = np.full(N_opt, 1.0 / N_opt)
    _push_top(top, _objective(uniform), uniform, TOP_K)

    # power‑ramps
    for p in np.linspace(0.2, 5.0, 13):          # 0.2,0.6,…,5.0
        for alp in np.linspace(0.0, 0.5, 6):
            a = _make_shape(N_opt, p, alp)
            _push_top(top, _objective(a), a, TOP_K)

    # beta shapes (U‑shaped)
    for alp in np.linspace(0.05, 0.9, 18):
        a = _make_beta_shape(N_opt, alp)
        _push_top(top, _objective(a), a, TOP_K)

    # edge‑power shapes
    for p in np.linspace(0.5, 5.0, 10):
        a = _make_edge_power_shape(N_opt, p)
        _push_top(top, _objective(a), a, TOP_K)

    # random Dirichlet like vectors (sparse)
    for _ in range(80):
        a = rng.gamma(0.3, 1.0, size=N_opt)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        _push_top(top, _objective(a), a, TOP_K)

    # mixed random combinations of the above families
    for _ in range(30):
        comps = rng.integers(2, 5)        # 2–4 components
        weights = rng.random(comps)
        weights /= weights.sum()
        mix = np.zeros(N_opt)
        for w in weights:
            kind = rng.integers(4)
            if kind == 0:
                comp = uniform
            elif kind == 1:
                comp = _make_shape(N_opt,
                                   rng.uniform(0.2, 5.0),
                                   rng.uniform(0.0, 0.5))
            elif kind == 2:
                comp = _make_beta_shape(N_opt,
                                        rng.uniform(0.05, 0.9))
            else:
                comp = _make_edge_power_shape(N_opt,
                                              rng.uniform(0.5, 5.0))
            mix += w * comp
        mix /= mix.sum()
        _push_top(top, _objective(mix), mix, TOP_K)

    # if a previously‑found global starter exists, incorporate it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N_opt:
                # simple linear up‑sampling / down‑sampling
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N_opt, endpoint=False)
                starter = np.interp(x_new, x_old,
                                    np.maximum(starter, 0.0))
            starter = _project_simplex(starter)
            _push_top(top, _objective(starter), starter, TOP_K)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the current best
    # ------------------------------------------------------------------
    best_score, best_vec = top[0][0], top[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Prolonged soft‑max Adam descent (the heavy lifting)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:               # ensure we have a few seconds left for polishing
        descent_deadline = time.time() + remaining - 5.0
        best_vec, best_score = _softmax_descent(best_vec,
                                                descent_deadline,
                                                start)

        print(f"[{time.time() - start:.2f}s] Post‑descent best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional tiny simulated‑annealing polish on the full size vector
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # up‑sample to the final length
        cand = _upsample(best_vec, final_N)

        # a very brief SA that moves tiny fractions of mass
        cur = cand.copy()
        cur_score = _objective(cur)
        best_final = cur.copy()
        best_final_score = cur_score

        max_move_frac = 0.05
        T_start = 0.2
        T_end = 1e-8

        it = 0
        while time.time() < deadline:
            it += 1
            elapsed = time.time() - start
            total = deadline - start
            frac = min(1.0, elapsed / total)
            T = T_start * (1.0 - frac) + T_end * frac

            src = rng.choice(final_N, p=cur)
            if cur[src] < 1e-16:
                continue
            dst = rng.integers(final_N)
            if src == dst:
                continue

            delta = cur[src] * max_move_frac * rng.random()
            a_prop = cur.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta

            new_score = _objective(a_prop)
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
                cur = a_prop
                cur_score = new_score
                if new_score < best_final_score:
                    best_final = cur.copy()
                    best_final_score = new_score
                    print(f"[{time.time() - start:.2f}s] SA polish it={it:,} "
                          f"score={new_score:.6f}")

        # adopt the polished version if it improved
        if best_final_score < best_score:
            best_vec = best_final
            best_score = best_final_score

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.shape[0])

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
