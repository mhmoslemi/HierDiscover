```
def _smooth_gradient(a, beta, L_conv, L_corr):
    conv = _conv_self(a, L=L_conv)
    max_c = conv.max()
    shifted = conv - max_c
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W*A, n=L_corr)
    start = a.shape[0] - 1
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad, conv
```

"""
Propose a non‑negative step function that minimizes the evaluation metric.

The search mixes several complementary strategies inside the 1000 s budget:

1️⃣  **Parametric sweep** – explores two‑parameter families
    (geometric · polynomial and Beta‑shaped densities) and mixes each
    shape with the uniform distribution.  The best candidate is retained.

2️⃣  **Smooth‑max Adam descent** – optimises a soft‑max approximation of the
    max‑convolution term with a temperature that grows up to 2000.
    Multiple random restarts increase diversity.

3️⃣  **Exact max‑subgradient descent** – works directly on the true max
    entry.  At each iteration the gradient of the dominant convolution lag
    is computed analytically and a projected step on the simplex is taken.
    A simple step‑size schedule drives convergence.

4️⃣  **Targeted hill‑climb** – repeatedly moves a small fraction of mass
    from the index that contributes most to the current peak towards a
    location that reduces it.  Only improving moves are kept; the step
    size adapts automatically.

All intermediate scores and timestamps are printed.  The final sequence is
normalised to sum 1 and returned as a plain list of floats.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    If L is given, zero‑pad to length L before the transform.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where
        conv = a*a (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) )
    The gradient is computed via FFT‑based correlation.
    """
    conv = _conv_self(a, L_conv)          # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (stable)
    shifted = conv - max_c                 # max becomes 0
    exp_vals = np.exp(beta * shifted)      # under‑flow for large negative values is harmless
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # correlation w * a_rev → gradient
    a_rev = a[::-1]                        # length n
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)   # linear convolution w * a_rev

    start = a.shape[0] - 1                 # n‑1
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type descent on the smooth‑max surrogate.
    β starts small (smooth landscape) and is slowly increased up to beta_max.
    The iterate is always projected onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 1.0               # start fairly smooth
    beta_mul = 1.015          # geometric increase per iteration

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.5 / (1.0 + 5e-5 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        beta = min(beta * beta_mul, beta_max)

        # tiny random jitter occasionally
        if t % 5000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑subgradient descent (directly on the true max term)
# ----------------------------------------------------------------------
def _max_subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    init_step: float = 0.5,
    decay: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent where the gradient is taken of the
    currently dominant convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    n = a.shape[0]
    step = init_step

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)                     # exact convolution
        max_val = conv.max()
        k_max = int(np.argmax(conv))
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={t:,} step={step:.6f} score={cur_score:.6f}"
            )

        # gradient of the active entry
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k_max - idx_i
        grad[idx_i] = 2.0 * a[idx_j]

        # trial step
        a_trial = a - step * grad
        a_trial = _project_simplex(a_trial)

        # evaluate trial
        conv_trial = _conv_self(a_trial)
        new_score = 2.0 * n * conv_trial.max() / (a_trial.sum() ** 2)

        if new_score < cur_score:
            a = a_trial
            step = min(step * 1.02, init_step)   # slightly enlarge after success
        else:
            step = max(step * decay, 1e-5)        # shrink otherwise

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (random + max‑entry attacks)
# ----------------------------------------------------------------------
def _targeted_hill_climb(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves and deterministic moves that pull weight
    away from the index pair that currently contributes most to the peak
    of the convolution.  Only improving moves are retained.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    step_factor = 0.12
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        # recompute convolution to locate the dominant lag
        conv = _conv_self(a, L_fft)
        k_max = int(np.argmax(conv))

        # indices that actually appear in this lag
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k_max - idx_i
        # contribution of each i to the peak
        contrib = a[idx_i] * a[idx_j]
        src_local = int(np.argmax(contrib))
        src = idx_i[src_local]

        # try a few candidate destinations and keep the best improving one
        best_candidate = None
        best_candidate_score = best_score
        for _ in range(3):
            dst = rng.integers(n)
            if dst == src:
                continue
            delta = a[src] * rng.random() * step_factor
            a_prop = a.copy()
            a_prop[src] -= delta
            a_prop[dst] += delta
            if a_prop[src] < 0.0:
                continue
            new_score = _eval_fast(a_prop, L_fft)
            if new_score < best_candidate_score:
                best_candidate = a_prop
                best_candidate_score = new_score

        if best_candidate is not None and best_candidate_score < best_score:
            a = best_candidate
            best_a = a.copy()
            best_score = best_candidate_score
            step_factor = max(step_factor, 0.12)   # reset step size
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Coarse‑to‑fine parametric sweep
# ----------------------------------------------------------------------
def _parametric_sweep(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore two families:
        (i) a_i ∝ r^i·(i+1)^p  (geometric × polynomial)
        (ii) a_i ∝ x^{α}·(1‑x)^{β}   (Beta‑type)
    each mixed with the uniform distribution.
    The sweep is stopped early if the remaining time becomes scarce.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=35)
    p_grid = np.linspace(-4.0, 3.0, 71)
    lam_grid = np.linspace(0.0, 1.0, 11)

    L_fft = _next_pow2(2 * n - 1)

    # ---- geometric × polynomial family ----
    for r in r_grid:
        if time.time() >= deadline - 60:
            break
        geo = np.power(r, np.arange(n, dtype=np.float64))
        for p in p_grid:
            if time.time() >= deadline - 60:
                break
            shape = geo * np.power(idx, p)
            s = shape.sum()
            if s <= 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep geo r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                    )

    # ---- Beta‑type family ----
    x = (np.arange(n, dtype=np.float64) + 0.5) / n        # centre of each bin
    eps = 1e-12
    alpha_grid = np.linspace(-0.9, 5.0, 30)
    beta_grid = np.linspace(-0.9, 5.0, 30)

    for alpha in alpha_grid:
        if time.time() >= deadline - 60:
            break
        for beta in beta_grid:
            if time.time() >= deadline - 60:
                break
            shape = (
                np.power(np.maximum(x, eps), alpha)
                * np.power(np.maximum(1.0 - x, eps), beta)
            )
            s = shape.sum()
            if s <= 0.0:
                continue
            shape /= s
            for lam in lam_grid:
                seq = lam * uniform + (1.0 - lam) * shape
                seq /= seq.sum()
                sc = _eval_fast(seq, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep beta α={alpha:.2f} β={beta:.2f} λ={lam:.2f} score={sc:.6f}"
                    )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects the 1000 s budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # problem size – 4096 offers a good balance
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise: uniform baseline (sum ≈ 1)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(
        f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}"
    )

    # optional global starter (if provided by the environment)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (coarse → refined)
    # ------------------------------------------------------------------
    if time.time() < deadline - 250:   # reserve ~4 min for later phases
        best_seq, best_score = _parametric_sweep(
            n, uniform, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # 2️⃣  Multiple smooth‑max Adam descent runs (diverse seeds)
    # ------------------------------------------------------------------
    descent_seeds = [seed, seed + 1, seed + 2, seed + 3]
    for ds in descent_seeds:
        if time.time() >= deadline - 120:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        a0 += 5e-4 * rng.standard_normal(n) * a0   # tiny perturbation
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 25000
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,          # leave a couple of seconds for polishing
            start_time,
            best_score,
            beta_max=2000.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth descent improvement score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # 3️⃣  Exact max‑subgradient descent (focus on the true max term)
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        rng = np.random.default_rng(seed + 10)
        a0 = best_seq.copy()
        a0 += 1e-3 * rng.standard_normal(n) * a0
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 15000
        a_opt, sc_opt = _max_subgradient_descent(
            a0,
            max_iters,
            deadline - 30,
            start_time,
            best_score,
            init_step=0.3,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgradient improvement score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb polish (random + targeted moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = uniform.copy()

    # sanity check with the official evaluator (optional)
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
