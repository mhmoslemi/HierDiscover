def _shape_vector(n, r, p, q, lam):
    idx = np.arange(n, dtype=np.float64) + 1
    rev = n - np.arange(n, dtype=np.float64)  # n - i
    shape = np.power(r, np.arange(n, dtype=np.float64))
    if p != 0:
        shape *= np.power(idx, p)
    if q != 0:
        shape *= np.power(rev, q)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0/n)
    else:
        shape /= shape_sum
    if lam > 0:
        shape = lam * np.full(n, 1.0/n) + (1 - lam) * shape
        shape /= shape.sum()
    return shape

"""
Propose a non‑negative step function (a list of floats) that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  A fast random‑parameter sweep over a two‑sided power‑exponential family
    a_i ∝ r**i * (i+1)**p * (n‑i)**q (mixed with the uniform vector).  
    This family is very expressive: p<0 concentrates mass near the start,
    q>0 concentrates mass near the end, and r close to 1 gives a slowly
    decaying envelope.  Many random draws are evaluated until a time budget
    slice expires; the best vector found so far is kept.

2️⃣  A refined sub‑gradient descent on the *exact* objective.
    For the current maximal convolution entry (or a small set of entries
    tied for the maximum) the exact sub‑gradient of
          max_k (a*a)[k]
    is computed (the factor 2 that was missing in the reference code is
    restored).  A diminishing step size together with a simple simulated‑annealing
    acceptance rule and Euclidean projection onto the probability simplex
    drives the vector downhill.

3️⃣  A microscopic hill‑climb that moves a tiny amount of mass between two
    random positions and accepts only improving moves.  This polishes the
    final candidate.

All phases respect the overall 1000 s budget, and the routine finally
returns a plain Python list of floats (sum ≈ 1).  The code is self‑contained
and uses only NumPy and the provided ``evaluate_sequence`` for final scoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum ≈ 1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Random parametric sweep (two‑sided power‑exponential family)
# ----------------------------------------------------------------------


def _random_param_sweep(n: int, L_fft: int, best_seq: np.ndarray,
                        best_score: float, deadline: float, start_time: float):
    """
    Explore the family
        a_i ∝ r**i * (i+1)**p * (n-i)**q
    mixed with uniform, using random draws.
    """
    rng = np.random.default_rng()
    uniform = np.full(n, 1.0 / n)

    # parameter ranges
    r_min, r_max = 0.85, 0.99999          # geometric decay
    p_min, p_max = -5.0, 5.0             # left‑hand power
    q_min, q_max = -5.0, 5.0             # right‑hand power
    lam_min, lam_max = 0.0, 0.5          # mixing weight with uniform

    idx = np.arange(n, dtype=np.float64) + 1.0          # i+1
    rev = np.arange(n, dtype=np.float64)[::-1] + 1.0   # n‑i

    while time.time() < deadline:
        # draw parameters
        r = np.exp(np.log(r_min) + rng.random() *
                   (np.log(r_max) - np.log(r_min)))
        p = rng.uniform(p_min, p_max)
        q = rng.uniform(q_min, q_max)
        lam = rng.uniform(lam_min, lam_max)

        # log‑space construction (stable for extreme exponents)
        log_shape = (np.log(r) * np.arange(n) +
                     p * np.log(idx) +
                     q * np.log(rev))
        # avoid overflow by shifting
        log_shape -= np.max(log_shape)
        shape = np.exp(log_shape)

        s = shape.sum()
        if s == 0.0:
            continue
        shape /= s

        if lam > 0.0:
            shape = lam * uniform + (1.0 - lam) * shape
            shape = shape / shape.sum()

        cur_score = _eval_fast(shape, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = shape.copy()
            print(f"[{time.time() - start_time:.2f}s] param sweep "
                  f"r={r:.5f} p={p:.2f} q={q:.2f} λ={lam:.2f} "
                  f"score={cur_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray, n: int, deadline: float,
                         start_time: float, best_score_initial: float):
    """
    Projected sub‑gradient descent on the true objective.
    The sub‑gradient is obtained from *all* maximal convolution entries.
    """
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.18   # initial step magnitude (tuned empirically)
    it = 0
    tol = 1e-12   # tolerance for selecting co‑maximal shifts

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        # indices whose value is within tolerance of the maximum
        max_idxs = np.where(conv >= max_val - tol)[0]

        # --- exact sub‑gradient for the set of maximal shifts ---------------
        g = np.zeros_like(a)
        for k in max_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_vals = k - i_vals
            # contribution 2 * a_j for each i (covers both i=j and i≠j cases)
            g[i_vals] += 2.0 * a[j_vals]

        if max_idxs.size > 1:
            g /= max_idxs.size  # average over the active set

        # --- step with diminishing magnitude ---------------------------------
        step = step0 / np.sqrt(it)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        # accept if improvement, otherwise occasional probabilistic acceptance
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.3e} score={cur_score:.6f}")
        else:
            # simulated‑annealing flavour
            cur_score_cur = _eval_fast(a, L_fft)
            T = max(1e-6, 0.5 / np.sqrt(it))
            prob = np.exp(-(cur_score - cur_score_cur) / T)
            if rng.random() < prob:
                a = a_candidate

        # occasional jitter to escape plateaus
        if it % 2500 == 0:
            jitter = 1e-4 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final microscopic hill‑climb
# ----------------------------------------------------------------------


def _final_hill_climb(a0: np.ndarray, deadline: float, start_time: float):
    """Tiny mass transfers accepted only if they improve the objective."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 5e-4   # ≤0.05 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine mixes a random‑parameter sweep, a precise
    sub‑gradient refinement and a final microscopic hill‑climb, all under the
    1000 s budget.
    Returns the best sequence as a list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15      # safety margin
    n = 4096                                     # moderate problem size
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter is present, try it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (two‑sided power‑exponential family)
    # ------------------------------------------------------------------
    time_for_params = min(deadline - time.time() - 300, 200)  # leave time for later phases
    if time_for_params > 20:
        sweep_deadline = time.time() + time_for_params
        best_seq, best_score = _random_param_sweep(
            n, L_fft, best_seq, best_score, sweep_deadline, start_time)

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent from several perturbed restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.time() - 120   # keep 2 min for final polishing
    if remaining > 30:
        n_restarts = 3
        per_restart = remaining / n_restarts
        for i in range(n_restarts):
            if time.time() >= deadline - 120:
                break
            rng = np.random.default_rng(seed + i + 12345)
            a0 = best_seq.copy()
            # small random perturbation to escape possible stagnation
            a0 += 0.01 * rng.standard_normal(n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            sub_deadline = time.time() + per_restart
            a_sub, sc_sub = _subgradient_descent(
                a0, n, sub_deadline, start_time, best_score)

            if sc_sub < best_score:
                best_score = sc_sub
                best_seq = a_sub.copy()
                print(f"[{time.time() - start_time:.2f}s] restart {i+1}/{n_restarts} "
                      f"improvement score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final microscopic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _final_hill_climb(best_seq, deadline, start_time)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb "
                  f"score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
