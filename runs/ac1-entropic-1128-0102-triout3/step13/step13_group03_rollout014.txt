    except Exception:
        pass

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a multi‑stage search:

1.  A cheap parametric sweep (geometric × power‑law shapes, optional
    left‑hand shift, mixed with the uniform distribution) on the full
    problem size to obtain a decent seed.
2.  Several runs of Adam‑type projected descent on a smooth‑max
    surrogate of the max‑convolution term (β is increased up to 2000).
3.  Adam on the *exact* sub‑gradient of the (unsmoothed) max‑convolution.
4.  A directed stochastic hill‑climb that repeatedly removes mass from
    the pair of indices that contribute most to the current maximal
    convolution entry and deposits it at a location that least aggravates
    that entry.
5.  Return the best vector (renormalised to sum 1) as a list of floats.

All steps respect a 1000 s budget; intermediate scores are printed.
"""

import time
import math
import numpy as np
from typing import List

# --------------------------------------------------------------
# Basic linear‑algebra utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear (full) self‑convolution a*a using FFT.
    Length of the result is 2*len(a)-1; if L is given the FFT is
    performed on a zero‑padded array of length L (must be ≥ 2*len(a)-1).
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (w≥0, sum w=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# --------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# --------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective for a vector whose sum is (close to) 1."""
    conv = _conv_self(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# --------------------------------------------------------------
# Smooth‑max gradient (soft‑max approximation of the max)
# --------------------------------------------------------------
def _smooth_gradient(
    a: np.ndarray, beta: float, L_conv: int, L_corr: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where

        conv = a * a            (full linear convolution)
        grad = ∂/∂a  ( (1/β)·log Σ_k exp(β·conv_k) ).

    The gradient is obtained via a correlation computed with FFTs.
    """
    conv = _conv_self(a, L=L_conv)               # length 2n‑1
    max_c = conv.max()

    # stable soft‑max weights
    shifted = conv - max_c               # max becomes 0
    exp_vals = np.exp(beta * shifted)   # ≤1
    w = exp_vals / exp_vals.sum()       # shape (2n‑1,)

    # correlation w * a_rev  → gradient
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_corr)
    A = np.fft.rfft(a_rev, n=L_corr)
    corr = np.fft.irfft(W * A, n=L_corr)   # linear convolution w * a_rev

    start = a.shape[0] - 1                 # index of a_rev[0] in corr
    grad = 2.0 * corr[start : start + a.shape[0]]
    return grad, conv


# --------------------------------------------------------------
# Adam‑style projected descent on the smooth‑max surrogate
# --------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    beta_max: float = 2000.0,
    symmetrize: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Adam descent on the smooth‑max surrogate.
    β grows geometrically up to `beta_max`.  After each projection the
    vector can optionally be symmetrised (a ← (a + a[::-1]) / 2) to
    encourage symmetric solutions.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    beta = 5.0               # start with a smooth landscape
    beta_mul = 1.02          # increase per iteration

    while time.time() < deadline and t < max_iters:
        t += 1

        # gradient of the smooth surrogate
        grad, conv = _smooth_gradient(a, beta, L_conv, L_corr)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth it={t:,} β={beta:.1f} score={cur_score:.6f}"
            )

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.5 / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        if symmetrize and (t % 500 == 0):
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)

        # temperature schedule
        beta = min(beta * beta_mul, beta_max)

        # tiny jitter every few thousand steps to escape plateaus
        if t % 3000 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# --------------------------------------------------------------
# Sub‑gradient (Adam) descent on the *exact* max‑convolution term
# --------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    start_time: float,
    best_score_initial: float,
    symmetrize: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Adam descent on the true max‑convolution (no smoothing).  The
    gradient for the active entry k_max is  g_i = 2·a_{k_max‑i}  (if
    indices are in range).  The method uses the same projection onto
    the probability simplex as the smooth version.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    base_lr = 0.5

    while time.time() < deadline and t < max_iters:
        t += 1

        conv = _conv_self(a, L_fft)
        k_max = int(np.argmax(conv))

        # build sub‑gradient of the max entry
        grad = np.zeros_like(a)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        j_range = k_max - i_range
        grad[i_range] = 2.0 * a[j_range]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = base_lr / (1.0 + 1e-4 * t)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        if symmetrize and (t % 5000 == 0):
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgrad it={t:,} score={cur_score:.6f}"
            )

    return best_a, best_score


# --------------------------------------------------------------
# Targeted hill‑climb (random + directed moves)
# --------------------------------------------------------------
def _targeted_hill_climb_enhanced(
    start_a: np.ndarray,
    deadline: float,
    start_time: float,
    L_fft: int,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers (≤15 % of source mass) mixed with *directed*
    moves that explicitly attack the currently dominant convolution entry.
    The directed move:
        – selects the source index that contributes most to the maximal entry,
        – selects a destination that would increase that entry the least
          (i.e. where a_{k_max‑dst} is minimal),
        – transfers a small fraction of the source mass.
    Only strictly improving moves are accepted.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    # step‑size control
    step_factor = 0.15
    min_factor = 1e-5
    decay = 0.995

    while time.time() < deadline:
        # compute current convolution once per iteration
        conv = _conv_self(a, L_fft)
        k_max = int(np.argmax(conv))

        # ----- source: index with largest contribution to the max entry -----
        idx = np.arange(n)
        valid_src = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros_like(a)
        contrib[valid_src] = a[idx][valid_src] * a[k_max - idx][valid_src]
        # pick the heaviest contributor that actually has mass
        src_candidates = np.where((a > 1e-15) & (contrib > 0))[0]
        if src_candidates.size == 0:
            # fall back to a pure random transfer
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.random() * step_factor
        else:
            src = src_candidates[np.argmax(contrib[src_candidates])]

            # ----- destination: index that would add the *least* to this entry -----
            # the increase would be a[src] * a[k_max - dst]; we minimise a[k_max‑dst]
            inc_weights = np.zeros_like(a)
            valid_dst = (k_max - idx >= 0) & (k_max - idx < n)
            inc_weights[valid_dst] = a[k_max - idx][valid_dst]
            # avoid picking src (or its pair) as destination
            inc_weights[src] = np.inf
            pair = k_max - src
            if 0 <= pair < n:
                inc_weights[pair] = np.inf
            dst = int(np.argmin(inc_weights))
            if dst == src:
                dst = rng.integers(n)
                if dst == src:
                    continue
            delta = a[src] * rng.random() * step_factor

        # propose the move
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < 0.0:
            continue

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = new_score
            step_factor = max(step_factor, 0.15)   # reset after success
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb score={best_score:.6f}"
            )
        else:
            step_factor = max(step_factor * decay, min_factor)

    return best_a, best_score


# --------------------------------------------------------------
# Coarse‑to‑fine parametric sweep (extended search space)
# --------------------------------------------------------------
def _parametric_sweep_extended(
    n: int,
    uniform: np.ndarray,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Explore a richer family

        a_i ∝ r^{i−shift}·(i−shift+1)^p   (for i≥shift)

    mixed with the uniform distribution.  The sweep is stopped early
    if the remaining time becomes small.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)           # i+1
    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), num=25)
    p_grid = np.linspace(-3.0, 2.0, 31)
    shift_grid = [0, n // 16, n // 8, n // 4]           # left‑hand truncation
    L_fft = _next_pow2(2 * n - 1)

    for shift in shift_grid:
        if time.time() >= deadline - 120:
            break
        length = n - shift
        if length <= 0:
            continue
        rel_idx = np.arange(length, dtype=np.float64)     # i‑shift
        for r in r_grid:
            if time.time() >= deadline - 120:
                break
            geo = np.power(r, rel_idx)                    # geometric factor
            for p in p_grid:
                if time.time() >= deadline - 120:
                    break
                shape = np.zeros(n, dtype=np.float64)
                # power‑law factor (i‑shift+1)^p
                shape[shift:] = geo * np.power(rel_idx + 1.0, p)
                s = shape.sum()
                if s == 0.0:
                    continue
                shape /= s
                for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
                    seq = lam * uniform + (1.0 - lam) * shape
                    seq /= seq.sum()
                    sc = _eval_fast(seq, L_fft)
                    if sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print(
                            f"[{time.time() - start_time:.2f}s] sweep shift={shift} r={r:.4f} p={p:.2f} λ={lam:.2f} score={sc:.6f}"
                        )
    return best_seq, best_score

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects the 1000 s budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # problem size – 8192 gives a finer search space than the previous 4096
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ---- baseline (uniform) -------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] baseline uniform score={best_score:.6f}")

    # ---- try a global starter (if provided) ---------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # interpolate / truncate to required length
            if starter.size != n:
                x_old = np.linspace(0, 1, starter.size)
                x_new = np.linspace(0, 1, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0.0:
                starter /= total
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ---- 1️⃣  Parametric sweep (coarse → refined) ---------------------------
    if time.time() < deadline - 300:
        best_seq, best_score = _parametric_sweep_extended(
            n, uniform, deadline, start_time, best_seq, best_score
        )

    # ---- 2️⃣  Multiple smooth‑max Adam descent runs -------------------------
    descent_seeds = [seed, seed + 1, seed + 2, seed + 3]
    for ds in descent_seeds:
        if time.time() >= deadline - 120:
            break
        rng = np.random.default_rng(ds)
        a0 = best_seq.copy()
        a0 += 1e-5 * rng.standard_normal(n) * a0   # tiny perturbation
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)

        max_iters = 80000
        a_opt, sc_opt = _smooth_descent(
            a0,
            max_iters,
            deadline - 30,          # leave a few seconds for the next phases
            start_time,
            best_score,
            beta_max=2000.0,
            symmetrize=(ds % 2 == 0),   # try symmetry on half of the runs
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth descent improvement score={sc_opt:.6f}"
            )

    # ---- 3️⃣  Adam on the exact max‑convolution (sub‑gradient) --------------
    if time.time() < deadline - 60:
        a_opt, sc_opt = _subgradient_descent(
            best_seq,
            max_iters=200000,
            deadline=deadline - 30,
            start_time=start_time,
            best_score=best_score,
            symmetrize=False,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] subgradient descent improvement score={sc_opt:.6f}"
            )

    # ---- 4️⃣  Final hill‑climb polish ---------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = _targeted_hill_climb_enhanced(
            best_seq, deadline, start_time, L_fft
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final hill‑climb score={sc_opt:.6f}"
            )

    # ---- Normalise finally (sum → 1) ----------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # ---- sanity check with the official evaluator (optional) -----------------
    try:
        true_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] true score via evaluate_sequence = {true_score:.6f}"
        )
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
