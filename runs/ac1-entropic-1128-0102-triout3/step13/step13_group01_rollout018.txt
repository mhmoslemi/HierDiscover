"""
A faster multi‑start optimiser for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The search builds a diverse initial pool (uniform, classic windows,
beta / power ramps, cosine and Kaiser windows, DPSS eigenvectors and
random Dirichlet draws).  The best pool members are locally refined by
projected sub‑gradient descent with a larger active set (all entries
within 2 % of the current maximum) and Adam‑style momentum.  Finally a
light simulated‑annealing phase moves tiny packets of mass between
indices to polish the solution.

All operations respect the 1000 s time budget; the routine returns a
plain Python list of non‑negative floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑free)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) (no safety checks)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex via the algorithm of Wang & Carreira‑Perpiñán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families of shapes
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _make_power_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """Two power ramps with a flat plateau, normalised to sum = 1."""
    alpha = max(0.0, min(0.5, alpha))
    plateau_len = int(round(alpha * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discretised beta distribution, normalised to sum = 1."""
    xs = (np.arange(n) + 0.5) / n   # midpoint rule on [0,1]
    shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
    total = shape.sum()
    if total > 0.0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_hann(n: int) -> np.ndarray:
    """Classic Hann window, shifted to be non‑negative and normalised."""
    i = np.arange(n)
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _make_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    shape /= shape.sum()
    return shape


def _make_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    shape = (0.42
             - 0.5 * np.cos(2 * np.pi * i / (n - 1))
             + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window via scipy (fallback to analytical if unavailable)."""
    try:
        from scipy.signal import windows
        shape = windows.kaiser(n, beta, sym=False)
    except Exception:
        # simple approximation using Bessel function (slow but safe)
        from scipy.special import i0
        i = np.arange(n) - (n - 1) / 2.0
        alpha = (n - 1) / 2.0
        shape = i0(beta * np.sqrt(1 - (i / alpha) ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_cosine_tip(n: int) -> np.ndarray:
    """Cosine tip: a_i ∝ 1+cos(π i/(n‑1)), vanishes at the ends."""
    i = np.arange(n)
    shape = 1.0 + np.cos(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_dpss(n: int, nw: float) -> np.ndarray:
    """First DPSS (Slepian) sequence, normalised to sum = 1."""
    try:
        from scipy.signal import windows
        vec = windows.dpss(n, nw, Kmax=1, return_ratios=False)[0]
        vec = np.maximum(vec, 0.0)
        vec /= vec.sum()
        return vec
    except Exception:
        # fallback – just return uniform
        return _make_uniform(n)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with a modest Adam‑style momentum
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         tol: float = 0.02,
                         top_k: int = None) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector by projected sub‑gradient descent.
    All conv entries within `tol`·max are taken as active (or the
    `top_k` largest if given).  Uses Adam‑style momentum.
    """
    a = a0.copy()
    n = a.shape[0]

    # sanity – keep sum == 1 after the first projection
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _objective(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # denominator =1 because sum(a)=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,}  score={cur_score:.6f}")

        # choose active indices
        if top_k is not None and top_k < conv.size:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            thresh = cur_max * (1.0 - tol)
            ks = np.where(conv >= thresh)[0]

        # gradient of the (smoothed) max‑conv
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(ks))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(it) + 1.0)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (mass‑swap moves)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Randomly move small packets of mass between positions,
    accepting according to a Metropolis rule with exponential cooling.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_steps = 500_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = max(1e-4, 0.1 * (0.05 / 0.1) ** (it / max_steps))

        src = rng.choice(n, p=a)          # source prefers large entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.4
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,}  best={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns a plain Python list of floats (sum normalised to 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Choose a length – a power of two makes FFT cheap
    # --------------------------------------------------------------
    n = 16384                     # 2¹⁴ – enough resolution, cheap enough

    # --------------------------------------------------------------
    # 2) Build a diverse pool of promising candidates
    # --------------------------------------------------------------
    pool = []                     # (score, vector)
    pool_cap = 12

    def _maybe_add(vec):
        sc = _objective(vec)
        if len(pool) < pool_cap:
            pool.append((sc, vec.copy()))
            pool.sort(key=lambda x: x[0])
        elif sc < pool[-1][0]:
            pool[-1] = (sc, vec.copy())
            pool.sort(key=lambda x: x[0])

    # uniform baseline
    _maybe_add(_make_uniform(n))

    # classic windows
    _maybe_add(_make_hann(n))
    _maybe_add(_make_hamming(n))
    _maybe_add(_make_blackman(n))
    for beta in np.linspace(0.0, 10.0, 7):
        _maybe_add(_make_kaiser(n, beta))

    # cosine tip
    _maybe_add(_make_cosine_tip(n))

    # power‑ramp family
    for p in np.linspace(0.2, 2.0, 9):
        for alpha in np.linspace(0.0, 0.5, 6):
            _maybe_add(_make_power_shape(n, p, alpha))

    # beta‑distribution family (expanded grid)
    a_grid = np.linspace(0.2, 8.0, 9)
    b_grid = np.linspace(0.2, 8.0, 9)
    for a_par in a_grid:
        for b_par in b_grid:
            _maybe_add(_make_beta_shape(n, a_par, b_par))

    # DPSS (Slepian) sequences for a few time‑bandwidth products
    for nw in [2.0, 3.0, 4.0, 5.0]:
        _maybe_add(_make_dpss(n, nw))

    # Random Dirichlet draws (moderate concentration)
    rng = np.random.default_rng()
    for _ in range(200):
        if time.time() >= deadline:
            break
        vec = rng.dirichlet(alpha=np.ones(n) * 0.3)
        _maybe_add(vec)

    # Report best of the initial pool
    best_score, best_seq = min(pool, key=lambda x: x[0])
    print(f"[{time.time() - start_time:.2f}s] Initial pool best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) If a globally supplied starter exists, try it
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    _maybe_add(starter)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4) Local refinement – sub‑gradient descent (full‑active set)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 80.0:
        sub_deadline = time.time() + 0.85 * remaining
        refined_seq, refined_score = _subgradient_descent(
            best_seq, sub_deadline, start_time, tol=0.02, top_k=None)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal improve score = {anneal_score:.6f}")

    # --------------------------------------------------------------
    # 6) Normalisation & conversion to a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
