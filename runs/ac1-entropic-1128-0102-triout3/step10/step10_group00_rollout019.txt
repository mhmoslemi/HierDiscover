"""
Search for a step‑function that minimizes the evaluation objective.
The algorithm:
1. Choose a target length (default 16384) to get a finer discretisation.
2. Warm‑start from the previous solution (if present) by interpolating it
   to the new length; otherwise use a simple power‑law “tent”.
3. Perform a smoothed gradient descent where the non‑smooth max is
   approximated by a soft‑max (log‑sum‑exp).  The gradient of the
   soft‑max is a correlation that can be computed with one FFT pair,
   giving O(n log n) per iteration.
4. After each step project onto the probability simplex and enforce
   symmetry (the optimum is known to be symmetric).
5. Keep the best solution seen; finish with a lightweight simulated‑annealing
   polish.
All helper functions are top‑level so the harness can pickle them.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(sum(seq)^2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
def _correlate_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the correlation (w ⊛ a)[i] = Σ_j w[j] * a[i+j] for i=0…n‑1.
    Both arguments are real, w length = 2·n‑1, a length = n.
    Result length = n.
    """
    n = a.shape[0]
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(np.conj(W) * A, n=L)
    # take the central n entries: indices n-1 .. 2n-2 (inclusive)
    start = n - 1
    end = start + n
    return corr_full[start:end]

# ----------------------------------------------------------------------
def _softmax(x: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max of alpha·x."""
    y = alpha * x
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    return exp_y / exp_y.sum()

# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent using a soft‑max approximation of the max.
    Returns the best iterate (lowest objective) seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    step0 = 0.4
    beta = 0.9
    momentum = np.zeros_like(a)

    it = 0
    alpha0 = 5.0          # starting temperature of the soft‑max
    alpha_max = 200.0

    while time.time() < deadline:
        it += 1
        # compute convolution and soft‑max weights
        conv = _conv_self(a)
        alpha = min(alpha_max, alpha0 * math.sqrt(it))
        w = _softmax(conv, alpha)                 # length 2n‑1, sums to 1

        # gradient = 2·correlation(w, a)
        grad = 2.0 * _correlate_w_a(w, a)

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project onto simplex and enforce symmetry
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5          # keep the sequence symmetric

        # evaluate
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} "
                  f"α={alpha:.1f} score={cur_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Baseline sub‑gradient descent (same as the previous solution)."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} "
                  f"score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5       # maintain symmetry

    return best_a, best_score

# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing polish."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        # source chosen proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:

"""
Search for a step‑function that minimizes the evaluation objective.

The algorithm proceeds as follows:

1. Choose a fairly large canvas (default 16384).  A larger n gives a finer
   discretisation of the continuous problem whose optimum is ≈ 1.5.
2. Warm‑start from the historic solution (if the global variable
   ``height_sequence_1`` is present) by linear interpolation, otherwise use a
   symmetric power‑law “tent’’ shape.
3. Phase 1 – a quick sub‑gradient descent (the same update rule that was used
   previously) to obtain a decent region of the search space.
4. Phase 2 – smooth gradient descent where the nonsmooth max is replaced by a
   soft‑max (log‑sum‑exp).  The gradient of the soft‑max is a correlation that
   can be evaluated with a single FFT pair, giving an O(n log n) step.
5. After each update the vector is projected onto the probability simplex and
   forced to be symmetric (the optimum is known to be symmetric).
6. Phase 3 – a very light simulated‑annealing polish.
7. The final vector is normalised to sum 1 and returned as a plain list.

All helper functions are top‑level so the harness can pickle them.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(sum(seq)^2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
def _correlate_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation (w ⊛ a)[i] = Σ_j w[j]·a[i+j] for i = 0…n‑1.
    w length = 2·n‑1, a length = n → result length = n.
    """
    n = a.shape[0]
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(np.conj(W) * A, n=L)
    start = n - 1
    end = start + n
    return corr_full[start:end]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max of α·x."""
    y = alpha * x
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    return exp_y / exp_y.sum()


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent using a soft‑max approximation of the max.
    Returns the best iterate seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    step0 = 0.4
    beta = 0.9
    momentum = np.zeros_like(a)

    it = 0
    alpha0 = 5.0          # start temperature for the soft‑max
    alpha_max = 200.0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)

        # temperature schedule – slowly sharpen the approximation
        alpha = min(alpha_max, alpha0 * math.sqrt(it))
        w = _softmax(conv, alpha)          # soft‑max weights, length 2n‑1

        # gradient = 2·correlation(w, a)
        grad = 2.0 * _correlate_w_a(w, a)

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto simplex and symmetry enforcement
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} "
                  f"α={alpha:.1f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Baseline projected sub‑gradient descent (same update rule as the previous
    solution).  Returns the best iterate seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} "
                  f"score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5

    return best_a, best_score


# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing polish."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
