def _optimize_with_adam(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2   # initial
    t = 0
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self_fft(a).max()
    # gamma schedule
    gamma0 = 0.5    # start small
    gamma_factor = 1.05
    epoch_len = 500  # iterations per gamma increase

    while time.time() < deadline:
        t += 1
        # compute conv and current objective
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t} score={cur_score:.6f}")

        # compute gamma
        gamma = gamma0 * (gamma_factor ** (t // epoch_len))

        # compute softmax weights
        # shift to avoid overflow
        shift = gamma * (conv - cur_max)
        exp_shift = np.exp(shift)
        w = exp_shift / exp_shift.sum()

        # compute gradient base (2 * sum w a)
        grad_base = _conv_weight_fft(w, a)   # shape (n,)
        # final gradient
        grad = grad_base   # incorporate scaling in learning rate

        # Adam update
        m = beta1 * m + (1 - beta1) * grad

        # projection onto simplex
        a = _project_simplex(a)

    return best_a, best_score

import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a):
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _conv_weight_fft(w, a):
    # compute gradient component = 2 * sum_{j} w_{i+j} a_j
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    tmp = np.fft.irfft(W * A_rev, n=L)
    start = n - 1
    grad = 2.0 * tmp[start:start + n]
    return grad

def _project_simplex(v):
    # Euclidean projection onto simplex {x >=0, sum=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective(a):
    conv = _conv_self_fft(a)
    n = a.shape[0]
    return float(2.0 * n * conv.max())

def _adam_optimize(a0, deadline, start_time):
    a = a0.copy()
    n = a.shape[0]
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.3   # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    # schedule for γ (soft‑max temperature)
    gamma0 = 0.5
    gamma_factor = 1.08
    epoch_len = 500  # after how many iterations γ is multiplied
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution and score
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")
        # temperature for soft‑max
        gamma = gamma0 * (gamma_factor ** (it // epoch_len))
        # compute soft‑max weights
        shift = gamma * (conv - cur_max)   # shift by max to keep numbers ≤0
        exp_shift = np.exp(shift)
        w = exp_shift / exp_shift.sum()
        # gradient of smoothed objective (up to a constant step size)
        grad_base = _conv_weight_fft(w, a)   # = 2 * Σ w a
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_base
        v = beta2 * v + (1 - beta2) * (grad_base * grad_base)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        # keep on the simplex
        a = _project_simplex(a)
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective non‑negative step sequence.
    The algorithm starts from a family of arcsine‑type densities,
    augments with several random restarts and refines each start
    by a projected Adam optimizer that minimises a soft‑max
    approximation of the max‑convolution objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – we keep the same size as the benchmark
    n = 4096

    # ------------------------------------------------------------------
    # 1️⃣  Initialise a few promising candidates (arcsine sweep + random)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # arcsine‑family sweep (α in [0.2,0.6])
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    alphas = np.linspace(0.20, 0.60, 21)
    for alpha in alphas:
        w = (x * (1.0 - x) + 1e-30) ** (-alpha)
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] α={alpha:.3f} score={score:.6f}")

    # a few random Dirichlet starts
    for _ in range(6):
        a = np.random.rand(n)
        a = a / a.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # print only occasionally to save time
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Local refinement with Adam (multiple restarts)
    # ------------------------------------------------------------------
    # Prepare a pool of start points – the best so far plus a handful of
    # fresh random vectors.  The total remaining time is split evenly.
    remaining = deadline - time.time()
    if remaining > 5.0:
        start_pool = [best_seq]
        for _ in range(4):
            a = np.random.rand(n)
            a = a / a.sum()
            start_pool.append(a)

        per_run = remaining / len(start_pool)
        for idx, a0 in enumerate(start_pool):
            sub_deadline = time.time() + per_run
            refined, refined_score = _adam_optimize(a0, sub_deadline, start_time)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined.copy()
                print(f"[{time.time() - start_time:.2f}s] Refined run {idx} score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final projection onto the simplex and conversion to list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best score = {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Propose a step‑function (non‑negative heights) with a low evaluation score.

The algorithm proceeds in three stages:

1️⃣  **Arc‑sine sweep** – a small family of symmetric beta‑densities
    a_i ∝ (x_i·(1−x_i))^{−α} with α∈[0.2,0.6] is scanned.  The best
    candidate is kept as a deterministic starting point.

2️⃣  **Projected Adam optimisation** – the objective 2·n·max(conv) is
    replaced by a smooth soft‑max surrogate
        f(a)=2·n/γ·log∑_k e^{γ·conv_k(a)} .
    Its gradient can be written using a convolution, which we evaluate
    by FFT.  Adam with bias‑correction updates the vector and each
    iterate is projected onto the probability simplex (non‑negative,
    sum = 1).  The temperature γ is increased gradually (continuation
    method) so the optimisation first flattens the whole convolution
    and then concentrates on the worst entry.

3️⃣  **Optional random‑restart polishing** – after the main run a few
    short Adam runs start from fresh random points; any improvement
    replaces the current best.

All work is performed with FFT‑based convolutions (O(n log n)) and the
total wall‑time never exceeds the provided budget.  The returned
sequence is normalised to sum = 1, which does not affect the
evaluation function because it is scale‑invariant.
"""

import time
import math

import numpy as np


# ----------------------------------------------------------------------
# Utility functions (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·n‑1) using FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_weight_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute   g_i = 2·∑_j w_{i+j}·a_j   for i=0..n‑1.

    This is exactly the gradient part
        2·∑_k w_k·a_{k-i}
    needed for the soft‑max surrogate.
    The convolution is performed by FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    L = _next_pow2(len(w) + n - 1)      # length ≥ (2n‑1)+n‑1 = 3n‑2
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    tmp = np.fft.irfft(W * A_rev, n=L)   # linear convolution
    start = n - 1
    grad = 2.0 * tmp[start:start + n]    # factor 2 as required
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation used for logging: 2·n·max(a*a) (the denominator is 1
    because a is always projected onto the simplex).
    """
    conv = _conv_self_fft(a)
    n = a.shape[0]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser for the soft‑max surrogate of max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises the soft‑max surrogate of
    max(conv).  The iterates are always kept on the simplex.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15                       # base learning rate (after scaling)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # continuation schedule for the soft‑max temperature γ
    gamma0 = 0.5
    gamma_factor = 1.07
    epoch_len = 500                # increase γ every ``epoch_len`` iter

    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution and exact objective (for bookkeeping)
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} score={cur_score:.6f}")

        # temperature γ (soft‑max sharpness)
        gamma = gamma0 * (gamma_factor ** (it // epoch_len))

        # soft‑max weights (numerically stable)
        shift = gamma * (conv - cur_max)          # ≤0
        exp_shift = np.exp(shift)
        w = exp_shift / exp_shift.sum()           # Σ w = 1

        # gradient of the surrogate (includes the factor 2·n)
        grad_base = _conv_weight_fft(w, a)        # = 2·∑ w a
        grad = 2.0 * n * grad_base                 # = 4 n ∑ w a (exact gradient up to constant)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)       # keep a on the simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a plain Python list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size (same as the benchmark that produced 1.508…)
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1️⃣  Initialise candidates (arc‑sine sweep + a few random starts)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # arc‑sine family: a_i ∝ (x_i·(1−x_i))^{−α}
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions in (0,1)
    alphas = np.linspace(0.20, 0.60, 21)            # 21 values, cheap sweep
    for alpha in alphas:
        w = (x * (1.0 - x) + 1e-30) ** (-alpha)
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] α={alpha:.3f} score={score:.6f}")

    # a handful of random Dirichlet initialisers
    for _ in range(6):
        a = np.random.rand(n)
        a /= a.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2️⃣  Main refinement with Adam (the bulk of the time budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # allocate the majority of the remaining budget to the best start
        main_deadline = time.time() + 0.85 * remaining
        refined_seq, refined_score = _adam_optimize(best_seq, main_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Main refinement score={refined_score:.6f}")

        # spare time for a few short random‑restart runs (polishing)
        spare = deadline - time.time()
        while spare > 5.0:
            a0 = np.random.rand(n)
            a0 /= a0.sum()
            # each extra run gets at most 30 s or half of the remaining spare time
            sub_deadline = time.time() + min(spare * 0.5, 30.0)
            cand_seq, cand_score = _adam_optimize(a0, sub_deadline, start_time)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Extra run improve score={cand_score:.6f}")
            spare = deadline - time.time()

    # --------------------------------------------------------------
    # 3️⃣  Final normalisation and conversion to list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)  # defensive clipping
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
