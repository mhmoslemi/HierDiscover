def _simulated_annealing(start_a, deadline, start_time, L_fft):
    a = start_a.copy()
    cur_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    T = 0.005
    T_min = 1e-6
    decay = 0.9999
    it = 0
    while time.time() < deadline:
        it += 1
        # propose move: pick src proportional to a (or maybe to a^2) and dst random
        src = rng.choice(len(a), p=a)
        dst = rng.integers(len(a))
        if src == dst or a[src] < 1e-12:
            continue
        eps = a[src] * rng.random() * 0.05  # move up to 5% of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # evaluate
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it} T={T:.2e} best={best_score:.6f}")
        # update temperature
        T = max(T_min, T * decay)
    return best_a, best_score

"""
Propose a non‑negative step function of length 4096 that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of four stages:

1. **Very dense parametric sweep** – geometric, power‑law, beta‑type
   families and simple convex combinations.  This stage is cheap (a few
   seconds) but often discovers a shape that already beats the previous
   best bound.

2. **Smoothed gradient descent with random restarts** – the hard max in
   the objective is approximated by a soft‑max (temperature τ).  The
   iterate stays on the probability simplex via Euclidean projection.
   Each restart runs a handful of seconds; the best restart is kept.

3. **Simulated‑annealing style local search** – tiny transfers of mass
   between coordinates are accepted either if they improve the true
   objective or with a probability that slowly cools down.  This helps
   to escape the basin of attraction of the gradient descent.

4. **Projected sub‑gradient refinement** followed by a cheap deterministic
   hill‑climb polish.  These final steps squeeze out the last fraction
   of improvement.

The whole routine respects the 1000 s budget, prints progress
information and finally returns a plain ``list[float]`` that can be fed
directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective for a (assumes sum(a)≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Dense parametric sweep
# ----------------------------------------------------------------------
def _parametric_sweep(n: int,
                     L_fft: int,
                     deadline: float,
                     start_time: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Explore many low‑dimensional families and keep the best candidate."""
    idx = np.arange(n, dtype=np.float64)

    # ----- geometric families (r = 1‑eps) ---------------------------------
    eps_grid = np.logspace(-7, -1, 400)        # finer near 0
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum

        # mix with uniform – a few mixing ratios are enough
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ----- power‑law families ---------------------------------------------
    p_grid = np.arange(-2.0, 5.01, 0.2)       # extend beyond the old range
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1.0, p)
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq /= pow_sum
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

        # mix with uniform as well (helps escaping flat spots)
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * pow_seq
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc2 = _eval_fast(seq, L_fft)
            if sc2 < best_score:
                best_score = sc2
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Power+Uniform p={p:.2f} lam={lam:.2f} score={sc2:.6f}")

    # ----- beta‑type families (a_i ∝ (i+1)^p (n‑i)^q) --------------------
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = np.power(idx + 1.0, p) * np.power(n - idx, q)
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq /= beta_sum
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smoothed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed objective:
        f(a) = 2 * n * softmax_τ(conv(a,a))
    The temperature τ is exponentially decayed.
    Iterate stays on the simplex via Euclidean projection.
    Returns the best sequence encountered.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    # start with a relatively warm temperature – allows wider steps
    tau = 0.5
    tau_min = 1e-8
    tau_decay = 0.9995
    step0 = 0.3

    while time.time() < deadline:
        it += 1

        # ----- convolution and soft‑max weights -----
        conv = _conv_self(a, L=L_fft)              # length 2n‑1
        maxc = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()              # simplex, sum(w)=1

        # ----- gradient via FFT: grad_j = 2 * Σ_i a_i w_{i+j} ----------
        # The required correlation is a (linear) convolution with reversed w
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        # the relevant slice (see original implementation)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # ----- projected gradient step -----
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        # ----- exact objective evaluation -----
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # ----- temperature decay -----
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Simulated‑annealing style local search
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic search: move a (small) fraction of mass from a
    source index to a destination index, accepting the move either if it
    improves the true objective or with a probability that decays with a
    temperature schedule.
    """
    a = start_a.copy()
    cur_score = _eval_fast(a, L_fft)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.shape[0]

    T = 0.01               # initial temperature
    T_min = 1e-7
    decay = 0.99995        # slow exponential cooling

    it = 0
    while time.time() < deadline:
        it += 1
        # source proportional to current mass – moves mass from "big" places
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 5 % of the source mass
        eps = a[src] * rng.random() * 0.05
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} T={T:.2e} best={best_score:.6f}")

        # temperature update
        T = max(T_min, T * decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Projected sub‑gradient descent (as in the reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum and a 1/√t step‑size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (or a small active set)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 5️⃣  Deterministic hill‑climb polish
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves accepted only if the true objective improves."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                # pick proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10      # up to 10 % of src
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Combines an extensive parametric sweep, smoothed gradient descent,
    a short simulated‑annealing hill‑climb, a final sub‑gradient
    refinement and deterministic polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_fft = _next_pow2(2 * n - 1)          # for self‑convolution
    L_grad = _next_pow2(3 * n)             # for gradient convolution

    # ------------------------------------------------------------------
    # Initial solution – use the global starter if it exists, otherwise uniform
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Dense parametric sweep
    # ------------------------------------------------------------------
    best_seq, best_score = _parametric_sweep(
        n, L_fft, deadline - 200.0, start_time, best_seq, best_score
    )

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent with several restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 120.0:
        # random perturbation around the current best (or uniform)
        base = best_seq if np.random.rand() < 0.7 else uniform
        noise = 0.07 * np.random.randn(n) * base       # 7 % relative noise
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 12.0)   # ≤12 s per restart
        cand_seq, cand_score = _smooth_descent(start, sub_deadline,
                                               start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after smooth restart score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing style local search (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        sub_deadline = deadline - 50.0
        cand_seq, cand_score = _simulated_annealing(best_seq, sub_deadline,
                                                   start_time, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SA improvement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient refinement (bulk of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Deterministic hill‑climb polish (only if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improve score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
