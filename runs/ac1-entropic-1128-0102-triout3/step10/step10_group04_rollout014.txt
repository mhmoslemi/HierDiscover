def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    n = 4096

    # param search
    best_seq, best_score = param_search(n, deadline)

    # if time remains, run exp gradient many iter
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + remaining * 0.7
        seq, scr = _run_exp_gradient(best_seq, n, phase2_deadline, start, eta0=0.6)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print("[...] exp grad improve: score=...")

    # greedy reduction if time
    if time.time() < deadline:
        seq, scr = _run_greedy_reduction(best_seq, n, deadline, start, best_score, best_seq)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print("[...] greedy improve: score=...")

    # random restart polishing
    while time.time() < deadline - 1.0:
        pert = best_seq + 0.01 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)
        pert /= pert.sum()
        tmp_deadline = time.time() + 1.0
        seq, scr = _run_exp_gradient(pert, n, tmp_deadline, start, eta0=0.5)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print("[...] random restart improve: score=...")

    # final normalization
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0/n)
    return [float(x) for x in best_seq]

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

The optimiser works in four complementary stages:

1.  Low‑dimensional parametric search.  Several simple families of
    shapes (power‑law, exponential, mixed, edge‑peaked and a
    two‑parameter “beta” family) are scanned on a coarse grid and then
    refined with a tiny hill‑climber on the parameters.  This quickly
    finds a good “macro” shape without touching the full 4096‑dimensional
    space.

2.  Long exponentiated‑gradient (mirror‑descent) on the true
    max‑convolution, starting from the shape discovered in stage 1.
    The update is multiplicative and each step is projected onto the
    probability simplex, directly reducing the quantity
    ``2·n·max(conv)`` (the objective for a normalised vector).

3.  Deterministic greedy peak‑flattening.  Mass is repeatedly moved
    from the index that contributes most to the current maximum to an
    index that contributes least, with an analytically chosen step size.
    The full convolution is recomputed only occasionally to avoid drift.

4.  A final random‑restart polishing loop (tiny Gaussian perturbations
    followed by a short exp‑gradient burst) uses any remaining budget.

All three stages respect the global time budget (default 1000 s) and
print progress information.  The function finally returns a plain list
of floats whose sum is 1, which can be fed directly to ``evaluate_sequence``.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v≥0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Scoring routine (identical to evaluate_sequence for normalised vectors)
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Return 2·n·max(conv) for a normalised (sum=1) non‑negative vector."""
    n = a.shape[0]
    if a.sum() < 1e-12:
        return np.inf
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric shape generation
# ----------------------------------------------------------------------
def _generate_shape(shape_type: str, params, n: int) -> np.ndarray:
    """Create a normalised non‑negative shape."""
    if shape_type == "power":
        p, c = params
        a = (np.arange(n, dtype=float) + c) ** (-p)
    elif shape_type == "exp":
        (lam,) = params
        a = np.exp(-lam * np.arange(n, dtype=float))
    elif shape_type == "mixed":
        p, c, lam = params
        a = ((np.arange(n, dtype=float) + c) ** (-p)) * np.exp(-lam * np.arange(n, dtype=float))
    elif shape_type == "edge":
        (p,) = params
        i = np.arange(n, dtype=float)
        a = ((i + 1) * (n - i)) ** (-p)
    elif shape_type == "general":
        alpha, beta = params
        i = np.arange(n, dtype=float) + 1
        a = (i ** alpha) * ((n - np.arange(n, dtype=float)) ** beta)
    else:
        raise ValueError(f"unknown shape_type {shape_type}")

    a = np.maximum(a, 0.0)
    a += 1e-30                     # avoid exact zeros
    a /= a.sum()
    return a

                print(f"[{time.time():.2f}s] general α={alpha:.2f} β={beta:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 0c) Tiny hill‑climb on the best parameterisation (if any)
    # ------------------------------------------------------------------
    if best_type is not None:
        # number of hill‑climb steps limited by remaining time
        hc_steps = 0
        while time.time() < deadline and hc_steps < 5000:
            hc_steps += 1
            # propose a perturbed parameter set
            if best_type == "power":
                p, c = best_params
                p_new = max(p * (1 + 0.1 * np.random.randn()), 0.01)
                c_new = max(c * (1 + 0.1 * np.random.randn()), 0.01)
                params_new = (p_new, c_new)
            elif best_type == "exp":
                (lam,) = best_params
                lam_new = max(lam * (1 + 0.1 * np.random.randn()), 0.0)
                params_new = (lam_new,)
            elif best_type == "mixed":
                p, c, lam = best_params
                p_new = max(p * (1 + 0.1 * np.random.randn()), 0.01)
                c_new = max(c * (1 + 0.1 * np.random.randn()), 0.01)
                lam_new = max(lam * (1 + 0.1 * np.random.randn()), 0.0)
                params_new = (p_new, c_new, lam_new)
            elif best_type == "edge":
                (p,) = best_params
                p_new = max(p * (1 + 0.1 * np.random.randn()), 0.01)
                params_new = (p_new,)
            elif best_type == "general":
                alpha, beta = best_params
                alpha_new = np.clip(alpha + 0.2 * np.random.randn(), -2.0, 2.0)
                beta_new = np.clip(beta + 0.2 * np.random.randn(), -2.0, 2.0)
                params_new = (alpha_new, beta_new)
            else:
                break

            a = _generate_shape(best_type, params_new, n)
            sc = _score_of(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                best_params = params_new
                print(f"[{time.time():.2f}s] hill‑climb {best_type} score={sc:.6f}")

    return best_seq, best_score, best_type, best_params


# ----------------------------------------------------------------------
# Phase 1 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) optimiser that directly minimises
    ``max(conv)`` while keeping the vector on the probability simplex.
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max   # sum(a)=1 by construction
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad iter={it:,} score={cur_score:.6f}")

        # --- gradient of the active (max) convolution index ---
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]   # d/d a_i of conv[k]

        # --- mirror‑descent (multiplicative) update ---
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # --- renormalise onto the simplex (exact sum = 1) ---
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Move a tiny amount of mass from the index that most contributes to the
    current peak to an index that contributes least, updating the full
    convolution incrementally.  The full convolution is recomputed every
    few thousand steps to wipe out numerical drift.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and every other candidate
        partner_vals = a[max_idx - idx]

        # choose destination with smallest partner value (and not src)
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Random‑hill‑climb on the full vector (tiny moves)
# ----------------------------------------------------------------------
def _run_random_hill(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     cur_best_score: float,
                     cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb: repeatedly choose two indices, move a
    random fraction of mass from the larger to the smaller, accept only
    if the objective improves.  Uses the exact convolution (via FFT) for
    each trial.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # ensure we move from a larger entry
        if a[i] < a[j]:
            i, j = j, i
        if a[i] <= 0.0:
            continue
        eps = a[i] * np.random.rand() * 0.3  # up to 30 % of source mass
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        sc = _score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] random‑hill iter={it:,} score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``
    under the prescribed time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # problem size – 4096 offers a sweet spot between flexibility and speed
    n = 4096

    # --------------------------------------------------------------
    # Stage 0 – parametric exploration
    # --------------------------------------------------------------
    best_seq, best_score, best_type, best_params = _param_search(n, deadline)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Stage 1 – long exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # devote ~60 % of the remaining budget to this stage
        phase_deadline = time.time() + remaining * 0.60
        seq, scr = _run_exp_gradient(best_seq, n, phase_deadline, start_time, eta0=0.5)
        if scr < best_score:
            best_score, best_seq = scr, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve score={scr:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – deterministic greedy flattening
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_greedy_reduction(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=8_000_000,
        )
        if scr < best_score:
            best_score, best_seq = scr, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve score={scr:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – lightweight random‑hill‑climb for any remaining time
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq, scr = _run_random_hill(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
        )
        if scr < best_score:
            best_score, best_seq = scr, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑hill improve score={scr:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (the objective is homogeneous)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return plain Python list
    return [float(x) for x in best_seq]
