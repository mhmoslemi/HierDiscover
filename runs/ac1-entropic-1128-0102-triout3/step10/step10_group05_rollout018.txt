"""
Improved step‑function optimiser for `evaluate_sequence`.

The optimiser works in three stages:

1.  Initialise with a short grid search over geometric decays and keep the
    best exponential seed (this already beats the naïve uniform start).

2.  Adam on a smooth log‑sum‑exp approximation of the maximal convolution.
    The schedule is longer than before and uses a slightly larger learning
    rate to move faster away from the plateau.

3.  Deterministic “optimal‑greedy’’ reduction of the peak.  At each iteration
    we locate the current maximal convolution entry, pick the pair that
    contributes most to it and move exactly the amount of mass that brings
    this entry down to the second‑largest one (or as much as allowed by the
    source weight).  The update of the full convolution is performed
    incrementally, and the exact convolution is recomputed periodically to
    erase accumulated drift.

All steps respect the global 1000 s time budget, keep the sequence on the
probability simplex (non‑negative, sum = 1) and finally return a plain
Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a using FFT (full length 2·len(a)‑1).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Helper – exponential seed
# ----------------------------------------------------------------------
def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution a_i ∝ r**i   (0 < r < 1).
    Returned vector sums to 1.
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a

# Phase 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.8,
                        iters_per_alpha: int = 40_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp smoothed max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    best_score = 2.0 * n * np.max(_conv_self(a, L))

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (multiplicative) on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0
    L = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # active index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Deterministic optimal‑greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_optimal_greedy_reduction(a0: np.ndarray,
                                  n: int,
                                  deadline: float,
                                  start_time: float,
                                  cur_best_score: float,
                                  cur_best_seq: np.ndarray,
                                  max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    At each step:
        * locate the current maximal convolution entry,
        * pick the pair (i, k−i) that contributes most to it,
        * move exactly the amount of mass that brings this entry down to the
          second‑largest value (or as far as the source permits).
    The full convolution is updated incrementally; a full recomputation
    is performed every 1000 iterations to avoid drift.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)                # length 2n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        # cheap O(n) extraction
        if max_idx == 0:
            second_max = conv[1:].max()
        elif max_idx == conv.size - 1:
            second_max = conv[:-1].max()
        else:
            second_max = max(conv[:max_idx].max(), conv[max_idx + 1:].max())

        # indices that actually appear in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        partners = a[max_idx - idx]
        contrib = a[idx] * partners

        src_local = int(np.argmax(contrib))
        src = idx[src_local]
        partner_src = partners[src_local]

        # choose a destination outside the active interval if possible
        if lo > 0:
            dst = lo - 1
        elif hi < n - 1:
            dst = hi + 1
        else:
            # fallback – pick a random index not equal to src
            dst = src
            while dst == src:
                dst = np.random.randint(0, n)

        # partner contribution of dst to the current max (zero if outside interval)
        if lo <= dst <= hi:
            partner_dst = a[max_idx - dst]
        else:
            partner_dst = 0.0

        delta_per_eps = -2.0 * partner_src + 2.0 * partner_dst
        if delta_per_eps >= 0.0:
            # moving mass does not lower the current peak – skip this iteration
            continue

        # epsilon needed to bring the peak down to the second largest value
        needed_eps = (max_val - second_max) / (-delta_per_eps)
        eps = min(a[src], needed_eps)
        if eps <= 0.0:
            continue

        # keep a copy of the old vector for the incremental update
        a_old = a.copy()

        # apply the mass transfer
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # ----- incremental convolution update -----
        # adding eps at dst adds 2*eps*a_old to conv[dst : dst+n]
        # removing eps from src subtracts 2*eps*a_old from conv[src : src+n]
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, L)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] optimal greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # reserve a small safety margin

    final_n = 4096

    # ------------------------------------------------------------------
    # Initialise – use the best exponential decay (grid search)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            a0 = np.interp(np.linspace(0, 1, final_n),
                           np.linspace(0, 1, starter.size),
                           starter)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            raise Exception()
    except Exception:
        best_score = float("inf")
        best_r = 0.99
        for r in np.linspace(0.90, 0.9999, 300):
            cand = _exponential_sequence(final_n, r)
            conv = _conv_self(cand, _next_pow2(2 * final_n - 1))
            score = 2.0 * final_n * conv.max()
            if score < best_score:
                best_score = score
                best_r = r
        a0 = _exponential_sequence(final_n, best_r)
        print(f"[0.00s] exponential seed r={best_r:.6f}, score={best_score:.6f}")

    # normalise strictly to a probability vector
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_seq = a0.copy()
    best_score = 2.0 * final_n * _conv_self(best_seq,
                                            _next_pow2(2 * final_n - 1)).max()
    print(f"[0.00s] initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smoothed max (≈15 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase1_deadline = time.time() + max(20.0, 0.15 * (deadline - time.time()))
        a_adam, score_adam = _run_adam_smoothing(
            best_seq,
            final_n,
            phase1_deadline,
            start_time,
            alphas=[0.05, 0.15, 0.5, 1.0, 3.0, 8.0, 20.0],
            lr0=0.8,
            iters_per_alpha=30_000,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
                  f"score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (≈30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + max(30.0, 0.30 * (deadline - time.time()))
        a_exp, score_exp = _run_exp_gradient(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            eta0=1.2,
        )
        if score_exp < best_score:
            best_score = score_exp
            best_seq = a_exp.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic optimal‑greedy reduction (remainder of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_greedy, score_greedy = _run_optimal_greedy_reduction(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=5_000_000,
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy improve: "
                  f"score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    final_obj = 2.0 * final_n * _conv_self(best_seq,
                                          _next_pow2(2 * final_n - 1)).max()
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
