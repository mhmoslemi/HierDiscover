def _proj_gradient_descent(a0, n, deadline, start_time, best_seq, best_score):
    a = a0.copy()
    best = best_seq.copy()
    best_val = best_score
    cur_score = _eval_fast(a, L_self) # compute once

    # m and v for Adam maybe not used; we just use line search.

    step0 = 0.5  # initial step
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L_self) # full conv
        M = conv.max()
        s = a.sum()
        if s <= 0:
            break
        k = int(conv.argmax())
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        idx = np.arange(lo, hi+1, dtype=int)
        grad_M = np.zeros_like(a)
        grad_M[idx] = 2.0 * a[k - idx]  # transposed

        # gradient of objective
        const = 2.0 * n / (s * s)
        grad = const * grad_M - (4.0 * n * M / (s ** 3)) * np.ones_like(a)

        # backtracking line search
        eta = step0
        # compute target for improvement: must be less than cur_score
        improved = False
        for _ in range(10):
            a_try = a - eta * grad
            # projection onto simplex
            a_try = _project_simplex(a_try)
            # evaluate fast (since sum should close to 1)
            new_score = _eval_fast(a_try, L_self)
            if new_score < cur_score - 1e-12:
                a = a_try
                cur_score = new_score
                improved = True
                if new_score < best_val:
                    best_val = new_score
                    best = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD it={it} score={new_score:.6f}")
                break
            else:
                eta *= 0.5
        if not improved:
            # no improvement; perhaps reduce step0 for future
            step0 *= 0.5
            # if step0 too small break
            if step0 < 1e-12:
                break
    return best, best_val

"""
Propose a step–function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for the fixed length n = 4096.

The algorithm is a lightweight meta‑heuristic:

1️⃣  Build several cheap initial candidates (uniform, half‑sine, a few
    geometric shapes and a few random Dirichlet vectors) and keep the best.

2️⃣  Refine the current best with a **projected Adam‑style gradient descent**
    on the exact objective.  The gradient is the sub‑gradient of the active
    convolution entry; after each step the vector is projected onto the
    probability simplex (non‑negative, sum = 1).

3️⃣  Finish with a short **simulated‑annealing mass‑transfer polish** that
    moves a small random amount of mass from one position to another.
    The annealer accepts downhill moves always and uphill moves with the
    usual Metropolis probability.

All heavy work is done with FFT‑based convolutions; the total wall‑clock
time never exceeds the supplied budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers (FFT convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution, not a circular one.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assuming sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on the exact objective
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_seq: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam updates on f(a)=2·n·max(a*a) / (sum a)^2.
    The iterate is always kept on the simplex.
    """
    a = a0.copy()
    best = best_seq.copy()
    best_val = best_score

    L_self = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    bias1 = 1.0
    bias2 = 1.0
    it = 0

    while time.time() < deadline:
        # ---------- evaluate current point ----------
        conv = _conv_self(a, L_self)
        M = conv.max()
        cur_score = 2.0 * n * M / (a.sum() ** 2)

        if cur_score < best_val:
            best_val = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # ---------- sub‑gradient of the max‑convolution ----------
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad_M = np.zeros_like(a)
        grad_M[idx] = 2.0 * a[k - idx]                     # ∂max/∂a

        # ---------- gradient of the objective (sum(a)=1 after projection) ----------
        # With sum(a)=1 we have ∇f = 2·n·grad_M - 4·n·M·1 .
        grad = 2.0 * n * (grad_M - 2.0 * M)

        # optional clipping to keep numbers well‑scaled
        max_abs = np.max(np.abs(grad))
        if max_abs > 10.0:
            grad = grad * (10.0 / max_abs)

        # ---------- Adam update ----------
        it += 1
        bias1 *= beta1
        bias2 *= beta2

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - bias1)
        v_hat = v / (1.0 - bias2)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)          # keep on simplex

    return best, best_val


# ----------------------------------------------------------------------
# Tiny simulated‑annealing polish (identical to reference)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

        best_seq, best_score = _annealing(
            best_seq, n, deadline, start_time,
            best_score, best_seq,
            start_temp=5e-2, end_temp=1e-7,
            steps=400_000)
        print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
