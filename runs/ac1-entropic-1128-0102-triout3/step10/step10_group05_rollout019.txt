"""
Improved step‑function optimiser.
Searches for the best non‑negative sequence of length 4096 that minimises
evaluate_sequence.  The optimiser consists of:
1. a high‑resolution ternary search over the decay factor r of a geometric
   distribution (the best known analytic family);
2. a short Adam optimiser on a log‑sum‑exp smooth approximation of the
   max‑convolution;
3. a long greedy “peak‑flattening’’ phase that repeatedly moves a small
   amount of mass from the coordinate that contributes most to the current
   maximal convolution value to a coordinate that contributes least.
All steps respect the 1000 s budget.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full self‑convolution a*a using FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Geometric distribution a_i ∝ r**i on n indices,
    normalised to sum 1 (r∈(0,1)).
    """
    if not (0.0 < r < 1.0):
        raise ValueError("r must be in (0,1)")
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # geometric sum 1‑r**n
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    return a

def _objective_from_seq(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)).  Assumes seq is non‑negative and already normalised."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(seq, L)
    return float(2.0 * n * conv.max())

def _search_best_r(n: int, low: float, high: float, iters: int, deadline: float, start_time: float):
    """
    Ternary search for the decay factor r that minimises the objective.
    Returns (best_r, best_score).
    """
    best_r = low
    best_score = float('inf')
    L = _next_pow2(2 * n - 1)

    for _ in range(iters):
        if time.time() >= deadline:
            break
        r1 = low + (high - low) / 3.0
        r2 = high - (high - low) / 3.0

        # evaluate r1
        a1 = _exponential_sequence(n, r1)
        conv1 = _conv_self(a1, L)
        score1 = float(2.0 * n * conv1.max())

        # evaluate r2
        a2 = _exponential_sequence(n, r2)
        conv2 = _conv_self(a2, L)
        score2 = float(2.0 * n * conv2.max())

        if score1 < best_score:
            best_score = score1
            best_r = r1
        if score2 < best_score:
            best_score = score2
            best_r = r2

        if score1 < score2:
            high = r2
        else:
            low = r1

    # one final dense scan around the final interval to avoid missing a narrow optimum
    fine_rs = np.linspace(low, high, 200)
    for r in fine_rs:
        if time.time() >= deadline:
            break
        a = _exponential_sequence(n, r)
        conv = _conv_self(a, L)
        sc = float(2.0 * n * conv.max())
        if sc < best_score:
            best_score = sc
            best_r = r

    return best_r, best_score

def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.5,
                        iters_per_alpha: int = 50000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth (log‑sum‑exp) approximation of the max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                continue
            w /= w_sum
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto simplex
            a = _project_simplex(a)

    return best_a, best_score

def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy reduction of the peak.
    At each step a small amount of mass is transferred from the
    index that contributes most to the current maximum convolution value
    to an index that contributes the least.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)  # full convolution (length 2n‑1)

    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2000

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build vector a_k = a[max_idx - i] for i in support of src
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose destination with smallest a_k value (different from src)
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.03 * a[src])  # a slightly larger step when we are not improving the slope
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-12)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.03 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        # update convolution in O(n) using vectorised slices
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Greedy improve: iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    # ------------------------------------------------------------------

        a_adam, score_adam = _run_adam_smoothing(
            best_seq,

    # ------------------------------------------------------------------
    # Normalise and return the result as a plain list.

"""
Improved step‑function optimiser.

Searches for a non‑negative step‑function (length 4096) that minimises
``evaluate_sequence``.  The optimiser proceeds in three stages:

1.  A high‑resolution ternary search over the decay factor *r* of a geometric
    distribution (a_i ∝ r**i) – the analytic family that is already very close
    to optimum.  The search is performed with a tight time budget and a final
    fine scan, yielding the best exponential seed.

2.  Adam optimisation on a smooth (log‑sum‑exp) surrogate of the max‑convolution
    to fine‑tune the vector while keeping it on the probability simplex.

3.  A long greedy “peak‑flattening’’ phase that repeatedly moves a small amount
    of mass from the coordinate that contributes most to the current maximal
    convolution value to a coordinate that contributes the least.  The loop
    runs until the overall time budget expires.

All steps are bounded by the 1000 s limit, print progress messages and finally
return the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a using FFT of length L.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Helper – exponential seed
# ----------------------------------------------------------------------
def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution (a_i ∝ r**i) on n points.
    r must lie in (0,1).  The vector sums to one.
    """
    if not (0.0 < r < 1.0):
        raise ValueError("r must be in (0,1)")
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise (finite sum = (1‑r**n))
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    return a


def _objective_from_seq(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)).  Assumes seq is non‑negative and normalised."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(seq, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1) Ternary search for the best exponential decay factor
# ----------------------------------------------------------------------
def _search_best_r(n: int,
                   low: float,
                   high: float,
                   iters: int,
                   deadline: float,
                   start_time: float) -> tuple[float, float]:
    """
    Perform a ternary search on r∈[low,high] to minimise the objective.
    Returns the best r found together with its score.
    """
    best_r = low
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)

    for _ in range(iters):
        if time.time() >= deadline:
            break
        r1 = low + (high - low) / 3.0
        r2 = high - (high - low) / 3.0

        a1 = _exponential_sequence(n, r1)
        conv1 = _conv_self(a1, L)
        score1 = float(2.0 * n * conv1.max())

        a2 = _exponential_sequence(n, r2)
        conv2 = _conv_self(a2, L)
        score2 = float(2.0 * n * conv2.max())

        if score1 < best_score:
            best_score = score1
            best_r = r1
        if score2 < best_score:
            best_score = score2
            best_r = r2

        if score1 < score2:
            high = r2
        else:
            low = r1

    # fine‑grained scan over the final interval
    fine_rs = np.linspace(low, high, 200)
    for r in fine_rs:
        if time.time() >= deadline:
            break
        a = _exponential_sequence(n, r)
        conv = _conv_self(a, L)
        sc = float(2.0 * n * conv.max())
        if sc < best_score:
            best_score = sc
            best_r = r

    return best_r, best_score


# ----------------------------------------------------------------------
# 2) Adam optimisation on a smooth max‑convolution proxy
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 60000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp smooth approximation of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max via log‑sum‑exp -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                continue
            w /= w_sum
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3) Greedy reduction of the maximal convolution entry
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 20_000_000) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy sweep: repeatedly move a small amount of mass from the
    index that most contributes to the current peak to an index that contributes
    the least.  The loop proceeds until the time budget expires.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length 2n‑1)

    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2000

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # choose a destination with the smallest partner value a[max_idx‑i]
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.03 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-12)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.03 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental convolution update (O(n)) -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid numerical drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Greedy improve: iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score
