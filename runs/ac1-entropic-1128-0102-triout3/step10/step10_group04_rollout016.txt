   plateaus of the pure max‑gradient method.

2. Run a long smooth projected‑gradient descent (PGD).  The
   temperature (β) of the soft‑max is increased during the run,
   turning the smoothed objective into the true max‑objective.
   After each PGD step the vector is projected onto the probability
   simplex (Euclidean projection) to keep it a valid step‑function.

3. Apply a deterministic greedy “peak‑flattening’’ phase identical
   to the one used by the reference code – it moves a tiny amount of
   mass from the index that contributes most to the current peak to
   an index that contributes least, updating the convolution
   incrementally.

4. Finish with a few random‑restart bursts of short PGD runs.

The whole routine respects a 1000 s budget, prints progress messages,
and finally returns the best sequence found as a plain Python list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


def _corr_valid_fft(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Valid correlation of w (len 2·n‑1) with a (len n):
    result[i] = sum_j w[i + j] * a[j]   for i=0,…,n‑1.
    Implemented via FFT (convolution with reversed a).
    """
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, n=L)
    Af = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(Wf * Af, n=L)
    # the valid part starts at index n-1
    start = a.shape[0] - 1
    return full[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Phase 2 – Smooth projected‑gradient descent on a soft‑max of the convolution
# ----------------------------------------------------------------------


def _run_smooth_pg(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   beta_start: float = 2.0,
                   beta_end: float = 200.0,
                   beta_mult: float = 1.25,
                   beta_change_iters: int = 5000,
                   eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed objective
        f(a) = 2·n·softmax_β(a*a)
    where softmax_β(z) = (1/β)·log Σ exp(β·z_i).

    Returns the best vector seen and its true objective value
    (2·n·max(a*a)).
    """
    a = np.asarray(a0, dtype=float).copy()
    L_conv = _next_pow2(2 * n - 1)                # for self‑convolution
    L_corr = _next_pow2(3 * n - 2)                # for correlation w * a_rev
    best_a = a.copy()
    # true objective (using max, not the smooth surrogate)
    conv = _conv_self_fft(a, L_conv)
    best_score = 2.0 * n * conv.max()

    beta = beta_start
    next_beta_it = beta_change_iters
    it = 0

    while time.time() < deadline:
        it += 1

        # --- current convolution & true objective ---
        conv = _conv_self_fft(a, L_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑PG iter={it:,} "
                  f"beta={beta:.1f} score={cur_score:.6f}")

        # --- soft‑max weights (dense sub‑gradient) ---
        maxc = cur_max
        # exponent arguments are ≤ 0, safe from overflow
        exp_arg = beta * (conv - maxc)
        exp_vals = np.exp(exp_arg)
        w = exp_vals / exp_vals.sum()            # sum_i w_i = 1

        # --- gradient of the smoothed objective w.r.t. a ---
        # grad = 4·n·corr_valid(w, a)
        grad = 4.0 * n * _corr_valid_fft(w, a, L_corr)

        # --- additive update + Euclidean projection onto simplex ---
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _project_simplex(a)

        # --- anneal β ---
        if it >= next_beta_it:
            beta = min(beta * beta_mult, beta_end)
            next_beta_it += beta_change_iters

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------


def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Greedy “move a tiny amount of mass from the index that most
    contributes to the current peak to an index that contributes
    least”, updating the full convolution incrementally.
    """
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest entry (ignore current max)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)

        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and every other candidate
        partner_vals = a[max_idx - idx]

        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size ε
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-30)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental convolution update -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        if it % recompute_interval == 0:
            conv = _conv_self_fft(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper – construction of a few deterministic seeds
# ----------------------------------------------------------------------


def _make_arcsine_shape(n: int) -> np.ndarray:
    """U‑shaped arcsine‑like distribution: 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    # shift by 0.5 to avoid division by zero at the boundaries
    raw = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    raw = np.maximum(raw, 0.0)
    raw /= raw.sum()
    return raw


def _make_initial_candidates(n: int):
    """Return a list of deterministic starting vectors (already normalised)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    cand.append(lin / lin.sum())

    # sqrt‑law
    sqrt_shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    cand.append(sqrt_shape / sqrt_shape.sum())

    # exponential decay (q close to 1)
    q = 0.9995
    exp_shape = (1 - q) * (q ** np.arange(n, dtype=float))
    cand.append(exp_shape / exp_shape.sum())

    # arcsine‑like U‑shape
    cand.append(_make_arcsine_shape(n))

    # optional: use a previously discovered construction if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            interp = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, starter.size),
                starter,
            )
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            cand.append(interp)
    except Exception:
        pass

    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the prescribed time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off between flexibility and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Initialise – short warm‑up runs on a few deterministic shapes.
    # ------------------------------------------------------------------
    initial_candidates = _make_initial_candidates(n)

    best_seq = None
    best_score = float("inf")

    for cand in initial_candidates:
        if time.time() >= deadline:
            break
        # give each candidate a few seconds of warm‑up
        warm_deadline = time.time() + min(5.0, (deadline - time.time()) * 0.03)
        seq, scr = _run_smooth_pg(
            cand,
            n,
            warm_deadline,
            start_time,
            beta_start=2.0,
            beta_end=60.0,
            beta_mult=1.5,
            beta_change_iters=2000,
            eta0=0.6,
        )
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] warm‑up improve: "
                  f"score={scr:.6f}")

    # fallback – uniform if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Phase 2 – long smooth projected‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + remaining * 0.65
        seq, scr = _run_smooth_pg(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            beta_start=5.0,
            beta_end=200.0,
            beta_mult=1.2,
            beta_change_iters=4000,
            eta0=0.5,
        )
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] long smooth‑PG improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑flattening.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_greedy_reduction(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=8_000_000,
        )
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Optional quick random‑restart polishing (use any remaining time).
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # small random perturbation of the current best
        pert = best_seq + 0.02 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)
        pert /= pert.sum()

        tmp_deadline = time.time() + 2.0
        seq, scr = _run_smooth_pg(
            pert,
            n,
            tmp_deadline,
            start_time,
            beta_start=10.0,
            beta_end=150.0,
            beta_mult=1.3,
            beta_change_iters=2500,
            eta0=0.4,
        )
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑restart improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (objective is homogeneous – any scaling works)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return as plain Python list
    return [float(x) for x in best_seq]
