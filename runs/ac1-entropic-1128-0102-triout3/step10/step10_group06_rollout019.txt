"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm (LP + simple greedy polishing)
-------------------------------------------------
1. **Random initialisation** – draw a handful of Dirichlet vectors with
   several concentration parameters, keep the one with the smallest
   exact score returned by ``evaluate_sequence``.
2. **Linear‑programming improvement** – given the current vector *f*,
   compute the maximum of its self‑convolution *M*.
   Solve the LP

       maximise    Σ b_j
       subject to  (f * b)[k] ≤ M   for all k = 0 … 2·n‑2
                   b_j ≥ 0

   The matrix of the constraints is the (2·n‑1)×n convolution matrix
   built from *f*.  The solution *b* has a larger ℓ₁‑norm than *f* while
   still respecting the convolution bound.
   Rescale ``b`` to a probability vector and move a fraction *t* towards it:
   ``f ← (1‑t)·f + t·b``.  Repeat this step many times (the LP is tiny, so
   a few dozen iterations are cheap).
3. **Greedy local polishing** – after the LP phase, repeatedly move a
   small amount of probability mass from the index that contributes most
   to the current convolution peak towards an index whose partner contribution
   to that peak is minimal.  Accept only strict improvements.
4. **Return** – clip, renormalise and output the best vector found.

The whole routine respects the supplied time budget (default 1000 s) and
prints progress information.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    return np.convolve(a, a)


# ----------------------------------------------------------------------
# Greedy local improvement
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> np.ndarray:
    """
    Deterministic greedy peak‑reduction.
    Moves a fraction of the coefficient that most contributes to the
    current convolution peak towards an index whose partner entry for
    that peak is smallest.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # Choose destination that hurts the peak the least:
        partner_vals = np.full(n, np.inf)
        valid = (k_max - np.arange(n) >= 0) & (k_max - np.arange(n) < n)
        partner_vals[valid] = a[k_max - np.arange(n)[valid]]
        # Exclude the source index itself
        partner_vals[src] = np.inf
        dst = np.argmin(partner_vals)

        eps = 0.25 * a[src]          # fraction of source mass to move
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement
# ----------------------------------------------------------------------


def _lp_improve(f: np.ndarray,
                t: float,
                max_iter: int,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described in the doc‑string.
    Returns the best vector seen and the corresponding exact score.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = float("inf")

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_self(f)
        M = conv.max()

        # Build the (2n‑1)×n convolution matrix of f.
        # Row k, column j contains f[k‑j] when 0≤k‑j<n, else 0.
        rows = np.arange(2 * n - 1)[:, None]  # (2n‑1,1)
        cols = np.arange(n)                    # (n,)
        # broadcasting gives a (2n‑1, n) array of k‑j
        idx = rows - cols                      # shape (2n‑1, n)
        mask = (idx >= 0) & (idx < n)
        A = np.zeros((2 * n - 1, n), dtype=float)
        A[mask] = f[idx[mask]]

        # Linear programme: maximise Σ b  (i.e. minimise –Σ b)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        # Use the 'highs' method – fast for modest sizes.
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # If the LP fails we abort the loop.
            print(f"[{time.time() - start_time:6.2f}s] LP failed at iter {it}: {res.message}")
            break

        b = res.x
        sumb = b.sum()
        if sumb <= 0:
            break
        # Normalise to a probability vector.
        g = b / sumb

        # Mix towards the new direction.
        f = (1.0 - t) * f + t * g
        f = np.clip(f, 0.0, None)
        s = f.sum()
        if s > 0:
            f /= s

        # Exact evaluation.
        sc = evaluate_sequence(f.tolist())
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve iter {it+1}: score={sc:.6f}")

    return best_f, best_score

# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Uses a small LP‑based refinement together with a final greedy polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a modest n keeps the LP cheap.
    # ------------------------------------------------------------------
    n = 300
    print(f"[{time.time() - start_time:6.2f}s] using length n={n}")

    # ------------------------------------------------------------------
    # Generate a diverse pool of random starts and keep the best.
    # ------------------------------------------------------------------
    starts = []

    # Use the previously known construction if it exists.
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # simple down‑sampling / up‑sampling by linear interpolation
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # Hand‑crafted simple starts.
    starts.append(np.full(n, 1.0 / n))                # uniform
    starts.append(np.exp(-np.arange(n) / (n / 6)))    # discrete Gaussian
    starts.append(np.power(0.985, np.arange(n)))     # mild exponential
    starts.append(np.power(0.9995, np.arange(n)))    # very mild exponential

    # Random Dirichlet draws with a variety of concentration parameters.
    conc_params = [0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_params:
        for _ in range(8):
            draws = rng.gamma(conc, 1.0, size=n)
            draws /= draws.sum()
            starts.append(draws)

    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} score={sc:.6f}")

    # ------------------------------------------------------------------
    # LP‑based polishing.
    # ------------------------------------------------------------------
    # Mixing fraction – start with a relatively aggressive move and slowly
    # decay it so that later iterations make finer adjustments.
    mix = 0.6
    lp_iters = 30                      # enough for a noticeable improvement
    polished_seq, polished_score = _lp_improve(best_seq, mix,
                                               max_iter=lp_iters,
                                               deadline=deadline,
                                               start_time=start_time)
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after LP phase: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy polishing (deterministic local search).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq = _greedy_local_improve(best_seq, deadline, start_time,
                                          max_iters=5000)
        final_score = evaluate_sequence(final_seq.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and safety‑check before returning.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
