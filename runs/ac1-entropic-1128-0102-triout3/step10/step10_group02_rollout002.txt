def subgradient_descent_active(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a, best_score = a.copy(), best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    L_grad = _next_pow2(3 * n - 2)
    iteration = 0
    # Tolerance for active set
    tol = 1e-12 * 0.01? Actually we need tolerance relative to max value.
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a, L=_next_pow2(2*n-1))
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={iteration} score={cur_score:.6f}")

        # Determine active set mask
        eps_active = 1e-12
        # relative tolerance
        rel = 1e-9
        active_mask = np.where(conv >= cur_max - rel * cur_max)[0]
        # Build weight w array length 2n-1 where w[k]=1 for active else 0
        w = np.zeros_like(conv)
        w[active_mask] = 1.0

        # Compute gradient via convolution w * a_rev
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # momentum + step
        momentum = beta * momentum + (1-beta) * grad
        step = step0 / math.sqrt(iteration)
        a = a - step * momentum
        # project onto simplex
        a = _project_simplex(a)

    return best_a, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------

1. Load a starter (global `height_sequence_1` if present) otherwise use a uniform
   probability vector.

2. **Smooth exponential‑gradient descent**  
   Optimises a smooth approximation of the max‑convolution (soft‑max
   with a geometrically increasing temperature).  The update is a
   mirror‑descent on the simplex (exponentiated‑gradient) and the best
   iterate is kept.

3. **Active‑set sub‑gradient descent**  
   Uses the true max‑convolution but, when several indices attain the
   maximal value, the sub‑gradient is the sum of all corresponding
   contributions (instead of a single active index).  Momentum and a
   1/√t step‑size are employed; each step is projected onto the simplex.

4. **Random local mass‑transfer search**  
   Repeatedly moves a small fraction of mass from a source index to a
   destination index and accepts the move if it improves the objective
   (or with a tiny Simulated‑Annealing probability).  This cheap
   neighbourhood search can escape shallow local minima.

5. **Final short annealing** (optional) – the same tiny mass‑transfer
   annealer as in the reference solution.

All phases are time‑guarded; the routine stops before the given budget
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    target_len = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(target_len)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:target_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective for vectors normalised to sum = 1
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max mirror descent
# ----------------------------------------------------------------------
def _smooth_descent_mod(a0: np.ndarray, n: int,
                        deadline: float, start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on a soft‑max approximation
    of max(conv).  A geometric schedule of the smoothing parameter
    `alpha` is used; the best iterate is returned.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # smoothing parameters – start very soft, finish very sharp
    alphas = np.geomspace(0.1, 5000.0, num=60)

    it = 0
    max_iters = 500_000  # safety cap

    for alpha in alphas:
        if time.time() >= deadline:
            break
        # inner loop for current alpha
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)       # stability shift
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)   # avoid exact zeros
            a = a / a.sum()

            # occasional real‑objective check
            if it % 1000 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        # break early if out of time
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – active‑set sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent_active(a0: np.ndarray, n: int,
                                deadline: float, start_time: float,
                                best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    All indices attaining the maximal convolution value contribute
    to the sub‑gradient (active set).  Momentum and a 1/√t stepsize
    are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    L_grad = _next_pow2(3 * n - 2)
    it = 0

    # tolerance for deciding “active” indices (relative to max)
    rel_tol = 1e-12

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=_next_pow2(2 * n - 1))
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} score={cur_score:.6f}")

        # ---- active set ----
        # pick all indices within relative tolerance of the maximal value
        mask = conv >= cur_max - rel_tol * cur_max
        w = np.zeros_like(conv)
        w[mask] = 1.0

        # gradient = 2 * (w * a_rev) (central slice)
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

def _random_local_search(a0: np.ndarray, n: int,
                         deadline: float, start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a random fraction of the mass
    from a source index (chosen proportionally to its current weight)
    to a destination index.  Improvements are kept; occasional
    uphill moves are accepted with a tiny temperature schedule.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    L_self = _next_pow2(2 * n - 1)

    it = 0
    # annealing temperature schedule
    start_T = 5e-2
    end_T   = 1e-7

    while time.time() < deadline:
        it += 1
        # source weighted by current mass, destination uniform
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 30 % of src mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L=L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a_prop.sum() ** 2)

        accept = False
        if cur_score < best_score:
            accept = True
        else:
            # very low‑temperature uphill acceptance
            elapsed = time.time() - start_time
            total = deadline - start_time
            T = start_T * (end_T / start_T) ** (elapsed / total)
            if math.exp(-(cur_score - best_score) / max(T, 1e-12)) > rng.random():
                accept = True

        if accept:
            a = a_prop
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                if it % 1000 == 0:
                    print(f"[{time.time() - start_time:.2f}s] LocalSearch it={it:,} best={best_score:.6f}")

    return best_a, best_score

# Phase 4 – final tiny annealing (same as reference)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray, n: int,
               deadline: float, start_time: float,
               cur_best_score: float, cur_best_seq: np.ndarray,
               start_temp: float = 5e-2, end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

            best_score = score_rand
            best_seq = a_rand.copy()
            print(f"[{time.time() - start_time:.2f}s] Random init improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Smooth‑max mirror descent (core of the new method)
    # --------------------------------------------------------------
    if time.time() < deadline - 200:
        # allocate ~15 % of remaining time to this phase
        remaining = deadline - time.time()
        smooth_deadline = time.time() + max(30.0, 0.15 * remaining)
        best_seq, best_score = _smooth_descent_mod(best_seq, n,
                                                   smooth_deadline, start_time,
                                                   best_score)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Active‑set sub‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        sub_deadline = deadline - 20.0  # keep a little room for later phases
        best_seq, best_score = _subgradient_descent_active(best_seq, n,
                                                           sub_deadline, start_time,
                                                           best_score)
        print(f"[{time.time() - start_time:.2f}s] After active sub‑grad best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Random local mass‑transfer search
    # --------------------------------------------------------------
    if time.time() < deadline - 5:
        local_deadline = deadline - 5.0
        best_seq, best_score = _random_local_search(best_seq, n,
                                                    local_deadline, start_time,
                                                    best_score)
        print(f"[{time.time() - start_time:.2f}s] After local search best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final tiny annealing (optional)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=1e-2, end_temp=1e-7,
                                       steps=300_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
