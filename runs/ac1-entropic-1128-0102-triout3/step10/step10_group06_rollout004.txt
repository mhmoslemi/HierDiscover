Propose a non‑negative step function (probability vector) that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Rich initialisation** – a handful of handcrafted windows
   (uniform, Hann, Hamming, Blackman, exponential, Gaussian) and a few
   Dirichlet draws with different concentrations.

2. **Exact smooth‑max optimisation** – for every start we run a few
   L‑BFGS refinements on the log‑sum‑exp surrogate of the convolution peak.
   The surrogate is
        fα(p)= (1/α)·log Σₖ exp(α·(p∗p)ₖ) .
   Its gradient is obtained analytically via FFT, and we optimise the
   *log‑probability* parameters `y` (`p = softmax(y)`).  A schedule of
   increasing smoothing parameters α∈{0.5,2,8,32,128,512,2048} drives the
   solution toward the true max‑convolution objective.

3. **Multiplicative mirror‑descent polishing** – the best vector returned
   by the L‑BFGS phase is further polished with the fast exponentiated‑
   gradient method from the reference code (full sub‑gradient, diminishing
   step‑size).  This phase runs for the majority of the time budget.

4. **Deterministic greedy tightening** – a final local search moves a
   fraction of the coefficient that contributes most to the current peak
   towards the smallest coefficient, accepting only strict improvements.

The routine respects the given time budget (default 1000 s) and returns the
best sequence found (normalised to sum = 1, all entries ≥ 0).  Only NumPy
(and SciPy if available) are required.
"""

import math
import numpy as np

# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

# Window generators (hand‑crafted starts)
# ----------------------------------------------------------------------


def _uniform_start(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _exponential_start(n: int, r: float = 0.985) -> np.ndarray:
    v = np.array([r ** i for i in range(n)], dtype=float)
    return v / v.sum()


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    x = np.arange(n)
    g = np.exp(-0.5 * ((x - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _hann_start(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return w / w.sum()


def _hamming_start(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman_start(n: int) -> np.ndarray:
    i = np.arange(n)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return w / w.sum()


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. log‑probabilities
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(y: np.ndarray, alpha: float, fft_len: int):
    """
    Returns (f, g) where

        f = (1/α)·log Σₖ exp(α·(p∗p)ₖ)          (smooth surrogate of max)
        g = ∂f/∂y  (gradient w.r.t. the log‑probability vector y)

    p = softmax(y) is the underlying probability distribution.
    """
    # ---- p = softmax(y) -------------------------------------------------
    p = _softmax(y)
    n = p.shape[0]

    # ---- convolution p∗p ------------------------------------------------
    conv = _conv_self(p, fft_len)                     # length 2n‑1

    # ---- smooth max (log‑sum‑exp) ---------------------------------------
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    f = lse / alpha

    # ---- gradient w.r.t. p ---------------------------------------------
    # weights of the soft‑max over the convolution entries
    w = np.exp(scaled - m)
    w /= w.sum()                                       # shape (2n‑1,)

    # convolution of p with reversed w gives ∂(Σ w_k (p∗p)_k)/∂p
    w_rev = w[::-1]
    Af = np.fft.rfft(p, n=fft_len)
    Wr = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * Wr, n=fft_len)
    Lw = w_rev.shape[0]                               # = 2n‑1
    corr = corr[: n + Lw - 1]
    grad_p = 2.0 * corr[Lw - 1 - np.arange(n)]        # ∂f/∂p (still needs chain rule)

    # ---- chain rule for softmax (∂p/∂y) --------------------------------
    dot = np.dot(grad_p, p)
    grad_y = p * (grad_p - dot)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS refinement on the smooth surrogate
# ----------------------------------------------------------------------


def _lbfgs_refine(p0: np.ndarray, alphas, fft_len, deadline, start_time):
    """
    Run a short L‑BFGS optimisation for each α in ``alphas``.
    The variable is the log‑probability vector ``y`` (so positivity is
    guaranteed).  Returns the refined probability vector and its exact
    score (via ``evaluate_sequence``).
    """
    # fall back to a simple gradient descent if SciPy is unavailable
    try:
        from scipy.optimize import minimize
        HAVE_SCIPY = True
    except Exception:  # pragma: no cover
        HAVE_SCIPY = False

    # initialise y
    eps = 1e-12
    y = np.log(np.maximum(p0, eps))

    for alpha in alphas:
        if time.time() >= deadline:
            break

        # number of function evals per α – we keep it modest to stay within budget
        maxfun = 120

        if HAVE_SCIPY:
            # SciPy expects a callable returning (f, grad)
            def fun(y_vec):
                return _smooth_obj_and_grad(y_vec, alpha, fft_len)

            res = minimize(fun,
                           y,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxfun': maxfun,
                                    'maxiter': maxfun,
                                    'ftol': 1e-12,
                                    'disp': False})
            y = res.x
        else:
            # very simple gradient descent with a diminishing step‑size
            eta0 = 0.3
            for it in range(maxfun):
                f, g = _smooth_obj_and_grad(y, alpha, fft_len)
                eta = eta0 / math.sqrt(it + 1)
                y -= eta * g
                if time.time() >= deadline:
                    break

    # final probability vector
    p = _softmax(y)
    score = float(evaluate_sequence(p.tolist()))
    return p, score


# ----------------------------------------------------------------------
# Multiplicative mirror‑descent (same as reference – fast polishing)
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """Sub‑gradient of maxₖ (a∗a)[k] built from the chosen active shifts."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       fft_len: int,
                       eta0: float = 0.6,
                       delta_frac: float = 0.001,
                       top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation using a multi‑gradient built from
    the (up to) ``top_k`` shifts that are within ``delta_frac`` of the current
    convolution maximum.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # active set
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        g = _multi_gradient(a, active)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        a = np.maximum(a, 1e-15)

        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak reduction (final touch)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]

    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]          # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and returns the best
    sequence found (normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse start pool
    # ------------------------------------------------------------------
    starts = []

    # previous best (if the global variable exists)
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation to required length
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # handcrafted windows
    starts.append(_uniform_start(n))
    starts.append(_hann_start(n))
    starts.append(_hamming_start(n))
    starts.append(_blackman_start(n))
    starts.append(_gaussian_start(n))
    starts.append(_exponential_start(n, r=0.9995))
    starts.append(_exponential_start(n, r=0.985))

    # Dirichlet draws with various concentrations
    conc_params = [0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_params:
        for _ in range(3):
            starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # ------------------------------------------------------------------
    # Evaluate raw starts, keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement on a smoothed max (≈10 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.10
    alphas = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]
    refined_seq, refined_score = _lbfgs_refine(best_seq,
                                               alphas,
                                               fft_len,
                                               phase1_deadline,
                                               start_time)
    if refined_score < best_score:
        best_score = refined_score
        best_seq = refined_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after L‑BFGS score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Heavy mirror‑descent polishing (≈80 % of budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.90
    polished_seq, polished_score = _expgrad_multi_opt(best_seq,
                                                      phase2_deadline,
                                                      start_time,
                                                      fft_len,
                                                      eta0=0.6,
                                                      delta_frac=0.001,
                                                      top_k=5)
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polishing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Greedy final improvement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(best_seq,
                                                       deadline,
                                                       start_time,
                                                       max_iters=5000)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and final safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
