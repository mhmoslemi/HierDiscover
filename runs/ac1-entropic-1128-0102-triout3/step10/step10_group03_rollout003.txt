    M = _conv_self(a).max()
    # Build sparse matrix A

"""
Propose a non‑negative step function (length 4096) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a hybrid of:

1.   A lightweight parametric sweep (geometric, power, beta families) to get a
     decent starting point.
2.   Smoothed gradient descent on a soft‑max approximation of the max‑convolution.
3.   A short simulated‑annealing phase that can escape local minima.
4.   One or several linear‑programming refinements that maximise the ℓ₁‑norm
     under the current max‑convolution bound (the “LP improvement” of the
     reference).
5.   A final cheap hill‑climb polish.

All helpers are top‑level functions; the total running time is kept below the
1000 s budget by explicit time checks.
"""

import time
import math
import numpy as np

# Optional SciPy – needed only for the LP refinement
try:
    from scipy.sparse import diags
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:                     # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣ Parametric sweep (tiny but useful)
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, L_fft: int, deadline: float,
                      start_time: float, cur_best_a, cur_best_score):
    """Explores a few classical families; returns (best_a, best_score)."""
    best_a, best_score = cur_best_a, cur_best_score
    idx = np.arange(n)

    # geometric families – 100 eps values, 7 λs
    eps_grid = np.logspace(-4, -0.8, 100)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 7):
            seq = lam * (1.0 / n) + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score, best_a = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} λ={lam:.2f} → {sc:.6f}")

    # power families – 50 exponents
    p_grid = np.linspace(-0.5, 2.0, 50)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq /= pow_sum
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score, best_a = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} → {sc:.6f}")

    # beta‑type families – a few (p,q) pairs
    beta_vals = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            beta_sum = beta_seq.sum()
            if beta_sum == 0:
                continue
            beta_seq /= beta_sum
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score, best_a = sc, beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p} q={q} → {sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Smoothed gradient descent (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray, deadline: float,
                    start_time: float, L_fft: int, L_grad: int):
    """Gradient descent on a soft‑max approximation of max(conv)."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    tau = 0.10        # initial temperature
    tau_min = 1e-9
    tau_decay = 0.9995
    step0 = 0.30

    while time.time() < deadline:
        it += 1
        # convolution and soft‑max weights
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # sum(w)=1

        # gradient via FFT: grad_j = 2 * Σ_i a_i w_{i+j}
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # projected step
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} τ={tau:.2e} → {cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣ Simulated annealing (non‑monotone local search)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float,
                         start_time: float, L_fft: int):
    """Simple SA that moves a random fraction of mass between two coordinates."""
    n = a0.shape[0]
    a = a0.copy()
    cur_score = _eval_fast(a, L_fft)
    best_a, best_score = a.copy(), cur_score
    rng = np.random.default_rng()
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        it += 1
        # linear temperature schedule (slow cooling)
        remaining = deadline - time.time()
        frac = remaining / total_time
        T = max(1e-6, 0.05 * (frac ** 2))

        src = rng.choice(n, p=a)               # source proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-18:
            continue
        eps = a[src] * rng.random() * 0.25     # move up to 25 % of source
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # renormalise (the move preserves total mass, only round‑off errors)
        a_prop /= a_prop.sum()

        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score
        if delta < 0 or np.exp(-delta / T) > rng.random():
            a, cur_score = a_prop, new_score
            if new_score < best_score:
                best_score, best_a = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} best={best_score:.6f}")

        if it % 5000 == 0:
            print(f"[{time.time() - start_time:.2f}s] SA it={it:,} cur={cur_score:.6f} T={T:.2e}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣ Linear‑programming refinement (single‑step “LP improvement”)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, deadline: float,
               start_time: float, L_fft: int):
    """
    Given a feasible vector a (sum=1), solve

        maximize   sum(b)
        subject to conv(a, b) ≤ M,    b ≥ 0,

    where M = max(conv(a, a)).
    The optimizer b is rescaled back to sum 1 and returned.
    """
    if not _SCIPY_OK:                         # pragma: no cover
        print("SciPy not available – skipping LP refinement.")
        return a, _eval_fast(a, L_fft)

    n = a.shape[0]
    M = _conv_self(a, L=L_fft).max()
    # Build the (2n‑1)×n convolution matrix via its diagonals.
    # Row k, column j contains a_{k‑j} when 0 ≤ k‑j < n.
    diagonals = [np.full(n, a_i) for a_i in a]   # each diagonal = constant a_i
    offsets = list(range(n))
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csc')
    b_ub = np.full(2 * n - 1, M)

    # maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n)

    # Solve using HiGHS (fast for this size)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
        return a, _eval_fast(a, L_fft)

    b_opt = res.x
    s = b_opt.sum()
    if s <= 0:
        return a, _eval_fast(a, L_fft)

    # Scale back to sum 1 and blend a little with the old vector.
    b_scaled = b_opt / s
    t = 0.5                                   # convex mixing parameter
    a_new = (1 - t) * a + t * b_scaled
    a_new = _project_simplex(a_new)

    new_score = _eval_fast(a_new, L_fft)
    print(f"[{time.time() - start_time:.2f}s] LP refinement → {new_score:.6f}")
    return a_new, new_score


# ----------------------------------------------------------------------
# 5️⃣ Deterministic hill‑climb (tiny mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, deadline: float,
                start_time: float, L_fft: int):
    """
    Very cheap local search: move a tiny fraction of mass from a source index
    to a destination index, keep the move only if the true objective improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                # pick proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # renormalise (numerical drift)
        a_prop /= a_prop.sum()
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score, best_a = new_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function of length 4096 that minimises the evaluation
    function. The routine mixes a quick parametric sweep, smoothed gradient
    descent, simulated annealing, a linear‑programming refinement and a final
    deterministic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.2

    n = 4096
    L_fft = _next_pow2(2 * n - 1)    # for all convolutions
    L_grad = _next_pow2(3 * n)       # for soft‑max gradient (a bit larger)

    # ------------------------------------------------------------------
    # 0️⃣  Baseline (uniform) – may be overwritten later
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣′  Use the globally supplied sequence if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=float)
            if hs.ndim == 1 and hs.size > 0:
                hs = np.maximum(hs, 0.0)
                if hs.sum() > 0:
                    hs = hs / hs.sum()
                    sc = _eval_fast(hs, L_fft)
                    if sc < best_score:
                        best_score, best_seq = sc, hs.copy()
                        print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
        except Exception:                     # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (geometric / power / beta)
    # ------------------------------------------------------------------
    best_seq, best_score = _parametric_sweep(
        n, L_fft, deadline, start_time, best_seq, best_score)

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent with random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 150.0:
        # perturb the current best a little (or start from uniform)
        base = best_seq if np.random.rand() < 0.7 else uniform
        noise = 0.03 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 12.0)
        cand_seq, cand_score = _smooth_descent(start, sub_deadline,
                                               start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after smooth → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated annealing (non‑monotone local search)
    # ------------------------------------------------------------------
    if time.time() < deadline - 90.0:
        sa_deadline = deadline - 60.0
        a_sa, sc_sa = _simulated_annealing(best_seq, sa_deadline,
                                           start_time, L_fft)
        if sc_sa < best_score:
            best_score, best_seq = sc_sa, a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] SA improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Linear‑programming refinement (a few iterations)
    # ------------------------------------------------------------------
    while _SCIPY_OK and time.time() < deadline - 30.0:
        a_new, sc_new = _lp_refine(best_seq, deadline, start_time, L_fft)
        if sc_new < best_score - 1e-6:
            best_score, best_seq = sc_new, a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] LP step improved → {best_score:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 5️⃣  Final deterministic hill‑climb (tiny moves only when improving)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (numerical drift) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
