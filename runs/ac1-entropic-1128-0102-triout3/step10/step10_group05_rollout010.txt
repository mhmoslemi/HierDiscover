def _run_frank_wolfe_lp(a0, n, deadline, start_time, max_iters=100, t0=0.2):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    L = 2 * n - 1
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(L))
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP improve iter={it}, score={score:.6f}")
        # Build constraint matrix A
        # Use dense numpy
        A = np.empty((L, n), dtype=float)
        # Fill each column
        for i in range(n):
            A[i:i+n, i] = a
        # Solve LP: maximize sum(b) subject to A @ b <= M, b >= 0
        c = -np.ones(n)  # minimize -sum(b)
        b_ub = np.full(L, M, dtype=float)
        bounds = [(0, None)] * n
        # Use scipy
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fallback: break or continue
            print("LP failed, breaking")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b
        # step size
        t = min(t0, 2.0/(it+2))
        a = (1 - t) * a + t * g
        # ensure on simplex (just renormalise)
        a = _project_simplex(a)
    return best_a, best_score
```

Propose a step‑function with a low value of
    2·n·max(conv(a,a)) / (sum(a))**2 .
The method is a lightweight Frank‑Wolfe style iteration that
solves a small linear programme at each step:

    * start from a strictly positive exponential seed (or the
      supplied `height_sequence_1`);
    * compute M = max(a*a);
    * maximise Σ b_j subject to a * b ≤ M (element‑wise) and b_j ≥ 0;
      this linear programme pushes mass towards indices where a_j is small;
    * mix the current a with the normalised LP solution (with a decreasing
      step size) and repeat;
    * finish with a short greedy sweep (the same incremental updater
      used in the reference implementation) to polish the result.

The LP uses the full convolution constraints, but its size is modest
(≤ 2 048 variables, ≤ 4 095 constraints) and therefore each solve
takes only a few milliseconds.  Several restarts with tiny random
perturbations are performed; the best sequence found before the time
budget expires is returned.
"""

import time
import math
import numpy as np

# optional – SciPy is used for the linear programme
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a using FFT (length L ≥ 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution a_i ∝ r**i  (0 < r < 1).
    The vector is already normalised to sum 1.
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise (finite sum = (1-r**n))
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


# ----------------------------------------------------------------------
# Frank‑Wolfe iteration (LP based)
# ----------------------------------------------------------------------


def _run_fw_lp(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               max_iters: int = 80,
               t0: float = 0.25) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style optimisation.
    At every iteration we solve
        maximise Σ b_j   s.t.   a * b ≤ M   (pointwise),  b ≥ 0
    where M = max(a*a).  The LP solution is normalised back to a
    probability vector and mixed with the current a.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = 2 * n - 1
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # current convolution and objective
        conv = _conv_self(a, _next_pow2(L))
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] FW iter={it}, score={cur_score:.6f}")

        # build the full constraint matrix A (shape L × n)
        # column i contains a shifted copy of a: rows i … i+n‑1
        A = np.empty((L, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = a

        # linear programme: maximise Σ b  ⇔ minimise −Σ b
        c = -np.ones(n, dtype=float)
        b_ub = np.full(L, M, dtype=float)
        bounds = [(0.0, None)] * n

        if linprog is None:  # pragma: no cover
            raise RuntimeError("SciPy is required for the LP based phase.")

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            # if the LP fails we stop the FW loop
            print("  LP solve failed – terminating FW loop")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break
        g = b / sum_b  # normalise to a probability vector

        # line‑search / step‑size (decreasing Frank‑Wolfe steps)
        # start with a modest step and halve it if the max‑convolution grows
        t = min(t0, 2.0 / (it + 2.0))
        for _ in range(5):
            a_new = (1.0 - t) * a + t * g
            conv_new = _conv_self(a_new, _next_pow2(L))
            M_new = conv_new.max()
            if M_new <= M:
                a = a_new
                break
            t *= 0.5
        # ensure we stay on the simplex (project after mixing)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy reduction (same as the reference implementation)
# ----------------------------------------------------------------------


def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """Small‑scale greedy sweep that moves a tiny amount of mass from the
    most responsible index to a least responsible one (incremental conv updates)."""
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and a prospective destination
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution (linear approximation) -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a set of problem sizes (small enough for the LP phase)
    # ------------------------------------------------------------------
    candidate_ns = [1024, 2048]          # try both, keep the best found
    best_seq = None
    best_score = float("inf")

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise – use the supplied construction if available,
        # otherwise fall back to a short exponential scan.
        # ------------------------------------------------------------------
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = starter / starter.sum()
                a0 = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, starter.size),
                               starter)
                a0 = np.maximum(a0, 0.0)
                a0 = _project_simplex(a0)
            else:
                raise Exception()
        except Exception:
            # quick scan for a good decay factor
            best_r = 0.999
            best_sc = float("inf")
            for r in np.linspace(0.90, 0.9999, 250):
                cand = _exponential_sequence(n, r)
                conv = _conv_self(cand, _next_pow2(2 * n - 1))
                sc = 2.0 * n * conv.max()
                if sc < best_sc:
                    best_sc = sc
                    best_r = r
            a0 = _exponential_sequence(n, best_r)
            print(f"[{time.time() - start_time:.2f}s] n={n} exponential seed r={best_r:.6f}, score={best_sc:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Frank‑Wolfe LP iteration (≈15 % of remaining budget)
        # ------------------------------------------------------------------
        phase1_deadline = time.time() + min(180.0, 0.15 * (deadline - time.time()))
        try:
            a_fw, score_fw = _run_fw_lp(a0, n, phase1_deadline, start_time,
                                         max_iters=80, t0=0.25)
        except Exception as e:   # pragma: no cover
            print("LP phase failed:", e)
            a_fw, score_fw = a0, 2.0 * n * _conv_self(a0, _next_pow2(2 * n - 1)).max()

        if score_fw < best_score:
            best_score = score_fw
            best_seq = a_fw.copy()
            print(f"[{time.time() - start_time:.2f}s] n={n} FW best score={score_fw:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – short greedy polish (≈10 % of remaining budget)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            phase2_deadline = deadline
            a_greedy, score_greedy = _run_greedy_reduction(
                a_fw,
                n,
                phase2_deadline,
                start_time,
                best_score,
                best_seq,
                max_iters=500_000,
            )
            if score_greedy < best_score:
                best_score = score_greedy
                best_seq = a_greedy.copy()
                print(f"[{time.time() - start_time:.2f}s] n={n} greedy polish score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return as plain list
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of modest length
        best_seq = np.full(256, 1.0 / 256)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    final_obj = 2.0 * best_seq.size * _conv_self(best_seq,
                                                _next_pow2(2 * best_seq.size - 1)).max()
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
