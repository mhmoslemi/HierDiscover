"""
Propose a step‑function (non‑negative heights) with a very low
`evaluate_sequence` score.

Algorithm
---------
* Use a relatively large canvas (n = 16384) – the optimum 1.5 is
  approached as n grows.
* Warm‑start from a family of symmetric “tent” shapes
  a_i ∝ min(i,n‑1‑i)^p (p scanned on a fine grid).
* Refine the best warm‑start with two complementary continuous
  optimisers:
    – Projected sub‑gradient descent with momentum (fast, rough).
    – Projected Adam optimiser on a smoothed version of the
      max‑convolution (soft‑max with a gradually sharpening
      temperature β).  The gradient can be evaluated by two FFTs,
      keeping the per‑iteration cost low.
* If time permits, a light simulated‑annealing polish moves tiny
  packets of mass.
* The final vector is projected onto the probability simplex
  (sum = 1, a_i ≥ 0) and returned as a plain Python list.

All heavy work is performed with FFT‑based convolutions; the code
fits comfortably into the 1000 s budget on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real → real)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fast coarse refinement)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact (non‑smooth) objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max‑conv entry
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam optimisation on a smoothed max (soft‑max) objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective
        Lβ(a) = 2·n·Mβ(conv)/ (∑a)²,
    where Mβ is the log‑sum‑exp (soft‑max) of the convolution and β
    grows slowly to sharpen the approximation of the true max.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # base learning rate

    it = 0
    beta_smooth = 5.0             # initial temperature for the soft‑max

    while time.time() < deadline:
        it += 1

        # ------- convolution -------
        L_self = _next_pow2(2 * n - 1)
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]
        max_c = conv.max()

        # ------- soft‑max over the convolution -------
        beta = beta_smooth * min(1.0 + it / 8000.0, 200.0)   # slowly increase β
        # stabilise exponentials
        exp_shift = beta * (conv - max_c)
        exps = np.exp(exp_shift)
        Z = exps.sum()
        w = exps / Z                     # soft‑max weights, sum to 1
        M = max_c + np.log(Z) / beta     # smooth approximation of max(conv)

        # ------- gradient of the smooth part (2·a‑weighted shift) -------
        w_rev = w[::-1]                 # reverse for convolution
        L_full = _next_pow2(3 * n - 2)
        A_full = np.fft.rfft(a, n=L_full)
        W_full = np.fft.rfft(w_rev, n=L_full)
        C_full = np.fft.irfft(A_full * W_full, n=L_full)[: (3 * n - 2)]
        idx = (2 * n - 2) - np.arange(n)    # extract the needed slice
        g = 2.0 * C_full[idx]               # gradient of Σ w_k·conv_k

        # ------- full gradient of the original objective -------
        s = a.sum()
        grad = (2.0 * n / (s * s)) * g - (4.0 * n * M / (s ** 3)) * np.ones_like(a)

        # ------- Adam update -------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ------- keep feasible (simplex) -------
        a = _project_simplex(a)

        # ------- keep the best exact score seen -------
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # occasionally boost the temperature a bit
        if it % 5000 == 0:
            beta_smooth = min(beta_smooth * 1.3, 250.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap annealing that moves tiny mass packets."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # simple exponential cooling
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point for the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger canvas gives a finer approximation.
    # --------------------------------------------------------------
    n = 16384

    # --------------------------------------------------------------
    # 1) Warm‑start: power‑tent family (a_i ∝ min(i,n‑1‑i)^p)
    # --------------------------------------------------------------
    x = np.arange(n, dtype=float) / (n - 1)          # points in [0,1]
    exponents = np.linspace(0.6, 1.3, 15)           # fine sweep
    best_seq = None
    best_score = float("inf")

    for p in exponents:
        w = np.minimum(x, 1.0 - x) ** p
        if w.sum() == 0.0:
            continue
        a = w / w.sum()                              # simplex normalisation
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] exponent={p:.3f} "
              f"score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # --------------------------------------------------------------
    # 2) A few random Dirichlet starts – cheap diversification
    # --------------------------------------------------------------
    for _ in range(3):
        if time.time() >= deadline:
            break
        a = np.random.random(n)
        a = a / a.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Random start score={score:.6f}")

    # --------------------------------------------------------------
    # 3) If a global starter is available, try to reuse it (resample)
    # --------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # simple nearest‑neighbour up‑sampling to length n
            idx = np.linspace(0, starter.size - 1, n).astype(int)
            starter = starter[idx]
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4) Local refinement – sub‑gradient descent (fast, many iters)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a few seconds for later phases
        sub_deadline = time.time() + remaining * 0.55
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Adam optimiser on the smoothed objective (complementary)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        adam_deadline = time.time() + (deadline - time.time()) * 0.40
        adam_seq, adam_score = _adam_optimize(best_seq, adam_deadline, start_time)
        if adam_score < best_score:
            best_score = adam_score
            best_seq = adam_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improved score={adam_score:.6f}")

    # --------------------------------------------------------------
    # 6) Very light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
