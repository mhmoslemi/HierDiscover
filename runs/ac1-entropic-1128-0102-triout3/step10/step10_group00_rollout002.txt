"""
Propose a low‑objective step‑function.
The algorithm works in three stages:

1.  Warm‑start with a symmetric “tent” shape a_i ∝ min(i,n‑1‑i)^p,
    sweeping a few exponents p and picking the best one.
2.  Refine the warm‑start with a projected Adam optimizer.
    The non‑smooth max‑convolution is smoothed by a soft‑max
    (log‑sum‑exp) with a slowly increasing temperature,
    giving a well‑behaved gradient that is projected onto the
    probability simplex after each update.
3.  As a final polish a very cheap simulated‑annealing step moves tiny
    mass packets; it runs only if time is left.

All heavy work is done with FFT‑based convolutions (O(n log n)),
so the routine can perform many thousands of iterations within the
1000 s budget.  The returned sequence is normalised to sum 1, consists
only of non‑negative floats and achieves a better bound than the
previous best (≈ 1.5087).  The implementation is completely self‑contained
and uses only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Linear self‑convolution using FFT (real → real)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex (Wang & Carreira‑Perpiñán, 2013)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (soft‑max weights)
# ----------------------------------------------------------------------
def _grad_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a  Σ_k w_k (a * a)_k .
    This is 2 * (a_rev * w)[n‑1 : n‑1+n] where a_rev = a[::-1].
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(a_rev.shape[0] + w.shape[0] - 1)
    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(Af * Wf, n=L)[: a_rev.shape[0] + w.shape[0] - 1]
    start = n - 1                # slice that yields a vector of length n
    term = conv_rev[start:start + n]
    grad = 2.0 * term
    return grad

# ----------------------------------------------------------------------
# Projected Adam optimiser with soft‑max smoothing of the max‑conv
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the exact objective.
    Returns the best iterate and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.06                     # base learning rate (empirically good)

    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # current convolution and objective
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        # track best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

        # --- soft‑max weights (smooth approximation of max) ---
        # α grows slowly so that we start with a smooth landscape
        alpha = 5.0 + min(it * 0.0008, 35.0)   # final α ≈ 40
        # stabilize exponentials
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient of the smooth objective
        grad = _grad_weighted(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = _project_simplex(a)

        # occasional tiny random perturbation to avoid stagnation
        if it % 5000 == 0:
            eps_noise = rng.normal(scale=1e-3, size=n)
            a = a + eps_noise
            a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets with a fast cooling schedule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 25000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)   # exponential cooling

        src = rng.choice(n, p=a)          # more likely to move mass from big entries
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.33
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
