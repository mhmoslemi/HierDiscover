"""
A fast, deterministic search for a low‑objective step‑function.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) so that
the evaluation function becomes  score = 2·n·max(conv(a,a)).
The main ideas are

1. **Large canvas** – we use n = 16384.  The theoretical optimum of 1.5
   is approached as 1/n, so doubling n already cuts the gap by about a
   factor two.

2. **Rich warm‑start** – a small sweep over several families
   (tent‑shapes with many exponents, discrete Gaussians, the previous
   global starter) is performed and the best candidate is kept.

3. **Projected sub‑gradient descent** – we minimise the non‑smooth
   objective  max(conv) on the simplex.  The sub‑gradient is given by the
   convolution entry that attains the maximum; we use a momentum term
   and a step size η_t = η₀ / √t.  After each update the iterate is
   projected back onto the simplex by an O(n log n) Euclidean projection.

4. **Light annealing polish** – a cheap simulated‑annealing phase moves
   tiny packets of mass between coordinates; accepted moves are those
   that improve the objective (or are accepted with the Metropolis rule).

The whole search respects the 1000 s time budget; the routine returns
the best sequence found (as a plain Python list) which will be passed
to ``evaluate_sequence`` by the harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real → real)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, works on the simplex)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (∑seq)².
    The denominator is 1 because the caller always works with a simplex
    vector; we keep the formula for safety.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpinán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on the objective.
    Returns the best iterate seen before ``deadline`` together with its
    objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # momentum vector
    momentum = np.zeros_like(a)
    beta = 0.9               # momentum coefficient
    step0 = 0.8              # initial step size
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index where the max is attained (break ties arbitrarily)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a (see derivation in the prompt)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer annealing.  Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # fixed budget of elementary moves
    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        # choose a source index proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 30 %) of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
