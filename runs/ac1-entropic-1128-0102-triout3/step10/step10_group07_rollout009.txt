def _soft_expgrad_optimize(a0, deadline, start_time, fft_len, eta0=0.8, alpha0=5.0, alpha_growth=1.0):
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] SoftExpGrad improve: iter={t}, score={cur_score:.6f}")
        # compute softmax weights
        max_conv = conv.max()
        # scale: using alpha * (conv - max_conv) to avoid overflow
        exp_arg = alpha0 * (conv - max_conv)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to argmax weight
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum
        # gradient: g_i = 2 * sum_{k} w_k * a_{k-i}
        # compute convolution of w with reversed a:
        a_rev = a[::-1]
        # compute conv2 = np.convolve(w, a_rev) using FFT length appropriate
        # The length needed: len(w) + len(a_rev) - 1 = (2*n-1)+(n)-1 = 3*n - 2
        # We'll compute using np.fft.rfft of appropriate length (next_pow2)
        total_len = len(w) + len(a_rev) - 1
        fft_len_grad = _next_pow2(total_len)
        Wf = np.fft.rfft(w, n=fft_len_grad)
        Af_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(Wf * Af_rev, n=fft_len_grad)
        # Slice to get indices n-1 .. n-1 + n-1 = 2n-2? Actually slice length n:
        grad = 2.0 * conv2[n-1 : n-1 + n]  # length n
        # update
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        # project onto simplex
        a = _project_simplex(a)
        # increase alpha maybe
        alpha0 = alpha0 * alpha_growth
    return best_a, best_score

"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

Algorithm
---------
1. Build a diversified pool of seeds (uniform, Gaussian, Dirichlet,
   sparse random, optionally the historic ``height_sequence_1``) and keep
   the best one according to the true objective.
2. Main optimisation – *soft* exponentiated‑gradient (mirror descent) on a
   smooth approximation of the max‑convolution.  The gradient is obtained
   by a soft‑max weighting of all convolution shifts (implemented via
   FFT‐based convolutions) and the iterate is projected onto the
   probability simplex.
3. Occasionally restart from fresh random seeds and run a short
   soft‑gradient to escape possible basins.
4. Deterministic greedy “peak‑reduction’’ (move mass from the coefficient
   most responsible for the current autocorrelation peak towards the
   smallest coefficient).
5. Final short stochastic hill‑climbing (random mass transfers) to squeeze
   out the last improvement.
6. Clip, renormalise to sum = 1 and return.

The routine respects the 1000 s wall‑clock limit, prints progress and
keeps the best sequence ever seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else np.full(n, 1.0 / n)


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Soft exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def _soft_expgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eta0: float = 0.6,
    alpha0: float = 5.0,
    alpha_growth: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation
        f(a) = (1/α) log Σ_k exp(α·(a*a)[k])
    Returns the best vector ever seen and its exact (unsmoothed) score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()                     # stay on the simplex
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    alpha = alpha0

    while time.time() < deadline:
        t += 1
        # --- convolution ---
        conv = _conv_self(a, fft_len)        # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max        # exact score (sum(a)=1)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Soft‑Grad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # --- soft‑max weights over all shifts ---
        max_conv = cur_max
        # stabilise exponentials
        exp_arg = alpha * (conv - max_conv)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback to pure arg‑max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # --- gradient of the smoothed objective ---
        # gradient_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        total_len = len(w) + len(a_rev) - 1          # = 3n‑2
        fft_len_grad = _next_pow2(total_len)
        Wf = np.fft.rfft(w, n=fft_len_grad)
        Af = np.fft.rfft(a_rev, n=fft_len_grad)
        conv2 = np.fft.irfft(Wf * Af, n=fft_len_grad)
        # slice indices n‑1 … n‑1+n‑1  => length n
        grad = 2.0 * conv2[n - 1 : n - 1 + n]

        # --- mirrored (multiplicative) update ---
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        # --- projection back onto simplex ---
        a = _project_simplex(a)

        # slowly increase α to focus on the true max
        alpha = alpha * alpha_growth

        # occasional exact check (cheap guard)
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] Exact check improvement: score={sc:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 8000,
) -> tuple[np.ndarray, float]:
    """
    Move a modest fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # pick destination – the smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # modest transfer
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] Greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine builds a diverse initial pool, then spends most of the time
    polishing the current best using a soft‑gradient method, interleaving
    random restarts, a final greedy sweep and a short stochastic hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # problem size (larger n gives finer granularity, but keep it manageable)
    n = 8000
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse set of initial candidates
    # ------------------------------------------------------------------
    starts = []

    # 1) historic construction, if present
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) classic handcrafted starters
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_gaussian_start(n))                  # bell‑shaped
    starts.append(_exponential_start(n, r=0.998))     # gentle decay
    starts.append(_exponential_start(n, r=0.985))     # stronger decay

    # 3) Dirichlet draws with several concentration parameters
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        starts.append(_dirichlet_start(n, conc, rng))

    # 4) sparse random supports – a few different densities
    for m in [800, 1600, 2400, 4000]:
        for _ in range(2):
            starts.append(_sparse_random_start(n, m, rng))

    # 5) pure exponential draw (continuous)
    exp_vec = rng.exponential(scale=1.0, size=n)
    starts.append(exp_vec / exp_vec.sum())

    # ------------------------------------------------------------------
    # Evaluate the raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(starts):
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] init #{idx+1:2d} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – soft exponentiated‑gradient polishing (≈70 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * (deadline - start_time)
    best_seq, best_score = _soft_expgrad_optimize(
        best_seq, phase1_deadline, start_time, fft_len,
        eta0=0.8, alpha0=7.0, alpha_growth=1.0
    )
    print(
        f"[{time.time() - start_time:6.2f}s] after soft‑grad best_score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase 2 – short random‑restart bursts (≈20 % of remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # draw a fresh start
        if rng.random() < 0.5:
            m = int(rng.integers(500, 3000))
            cand_start = _sparse_random_start(n, m, rng)
        else:
            conc = 10 ** rng.uniform(-1.5, 1.0)
            cand_start = _dirichlet_start(n, conc, rng)

        sub_deadline = time.time() + 5.0
        cand_seq, cand_score = _soft_expgrad_optimize(
            cand_start, sub_deadline, start_time, fft_len,
            eta0=0.5, alpha0=5.0, alpha_growth=1.0
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL from restart score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=8000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 4 – short stochastic hill‑climbing (use remaining seconds)
    # ------------------------------------------------------------------
    while time.time() < deadline - 0.1:
        i = rng.integers(0, n)
        if best_seq[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        eps = best_seq[i] * rng.uniform(0.05, 0.2)
        a_prop = best_seq.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        sc = evaluate_sequence(a_prop.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_prop
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    return [float(x) for x in best_seq.tolist()]
