"""
Propose a low‑objective step sequence.

The algorithm combines three ideas:

1. **Parametric sweep** – A fine grid over the Beta‑family
   a_i ∝ (x_i·(1−x_i)+ε)^{−α} (α≈½, ε tiny) is evaluated with a fast FFT‑based
   self‑convolution.  The best candidate is kept.

2. **Smoothed sub‑gradient descent (Adam)** –  The non‑smooth objective
   2·n·max(conv)/sum² is approximated by a log‑sum‑exp smoothing.
   The gradient of the smooth surrogate can be expressed with a single
   extra convolution, so each iteration needs two FFTs.
   An Adam optimiser with a linearly increasing smoothing parameter (β)
   drives the sequence towards the (conjectured) optimal shape.
   After every few iterations the exact objective is evaluated and the
   best iterate is saved.

3. **Light annealing polish** –  A few thousand random mass‑transfer moves
   (simulated‑annealing style) are performed if time permits.

All steps respect the 1000 s wall‑time limit, keep the vector on the
probability simplex (non‑negative, sum = 1), and finally return a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w≥0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    Computed with an FFT based linear convolution.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) with a very low
    evaluation score under the given time budget.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 0.5

    # size of the problem – a power of two works well with the FFTs
    n = 4096

    # ------------------------------------------------------------------
    # 1) Fine sweep over Beta‑family parameters (α, ε)
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n           # mid‑bin positions
    best_seq = None
    best_score = float("inf")

    alphas = np.linspace(0.45, 0.55, 41)                    # step ≈0.0025
    epsilons = [0.0, 1e-8, 1e-6, 1e-4]                     # edge‑smoothing

    for alpha in alphas:
        for eps in epsilons:
            w = (x * (1.0 - x) + eps) ** (-alpha)
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        # early‑exit if we run out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Optional global starter (if the harness provided one)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():

                if starter.sum() > 0:

                    cur_score = _objective(starter)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = starter.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Smoothed Adam optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if best_seq is None:
        # safe fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)

    a = best_seq.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.02

    # FFT length that can accommodate both convolutions (self‑conv and
    # weight‑conv).  Needs to be ≥ 3·n‑2.
    L = _next_pow2(3 * n)

    iteration = 0
    # smoothing schedule – from a gentle β to a sharp one (≈200)
    beta_start, beta_end = 5.0, 200.0

    while time.time() < deadline:
        iteration += 1

        # ---- self‑convolution (linear, via FFT) --------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]                     # length 2n‑1

        # ---- smoothing parameter (increases with elapsed time) ----
        elapsed = time.time() - start_time
        prop = min(1.0, elapsed / (budget_s - 0.5))
        beta = beta_start + (beta_end - beta_start) * prop

        # ---- soft‑max weights for the log‑sum‑exp surrogate ------
        beta_conv = beta * conv
        max_beta = beta_conv.max()
        exp_shift = np.exp(beta_conv - max_beta)
        sum_exp = exp_shift.sum()
        w = exp_shift / sum_exp               # shape (2n‑1,)

        # B·a = correlation of w with a (cross‑correlation)

        gradient = 4.0 * n * Ba               # ∇F_s = 4·n·B·a

        # ---- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * gradient
        v = beta2 * v + (1.0 - beta2) * (gradient ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        # keep a on the simplex (non‑negative, sum = 1)
        a = _project_simplex(a)

        # ---- keep the best exact objective seen so far ------------
        if iteration % 50 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{elapsed:.2f}s] iter={iteration:,}  best={cur_score:.6f}")

        # leave a couple of seconds for the final polish
        if time.time() >= deadline - 2.0:
            break

    # ------------------------------------------------------------------
    # 4) Very light simulated‑annealing polish (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_moves = 30_000

        for t in range(1, total_moves + 1):
            if time.time() >= deadline:
                break
            # source chosen proportionally to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # move a modest fraction (≤10 %) of the source mass
            eps = a[src] * rng.random() * 0.10
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={t:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"Returning sequence of length {n} with final objective {best_score:.6f}")
    return [float(v) for v in best_seq]
