"""
Propose:  gradient descent on a smooth approximation of the max‑convolution.
The algorithm

1.  builds a fine family of symmetric “Beta‑type’’ step functions
    a_i ∝ (x_i·(1−x_i))^{−α} (the arcsine family) and picks the best α.
2.  starting from that vector it minimises a smoothed version of the
    objective  F(a)=2·n·max(a∗a)/‖a‖_1²
    by standard Adam updates on the simplex.
    The smoothing uses a soft‑max (log‑sum‑exp) with an annealing
    inverse temperature β that grows from 5 to ≈150.
3.  the best iterate (according to the exact objective) is kept.
4.  if a few seconds remain, a very light simulated‑annealing polish
    (tiny mass transfers) is performed.
The whole search respects the 1000 s wall‑time limit and never produces
negative or >1000 values – the final simplex projection guarantees a
valid step‑function for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)−1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _cross_corr(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Cross‑correlation g_i = Σ_j p_{i+j}·a_j for i=0..len(a)-1."""
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(p) + len(a_rev) - 1)
    P = np.fft.rfft(p, n=L)
    A = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(P * A, n=L)  # length L
    start = n - 1                     # position of g_0 in the full convolution
    return conv[start : start + n]   # size n


# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact evaluation: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Smoothed max and its gradient ----------------------------------------
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_max, grad) where
    smooth_max ≈ max(a∗a) (log‑sum‑exp with temperature 1/β)
    grad = ∂ smooth_max / ∂ a   (shape: len(a))
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    # log‑sum‑exp (stable)
    m = conv.max()
    exp_terms = np.exp(beta * (conv - m))
    sum_exp = exp_terms.sum()
    smooth_max = (np.log(sum_exp) + beta * m) / beta

    # soft‑max weights
    p = exp_terms / sum_exp                  # sum(p)=1, length 2n‑1

    # ∂conv/∂a_i = 2·a_{j-i} (zero outside)
    # grad_i = Σ_j p_j * ∂conv_j/∂a_i = 2·Σ_j p_j·a_{j-i}
    grad = 2.0 * _cross_corr(p, a)           # length n
    return smooth_max, grad


# ----------------------------------------------------------------------
# Adam optimiser --------------------------------------------------------
# ----------------------------------------------------------------------
def _adam_step(a: np.ndarray,
               grad: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Perform one Adam update (without projection)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    return a_new, m, v


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """

    start_time = time.time()
    # keep a tiny margin for the harness
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------

    # --------------------------------------------------------------

        w = (x * (1.0 - x) + 1e-30) ** (-alpha)

        # tiny time‑check

        # actually we reuse the previously found best alpha

                                  min(0.99, alpha0 + 0.015), 31)

            score = _objective(a)
            if score < best_score:
                best_score = score

            if time.time() >= deadline:

    # --------------------------------------------------------------

    base_lr = 0.01

    beta_start = 5.0
    beta_end = 150.0

    while time.time() < deadline - 5.0:     # leave a few seconds for polishing

        cur_score = _objective(a)

            best_seq = a.copy()

    if time.time() < deadline:
