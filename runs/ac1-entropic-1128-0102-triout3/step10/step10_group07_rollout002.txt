def gradient_softmax(a, beta, N_self, N_grad):
    conv = conv_self_fft(a, N_self)
    maxc = conv.max()
    # compute weights
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # produce gradient
    a_rev = a[::-1]
    cross = conv_full_fft(w, a_rev, N_grad)
    start = len(a) - 1
    grad = 2.0 * cross[start:start+len(a)]

def _conv_self_fft(a: np.ndarray, N: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT of size N."""
    Af = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Af * Af, n=N)
    return conv[: 2 * a.shape[0] - 1]


def _conv_full_fft(x: np.ndarray, y: np.ndarray, N: int) -> np.ndarray:
    """Full linear convolution x*y via FFT of size N."""
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _gradient_softmax(a: np.ndarray,
                      beta: float,
                      N_self: int,
                      N_grad: int) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of   f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
    and return (grad, current_max_of_a*a).
    """
    # a*a
    conv = _conv_self_fft(a, N_self)
    cur_max = conv.max()

    # softmax weights (already shifted by cur_max for stability)
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    if sum_w == 0.0 or not np.isfinite(sum_w):
        # fallback – uniform weights (should not happen)
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= sum_w

    # gradient = 2 * (w * a_rev) convolution
    a_rev = a[::-1]
    cross = _conv_full_fft(w, a_rev, N_grad)
    start = a.shape[0] - 1
    grad = 2.0 * cross[start:start + a.shape[0]]
    return grad, cur_max


# ----------------------------------------------------------------------
# Deterministic greedy peak reduction (same idea as the previous code)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Repeatedly shift a modest fraction of the coefficient that most
    contributes to the current convolution peak towards the smallest
    coefficient.  Accept only strictly decreasing max‑convolution values.
    """
    a = a0.copy()
    n = a.shape[0]
    N_self = _next_pow2(2 * n - 1)

    best_score = 2.0 * n * _conv_self_fft(a, N_self).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self_fft(a, N_self)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = smallest coefficient that is not the source
        candidates = np.delete(np.arange(n), src)
        dst = candidates[a[candidates].argmin()]

        eps = 0.20 * a[src]          # move 20 % of the source mass
        if eps <= 1e-15:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self_fft(a_prop, N_self).max()
        if new_max < cur_max - 1e-15:   # strict improvement
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    DEFAULT_N = 4800
    n = DEFAULT_N

    # ------------------------------------------------------------------
    # Initialise from the previous best construction if possible
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=float).ravel()
        if seq.size != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, seq.size)
            new_x = np.linspace(0.0, 1.0, n)
            seq = np.interp(new_x, old_x, seq)
        a = np.maximum(seq, 0.0)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
    else:
        # fallback: uniform distribution
        a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths
    # ------------------------------------------------------------------
    N_self = _next_pow2(2 * n - 1)      # for a*a
    N_grad = _next_pow2(3 * n - 2)      # for weight * a_rev

    # ------------------------------------------------------------------
    # Track the best solution found so far
    # ------------------------------------------------------------------
    cur_max = _conv_self_fft(a, N_self).max()
    best_score = 2.0 * n * cur_max
    best_a = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed ℓ∞ descent (≈ 80 % of the budget)
    # ------------------------------------------------------------------
    beta0 = 5.0
    beta_max = 5000.0
    eta0 = 0.8
    t = 0

    while time.time() < deadline - 12.0:   # keep a little time for later phases
        t += 1
        beta = min(beta_max, beta0 * math.sqrt(t))
        grad, cur_max = _gradient_softmax(a, beta, N_self, N_grad)

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] new best (grad) score={cur_score:.6f} iter={t}")

        # multiplicative (exponentiated) update on the simplex
        eta = eta0 / math.sqrt(t)
        grad_clip = np.clip(grad, -1e6, 1e6)   # protect from overflow
        a = a * np.exp(-eta * grad_clip)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    # ------------------------------------------------------------------
    # Phase 2 – cheap random‑restart bursts (≈ 10 % of remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # random sparse start
        m = int(rng.integers(200, 1500))
        cand = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=m, replace=False)
        cand[idx] = 1.0 / m

        # short smoothed‑gradient run (≈ 3 s)
        sub_deadline = time.time() + 3.0
        a_local = cand.copy()
        t_local = 0
        while time.time() < sub_deadline:
            t_local += 1
            # use a modest constant β for speed
            grad, cur_max = _gradient_softmax(a_local, beta=30.0,
                                             N_self=N_self, N_grad=N_grad)
            eta = 0.5 / math.sqrt(t_local)
            a_local = a_local * np.exp(-eta * np.clip(grad, -1e6, 1e6))
            a_local = a_local / a_local.sum()

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_local.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] new best (restart) score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak reduction (final polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_a, final_score = _greedy_local_improve(best_a, deadline,
                                                      start_time,
                                                      max_iters=5000)
        if final_score < best_score:
            best_score = final_score
            best_a = final_a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] new best after greedy score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety round‑trip
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0.0 and np.isfinite(total):
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_reported = evaluate_sequence(best_a.tolist())
    elapsed = time.time() - start_time
    print(f"[{elapsed:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(v) for v in best_a.tolist()]
