def generate_beta_seq(n, alpha):
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    # ensure no zeros or ones
    eps = 1e-12
    x = np.clip(x, eps, 1-eps)
    a = (x**(alpha-1)) * ((1-x)**(alpha-1))
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    else:
        a = np.full(n, 1.0/n)
    return a
```

    deadline = start + budget_s - 1.0

            best_seq = cand

"""
Multi‑start optimiser for the self‑convolution bound.

The routine builds a set of candidate step‑functions from a few simple
parameterised families (geometric decay, symmetric U‑shaped Beta‑like
densities, double‑ended exponentials and random Dirichlet vectors).
The best candidate is then refined by three fast stages:

1. Adam optimisation on a smooth (log‑sum‑exp) surrogate of the
   max‑convolution.
2. Exponentiated‑gradient (mirror) descent on the true max‑convolution.
3. Deterministic greedy mass transfer that continuously reduces the
   peak of the self‑convolution.

All iterates are kept on the probability simplex (non‑negative,
unit‑sum) – the objective is homogeneous, therefore this does not
affect optimality.  The algorithm respects a 1000 s time budget,
prints progress messages and finally returns the best sequence found
as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT of length L (L must be a power of two).
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Families of initial sequences
# ----------------------------------------------------------------------


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution a_i ∝ r^i on n points; 0<r<1.
    """
    if not (0.0 < r < 1.0):
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # geometric series 1+r+…+r^{n‑1} = (1‑r^n)/(1‑r)
    norm = (1.0 - r ** n) / (1.0 - r)
    a = a / norm
    return a


def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric U‑shaped density based on a Beta(alpha,alpha) law.
    For alpha<1 the distribution is U‑shaped, for alpha=1 it is uniform.
    The vector is normalised to sum 1.
    """
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    # log‑space to avoid overflow for alpha<1
    log_a = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    log_a -= np.max(log_a)               # shift to avoid huge exp
    a = np.exp(log_a)
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def _double_exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Symmetric two‑ended exponential: a_i ∝ r^i + r^{n‑1‑i}.
    """
    if not (0.0 < r < 1.0):
        r = 0.999
    i = np.arange(n, dtype=float)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Objective (same as evaluate_sequence for unit‑sum vectors)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, L: int) -> float:
    """
    Returns the evaluation value for a simplex‑normalised vector a.
    For sum(a)=1 this equals 2·n·max(conv(a,a)).
    """
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _conv_self(a, L)
    mx = conv.max()
    n = a.shape[0]
    return 2.0 * n * mx / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------


def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 20000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve "
                      f"iter={it:,} score={cur_score:.6f}")

            # ----- smooth max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp‑grad improve "
                  f"iter={it:,} score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative (mirror) update
        eta = eta0 / math.sqrt(it)
        # guard against overflow/underflow
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum <= 0.0 or np.isnan(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------


def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L_full = 2 * n - 1
    L_fft = _next_pow2(L_full)
    conv = _conv_self(a, L_fft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices which actually contribute to the current maximum
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build the reflected window a_k
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose destination with minimal reflected value (but not the source)
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partners for src and dst in the currently maximal convolution term
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            # theoretical epsilon to bring max down to second_max
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental convolution update -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # periodic exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, L_fft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Greedy improve "
                  f"iter={it:,} score={cur_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.

    The routine follows the strategy described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin so that we always finish before the hard limit
    deadline = start_time + budget_s - 1.0

    final_n = 4096
    L_fft = _next_pow2(2 * final_n - 1)

    # ------------------------------------------------------------------
    # 1. Scan a few parametric families for a good starting point.
    # ------------------------------------------------------------------
    best_seq = None
    best_obj = float("inf")

    # exponential family
    for r in np.linspace(0.90, 0.9999, 150):
        cand = _exponential_sequence(final_n, r)
        obj = _objective(cand, L_fft)
        if obj < best_obj:
            best_obj = obj
            best_seq = cand

    # symmetric Beta‑like family
    for alpha in np.linspace(0.1, 2.0, 190):
        cand = _beta_sequence(final_n, alpha)
        obj = _objective(cand, L_fft)
        if obj < best_obj:
            best_obj = obj
            best_seq = cand

    # double‑ended exponential
    for r in np.linspace(0.90, 0.9999, 120):
        cand = _double_exponential_sequence(final_n, r)
        obj = _objective(cand, L_fft)
        if obj < best_obj:
            best_obj = obj
            best_seq = cand

    # random Dirichlet seeds
    for _ in range(10):
        cand = np.random.dirichlet(np.ones(final_n))
        obj = _objective(cand, L_fft)
        if obj < best_obj:
            best_obj = obj
            best_seq = cand

    # reuse a previously discovered construction if it exists
    try:
        prior = np.asarray(height_sequence_1, dtype=float)
        if prior.ndim == 1 and prior.size > 0:
            # interpolate to the required length
            prior = np.interp(
                np.linspace(0, 1, final_n),
                np.linspace(0, 1, prior.size),
                prior,
            )
            prior = np.maximum(prior, 0.0)
            prior_sum = prior.sum()
            if prior_sum > 0:
                prior = prior / prior_sum
                obj = _objective(prior, L_fft)
                if obj < best_obj:
                    best_obj = obj
                    best_seq = prior
                    print(f"[0.00s] Prior sequence taken as start, obj={obj:.6f}")
    except Exception:
        pass

    # safety: ensure we have a valid simplex vector
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    print(f"[0.00s] Initial best objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam smoothing (short, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = min(deadline,
                             time.time() + max(20.0, 0.15 * (deadline - time.time())))
        a_adam, obj_adam = _run_adam_smoothing(
            best_seq,
            final_n,
            phase_deadline,
            start_time,
            alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0],
            lr0=0.6,
            iters_per_alpha=25000,
        )
        if obj_adam < best_obj:
            best_obj = obj_adam
            best_seq = a_adam
            print(f"[{time.time() - start_time:.2f}s] Adam improved to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Exponentiated‑gradient descent on the true max.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = min(deadline,
                             time.time() + max(30.0, 0.25 * (deadline - time.time())))
        a_exp, obj_exp = _run_exp_gradient(
            best_seq,
            final_n,
            phase_deadline,
            start_time,
            eta0=0.9,
        )
        if obj_exp < best_obj:
            best_obj = obj_exp
            best_seq = a_exp
            print(f"[{time.time() - start_time:.2f}s] Exp‑grad improved to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Aggressive deterministic greedy reduction.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate the remaining time to greedy reduction
        a_greedy, obj_greedy = _run_greedy_reduction(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_obj,
            best_seq,
            max_iters=2_500_000,
        )
        if obj_greedy < best_obj:
            best_obj = obj_greedy
            best_seq = a_greedy
            print(f"[{time.time() - start_time:.2f}s] Greedy improved to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing with a short Adam pass (high smoothing).
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        phase_deadline = min(deadline, time.time() + 5.0)
        a_final, obj_final = _run_adam_smoothing(
            best_seq,
            final_n,
            phase_deadline,
            start_time,
            alphas=[10.0, 20.0, 40.0],
            lr0=0.4,
            iters_per_alpha=5000,
        )
        if obj_final < best_obj:
            best_obj = obj_final
            best_seq = a_final
            print(f"[{time.time() - start_time:.2f}s] Final Adam polish improved to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return the result.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    final_obj = _objective(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Final objective = {final_obj:.6f}")

    # convert to plain Python list
    return [float(x) for x in best_seq]
