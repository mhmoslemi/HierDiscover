"""
Gradient‑descent on a smooth approximation of the max‑convolution
objective.  The method uses a soft‑max (temperature α) to turn the
non‑smooth max into a smooth convex function, then applies Adam with
projection onto the probability simplex.  A warm‑start is obtained from
a small family of symmetric “tent” shapes or from the previously found
global construction (height_sequence_1).  After the main optimisation
a light simulated‑annealing polish is performed.  The whole search is
confined to the allotted budget (≈ 1000 s) and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic linear‑algebra helpers
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {w ≥ 0, Σ w = 1}.
    O(n log n) implementation (Wang & Carreira‑Perpinán, 2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the evaluation harness:
          2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The function is scale‑invariant, therefore the sequence can be
    assumed to be normalised to sum 1.
    """
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ------------------------------------------------------------
# Annealing polish (very cheap; runs only if time remains)
# ------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer annealing, keeping the best exact score seen."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 40_000               # a bit more than the previous run
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = 0.02 * (1e-4 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)       # more mass → more likely to be source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 35 % of src mass
        eps = a[src] * rng.random() * 0.35
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={new_score:.6f}")

    return best_a, best_score

        conv_self = np.fft.irfft(Fa * Fa, n=L_big)[: 2 * n - 1]

        # ----- soft‑max weights for the smoothed max -----
        max_c = conv_self.max()
        shifted = conv_self - max_c            # numerical stabilisation
        exps = np.exp(alpha * shifted)
        p = exps / exps.sum()

        # ----- FFTs of the weight vectors -----
        Fp = np.fft.rfft(p, n=L_big)

        # ----- a * p and a * rev(p) -----

        # ----- gradient of the smooth objective -----

        grad = 2.0 * (conv_ap[:n] + conv_ar[rev_idx])

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = 0.5 / math.sqrt(iteration)           # simple decay
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep a on the simplex -----
