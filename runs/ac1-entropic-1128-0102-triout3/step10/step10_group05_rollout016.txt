import cvxpy as cp

def lp_optimize(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    # convolution constraints
    # compute for each k: sum_{i} f[i] * b[k-i] <= M
    constraints = []
    for k in range(2*n-1):
        # compute range of i
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # sum_{i=lo..hi} f[i] * b[k-i]
        expr = cp.sum(cp.multiply(f[lo:hi+1], b[k - np.arange(lo, hi+1)]))
        constraints.append(expr <= M)
    # objective: maximize sum(b)
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if b.value is None: return None
    return b.value

import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2*n-1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af*Af, n=L)[:2*n-1]
    return conv

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho+1)
        w = np.maximum(v - theta, 0)
    else:
        w = np.full_like(v, 1.0/n)
    return w

def _exponential_sequence(n, r):
    if r <= 0 or r >= 1:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * r
    norm = 1.0 - r**n
    a = a * (1.0 - r) / norm
    a[a < 0] = 0.0
    return a

def _mixture_sequence(n, r1, r2, w):
    a1 = _exponential_sequence(n, r1)
    a2 = _exponential_sequence(n, r2)
    # each already sum to 1
    a = w * a1 + (1.0 - w) * a2
    # ensure normalization (should be 1)
    total = a.sum()
    if total > 0:
        a = a / total
    else:
        a = np.full(n, 1.0/n)
    return a

"""
Improved step‑function optimiser.

The optimiser proceeds in four stages:

1.  **Diverse initialisation** – a quick scan of single‑exponential,
    plus a random search over mixtures of two exponentials.  The best
    candidate (or a supplied historic construction) becomes the start
    point.

2.  **Adam on a log‑sum‑exp smooth approximation** of the max‑convolution.
    The schedule is longer and covers a broader range of smoothing
    parameters, producing a high‑quality continuous descent.

3.  **Exponentiated‑gradient descent** on the true max‑convolution.
    This cheap multiplicative update quickly sharpens the distribution.

4.  **Targeted simulated annealing** followed by an **aggressive greedy
    reduction** of the peak.  The greedy step moves mass from the
    most‑responsible coordinate to a low‑partner coordinate with a
    theoretically‑derived step size that would equalise the current
    maximum and the second‑largest entry.

All phases respect the global 1000 s budget and the routine prints
progress messages.  The final sequence is normalised to sum 1 and
returned as a plain Python list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Full self‑convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Helper – exponential / mixture seeds
# ----------------------------------------------------------------------
def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution on n points:
    a_i ∝ r**i  (0 < r < 1).
    The vector is already normalised to sum = 1.
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise (finite sum = 1‑r**n)
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


def _mixture_sequence(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    """
    Convex combination of two normalised exponentials.
    w∈[0,1] is the weight of the first component.
    """
    a1 = _exponential_sequence(n, r1)
    a2 = _exponential_sequence(n, r2)
    a = w * a1 + (1.0 - w) * a2
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 30000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative / mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing (longer schedule)
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-9,
                            steps: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # locate current max convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.25     # slightly larger move budget
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 4 – Aggressive greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2000
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and a prospective dst
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0

        # Choose a destination with small partner (heuristic)
        a_rev = np.zeros(n)
        a_rev[idx] = a[max_idx - idx]          # a_rev[i] = a[max_idx-i] where defined
        dst_candidates = np.where(a_rev == a_rev.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)
        else:
            dst = rng.choice(dst_candidates)

        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            # source does not help – take a modest step
            eps = min(a[src], 0.02 * a[src])
        else:
            # theoretical step that would bring the peak down to the runner‑up
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:                     # fall back to a tiny move
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the prescribed time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    final_n = 4096

    # --------------------------------------------------------------
    # 1️⃣  Initialise – exponential baseline + random mixtures
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # single‑exponential scan (coarse)
    for r in np.linspace(0.90, 0.9999, 250):
        cand = _exponential_sequence(final_n, r)
        conv = _conv_self(cand)
        score = 2.0 * final_n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            best_r = r
    print(f"[0.00s] exponential baseline r={best_r:.6f}, score={best_score:.6f}")

    # random mixtures of two exponentials
    rng = np.random.default_rng()
    mix_trials = 800
    for _ in range(mix_trials):
        w = rng.random()
        r1 = rng.uniform(0.85, 0.9995)
        r2 = rng.uniform(0.85, 0.9995)
        if r1 > r2:
            r1, r2 = r2, r1
        cand = _mixture_sequence(final_n, r1, r2, w)
        conv = _conv_self(cand)
        score = 2.0 * final_n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] mixture improve w={w:.3f} r1={r1:.5f} r2={r2:.5f} score={score:.6f}")

    # possibly reuse a historic construction if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            a0 = np.interp(np.linspace(0, 1, final_n),
                           np.linspace(0, 1, starter.size),
                           starter)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            conv = _conv_self(a0)
            score = 2.0 * final_n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = a0.copy()
                print(f"[{time.time() - start_time:.2f}s] historic starter improves score={score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2️⃣  Phase 1 – Adam on a smooth max (≈15 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase1_deadline = time.time() + min(30.0, 0.15 * (deadline - time.time()))
        best_seq, best_score = _run_adam_smoothing(
            best_seq,
            final_n,
            phase1_deadline,
            start_time,
            alphas=[0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0,
                    20.0, 40.0, 80.0, 160.0],
            lr0=0.8,
            iters_per_alpha=40000,
        )
        print(f"[{time.time() - start_time:.2f}s] after Adam score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Phase 2 – Exponentiated‑gradient descent (≈35 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + min(300.0, 0.35 * (deadline - time.time()))
        best_seq, best_score = _run_exp_gradient(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            eta0=1.2,
        )
        print(f"[{time.time() - start_time:.2f}s] after exp‑grad score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Phase 3 – Targeted annealing (≈30 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = time.time() + min(300.0, 0.30 * (deadline - time.time()))
        best_seq, best_score = _run_targeted_annealing(
            best_seq,
            final_n,
            phase3_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-9,
            steps=800_000,
        )
        print(f"[{time.time() - start_time:.2f}s] after annealing score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Phase 4 – Aggressive greedy reduction (≈20 % of the budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase4_deadline = deadline
        best_seq, best_score = _run_greedy_reduction(
            best_seq,
            final_n,
            phase4_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=8_000_000,
        )
        print(f"[{time.time() - start_time:.2f}s] after greedy score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    final_obj = 2.0 * final_n * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
