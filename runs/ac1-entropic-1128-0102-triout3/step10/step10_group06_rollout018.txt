"""
Propose a non‑negative step function that (almost) minimises
`evaluate_sequence`.  

The routine combines several complementary optimisation ideas:

1. **Diversified initialisation** – a mixture of random Dirichlet draws,
   exponential decays and a deterministic *interleaving* of sorted values
   (large numbers placed at opposite ends of the vector).  This creates
   starting points where the large coefficients are paired with small ones
   in the central convolution, which is already helpful for reducing the
   peak.

2. **Adam warm‑up on a log‑sum‑exp surrogate** – a few thousand Adam steps
   on a smooth approximation of the max‑convolution with an extended
   smoothing schedule (α up to 8192).  This quickly pushes the solution
   out of bad regions.

3. **Projected exponentiated‑gradient (mirror‑descent)** – the true max‑
   convolution is tackled using a *multi‑gradient*: all shifts whose
   value is within a small tolerance of the current maximum are taken as
   active constraints.  A larger active set (top‑20 shifts, tolerance
   0.5 %) and a more aggressive step‑size (η₀ = 0.8) give a stronger
   descent.

4. **Simulated‑annealing fine‑tuning** – on the remaining time we run a
   fast O(n) update of the full convolution when moving a small mass
   from one coefficient to another.  The move is accepted either if it
   improves the peak or with a temperature‑dependent probability, allowing
   the search to escape shallow local minima.

5. **Deterministic greedy peak reduction** – a final local search that
   transfers a fraction of the coefficient that contributes most to the
   current peak towards the smallest coefficient, accepting only strict
   improvements.

All phases respect the 1000 s budget, keep the best vector ever seen,
and finally normalise the answer to a probability simplex (the scaling
does not affect the objective because the evaluation function is
scale‑invariant apart from the clipping‑to‑1000 that never triggers with
our normalised vectors).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    total = x.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return x / total


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _interleaved_order(v: np.ndarray) -> np.ndarray:
    """
    Place the sorted values alternately at the leftmost and rightmost
    free positions (large‑small‑large‑small …).  This tends to pair large
    numbers with small ones in the central convolution.
    """
    n = v.shape[0]
    out = np.empty_like(v)
    lo, hi = 0, n - 1
    turn = True
    for x in v:
        if turn:
            out[lo] = x
            lo += 1
        else:
            out[hi] = x
            hi -= 1
        turn = not turn
    return out


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and Adam optimiser (warm‑up)
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    alpha_schedule=None,
    iters_per_alpha=4000,
    eps_log: float = 1e-16,
):
    """
    Adam optimisation on the smoothed max‑convolution objective.
    Returns the best vector and its exact score.
    """
    if alpha_schedule is None:
        alpha_schedule = [0.5, 2, 8, 32, 128, 512, 2048, 8192]

    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max to a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]

            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.0f}, score={sc:.6f}"
                    )
    return best_a, best_score


# ----------------------------------------------------------------------
# Multiplicative mirror‑descent with multi‑gradient (main optimisation)
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of maxₖ (a*a)[k] obtained by summing the gradients of the
    shifts listed in ``active_shifts``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eta0: float = 0.8,
    delta_frac: float = 0.005,
    top_k: int = 20,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    The active set contains every shift within ``delta_frac`` of the
    current maximum, limited to the ``top_k`` largest values.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Exp‑Grad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ----- active shifts -----
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ----- sub‑gradient -----
        g = _multi_gradient(a, active)

        # ----- step size (diminishing) -----
        eta = eta0 / math.sqrt(t)

        # ----- exponentiated update -----
        a = a * np.exp(-eta * g)

        # ----- re‑normalise -----
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        a = np.maximum(a, 1e-15)

        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] Periodic exact eval: score={sc:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _simulated_annealing_opt(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iter: int = 2_000_000,
    init_T_factor: float = 0.05,
    eps_start: float = 0.1,
    decay_rate: float = 0.9995,
    eps_min: float = 1e-6,
):
    """
    Fast O(n) simulated‑annealing using an incremental update of the full
    convolution.  Mass is moved between two random indices; the move is
    accepted if it reduces the current peak or with a temperature‑dependent
    probability.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    conv = _conv_self(a, fft_len)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_a = a.copy()
    best_score = cur_score
    best_max = cur_max

    rng = np.random.default_rng()
    T = init_T_factor * cur_max
    eps = eps_start
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if a[i] < 1e-12:
            continue

        # size of the transferred mass
        delta = min(a[i], eps * a[i])

        # incremental convex‑convolution update
        conv_new = conv.copy()
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        new_max = conv_new.max()

        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-15))
            accept = rng.random() < prob

        if accept:
            conv = conv_new
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            cur_score = 2.0 * n * cur_max

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_max = cur_max
                print(
                    f"[{time.time() - start_time:6.2f}s] SA improve: iter={it:,}, score={best_score:.6f}"
                )

        # annealing schedule
        if it % 1000 == 0:
            T *= decay_rate
            eps = max(eps * decay_rate, eps_min)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local improvement (same as before)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine follows the multi‑phase strategy described in the module
    docstring and returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a rich set of starting points
    # ------------------------------------------------------------------
    starts = []

    # 1) known best (if available)
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            seq = np.maximum(seq, 0.0)
            seq = np.minimum(seq, 1000.0)
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            seq = seq / seq.sum()
            starts.append(seq)
    except Exception:
        pass

    # 2) handcrafted / deterministic starts
    starts.append(np.full(n, 1.0 / n))                     # uniform
    starts.append(_exponential_start(n, r=0.9995))        # very gentle decay
    starts.append(_exponential_start(n, r=0.985))         # stronger decay
    starts.append(_gaussian_start(n))                     # bell‑shaped
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # 3) interleaved sorted Dirichlet draws
    for _ in range(4):
        a = _dirichlet_start(n, conc=0.5, rng=rng)
        a = np.sort(a)[::-1]               # descending
        a = _interleaved_order(a)           # large‑small alternating
        starts.append(a)

    # 4) random Dirichlet draws with varied concentrations
    conc_params = [0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_params:
        for _ in range(3):
            starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # ------------------------------------------------------------------
    # Evaluate raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:02d} raw score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam warm‑up (≈10 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + budget_s * 0.10
    cand_seq, cand_score = _adam_optimize(
        best_seq, explore_deadline, start_time, fft_len
    )
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – Multi‑gradient exponentiated‑gradient polishing
    #          (≈70 % of the budget)
    # ------------------------------------------------------------------
    polish_deadline = start_time + budget_s * 0.80
    polished_seq, polished_score = _expgrad_multi_opt(
        best_seq,
        polish_deadline,
        start_time,
        fft_len,
        eta0=0.8,
        delta_frac=0.005,
        top_k=20,
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polishing score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing fine‑tuning (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_seq, sa_score = _simulated_annealing_opt(
            best_seq, deadline, start_time, max_iter=2_000_000
        )
        if sa_score < best_score:
            best_score = sa_score
            best_seq = sa_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after SA score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 4 – Deterministic greedy peak reduction (final touch)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
