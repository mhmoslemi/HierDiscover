"""
Multi‑stage stochastic / deterministic optimizer.

The algorithm works as follows:

1.  Try to reuse the previously found construction (if the global variable
    ``height_sequence_1`` exists).  Otherwise initialise with a geometric
    (exponential) distribution whose decay factor is tuned by a cheap grid
    search.

2.  Run a few independent restarts.  Each restart draws a random decay
    factor, normalises the resulting vector and then performs a fast
    exponentiated‑gradient descent that minimises the *smooth* max‑convolution
    (log‑sum‑exp surrogate).  The best vector of the restarts is kept.

3.  Refine the current best vector with a deterministic greedy “peak‑reduction”
    sweep that moves a small amount of mass from the coordinate that contributes
    most to the current maximum of the self‑convolution to a coordinate that
    contributes the least.  The convolution is updated incrementally, and the
    sweep runs until the time budget is exhausted.

4.  The best vector found (always kept normalised to sum 1) is returned as a
    plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT of size a power of two.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Compute the evaluation function (without the caps performed inside
    ``evaluate_sequence``) for a vector that is already normalised to sum 1:

        2 * n * max(a*a)

    where ``a*a`` is the full discrete convolution.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution a_i ∝ r**i,   0<r<1.
    The returned vector sums to exactly 1.
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise the truncated geometric series
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


# ----------------------------------------------------------------------
# Phase 1 – fast exponentiated‑gradient descent on the *smooth* max
# ----------------------------------------------------------------------
def _exp_grad_smooth(a0: np.ndarray,
                     deadline: float,
                     start: float,
                     alphas,
                     iters_per_alpha: int = 25000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a log‑sum‑exp approximation of the max‑convolution.
    The routine runs for the given time budget and returns the best vector seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    it = 0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            # ----- convolution and smooth max -----
            Af = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time()-start:.2f}s] exp‑grad improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # log‑sum‑exp smoothing
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()                     # weights that approximate the arg‑max
            w_rev = w[::-1]

            # gradient of the smoothed max (see derivation in the comment header)
            A_f = Af
            W_rev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(A_f * W_rev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # mirror descent update (exponentiated gradient)
            eta = 0.6 / math.sqrt(it)
            a = a * np.exp(-eta * grad)

            # project back onto the simplex
            total = a.sum()
            if total > 0.0:
                a = a / total
            else:
                a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – deterministic greedy peak reduction (incremental update)
# ----------------------------------------------------------------------
def _greedy_peak_reduction(a0: np.ndarray,
                           deadline: float,
                           start: float,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from the coordinate that
    contributes most to the current maximal convolution entry to a coordinate
    that contributes the least.  The convolution is updated incrementally,
    which makes the sweep very cheap.
    """
    a = a0.copy()
    n = a.shape[0]
    L_full = 2 * n - 1
    L_fft = _next_pow2(L_full)
    conv = _conv_self(a)                     # full convolution
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0
    recompute_every = 1000

    while time.time() < deadline - 0.4 and it < max_iters:
        it += 1

        # ---- locate the current peak ----
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest value (ignoring the current maximum)
        if max_idx == 0:
            second_max = conv[1:].max()
        elif max_idx == L_full - 1:
            second_max = conv[:max_idx].max()
        else:
            second_max = max(conv[:max_idx].max(), conv[max_idx + 1:].max())

        # indices that appear in this convolution entry
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)

        # contribution of each pair (i, max_idx-i)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and for a prospective destination
        a_k = np.zeros(n, dtype=float)
        a_k[idx] = a[max_idx - idx]

        # choose a destination with the smallest partner value (but not src)
        min_partner = a_k.min()
        dst_cands = np.where((a_k == min_partner) & (np.arange(n) != src))[0]
        if dst_cands.size:
            dst = np.random.choice(dst_cands)
        else:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # ---- decide how much mass to move ----
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            # try to bring the peak down to the second largest value
            if delta_a != 0.0:
                eps_target = (max_val - second_max) / (2.0 * delta_a)
                eps = min(a[src], max(0.0, 0.9 * eps_target))
                if eps < 1e-12:
                    eps = min(a[src], 0.02 * a[src])
            else:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[src:src + n] -= 2.0 * eps * a_old
        conv[dst:dst + n] += 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # periodic exact recomputation to avoid numerical drift
        if it % recompute_every == 0:
            conv = _conv_self(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start:.2f}s] greedy improve: iter={it:,}, "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine combines random exponential seeds, a fast exponentiated‑gradient
    descent on a smooth surrogate of the max‑convolution and a deterministic
    greedy peak‑reduction sweep.  It respects the allocated time budget and
    returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0.  Try to reuse an existing construction (if present)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            base_n = starter.size
        else:
            raise Exception()
    except Exception:
        starter = None
        base_n = None

    # ------------------------------------------------------------------
    # 1.  Choose a set of problem sizes (larger n gives a finer resolution)
    # ------------------------------------------------------------------
    candidate_sizes = [1024, 2048, 4096, 8192]

    best_seq = None
    best_score = float("inf")

    for n in candidate_sizes:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # 2.  Obtain an initial vector for this size
        # ------------------------------------------------------------------
        if starter is not None and base_n == n:
            a0 = starter.copy()
        else:
            # quick grid search for a good decay factor
            r_grid = np.linspace(0.85, 0.9999, 120)
            best_r = r_grid[0]
            best_init = float("inf")
            for r in r_grid:
                cand = _exponential_sequence(n, r)
                sc = _objective(cand)
                if sc < best_init:
                    best_init = sc
                    best_r = r
            a0 = _exponential_sequence(n, best_r)

        # ------------------------------------------------------------------
        # 3.  Several independent restarts with random decay factors
        # ------------------------------------------------------------------
        restarts = 3
        for _ in range(restarts):
            if time.time() >= deadline:
                break
            # random decay factor – a little perturbation around the grid optimum
            r = best_r * (1.0 + np.random.uniform(-0.03, 0.03))
            a = _exponential_sequence(n, r)

            # ----- phase A: smooth‑max exponentiated‑gradient -----
            alphas = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
            a_opt, sc_opt = _exp_grad_smooth(
                a,
                deadline,
                start_time,
                alphas,
                iters_per_alpha=20000,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time()-start_time:.2f}s] new best after exp‑grad "
                      f"n={n} score={sc_opt:.6f}")

        # ------------------------------------------------------------------
        # 4.  Greedy peak‑reduction on the current best of this size
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            a_greedy, sc_greedy = _greedy_peak_reduction(
                best_seq,
                deadline,
                start_time,
                max_iters=3_000_000,
            )
            if sc_greedy < best_score:
                best_score = sc_greedy
                best_seq = a_greedy.copy()
                print(f"[{time.time()-start_time:.2f}s] post‑greedy improve "
                      f"n={n} score={sc_greedy:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final normalisation and safe return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a simple uniform distribution of length 4096
        n = 4096
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(best_seq.shape[0], 1.0 / best_seq.shape[0])

    final_obj = _objective(best_seq)
    print(f"[{time.time()-start_time:.2f}s] final objective ≈ {final_obj:.6f}")

    # convert to plain Python list
    return [float(x) for x in best_seq]
```

This implementation respects the 1000 s time limit, performs several independent stochastic restarts, refines each candidate with a fast exponentiated‑gradient descent on a smooth surrogate of the max‑convolution, and then applies an aggressive deterministic greedy sweep that incrementally reduces the peak of the self‑convolution.  The best sequence discovered within the budget is returned as a list of non‑negative heights.
