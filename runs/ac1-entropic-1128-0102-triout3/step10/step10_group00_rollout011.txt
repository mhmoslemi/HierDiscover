def _conv_fft(x, y):
    n = len(x)
    m = len(y)
    L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n+m-1]
    return conv

    """
    Linear self‑convolution a*a using a real FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    L = _next_pow2(n + m - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (Σ seq)²  (here Σ seq = 1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())   # Σ seq = 1  → denominator drops out


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (identical to the one used before)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap annealing that moves tiny mass packets."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule (fast cooling)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)                # more likely to move mass from a large entry
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30      # move up to 30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8192 works well for the given budget.
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # 1) Generate a rich family of warm‑starts.
    # ------------------------------------------------------------------
    warm_starts = []

    # power‑law decays (monotone decreasing)
    for p in np.linspace(0.2, 3.0, 15):
        w = (1.0 - np.arange(n) / (n - 1)) ** p
        if w.sum() > 0:
            warm_starts.append(w / w.sum())

    # exponentials
    for lam in np.linspace(0.0, 5.0, 11):
        w = np.exp(-lam * np.arange(n))
        if w.sum() > 0:
            warm_starts.append(w / w.sum())

    # uniform
    warm_starts.append(np.full(n, 1.0 / n))

    # the global starter, if present – up‑sample to length n
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # linear interpolation to the new grid
            x_old = np.linspace(0.0, 1.0, starter.size)
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, starter)
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                warm_starts.append(up / up.sum())
    except Exception:
        pass

    # keep the best warm‑start seen so far
    best_a = None
    best_score = float("inf")
    for a in warm_starts:
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        print(f"[{time.time() - start:.2f}s] warm‑start score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2) Refine with a smoothed Adam optimiser.
    # ------------------------------------------------------------------
    if best_a is None:                 # should never happen, but guard against it
        best_a = np.full(n, 1.0 / n)
        best_score = _objective(best_a)

    a = best_a.copy()
    m = np.zeros_like(a)               # first moment (Adam)
    v = np.zeros_like(a)               # second moment (Adam)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.08                         # base learning‑rate
    it = 0

    # temperature schedule for the soft‑max (smooth → hard)
    tau_min = 3.0
    tau_max = 2000.0

    while time.time() < deadline:
        it += 1

        # self‑convolution and soft‑max weights
        conv = _conv_self(a)                            # length 2·n‑1
        max_c = conv.max()
        # linear interpolation of the temperature
        elapsed = time.time() - start
        tau = tau_min + (tau_max - tau_min) * (elapsed / (deadline - start))

        # stable soft‑max
        w_exp = np.exp(tau * (conv - max_c))
        w = w_exp / w_exp.sum()                         # shape (2·n‑1,)

        # gradients of the smoothed max (see analysis)
        w_rev = w[::-1]

        term2 = _conv_fft(w, a)[:n]                     # conv(w, a)  → first n entries
        term1 = _conv_fft(w_rev, a)[(n - 1):(2 * n - 1)][::-1]
        grad = 2.0 * (term1 + term2)                    # ∇_a Σ w_i·conv_i

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)                         # keep feasibility

        # occasional exact evaluation (cheap because Σ a = 1)
        if it % 200 == 0:
            cur_score = 2.0 * n * max_c                 # same as _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] iter={it:,} "
                      f"best={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Light annealing polish (only if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _anneal(best_a, deadline, start)

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    final_val = _objective(best_a)
    print(f"[{time.time() - start:.2f}s] final objective = {final_val:.6f}")

    return [float(v) for v in best_a]
