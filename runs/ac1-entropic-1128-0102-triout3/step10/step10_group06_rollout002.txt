"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
--------------------------------
1. **Rich initial pool** – a handful of deterministic shapes
   (uniform, exponential, Gaussian, sparse masks) together with many
   random Dirichlet draws.  The best of those is kept as a warm start.

2. **Fast exponentiated‑gradient (multi‑gradient)** – a mirror‑descent
   on the true max‑convolution objective using the top‑K active shifts.
   This is the same idea as the reference code but with a more aggressive
   learning‑rate schedule and a tighter active‑set tolerance.

3. **Stochastic local mass‑transfer** – while time permits we repeatedly
   pick a “source’’ coefficient (large) and a “target’’ coefficient (small)
   and move a modest fraction of mass.  The effect on the objective is
   evaluated with a cheap FFT‑based convolution; only improving moves are
   kept.  This simple simulated‑annealing‑style step often breaks the
   plateau left by gradient descent.

4. **Deterministic greedy peak reduction** – a final fine‑grained sweep
   that moves mass from the index contributing most to the current peak
   towards the smallest coefficient, accepting only strict improvements.

All phases respect the global time budget and always keep the best
sequence encountered (normalised to sum = 1).  The routine works on a
fixed length `n = 4800` (the size used by the reference solution) and
uses only NumPy – no external solvers are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _obj_from_seq(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a *normalised*
    sequence (sum(a)==1).  Returns ``2·n·max(conv)``.
    """
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sequence generators (hand‑crafted / random)
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return x / s


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n, dtype=float)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_start(n: int, sparsity: float, rng: np.random.Generator) -> np.ndarray:
    """Randomly keep a fraction ``sparsity`` of entries, Dirichlet on them."""
    k = max(2, int(sparsity * n))
    mask = rng.choice(n, size=k, replace=False)
    sub = rng.random(k)
    sub /= sub.sum()
    seq = np.zeros(n, dtype=float)
    seq[mask] = sub
    return seq


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution: sum of the gradients of the
    shifts listed in ``active_shifts``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eta0: float = 0.6,
    delta_frac: float = 0.001,
    top_k: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution
    objective, using a *multi‑gradient* built from the top‑K (or all)
    shifts that are within ``delta_frac`` of the current maximum.
    Returns the best vector ever seen and its exact score (2·n·maxConv).
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Multi‑ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ----- active set -----
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ----- gradient -----
        g = _multi_gradient(a, active)

        # ----- step size -----
        eta = eta0 / math.sqrt(t)

        # ----- exponentiated update -----
        a = a * np.exp(-eta * g)

        # ----- re‑normalise onto the simplex -----
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # protect against exactly zero values
        a = np.maximum(a, 1e-15)

        # occasional exact evaluation (guard against drift)
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic local mass‑transfer (cheap surrogate)
# ----------------------------------------------------------------------


def _random_local_search(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    max_trials: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly move a fraction of mass from a
    relatively large coefficient to a relatively small one and keep the
    move if it strictly reduces the objective (2·n·maxConv).  The
    convolution is evaluated with the fast FFT routine, so thousands of
    trials are cheap.
    """
    a = a0.copy()
    best_a = a.copy()
    best_obj = _obj_from_seq(a, fft_len, n)

    trials = 0
    while time.time() < deadline and trials < max_trials:
        trials += 1

        # source ~ weighted by current mass
        src = rng.choice(n, p=a / a.sum())
        if a[src] < 1e-12:
            continue

        # target – favour small entries (inverse weighting)
        inv_weights = 1.0 / (a + 1e-12)
        inv_weights[src] = 0.0
        inv_weights /= inv_weights.sum()
        dst = rng.choice(n, p=inv_weights)

        eps = 0.1 * a[src]            # move 10 % of the source mass
        if eps < 1e-14:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # objective (no need to call ``evaluate_sequence``)
        obj_new = _obj_from_seq(a_new, fft_len, n)

        # accept only improving moves
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new            # continue from the improved point
            # occasionally report progress
            if trials % 2000 == 0:
                print(
                    f"[{time.time() - start_time:6.2f}s] RandomLocal improve: trials={trials}, score={best_obj:.6f}"
                )

    return best_a, best_obj


# ----------------------------------------------------------------------
# Deterministic greedy peak reduction (as in the reference)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    fft_len = _next_pow2(2 * n - 1)

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = index with smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]          # a modest fraction of the source mass
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, fft_len).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin – the caller aborts if we exceed budget_s seconds
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size (same as reference, good trade‑off between quality and cost)
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse collection of initial candidates
    # ------------------------------------------------------------------
    starts = []

    # deterministic shapes
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_exponential_start(n, r=0.9995))
    starts.append(_exponential_start(n, r=0.985))
    starts.append(_exponential_start(n, r=0.95))
    starts.append(_gaussian_start(n))

    # random Dirichlet draws with several concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]:
        for _ in range(3):
            starts.append(_dirichlet_start(n, conc, rng))

    # sparse random patterns
    for sparsity in [0.05, 0.10, 0.20]:
        for _ in range(2):
            starts.append(_sparse_start(n, sparsity, rng))

    # previous best if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Pick the best raw start (exact evaluation)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] init #{idx+1:02d} score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase A – intensive multi‑gradient exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    # Allocate roughly 80 % of the total time for this phase (the rest
    # is kept for stochastic local search and the final greedy sweep).
    phase_a_deadline = start_time + 0.80 * budget_s
    if phase_a_deadline > deadline:
        phase_a_deadline = deadline

    a_polished, score_polished = _expgrad_multi_opt(
        best_seq,
        phase_a_deadline,
        start_time,
        fft_len,
        eta0=0.8,          # a slightly larger initial step‑size
        delta_frac=5e-4,   # tighter active‑set tolerance
        top_k=7,
    )
    if score_polished < best_score:
        best_score = score_polished
        best_seq = a_polished.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polishing score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase B – stochastic local mass‑transfer (many short restarts)
    # ------------------------------------------------------------------
    # The remaining budget is split into short “restart’’ windows.
    while time.time() < deadline:
        remaining = deadline - time.time()
        # each restart gets at most ~20 s (exp‑grad + local search)
        if remaining < 25.0:
            break

        restart_deadline = time.time() + 20.0

        # random Dirichlet start with a random concentration parameter
        conc = rng.uniform(0.1, 8.0)
        a0 = _dirichlet_start(n, conc, rng)

        # short exponentiated‑gradient polishing (≈10 s)
        sub_deadline = time.time() + 10.0
        a_mid, score_mid = _expgrad_multi_opt(
            a0,
            sub_deadline,
            start_time,
            fft_len,
            eta0=0.7,
            delta_frac=0.001,
            top_k=5,
        )
        if score_mid < best_score:
            best_score = score_mid
            best_seq = a_mid.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after short‑expgrad score={best_score:.6f}"
            )

        # stochastic local search (use the rest of the restart window)
        a_loc, score_loc = _random_local_search(
            a_mid, restart_deadline, start_time, n, fft_len, rng, max_trials=5000
        )
        if score_loc < best_score:
            best_score = score_loc
            best_seq = a_loc.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after random search score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase C – final deterministic greedy refinement (any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after final greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # final verification (exact evaluation)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_score:.6f}")

    return [float(x) for x in best_seq.tolist()]
