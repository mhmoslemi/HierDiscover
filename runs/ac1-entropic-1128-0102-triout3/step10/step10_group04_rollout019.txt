"""
optimize_step_function – a three‑stage optimiser.

Stage 1
-------
Enumerate a modest collection of parametric families (uniform,
linear, power‑law, beta and exponential).  The best candidate of this
cheap sweep is taken as a warm start.

Stage 2
-------
Projected gradient descent on a smooth approximation of the
ℓ∞‑norm.  We replace  max(conv) by a soft‑max:
    f(a)=log(∑exp(α·conv))/α .
Its gradient can be written as a convolution of the current vector with
the soft‑max weights, which is evaluated with FFTs.  After each step the
vector is projected onto the probability simplex (exact ℓ₁‑normalisation
and non‑negativity).

Stage 3
-------
A deterministic “peak‑flattening’’ pass – exactly the greedy routine
used in the reference solution – cleans the remaining small peaks.
All three stages respect the global time budget (default 1000 s).

The function returns a plain Python list of floats; any scaling of the
vector is irrelevant for the objective because the evaluation function
is homogeneous.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative,
    sum = 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # numeric fallback
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact objective value for a given self‑convolution."""
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def _compute_score(a: np.ndarray) -> float:
    """Fast wrapper – assumes a is non‑negative and not all zero."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return _score_from_conv(conv, n, a.sum())


# ----------------------------------------------------------------------
# Stage‑1 – parametric initialisation
# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Return a list of a few hand‑crafted shapes (all summed to 1)."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n))

    # 2) linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    cand.append(lin / lin.sum())

    # 3) power‑law (a_i ∝ (i+1)^p, p in [-2,2])
    for p in np.linspace(-2.0, 2.0, 9):
        pw = (np.arange(1, n + 1, dtype=float) ** p)
        pw = np.maximum(pw, 0.0)
        cand.append(pw / pw.sum())

    # 4) beta‑law (a_i ∝ x^{α‑1}(1‑x)^{β‑1})
    xs = np.linspace(0.0, 1.0, n)
    for α in [0.4, 0.6, 1.0, 2.0, 4.0]:
        for β in [0.4, 0.6, 1.0, 2.0, 4.0]:
            b = (xs ** (α - 1.0)) * ((1.0 - xs) ** (β - 1.0))
            b = np.maximum(b, 0.0)
            cand.append(b / b.sum())

    # 5) exponential decay (parameter q)
    for q in [0.995, 0.997, 0.998, 0.999, 0.9995, 0.9998]:
        expv = q ** np.arange(n, dtype=float)
        expv = np.maximum(expv, 0.0)
        cand.append(expv / expv.sum())

    return cand


# ----------------------------------------------------------------------
# Stage‑2 – smooth (soft‑max) projected gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 300_000,
                    init_alpha: float = 5.0,
                    final_alpha: float = 150.0,
                    init_lr: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Minimises the true max‑convolution via a smooth approximation.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)          # length for self‑convolution
    best_a = a.copy()
    best_score = _compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- convolution of the current vector ---
        conv = _conv_self(a, L)                     # length 2·n‑1
        cur_max = conv.max()

        # --- soft‑max weights (smooth approximation of max) ---
        alpha = init_alpha + (final_alpha - init_alpha) * (it / max_iters)
        shift = cur_max
        # numerical stability: subtract the max before exponentiation
        exp_terms = np.exp(alpha * (conv - shift))
        w = exp_terms / exp_terms.sum()             # shape (2n‑1,)

        # --- gradient of the smooth objective ---
        # we need: grad_i = 2 * Σ_k w_k * a_{k‑i}
        # which is a convolution of a with the reversed weight vector
        w_rev = w[::-1]
        conv_len = 3 * n - 2
        Nfft = _next_pow2(conv_len)
        fft_a = np.fft.rfft(a, n=Nfft)
        fft_w = np.fft.rfft(w_rev, n=Nfft)
        full = np.fft.irfft(fft_a * fft_w, n=Nfft)[:conv_len]
        offs = 2 * n - 2
        grad = 2.0 * full[offs - np.arange(n)]

        # --- step size (decays like 1/√t) and projected update ---
        lr = init_lr / math.sqrt(it)
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)

        a = a_new

        # --- keep the best vector seen ---
        cur_score = _compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑grad it={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage‑3 – deterministic greedy peak‑flattening (as in the reference)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a tiny amount of mass from the index that contributes
    most to the current peak to an index that contributes least, updating
    the convolution incrementally.
    """
    a = a0.copy()
    n = a.shape[0]
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, Lfft)            # full self‑convolution (2·n‑1)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < 10_000_000:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second biggest entry (ignore the current max)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # choose a destination with the smallest partner value
        partner_vals = a[max_idx - idx]
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        # size of the transfer
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-30)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ---- incrementally update the full convolution ----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # occasional full recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = _compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy it={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size – 4096 is a good trade‑off between flexibility and speed
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Stage 1 – cheap parametric sweep
    # --------------------------------------------------------------
    candidates = _initial_candidates(n)
    best_seq = None
    best_score = float("inf")

    for cand in candidates:
        if time.time() >= deadline:
            break
        scr = _compute_score(cand)
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] init candidate score={scr:.6f}")

    # fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _compute_score(best_seq)

    # --------------------------------------------------------------
    # Stage 2 – smooth soft‑max projected gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + remaining * 0.55   # allocate ~55 % of left time
        seq, scr = _smooth_descent(best_seq, phase2_deadline, start_time)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑grad improve: "
                  f"score={scr:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – deterministic greedy peak flattening
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_greedy_reduction(best_seq, deadline, start_time, best_score)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy improve: "
                  f"score={scr:.6f}")

    # --------------------------------------------------------------
    # Very short random‑restart polishing if any seconds remain
    # --------------------------------------------------------------
    while time.time() < deadline - 2.0:
        pert = best_seq + 0.01 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)
        pert /= pert.sum()
        # a quick few‑step smooth descent (2 s)
        tmp_deadline = time.time() + 2.0
        seq, scr = _smooth_descent(pert, tmp_deadline, start_time,
                                   max_iters=30_000,
                                   init_alpha=10.0,
                                   final_alpha=80.0,
                                   init_lr=0.3)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑restart improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (the objective is homogeneous, any scaling works)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
