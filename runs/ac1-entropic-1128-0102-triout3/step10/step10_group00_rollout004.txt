assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function `a = (a₀,…,aₙ₋₁)` that minimizes  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear (self) convolution.  
The objective is scale‑invariant, therefore we keep the sequence on the
probability simplex (non‑negative, sum = 1).  

The search consists of three stages

1. **Analytic warm‑starts** – a fine sweep of simple symmetric families  
   *tent* (`aᵢ ∝ min(i,n‑1‑i)^p`) and *trapezoid* (flat top with linear
   slopes).  The best of these is taken as the initial point.

2. **Multi‑index projected sub‑gradient descent** – instead of using the
   sub‑gradient coming from a single maximal convolution entry, we average
   the sub‑gradients of the `k` largest entries (default = 5).  This gives a
   richer descent direction and lets the algorithm escape shallow local
   minima.  Momentum and a `1/√t` step‑size are used; after each step the
   iterate is projected back onto the simplex.

3. **Light simulated‑annealing polish** – tiny random transfers of mass
   between positions are accepted with a Metropolis rule.  The annealer
   runs only while time permits and usually yields a final improvement.

All calculations are performed with FFT‑based convolutions (`O(n log n)`);
the algorithm respects the 1000 s time limit by constantly checking the
deadline.  The final vector is normalised to sum = 1 and returned as a
plain Python list, ready for `evaluate_sequence`.

```python
"""
search for a low‑objective non‑negative step‑function.
Algorithm
---------
1. Create a set of analytic shapes (tent with power p, trapezoid with
   side‑fraction s) on a reasonably large grid (n = 16384).  Keep the
   best shape (lowest exact objective).
2. Refine that shape with projected multi‑index sub‑gradient descent.
   At each iteration we average the sub‑gradients of the k largest
   convolution entries (k≈5), use momentum and a 1/√t step‑size,
   and project onto the probability simplex.
3. Finish with a cheap simulated‑annealing polish that moves tiny mass
   packets between positions.
The whole routine stops before the supplied time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale invariant, sum(a) == 1 ⇒ F = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Multi‑index projected sub‑gradient descent
# ----------------------------------------------------------------------
def _subgradient_descent_multi(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that averages the sub‑gradients of the
    top_k convolution entries (k ≥ 1).  Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9                # momentum decay
    step0 = 1.0               # initial step size
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"sub‑grad iter={it:,} score={cur_score:.6f}")

        # ---- sub‑gradient from the top_k entries ---------------------------------
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]

        grad = np.zeros_like(a)
        for i in idxs:
            lo = max(0, i - (n - 1))
            hi = min(n - 1, i)
            j = np.arange(lo, hi + 1, dtype=int)
            grad[j] += 2.0 * a[i - j]
        grad /= len(idxs)          # average over the chosen indices

        # ---- momentum, step, projection -----------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (very cheap – only a few ten‑thousands steps)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            steps: int = 30000) -> tuple[np.ndarray, float]:
    """Move tiny mass packets, accept with Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / steps)

        src = rng.choice(n, p=a)          # more mass ⇒ more likely to lose it
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # move up to 30 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a list of non‑negative floats (the vector is kept on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger grid gives a finer discretisation.
    # 16384 works comfortably within the time budget.
    # --------------------------------------------------------------
    n = 16384

    # --------------------------------------------------------------
    # 1) Analytic warm‑starts (tent & trapezoid families)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ----- symmetric tent a_i ∝ min(i,n‑1‑i)^p  --------------------
    exponents = np.linspace(0.7, 1.4, 15)      # fine sweep
    for p in exponents:
        x = np.arange(n, dtype=float) / (n - 1)          # points in [0,1]
        w = np.minimum(x, 1.0 - x) ** p
        if w.sum() == 0:
            continue
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score, best_seq = score, a.copy()
        if time.time() >= deadline:
            break

    # ----- trapezoid (flat top + linear sides) --------------------
    side_fracs = np.linspace(0.05, 0.45, 9)   # fraction of total length used for each side
    for f in side_fracs:
        Ls = int(f * n)                         # length of one sloping side
        if Ls < 1 or 2 * Ls >= n:
            continue
        Lc = n - 2 * Ls                         # length of flat top
        h = 1.0 / (n - Ls)                      # height such that Σa = 1 (≈ exact)
        # left slope: values from 0 (inclusive) to h (exclusive)
        left = h * np.linspace(0, 1, Ls, endpoint=False)
        right = left[::-1]
        plateau = np.full(Lc, h)
        a = np.concatenate([left, plateau, right])
        a = a / a.sum()
        score = _objective(a)
        if score < best_score:
            best_score, best_seq = score, a.copy()
        if time.time() >= deadline:
            break

    # ----- a few random Dirichlet samples (optional) ---------------
    for _ in range(5):
        a = np.random.dirichlet(np.ones(n))
        score = _objective(a)
        if score < best_score:
            best_score, best_seq = score, a.copy()
        if time.time() >= deadline:
            break

    # fallback – uniform (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 2) Refine the best warm‑start with multi‑index sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:                       # keep a few seconds for polishing
        sub_deadline = time.time() + remaining * 0.85
        refined_seq, refined_score = _subgradient_descent_multi(
            best_seq, sub_deadline, start_time, top_k=5)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 3) Light simulated‑annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time,
                                          steps=50000)
        if anneal_score < best_score:
            best_score, best_seq = anneal_score, anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal score={anneal_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
```
