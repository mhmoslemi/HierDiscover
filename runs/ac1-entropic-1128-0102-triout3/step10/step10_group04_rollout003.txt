"""
Propose a non‑negative step function that (approximately) minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Initial shape** – start from the previously found construction (if
   ``height_sequence_1`` exists) or from a simple decreasing profile.

2. **Smoothed gradient descent** – minimise a smooth surrogate of the
   ``max`` term:
        f(a) = log‑sum‑exp( β·(a*a) )
   The gradient of f is a weighted sum of the gradients of all convolution
   entries; the weights are a soft‑max with temperature β.
   After each gradient step the vector is projected back onto the probability
   simplex (non‑negative, sum = 1).  β is increased slowly so the optimisation
   focuses on the true maximum later in the run.  The step size decays like
   1/√iteration.

   All convolutions are performed by FFT with a single zero‑padded length
   ``L = 2^⌈log₂(3·n‑2)⌉`` – this length is large enough for both the
   self‑convolution (length 2·n‑1) and the correlation needed for the gradient.

3. **Optional greedy polishing** – if time remains, a deterministic,
   O(n) “peak‑flattening’’ pass (identical to the greedy routine used in the
   previous optimiser) is run for a few thousand moves.

The whole routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence found as a plain Python list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # implementation of [Wang & Carreira‑Perpiñán, 2013]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # fallback – should not happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Smoothed gradient descent on the log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------


def _run_softmax_gradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          beta_max: float = 30.0,
                          step0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on   f(a)=log∑exp(β·(a*a)).
    Returns the best vector seen and its true objective value
    (2·n·max(a*a) because the sum of a is 1).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    # one FFT length works for both the self‑convolution (2·n‑1) and the
    # correlation needed for the gradient (3·n‑2)
    L = _next_pow2(3 * n - 2)

    it = 0
    while time.time() < deadline:
        it += 1

        # --- self‑convolution (linear, via zero‑padded FFT) ----------------
        Fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Fa * Fa, n=L)          # length L
        conv = conv_full[: 2 * n - 1]                  # linear part only
        max_conv = conv.max()

        # true objective (sum a = 1)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,}  score={cur_score:.6f}")

        # --- soft‑max weights over the convolution entries -----------------
        # temperature schedule: start small, increase slowly to beta_max
        beta = min(beta_max,
                   0.5 + (beta_max - 0.5) * min(1.0, it / 200_000.0))
        # stability: subtract the maximal value before exponentiating
        exp_vals = np.exp(beta * (conv - max_conv))
        p = exp_vals / exp_vals.sum()                 # shape (2·n‑1,)

        # --- gradient of the surrogate ------------------------------------
        # grad = 2 * (p ⋆ a)   where ⋆ is the linear cross‑correlation
        a_rev = a[::-1]                               # reversed a
        Fa_rev = np.fft.rfft(a_rev, n=L)
        Fp = np.fft.rfft(p, n=L)
        corr_full = np.fft.irfft(Fp * Fa_rev, n=L)    # linear correlation
        # extract the n entries that correspond to grads[0..n‑1]
        grad = 2.0 * corr_full[n - 1: n - 1 + n]

        # --- step size ----------------------------------------------------
        step = step0 / (1.0 + math.sqrt(it))

        # --- projected gradient step ---------------------------------------
        a = a - step * grad
        a = np.maximum(a, 0.0)         # clip negatives before projection
        a = _project_simplex(a)        # force sum = 1 and non‑negativity

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑flattening (same as in the previous script)
# ----------------------------------------------------------------------


def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Move a tiny amount of mass from the index that contributes most to the
    current convolution peak to an index that contributes least, updating the
    full convolution incrementally.  This very cheap local search can shave
    the last few 10⁻³ off the objective.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        partner_vals = a[max_idx - idx]
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a tiny move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # incremental update of the convolution (linear‑time)
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the given time budget and returns a plain Python list
    of floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – the harness adds its own timeout
    deadline = start_time + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem size – 4096 provides a good balance between flexibility
    # and speed given the 1000 s budget.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise – try to reuse the previously discovered construction if
    # it is present, otherwise fall back to a simple monotone profile.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a0 = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, starter.size),

            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:

    except Exception:
        # a decreasing exponential – empirically a good starting point
        q = 0.9995
        a0 = (1 - q) * (q ** np.arange(n, dtype=float))
        a0 = a0 / a0.sum()

    # --------------------------------------------------------------
    # Phase 1 – smoothed gradient descent on log‑sum‑exp surrogate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # reserve a little room for the optional greedy polishing

    best_seq, best_score = _run_softmax_gradient(
        a0, n, phase1_deadline, start_time,
        beta_max=30.0, step0=5.0)

    # --------------------------------------------------------------
    # Phase 2 – deterministic greedy peak‑flattening (uses any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:

            best_seq, n, deadline, start_time,

    # --------------------------------------------------------------
    # Final normalisation (the objective is homogeneous, any scaling works)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
