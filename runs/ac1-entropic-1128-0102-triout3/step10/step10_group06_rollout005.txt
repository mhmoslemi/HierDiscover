    while time.time() < deadline:
        # compute conv for current a

        cur_max = conv.max()
        k_max = conv.argmax()
        
        lo = max(0, k_max - (len(a) - 1))
        hi = min(len(a)-1, k_max)
        i_idx = np.arange(lo, hi+1,dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]  # source of max pair

                a = a_new
                # early break to continue from new state

"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Rich initial pool** – deterministic constructions (uniform,
   exponential, Gaussian) and a handful of Dirichlet draws with
   various concentration parameters.  The best of them (by the exact
   objective) seeds the optimisation.

2. **Warm‑up** – a few thousand Adam steps on the smooth
   log‑sum‑exp surrogate of the convolution peak.  The surrogate
   parameter α is gradually increased so the optimizer first sees a
   very smooth landscape and then focuses on the true maximum.

3. **Adaptive exponentiated‑gradient (Exp‑Grad)** – mirror‑descent on
   the true ``max(conv)`` objective.  The active set of shifts is
   chosen adaptively: early iterations use a generous tolerance and a
   large top‑K set (many shifts influence the gradient); later they
   focus on the few dominant shifts.  The learning‑rate decays as
   1/√t.

4. **Local stochastic hill‑climbing** – while time permits we repeatedly
   move a small fraction of mass from a source index (the one that
   contributes most to the current peak) to a destination index with
   the smallest coefficient.  Moves are accepted only if they improve
   the exact ``evaluate_sequence`` score (with a tiny chance to accept
   equal scores to escape plateaus).  This often squeezes the remaining
   few hundredths out of the bound.

All stages respect the total time budget (default 1000 s).  The final
sequence is normalised to sum = 1 (the objective is scale‑invariant) and
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Start‑point generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# 1) Warm‑up with Adam on a smooth surrogate
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate.  The routine stops
    as soon as the deadline is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule – we go a little farther than the reference
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]
    iters_per_alpha = 2000

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # snapshot probability vector from logits
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            # w_rev needed because conv gradient is a*rev(w)
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 2) Adaptive exponentiated‑gradient on the true max‑conv objective
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] obtained by summing the gradients of
    the shifts listed in ``active_shifts``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt_adaptive(a0: np.ndarray,
                                deadline: float,
                                start_time: float,
                                fft_len: int,
                                eta0: float = 0.9,
                                delta_start: float = 0.02,
                                delta_end: float = 0.001,
                                topk_start: int = 30,
                                topk_end: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
    objective.  The tolerance for the active set and the size of the
    top‑K set decay linearly from a generous ``delta_start`` /
    ``topk_start`` to a tight ``delta_end`` / ``topk_end``.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Exp‑Grad improve: iter={t:,}, score={cur_score:.6f}")

        # ---- adaptivity ----
        progress = (time.time() - start_time) / (deadline - start_time + 1e-12)
        delta = max(delta_end, delta_start * (1.0 - progress))
        topk = max(topk_end, int(topk_start * (1.0 - progress) + topk_end * progress))

        # ---- active set ----
        thresh = cur_max - max(delta * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]
        if active.size > topk:
            part = np.argpartition(-conv[active], topk - 1)[:topk]
            active = active[part]

        # ---- sub‑gradient ----
        g = _multi_gradient(a, active)

        # ---- step‑size schedule ----
        eta = eta0 / math.sqrt(t)

        # ---- exponentiated update ----
        a = a * np.exp(-eta * g)

        # ---- re‑normalise onto the simplex ----
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        a = np.maximum(a, 1e-15)

        # occasional exact check
        if t % 400 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Exp‑Grad exact improve: iter={t:,}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 3) Stochastic hill‑climbing (fine‑grained local search)
# ----------------------------------------------------------------------


def _local_hill_climb(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly transfer a modest fraction of the coefficient that most
    contributes to the current convolution peak to the smallest
    coefficient.  Moves are accepted only if they strictly improve the
    exact score (ties are accepted with a tiny probability to avoid
    stagnation).  The routine stops when time runs out.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_best_score
    rng = np.random.default_rng(int(time.time() * 1e6) & 0xFFFFFFFF)

    # pre‑compute FFT length for conv used inside the loop
    fft_len = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        # decide whether to perform a *targeted* move (50 %) or a
        # completely random one (50 %).
        if rng.random() < 0.5:
            # --- targeted move based on the current peak ---
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            k_max = conv.argmax()
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            i_idx = np.arange(lo, hi + 1, dtype=int)
            j_idx = k_max - i_idx
            contrib = a[i_idx] * a[j_idx]
            src = i_idx[contrib.argmax()]          # index that hurts the peak most
            # destination = index with the smallest coefficient (but not src)
            dst_candidates = np.delete(np.arange(n), src)
            dst = dst_candidates[a[dst_candidates].argmin()]
            # take a modest fraction of the source mass
            eps = 0.22 * a[src]                     # 22 % of the source mass
        else:
            # --- purely random move (helps escape tiny plateaus) ---
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)
            eps = 0.05 * a[src]                     # 5 % of the source mass

        if eps < 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < 0.0:
            continue

        sc = evaluate_sequence(a_prop.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb improve: score={best_score:.6f}")
        else:
            # occasional tie‑acceptance to keep the walk alive
            if sc == best_score and rng.random() < 0.001:
                a = a_prop

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – orchestrates the whole search
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine respects the given time budget (default 1000 s) and
    returns the best sequence found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (same as in the reference implementation)
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of start vectors
    # ------------------------------------------------------------------
    starts = []

    # possibly reuse a previously discovered construction
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # handcrafted simple constructions
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_exponential_start(n, r=0.9995))    # very gentle decay
    starts.append(_exponential_start(n, r=0.985))     # stronger decay
    starts.append(_gaussian_start(n))                 # bell‑shaped
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # a few random Dirichlet draws with several concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # ------------------------------------------------------------------
    # 2) Evaluate raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Allocate time to the three optimisation phases
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = deadline - time.time()
    # Rough split: 7 % warm‑up, 78 % adaptive Exp‑Grad, 15 % hill‑climb
    warm_deadline = start_time + budget_s * 0.07
    expgrad_deadline = start_time + budget_s * 0.85
    hill_deadline = deadline

    # ------------------------------------------------------------------
    # Phase 1 – Adam warm‑up
    # ------------------------------------------------------------------
    cand_seq, cand_score = _adam_optimize(best_seq, warm_deadline, start_time, fft_len)
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adaptive exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    polished_seq, polished_score = _expgrad_multi_opt_adaptive(
        best_seq, expgrad_deadline, start_time, fft_len,
        eta0=0.9, delta_start=0.02, delta_end=0.001,
        topk_start=30, topk_end=5
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Exp‑Grad score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Stochastic hill‑climbing (final fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < hill_deadline:
        final_seq, final_score = _local_hill_climb(
            best_seq, hill_deadline, start_time, best_score
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq.tolist()]
