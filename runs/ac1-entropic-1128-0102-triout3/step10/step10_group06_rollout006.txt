"""
Sidon‐set based construction for the step‑function minimisation.

The key idea is to pick a subset of indices S⊂{0,…,n−1} such that all
pairwise sums i+j (including the “diagonal’’ sums 2i) are distinct.
With equal mass on each chosen index the autoconvolution at any shift
contains at most one ordered pair, hence its maximum equals 2·w² where
w=1/|S|.  The resulting score is

    score = 2·n·max_b / (∑a)² = 4·n / |S|².

For n=4800 a Sidon set of size ≥115 yields a score below the target
1.503 (|S|≈120 gives ≈1.33, |S|≈138 gives ≈1.01).

The implementation repeatedly builds random Sidon sets (greedy selection
in a random order) until the time budget expires or a sufficiently
large set is found.  The best sequence found is returned.
"""

import time
import numpy as np


def _build_sidon_set(n: int, rng: np.random.Generator) -> list[int]:
    """
    Greedy construction of a Sidon set (all unordered sums distinct)
    on {0,…,n‑1}.  A random permutation of candidates is tried, and a
    candidate is kept iff none of its new sums collide with existing
    ones (including its diagonal sum 2·p).

    Returns a list of chosen indices.
    """
    # random order of candidates
    cand = np.arange(n, dtype=int)
    rng.shuffle(cand)

    chosen = []                # list of indices already accepted
    used_sums = set()          # set of all i+j already taken

    for p in cand:
        # diagonal sum must be free
        if 2 * p in used_sums:
            continue

        # check pairwise sums against used_sums
        conflict = False
        for q in chosen:
            if q + p in used_sums:
                conflict = True
                break
        if conflict:
            continue

        # accept p
        for q in chosen:
            used_sums.add(q + p)
        used_sums.add(2 * p)
        chosen.append(int(p))

    return chosen


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step function with a low value of ``evaluate_sequence``.
    The method builds many random Sidon sets and keeps the one that yields
    the smallest score (equal weight on each selected index).  The routine
    respects the time budget and prints intermediate progress.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4800                                 # length used in the reference runs
    best_seq = None
    best_score = float("inf")
    best_m = 0

    # The theoretical maximum size of a Sidon set in [0,n‑1] satisfies
    # m(m+1)/2 ≤ 2·n‑1.  For n=4800 this bound is 138, and a score ≈1.01
    # is attainable if we reach that size.
    target_score = 1.503
    attempt = 0

    while time.time() < deadline:
        attempt += 1
        S = _build_sidon_set(n, rng)
        m = len(S)
        # ignore tiny sets that cannot beat the current best
        if m <= best_m:
            continue

        # equal weight on each selected index
        seq = np.zeros(n, dtype=float)
        w = 1.0 / m
        seq[S] = w

        # evaluate (exact) – cheap because the vector is very sparse
        score = evaluate_sequence(seq.tolist())

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_m = m
            print(
                f"[{time.time() - start_time:6.2f}s] attempt {attempt:4d}: "
                f"Sidon size={m:3d}, score={score:.6f}"
            )
            # stop early if we already beat the target
            if best_score < target_score:
                break

        # optional early stop if we hit the proven maximum size
        # (m*(m+1)//2 == 2*n-1 is a sufficient condition)
        if m * (m + 1) // 2 >= 2 * n - 1:
            # we cannot enlarge the set any further
            break

    # Fallback to uniform distribution if something went wrong
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] fallback uniform, score={best_score:.6f}")

    # final sanity checks and normalisation (the optimiser already enforces sum=1)
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=float)

    final_score = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final score = {final_score:.6f} "
        f"(Sidon size={best_m})"
    )
    return [float(x) for x in best_seq.tolist()]
