Propose a step function that (almost) minimises ``evaluate_sequence``.
====================================================================

Algorithm
---------
1. **Diverse initial pool** – generate a handful of handcrafted vectors
   (uniform, Gaussian, exponential, Dirichlet, sparse supports) and,
   if available, the previously known construction stored in
   ``height_sequence_1``.  All candidates are normalised to sum 1 and the
   best one (lowest true score) is kept.

2. **Smooth exponential‑gradient polishing** – instead of using only the
   single active shift (as in the previous code), we minimise a smooth
   approximation of the max‑convolution:
       f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
   The gradient of f_β can be computed via a single convolution with the
   soft‑max weights w_k ∝ exp(β·conv_k).  A mirror‑descent (exponentiated

   projection onto the probability simplex.  The smoothing parameter β
   starts small (≈1) and is increased slowly (×1.02 per iteration) so
   that the optimisation first sees a broad landscape and finally focuses
   on the true peak.  The step size η decays like 1/√t.

   only strict improvements.  This often squeezes a few extra digits.

5. **Safety normalisation** – clip to non‑negative, renormalise to sum 1

    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_full(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution x*y (length len(x)+len(y)-1) via FFT."""
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Simple simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw Dirichlet( conc,…,conc ) and normalise to sum 1."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Discrete Gaussian (σ≈n/6) normalised to sum 1."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Choose `m` positions uniformly, put equal mass on them."""
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the true max‑convolution objective
# ----------------------------------------------------------------------


def _expgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len_self: int,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on f(a)=max_k (a*a)[k].
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # sub‑gradient from the single shift attaining the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]  # gradient for active shift

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth (soft‑max) exponentiated‑gradient polishing
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len_self: int,
    beta0: float = 1.0,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation
        f_β(a) = (1/β) log Σ_k exp(β·(a*a)[k]).
    Uses a soft‑max weight vector and a multiplicative update.
    Returns best vector seen and its *exact* evaluate_sequence score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    t = 0
    beta = beta0

    # pre‑compute FFT length for the gradient convolution
    fft_len_grad = _next_pow2((2 * n - 1) + n - 1)

    while time.time() < deadline:
        t += 1

        # 1) self‑convolution (full)
        conv = _conv_self(a, fft_len_self)  # length 2n‑1

        # 2) soft‑max weights w_k ∝ exp(β·conv_k)
        scaled = beta * conv
        max_scaled = scaled.max()
        w = np.exp(scaled - max_scaled)  # stability
        w = w / w.sum()

        # 3) gradient = 2 * (w ⋆ a) where ⋆ is cross‑correlation
        a_rev = a[::-1]
        conv_grad = _conv_full(w, a_rev, fft_len_grad)
        grad = 2.0 * conv_grad[n - 1 : n - 1 + n]  # length n

        # 4) multiplicative (mirror‑descent) update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 1e-15, None)
        a = a / a.sum()

        # occasional exact evaluation
        if t % 100 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] SmoothExpGrad improve: iter={t}, score={sc:.6f}"
                )

        # gradually increase smoothing to focus on the true max
        beta = min(beta * 1.02, 50.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic peak‑reduction
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Transfer a modest fraction of the coefficient that most contributes to
    the current convolution peak towards the smallest coefficient.
    Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # modest move
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine builds a rich pool of starters, performs a long smooth‑gradient
    polishing phase, refines the result with the original max‑gradient method
    and finishes with a deterministic greedy local improvement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size (feel free to change, but 4800 works well under the time cap)
    # ------------------------------------------------------------------
    n = 4800
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse set of initial candidates
    # ------------------------------------------------------------------
    starts = []

    # 1) previously known construction (if provided)
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) classic handcrafted starters
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_gaussian_start(n))                  # bell
    starts.append(_exponential_start(n, r=0.985))     # stronger decay
    starts.append(_exponential_start(n, r=0.9995))    # gentle decay

    # 3) Dirichlet draws with a range of concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # 4) Sparse random supports
    for m in [400, 800, 1200, 2000]:
        for _ in range(2):
            starts.append(_sparse_random_start(n, m, rng))

    # 5) Pure exponential draw (continuous)
    exp_vec = rng.exponential(scale=1.0, size=n)
    starts.append(exp_vec / exp_vec.sum())

    # ------------------------------------------------------------------
    # Evaluate raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – long smooth exponential‑gradient polishing (~80% of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.80 * (deadline - start_time)
    smooth_seq, smooth_score = _smooth_expgrad_optimize(
        best_seq, phase1_deadline, start_time, fft_len_self, beta0=1.0, eta0=0.8
    )
    if smooth_score < best_score:
        best_score = smooth_score
        best_seq = smooth_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after smooth phase score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – fine‑grained max‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        expgrad_seq, expgrad_score = _expgrad_optimize(
            best_seq, deadline, start_time, fft_len_self, eta0=0.5
        )
        if expgrad_score < best_score:
            best_score = expgrad_score
            best_seq = expgrad_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after exp‑grad polish score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final safety normalisation / checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    return [float(x) for x in best_seq.tolist()]
