"""
Improved step‑function optimiser.

The algorithm combines a fine‑grained search over simple parametrised families
(geometric, mixtures of two geometric sequences) with a short but deeper
gradient‑based refinement and an aggressive greedy sweep.  The search is
repeated with several random seeds and the best sequence found is returned.
All steps respect the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a using the FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i ∝ r**i, normalised to sum 1."""
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    norm = 1.0 - r ** n                      # sum of the un‑normalised series
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    a /= a.sum()
    return a


def _mixture_exponential(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    """
    Mixture of a left‑hand geometric decay and a right‑hand (reversed) decay.
    The mixture weight w ∈ [0,1] controls the contribution of the left part.
    """
    left = _exponential_sequence(n, r1)
    right = np.flip(_exponential_sequence(n, r2))
    seq = w * left + (1.0 - w) * right
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 30000) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth (log‑sum‑exp) approximation of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it + 1.0)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Deterministic greedy sweep that moves a little mass from the most
    responsible index to the least responsible one."""
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # vector a_k = a[max_idx - i] (zero outside)
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # choose step size
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-30)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines a fine search over a few parametrised families,
    a short Adam refinement, exponentiated‑gradient descent and a long greedy
    reduction.  All stages respect the 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    final_n = 4096
    L = _next_pow2(2 * final_n - 1)

    # --------------------------------------------------------------
    # 1.  Coarse search: pure exponential families
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # fine grid for r – the most promising range is close to 1
    for r in np.linspace(0.990, 0.9999, 1500):
        cand = _exponential_sequence(final_n, r)
        conv = _conv_self(cand, L)
        score = 2.0 * final_n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand
            best_r = r
    print(f"[{time.time() - start_time:.2f}s] best geometric r={best_r:.6f}, obj={best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Random mixture of two exponentials – cheap Monte‑Carlo search
    # --------------------------------------------------------------
    for _ in range(300):
        r1 = np.random.uniform(0.90, 0.9999)
        r2 = np.random.uniform(0.90, 0.9999)
        w = np.random.uniform(0.3, 0.7)
        cand = _mixture_exponential(final_n, r1, r2, w)
        conv = _conv_self(cand, L)
        score = 2.0 * final_n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand
            best_params = (r1, r2, w)
    if best_seq is not None:
        print(f"[{time.time() - start_time:.2f}s] after mixture search best obj={best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Adam refinement on a smooth approximation of max(conv)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase1_deadline = time.time() + min(30.0, 0.12 * (deadline - time.time()))
        a_adam, score_adam = _run_adam_smoothing(
            best_seq,
            final_n,
            phase1_deadline,
            start_time,
            alphas=[0.2, 0.5, 1., 2., 5., 10., 20., 40., 80., 160.],
            lr0=0.6,
            iters_per_alpha=40000,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam
            print(f"[{time.time() - start_time:.2f}s] after Adam improve: obj={score_adam:.6f}")

    # --------------------------------------------------------------
    # 4.  Exponentiated‑gradient descent (many cheap iterations)
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + min(250.0, 0.30 * (deadline - time.time()))
        a_exp, score_exp = _run_exp_gradient(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            eta0=1.0,
        )
        if score_exp < best_score:
            best_score = score_exp
            best_seq = a_exp
            print(f"[{time.time() - start_time:.2f}s] after exp‑grad improve: obj={score_exp:.6f}")

    # --------------------------------------------------------------
    # 5.  Aggressive greedy reduction – runs until the deadline
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=8_000_000,
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy
            print(f"[{time.time() - start_time:.2f}s] after greedy improve: obj={score_greedy:.6f}")

    # --------------------------------------------------------------
    # 6.  Very short random‑restart hill‑climb (if any time left)
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # generate a random Dirichlet start, run a tiny exp‑grad burst
        cand = np.random.dirichlet(np.ones(final_n))
        sub_deadline = time.time() + 8.0
        a_tmp, score_tmp = _run_exp_gradient(
            cand,
            final_n,
            sub_deadline,
            start_time,
            eta0=0.8,
        )
        if score_tmp < best_score:
            best_score = score_tmp
            best_seq = a_tmp
            print(f"[{time.time() - start_time:.2f}s] random restart improve: obj={score_tmp:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    final_obj = 2.0 * final_n * _conv_self(best_seq, L).max()
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
