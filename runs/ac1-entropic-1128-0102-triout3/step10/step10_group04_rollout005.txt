def _run_softmax_gradient(a0, n, deadline, start_time, eta0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline:
        it += 1
        # compute self convolution via FFT
        L_self = _next_pow2(2 * n - 1)
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]   # conv length 2n-1
        # compute max and objective
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv  # sum a =1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)
        # compute smooth max using log-sum-exp
        # set tau = max_conv * alpha (e.g., 0.15) or bound.
        alpha = 0.2
        tau = max_conv * alpha if max_conv>0 else 1e-6
        # stable log-sum-exp
        # compute shifted conv = conv - max_conv
        shifted = conv - max_conv
        exp_shifted = np.exp(shifted / tau)
        sum_exp = exp_shifted.sum()
        smooth_max = max_conv + tau * np.log(sum_exp)
        # compute weights w = exp(shifted/tau)/sum_exp
        w = exp_shifted / sum_exp   # shape (2n-1)
        # Compute c = sum_k w_k a_{k-i}
        a_rev = a[::-1]
        L_wa = _next_pow2(len(w) + len(a_rev) - 1)  # = next_pow2(3n-2)
        W = np.fft.rfft(w, n=L_wa)
        A_rev = np.fft.rfft(a_rev, n=L_wa)
        conv_wa = np.fft.irfft(W * A_rev, n=L_wa)[:len(w) + len(a_rev) - 1]
        # slice out c_i
        start = n - 1
        c = conv_wa[start:start + n]
        # gradient ascent? we want to descent: gradient = 4n * c
        grad = 4.0 * n * c   # shape n
        # step size
        eta = eta0 / math.sqrt(it)
        # gradient descent update
        a = a - eta * grad
        # project onto simplex
        a = _project_simplex(a)
        # ensure no tiny negatives
        a = np.maximum(a, 0.0)
        # (Optionally renorm)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)
    return best_a, best_score

"""
propose_candidate
=================

A three‑stage optimiser for the “max‑self‑convolution” objective used by
``evaluate_sequence``.

1. **Initialisation** – try to reuse a previously discovered shape
   (the global ``height_sequence_1`` if present); otherwise start from a
   slightly perturbed uniform distribution on the probability simplex.

2. **Soft‑max gradient descent** – minimise the smooth surrogate
   `f(a)=2·n·τ·log(∑exp(conv/τ))` where `conv=a*a`.  The gradient of
   `f` is `4·n·c` with `c_i=∑_k w_k·a_{k-i}` and `w` the soft‑max
   weights.  Each iteration performs a projected gradient step onto the
   simplex.  The temperature `τ` is taken as a fixed fraction of the
   current maximum convolution value, so the algorithm automatically
   sharpens as it approaches a good solution.

3. **Deterministic greedy “peak‑flattening”** – identical to the
   previous routine: repeatedly move a tiny amount of mass from the
   index that contributes most to the current peak to an index that
   contributes least, updating the convolution incrementally.

The whole search respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence found as a plain Python
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    The returned array has length exactly 2*len(a)-1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # fallback – should never happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Stage 2 – soft‑max (smooth) gradient descent
# ----------------------------------------------------------------------


def _run_softmax_gradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    eta0: float = 5e-3,
    alpha_tau: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth surrogate
        f(a) = 2·n·τ·log(∑_k exp(conv[k]/τ)),
    where conv = a*a.  Returns the best vector seen and its (true)
    objective value 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- self‑convolution ----------
        L_self = _next_pow2(2 * n - 1)
        conv = _conv_self(a, L_self)          # length = 2*n-1
        max_conv = conv.max()

        # true objective (sum(a)=1 after projection)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] soft‑grad iter={it:,} "
                f"score={cur_score:.6f}"
            )

        # ---------- smooth max (log‑sum‑exp) ----------
        tau = max(alpha_tau * max_conv, 1e-12)          # temperature
        shifted = conv - max_conv                       # for stability
        exp_shifted = np.exp(shifted / tau)
        sum_exp = exp_shifted.sum()
        # smooth max value (not used directly for the update, only for diagnostic)
        smooth_max = max_conv + tau * np.log(sum_exp)

        # ---------- soft‑max weights ----------
        w = exp_shifted / sum_exp                        # length 2*n-1, sum(w)=1

        # ---------- compute c_i = Σ_k w_k a_{k-i} ----------
        a_rev = a[::-1]                                 # reversed a
        conv_wa = _conv_fft(w, a_rev)                   # length 3*n-2
        start = n - 1
        c = conv_wa[start : start + n]                  # length n

        # ---------- gradient of the smooth objective ----------
        # ∇f = 4·n·c   (since sum(a)=1 after projection)
        grad = 4.0 * n * c

        # ---------- step size ----------
        eta = eta0 / math.sqrt(it)

        # ---------- projected gradient step ----------
        a = a - eta * grad
        a = _project_simplex(a)                         # back to simplex

        # ---------- tiny re‑normalisation safeguard ----------
        total = a.sum()
        if total > 0.0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stage 3 – deterministic greedy reduction (unchanged)
# ----------------------------------------------------------------------


def _run_greedy_reduction(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    cur_best_score: float,
    cur_best_seq: np.ndarray,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the index that most
    contributes to the current peak to an index that contributes least,
    updating the full convolution incrementally (O(n) per move).
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that take part in this maximum
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and every other candidate
        partner_vals = a[max_idx - idx]

        # choose destination with smallest partner value (and not src)
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst : dst + n] += 2.0 * eps * a_old
        conv[src : src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                f"score={cur_score:.6f}"
            )

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------

    """

    greedy flattening) within the allotted time budget.

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    best_seq = a0.copy()

    conv0 = _conv_self(best_seq, _next_pow2(2 * n - 1))

            n,
            deadline,
            start_time,
            best_score,
            best_seq,

        )

            best_score = scr
            best_seq = seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] final greedy improve: score={scr:.6f}"

    # --------------------------------------------------------------
    while time.time() < deadline - 2.0:
        pert = best_seq + 0.005 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)

        tmp_deadline = time.time() + 1.5
        seq, scr = _run_softmax_gradient(
            pert, n, tmp_deadline, start_time, eta0=5e-3, alpha_tau=0.20

            best_seq = seq.copy()

    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)

        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # return as plain Python list
    return [float(x) for x in best_seq]
