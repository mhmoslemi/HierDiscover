"""
Propose a step function (non‑negative vector) that minimizes
``evaluate_sequence``.

The algorithm consists of three stages:

1. **Diverse initial pool** – several handcrafted and random starts
   (uniform, Gaussian, Dirichlet, sparse supports, exponential).

2. **Smooth exponentiated‑gradient optimisation** – instead of using the
   sub‑gradient of a single active convolution shift, we minimise a
   smooth approximation of the max‑convolution:
   f(a)= (1/β)·log Σ_k exp(β·(a∗a)_k).  Its gradient is a weighted sum
   of all shift‑gradients, which is computed efficiently via FFTs.
   Multiplicative (KL‑mirror) updates keep the iterate on the simplex.
   The temperature β is slowly increased (continuation) and the step
   size decays as η₀/√t.

3. **Deterministic greedy peak‑reduction** – after the long run we
   locally move a small fraction of mass from the coefficient that
   contributes most to the current peak towards the smallest coefficient.
   Only strict improvements are accepted.

Random‑restart bursts are interleaved while time permits.  The whole
procedure respects the 1000 s wall‑clock limit.  The returned vector is
clipped, renormalised to sum 1 and converted to plain Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_full(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution x*y using FFT of length ``fft_len``.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc` and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian, normalised to sum‑1."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose ``m`` indices, give each equal mass 1/m, zero elsewhere.
    """
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Smooth‑gradient optimisation (exponentiated‑gradient on a softmax of the
# convolution) ---------------------------------------------------------
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len_self: int,
    fft_len_grad: int,
    beta_max: float = 30.0,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the smoothed objective

        f(a) = (1/β)·log Σ_k exp(β·(a∗a)[k]).

    Returns the best vector seen and its exact score (using the original
    ``evaluate_sequence`` objective).
    """
    n = a0.shape[0]
    a = np.maximum(a0, 1e-15)          # avoid exact zeros
    a = a / a.sum()                    # project onto simplex

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # ----- convolution a∗a ------------------------------------------------
        Af = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(Af * Af, n=fft_len_self)
        conv = conv[: 2 * n - 1]          # linear convolution length

        # ----- softmax weights ------------------------------------------------
        # continuation on beta: start cold (beta≈0) and increase to beta_max
        beta = min(beta_max, 0.01 + beta_max * (t / 5000.0))
        m = conv.max()
        exp_terms = np.exp(beta * (conv - m))
        w = exp_terms / exp_terms.sum()   # shape (2n‑1,)

        # ----- gradient of the smoothed max ----------------------------------
        a_rev = a[::-1]                   # reversed copy
        # convolution w * a_rev  (length 3n‑2)
        conv_rev_full = _conv_full(w, a_rev, fft_len_grad)
        # relevant slice corresponds to indices n‑1 … n‑1+n‑1
        g = 2.0 * conv_rev_full[n - 1 : n - 1 + n]   # length n

        # ----- current exact score (sum=1 ⇒ 2·n·max) -----------------------
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] smooth‑expgrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ----- multiplicative update (mirror descent on KL) -----------------
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # ----- re‑normalise --------------------------------------------------
        s = a.sum()
        if s <= 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- occasional exact evaluation (guard against drift) ------------
        if t % 300 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] smooth‑expgrad exact improve: iter={t:,}, score={sc:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic peak‑reduction (same as the previous version)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Shift a modest fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # transfer a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(
                f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}"
            )

    return a, best_score

# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns a list of floats (the best found sequence).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # problem size -------------------------------------------------------
    n = 4800
    fft_len_self = _next_pow2(2 * n - 1)          # for a∗a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w∗a_rev

    # ------------------------------------------------------------------
    # Build a diverse set of initial candidates
    # ------------------------------------------------------------------
    starts = []

    # 1) previously known construction, if it exists
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation to required length
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) handcrafted starters
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_gaussian_start(n))                  # bell‑shaped
    starts.append(_exponential_start(n, r=0.9995))    # very gentle decay
    starts.append(_exponential_start(n, r=0.985))     # stronger decay

    # 3) Dirichlet draws with several concentration parameters
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # 4) Sparse random supports – try a few densities
    for m in [400, 800, 1200, 2000]:
        for _ in range(2):
            starts.append(_sparse_random_start(n, m, rng))

    # 5) pure exponential random draw (continuous)
    exp_vec = rng.exponential(scale=1.0, size=n)
    starts.append(exp_vec / exp_vec.sum())

    # ------------------------------------------------------------------
    # Evaluate the raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – long smooth‑gradient polishing (≈85 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.85 * (deadline - start_time)
    polished_seq, polished_score = _smooth_expgrad_optimize(
        best_seq,
        phase1_deadline,
        start_time,
        fft_len_self,
        fft_len_grad,
        beta_max=30.0,
        eta0=0.8,
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after smooth‑grad score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – cheap random‑restart bursts (≈10 % of remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # fresh random start (sparse or Dirichlet)
        if rng.random() < 0.5:
            m = int(rng.integers(300, 2500))
            cand_start = _sparse_random_start(n, m, rng)
        else:
            conc = 10 ** rng.uniform(-1.5, 1.0)   # roughly 0.03 … 10
            cand_start = _dirichlet_start(n, conc=conc, rng=rng)

        sub_deadline = time.time() + 5.0
        cand_seq, cand_score = _smooth_expgrad_optimize(
            cand_start,
            sub_deadline,
            start_time,
            fft_len_self,
            fft_len_grad,
            beta_max=20.0,
            eta0=0.5,
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from restart score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – final deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    # return plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
