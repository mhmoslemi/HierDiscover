```
    return best_a, best_score
```

assistantfinal**Solution Overview**

The evaluation function is  

\[
\operatorname{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

Because the objective is homogeneous we may keep the vector on the
probability simplex \(\sum a_i=1\).  
For a fixed vector the convolution `a*a` is the (aperiodic) autocorrelation
of `a`.  Minimising the maximum autocorrelation entry is a non‑convex
problem; a pure “max‑gradient’’ (as used previously) often gets stuck in a
local plateau.

The new optimiser uses a **smooth approximation of the max**:

\[
w_k(\beta)=\frac{e^{\beta (c_k-M)}}{\sum_j e^{\beta (c_j-M)}},
\qquad 
c_k=(a*a)_k,\; M=\max_k c_k .
\]

For \(\beta=0\) the weights are uniform, for large \(\beta\) they concentrate
on the current peak.  
The (sub‑)gradient of the smoothed objective is

\[
\nabla = 2\,\bigl(w\star a_{\text{reversed}}\bigr)_{[n-1:n-1+n]} .
\]

All convolutions are performed with FFTs, yielding an
\(O(n\log n)\) per‑iteration cost (≈ 1 ms for \(n=4096\)).  
The optimiser proceeds in three phases

1. **Coarse optimisation** on a short vector (1024 entries) – cheap, easy to
   explore many shapes.
2. **Upsampling** to the full size (4096) and a long smoothed‑gradient run.
   `β` is increased slowly from 1 to a large value (≈ 5000), “softening’’ the
   max at the beginning and sharpening it later.
3. **Random‑restart polishing**: occasional Gaussian perturbations followed
   by a short gradient burst, keeping the best sequence seen.

The whole routine respects the 1000 s budget, prints progress, and finally
returns the best vector found as a plain Python list.

```python
"""
Propose a non‑negative step function (vector) that minimises
`evaluate_sequence`.  The algorithm uses a smoothed‑max gradient descent
with an exponential “temperature’’ schedule, a coarse‑to‑fine optimisation
scheme and occasional random restarts.  All convolutions are performed
via FFT for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    The result is a real array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * n - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (non‑negative, sum=1). """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # fallback – should never happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Smoothed‑max gradient step
# ----------------------------------------------------------------------


def _smooth_gradient_step(a: np.ndarray,
                          beta: float,
                          lr: float,
                          L_conv: int,
                          L_corr: int) -> np.ndarray:
    """
    Perform one gradient descent step on the smoothed max:
        w_k ∝ exp(beta * (conv_k - max_conv))
        grad = 2 * (w ⋆ a_rev)[n-1 : n-1+n]
    Afterwards project onto the simplex.
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _conv_self(a, L_conv)
    max_c = conv.max()

    # 2) soft‑max weights (stable)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient via correlation w ⋆ a_rev
    #    we need the slice [n‑1 … n‑1+n] of the full correlation
    w_fft = np.fft.rfft(w, n=L_corr)
    a_rev_fft = np.fft.rfft(a[::-1], n=L_corr)
    corr_full = np.fft.irfft(w_fft * a_rev_fft, n=L_corr)
    grad = 2.0 * corr_full[n - 1:n - 1 + n]

    # 4) gradient step + projection
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_proj = _project_simplex(a_new)
    return a_proj, conv.max()


# ----------------------------------------------------------------------
# Full smoothed‑max optimiser
# ----------------------------------------------------------------------


def _run_smooth_gradient(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         lr0: float = 0.5,
                         beta_start: float = 1.0,
                         beta_max: float = 5000.0,
                         beta_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply _smooth_gradient_step until the deadline expires.
    Returns the best vector seen and its objective value (2·n·max_conv).
    """
    a = a0.copy()
    n = a.shape[0]
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    best_a = a.copy()
    # initial score
    cur_max = _conv_self(a, L_conv).max()
    best_score = 2.0 * n * cur_max
    it = 0
    beta = beta_start

    while time.time() < deadline:
        it += 1
        # learning‑rate schedule (classic 1/√t)
        lr = lr0 / math.sqrt(it)

        a, cur_max = _smooth_gradient_step(a, beta, lr, L_conv, L_corr)

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth iter={it:,} "
                  f"score={cur_score:.6f}")

        # increase temperature (β) geometrically, clamp at β_max
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to build simple initial shapes
# ----------------------------------------------------------------------


def _initial_shapes(n: int):
    """Return a list of plausible starting vectors (all normalised to sum=1)."""
    shapes = []

    # 1) uniform
    shapes.append(np.full(n, 1.0 / n))

    # 2) linearly decreasing (a_i ∝ n‑i)
    lin = np.arange(n, 0, -1, dtype=float)
    shapes.append(lin / lin.sum())

    # 3) sqrt‑law (a_i ∝ 1/√(i+1))
    sqrt_shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    shapes.append(sqrt_shape / sqrt_shape.sum())

    # 4) exponential decay (q close to 1)
    q = 0.9995
    exp_shape = (1 - q) * (q ** np.arange(n, dtype=float))
    shapes.append(exp_shape / exp_shape.sum())

    # 5) U‑shaped (large at both ends, small in the centre)
    idx = np.arange(n, dtype=float)
    u_shape = 1.0 / (idx + 1) + 1.0 / (n - idx)
    shapes.append(u_shape / u_shape.sum())

    # 6) a small random perturbation of the uniform vector
    rnd = np.full(n, 1.0 / n) + 0.02 * np.random.randn(n)
    rnd = np.clip(rnd, 0.0, None)
    shapes.append(rnd / rnd.sum())

    return shapes


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the surrounding harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Coarse optimisation on a smaller problem (n_small = 1024)
    # ------------------------------------------------------------------
    n_small = 1024
    init_vecs = _initial_shapes(n_small)

    # pick the most promising start after a very short run
    best_small = None
    best_small_score = float('inf')
    for vec in init_vecs:
        if time.time() >= deadline:
            break
        # give each candidate a few hundred cheap iterations (≈ 2 s)
        tmp_deadline = time.time() + min(2.0, (deadline - time.time()) * 0.02)
        cand, scr = _run_smooth_gradient(
            vec, tmp_deadline, start_time,
            lr0=0.6, beta_start=1.0, beta_max=2000.0, beta_factor=1.0015
        )
        if scr < best_small_score:
            best_small_score = scr
            best_small = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] coarse start improved: score={scr:.6f}")

    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # 2) Upsample to the full size (n = 4096) and run the main optimisation
    # ------------------------------------------------------------------
    n_full = 4096
    best_full = np.interp(
        np.linspace(0, 1, n_full),
        np.linspace(0, 1, n_small),
        best_small,
    )
    best_full = np.maximum(best_full, 0.0)
    best_full = best_full / best_full.sum()

    # main smooth‑gradient phase – consumes the majority of the budget
    if time.time() < deadline:
        remaining = deadline - time.time()
        # leave a small buffer for the random‑restart polishing stage
        main_deadline = time.time() + max(0.0, remaining - 20.0)
        best_full, best_score = _run_smooth_gradient(
            best_full, main_deadline, start_time,
            lr0=0.5, beta_start=1.0, beta_max=5000.0, beta_factor=1.001
        )
        print(f"[{time.time() - start_time:.2f}s] main optimisation final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Tiny random‑restart polishing (the last ~20 s)
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # add a small Gaussian jitter, re‑normalise
        pert = best_full + 0.015 * np.random.randn(n_full)
        pert = np.clip(pert, 0.0, None)
        pert /= pert.sum()

        # short burst (≈ 1 s) of gradient descent
        tmp_deadline = time.time() + 1.0
        cand, scr = _run_smooth_gradient(
            pert, tmp_deadline, start_time,
            lr0=0.4, beta_start=10.0, beta_max=5000.0, beta_factor=1.002
        )
        if scr < best_score:
            best_score = scr
            best_full = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑restart improved: score={scr:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (objective is scale‑invariant)
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_full, 0.0)
    if final_seq.sum() > 0.0:
        final_seq = final_seq / final_seq.sum()
    else:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)

    # Return a plain Python list as required by the harness
    return [float(x) for x in final_seq]
```
