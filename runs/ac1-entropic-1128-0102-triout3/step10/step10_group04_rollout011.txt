assistantfinal**Solution Overview**  
The optimizer works in three stages:

1. **Initialisation** – a small collection of deterministic shapes (uniform,
   linear, √‑law, exponential) and the previously discovered construction are
   normalised to a probability simplex.  The best of these is kept as a start
   point.

2. **Smoothed max‑gradient descent** – instead of using the hard maximum
   `max(conv)` we minimise the smooth approximation  

   \[
   f_\beta(a)=\frac{1}{\beta}\log\Bigl(\sum_{k}e^{\beta\,
   (a*a)_k}\Bigr) .
   \]

   For a given temperature β the gradient is  

   \[
   \nabla f_\beta(a)=2\;p\star a^{\text{rev}},
   \qquad  
   p_k=\frac{e^{\beta (a*a)_k}}{\sum_j e^{\beta (a*a)_j}},
   \]

   where “★’’ denotes the (linear) correlation.  Using FFTs the convolution
   and the correlation are computed in O(n log n).  β is increased slowly from
   a small value (smooth) to a large value (close to the true max), while a
   decreasing stepsize `η_t = η0/√t` is used.  After each update the vector is
   projected back onto the probability simplex.

3. **Fine‑grained greedy flattening** – the same deterministic “move a tiny
   amount of mass from the most responsible index to the least responsible
   one’’ routine from the previous state‑of‑the‑art is applied to shave the
   remaining few 10⁻³ off the objective.

All stages respect the 1000 s budget.  The final sequence is returned as a plain
list of floats; any scaling of the vector leaves the objective unchanged, so we
normalise it to sum = 1 before returning.

```python
"""
Optimiser for the step‑function evaluation problem.

The algorithm consists of:
1. A few deterministic initial shapes (uniform, linear, sqrt‑law, exponential)
   plus an optional historic construction.
2. Smoothed max‑gradient descent on the self‑convolution peak; the gradient
   is obtained via a soft‑max weighting of the convolution entries and a
   correlation computed with FFTs.  The temperature β is increased gradually.
3. A deterministic greedy “peak‑flattening’’ phase (identical to the previous
   best implementation) for final polishing.

The whole routine fits into a 1000 s time budget and returns the best
non‑negative sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # fallback – shouldn't happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Phase 2 – Smoothed (soft‑max) gradient descent on the max convolution
# ----------------------------------------------------------------------
def _run_softmax_gd(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    beta_start: float = 5.0,
                    beta_max: float = 80.0,
                    eta0: float = 0.05,
                    beta_ramp_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation of max(conv) using a soft‑max
    weighting.  Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    it = 0

    L_self = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)          # for correlation (p * a)

    while time.time() < deadline:
        it += 1

        # ----- self‑convolution -----
        conv = _conv_self(a, L_self)               # length 2·n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] soft‑max GD iter={it:,} "
                  f"score={cur_score:.6f}")

        # ----- temperature schedule (smooth → sharp) -----
        beta = beta_start + (beta_max - beta_start) * min(1.0,
                                                          it / beta_ramp_iters)

        # ----- soft‑max weights over the convolution entries -----
        # shift by max to keep numbers stable
        exp_terms = np.exp(beta * (conv - cur_max))
        p = exp_terms / exp_terms.sum()                # length 2·n‑1

        # ----- correlation p ★ a (linear, not circular) -----
        Pfft = np.fft.rfft(p, n=L_corr)
        Afft = np.fft.rfft(a, n=L_corr)
        corr = np.fft.irfft(Pfft * np.conj(Afft), n=L_corr)
        corr = corr[: (2 * n - 1) + n - 1]              # keep linear part (len = 3·n‑2)

        # gradient: 2 * corr[n‑1‑j] for j = 0..n‑1
        grad = 2.0 * corr[n - 1 :: -1][:n]

        # ----- stepsize (decreasing) -----
        eta = eta0 / math.sqrt(it)

        # ----- gradient step + projection onto the simplex -----
        a = a - eta * grad
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Deterministic greedy reduction of the peak (unchanged)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the index that most
    contributes to the current peak to an index that contributes least,
    updating the full convolution incrementally.  A full recomputation is
    performed every few thousand steps to remove numerical drift.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and every other candidate
        partner_vals = a[max_idx - idx]

        # choose destination with smallest partner value (and not src)
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score

    while time.time() < deadline - 2.0:
        pert = best_seq + 0.01 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)
        pert /= pert.sum()

        tmp_deadline = time.time() + 2.0
        seq, scr = _run_softmax_gd(
            pert,
            n,
            tmp_deadline,
            start_time,
            beta_start=10.0,
            beta_max=80.0,
            eta0=0.04,
        )
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑restart improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (objective is homogeneous, any scaling works)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return as plain Python list
    return [float(x) for x in best_seq]
```
