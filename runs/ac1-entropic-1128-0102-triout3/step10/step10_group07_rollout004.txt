def _smooth_expgrad_optimize(a0, deadline, start_time, fft_len, eta0=0.5, alpha_start=50.0, alpha_end=2000.0, alpha_ramp_iters=2000):

    a = np.maximum(a0, 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline:
        t += 1
        # conv via FFT
        Af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)
        conv = conv[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth expgrad improve: iter={t:,}, score={cur_score:.6f}")
        # compute alpha
        if t <= alpha_ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * (t / alpha_ramp_iters)
        else:
            alpha = alpha_end
        # softmax
        max_conv = cur_max
        weights = np.exp(alpha * (conv - max_conv))
        p = weights / weights.sum()
        # gradient via correlation
        a_rev = a[::-1]
        L2 = _next_pow2(len(p) + len(a_rev) - 1)
        P_fft = np.fft.rfft(p, n=L2)
        A_rev_fft = np.fft.rfft(a_rev, n=L2)
        corr_full = np.fft.irfft(P_fft * A_rev_fft, n=L2)
        start = n - 1
        g = 2.0 * corr_full[start : start + n]
        # multiplicative update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        # occasional exact eval with evaluate_sequence for verification
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

"""
Smooth‑exponential‑gradient optimiser for the max‑convolution objective.

Algorithm
---------
1. Build a diverse pool of start vectors (uniform, Gaussian, exponentials,
   Dirichlet draws, sparse supports, triangular, …) and keep the best.
2. Run a long “soft‑max’’ exponentiated‑gradient phase.
   * Instead of using the sub‑gradient from the single active shift,
     we form a smooth approximation of the max via a soft‑max
     (temperature α).  The resulting gradient is a weighted sum of all
     shift‑gradients and is obtained efficiently by a single FFT‑based
     correlation.
   * Multiplicative updates on the simplex (a_i ← a_i·exp(‑η·g_i))
     with a diminishing step size η = η₀ / √t.
   * α is increased slowly (α≈50 → α≈2000) so that early iterations
     see a smooth landscape and later iterations concentrate on the
     true maximum.
3. While time permits perform cheap random‑restart bursts (short
   smooth‑gradient runs) to escape possible basins.
4. Finish with the deterministic greedy peak‑reduction from the
   previous state‑of‑the‑art code.
5. Clip to [0,∞), renormalise to sum 1 and return.

The routine respects the 1000 s wall‑clock limit and prints progress
messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc` and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose `m` indices, give each equal mass, zero elsewhere."""
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


def _triangular_start(n: int) -> np.ndarray:
    """Symmetric triangular shape (1,…,mid,…,1) normalised."""
    if n % 2 == 0:
        half = n // 2
        left = np.arange(1, half + 1, dtype=float)
        vec = np.concatenate([left, left[::-1]])
    else:
        half = n // 2
        left = np.arange(1, half + 1, dtype=float)
        middle = np.array([half + 1], dtype=float)
        vec = np.concatenate([left, middle, left[::-1]])
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             fft_len: int,
                             eta0: float = 0.5,
                             alpha_start: float = 50.0,
                             alpha_end: float = 2000.0,
                             ramp_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max‑convolution objective.
    Returns the best vector seen and its exact score.
    """
    # protect against zeros and normalise to the simplex
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    it = 0

    # pre‑allocate FFT buffers sizes
    corr_fft_len = _next_pow2(3 * n - 2)          # enough for p (2n‑1) ⊛ a_rev (n)

    while time.time() < deadline:
        it += 1

        # ----- 1. evaluate current objective -----
        conv = _conv_self(a, fft_len)                # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth‑expgrad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # ----- 2. compute smoothed gradient -----
        # temperature schedule
        if it <= ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * it / ramp_iters
        else:
            alpha = alpha_end

        # soft‑max over the convolution values
        max_conv = cur_max
        # all entries ≤ max_conv, so exponentials stay ≤ 1 → no overflow
        exp_weights = np.exp(alpha * (conv - max_conv))
        p = exp_weights / exp_weights.sum()          # length 2n‑1, sums to 1

        # correlation p ⋆ a (equivalently 2·corr(p,a) = gradient)
        a_rev = a[::-1]
        P_fft = np.fft.rfft(p, n=corr_fft_len)
        A_rev_fft = np.fft.rfft(a_rev, n=corr_fft_len)
        corr_full = np.fft.irfft(P_fft * A_rev_fft, n=corr_fft_len)

        # extract the n entries needed: corr_full[n‑1 : n‑1+n]
        start = n - 1
        g = 2.0 * corr_full[start:start + n]        # gradient of the smoothed objective

        # ----- 3. multiplicative update on the simplex -----
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s <= 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact check (guard against numerical drift)
        if it % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak reduction (same as before)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a modest fraction of the coefficient that contributes most
    to the current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # move a modest fraction
        if eps <= 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

# Main driver – the function that will be called by the judge
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine combines a rich initial pool, a long smooth‑exp‑gradient
    polishing phase, occasional random‑restart bursts and a final
    deterministic greedy refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin so we never overrun the hard limit
    deadline = start_time + budget_s - 0.5

    # problem size -------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)          # used for a*a convolution

    # ------------------------------------------------------------------
    # 1) Build a diverse set of initial candidates
    # ------------------------------------------------------------------
    starts = []

    # use the previously known construction if it exists
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation to required length
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # handcrafted starters
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_gaussian_start(n))                  # bell‑shaped
    starts.append(_exponential_start(n, r=0.9995))    # very gentle decay
    starts.append(_exponential_start(n, r=0.985))     # stronger decay
    starts.append(_triangular_start(n))               # symmetric triangle

    # Dirichlet draws (several concentrations)
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # sparse random supports – several densities
    for m in [400, 800, 1200, 2000]:
        for _ in range(2):
            starts.append(_sparse_random_start(n, m, rng))

    # a pure exponential (continuous) draw
    exp_vec = rng.exponential(scale=1.0, size=n)
    starts.append(exp_vec / exp_vec.sum())

    # ------------------------------------------------------------------
    # 2) Evaluate raw starts, keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(starts):
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – long smooth‑exp‑gradient polishing (≈ 80 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.80 * (deadline - start_time)
    polished_seq, polished_score = _smooth_expgrad_optimize(
        best_seq,
        phase1_deadline,
        start_time,
        fft_len,
        eta0=0.8,
        alpha_start=50.0,
        alpha_end=2000.0,
        ramp_iters=3000
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after long polish "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – cheap random‑restart bursts (≈ 15 % of remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # draw a fresh random start (sparse or Dirichlet)
        if rng.random() < 0.5:
            m = int(rng.integers(300, 2500))
            cand_start = _sparse_random_start(n, m, rng)
        else:
            conc = 10 ** rng.uniform(-1.5, 1.0)   # 0.03 … 10
            cand_start = _dirichlet_start(n, conc=conc, rng=rng)

        # short run – about 5 seconds
        sub_deadline = time.time() + 5.0
        cand_seq, cand_score = _smooth_expgrad_optimize(
            cand_start,
            sub_deadline,
            start_time,
            fft_len,
            eta0=0.5,
            alpha_start=30.0,
            alpha_end=1500.0,
            ramp_iters=800
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from restart "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase 3 – final deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq,
            deadline,
            start_time,
            max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final safety round‑trip
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    # return a plain Python list, as required by the judge
    return [float(x) for x in best_seq.tolist()]
