def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step function with a low evaluation score using
    multiple random restarts, a smooth log‑sum‑exp surrogate of the
    max‑convolution, Adam‑style projected gradient descent and a
    final light simulated‑annealing polish.  The algorithm works on
    the probability simplex (non‑negative entries, sum = 1).  It
    prefers a larger number of steps (default 8192) to get a finer
    discretisation of the optimal continuous shape.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # problem size
    n = 8192

    best_a = None
    best_score = float('inf')

    # Pre‑compute x positions for arcsine seed
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    # main loop: keep launching restarts while time permits
    restart = 0
    while time.time() < deadline:
        # --- initialise a candidate -----------------------------------
        # random mixture of arcsine and uniform
        alpha = rng.uniform(0.25, 0.55)      # exponent for arcsine‑like term

        # small random perturbation from a Dirichlet too
        if rng.random() < 0.3:
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
            dirich_pt = rng.dirichlet(np.ones(n))
            a0 = 0.5 * a0 + 0.5 * dirich_pt
            a0 = a0 / a0.sum()

        # --- optimizer ------------------------------------------------
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        beta_smooth = 1.0
        beta_factor = 1.005     # geometric increase of the smoothing parameter
        beta_max = 300.0

        iter_num = 0

            iter_num += 1

            conv = _conv_self(a)      # length 2n-1
            # objective (sum of a is 1 after projection)
            cur_score = 2.0 * n * conv.max()

                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] restart={restart} iter={iter_num} new best score={cur_score:.6f}")

            conv_max = conv.max()

def _conv_self(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _conv_fft(a, b):
    n = a.shape[0] + b.shape[0] - 1
    L = _next_pow2(n)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:n]
    return conv

def _project_simplex(v):
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

"""
Propose a low‑objective step function.

The algorithm works on the probability simplex (non‑negative entries,
summing to 1) and consists of three phases:

1. **Random restarts** – each start is a mixture of an arcsine‑like
   distribution (parameter α∈[0.25,0.55]), a uniform component and a
   Dirichlet random vector.  This gives a diverse collection of initial
   shapes.

2. **Smoothed Adam descent** – we minimise a smooth surrogate of the
   max‑convolution obtained by the log‑sum‑exp trick.
   For a vector `a` we compute the self‑convolution `c = a*a`,
   weights `w ∝ exp(β·c)` and the gradient
   `∇ = 2·(a_rev ⊛ w)`.  The smoothing parameter β is increased
   geometrically so that the optimisation first sees a smooth landscape
   and finally focuses on the true maximum.  After each step we project
   back onto the simplex.

3. **Light annealing polish** – a few hundred thousand tiny mass‑transfer
   moves are performed to escape any remaining shallow local minima.

A relatively large problem size (`n = 8192`) is used; the discretisation
error of the optimal continuous shape decreases with *n*, allowing the
objective to drop below the target 1.503.  All intermediate values are
kept inside the allowed range, and the final vector is normalised before
returning it as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary 1‑D arrays using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    L = _next_pow2(n)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[:n]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(seq)²."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # problem dimension – a larger n gives a finer discretisation
    n = 8192

    # keep the best solution seen so far
    best_a = None
    best_score = float("inf")

    # pre‑compute the mid‑bin positions (used for arcsine‑like seeds)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # values in (0,1)

    restart = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) initialise a random candidate
        # ------------------------------------------------------------------
        alpha = rng.uniform(0.25, 0.55)                     # arcsine exponent
        w_arcsine = (x * (1.0 - x) + 1e-30) ** (-alpha)    # arcsine‑like weights
        w_uniform = np.ones_like(w_arcsine)
        mix = rng.random()
        w = mix * w_arcsine + (1.0 - mix) * w_uniform      # blend arcsine & uniform
        a0 = w / w.sum()                                   # normalise to sum‑1

        # occasionally mix with a Dirichlet sample for extra diversity
        if rng.random() < 0.3:
            dirich = rng.dirichlet(np.ones(n))
            a0 = 0.5 * a0 + 0.5 * dirich
            a0 = a0 / a0.sum()

        # ------------------------------------------------------------------
        # 2) Adam‑style projected gradient descent on a smooth surrogate
        # ------------------------------------------------------------------
        a = a0.copy()
        m = np.zeros_like(a)           # first‑moment estimate
        v = np.zeros_like(a)           # second‑moment estimate
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.3                       # base learning rate
        beta_smooth = 1.0
        beta_factor = 1.005
        beta_max = 300.0
        it = 0

        while time.time() < deadline:
            it += 1

            # self‑convolution and current objective
            conv = _conv_self(a)                     # length 2n‑1
            cur_score = 2.0 * n * conv.max()         # sum(a) == 1 after projection
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] restart={restart} "
                    f"iter={it:,} new best score={cur_score:.6f}"
                )

            # ----- smooth approximation of the max (log‑sum‑exp) -----
            cmax = conv.max()
            # exponentiate after shifting by the maximum to avoid overflow
            exp_vals = np.exp(beta_smooth * (conv - cmax))
            w_soft = exp_vals / exp_vals.sum()       # probability weights

            # ----- gradient of the smooth surrogate -----
            a_rev = a[::-1]                          # reversed a (view, not copy)
            conv_rev = _conv_fft(a_rev, w_soft)      # length 3n‑2
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]  # slice of length n

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_simplex(a)                  # stay on the simplex

            # increase smoothing (focus more on the true max)
            beta_smooth = min(beta_smooth * beta_factor, beta_max)

        restart += 1

    # ------------------------------------------------------------------
    # 3) Light simulated‑annealing polish (if time permits)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = _objective(best_a)

    if time.time() < deadline:
        a = best_a.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        anneal_rng = np.random.default_rng(seed + 12345)
        total_moves = 200_000
        for move in range(1, total_moves + 1):
            if time.time() >= deadline:
                break

            # simple exponential cooling
            T = 0.02 * (1e-5 / 0.02) ** (move / total_moves)

            # choose a source index proportionally to its current mass
            src = anneal_rng.choice(n, p=a)
            dst = anneal_rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            # move a small fraction (up to 15 %) of the source mass to dst
            eps_move = a[src] * anneal_rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            new_score = _objective(a_prop)
            delta = new_score - cur_score
            if delta < 0.0 or math.exp(-delta / T) > anneal_rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] anneal step={move:,} "
                        f"best={new_score:.6f}"
                    )

        if best_local_score < best_score:
            best_score = best_local_score
            best_a = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to a plain Python list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a]
