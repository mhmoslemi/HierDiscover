"""
Propose a low‑evaluation step‑function.

The algorithm works in three phases

1. **Arcsine family warm‑start** –  search a short grid of
   exponents α for the discrete arcsine density
   a_i ∝ (x_i·(1−x_i))^{−α} (x_i centered in each bin) and pick the
   best‑scoring vector (objective is evaluated with an FFT‑based
   convolution).

2. **Smoothed‑max gradient descent** –  minimise a smooth approximation
   of  max(conv(a,a)).  For a smoothing parameter β we use the
   soft‑max
        p_k = exp(β·conv_k) / Σ_l exp(β·conv_l) .
   The gradient of the smoothed objective
        φ(a)= (1/β)·log Σ_k exp(β·conv_k)
   is
        ∇φ = 2·(a_rev * p)_{n-1…2n-2}
   where a_rev is a reversed, and * denotes linear convolution.
   The gradient step is performed with the Adam optimiser and the
   iterate is projected onto the probability simplex (non‑negative,
   sum = 1).  β is increased gradually (0.5 → 1 → 2 → 5 → …) so that
   the optimisation first sees a smooth landscape and later refines
   the true max‑objective.

3. **Return the best iterate** –  after the time budget the best
   sequence seen (lowest exact objective) is normalised and returned
   as a plain list of floats.

All heavy work (convolutions, FFTs) is O(n log n) and the length is
chosen as n=8192 – large enough to approach the continuous optimum
while still allowing many optimisation steps inside the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real‑FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:       # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


def _gradient_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) objective.
    a  – current probability vector (sum = 1)
    p  – soft‑max probabilities (same length as conv(a,a))
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    L = _next_pow2(3 * n - 2)           # length needed for linear conv
    A_rev = np.fft.rfft(a_rev, n=L)
    P = np.fft.rfft(p, n=L)
    conv_rev = np.fft.irfft(A_rev * P, n=L)   # linear convolution
    # slice n‑1 … 2n‑2 (inclusive) – exactly n entries
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad

    else:
