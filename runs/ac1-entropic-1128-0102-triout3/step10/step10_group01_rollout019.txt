def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length L."""
    n = a.shape[0]
    a_pad = np.zeros(L, dtype=np.float64)
    a_pad[:n] = a
    Af = np.fft.rfft(a_pad)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv, Af


def _conv(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*b using FFT of length L."""
    na, nb = a.shape[0], b.shape[0]
    a_pad = np.zeros(L, dtype=np.float64)
    b_pad = np.zeros(L, dtype=np.float64)
    a_pad[:na] = a
    b_pad[:nb] = b
    Af = np.fft.rfft(a_pad)
    Bf = np.fft.rfft(b_pad)
    conv = np.fft.irfft(Af * Bf, n=L)[: na + nb - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective(seq: np.ndarray) -> float:

    n = seq.shape[0]

    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:

    return float(2.0 * n * max_conv / (sum_seq ** 2))

        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off
    # --------------------------------------------------------------
    n = 8192                     # larger n gives more flexibility

    # --------------------------------------------------------------
    # 1) Seed with the best arcsine‑family member
    # --------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    alphas = np.linspace(0.20, 0.60, 21)
    best_seq = None
    best_score = float("inf")
    for alpha in alphas:
        w = (x * (1.0 - x) + 1e-30) ** (-alpha)
        a = w / w.sum()
        score = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] α={alpha:.3f} score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.copy()

    # --------------------------------------------------------------
    # 2) Use a global starter if it exists and is better
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur_score = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur_score:.6f}")
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
    except Exception:
        pass

    # ensure we have a valid start
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    a = best_seq.copy()
    a = _project_simplex(a)                # guarantee sum == 1

    # --------------------------------------------------------------
    # 3) Iterated capacity step (de‑convolution + convex mixing)
    # --------------------------------------------------------------
    L = _next_pow2(2 * n)                  # FFT length for all convolutions
    # FFT of the all‑ones target (length 2n‑1)
    ones_pad = np.zeros(L, dtype=np.float64)
    ones_pad[: 2 * n - 1] = 1.0
    Bf = np.fft.rfft(ones_pad)

    iteration = 0
    eps = 1e-12
    while time.time() < deadline - 5.0:    # keep a few seconds for polishing
        iteration += 1

        # ----- self‑convolution and current max -----
        conv_self, Af = _conv_self(a, L)
        M = conv_self.max()

        # ----- de‑convolution: solve a * c ≈ 1  (c ≥ 0) -----
        Af_mod = np.where(np.abs(Af) < eps, eps + 0j, Af)   # avoid division by zero
        Cf = Bf / Af_mod
        c_full = np.fft.irfft(Cf, n=L)[:n]
        c = np.maximum(c_full, 0.0)          # enforce non‑negativity
        sum_c = c.sum()
        if sum_c < 1e-12:
            print("[deconv] zero solution – stopping.")
            break

        # ----- candidate g (mass‑preserving) -----
        g = c / sum_c                        # already sums to 1

        # ----- try a full convex move, backtrack if needed -----
        t = 1.0
        accepted = False
        while t > 1e-6:
            a_new = (1.0 - t) * a + t * g
            a_new = _project_simplex(a_new)
            conv_new, _ = _conv_self(a_new, L)
            M_new = conv_new.max()
            if M_new <= M * (1.0 + 1e-12):
                a = a_new
                M = M_new
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # no progress – exit the loop
            print("[mix] cannot improve further – exiting.")
            break

        cur_score = 2.0 * n * M / (a.sum() ** 2)   # a.sum() == 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] iter={iteration:,} score={cur_score:.6f}")

        # a very small improvement threshold to avoid endless loops
        if iteration % 20 == 0:
            if best_score / cur_score > 0.999999:
                # progress has stalled
                break

    # --------------------------------------------------------------
    # 4) Light polishing: sub‑gradient descent + annealing (time‑budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + 0.7 * remaining
        refined_seq, refined_score = _subgradient_descent(
            best_seq, sub_deadline, start_time
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # Tiny simulated‑annealing polish (optional, very cheap)
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_steps = 20000
        for it in range(1, total_steps + 1):
            if time.time() >= deadline:
                break
            T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)   # gentle cooling
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps_move = a[src] * rng.random() * 0.20
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move
            new_score = _objective(a_prop)
            delta = new_score - cur_score
            if delta < 0.0 or math.exp(-delta / T) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score={_objective(best_seq):.6f}")
    return [float(v) for v in best_seq]
