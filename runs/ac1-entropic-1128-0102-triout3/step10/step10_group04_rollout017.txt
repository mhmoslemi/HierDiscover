    start = time.time()
    deadline = start + budget_s - 1.0

    n = 4096

    # Precompute FFT length
    L_fft = _next_pow2(2*n - 1)

    # Define objective evaluation: returns ratio (float)
    def compute_obj(a):
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a < 0.01:
            return float('inf')
        # Normalize to keep scaling invariance? Not needed.
        # but we compute conv directly.
        conv = _conv_self(a, L_fft)
        max_b = conv.max()
        return 2.0 * n * max_b / (sum_a * sum_a)

    # Generating candidates
    candidates = []

    # 1. Uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2. Linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    candidates.append(lin / lin.sum())

    # 3. Linear increasing (mirror)
    lin_inc = np.arange(1, n+1, dtype=float)
    candidates.append(lin_inc / lin_inc.sum())

    # 4. Power law with p in [0.1..1]

    i = np.arange(n)

"""
Propose a step‑function (non‑negative vector) that (almost) minimizes
``evaluate_sequence``.

The optimiser consists of three stages:

1.  **Rapid shape scan** – a handful of analytically‑motivated families
    (uniform, linear, power‑law, inverse‑Hann, Kaiser, exponential,
    Dirichlet draws, …) are generated for a fixed length `n`.  Each shape
    is normalised to unit sum and its objective
    ``2*n*max(conv(a))/sum(a)^2`` is evaluated via a fast FFT convolution.
    The best shape becomes the starting point.

2.  **Accelerated exponentiated‑gradient (Adam‑style) descent** –
    the objective is the infinity‑norm of the self‑convolution.  At each
    iteration the active index (the one attaining the current maximum)
    supplies a sub‑gradient; a multiplicative Adam update is applied on the
    log‑variables and the vector is projected onto the probability simplex.
    The best vector seen during the descent is retained.

3.  **Greedy peak‑flattening** – starting from the best vector of stage 2,
    tiny masses are moved from the index that contributes most to the
    current maximum to an index that contributes least.  The convolution is
    updated incrementally (O(n) per move) and the full FFT recomputed every
    few thousand steps to avoid numerical drift.  This stage typically
    shaves the last few 10⁻³ from the objective.

All work is stopped well before the 1000 s budget to avoid a timeout.
The final vector is renormalised to sum to 1 and returned as a plain
Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2*len(a)-1) via FFT.
    The result is truncated to the exact length.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Af * Af, n=fft_len)
    return conv_full[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                         # fallback – should never happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Objective helper (exact same formula used inside evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2*n*max(a*a)/(sum(a)^2).  a is assumed non‑negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_self(a, fft_len)
    m = conv.max()
    return 2.0 * n * m / (s * s)


# ----------------------------------------------------------------------
# Stage 2 – Adam‑style multiplicative gradient descent on the true max‑conv
# ----------------------------------------------------------------------
def _adam_exp_grad(a0: np.ndarray,
                   n: int,
                   fft_len: int,
                   deadline: float,
                   start: float,
                   eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Adam on the log‑variables; returns best vector and its objective."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time() - start:.2f}s] adam‑exp‑grad it={it:,} score={cur_val:.6f}")

        # gradient of the active (max) convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # ∂max/∂a_i

        # Adam on the log‑variables (multiplicative update)
        g = -grad                                 # descent direction for log(a)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive per‑coordinate step size
        step = eta0 / (np.sqrt(v_hat) + eps)

        log_a = np.log(np.maximum(a, 1e-12))
        log_a += step * m_hat          # note: m_hat already has the sign of –grad
        a = np.exp(log_a)

        # project back onto the simplex (exact unit sum)
        total = a.sum()
        if total > 0.0 and np.isfinite(total):
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – Deterministic greedy peak‑flattening
# ----------------------------------------------------------------------
def _greedy_flatten(a0: np.ndarray,
                    n: int,
                    fft_len: int,
                    deadline: float,
                    start: float,
                    cur_best_val: float,
                    cur_best_a: np.ndarray,
                    max_iters: int = 4_000_000) -> tuple[np.ndarray, float]:
    """Move small mass from the “worst’’ contributor to the “best’’ one."""
    a = a0.copy()
    conv = _conv_self(a, fft_len)
    best_a = a.copy()
    best_val = cur_best_val
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore current max)
        tmp = np.partition(conv, -2)
        second_max = tmp[-2] if conv.size > 1 else max_val

        # indices that actually take part in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)

        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for each candidate dst
        partner_vals = a[max_idx - idx]

        # choose destination with smallest partner (but not the source)
        min_part = partner_vals.min()
        dst_cand = idx[(partner_vals == min_part) & (idx != src)]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # pick a safe epsilon
        if delta_a <= 0.0:
            eps_move = min(a[src], 0.02 * a[src])
        else:
            # try to bring the peak down to the second‑largest value
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-12)
            eps_move = min(a[src], max(0.0, 0.9 * eps_target))
            if eps_move < 1e-12:
                eps_move = min(a[src], 0.02 * a[src])

        if eps_move <= 0.0:
            continue

        # incremental convolution update (O(n))
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps_move * a_old
        conv[src:src + n] -= 2.0 * eps_move * a_old
        a[src] -= eps_move
        a[dst] += eps_move
        a[a < 0.0] = 0.0

        if it % recompute_interval == 0:
            conv = _conv_self(a, fft_len)

        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time() - start:.2f}s] greedy it={it:,} score={cur_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4096 gave the best trade‑off in earlier experiments.
    # ------------------------------------------------------------------
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Scan a variety of analytically‑motivated shapes.
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # linear decreasing (and its mirror)
    lin = np.arange(n, 0, -1, dtype=float)
    candidates.append(lin / lin.sum())
    candidates.append(lin[::-1] / lin.sum())

    # simple power‑law families a_i ∝ (i+½)^{-p}
    for p in [0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40]:
        idx = np.arange(n, dtype=float) + 0.5
        a = idx ** (-p)
        candidates.append(a / a.sum())

    # symmetric edge‑enhanced power‑law:
    # a_i ∝ ((i+½)*(n‑i‑½))^{-p/2}
    for p in [0.10, 0.20, 0.30, 0.40, 0.50]:
        idx = np.arange(n, dtype=float)
        prod = (idx + 0.5) * (n - idx - 0.5)
        a = prod ** (-p / 2.0)
        candidates.append(a / a.sum())

    # inverse‑Hann (cos²) – low centre, high edges
    i = np.arange(n, dtype=float)
    cos2 = np.cos(np.pi * i / (n - 1)) ** 2
    candidates.append(cos2 / cos2.sum())

    # exponential decays (both directions)
    for q in [0.9999, 0.9995, 0.998, 0.995]:
        exp_fwd = (1 - q) * (q ** np.arange(n, dtype=float))
        candidates.append(exp_fwd / exp_fwd.sum())
        candidates.append(exp_fwd[::-1] / exp_fwd.sum())

    # Kaiser windows with a few betas (np.i0 is the modified Bessel function)
    for beta in [0.5, 2.0, 4.0, 6.0, 8.0]:
        w = np.i0(np.pi * beta *
                  np.sqrt(1.0 - ((2.0 * i) / (n - 1) - 1.0) ** 2))
        candidates.append(w / w.sum())

    # Random Dirichlet draws – different concentration parameters
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            a = np.random.gamma(shape=alpha, scale=1.0, size=n)
            a = a / a.sum()
            candidates.append(a)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Evaluate all candidates (very cheap) and keep the best one.
    # ------------------------------------------------------------------
    for cand in candidates:
        if time.time() >= deadline:
            break
        scr = _objective(cand, n, fft_len)
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[{time.time() - start:.2f}s] init shape score={scr:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam‑style exponentiated gradient descent (≈60 % of remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + remaining * 0.65
        seq2, scr2 = _adam_exp_grad(best_seq, n, fft_len,
                                    phase2_deadline, start, eta0=0.7)
        if scr2 < best_score:
            best_score = scr2
            best_seq = seq2.copy()
            print(f"[{time.time() - start:.2f}s] post‑adam score={scr2:.6f}")

    # ------------------------------------------------------------------
    # 3) Deterministic greedy flattening (use all time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq3, scr3 = _greedy_flatten(best_seq, n, fft_len,
                                    deadline, start,
                                    best_score, best_seq,
                                    max_iters=8_000_000)
        if scr3 < best_score:
            best_score = scr3
            best_seq = seq3.copy()
            print(f"[{time.time() - start:.2f}s] final greedy score={scr3:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (the objective is homogeneous, any scaling works)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] DONE – best objective ≈ {best_score:.6f}")
    # return plain Python list as required by the outer harness
    return [float(x) for x in best_seq]
