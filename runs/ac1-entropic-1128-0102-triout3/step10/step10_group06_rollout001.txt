"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Generate a rich pool of starting distributions** – uniform, several
   deterministic shapes (exponential, Gaussian) and dozens of Dirichlet
   draws with different concentration parameters.

2. **Select the best raw start** (exact objective computed with a fast
   FFT based routine).

3. **Warm‑up** – a short Adam optimisation on a smooth log‑sum‑exp
   surrogate of the convolution peak.  This moves the distribution away
   from very bad regions without committing many iterations.

4. **Full exponentiated‑gradient descent on the exact max‑convolution**
   objective.  At each iteration we:
   * compute the self‑convolution via FFT,
   * select *all* shifts whose value is within a tiny relative tolerance
     of the current maximum (the true sub‑gradient of the max‑function),
   * build the sub‑gradient by summing the gradients of those active
     shifts,
   * perform an exponentiated‑gradient (mirror‑descent on the
     probability simplex) update with a diminishing step‑size.
   If no improvement is seen for a long stretch we inject a tiny random
   mass transfer to escape possible plateaus.

5. **Deterministic greedy peak‑reduction** – move a modest fraction of
   the coefficient that most contributes to the current convolution peak
   towards the smallest coefficient, accepting only strict improvements.

6. **Return the best sequence** (clipped to `[0,1000]` and renormalised).

The routine respects the supplied time budget (default 1000 s) and prints
progress information.  All heavy linear algebra is performed with FFTs,
so even with 4800 coefficients millions of iterations fit comfortably
within the time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT‑based convolution, utilities)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _compute_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective 2·n·max(conv)/(sum(a)**2)  – cheap version using FFT.
    """
    conv = _conv_self(a, fft_len)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] obtained by summing the gradients of
    the shifts listed in *active_shifts*.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # each pair contributes 2*a[j] to the gradient of a[i]
        g[i] += 2.0 * a[j]
    return g


def _active_shifts(conv: np.ndarray,
                   cur_max: float,
                   delta_frac: float,
                   max_active: int | None) -> np.ndarray:
    """
    Return the indices of shifts that are “almost” maximal.
    If max_active is given we keep only the top‑max_active among them.
    """
    thresh = cur_max - max(delta_frac * cur_max, 1e-14)
    active = np.where(conv >= thresh)[0]
    if max_active is not None and active.size > max_active:
        # keep the `max_active` shifts with largest conv values
        part = np.argpartition(-conv[active], max_active - 1)[:max_active]
        active = active[part]
    return active


# ----------------------------------------------------------------------
# Warm‑up: Adam on a smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray,
                      alpha: float,
                      fft_len: int) -> tuple[float, np.ndarray]:
    """Log‑sum‑exp (soft‑max) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_warmup(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 fft_len: int,
                 n: int) -> tuple[np.ndarray, float]:
    """
    A short Adam optimisation on the smooth surrogate.
    Returns the best vector encountered and its exact (fast) score.
    """
    a = a0.copy()
    # work in the log‑domain to keep positivity automatically
    y = np.log(np.maximum(a, 1e-16))

    # smoothing schedule (gradually tighten the surrogate)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 2000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = _compute_score(a, n, fft_len)

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # softmax → probability simplex
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of smooth surrogate w.r.t a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()                              # soft‑max weights over shifts
            # convolution of a with reversed weights gives gradient
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w[::-1], n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w) - 1]
            grad_a = 2.0 * corr[len(w) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact check
            if t % 500 == 0:
                sc = _compute_score(a, n, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}")

    # final projection onto simplex (just in case)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main optimisation: exponentiated‑gradient on the exact max‑convolution
# ----------------------------------------------------------------------


def _expgrad_opt(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 fft_len: int,
                 n: int,
                 eta0: float = 0.6,
                 delta_frac: float = 1e-5,
                 max_active: int | None = None,
                 plateau_patience: int = 20000,
                 rng: np.random.Generator | None = None) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) minimising the exact objective.
    *max_active* limits the number of active shifts (None → all).
    *plateau_patience* controls how often a random perturbation is applied
    when no improvement has been seen.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _compute_score(a, n, fft_len)
    last_improve_iter = 0
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() * a.sum())

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve_iter = t
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # --------------------------------------------------------------
        #  sub‑gradient of the max‑convolution
        # --------------------------------------------------------------
        active = _active_shifts(conv, cur_max, delta_frac, max_active)
        g = _multi_gradient(a, active)

        # --------------------------------------------------------------
        #  exponentiated update with diminishing step size
        # --------------------------------------------------------------
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # --------------------------------------------------------------
        #  re‑normalise onto the simplex & protect against underflow
        # --------------------------------------------------------------
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        a = np.maximum(a, 1e-15)

        # --------------------------------------------------------------
        #  random perturbation when stuck on a plateau
        # --------------------------------------------------------------
        if t - last_improve_iter > plateau_patience:
            # move a modest fraction of mass from a large entry to a small one
            src = rng.choice(np.where(a > 1e-12)[0],
                             p=a / a.sum())
            dst_candidates = np.delete(np.arange(n), src)
            dst = dst_candidates[a[dst_candidates].argmin()]
            eps = min(0.05 * a[src], 0.01)          # keep move tiny
            a[src] -= eps
            a[dst] += eps
            a = a / a.sum()
            last_improve_iter = t         # reset counter after perturb

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic peak reduction
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          n: int) -> tuple[np.ndarray, float]:
    """
    Transfer a fraction of the coefficient responsible for the current
    convolution peak towards the smallest coefficient.  Accept only
    strictly decreasing moves.
    """
    a = a0.copy()
    best_score = _compute_score(a, n, _next_pow2(2 * n - 1))
    it = 0

    while time.time() < deadline and it < 5000:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]          # modest fraction of the source mass
        if eps <= 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _compute_score(a_prop, n, _next_pow2(2 * n - 1))
        if new_score < cur_max * 2.0 * n / (a.sum() ** 2):
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: score={new_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Start‑point generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    vec /= vec.sum()
    return vec


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size (keep the size used by the reference implementation)
    # --------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse set of start points
    # --------------------------------------------------------------
    starts = []

    # 1) previous best if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) handcrafted seeds
    starts.append(np.full(n, 1.0 / n))                     # uniform
    starts.append(_exponential_start(n, r=0.9995))
    starts.append(_exponential_start(n, r=0.985))
    starts.append(_gaussian_start(n))
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # 3) random Dirichlet draws with a spread of concentrations
    conc_params = [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for conc in conc_params:
        for _ in range(8):                 # 8 draws per concentration
            starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # --------------------------------------------------------------
    # Evaluate raw starts and keep the best one (fast exact score)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = _compute_score(a0, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – short Adam warm‑up (≈5 % of the time budget)
    # --------------------------------------------------------------
    warmup_deadline = start_time + 0.05 * budget_s
    if time.time() < warmup_deadline:
        cand_seq, cand_score = _adam_warmup(best_seq,
                                            warmup_deadline,
                                            start_time,
                                            fft_len,
                                            n)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam warm‑up: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – extensive exponentiated‑gradient optimisation
    #          (≈80 % of the remaining budget)
    # --------------------------------------------------------------
    expgrad_deadline = start_time + 0.90 * budget_s
    polished_seq, polished_score = _expgrad_opt(best_seq,
                                                expgrad_deadline,
                                                start_time,
                                                fft_len,
                                                n,
                                                eta0=0.7,
                                                delta_frac=1e-6,
                                                max_active=None,
                                                plateau_patience=25000,
                                                rng=rng)
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after ExpGrad: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic greedy peak reduction (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(best_seq,
                                                       deadline,
                                                       start_time,
                                                       n)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final safety checks & report
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
