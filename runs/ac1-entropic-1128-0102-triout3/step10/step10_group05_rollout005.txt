def _run_lbfgs_smoothing(a_init, n, deadline, start_time, alphas, maxfun_per_alpha=300):
    x = np.log(a_init + 1e-20)   # log of a_init
    best_a = a_init.copy()
    best_score = evaluate_sequence(best_a.tolist())
    for alpha in alphas:
        if time.time() >= deadline: break
        # define closure function for objective+grad
        def func_and_grad(x_local):
            obj, grad = smoothed_obj_and_grad(x_local, alpha, n)
            return obj, grad
        # Use minimize
        res = minimize(fun=lambda x_: smoothed_obj_and_grad(x_, alpha, n)[0],
                       x0=x,
                       jac=lambda x_: smoothed_obj_and_grad(x_, alpha, n)[1],
                       method='L-BFGS-B',

            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LBFGS improve at alpha {alpha}: {cur_score:.6f}")

"""
Optimise a non‑negative step function for the evaluation function
``evaluate_sequence``.

The optimiser works in three stages:

1.  Initialise with a finely tuned exponential decay (grid‑search over the
    decay factor).  The exponential distribution is already normalised,
    which means the objective is simply ``2·n·max(conv)``.

2.  Refine the shape by minimising a *smoothed* version of the max‑convolution
    using the L‑BFGS quasi‑Newton method.  The smoothing is the
    log‑sum‑exp (a standard smooth approximation of the max).  The variables
    are the unconstrained logits ``x`` of a soft‑max representation
    ``a = softmax(x)``; this guarantees non‑negativity and the ``sum(a)=1``
    constraint automatically.  For a sequence length of 4096 we run a short
    L‑BFGS optimisation for a series of increasing smoothing parameters,
    gradually approaching the true (non‑smooth) objective.

3.  A tiny stochastic hill‑climbing finish: repeatedly move a small random
    amount of mass from a source index to a destination index and keep the
    move only if the true objective improves.  This clears the remaining few
    10⁻⁴ of the objective that the quasi‑Newton phase cannot see because of
    the smoothing.

All phases respect the global time budget (default 1000 s).  The best
sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a using an FFT of length L.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i ∝ r**i, normalised to sum 1."""
    if not (0.0 < r < 1.0):
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    norm = 1.0 - r ** n                     # (1‑rⁿ)
    a = a * (1.0 - r) / norm               # normalise to unit sum
    return a


def _grad_a_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of ``sum_i w_i conv_i`` w.r.t. a (conv = a*a).
    w must have length 2·n‑1 and be non‑negative (weights of the smoothed max).
    """
    n = a.shape[0]
    w_rev = w[::-1]
    L = _next_pow2(n + len(w_rev) - 1)
    Af = np.fft.rfft(a, n=L)
    Wrev = np.fft.rfft(w_rev, n=L)
    corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
    grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]
    return grad


def _smoothed_obj_and_grad(x: np.ndarray, alpha: float, n: int):
    """
    For a given smoothing parameter ``alpha`` compute

        f(x) = 2·n·smooth_max(conv(a))

    where a = softmax(x) and ``smooth_max`` = (1/α)·log∑exp(α·conv).
    Returns the scalar value and the gradient w.r.t. the logits x.
    """
    a = _softmax(x)

    # full self‑convolution (length 2·n‑1)
    Lfft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=Lfft)
    conv = np.fft.irfft(Af * Af, n=Lfft)[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    sum_exp = exp_scaled.sum()
    smooth_max = (max_scaled + np.log(sum_exp)) / alpha
    obj = 2.0 * n * smooth_max

    # weights for the gradient (softmax over α·conv)
    w = exp_scaled / sum_exp
    grad_a = _grad_a_from_weights(a, w)

    # chain rule for the soft‑max parametrisation
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


# ----------------------------------------------------------------------
# Optimisation stages
# ----------------------------------------------------------------------
def _stage_initialise(n: int, deadline: float, start_time: float):
    """
    Scan the decay factor of an exponential seed and return the best
    normalised sequence (under the true objective, i.e. 2·n·max(conv)).
    """
    best_score = float("inf")
    best_seq = None
    # Fine grid – denser near 1 to capture the optimum.
    for r in np.linspace(0.97, 0.99995, 800):
        cand = _exponential_sequence(n, r)
        L = _next_pow2(2 * n - 1)
        conv = _conv_self(cand, L)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand
    print(f"[{time.time() - start_time:.2f}s] init exponential r≈{r:.6f} score={best_score:.6f}")
    return best_seq, best_score


def _stage_lbfgs(a0: np.ndarray, n: int,
                deadline: float, start_time: float,
                alphas, maxfun_per_alpha=250):
    """
    Run L‑BFGS on the smoothed objective for a series of smoothing
    parameters (α).  Starts from ``a0`` and returns the best sequence found.
    """
    # logits for the soft‑max parametrisation
    x = np.log(a0 + 1e-30)

    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())

    for alpha in alphas:
        if time.time() >= deadline:
            break

        # wrap the objective / gradient in lightweight callables
        def fun(x_local):
            f, _ = _smoothed_obj_and_grad(x_local, alpha, n)
            return f

        def jac(x_local):
            _, g = _smoothed_obj_and_grad(x_local, alpha, n)
            return g

        # run a short L‑BFGS optimisation
        res = minimize(fun=fun,
                       x0=x,
                       jac=jac,
                       method='L-BFGS-B',
                       options={'maxfun': maxfun_per_alpha,
                                'maxiter': maxfun_per_alpha,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})

        x = res.x
        a = _softmax(x)

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] L‑BFGS (α={alpha}) improve: {cur_score:.6f}")

    return best_a, best_score


def _stage_hill_climb(a0: np.ndarray, n: int,
                     deadline: float, start_time: float,
                     cur_best_score: float, cur_best_seq: np.ndarray,
                     moves: int = 200_000):
    """
    Very cheap stochastic hill‑climbing: move a random tiny amount of mass
    from a source index to a destination index and keep it if the true
    objective improves.  The loop stops when the time budget is exhausted
    or after ``moves`` iterations.
    """
    a = a0.copy()
    best_a = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < moves:
        it += 1

        # pick a source with non‑trivial mass
        src = rng.choice(np.where(a > 1e-12)[0])
        dst = rng.integers(0, n)
        if src == dst:
            continue

        # move a small random fraction of src's mass
        eps = a[src] * rng.random() * 0.05  # ≤5 % of src
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate quickly via the provided function (includes convolution)
        new_score = evaluate_sequence(a_prop.tolist())
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] hill‑climb iter={it:,} score={new_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses an exponential seed, L‑BFGS on a smoothed max‑convolution and a

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Initial exponential seed (length 4096)
    # ------------------------------------------------------------------
    n = 4096

        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()

                           np.linspace(0, 1, starter.size),
                           starter)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            raise Exception()
    except Exception:
        a0, _ = _stage_initialise(n, deadline, start_time)

    best_seq = a0.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  L‑BFGS refinement on a smoothed max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # smoothing parameters – start gentle, finish very sharp
        alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
                  80.0, 160.0, 320.0, 640.0, 1280.0, 2560.0]
        a_lbfgs, score_lbfgs = _stage_lbfgs(best_seq, n,
                                            deadline, start_time,
                                            alphas,
                                            maxfun_per_alpha=300)
        if score_lbfgs < best_score:
            best_score = score_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"[{time.time() - start_time:.2f}s] after L‑BFGS score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Tiny random hill‑climb to prune the remaining gap
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, score_hc = _stage_hill_climb(best_seq, n,
                                           deadline, start_time,
                                           best_score, best_seq,
                                           moves=200_000)
        if score_hc < best_score:
            best_score = score_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] after hill‑climb score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (the objective is scale‑invariant, but we keep sum=1)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_obj = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(v) for v in best_seq]
