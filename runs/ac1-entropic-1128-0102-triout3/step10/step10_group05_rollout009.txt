def _run_random_hill_climbing(a0, n, L, deadline, start_time, best_score):
    a = a0.copy()
    conv = _conv_self(a, L)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_local = a.copy()
    best_local_score = cur_score

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        # choose src-prob weighted by a   (mass)
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.1  # up to 10% of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # compute convolution and objective
        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_prop
            conv = conv_new
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill climb improve: iter={it}, score={new_score:.6f}")
    return best_local, best_local_score
```

"""
Improved step‑function optimiser.

The optimiser builds a probability vector (non‑negative entries summing to 1)
of configurable length.  It starts from a finely tuned exponential decay,
refines it with a few L‑BFGS runs on a smooth approximation of the max‑
convolution, performs a short Adam phase, then a deterministic greedy sweep,
and finally a simple stochastic hill‑climbing / mixing stage.  All phases
respect the global time budget (default 1000 s).  The returned list is a plain
Python list of floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS optimisation)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import minimize  # type: ignore
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self–convolution a*a using FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution a_i ∝ r**i (0<r<1).
    The vector is automatically normalised to sum 1.
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


# ----------------------------------------------------------------------
# Smooth‐max objective (log‑sum‑exp) and its gradient w.r.t. soft‑max params
# ----------------------------------------------------------------------
def _smooth_max_and_grad(y: np.ndarray, alpha: float, L: int, n: int):
    """
    Given real parameters y, define a = softmax(y) (∑a=1, a≥0).
    Return a smooth approximation of max(conv) together with its gradient
    d/ dy ( (1/α)·log ∑ exp(α·conv) ).
    """
    # soft‑max
    y_shift = y - np.max(y)
    exp_y = np.exp(y_shift)
    a = exp_y / exp_y.sum()

    # convolution a*a
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    value = (max_scaled + np.log(sum_exp)) / alpha  # smooth max

    # weights for gradient (∂value/∂conv = w)
    w = exp_scaled / sum_exp

    # gradient w.r.t. a : 2·(a ∘ w_rev)   (∘ denotes correlation)
    w_rev = w[::-1]
    corr = np.fft.irfft(Af * np.fft.rfft(w_rev, n=L), n=L)
    idx = len(w_rev) - 1 - np.arange(n)
    grad_a = 2.0 * corr[idx]

    # chain rule through the soft‑max
    dot = np.dot(a, grad_a)
    grad_y = a * (grad_a - dot)

    return value, grad_y


def _optimize_l_bfgs(a_init: np.ndarray, alpha: float,
                     max_iter: int, deadline: float) -> np.ndarray:
    """
    Run a small L‑BFGS optimisation of the smooth max for a fixed α.
    Stops early if the global deadline is exceeded.
    """
    if not _HAS_SCIPY:               # pragma: no cover
        return a_init.copy()

    n = a_init.shape[0]
    L = _next_pow2(2 * n - 1)
    y0 = np.log(a_init + 1e-12)

    def fun(y):
        val, grad = _smooth_max_and_grad(y, alpha, L, n)
        return val, grad

    options = {"maxiter": max_iter, "disp": False}
    res = minimize(fun, y0, method="L-BFGS-B", jac=True, options=options)

    a_opt = np.exp(res.x - np.max(res.x))
    a_opt /= a_opt.sum()
    return a_opt


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 20000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max weights (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)  # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build vector a_k[i] = a[max_idx - i] (zero outside the range)
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination index with the smallest a_k value
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partner values for src and dst in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Simple stochastic hill climbing (accept‑only‑improving moves)
# ----------------------------------------------------------------------
def _run_random_hill(a0: np.ndarray,
                     n: int,
                     L: int,
                     deadline: float,
                     start_time: float,
                     cur_best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _conv_self(a, L)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        # pick a source index weighted by its contribution to the current peak
        max_idx = int(conv.argmax())
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        if contrib.sum() > 0:
            src = rng.choice(idx, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * 0.05 * rng.random()
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_prop
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] random hill improve: "
                      f"iter={it:,}, score={new_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # ------------------------------------------------------------------
    final_n = 4096
    L = _next_pow2(2 * final_n - 1)

    # ------------------------------------------------------------------
    # Initialise – refined exponential decay.
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_r = 0.99
    # a fairly fine scan (still cheap thanks to FFT)
    for r in np.linspace(0.90, 0.99995, 800):
        cand = _exponential_sequence(final_n, r)
        conv = _conv_self(cand, L)
        score = 2.0 * final_n * conv.max()
        if score < best_score:
            best_score = score
            best_r = r
    a0 = _exponential_sequence(final_n, best_r)
    print(f"[0.00s] exponential seed r={best_r:.6f}, score={best_score:.6f}")

    best_seq = a0.copy()
    best_score = 2.0 * final_n * _conv_self(best_seq, L).max()
    print(f"[0.00s] initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase A – L‑BFGS on a smooth max (if SciPy is present)
    # ------------------------------------------------------------------
    if _HAS_SCIPY and time.time() < deadline:
        for alpha in (5.0, 20.0, 80.0):
            if time.time() >= deadline:
                break
            a_opt = _optimize_l_bfgs(best_seq, alpha, max_iter=200, deadline=deadline)
            score = 2.0 * final_n * _conv_self(a_opt, L).max()
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] L‑BFGS α={alpha:.1f} improve: "
                      f"score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Adam on a smoothed max (≈15 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = time.time() + min(180.0, deadline - time.time())
        a_adam, score_adam = _run_adam_smoothing(
            best_seq,
            final_n,
            phase_deadline,
            start_time,
            alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            lr0=0.6,
            iters_per_alpha=25000,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Aggressive greedy reduction (≈30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = deadline
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq,
            final_n,
            phase_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=3_000_000,
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] Greedy improve: score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Stochastic hill climbing (≈15 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = deadline
        a_hill, score_hill = _run_random_hill(
            best_seq,
            final_n,
            L,
            phase_deadline,
            start_time,
            best_score,
        )
        if score_hill < best_score:
            best_score = score_hill
            best_seq = a_hill.copy()
            print(f"[{time.time() - start_time:.2f}s] Random hill improve: "
                  f"score={score_hill:.6f}")

    # ------------------------------------------------------------------
    # Phase E – Simple convex mixes with uniform / reversed version
    # ------------------------------------------------------------------
    if time.time() < deadline:
        uniform = np.full(final_n, 1.0 / final_n)
        reversed_seq = best_seq[::-1]
        for mix in np.linspace(0.0, 1.0, 21):
            if time.time() >= deadline:
                break
            # mix with uniform
            a_mix = (1.0 - mix) * best_seq + mix * uniform
            a_mix = a_mix / a_mix.sum()
            score = 2.0 * final_n * _conv_self(a_mix, L).max()
            if score < best_score:
                best_score = score
                best_seq = a_mix.copy()
                print(f"[{time.time() - start_time:.2f}s] Mix‑uniform α={mix:.2f} improve: "
                      f"score={score:.6f}")

            # mix with reversed copy
            a_mix2 = (1.0 - mix) * best_seq + mix * reversed_seq
            a_mix2 = a_mix2 / a_mix2.sum()
            score2 = 2.0 * final_n * _conv_self(a_mix2, L).max()
            if score2 < best_score:
                best_score = score2
                best_seq = a_mix2.copy()
                print(f"[{time.time() - start_time:.2f}s] Mix‑reverse α={mix:.2f} improve: "
                      f"score={score2:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    final_obj = 2.0 * final_n * _conv_self(best_seq, L).max()
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    # Return as plain Python list
    return [float(x) for x in best_seq]
