"""
Hybrid optimizer for the 2 · n · max(conv(a,a)) / (sum(a)**2) objective.
The method consists of three stages:

1️⃣  Exponentiated‑gradient (smooth‑max) mirror descent – gradually increases a
    temperature α so that the soft‑max approximates the true max.  This quickly
    reduces large peaks of the self‑convolution.

2️⃣  Projected sub‑gradient descent with momentum – works directly on the
    nonsmooth max‑convolution and refines the solution obtained by stage 1.

3️⃣  Directed mass‑transfer refinement – after the sub‑gradient phase we
    repeatedly move probability mass from indices that contribute most to the
    active (near‑maximal) convolution entries to those that contribute the
    least.  The move direction is obtained analytically from the active set,
    so each iteration makes a noticeable decrease in the true objective.

If time permits a tiny random annealing polish is applied at the very end.
All operations stay on the probability simplex (sum = 1, a_i ≥ 0) and use
FFT‑based convolutions for speed.  The whole routine respects a wall‑clock
budget of ≤ 1000 s.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT sizes, convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear convolution of two real vectors u and v using FFT.
    Returns the full linear convolution (len(u)+len(v)-1 elements).
    """
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (vector assumed to sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (same idea as the reference)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # geometric schedule for the smoothing parameter α
    alphas = np.geomspace(0.5, 200.0, num=30)

    it = 0
    max_iterations = 250_000

    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # 1) self‑convolution a*a
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights w ∝ exp(α·conv)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient of the smooth maximum w.r.t. a
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)
            a /= a.sum()

            # occasional exact‑objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (as in the reference solution)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.31          # slightly larger than the reference
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (first max – other indices behave similarly)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max‑convolution at index k
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : deterministic active‑set mass‑transfer refinement
# ----------------------------------------------------------------------
def _active_mass_transfer(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Greedy moves mass from indices that contribute most to the active
    (near‑maximum) convolution entries toward indices that contribute the
    least.  The active set is defined by a relative tolerance.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # parameters
    rel_tol = 1e-4          # indices with conv >= (1‑rel_tol)·max are active
    max_iter_without_gain = 5000
    it_without_gain = 0

    while time.time() < deadline:
        # 1) evaluate current objective
        conv = _conv_self(a, L=L_self)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            it_without_gain = 0
            print(f"[{time.time() - start_time:.2f}s] Mass‑transfer best={score:.6f}")
        else:
            it_without_gain += 1
            if it_without_gain >= max_iter_without_gain:
                # stagnation – apply a tiny random perturbation to escape
                src = np.random.choice(n, p=a)
                dst = np.random.randint(n)
                if src != dst and a[src] > 1e-12:
                    eps = a[src] * (0.02 + 0.03 * np.random.rand())
                    a[src] -= eps
                    a[dst] += eps
                it_without_gain = 0
                continue

        # 2) build active‑set indicator (length 2n‑1)
        active = conv >= M * (1.0 - rel_tol)

        # keep the active set compact (max 30 indices)
        if active.sum() > 30:
            # keep the top‑30 peaks
            top_idx = np.argsort(-conv)[:30]
            active[:] = False
            active[top_idx] = True

        # 3) compute impact vector h_j = Σ_{i∈S} a_{i‑j}
        s = active.astype(float)               # shape (2n‑1,)
        s_rev = s[::-1]
        # convolution a (len n) with s_rev (len 2n‑1) → length 3n‑2
        h_full = _conv_linear(a, s_rev, L=L_grad)
        # indices that correspond to j = 0 … n‑1:
        idx = 2 * n - 2 - np.arange(n)
        h = h_full[idx]

        # 4) choose source (large h) and destination (small h)
        src = int(np.argmax(h))
        dst = int(np.argmin(h))
        if src == dst:
            # fallback: random move
            src = np.random.choice(n, p=a)
            dst = np.random.randint(n)

        # 5) transfer a fraction of mass from src to dst
        max_move = a[src] * 0.5          # never move more than half the source mass
        if max_move < 1e-12:
            # source essentially empty – pick another source
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                break
            src = int(np.random.choice(nonzero))
            max_move = a[src] * 0.5
        eps = max_move * np.random.rand()
        a[src] -= eps
        a[dst] += eps
        # numerical safety
        a = np.maximum(a, 0.0)
        a /= a.sum()   # keep exact simplex (sum is invariant under the move, but re‑normalise for safety)

    return best_seq, best_score

               steps: int = 100_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Load an initial vector (previous best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Smooth‑max mirror descent (core of the new method)
    # ------------------------------------------------------------------
    if time.time() < deadline - 15.0:
        smooth_deadline = deadline - 12.0
        best_seq, best_score = _smooth_descent(best_seq, n, smooth_deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient refinement (as in the reference)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        sub_deadline = deadline - 3.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Directed active‑set mass transfer (new deterministic polish)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        mt_deadline = deadline - 0.5
        a_mt, sc_mt = _active_mass_transfer(best_seq, n, mt_deadline,
                                            start_time, best_score, best_seq)
        if sc_mt < best_score:
            best_score = sc_mt
            best_seq = a_mt.copy()
            print(f"[{time.time() - start_time:.2f}s] Mass‑transfer polish improved score = {sc_mt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Very short random annealing polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=1e-2, end_temp=1e-7,
                                       steps=200_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
