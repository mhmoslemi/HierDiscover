"""
A fast “soft‑max + projected gradient” optimizer for the step‑function bound.

Idea
----
* The objective 2·n·max(conv)/(∑a)² is scale‑invariant, so we work on the
  probability simplex (∑a = 1).
* Instead of the non‑smooth max we minimise a smooth surrogate
  Lβ(a)=log‑∑exp(β·(a*a)).  When β→∞ the surrogate converges to the true max.
* The gradient of Lβ can be written in closed form using a correlation
  between the current vector a and the soft‑max weights p.
* Each iteration therefore needs two FFT‑based convolutions:
    – the self‑convolution a*a  (to build p)
    – the correlation a⋆p      (the gradient)
* After a gradient step we project back onto the simplex
  (Euclidean projection).  Symmetrising a after every step keeps the
  search space small without hurting optimality.
* A simple schedule increases β during the run (the objective becomes
  sharper) while the step‑size decays as 1/√t.  The best iterate seen
  is stored and returned.
* A few thousand cheap sub‑gradient steps and a light annealing polish
  are performed at the end if time permits.

The whole routine respects the 1000 s budget and never exceeds it.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution (real → real) using FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation a ⋆ p  (grad_i = Σ_l a_l p_{i+l})
# ----------------------------------------------------------------------
def _conv_corr(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Returns the vector g where
        g[i] = Σ_l a[l] * p[i + l]   ,  i = 0,…,len(a)-1
    (zero padding outside the range of p).
    Implemented via convolution with the reversed p.
    """
    n = a.shape[0]
    m = p.shape[0]                     # = 2·n‑1
    L = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=L)
    Pr = np.fft.rfft(p[::-1], n=L)     # reversed p
    corr_full = np.fft.irfft(A * Pr, n=L)
    start = m - 1
    return corr_full[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Implementation of the O(n log n) algorithm of Wang & Carreira‑Perpinán."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                               # should never happen for non‑negative v
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise  Lβ(a)=log∑exp(β·(a*a))  on the simplex using projected gradient.
    The temperature β is increased gradually so that the surrogate
    becomes sharper.  Returns the best iterate seen.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    step0 = 1.0               # initial step size (larger than in the legacy code)
    beta = 1.0                # start smoothing parameter
    beta_target = 200.0       # final β – large enough to approximate max
    beta_mul = 1.001          # multiplicative increase per iteration

    it = 0
    while time.time() < deadline:
        it += 1
        # 1) self‑convolution (used for the surrogate and the true score)
        conv = _conv_self(a)

        # true objective for bookkeeping
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑iter={it:,} "
                  f"score={cur_score:.6f}")

        # 2) soft‑max weights (probability distribution over the 2n‑1 sums)
        scaled = beta * conv
        max_scaled = np.max(scaled)
        exp_shifted = np.exp(scaled - max_scaled)          # safe exponentiation
        p = exp_shifted / exp_shifted.sum()                # ∑ p = 1

        # 3) gradient of the smooth surrogate (see derivation in the doc‑string)
        grad = 2.0 * _conv_corr(a, p)                      # length‑n vector

        # 4) momentum + step‑size
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # 5) project onto the simplex and enforce symmetry
        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5        # keep the vector symmetric

        # 6) anneal β towards the target
        if beta < beta_target:
            beta = min(beta * beta_mul, beta_target)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light projected sub‑gradient descent (exact max) – fine polishing
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient descent on the exact max‑objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index of a maximal convolution entry (any one is fine)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5      # keep symmetry

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny random‑mass transfers with a fast cooling schedule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (very fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)          # more mass → higher chance to be moved
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function within the given time budget.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a larger canvas improves the continuous limit.
    # --------------------------------------------------------------
    n = 16384                     # power of two → FFT friendly

    # --------------------------------------------------------------
    # 1) Analytic warm‑start: sweep a few parametric families
    # --------------------------------------------------------------
    x = np.arange(n, dtype=float) / (n - 1)      # points in [0,1]

    families = [
        lambda p: np.minimum(x, 1.0 - x) ** p,               # tent‑type
        lambda p: np.sin(np.pi * x) ** p,                    # sinusoid
        lambda p: (x * (1.0 - x)) ** p,                      # parabola‑type
    ]

    best_seq = None
    best_score = float("inf")

    exponents = np.linspace(0.2, 2.0, 25)   # fine sweep

    for fam in families:
        for p in exponents:
            w = fam(p)
            if w.sum() == 0.0:
                continue
            a = w / w.sum()                     # simplex normalisation
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] fam={fam.__code__.co_consts[1]:<6}"
                      f" p={p:.3f} score={score:.6f}")

    # --------------------------------------------------------------
    # 2) Keep the globally stored construction if it is better
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3) Smoothed gradient descent (most of the time is spent here)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:                              # leave room for polishing
        smooth_deadline = time.time() + remaining * 0.80
        smooth_seq, smooth_score = _smooth_gradient_descent(best_seq,
                                                            smooth_deadline,
                                                            start_time)
        if smooth_score < best_score:
            best_score = smooth_score
            best_seq = smooth_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth‑descent best={smooth_score:.6f}")

    # --------------------------------------------------------------
    # 4) Extra few thousand sub‑gradient steps (fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 5.0
        sub_seq, sub_score = _subgradient_descent(best_seq,
                                                  sub_deadline,
                                                  start_time)
        if sub_score < best_score:
            best_score = sub_score
            best_seq = sub_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish best={sub_score:.6f}")

    # --------------------------------------------------------------
    # 5) Light annealing polish (if any time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score = {_objective(best_seq):.6f}")
    return [float(v) for v in best_seq]
