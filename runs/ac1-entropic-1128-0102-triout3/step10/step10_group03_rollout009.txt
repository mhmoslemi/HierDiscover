def simulated_annealing(a, conv, start_score, deadline, start_time):
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score
    T = 0.001 # initial temperature
    T_min = 1e-7
    T_decay = 0.9995
    n = a.shape[0]
    rng = np.random.default_rng()
    # pre-built range arrays for efficiency? We'll just compute dynamic each iteration.

    while time.time() < deadline:
        # pick source in active interval
        k = int(cur_conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        src_candidates = np.arange(lo, hi+1)
        # sample src weighted by a[src] maybe
        prob = cur_a[src_candidates]
        prob_sum = prob.sum()
        if prob_sum == 0:
            src = rng.integers(lo, hi+1)
        else:
            prob = prob / prob_sum
            src = rng.choice(src_candidates, p=prob)

        dst = rng.integers(n)
        # if src == dst or cur_a[src] < 1e-12: continue
        if src == dst or cur_a[src] < 1e-12:
            continue
        # epsilon proportion
        eps = cur_a[src] * rng.random() * 0.1 # up to 10% of src mass

        # compute new conv (copy)
        new_conv = cur_conv.copy()
        # linear term:
        new_conv[src:src+n] -= 2.0 * eps * cur_a
        new_conv[dst:dst+n] += 2.0 * eps * cur_a
        # quadratic term
        idx2s = 2*src
        idx2d = 2*dst
        new_conv[idx2s] += eps*eps
        new_conv[idx2d] += eps*eps
        new_conv[src+dst] -= 2.0*eps*eps
        # compute new max conv:
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max # since sum=1

        # acceptance
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                prob_accept = math.exp(-delta / T)
                if rng.random() < prob_accept:
                    accept = True
        if accept:
            # update a and conv
            cur_a[src] -= eps
            cur_a[dst] += eps
            cur_conv = new_conv
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[{time.time()-start_time:.2f}s] SA improve score={best_score:.6f}")
        # temperature decay
        if T > T_min:
            T = max(T_min, T * T_decay)

    return best_a, best_score

assistantfinal**Solution Overview**

The evaluator penalises large values of the discrete auto‑correlation  

\[
M(a)=\max_{k}(a*a)_k\; ,\qquad   
\text{score}(a)=\frac{2\,n\,M(a)}{(\sum a)^2}\; .
\]

All feasible vectors lie on the probability simplex (non‑negative, sum = 1).  
The algorithm consists of four independent phases that together explore the
search space much more aggressively than the reference implementation.

1. **Extended parametric sweep** – geometric, power, beta and exponential
   families are scanned on a fine grid, plus mixtures with the uniform
   distribution.

2. **Smoothed gradient descent** – we minimise the soft‑max version of the
   objective with an Adam optimiser.  The temperature of the soft‑max is
   exponentially decayed, giving a differentiable surrogate for the true
   `max`.

3. **Projected sub‑gradient descent** – a classic sub‑gradient method with
   Nesterov momentum refines the result of the smoothed phase.

4. **Simulated‑annealing polishing** – tiny “mass‑transfer’’ moves are applied
   to the current best vector.  The convolution is updated **incrementally**
   (no FFT) so millions of proposals can be examined in the remaining time.
   An annealing schedule allows occasional uphill moves, helping to escape
   shallow local minima.

The whole pipeline respects the 1000 s budget, prints progress messages and
finally returns a normalised list of non‑negative numbers compatible with
`evaluate_sequence`.  

```python
"""
Advanced search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of:
1. a comprehensive parametric sweep (geometric, power, beta, exponential);
2. smoothed gradient descent with Adam and a soft‑max approximation of the max;
3. projected sub‑gradient descent with Nesterov momentum;
4. a simulated‑annealing polish that updates the convolution incrementally.

All helpers are top‑level functions; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if np.any(mask):
        rho = np.nonzero(mask)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return the exact score for a (sum≈1) sequence."""
    conv = _conv_self_fft(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Extended parametric sweep
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, L_fft: int, deadline: float,
                     best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Explore a wide range of analytic families."""
    start = time.time()
    idx = np.arange(n)

    # ---- geometric (already in reference) ----
    eps_grid = np.logspace(-4, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 21):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power family (finer grid) ----
    p_grid = np.linspace(-1.0, 3.0, 81)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # ---- beta family (continuous grid) ----
    p_vals = np.arange(0.1, 4.1, 0.3)
    q_vals = np.arange(0.1, 4.1, 0.3)
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = np.power(idx + 1, p) * np.power(n - idx, q)
            s = beta_seq.sum()
            if s == 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, beta_seq.copy()
                print(f"[{time.time() - start:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ---- exponential family (pure and mixed) ----
    lam_grid = np.logspace(-4, -0.5, 30)
    for lam in lam_grid:
        if time.time() >= deadline - 30:
            break
        exp_seq = np.exp(-lam * idx)
        s = exp_seq.sum()
        if s == 0:
            continue
        exp_seq /= s
        # pure exponential
        sc = _eval_fast(exp_seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, exp_seq.copy()
            print(f"[{time.time() - start:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
        # mixture with uniform
        for alpha in np.linspace(0.0, 1.0, 11):
            mix = alpha * np.full(n, 1.0 / n) + (1 - alpha) * exp_seq
            mix = np.maximum(mix, 0.0)
            mix /= mix.sum()
            sc = _eval_fast(mix, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, mix.copy()
                print(f"[{time.time() - start:.2f}s] Mix exp λ={lam:.1e} α={alpha:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smoothed gradient descent with Adam
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smoothed objective:
        f(a) = 2 * n * softmax_tau(conv(a,a))
    The temperature τ is exponentially decayed.
    The iterate stays on the simplex via Euclidean projection.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    tau = 0.10                     # initial soft‑max temperature
    tau_min = 1e-8
    tau_decay = 0.9997
    step0 = 0.25                   # base step size (multiplied by Adam’s “α”)

    while time.time() < deadline:
        it += 1

        # ---------- convolution and soft‑max ----------
        conv = _conv_self_fft(a, L=L_fft)   # length 2n‑1
        maxc = conv.max()
        # numerically stable soft‑max
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()       # sum(w)=1

        # ---------- gradient (via FFT) ----------
        w_rev = w[::-1]                     # reversed weights
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps_adam)
        a = a - step * m_hat
        a = _project_simplex(a)

        # ---------- exact score ----------
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # ---------- temperature ----------
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Projected sub‑gradient descent (Nesterov momentum)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent with Nesterov momentum.
    Optimises   f(a)=2·n·max(conv(a,a))   under a∈simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    # Nesterov momentum parameters
    beta = 0.9
    momentum = np.zeros_like(a)
    step0 = 0.30
    it = 0
    L_grad = _next_pow2(3 * n)

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (one of them is enough for a sub‑gradient)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # Nesterov‑style update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Simulated‑annealing polish (incremental convolution update)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         conv: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves accepted with a Metropolis rule.
    The convolution is updated incrementally – O(n) per proposal.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_score = 2.0 * n * cur_conv.max()      # sum(cur_a) == 1
    best_a, best_score = cur_a.copy(), cur_score

    rng = np.random.default_rng()
    T = 5e-4                # initial temperature
    T_min = 1e-7
    decay = 0.9996

    it = 0
    while time.time() < deadline:
        it += 1

        # focus on the region that currently attains the maximum
        k = int(cur_conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        src_candidates = np.arange(lo, hi + 1, dtype=int)

        # source weighted by current mass (more likely to move mass that really
        # participates in the peak)
        probs = cur_a[src_candidates]
        prob_sum = probs.sum()
        if prob_sum > 0:
            probs = probs / prob_sum
            src = rng.choice(src_candidates, p=probs)
        else:
            src = rng.integers(lo, hi + 1)

        dst = rng.integers(n)
        if src == dst or cur_a[src] < 1e-14:
            continue

        # move at most 10 % of the source mass
        eps = cur_a[src] * rng.random() * 0.10

        # ----- compute the new convolution (incremental) -----
        new_conv = cur_conv.copy()
        # linear contribution 2*eps*(shift(dst)‑shift(src))
        new_conv[src:src + n] -= 2.0 * eps * cur_a
        new_conv[dst:dst + n] += 2.0 * eps * cur_a
        # quadratic contribution eps²
        new_conv[2 * src] += eps * eps
        new_conv[2 * dst] += eps * eps
        new_conv[src + dst] -= 2.0 * eps * eps

        new_max = new_conv.max()
        new_score = 2.0 * n * new_max

        # ----- Metropolis acceptance -----
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_a[src] -= eps
            cur_a[dst] += eps
            cur_conv = new_conv
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} best={best_score:.6f}")

        # temperature decay
        if T > T_min:
            T = max(T_min, T * decay)

    return best_a, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine mixes an extended parametric sweep, Adam‑based
    smoothed gradient descent, a sub‑gradient refinement and a
    simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)   # for exact evaluation
    L_grad = _next_pow2(3 * n)      # for gradient convolutions

    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ---- optionally start from a known construction ----
    try:

            else:

            sc = _eval_fast(starter, L_fft)
            if sc < best_score:

                print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:

    # ------------------------------------------------------------------
    # 1️⃣  Extended parametric sweep
    # ------------------------------------------------------------------
    best_seq, best_score = _parametric_sweep(

    # ------------------------------------------------------------------

        cand_seq, cand_score = _adam_smooth_descent(

            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after Adam restart score={best_score:.6f}")

    # 3️⃣  Projected sub‑gradient refinement (bulk of the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(
            best_seq, n, sub_deadline, start_time, best_score)
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.copy()

    # ------------------------------------------------------------------

        a_hc, sc_hc = _simulated_annealing(
            best_seq, cur_conv, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] SA final improve score={sc_hc:.6f}")

    # ------------------------------------------------------------------

    return [float(x) for x in best_seq]
```
