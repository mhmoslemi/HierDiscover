"""
Propose a step function (non‑negative vector that sums to 1) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Diverse initial pool** – uniform, Gaussian, exponential, Dirichlet,
   sparse supports and Beta‑shaped vectors are generated.  The best raw
   candidate (according to ``evaluate_sequence``) is kept.

2. **Soft‑max gradient descent** – we minimise a smooth approximation of the
   max‑convolution:
   fβ(a)= (1/β)·log Σk exp(β·(a∗a)k).  Its gradient is
   g = 2·(a_rev ∗ w) where w are the soft‑max weights.
   An additive (projected) gradient step on the probability simplex is used.
   The smoothing parameter β is slowly increased (annealing).

3. **Exponentiated‑gradient polishing** – the original sub‑gradient mirror
   descent (active shift only) is applied to the result of step 2; this
   often yields a further small improvement.

4. **Short random‑restart bursts** – while time permits, fresh random starts
   are drawn and a brief soft‑max optimisation is performed; any improvement
   replaces the current best.

5. **Deterministic greedy peak‑reduction** – finally a simple pairwise mass
   transfer is repeated until no strict improvement of the max‑convolution
   peak is possible.

All steps respect the 1000 s wall‑clock limit and work entirely with NumPy
(FFT based convolutions).  The returned list is clipped to ``[0,∞)`` and
normalised to sum 1, which is exactly what ``evaluate_sequence`` expects.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


def _beta_start(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shaped vector on [0,1] (alpha,beta>0)."""
    xs = np.linspace(0.0, 1.0, n)
    vec = (xs ** (alpha - 1)) * ((1 - xs) ** (beta - 1))
    if vec.sum() == 0:
        vec = np.full(n, 1.0 / n)
    else:
        vec = vec / vec.sum()
    return vec

def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on max_k (a*a)[k]."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _softmax_grad_optimize(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           fft_len: int,
                           beta0: float = 0.5,
                           beta_factor: float = 1.02,
                           step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected additive gradient descent on the smooth max:
        fβ(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

    The gradient is   g = 2·(a_rev ∗ w)   with   w_k ∝ exp(β·c_k).
    An annealing schedule slowly increases β.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    beta = beta0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)                     # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Softmax improve: iter={t:,}, score={cur_score:.6f}")

        # soft‑max weights (stable version)
        shifted = conv - conv.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient = 2 * (a_rev ∗ w)  (valid indices n‑1 … n‑1+n‑1)
        a_rev = a[::-1]
        grad_full = np.convolve(a_rev, w)                 # length 3n‑2
        g = 2.0 * grad_full[n - 1:n - 1 + n]

        # diminishing additive step
        step = step0 / math.sqrt(t)
        a = a - step * g
        # projection onto simplex
        a = _project_simplex(a)

        # increase smoothing
        beta = min(beta * beta_factor, 20.0)

        # occasional exact check
        if t % 400 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Shift a fraction of the mass from the most damaging coefficient."""
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0
    fft_len = _next_pow2(2 * n - 1)

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, fft_len).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score
