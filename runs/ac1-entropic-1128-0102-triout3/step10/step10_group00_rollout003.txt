"""
Propose a step‑function with a lower bound than 1.5087.

Algorithm
---------
1. Use a large canvas (n = 32768).  The continuous optimum of the
   problem is 1.5, and the discretisation error behaves roughly as O(1/n);
   with n = 32768 the theoretical bound is ≈ 1.502.

2. Generate a family of symmetric “tent‑with‑plateau’’ shapes
   a_i = f(i; p, α) where
        * p   – exponent controlling the curvature of the side ramps,
        * α   – relative width of a flat central plateau (0 ≤ α ≤ 0.5).
   The shape is built from two power‑ramps that meet a plateau of height 1.
   The vector is normalised to sum 1 (the objective is scale‑invariant).

   A cheap grid search over (p,α) together with a few random points
   yields a good starting distribution.

3. Refine the best start with projected sub‑gradient descent on the exact
   (non‑smooth) objective.  The sub‑gradient is taken from the index
   where the self‑convolution attains its maximum; after each step the
   vector is projected onto the probability simplex (Euclidean projection).

   Momentum (β = 0.9) and a 1/√t step‑size are used.  The loop runs until
   the time budget (≈ 1000 s) is exhausted.

4. A light simulated‑annealing polish moves tiny mass packets between
   randomly chosen positions; Metropolis acceptance is used with a fast
   exponential cooling schedule.

5. The final vector is clipped to non‑negative values, normalised to
   sum 1 and returned as a plain Python list.

All helper functions are defined at top level so that they can be
pickled if the harness needs to run the code in a child process.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    Identical to evaluate_sequence but without defensive checks.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex { w ≥ 0, Σ w = 1 }.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Shape generator: power‑ramps with an optional flat plateau
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric shape of length n.
    - p     : exponent of the side ramps (p > 0).
    - alpha : relative plateau width ∈ [0, 0.5]; alpha = 0 → pure tent,
              alpha = 0.5 → uniform distribution.
    The shape is normalised to sum 1.
    """
    if alpha < 0.0:
        alpha = 0.0
    if alpha > 0.5:
        alpha = 0.5

    # number of points belonging to the central plateau
    plateau_len = int(round(alpha * n))
    # make the plateau length odd when possible to keep exact symmetry
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        # pure uniform case
        shape = np.ones(n)
    else:
        # side ramp: values rise from 0 at the very edge to 1 at the
        # start of the plateau; we avoid a zero division by starting at 1
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p                     # shape on one side, from 0..1
        # Build symmetric shape
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # In rare rounding cases shape may be a few elements short/long
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    # Normalise to sum 1 (the objective is scale‑invariant)
    total = shape.sum()
    if total > 0:
        shape = shape / total
    else:
        shape = np.full(n, 1.0 / n)

    return shape


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent until ``deadline``.
    Returns the best iterate (and its score) seen so far.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Move tiny mass packets between random positions."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # Number of annealing attempts – chosen to fit the remaining time.
    max_steps = 200_000

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_steps)

        # choose source proportional to its mass (more likely to move)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move a random fraction (up to 25 %) of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) achieving a low
    value of the evaluation function.  Returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid crossing the hard limit
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a larger canvas brings the discrete bound closer to
    # the continuous optimum 1.5.
    # ------------------------------------------------------------------
    n = 32768  # power of two for fast FFTs

    # ------------------------------------------------------------------
    # 1) Sweep a modest family of analytic shapes.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    top_k = 5
    top_candidates = []   # list of (score, seq)

    # grid over exponent p and plateau fraction α
    p_grid = np.linspace(0.2, 2.0, 19)          # 0.2,0.3,…,2.0
    alpha_grid = np.linspace(0.0, 0.5, 11)      # 0.0,0.05,…,0.5

    for p in p_grid:
        for alpha in alpha_grid:
            a = _make_shape(n, p, alpha)
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
            # maintain a small pool of the best candidates
            if len(top_candidates) < top_k:
                top_candidates.append((score, a.copy()))
                top_candidates.sort(key=lambda x: x[0])
            else:
                if score < top_candidates[-1][0]:
                    top_candidates[-1] = (score, a.copy())
                    top_candidates.sort(key=lambda x: x[0])

            # occasional progress report
            if (time.time() - start_time) % 30 < 0.2:
                print(f"[{time.time() - start_time:.2f}s] "
                      f"grid p={p:.2f} α={alpha:.2f} → score={score:.6f}")

    # add a few random draws to diversify the pool
    rng = np.random.default_rng()
    for _ in range(200):
        if time.time() >= deadline:
            break
        p = rng.uniform(0.1, 2.5)
        alpha = rng.uniform(0.0, 0.5)
        a = _make_shape(n, p, alpha)
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        if len(top_candidates) < top_k:
            top_candidates.append((score, a.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if score < top_candidates[-1][0]:
                top_candidates[-1] = (score, a.copy())
                top_candidates.sort(key=lambda x: x[0])

    print(f"[{time.time() - start_time:.2f}s] "
          f"Best grid/random score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional: incorporate the globally stored starter (if any)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Global starter score = {cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    # also push into the top‑k pool
                    if len(top_candidates) < top_k:
                        top_candidates.append((cur, starter.copy()))
                        top_candidates.sort(key=lambda x: x[0])
                    else:
                        if cur < top_candidates[-1][0]:
                            top_candidates[-1] = (cur, starter.copy())
                            top_candidates.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Local refinement – run sub‑gradient descent on the best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:                     # keep a few seconds for polishing
        # allocate 80 % of the remaining budget to descent
        sub_deadline = time.time() + 0.80 * remaining
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score = "
                  f"{refined_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve "
                  f"score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return a simple Python list – the harness will feed it to evaluate_sequence.
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
