import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp_for_g(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    # for each k, iterate i where g_{k-i} appears: i from max(0, k-(n-1)) to min(n-1, k)
    for k in range(2*n - 1):
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        # for each i, variable index j = k-i
        # Actually coefficient for g_j is f_i, where j = k - i
        # but we treat each variable index j, not i.
        # equivalently for each i, we set entry for variable j = k-i.
        # So loop i loops over possible i, set column j = k-i with value f_i.
        for i in range(i0, i1+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    # constraint A @ g <= M
    # also g >= 0 implicit in bounds
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    # highs returns solution.x
    g = res.x
    return g
```

"""
Propose a step function minimizing ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Rich initial pool** – several handcrafted and random starts (uniform,
   Gaussian, exponential, Dirichlet with various concentrations, sparse
   supports).  The best raw candidate is kept.

2. **Smoothed exponentiated‑gradient (mirror‑descent)** – we minimise a
   smooth surrogate of the max‑convolution:
   `H(a)=log Σ_k exp(θ·(a*a)_k)`.  A soft‑max weight vector is built from the
   current convolution, the exact gradient `∇H` is obtained via FFT‑based
   convolution, and a multiplicative update `a_i←a_i·exp(−η·∇H_i)` keeps the
   vector on the simplex.  The temperature `θ` is increased gradually,
   while the stepsize `η` decays as `1/√t`.  This phase consumes ~80 % of
   the time budget.

3. **Short random‑restart bursts** – while time permits we draw fresh random
   starts and run a brief smoothed‑exp‑gradient run (≈ 5 s).  Any improvement
   updates the global best.

4. **Deterministic greedy peak‑reduction** – a final local search moves a
   modest fraction of the coefficient that contributes most to the current
   convolution peak towards the smallest coefficient, accepting only strict
   improvements.

5. **Safety post‑processing** – clip to non‑negative, renormalise to sum = 1
   and return the best sequence found.

The routine respects the 1000 s wall‑clock limit and works for any length.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _conv_general(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*b via FFT (full length)."""
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Af * Bf, n=fft_len)
    return conv[: a.shape[0] + b.shape[0] - 1]

# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Pick `m` indices uniformly, give each equal mass 1/m, zero elsewhere."""
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimisation
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             fft_len: int,
                             theta0: float = 0.5,
                             theta_max: float = 20.0,
                             eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max surrogate of the max‑convolution.
    Returns the best vector observed and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    theta = theta0

    # pre‑allocate a larger FFT length (covers both self‑conv and a*w_rev)
    fft_len_g = fft_len  # already ≥ 3·n‑2 in the caller

    while time.time() < deadline:
        t += 1

        # ----- convolution a*a (needed for the soft‑max) -----
        Af = np.fft.rfft(a, n=fft_len_g)
        conv_full = np.fft.irfft(Af * Af, n=fft_len_g)
        conv_self = conv_full[: 2 * n - 1]          # length 2n‑1

        # ----- soft‑max weights -----
        scaled = theta * conv_self
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # ----- gradient of the smooth surrogate -----
        w_rev = w[::-1]
        Bf = np.fft.rfft(w_rev, n=fft_len_g)
        conv_a_wrev = np.fft.irfft(Af * Bf, n=fft_len_g)
        # conv_a_wrev length = 3n‑2, we need entries (2n‑2‑i) for i=0..n‑1
        idx = 2 * n - 2 - np.arange(n)
        g = 2.0 * conv_a_wrev[idx]

        # ----- mirror‑descent update (exponentiated gradient) -----
        eta = eta0 / math.sqrt(t)
        upd = -eta * g
        # clip exponent to avoid overflow/underflow
        upd = np.clip(upd, -700.0, 700.0)
        a = a * np.exp(upd)

        # re‑normalise onto the simplex, re‑inject a tiny floor
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- occasional exact evaluation -----

                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] smooth‑expgrad iter={t:,} score={sc:.6f} theta={theta:.2f}"
                )

        # gradually increase temperature (more focus on the true max)
        if theta < theta_max:
            theta = min(theta * 1.02, theta_max)

    # final evaluation
    final_sc = evaluate_sequence(a.tolist())
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
        print(f"[{time.time() - start_time:6.2f}s] final smooth‑expgrad score={final_sc:.6f}")

    return best_a, best_score

def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Transfer a modest fraction from the coefficient that most contributes
    to the current convolution peak towards the smallest coefficient.
    Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    best_score = float("inf")
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # move a modest fraction
        if eps <= 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, fft_len).max()
        if new_max < cur_max:
            a = a_prop
            cur_score = 2.0 * n * new_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] greedy improve: iter={it:,} max={new_max:.6f}")

    # evaluate exact score of the final vector
    final_sc = evaluate_sequence(best_a.tolist())
    return best_a, final_sc
