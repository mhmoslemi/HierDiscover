Propose a step‑function with a very low evaluation score.

The strategy is a three‑stage search:

1. **Rich initialisation** – besides the usual uniform / exponential /
   Gaussian starts we also try discretised Beta‑distributions
   (`p_i ∝ x_i^{α‑1}(1‑x_i)^{β‑1}`) which put a lot of mass at the
   edges of the interval.  All candidates are normalised to a
   probability vector and the best raw score (exact ``evaluate_sequence``)
   is kept.

2. **Adam warm‑up on a smooth surrogate** – we minimise a
   log‑sum‑exp approximation of the peak of ``p∗p`` using Adam on the
   log‑probabilities.  The schedule is longer than in the reference code
   (six values of the smoothing parameter, 6000 iterations each) and
   the learning‑rate is slightly larger.

3. **Aggressive multiplicative mirror‑descent** – the true
   ``max(p∗p)`` is driven down by an exponentiated‑gradient update.
   Instead of the tiny active set used before we keep the top‑15 shifts
   whose convolution values are within 0.5 % of the current maximum.
   The initial step size is `η0 = 1.0` and decays as `1/√t`.  Every
   5000 iterations a tiny random perturbation is added if no improvement
   has been seen, helping to escape plateaus.

4. **Deterministic greedy polishing** – finally we repeatedly move a
   fraction of the coefficient that contributes most to the current

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution ``a*a`` (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc``."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6), normalised."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised."""
    xs = (np.arange(n) + 0.5) / n      # avoid exact 0 or 1
    # use np.power which handles non‑integer exponents
    p = np.power(xs, a - 1.0) * np.power(1.0 - xs, b - 1.0)
    # replace NaNs/Inf (possible when a<1 and xs≈0) by 0
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s == 0.0:
        p = np.full(n, 1.0 / n)
    else:
        p = p / s
    return p


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation that preserves total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv

                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score


# ----------------------------------------------------------------------
# Multiplicative mirror‑descent with a richer active set
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """Sub‑gradient of Σ_{k∈active} (a∗a)[k]."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eta0: float = 1.0,
    delta_frac: float = 0.005,
    top_k: int = 15,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution objective.
    Uses the top‑``top_k`` shifts whose values lie within ``delta_frac`` of
    the current maximum.  Random jitter is injected every 5000 steps if
    no improvement has been observed.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    last_improve_iter = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve_iter = t
            print(
                f"[{time.time() - start_time:6.2f}s] Multi‑ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ------- active set -------------------------------------------------
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            # keep only the largest ``top_k`` entries
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ------- gradient ----------------------------------------------------
        g = _multi_gradient(a, active)

        # ------- step‑size ---------------------------------------------------
        eta = eta0 / math.sqrt(t)

        # ------- exponentiated update ----------------------------------------
        a = a * np.exp(-eta * g)

        # ------- re‑normalise ------------------------------------------------
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # protect against exact zero (helps stability)
        a = np.maximum(a, 1e-15)

        # ------- occasional perturbation ------------------------------------
        if t % 5000 == 0 and (t - last_improve_iter) > 3000:
            # add a tiny random jitter and renormalise
            jitter = np.random.randn(n) * 0.001
            a = np.maximum(best_a + jitter, 1e-15)
            a = a / a.sum()
            last_improve_iter = t  # reset counter to avoid massive jitter loops

        # occasional exact evaluation
        if t % 1000 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve_iter = t

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy local polishing (same idea as the reference)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Move a fraction of the coefficient that most contributes to the
    current peak towards the smallest coefficient, accepting only strict
    improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.35 * a[src]          # a larger fraction than the reference
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver -----------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of a diversified initialisation, an Adam warm‑up,
    a fast exponentiated‑gradient polishing with a richer active set,
    and a final greedy refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size (keep the reference size, but you may change it)
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of starting points
    # ------------------------------------------------------------------
    starts = []

    # 0) previous best, if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 1) handcrafted seeds
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_exponential_start(n, r=0.9995))    # gentle decay
    starts.append(_exponential_start(n, r=0.985))     # stronger decay
    starts.append(_gaussian_start(n))                 # bell‑shaped
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # 2) random Dirichlet draws with various concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # 3) discretised Beta shapes (U‑shaped when α=β<1)
    beta_params = [0.3, 0.4, 0.5, 0.6, 0.8, 1.0]
    for a in beta_params:
        for b in beta_params:
            starts.append(_beta_shape(n, a, b))

    # ------------------------------------------------------------------
    # Evaluate raw seeds and keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:02d} raw score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam warm‑up (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.15
    cand_seq, cand_score = _adam_optimize(
        best_seq, phase1_deadline, start_time, fft_len, iters_per_alpha=6000
    )
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – Multi‑gradient exponentiated‑gradient polishing
    #          (≈75 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.90
    polished_seq, polished_score = _expgrad_multi_opt(
        best_seq,
        phase2_deadline,
        start_time,
        fft_len,
        eta0=1.0,
        delta_frac=0.005,
        top_k=15,
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polishing score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic greedy peak reduction (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
