"""
Propose a non‑negative step‑function (list of heights) that minimizes

    evaluate_sequence(seq)  =  2·n·max(conv(seq,seq)) / (sum(seq)**2)

for n = 4096.

The algorithm works in three phases:

1️⃣  Adam‑based smooth‑max mirror descent – a smooth approximation of the
    infinity‑norm of the self‑convolution is minimised.  The smoothing
    parameter α is increased geometrically; per‑parameter Adam updates
    (with projection onto the probability simplex) are used.

2️⃣  Adam‑accelerated sub‑gradient descent on the true objective.
    The sub‑gradient is the average of the gradients of all (near) maximal
    convolution entries.  Adam gives adaptive step‑sizes and momentum.

3️⃣  A very short simulated‑annealing “mass‑transfer’’ polish to escape
    remaining tiny local‑optima.

The three phases are allocated a fixed portion of the total budget
(≈25 % smooth, ≈60 % sub‑gradient, the rest annealing) and the code never
exceeds the given wall‑clock budget.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------
# Helper routines (FFT‑based convolutions, simplex projection)
# ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ---------------------------------------------------------
# Phase 1 – Adam smooth‑max mirror descent
# ---------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         L_grad: int,
                         deadline: float,
                         start_time: float,
                         best_score_init: float) -> tuple[np.ndarray, float]:
    """Minimise a smooth approximation of max(conv) using Adam."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_init

    # geometric schedule for the smoothing parameter
    alphas = np.geomspace(0.5, 300.0, num=30)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.2

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        # re‑initialise Adam for each α (helps stability)
        m.fill(0.0)
        v.fill(0.0)
        beta1_pow = 1.0
        beta2_pow = 1.0

        while time.time() < deadline - 0.2:
            it += 1
            beta1_pow *= beta1
            beta2_pow *= beta2

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights  w ∝ exp(α·conv)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # shift for stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient of the smooth max (see derivation in comment)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update + projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            step = base_lr / math.sqrt(it)
            a = a - step * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a)

            # occasional exact‑objective check
            if it % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time()-start_time:.2f}s] Smooth‑Adam α={alpha:.2f} it={it:,} score={cur_score:.6f}")

            if time.time() >= deadline:
                break

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ---------------------------------------------------------
# Phase 2 – Adam sub‑gradient descent on the true objective
# ---------------------------------------------------------
def _adam_subgradient_descent(a0: np.ndarray,
                              n: int,
                              L_self: int,
                              L_grad: int,
                              deadline: float,
                              start_time: float,
                              best_score_init: float) -> tuple[np.ndarray, float]:
    """Projected Adam sub‑gradient descent on 2·n·max(conv)/sum²."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_init

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.1

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    while time.time() < deadline:
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _conv_self(a, L=L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam‑subgrad it={it:,} score={cur_score:.6f}")

        # gather all (near) maximal indices (numerical tolerance)
        max_tol = 1e-12
        idx_max = np.flatnonzero(conv >= cur_max - max_tol)

        # sub‑gradient: average over all active indices
        grad = np.zeros_like(a)
        for k in idx_max:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad = grad / max(1, len(idx_max))

        # Adam update + simplex projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = base_lr / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_score


# ---------------------------------------------------------
# Phase 3 – Tiny simulated‑annealing polish
# ---------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling schedule
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # pick source according to current mass distribution
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # transfer a random fraction of source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

        best_seq, best_score = _adam_subgradient_descent(
            best_seq, n, L_self, L_grad, subgrad_deadline, start_time, best_score)
        print(f"[{time.time()-start_time:.2f}s] After Adam sub‑grad best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Light annealing polish (uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _annealing(
            best_seq, n, deadline, start_time,
            best_score, best_seq,
            start_temp=1e-2, end_temp=1e-7, steps=300_000)
        print(f"[{time.time()-start_time:.2f}s] Final anneal best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
