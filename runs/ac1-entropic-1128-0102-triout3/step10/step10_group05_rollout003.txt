g = cp.Variable(n, nonneg=True)

conv_constraints = []
for k in range(2*n-1):
    # compute linear expression c_k = sum_i f[i] * g[k - i] for i with valid indices
    # This is a dot product of vector f and shifted g.
    # But building each as sum over i is O(n^2). With n=512, that's okay.
    expr = 0
    lo = max(0, k-(n-1))
    hi = min(n-1, k)
    # sum over i: f[i] * g[k-i] = sum_{i=lo..hi} f[i] * g[k-i]
    # For each i, g index j = k-i
    # So we iterate i.
    for i in range(lo, hi+1):

    conv_constraints.append(expr <= M)

objective = cp.Maximize(cp.sum(g))

prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1000)

"""
Propose a step‑function sequence with a low value of ``evaluate_sequence``.

The algorithm works in two stages.

Stage 1 – small scale convex‑programming refinement.
    * Use a modest length (n≈256) so that a linear program can be solved quickly.
    * Initialise with an arcsine (Beta(½,½)) distribution – a shape known to give a
      relatively flat self‑convolution.
    * Repeatedly solve the LP
          maximise   sum(g)
          subject to f * g ≤ M   (component‑wise)
                     g ≥ 0
      where * denotes linear convolution and M = max(f*f).  The solution g pushes
      mass into positions that do not increase the current peak.  After each LP
      step we mix the old vector with the new one and keep the best seen.

Stage 2 – up‑sample and smooth on the full length (n=4096).
    * Interpolate the best small‑scale vector to the target length.
    * Apply a short Adam optimisation on a smooth (log‑sum‑exp) surrogate of
      the max‑convolution to polish the result.
    * Finish with a few greedy “peak‑reduction’’ moves that shift a tiny amount
      of mass from the most responsible coordinate to the least responsible one.

Both stages respect the global time budget (default 1000 s).  The final
sequence is normalised to have unit sum (scaling does not affect the objective)
and returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a using FFT.
    Returns the full linear convolution of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Return a discretised Beta(½,½) (arcsine) density on {0,…,n‑1}
    and normalise it to sum 1.
    """
    i = np.arange(n, dtype=float)
    # avoid zero denominator by using (i+0.5) and (n‑i‑0.5)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _local_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective (identical to ``evaluate_sequence``
    up to the clamping performed there).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-8:
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Stage 1 – LP driven flattening on a small support
# ----------------------------------------------------------------------
def _lp_flatten_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    One LP iteration: maximise sum(g) subject to
        f * g ≤ M  (linear convolution, component‑wise)
        g ≥ 0
    Returns the optimal g (or the original f if the LP fails).
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the (rows × n) convolution matrix A where (A g)_k = sum_i f_i * g_{k-i}
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f[i]

    # Linear program
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(rows)]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=200, verbose=False)
    except Exception:  # fallback in case the solver crashes
        return f.copy()

    if g.value is None:
        # Solver failed – return the original vector
        return f.copy()

    g_opt = np.maximum(g.value, 0.0)
    return g_opt


def _stage1_small_lp(initial_n: int, deadline: float, start_time: float):
    """
    Perform a sequence of LP refinements on a small vector.
    Returns the best vector found (normalised to sum 1) and its score.
    """
    # start from the arcsine distribution
    a = _arcsine_sequence(initial_n)
    best_a = a.copy()
    best_sc = _local_score(best_a)

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # current worst convolution value
        L = _next_pow2(2 * a.shape[0] - 1)
        conv = _conv_self(a, L)
        M = conv.max()

        # LP step
        g = _lp_flatten_step(a, M)

        # Mix old and new (t controls the step size)
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        sc = _local_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter={iteration:,} score={sc:.6f}")

        # Prepare for next iteration
        a = a_new

        # stop after a reasonable number of iterations (prevent endless loop)
        if iteration >= 30:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stage 2 – Adam smoothing on the full size
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 5000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth (log‑sum‑exp) surrogate of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    n = a.shape[0]
    Lfft = _next_pow2(2 * n - 1)

    # initial score
    conv = _conv_self(a, Lfft)
    best_score = 2.0 * n * conv.max()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, Lfft)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: iter={it:,} score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=Lfft)
            Wrev = np.fft.rfft(w_rev, n=Lfft)
            corr = np.fft.irfft(Af * Wrev, n=Lfft)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


def _greedy_peak_reduction(a0: np.ndarray,
                           deadline: float,
                           start_time: float,
                           max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy reductions of the current peak.
    Moves a small amount of mass from the index that contributes most to the
    maximal convolution entry to an index that contributes the least.
    """
    a = a0.copy()
    n = a.shape[0]
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, Lfft)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # locate the contributing pair with the largest product a[i]*a[max_idx-i]
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        prod = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(prod)]

        # partner values for src in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0

        # build the vector a_rev where a_rev[i] = a[max_idx - i]
        a_rev = np.zeros_like(a)
        a_rev[idx] = a[max_idx - idx]

        # pick destination where a_rev is minimal (i.e. contributes least)
        dst_candidates = np.where(a_rev == a_rev.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = (src + 1) % n
        else:
            dst = np.random.choice(dst_candidates)

        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta = partner_src - partner_dst

        # step size – try to cut the peak but stay safe
        if delta > 0:
            eps = min(a[src], (max_val - conv.max()) / (2.0 * delta + 1e-12))
        else:
            eps = min(a[src], 0.01 * a[src])

        eps = max(eps, 1e-12)
        if eps <= 0:
            continue

        # apply the move (exact update of convolution)
        a_old = a.copy()
        # shift mass
        a[src] -= eps
        a[dst] += eps

        # update convolution incrementally
        # contributions of src disappear, of dst appear
        conv[src:src + n] -= 2.0 * eps * a_old
        conv[dst:dst + n] += 2.0 * eps * a

        # stay non‑negative
        a = np.maximum(a, 0.0)

        # periodic exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: iter={it:,} score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``
    under the given time budget (default 1000 s).  Returns the best sequence found
    as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Stage 1 – small‑scale LP flattening (n≈256)
    # ------------------------------------------------------------------
    small_n = 256
    stage1_deadline = min(deadline, start_time + 120.0)  # give ~2 min to LP stage
    best_small, best_small_score = _stage1_small_lp(small_n, stage1_deadline, start_time)
    print(f"[{time.time() - start_time:.2f}s] stage 1 best score={best_small_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to the target length and polish
    # ------------------------------------------------------------------
    big_n = 4096
    # linear interpolation to the larger size
    x_small = np.linspace(0.0, 1.0, small_n)
    x_big = np.linspace(0.0, 1.0, big_n)
    a_big = np.interp(x_big, x_small, best_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    # Adam smoothing (light weight)
    if time.time() < deadline:
        phase2_deadline = min(deadline, start_time + 300.0)  # up to ~5 min
        alphas = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2]
        a_big, big_score = _run_adam_smoothing(
            a_big,
            phase2_deadline,
            start_time,
            alphas=alphas,
            lr0=0.6,
            iters_per_alpha=8000,
        )
        print(f"[{time.time() - start_time:.2f}s] adam polish score={big_score:.6f}")

    # Greedy peak reduction (very cheap, many cheap iterations)
    if time.time() < deadline:
        phase3_deadline = deadline
        a_big, big_score = _greedy_peak_reduction(
            a_big,
            phase3_deadline,
            start_time,
            max_iters=2_000_000,
        )
        print(f"[{time.time() - start_time:.2f}s] final greedy score={big_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise the result and return a plain Python list
    # ------------------------------------------------------------------
    a_big = np.maximum(a_big, 0.0)
    total = a_big.sum()
    if total > 0.0:
        a_big = a_big / total
    else:
        a_big = np.full(big_n, 1.0 / big_n)

    final_obj = _local_score(a_big)
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in a_big]
