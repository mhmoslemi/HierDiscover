def _lp_optimize(f, M, active_idx):
    n = f.shape[0]
    K = len(active_idx)
    # Build A matrix
    A = np.zeros((K, n))
    for row, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        cols = np.arange(lo, hi + 1)
        A[row, cols] = f[k - cols]  # since f index is k-j
    b = np.full(K, M)
    c = -np.ones(n)  # maximize sum(g)
    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a step function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Diverse starts** – a few handcrafted sequences (uniform,
   Gaussian, exponential) plus a few random Dirichlet / sparse draws.
2. **Evaluation** – keep the best start (objective = 2·n·max(conv)).
3. **LP‑refinement** – for the current best vector `a`
   * compute the self‑convolution `c = a*a`;
   * take the `K` largest entries of `c` (active shifts);
   * solve a tiny linear program
        maximise   Σ g_j
        s.t.       (a * g)[k] ≤ M   for every active shift `k`
                   g_j ≥ 0
     (here `M = max(c)` and the convolution `(a*g)[k]` is linear in `g`);
   * normalise `g` to sum‑1 and mix a little:
        `a ← (1‑t)·a + t·g`   (t≈0.2) and renormalise.
   * repeat while we have time – each LP uses at most a few dozen
     constraints, so it is cheap.
4. **Exponentiated‑gradient polishing** – the standard mirror‑descent
   update as used in the previous best solution, run on the remaining
   time budget.
5. **Greedy peak reduction** – a deterministic local move that shifts a
   fraction of the coefficient that contributes most to the current
   convolution peak toward the smallest coefficient.
6. Return the best vector seen, clipped to `[0,∞)` and normalised to
   sum‑1.

The LP‑step tends to push mass away from positions that create the
largest convolution peaks, which complements the exponentiated‑gradient
phase and often breaks the plateaus that limited the earlier approach.
The whole routine respects the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT helpers (full linear convolution a*a)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Simple score – we keep the vector normalised to sum‑1, so
#   score = 2·n·max(conv)
# ----------------------------------------------------------------------


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Linear‑program step
# ----------------------------------------------------------------------


def _lp_optimize(f: np.ndarray,
                 M: float,
                 active_idx: np.ndarray,
                 t_mix: float = 0.2) -> np.ndarray | None:
    """
    Solve the tiny LP:
        maximise   Σ g_j
        s.t.       (f * g)[k] ≤ M   for every k in active_idx
                   g_j ≥ 0
    Returns the mixed vector a' = (1‑t)·f + t·g / Σg,
    or None if the LP fails.
    """
    n = f.shape[0]
    K = active_idx.size
    if K == 0:
        return None

    # Build the (K × n) constraint matrix.
    # Row for shift k has entry f[k‑j] at column j (if 0≤k‑j<n).
    A = np.zeros((K, n), dtype=float)
    for row, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        cols = np.arange(lo, hi + 1, dtype=int)
        A[row, cols] = f[k - cols]

    b = np.full(K, M, dtype=float)

    # maximise Σ g_j  →  minimise −Σ g_j
    c = -np.ones(n, dtype=float)

    # bounds: g_j ≥ 0
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    g_norm = g / s
    # mixing
    a_new = (1.0 - t_mix) * f + t_mix * g_norm
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (same as the previous best solution)
# ----------------------------------------------------------------------


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on max‑convolution."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    cur_conv = _conv_self(a, fft_len)
    best_score = _score_from_conv(cur_conv, n)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # sub‑gradient w.r.t. the *single* shift attaining the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact check (cheap guard against drift)
        if t % 500 == 0:
            sc = _score_from_conv(_conv_self(a, fft_len), n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic improvement (peak reduction)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Move a fraction of the coefficient that builds the current peak
    towards the smallest coefficient, accepting only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)

    best_score = _score_from_conv(_conv_self(a, fft_len), n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # smallest coefficient (ignore src itself)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # modest fraction
        if eps <= 1e-14:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, fft_len).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec
