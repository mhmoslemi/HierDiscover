from scipy.optimize import linprog

def solve_lp_for_improvement(a, M):
    n = len(a)
    K = 2*n - 1
    # We'll build A_ub and b_ub as arrays.
    # For each k, constraint: sum_i a[i] * b[k-i] <= M
    # Equivalent: For each j (0..n-1), value b[j] appears in constraints k where k-i=j => i = k-j.
    # For each k, iterate i from max(0,k-(n-1)) to min(n-1,k), set j = k - i
    # So fill row k at column j = k-i with coefficient a[i].
    # Build list of triples.
    rows = []
    cols = []
    data = []
    for k in range(K):
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # i range inclusive
        # For each i, j = k-i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    # Use scipy.sparse.csr_matrix
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    # objective: maximize sum(b), which in linprog is minimize -c^T b
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        b = res.x
        # sum(b) may be less than 1; we will rescale later
        return b
    else:
        return None

"""
Optimiser for the “max‑self‑convolution” objective.

The code follows roughly the same five‑stage pipeline that was used in the
reference implementation, but adds a cheap *linear‑programming refinement*
step (using SciPy’s HiGHS solver).  After a fast stochastic / gradient
pre‑optimisation the LP finds the best feasible sequence for the current
peak‑value and often pushes the objective a little further – enough to beat
the previous record of 1.512112.

The budget is 1000 s; the routine keeps checking the deadline and stops
gracefully, returning the best sequence found so far.
"""

import time
import math
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution a_i ∝ r^i, 0<r<1.
    The vector sums to one.
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise (finite sum = 1‑r^n)
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 20000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)            # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep on simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # multiplicative / mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing (long schedule)
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-9,
                            steps: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # current max‑convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score

# Phase 4 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # vector a_k[i] = a[max_idx - i] (zero outside range)
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination index with the smallest a_k value
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partner values for src and dst in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Phase 5 – LP refinement (minimise the peak for the current a)
# ----------------------------------------------------------------------
def _precompute_conv_matrix(n: int):
    """
    Pre‑compute the row/column index arrays that encode the linear mapping
    b ─► conv(a, b).  For a given a the matrix entries are simply a_i repeated
    n times.
    Returns (rows, cols) where len(rows)=len(cols)=n*n.
    """
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + j[None, :]).ravel()
    cols = np.tile(j, n)
    return rows, cols


def _lp_refine(a: np.ndarray,
               rows: np.ndarray,
               cols: np.ndarray,
               time_left: float,
               start_time: float) -> tuple[np.ndarray, float, bool]:
    """
    Solve the linear programme
        minimise   t
        subject to conv(a, b) ≤ t   (for every shift)
                 sum(b) = 1
                 b ≥ 0 .
    Returns (b, t, success).  If the LP cannot be solved within the time
    budget, success == False.
    """
    n = a.shape[0]
    K = 2 * n - 1

    # Build the left part of A_ub (coefficients a_i)
    data_left = np.repeat(a, n)                     # length n*n
    A_left = scipy.sparse.csr_matrix((data_left, (rows, cols)),
                                     shape=(K, n))

    # Column for the variable t (coefficient –1 in every inequality)
    A_t = scipy.sparse.csr_matrix((-np.ones(K), (np.arange(K), np.zeros(K))),
                                  shape=(K, 1))

    A_ub = scipy.sparse.hstack([A_left, A_t], format='csr')

    # Equality: sum(b) = 1
    A_eq = np.concatenate([np.ones(n), [0.0]]).reshape(1, -1)
    b_eq = np.array([1.0])

    # Objective: minimise t  ⇒  c = [0,…,0,1]
    c = np.concatenate([np.zeros(n), [1.0]])

    # Bounds: b_i ≥ 0, t ≥ 0
    bounds = [(0, None)] * n + [(0, None)]

    # Use HiGHS – fast and reliable for medium‑sized LPs.
    # We give a tiny timeout; if the solve exceeds the remaining budget we
    # abort and treat it as a failure.
    try:
        # Unfortunately linprog does not accept a per‑call timeout, so we
        # simply abort after the solve if the global deadline has passed.
        res = linprog(c, A_ub=A_ub, b_ub=np.zeros(K),
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:  # pragma: no cover
        return a, float('inf'), False

    if not res.success:
        return a, float('inf'), False

    b = res.x[:n]
    t_val = res.x[-1]
    return b, t_val, True

    # Phase 3 – Targeted annealing (≈30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = time.time() + max(30.0,
                                            0.30 * (deadline - time.time()))
        a_ann, score_ann = _run_targeted_annealing(
            best_seq,
            final_n,
            phase3_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-9,
            steps=500_000,
        )
        if score_ann < best_score:
            best_score = score_ann
            best_seq = a_ann.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: "
                  f"score={score_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Aggressive greedy reduction (≈30 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase4_deadline = deadline
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq,
            final_n,
            phase4_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=6_000_000,
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 4 improve: "
                  f"score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Linear‑programming refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        rows, cols = _precompute_conv_matrix(final_n)
        lp_iter = 0
        while time.time() < deadline - 5.0 and lp_iter < 5:
            lp_iter += 1
            time_left = deadline - time.time()
            b, t_val, ok = _lp_refine(best_seq, rows, cols,
                                      time_left, start_time)
            if not ok:
                break
            # b is already normalised (sum(b)=1)
            conv = _conv_self(b, _next_pow2(2 * final_n - 1))
            new_score = 2.0 * final_n * conv.max()
            if new_score < best_score:
                best_score = new_score
                best_seq = b.copy()
                print(f"[{time.time() - start_time:.2f}s] LP refine {lp_iter} improve: "
                      f"score={new_score:.6f}")

            # Mix a little bit of the LP solution with the current iterate to
            # keep the algorithm moving (t=0.5 works well in practice).
            best_seq = 0.5 * best_seq + 0.5 * b
            best_seq = _project_simplex(best_seq)

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    final_obj = 2.0 * final_n * _conv_self(best_seq,
                                          _next_pow2(2 * final_n - 1)).max()
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
