    constraints = []

    for k in range(2*n-1):
        # sum_i f_i * b_{k-i}

"""
Alternating Linear‑Program (ALP) optimiser for the step‑function challenge.

The algorithm proceeds as follows:

1.  Initialise a non‑negative simplex vector using a quick grid search
    over geometric decay factors (the “exponential seed”).

2.  Repeatedly improve the sequence by solving a linear programme:
    for the current distribution `a` we compute
        M = max (a * a) .
    Keeping the peak `M` fixed we maximise the total mass of a new
    vector `b` subject to the linear constraints
        (a * b)_k ≤ M   for every convolution index k.
    The LP is solved with CVXPY (ECOS).  The optimum `b` is renormalised
    to sum to one, then mixed into the current iterate
        a ← (1‑t)·a + t·b
    with a back‑tracking line search on `t`.  Each successful mix
    strictly reduces the objective `2·n·max(conv)`.

3.  When the time budget is almost exhausted a deterministic greedy
    sweep (the same routine used in the previous state‑of‑the‑art
    optimiser) is applied to shave off the last bits of the peak.

The whole routine respects the 1000 s time limit, prints occasional
progress messages, and finally returns a plain Python list of floats.
"""

import time
import math
import numpy as np
import sys

# Optional imports – if a solver is unavailable we fall back to the older method.
try:
    import cvxpy as cp
    _HAVE_CVX = True
except Exception:                     # pragma: no cover
    _HAVE_CVX = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a using FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution a_i ∝ r^i (0<r<1) on n points.
    The vector is automatically normalised to sum 1.
    """
    if not (0.0 < r < 1.0):
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    norm = 1.0 - r ** n                     # = sum_{i=0}^{n-1} r^i
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    a /= a.sum()                            # safety normalisation
    return a


# ----------------------------------------------------------------------
# Linear‑programmer (maximise total mass under peak constraint)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float, timeout: float):
    """
    Solve:
        maximise  Σ b_i
        subject to  (f * b)_k ≤ M   for all k
                    b_i ≥ 0
    Returns the optimal b (as a NumPy array) or None on failure.
    """
    n = f.shape[0]
    # Build CVXPY problem (conv is linear in the variable b)
    b = cp.Variable(n, nonneg=True)
    f_const = cp.Constant(f)
    conv_fb = cp.conv(f_const, b)                # length 2·n‑1
    constraints = [conv_fb <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS does not support an explicit timeout; we rely on the outer budget.
    try:
        prob.solve(solver=cp.ECOS, verbose=False, warm_start=False,
                   # mild tolerance to accelerate convergence
                   abstol=1e-9, reltol=1e-9, feastol=1e-9,
                   max_iters=5000)
    except Exception:                            # pragma: no cover
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    b_val = b.value
    if b_val is None:
        return None
    # numerical clean‑up
    b_val = np.maximum(b_val, 0.0)
    return b_val


# ----------------------------------------------------------------------
# Greedy reduction (identical to the previous implementation)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build vector a_k[i] = a[max_idx - i] (zero outside the range)
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination index with the smallest a_k value
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partner values for src and dst in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The method combines an exponential seed, alternating linear‑programming,
    and a final greedy sweep.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0. Initialise – exponential decay (replicates the earlier good seed)
    # ------------------------------------------------------------------
    best_r = 0.99
    best_score = float("inf")
    for r in np.linspace(0.90, 0.9999, 250):
        cand = _exponential_sequence(n, r)
        conv = _conv_self(cand, L_fft)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_r = r
    a = _exponential_sequence(n, best_r)
    cur_score = 2.0 * n * _conv_self(a, L_fft).max()
    print(f"[0.00s] exponential seed r={best_r:.6f}, score={cur_score:.6f}")

    best_a = a.copy()
    best_obj = cur_score

    # ------------------------------------------------------------------
    # 1. Alternating LP phase
    # ------------------------------------------------------------------
    if _HAVE_CVX:
        lp_iter = 0
        while time.time() < deadline - 30.0:    # keep ~30 s for the final sweep
            lp_iter += 1
            conv = _conv_self(a, L_fft)
            M = conv.max()

            # Solve the LP; give each LP at most 10 s (hard timeout handled
            # by the outer deadline).
            remain = deadline - time.time()
            if remain < 5.0:
                break
            b = _solve_lp_max_sum(a, M, timeout=min(10.0, remain))
            if b is None:
                print(f"[{time.time() - start_time:.2f}s] LP failed at iter {lp_iter}")
                break

            total_b = b.sum()
            if total_b <= 1e-12:
                break
            g = b / total_b                     # normalised new candidate

            # back‑tracking line search on the mixing parameter t
            improved = False
            for t in [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new /= a_new.sum()
                conv_new = _conv_self(a_new, L_fft)
                new_score = 2.0 * n * conv_new.max()
                if new_score < best_obj - 1e-9:
                    a = a_new
                    best_a = a_new.copy()
                    best_obj = new_score
                    improved = True
                    print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter} improve: "
                          f"t={t:.5f}, score={new_score:.6f}")
                    break
            if not improved:
                # unable to improve – exit the LP phase
                print(f"[{time.time() - start_time:.2f}s] LP phase stagnated after {lp_iter} iterations.")
                break
    else:
        print("[warning] CVXPY not available – skipping LP phase (fallback to greedy only).")

    # ------------------------------------------------------------------
    # 2. Final greedy reduction – use remaining time budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        a_greedy, score_greedy = _run_greedy_reduction(
            best_a,
            n,
            deadline,
            start_time,
            best_obj,
            best_a,
            max_iters=5_000_000,
        )
        if score_greedy < best_obj:
            best_obj = score_greedy
            best_a = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy improve: score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    final_obj = 2.0 * n * _conv_self(best_a, L_fft).max()
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_a]
