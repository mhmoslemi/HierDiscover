"""
Improved step‑function optimiser.

The optimiser works in four stages:

1.  Exhaustive grid search for a pure exponential decay.  The grid is
    refined around the best value and the objective (2·n·max(conv))
    is evaluated using an FFT‑based convolution.

2.  Randomised search over low‑dimensional families:
    * mixtures of two exponentials,
    * polynomial decays a_i ∝ (1‑i/(n‑1))^p,
    * triangular (linear) decay with a finite support.
    Several thousand candidates are examined; the best one is kept.

3.  Adam optimisation of a smooth approximation of the max‑convolution.
    The smoothness parameter (β) is increased gradually, the learning‑rate
    follows a 1/√t schedule and the iterate is projected onto the simplex.
    This stage consumes the bulk of the time budget.

4.  Fast greedy hill‑climbing: repeatedly move a small amount of mass
    from the index that contributes most to the current peak to a
    location that reduces the peak the most (estimated from the linear
    model of the convolution).  The move size is adaptively chosen,
    the convolution is updated incrementally, and the best solution is
    recorded.

The whole routine respects the supplied time budget (default 1000 s),
prints progress messages and finally returns the best sequence found as
a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Self‑convolution a*a using FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Families of candidate sequences
# ----------------------------------------------------------------------
def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric sequence a_i ∝ r**i  (0 < r < 1).
    """
    if r <= 0.0 or r >= 1.0:
        raise ValueError("r must be in (0,1)")
    # geometric series a_i = (1-r) r**i  (infinite sum =1)
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # renormalise finite sum
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


def _mix_two_exponentials(n: int, r1: float, r2: float, p: float) -> np.ndarray:
    """
    Mixture p·exp(r1) + (1-p)·exp(r2).  Each component is already normalised,
    therefore the mixture sums to one.
    """
    a1 = _exponential_sequence(n, r1)
    a2 = _exponential_sequence(n, r2)
    return p * a1 + (1.0 - p) * a2


def _poly_decay_sequence(n: int, power: float) -> np.ndarray:
    """
    a_i ∝ (1 - i/(n-1))**power  on the interval [0,n-1].
    The vector is normalised to sum 1.
    """
    if power < 0.0:
        raise ValueError("power must be non‑negative")
    idx = np.arange(n, dtype=float)
    a = (1.0 - idx / (n - 1)) ** power
    a[a < 0.0] = 0.0
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _triangular_sequence(n: int, support: int) -> np.ndarray:
    """
    Linear decreasing sequence that is zero after `support` entries.
    The non‑zero part is a triangle that sums to one.
    """
    if support <= 0 or support > n:
        raise ValueError("invalid support")
    a = np.zeros(n, dtype=float)
    # values: a[i] = 2*(support-i)/support/(support+1) for i<support
    denom = support * (support + 1) / 2.0
    for i in range(support):
        a[i] = (support - i) / denom
    return a


# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, L: int, n: int) -> float:
    """
    Return 2·n·max(conv(a,a)).  The input is assumed to be non‑negative and
    sum(a)≈1 (normalised).  L is the FFT length (pre‑computed for speed).
    """
    conv = _conv_self(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   alphas,
                   lr0: float = 0.6,
                   iters_per_alpha: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp (smooth max) proxy.
    Returns the best vector seen and its unsmoothed objective.
    """
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = _objective(best_a, L, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)   # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto the simplex
            a = _project_simplex(a)

    final_score = _objective(a, L, n)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing (pairwise mass transfer)
# ----------------------------------------------------------------------
def _greedy_hill_climb(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       start_time: float,
                       cur_best_score: float,
                       cur_best_seq: np.ndarray,
                       max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Move a little mass from the dominant contributor of the current peak
    to a position that reduces the peak.  The convolution is updated
    incrementally; the best vector is recorded.
    """
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, Lfft)               # full convolution
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 2000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # contribution of each coefficient to this peak
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src (the coefficient it meets at the peak)
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0

        # choose destination: index where a[max_idx - j] is smallest
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]          # partner values for the peak
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            # fallback – pick a random index different from src
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # size of the transfer; heuristic based on the gap to the second peak
        second_max = np.partition(conv.ravel(), -2)[-2]  # second largest value
        gap = max_val - second_max
        if delta_a > 0.0 and gap > 0.0:
            eps_target = gap / (2.0 * delta_a)
            eps = min(a[src], max(1e-12, 0.9 * eps_target))
        else:
            eps = min(a[src], max(1e-12, 0.02 * a[src]))

        if eps <= 0.0:
            continue

        # ----- incremental convolution update -----
        a_old = a.copy()
        # Adding eps at dst adds 2·eps·a_old to conv[dst:dst+n]
        conv[dst:dst + n] += 2.0 * eps * a_old
        # Removing eps from src subtracts 2·eps·a_old from conv[src:src+n]
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # periodic exact recomputation (removes numerical drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective within the given time budget.
    The returned object is a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n usually yields a better bound, but
    # the FFT cost grows only mildly.  8192 is a good compromise.
    # ------------------------------------------------------------------
    final_n = 8192
    L_fft = _next_pow2(2 * final_n - 1)

    # ------------------------------------------------------------------
    # Initialise from a known global (if present) otherwise start from an
    # exponential seed that is tuned by a quick grid search.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter /= starter.sum()
            a0 = np.interp(np.linspace(0, 1, final_n),
                           np.linspace(0, 1, starter.size),
                           starter)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print("started from existing height_sequence_1")
        else:
            raise Exception()
    except Exception:
        # quick exponential grid search
        print("starting exponential grid search")
        best_score = float("inf")
        best_r = 0.99
        # coarse grid
        for r in np.linspace(0.90, 0.9999, 200):
            cand = _exponential_sequence(final_n, r)
            score = _objective(cand, L_fft, final_n)
            if score < best_score:
                best_score = score
                best_r = r
        # refined grid around the best coarse r
        for r in np.linspace(best_r * 0.99, min(best_r * 1.01, 0.999999), 200):
            cand = _exponential_sequence(final_n, r)
            score = _objective(cand, L_fft, final_n)
            if score < best_score:
                best_score = score
                best_r = r
        a0 = _exponential_sequence(final_n, best_r)
        print(f"[{time.time() - start_time:.2f}s] exponential seed r={best_r:.8f}, score={best_score:.6f}")

    best_seq = a0.copy()
    best_score = _objective(best_seq, L_fft, final_n)
    print(f"[{time.time() - start_time:.2f}s] initial objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Randomised search over low‑dimensional families
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("phase 1 – randomised family search")
        family_start = time.time()
        while time.time() < deadline - 30.0:
            choice = np.random.randint(0, 5)
            if choice == 0:           # pure exponential (different r)
                r = np.random.uniform(0.95, 0.99999)
                cand = _exponential_sequence(final_n, r)
            elif choice == 1:         # mixture of two exponentials
                r1 = np.random.uniform(0.90, 0.999)
                r2 = np.random.uniform(r1, 0.99999)
                p = np.random.rand()
                cand = _mix_two_exponentials(final_n, r1, r2, p)
            elif choice == 2:         # polynomial decay
                pwr = np.random.uniform(0.0, 10.0)
                cand = _poly_decay_sequence(final_n, pwr)
            elif choice == 3:         # triangular with random support
                sup = np.random.randint(final_n // 8, final_n)
                cand = _triangular_sequence(final_n, sup)
            else:                     # linear combination of exponential + poly
                r = np.random.uniform(0.95, 0.9999)
                pwr = np.random.uniform(0.0, 5.0)
                lam = np.random.rand()
                cand = lam * _exponential_sequence(final_n, r) + (1.0 - lam) * _poly_decay_sequence(final_n, pwr)
                cand /= cand.sum()

            score = _objective(cand, L_fft, final_n)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] family improve: score={score:.6f}")

        print(f"phase 1 finished (elapsed {time.time() - family_start:.1f}s)")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smooth max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("phase 2 – Adam optimisation")
        phase2_deadline = min(deadline, time.time() + 0.30 * (deadline - time.time()))
        alphas = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
        a_adam, score_adam = _adam_optimize(
            best_seq,
            final_n,
            phase2_deadline,
            start_time,
            alphas,
            lr0=0.6,
            iters_per_alpha=25000,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Greedy hill‑climbing (deterministic)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("phase 3 – greedy hill‑climbing")
        a_greedy, score_greedy = _greedy_hill_climb(
            best_seq,
            final_n,
            deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=2_000_000,
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(final_n, 1.0 / final_n)

    final_obj = _objective(best_seq, L_fft, final_n)
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    # Return as simple Python list
    return [float(x) for x in best_seq]
