def _run_multi_exp_gradient(a0, n, deadline, start_time, eta0=0.6, top_k=3, alpha=5.0, beta=0.9):
    a = a0.copy()
    best_a = a.copy()
    best_score = np.inf
    L = _next_pow2(2*n-1)
    grad_ema = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution via FFT
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)
        # select top_k indices
        if top_k >= len(conv):
            idxs = np.arange(len(conv))
        else:
            idxs = np.argpartition(-conv, top_k)[:top_k]
        # compute softmax weights over top_k
        top_vals = conv[idxs]
        w = np.exp(alpha * top_vals - np.max(alpha * top_vals))
        w /= w.sum()
        grad = np.zeros_like(a)
        for idx, weight in zip(idxs, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            sub_idx = np.arange(lo, hi + 1)
            grad[sub_idx] += weight * (2.0 * a[idx - sub_idx])   # each contribution weight?
        # exponential moving average
        grad_ema = beta * grad_ema + (1 - beta) * grad
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad_ema)
        # renormalize onto simplex
        a_sum = a.sum()
        if a_sum <= 0 or np.isnan(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum
    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence.

Algorithm overview
------------------
* **Candidate pool** – generate a dozen simple deterministic shapes
  (uniform, linear, sqrt, exponential, sine, quadratic, Gaussian)
  and keep the one giving the smallest self‑convolution peak after a
  very short “multi‑gradient” run.

* **Phase 2 – multi‑index mirror descent** – a variant of the
  exponentiated‑gradient (mirror descent) optimiser.
  Instead of following the sub‑gradient of only the current maximum
  convolution entry, we take a soft‑max over the *top‑k* entries
  (controlled by a temperature α).  The resulting gradient is an
  EMA‑averaged weighted sum of the corresponding sub‑gradients,
  which gives a smoother descent direction and helps escape shallow
  local minima.  The step size decays as η₀/√t.

* **Phase 3 – deterministic greedy peak‑flattening** – the same
  incremental mass‑shifting routine as in the reference code, but
  driven by the current best sequence from Phase 2.

* **Optional random‑restart polishing** – if a few seconds remain,
  perturb the best sequence with small Gaussian noise and run a short
  multi‑gradient burst; keep any improvement.

All vectors are kept on the probability simplex (∑aᵢ = 1), which is
invariant for the objective.  The routine respects the 1000 s time
budget and prints progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # fallback – shouldn't happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Phase 2 – Multi‑index exponentiated gradient
# ----------------------------------------------------------------------
def _run_multi_exp_gradient(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           eta0: float = 0.8,
                           top_k: int = 5,
                           alpha: float = 5.0,
                           beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimiser that uses a soft‑max over the top‑k
    convolution entries (smooth approximation of the max) and an EMA
    of the gradients.  Returns the best vector seen and its objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    grad_ema = np.zeros_like(a)
    it = 0

    # pre‑allocate arrays to avoid repeated allocation where possible
    while time.time() < deadline:
        it += 1

        # --- convolution (full) ---
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] multi‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # --- top‑k soft‑max selection of indices ---
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(-conv, top_k)[:top_k]
        vals = conv[idxs]

        # soft‑max weights (temperature α)
        max_scaled = np.max(alpha * vals)
        w = np.exp(alpha * vals - max_scaled)
        w /= w.sum()

        # --- weighted sub‑gradient (sum of contributions of chosen indices) ---
        grad = np.zeros_like(a)
        for idx, weight in zip(idxs, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            sub_idx = np.arange(lo, hi + 1, dtype=int)
            grad[sub_idx] += weight * (2.0 * a[idx - sub_idx])

        # --- EMA of gradient ---
        grad_ema = beta * grad_ema + (1.0 - beta) * grad

        # --- mirror‑descent multiplicative update ---
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad_ema)

        # --- re‑project onto the simplex ---
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the index that most
    contributes to the current peak to an index that contributes least,
    updating the full convolution incrementally.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and every other candidate
        partner_vals = a[max_idx - idx]

        # choose destination with smallest partner value (and not src)
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score
