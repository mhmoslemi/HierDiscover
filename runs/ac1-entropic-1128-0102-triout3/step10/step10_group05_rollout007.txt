"""
Propose a step‑function (non‑negative vector) with a low value of
    2·n·max(conv(a,a)) / (sum(a)²)

The search consists of three stages:

1.  A fast scan for a good exponential seed (geometric decay) on the
    chosen length *n*.
2.  A continuation method that runs Adam on a smooth soft‑max
    approximation of the max‑convolution.  The smoothing parameter
    α is increased from very small values to several‑thousand, which
    drives the solution towards the true ℓ∞‑objective.
    Multiple independent restarts (Dirichlet random starts) are performed
    while time allows.
3.  A short deterministic greedy sweep that moves a tiny amount of mass
    from the most responsible coordinate (the one contributing most to
    the current maximum convolution entry) to the least responsible one.
    This fine‑tunes the solution after the Adam phase.

All operations respect the 1000 s budget; progress is printed
periodically.  The final vector is normalised to sum = 1 and returned as
a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a using FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Return a normalised geometric distribution (a_i ∝ r^i) on n points.
    r must be in (0,1); the vector is automatically normalised to sum 1.
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


# ----------------------------------------------------------------------
# Adam optimisation of a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _run_adam_softmax(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      alphas,
                      iters_per_alpha: int = 30000,
                      lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate
        f(a)=log Σ_k exp(α·conv_k(a))

    Returns the best vector seen and its corresponding objective
    2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)
    best_score = 2.0 * n * _conv_self(best_a, L).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    idx = np.arange(n)

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1

            # forward pass: convolution and soft‑max weights
            Af = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # smooth‑max (log‑sum‑exp) weights
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # gradient of the smooth objective
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - idx]

            # Adam update
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # stay on the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy reduction of the peak (deterministic fine‑tuning)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Move a tiny amount of mass from the most responsible index for the
    current maximum convolution entry to the least responsible one.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # vector a_k = a[max_idx - i] on the support of the max
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # destination with smallest a_k value (i.e. smallest partner)
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partner values for src and dst in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # choose an epsilon that reduces the peak but stays safe
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-30)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final return
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4096 balances quality and speed
    # ------------------------------------------------------------------
    n = 4096
    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialisation – try to reuse a historic construction if available,
    # otherwise fall back to a cheap exponential scan.
    # ------------------------------------------------------------------
    a0 = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = starter / starter.sum()
            # Interpolate / truncate to length n
            if starter.size != n:
                a0 = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, starter.size),
                               starter)
            else:
                a0 = starter.copy()
    except Exception:
        pass

    if a0 is None:
        # quick sweep for a good geometric decay
        best_r = 0.999
        best_score = float("inf")
        for r in np.linspace(0.90, 0.9999, 200):
            cand = _exponential_sequence(n, r)
            conv = _conv_self(cand, L)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_r = r
        a0 = _exponential_sequence(n, best_r)
        print(f"[0.00s] exponential seed r={best_r:.6f}, "
              f"score={best_score:.6f}")

    # normalise to sum = 1
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_seq = a0.copy()
    best_obj = 2.0 * n * _conv_self(best_seq, L).max()
    print(f"[0.00s] initial objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a soft‑max surrogate, many continuation steps.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a fairly long schedule of α values (geometric growth up to ~1e4)
        alphas = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0,
                  20.0, 40.0, 80.0, 160.0, 320.0,
                  640.0, 1280.0, 2560.0, 5120.0, 10240.0]
        phase1_deadline = time.time() + max(30.0, 0.30 * (deadline - time.time()))
        a_adam, obj_adam = _run_adam_softmax(
            best_seq,
            n,
            phase1_deadline,
            start_time,
            alphas=alphas,
            iters_per_alpha=25000,
            lr0=0.6,
        )
        if obj_adam < best_obj:
            best_obj = obj_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
                  f"score={obj_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – several independent random restarts of the same Adam
    #            procedure (shorter runs).  This helps escape local minima.
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # remaining budget for a single restart
        remaining = deadline - time.time()
        # allocate at most 20 s per restart (more if time allows)
        run_time = min(20.0, remaining * 0.25)
        restart_deadline = time.time() + run_time

        # random Dirichlet start (moderately sparse)
        a_start = np.random.dirichlet(np.ones(n))
        a_opt, obj_opt = _run_adam_softmax(
            a_start,
            n,
            restart_deadline,
            start_time,
            alphas=[0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0],
            iters_per_alpha=5000,
            lr0=0.6,
        )
        if obj_opt < best_obj - 1e-12:
            best_obj = obj_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑restart improve: "
                  f"score={obj_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic greedy fine‑tuning (as much as time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_greedy, obj_greedy = _run_greedy_reduction(
            best_seq,
            n,
            deadline,
            start_time,
            best_obj,
            best_seq,
            max_iters=5_000_000,
        )
        if obj_greedy < best_obj - 1e-12:
            best_obj = obj_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy improve: "
                  f"score={obj_greedy:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to a Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # a quick sanity check using the provided evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] final evaluate_sequence = {final_val:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
