def _run_sa(a, n, deadline, start_time):
    Lfft = _next_pow2(2*n - 1)
    conv = _conv_self(a, Lfft)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    total_steps_est = 2000000  # fallback, may stop earlier
    start_temp = cur_score * 0.1
    end_temp = 1e-9
    recompute_interval = 2000
    step = 0
    while time.time() < deadline and step < total_steps_est:
        step += 1
        # cooling
        T = start_temp * (end_temp / start_temp) ** (step / total_steps_est)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[k - idx]
        if contrib.sum() > 0:
            src = rng.choice(idx, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)
        # destination biased toward low values
        prob_dst = 1.0 - a
        prob_dst = prob_dst / prob_dst.sum()
        dst = rng.choice(n, p=prob_dst)
        if src == dst:
            continue
        src_mass = a[src]
        if src_mass <= 1e-12:
            continue
        eps = src_mass * rng.random() * 0.2  # up to 20%
        a_old = a.copy()
        # update conv using a_old
        conv[src:src+n] -= 2 * eps * a_old
        conv[dst:dst+n] += 2 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        # ensure non-negativity
        a = np.maximum(a, 0.0)
        # optional renorm to keep sum exactly 1 to avoid drift
        s = a.sum()
        if s != 0:
            a /= s
        # compute new score
        new_score = 2.0 * n * conv.max()
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if math.exp(-delta / (T + 1e-12)) > rng.random():
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve: iter={step}, score={new_score:.6f}")
        else:
            # revert
            a = a_old
            # recompute conv fully to avoid drift
            conv = _conv_self(a, Lfft)
            cur_score = 2.0 * n * conv.max()
        if step % recompute_interval == 0:

"""
Search for a low‑value step function.

The algorithm consists of three stages:

1.  **Fast shape scan** – generate many candidates from simple
    parametric families (exponential, mixtures with the uniform,
    power‑law and logistic shapes).  The best candidate is kept as a
    starting point.

2.  **Targeted simulated annealing** – repeatedly move a small amount
    of mass from the index that contributes most to the current peak
    of the self‑convolution to an index that currently carries little
    mass.  The move is accepted according to a classic temperature
    schedule, allowing occasional uphill steps and thus escaping local
    minima.

3.  **Deterministic greedy sweep** – a final deterministic pass that
    moves mass from the “source’’ coordinate of the maximal convolution
    entry toward a coordinate with the smallest partner value.
    This phase often shaves the last few 10⁻⁴ from the objective.

All steps respect the 1000 s budget, keep the sequence non‑negative
and normalized (the objective is invariant under scaling, but a sum‑1
normalisation makes the code numerically stable).  The function
``propose_candidate`` returns the best sequence found as a plain
Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a using FFT.
    Returns the full convolution (length 2·len(a)‑1).
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to sum 1."""
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise (finite sum = (1‑r**n))
    a = a * (1.0 - r) / (1.0 - r ** n)
    a[a < 0.0] = 0.0
    a /= a.sum()
    return a


def _power_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum 1."""
    x = np.arange(1, n + 1, dtype=float)
    a = x ** (-p)
    a = a / a.sum()
    return a


def _logistic_sequence(n: int, steep: float, offset: float) -> np.ndarray:
    """Logistic decay a_i = 1/(1+exp(steep*(i‑offset)))."""
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp(steep * (i - offset)))
    a = a / a.sum()
    return a


def _mix_exponential_uniform(n: int, r: float, gamma: float) -> np.ndarray:
    """Convex combination (1‑γ)·uniform + γ·exponential(r)."""
    uniform = np.full(n, 1.0 / n)
    exp_seq = _exponential_sequence(n, r)
    a = (1.0 - gamma) * uniform + gamma * exp_seq
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – fast scan of simple parametric families
# ----------------------------------------------------------------------
def _scan_initial_shapes(n: int, budget_deadline: float):
    """Return the best sequence among a handful of parametric families."""
    best_a = None
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)

    # exponential family (coarse sweep)
    for r in np.linspace(0.90, 0.9999, 200):
        a = _exponential_sequence(n, r)
        conv = _conv_self(a, L)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a

    # mixture of uniform + exponential
    uniform = np.full(n, 1.0 / n)
    for r in np.linspace(0.90, 0.9999, 100):
        exp_seq = _exponential_sequence(n, r)
        for gamma in np.linspace(0.0, 1.0, 11):
            a = (1.0 - gamma) * uniform + gamma * exp_seq
            a = a / a.sum()
            conv = _conv_self(a, L)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_a = a

    # power‑law family
    for p in np.linspace(0.0, 2.5, 101):
        a = _power_sequence(n, p)
        conv = _conv_self(a, L)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a

    # logistic family (steepness sweep)
    for steep in np.linspace(0.001, 0.2, 31):
        offset = n / 2.0
        a = _logistic_sequence(n, steep, offset)
        conv = _conv_self(a, L)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a

    # make sure we return a valid array
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _conv_self(best_a, L).max()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – targeted simulated annealing
# ----------------------------------------------------------------------
def _run_sa(a0: np.ndarray, n: int, deadline: float, start_time: float):
    """
    Simulated annealing that moves a small amount of mass from the index
    responsible for the current peak towards a low‑mass index.
    """
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a, Lfft)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    # annealing schedule
    start_temp = cur_score * 0.1 + 1e-6
    end_temp = 1e-9
    total_steps = 2_000_000               # generous upper bound
    recompute_interval = 5_000
    step = 0

    while time.time() < deadline and step < total_steps:
        step += 1
        # --- temperature ---
        T = start_temp * (end_temp / start_temp) ** (step / total_steps)

        # --- identify source index (largest contributor to current peak) ---
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idx] * a[k - idx]
        if contrib.sum() > 0.0:
            src = rng.choice(idx, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        # --- choose destination biased towards low mass ---
        prob_dst = 1.0 - a
        prob_dst = prob_dst / prob_dst.sum()
        dst = rng.choice(n, p=prob_dst)

        if src == dst:
            continue

        src_mass = a[src]
        if src_mass <= 1e-14:
            continue

        # epsilon = up to 20 % of the source mass, chosen randomly
        eps = src_mass * rng.random() * 0.20
        a_old = a.copy()                         # keep a snapshot for possible revert

        # ----- incremental update of the convolution using the *old* a -----
        conv[src:src + n] -= 2.0 * eps * a_old
        conv[dst:dst + n] += 2.0 * eps * a_old

        # ----- modify the sequence -----
        a[src] -= eps
        a[dst] += eps
        a = np.maximum(a, 0.0)                   # safety

        # renormalise (drift due to floating point is tiny but we keep sum = 1)
        s = a.sum()
        if s != 0.0:
            a /= s

        new_score = 2.0 * n * conv.max()
        delta = new_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            if math.exp(-delta / (T + 1e-12)) > rng.random():
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improve: iter={step:,}, "
                      f"score={new_score:.6f}")
        else:
            # revert everything
            a = a_old
            conv = _conv_self(a, Lfft)          # full recompute after a rejection
            cur_score = 2.0 * n * conv.max()

        # periodic exact recomputation to avoid accumulated drift
        if step % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – deterministic greedy reduction (same spirit as the previous code)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass from the source of the current peak toward a
    coordinate with the smallest partner value.  The loop terminates on time
    or after ``max_iters`` iterations."""
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)  # full convolution (length 2·n‑1)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest value (used for dynamic step size)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually appear in this maximal convolution entry
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and a prospective destination
        a_k = np.zeros(n, dtype=float)
        a_k[idx] = a[max_idx - idx]

        # pick destination where the partner value is minimal (and not the source)
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = (src + 1) % n
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution (exact) -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

        print(f"[{time.time() - start_time:.2f}s] estimated objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
