``evaluate_sequence`` objective.

The algorithm combines
(1) a diverse set of random Dirichlet initialisations,
(2) a smooth‑max exponentiated‑gradient descent where the gradient is
    computed by a weighted convolution (fast FFT based),
    with an increasing “temperature” that turns the smooth‑max into the
    true maximum,
(3) a short deterministic greedy peak‑reduction phase,
(4) a final stochastic local‑search that moves tiny amounts of mass.
All phases respect the 1000 s budget and the best vector found is

import time, math, numpy as np

# helper functions
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT, length 2·len(a)‑1."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _gradient_weighted(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a*a)[k] w.r.t. a.
    w must have length 2·len(a)‑1.
    """
    w_rev = w[::-1]
    Af = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(w_rev, n=fft_len)
    corr = np.fft.irfft(Af * W, n=fft_len)
    n = a.shape[0]
    offset = len(w_rev) - 1
    grad = 2.0 * corr[offset - np.arange(n)]
    return grad

    else:

    return x

def _score_from_vector(a: np.ndarray, n: int) -> float:
    """Exact score when the vector is normalised to sum‑1."""
    conv = _conv_self_fft(a, _next_pow2(3 * n))
    return 2.0 * n * conv.max()

def _smooth_expgrad_opt(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        fft_len: int,
                        n: int) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    A smooth‑max weight w = exp(α·(conv−max)) is used; α grows
    from a small value to a large one, turning the smooth‑max into an
    exact max.  The step‑size decays as 1/√t.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_from_vector(a, n)

    t = 0
    eta0 = 0.7
    alpha = 2.0          # initial temperature
    alpha_max = 500.0
    alpha_inc = 0.08     # increase per iteration (will quickly reach max)

    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth‑expgrad improve iter={t:,} score={cur_score:.6f}")

        # build smooth‑max weights
        # w_k = exp(α·(conv_k - cur_max))  ≤ 1, =1 for the current max.
        w = np.exp(alpha * (conv - cur_max))

        # gradient of Σ_k w_k·(a*a)[k]
        g = _gradient_weighted(a, w, fft_len)

        # step‑size
        eta = eta0 / math.sqrt(t)

        # exponentiated update (mirror descent on the simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # keep numerical hygiene
        a = np.maximum(a, 1e-15)

        # increase the temperature slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # occasional exact evaluation for safety (once per 2000 iters)
        if t % 2000 == 0:
            exact = _score_from_vector(a, n)
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] exact check improve score={exact:.6f}")

    return best_a, best_score

def _greedy_reduce(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   n: int,
                   fft_len: int,
                   max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: move a fraction of the coefficient that
    contributes most to the current peak towards the smallest coefficient.
    Accept only strict improvements.
    """
    a = a0.copy()
    best_score = _score_from_vector(a, n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self_fft(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination is the index with smallest coefficient (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # move a modest amount (25 % of the source mass)
        eps = 0.25 * a[src]
        if eps < 1e-12:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score_from_vector(a_prop, n)
        if new_score < best_score - 1e-12:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:6.2f}s] greedy improve iter={it} score={new_score:.6f}")

    return a, best_score

def _random_local_search(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         max_trials: int = 200000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: repeatedly move a tiny random
    amount of mass from a randomly chosen donor to a random receiver.
    Accept only strict improvements.
    """
    a = a0.copy()
    best_score = _score_from_vector(a, n)

    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        i = rng.integers(n)
        if a[i] < 1e-9:
            continue
        # pick recipient distinct from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # transfer a small fraction of the donor
        eps = min(a[i] * 0.02, 1e-5)
        if eps <= 0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps

        cur_score = _score_from_vector(a_candidate, n)
        if cur_score < best_score:
            a = a_candidate
            best_score = cur_score
            print(f"[{time.time() - start_time:6.2f}s] rand‑search improve score={cur_score:.6f}")

    return a, best_score

    # handcrafted baselines
    starts.append(np.full(n, 1.0 / n))          # uniform
    starts.append(_exponential_start(n, r=0.9995))
    starts.append(_exponential_start(n, r=0.98))
    starts.append(_gaussian_start(n))

    # Dirichlet draws with several concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            starts.append(_dirichlet_start(n, conc, rng))

    # previous best, if the global variable exists
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                # linear interpolation to required length
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            seq = np.maximum(seq, 0.0)
            if seq.sum() == 0.0:
                seq = np.full(n, 1.0 / n)
            else:
                seq = seq / seq.sum()
            starts.append(seq)
    except Exception:
        pass

    return starts


def _score_from_vector(a: np.ndarray, n: int) -> float:
    """
    Exact score for a normalised (sum = 1) vector:
        score = 2·n·max(a*a)
    """
    fft_len = _next_pow2(3 * n)
    conv = _conv_self_fft(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def _smooth_expgrad_opt(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        fft_len: int,
                        n: int) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with a smooth‑max weight.
    The temperature α starts small and is increased until the weight
    concentrates on the true maximum.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_from_vector(a, n)

    t = 0
    eta0 = 0.7
    alpha = 2.0          # initial temperature
    alpha_max = 500.0
    alpha_inc = 0.08     # how fast the temperature rises

    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth‑expgrad improve iter={t:,} "
                  f"score={cur_score:.6f}")

        # smooth‑max weights (w_k = exp(α·(conv_k – cur_max)))
        w = np.exp(alpha * (conv - cur_max))

        # gradient of Σ_k w_k·(a*a)[k]
        g = _gradient_weighted(a, w, fft_len)

        # diminishing step size
        eta = eta0 / math.sqrt(t)

        # exponentiated update (mirror descent on the simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        a = np.maximum(a, 1e-15)

        # increase temperature slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # occasional exact re‑evaluation (helps catch numerical drift)
        if t % 2000 == 0:
            exact = _score_from_vector(a, n)
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] exact check improve score={exact:.6f}")

    return best_a, best_score


def _greedy_reduce(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   n: int,
                   fft_len: int,
                   max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: move a fraction of the coefficient that
    contributes most to the current peak towards the smallest coefficient.
    """
    a = a0.copy()
    best_score = _score_from_vector(a, n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self_fft(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination = index with smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # move 25 % of donor mass
        if eps < 1e-12:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score_from_vector(a_prop, n)
        if new_score < best_score - 1e-12:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:6.2f}s] greedy improve iter={it} score={new_score:.6f}")

    return a, best_score


def _random_local_search(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         max_trials: int = 200000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing.  Repeatedly transfer a tiny
    random amount of mass from a random donor to a random receiver,
    accepting only strict improvements.
    """
    a = a0.copy()
    best_score = _score_from_vector(a, n)

    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        eps = min(a[i] * 0.02, 1e-5)
        if eps <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps

        sc = _score_from_vector(cand, n)
        if sc < best_score:
            a = cand
            best_score = sc
            print(f"[{time.time() - start_time:6.2f}s] rand‑search improve score={sc:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine mixes a smooth‑max
    exponentiated‑gradient descent, a greedy peak‑reduction and a final
    short random local search, respecting the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    n = 4800
    fft_len = _next_pow2(3 * n)   # large enough for all convolutions we need

    # ------------------------------------------------------------------
    # 1) generate starting vectors and keep the best one
    # ------------------------------------------------------------------
    candidates = _initial_candidates(n, rng)
    best_seq = None
    best_score = float('inf')
    for idx, a0 in enumerate(candidates):
        sc = _score_from_vector(a0, n)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] init #{idx+1:2d} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max exponentiated‑gradient descent
    # ------------------------------------------------------------------
    opt_deadline = start_time + budget_s * 0.80
    opt_seq, opt_score = _smooth_expgrad_opt(best_seq, opt_deadline,
                                            start_time, fft_len, n)
    if opt_score < best_score:
        best_score = opt_score
        best_seq = opt_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST after smooth‑expgrad {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Deterministic greedy peak reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        greedy_deadline = deadline - 0.3   # keep a small buffer for the final phase
        gre_seq, gre_score = _greedy_reduce(best_seq, greedy_deadline,
                                            start_time, n, fft_len)
        if gre_score < best_score:
            best_score = gre_score
            best_seq = gre_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST after greedy {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Very cheap random local search (last seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fin_seq, fin_score = _random_local_search(best_seq, deadline,
                                                  start_time, n, fft_len, rng)
        if fin_score < best_score:
            best_score = fin_score
            best_seq = fin_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST after random search {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation & reporting
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
