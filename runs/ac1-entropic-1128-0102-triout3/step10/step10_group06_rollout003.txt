assistantfinal**Solution Overview**

The code below implements a *hybrid* optimiser for the black‑box function
`evaluate_sequence`.  
The optimiser works in several stages

1. **Diverse initialisation** – a collection of handcrafted and random
   Dirichlet draws (different concentration parameters) are generated.
2. **Adam warm‑up** – a few thousand Adam steps on a smooth
   log‑sum‑exp surrogate of the convolution peak move the distribution out
   of bad regions quickly.
3. **Adaptive multiplicative mirror‑descent** – exponentiated‑gradient
   updates are performed on the exact objective.  At each iteration the
   active shifts (those whose convolution value is within a tiny fraction
   of the current maximum) are collected and a *multi‑gradient* is built.
   The step size decays like `1/√t`.  If the algorithm stalls a small
   multiplicative perturbation is injected to escape local minima.
4. **Random hill‑climbing** – mass is transferred between two random
   positions; a move is kept only if the exact score improves (with an
   optional tiny simulated‑annealing acceptance).  This explores the
   neighbourhood that the gradient step may miss.
5. **Deterministic greedy refinement** – finally a deterministic
   “move the most harmful mass towards the smallest coefficient’’ step
   is applied until no further improvement is possible.

All phases respect the global time budget (default 1000 s).  The best
sequence found during the whole run is returned (normalised to sum = 1 and
all entries ≥ 0).  The implementation uses only `numpy`, `time`,
`math` and the supplied `evaluate_sequence`.

```python
"""
Hybrid optimiser for `evaluate_sequence`.

The optimiser builds many random starts, performs a short Adam warm‑up on a
smooth surrogate of the max‑convolution, then runs an adaptive
exponentiated‑gradient (mirror‑descent) on the exact objective.
If progress stalls a small multiplicative perturbation is applied.
A random hill‑climbing phase follows, and finally a deterministic greedy
local improvement is executed.  The whole routine stops after the given
time budget (default 1000 s) and returns the best non‑negative step
function it found.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast version – identical to evaluate_sequence for valid data)
# ----------------------------------------------------------------------


def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv)/sum(a)^2 (no clipping, assumes a ≥ 0)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Smooth surrogate and Adam warm‑up
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Log‑sum‑exp approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    A few Adam steps on the smoothed max‑convolution objective.
    Returns the best probability vector and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule (more gradual than the reference)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 2500

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max → valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate (see reference implementation)
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score


# ----------------------------------------------------------------------
# Multi‑gradient exponentiated‑gradient (adaptive variant)
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] obtained by summing the gradients of the
    (selected) active shifts.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt_adaptive(a0: np.ndarray,
                                deadline: float,
                                start_time: float,
                                fft_len: int,
                                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adaptive exponentiated‑gradient on the exact max‑convolution objective.
    *   active shifts are those within a decreasing tolerance;
    *   if no improvement for a while, a small random perturbation is added.
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _fast_score(a, fft_len)
    t = 0
    last_improve = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # sum(a) = 1 ⇒ denominator = 1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = t
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ----- stagnation handling -----
        if t - last_improve > 50000:
            sigma = 0.04
            perturb = rng.normal(0.0, sigma, size=n)
            a = a * np.exp(perturb)        # multiplicative noise
            a = np.clip(a, 1e-15, None)
            a = a / a.sum()
            print(f"[{time.time() - start_time:6.2f}s] Perturbation at iter={t}")
            last_improve = t

        # ----- active set (fraction of the current peak) -----
        # tolerance shrinks slowly with the iteration count
        delta_frac = max(0.0002, 0.001 / math.sqrt(t))
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        # limit the number of active shifts for speed
        top_k = 7
        if active.size > top_k:
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ----- sub‑gradient -----
        g = _multi_gradient(a, active)

        # ----- step‑size schedule -----
        eta0 = 0.6
        eta = eta0 / math.sqrt(t)

        # ----- exponentiated update & renormalisation -----
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        a = np.maximum(a, 1e-15)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random hill‑climbing phase
# ----------------------------------------------------------------------


def _hill_climb_random(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       fft_len: int,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer hill‑climbing.
    A move (source → destination) is kept if it strictly improves the exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _fast_score(a, fft_len)
    best_a = a.copy()

    while time.time() < deadline:
        # pick a source with non‑zero mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)

        # pick a different destination uniformly
        dst = rng.integers(0, n - 1)
        while dst == src:
            dst = rng.integers(0, n - 1)

        # amount to move – a random fraction of the source mass
        eps_frac = rng.uniform(0.001, 0.25)
        eps = eps_frac * a[src]
        if eps <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        sc = _fast_score(a_new, fft_len)
        if sc < best_score:
            a = a_new
            best_score = sc
            best_a = a_new.copy()
            # occasional progress report
            if (int(best_score * 1e6) % 1000) == 0:
                print(
                    f"[{time.time() - start_time:6.2f}s] hill‑climb improve: score={best_score:.6f}"
                )
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy refinement (same as reference)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the current
    convolution peak towards the smallest coefficient.  Accept only strict
    improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Start‑vector factories
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with a given concentration."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Discrete Gaussian centred in the middle (σ≈n/6)."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving total mass."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine mixes random Dirichlet starts, a short Adam warm‑up,
    an adaptive exponentiated‑gradient descent, a random hill‑climbing
    stage and a final deterministic greedy refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 works well with a FFT length of 16384
    # ------------------------------------------------------------------
    n = 8000
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a rich set of starting points
    # ------------------------------------------------------------------
    starts = []

    # uniform
    starts.append(np.full(n, 1.0 / n))

    # exponential decays (different rates)
    for r in (0.9999, 0.998, 0.99, 0.95):
        starts.append(_exponential_start(n, r))

    # Gaussian bell
    starts.append(_gaussian_start(n))

    # Sine‑shaped (non‑negative on [0,π])
    xs = np.arange(n)
    sin_seq = np.sin(np.pi * xs / (n - 1))
    sin_seq = np.maximum(sin_seq, 0.0)
    starts.append(sin_seq / sin_seq.sum())

    # Dirichlet draws with assorted concentrations
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            starts.append(_dirichlet_start(n, conc, rng))

    # optionally include the previously best known construction
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())

    # ------------------------------------------------------------------
    # Evaluate raw starts and keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(starts):
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1}/{len(starts)}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam warm‑up (≈10 % of the budget)
    # ------------------------------------------------------------------
    warm_deadline = start_time + budget_s * 0.10
    cand_seq, cand_score = _adam_optimize(best_seq, warm_deadline, start_time, fft_len)
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adaptive exponentiated‑gradient (≈60 % of the budget)
    # ------------------------------------------------------------------
    md_deadline = start_time + budget_s * 0.70
    md_seq, md_score = _expgrad_multi_opt_adaptive(best_seq, md_deadline, start_time, fft_len, rng)
    if md_score < best_score:
        best_score = md_score
        best_seq = md_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after MD  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Random hill‑climbing (≈20 % of the budget)
    # ------------------------------------------------------------------
    hc_deadline = start_time + budget_s * 0.90
    hc_seq, hc_score = _hill_climb_random(best_seq, hc_deadline, start_time, fft_len, rng)
    if hc_score < best_score:
        best_score = hc_score
        best_seq = hc_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after hill‑climb  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Deterministic greedy refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
```
