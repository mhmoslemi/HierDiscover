"""
Propose a non‑negative step function (length 4096) that minimises
2 * n * max(conv(a, a)) / (sum(a)**2).

Algorithm
---------

1. **Baseline & global starter** – start from the uniform sequence
   and optionally from the provided `height_sequence_1`.

2. **Rich parametric sweep** – explore
   * geometric families (fine grid around the best epsilon),
   * power families,
   * exponential‑power families a_i = exp(-c·(i/(n‑1))⁽ᵖ⁾),
   * random mixtures of a few exponentials.
   The best shape from this inexpensive search seeds the next stage.

3. **Mirror‑descent on a soft‑max approximation** – a
   *soft‑max* replaces the max in the objective;
   we perform exponentiated‑gradient (mirror) updates
   `a_i ← a_i·exp(-η·∇_i)` and renormalise to the simplex.
   Multiple short random restarts are performed.

4. **Multi‑active sub‑gradient refinement** – the exact max is
   approximated by the convex hull of all (near‑maximal) convolution
   entries; the sub‑gradient is obtained by convolving the current
   sequence with an indicator of the active set.

5. **Deterministic hill‑climbing polish** – tiny mass transfers
   between coordinates are accepted only if the true objective
   improves.

All stages respect the 1000 s budget, stop well before the deadline,
and keep the best sequence encountered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (w≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast exact objective for a sequence whose sum ≈ 1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweep (extended)
# ----------------------------------------------------------------------
def _parametric_sweep(best_seq: np.ndarray,
                      best_score: float,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:

    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    # ---- geometric families ----
    eps_grid = np.logspace(-5, -1, 300)          # finer than before
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        # a few lam values – dense near 0 and 1
        for lam in np.linspace(0.0, 1.0, 21):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power families ----
    p_grid = np.linspace(-1.0, 2.0, 81)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = (idx + 1) ** p
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq /= pow_sum
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # ---- exponential‑power families ----
    c_grid = np.logspace(-6, -1, 30)   # decay coefficient
    p_grid = np.linspace(0.3, 3.0, 27)  # power exponent
    for c in c_grid:
        if time.time() >= deadline - 30:
            break
        for p in p_grid:
            seq = np.exp(-c * (idx / (n - 1)) ** p)
            seq_sum = seq.sum()
            if seq_sum == 0:
                continue
            seq /= seq_sum
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Exp‑pow c={c:.1e} p={p:.2f} score={sc:.6f}")

    # ---- random mixtures of exponentials ----
    rng = np.random.default_rng()
    for _ in range(2000):
        if time.time() >= deadline - 30:
            break
        K = rng.integers(2, 5)               # 2‑4 components
        rates = rng.uniform(0.90, 0.9999, size=K)   # slower than 1
        weights = rng.random(K)
        weights /= weights.sum()
        # build mixture: Σ w_k * rates_k**i
        seq = np.zeros(n, dtype=np.float64)
        for wk, rk in zip(weights, rates):
            seq += wk * (rk ** idx)
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq /= seq_sum
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] MixExp K={K} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Mirror‑descent on soft‑max (exponentiated gradient)
# ----------------------------------------------------------------------
def _mirror_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on f(a)=2·n·softmaxτ(conv(a,a))."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    tau = 0.2
    tau_min = 1e-8
    tau_decay = 0.9996
    eta0 = 0.12

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L_fft)
        maxc = conv.max()
        # stable soft‑max
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient: 2·Σ_{k∈active} a_{k‑i}
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # exponentiated (mirror) update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        # renormalise onto the simplex
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Mirror it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Multi‑active sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_descent_multi(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using all near‑maximal convolution entries."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.28
    it = 0
    L_grad = _next_pow2(3 * n)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Multi‑sub it={it:,} score={cur_score:.6f}")

        # active set (all entries within relative 1e‑8 of the max)
        tol = cur_max * 1e-8 + 1e-12
        active = (conv >= cur_max - tol).astype(np.float64)

        # gradient via convolution with the active‑set mask
        w_rev = active[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # momentum + Euclidean projection onto the simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Deterministic hill‑climb polish
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between coordinates, keep improvements."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # prob‑proportional pick
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.12   # up to 12 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score

    """
    Search for a non‑negative step function of length 4096 that minimises

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    L_grad = _next_pow2(3 * n)      # for gradient‑related convolutions

    best_seq = uniform.copy()

                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Extended parametric sweep
    # ------------------------------------------------------------------
    best_seq, best_score = _parametric_sweep(best_seq, best_score,

    while time.time() < deadline - 30.0:

        base = best_seq if np.random.rand() < 0.75 else uniform

        start = base + noise

            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mirror‑descent new global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Multi‑active sub‑gradient refinement (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent_multi(best_seq, n,
                                                   sub_deadline, start_time,
                                                   best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Multi‑active sub‑grad improve score={best_score:.6f}")

    # ------------------------------------------------------------------
    if time.time() < deadline:

        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improve score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
