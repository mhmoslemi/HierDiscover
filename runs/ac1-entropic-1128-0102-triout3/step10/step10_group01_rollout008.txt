"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. **Arcsine‑family sweep** –  Scan a one‑parameter family
       a_i ∝ (x_i·(1−x_i))^{−α},   x_i = (i+½)/n .
   The best exponent α gives a good starting point.

2. **Smoothed max gradient descent** –  Replace the non‑smooth
   max‑convolution by a log‑sum‑exp (smooth max) with a
   decreasing temperature ε.  The exact gradient is computed by
   FFT‑based cross‑correlation, then a projected Adam step
   keeps the vector on the probability simplex.

3. **Light simulated‑annealing polish** –  Tiny random mass‑transfer
   moves are accepted with a Metropolis criterion to escape any
   remaining local minima.

All operations respect the 1000 s wall‑time limit and never
produce negative or >1000 entries (the normalisation onto the
simplex guarantees a sum of 1).  The code is self‑contained and
uses only NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Return the cross‑correlation g[j] = Σ_i w[i]·a[i‑j] (zero‑padded)
    for j = 0 … n‑1.  Implemented with a single FFT pair.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A, n=L)          # length = len(w)+n‑1 = 3n‑2
    start = n - 1                             # index of the first element of length‑n slice
    return full[start:start + n]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negativity and Σ a = 1
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence but without safety checks)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a (already normalised)."""
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a ** 2))


# ----------------------------------------------------------------------
# Smoothed‑max gradient of the objective
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray,
                 eps: float,
                 const: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of   J(a) = const * max(a*a)
    using a log‑sum‑exp approximation with temperature eps.
    Returns (grad_J, smoothed_max).
    """
    # self‑convolution
    conv = _conv_self(a)

    # stable log‑sum‑exp (max subtracted)
    m = conv.max()
    shifted = conv - m                # ≤ 0
    exp_shift = np.exp(shifted / eps)  # under‑flow is fine
    Z = exp_shift.sum()
    w = exp_shift / Z

    # smoothed max value
    f_eps = m + eps * np.log(Z)

    # gradient of smoothed max w.r.t. a: 2 * cross‑correlation(w, a)
    grad_f = 2.0 * _cross_corr(w, a)

    # full gradient of J (const = 2*n / (sum(a)^2), but sum(a)=1 after projection)
    # we keep the generic formula that works even if sum deviates slightly.
    s = a.sum()
    grad_J = const * (grad_f - (2.0 * f_eps / s) * np.ones_like(a))
    return grad_J, f_eps
