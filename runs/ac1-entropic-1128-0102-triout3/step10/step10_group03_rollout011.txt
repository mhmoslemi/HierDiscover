"""
Propose a non‑negative step function (length 4096) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1. **Rich parametric sweep** – a three‑parameter family
   a_i ∝ (i+1)^p·(n‑i)^q·r^i   (p,q ≥ 0, 0.9 ≤ r ≤ 1) is explored
   on a fine grid together with a few convex mixtures with the uniform
   distribution.  The best vector from this exhaustive search seeds the
   subsequent refinements.

2. **Adam‑accelerated smoothed descent** – the hard max is replaced by a
   soft‑max (temperature τ) and the gradient is obtained with two FFTs.
   An Adam optimiser (bias‑corrected 1st/2nd moments) together with an
   exponential τ‑schedule performs a powerful local optimisation.
   Several short restarts from perturbed seeds are executed.

3. **Projected sub‑gradient descent** – after the Adam phase we run a
   classic sub‑gradient method (step size ∝ 1/√t) with momentum, staying
   on the probability simplex.

4. **Deterministic hill‑climbing polish** – tiny mass‑transfer moves
   (up to 10 % of the source bucket) are accepted only if the true
   objective improves.

All helpers are top‑level, no closures or lambdas are used.  The
implementation respects the 1000 s budget, printing progress along the
way and finally returning a list compatible with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Return 2·n·max(conv(a,a)) assuming sum(a)≈1.
    """
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑accelerated smoothed descent (soft‑max of the max)
# ----------------------------------------------------------------------
def _adam_descent(start_a: np.ndarray,
                  deadline: float,
                  start_time: float,
                  L_fft: int,
                  L_grad: int) -> tuple[np.ndarray, float]:
    """
    Smoothed optimisation using Adam.  The objective is
        f(a) = 2·n·softmax_τ(conv(a,a))
    while keeping a on the simplex.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.15

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    tau = 0.12               # initial temperature
    tau_min = 1e-8
    tau_decay = 0.9996       # geometric decay per iteration

    while time.time() < deadline:
        it += 1

        # --- convolution and soft‑max -------------------------------------------------
        conv = _conv_self(a, L=L_fft)                 # length 2n‑1
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)       # stable soft‑max
        w = exp_vals / exp_vals.sum()                # weights sum to 1

        # --- gradient via FFT (grad_j = 2·Σ_i a_i w_{i+j}) ----------------------------
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # --- Adam update -------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)                     # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- projection onto the simplex ---------------------------------------------
        a = _project_simplex(a)

        # --- true objective ---------------------------------------------------------
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} "
                  f"score={cur_score:.6f}")

        # --- temperature decay -------------------------------------------------------
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (identical to reference, but slightly tuned)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a) = 2·n·max(conv(a,a)).
    Uses momentum (β≈0.9) and a 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.35
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} "
                  f"score={cur_score:.6f}")

        # active index of the current maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of the max

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a tiny fraction of mass from a source
    index to a destination index, keep the move only if the true objective
    improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)           # pick proportionally to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10   # up to 10 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} "
                  f"best={best_score:.6f}")

    return best_a, best_score

    while time.time() < deadline - 30.0:
        # random perturbation of the current best (or uniform)
        base = best_seq if np.random.rand() < 0.7 else uniform
        noise = 0.03 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 10.0)   # ≤10 s per restart
        cand_seq, cand_score = _adam_descent(start, sub_deadline,
                                             start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after Adam "
                  f"restart score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient refinement (most of the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve "
                  f"score={sc_sub:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Deterministic hill‑climbing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improve "
                  f"score={sc_hc:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ "
          f"{best_score:.6f}")

    # Convert to plain Python floats for the external evaluator
    return [float(x) for x in best_seq]
