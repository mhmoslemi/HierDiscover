    it = 0
    tau_factor = tau_factor_init
    while time.time() < deadline:
        it += 1
        # compute conv and Af, ...
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af * Af, n=L)
        conv = conv_full[:2*n-1]
        M = conv.max()
        cur_score = 2 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        tau = max(tau_factor * M, 1e-9) # avoid zero

        shifted = conv - M  # negative values

        w_sum = w.sum()

            a = np.full_like(a, 1.0 / n)
        # decay tau_factor slowly
        tau_factor = max(tau_factor * tau_decay, min_tau_factor)
    return best_a, best_score

The evaluator is  
\[
\text{score}= \frac{2n\;\max (a*a)}{(\sum a)^2},
\]
and the score does **not** depend on a global scaling of the vector.  
Hence we may (and do) keep the vector on the probability simplex
(\(\sum a_i = 1,\;a_i\ge 0\)).

The optimisation problem is non‑convex.  The following strategy works well
inside the 1000 s budget:

1. **Deterministic initialisations** – uniform, linear, √‑law, exponential
   and (if present) the historic `height_sequence_1`. The best one is kept
   as a start point.

2. **Smoothed mirror‑descent** – minimise a smooth approximation of the
   max‑convolution  
   \[
   F_\tau(a)=\tau\log\!\Big(\sum_k e^{(a*a)_k/\tau}\Big) .
   \]
   For a given temperature τ the gradient is a weighted sum of the usual
   max‑gradient.  The weights are a soft‑max over the convolution values.
   We perform exponentiated‑gradient updates  

   \[
   a_i \leftarrow a_i\exp\!\bigl(-\eta_t\,\nabla_iF_\tau(a)\bigr),
   \qquad a\leftarrow a/\sum a,
   \]

   with a step size \(\eta_t=\eta_0/\sqrt t\) and a geometric decay of τ.
   The whole loop runs as long as time permits (≈ 55 % of the budget).

3. **Linear‑programming improvement** – after the smooth descent we know the
   current maximal convolution value \(M\).  
   We solve the LP  

   \[
   \max\sum_j b_j\quad\text{s.t.}\quad (a*b)_k\le M,\;b_j\ge0,
   \]

   but only for the *active* indices where the convolution is within a
   tiny tolerance of \(M\).  The LP (handled by `scipy.optimize.linprog`)
   yields a direction \(g\); a convex combination
   \(a\leftarrow(1-t)a+tg\) (with \(t\approx0.3\)) strictly reduces the peak.
   The LP step is repeated while it yields improvement (≈ 20 % of the budget).

4. **Deterministic greedy flattening** – the same routine that was used in
   the reference solution, now applied to the current best vector.
   It moves a tiny amount of mass from the index that contributes most to
   the peak to an index that contributes least, updating the convolution
   incrementally.  This cleans up the last few digits (≈ 25 % of the budget).

5. **Optional random‑restart polishing** – if a few seconds remain we
   perturb the best vector slightly and run a short smoothed descent again.

All helper functions are top‑level, there are no closures or lambda
expressions, and the final vector is returned as a plain Python list of
floats.

5. optional random‑restart polishing.

The whole routine respects the 1000 s budget and returns a list of
non‑negative heights whose sum is 1 (the evaluator is scale‑invariant)."""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # should never happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Stage 2 – smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _softmax_descent(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     eta0: float = 0.5,
                     tau_init_factor: float = 0.5,
                     tau_decay: float = 0.97,
                     min_tau_factor: float = 0.001,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max objective.
    Returns the best vector seen and its score (2*n*max(conv)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(3 * n - 2)               # length needed for correlation
    tau_factor = tau_init_factor
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convolution of a with itself -----
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af * Af, n=L)
        conv = conv_full[: 2 * n - 1]        # linear self‑convolution
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] softmax iter={it:,} "
                  f"score={cur_score:.6f}")

        # ----- temperature (smoothness) -----
        tau = max(tau_factor * M, 1e-12)

        # ----- soft‑max weights -----
        shifted = conv - M                     # max‑shifting for stability
        w = np.exp(shifted / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (via correlation) -----
        Fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(Fw * np.conj(Af), n=L)
        grad = 2.0 * corr_full[:n]           # linear term of d(a*a)/da

        # ----- exponentiated‑gradient update -----
        eta = eta0 / math.sqrt(it)
        # clip exponent to avoid overflow
        factor = np.exp(np.clip(-eta * grad, -50, 50))
        a = a * factor

        # ----- renormalise onto the simplex -----
        total = a.sum()
        if total > 0.0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

        # ----- anneal temperature -----
        tau_factor = max(tau_factor * tau_decay, min_tau_factor)

    return best_a, best_score


def _softmax_descent_short(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    """A very short smoothed descent (used for quick initial screening)."""
    return _softmax_descent(a0, n, deadline, start_time,
                            eta0=0.6, tau_init_factor=0.7,
                            tau_decay=0.98, min_tau_factor=0.001,
                            max_iters=10_000)


# ----------------------------------------------------------------------
# Stage 2b – linear‑programming improvement
# ----------------------------------------------------------------------
def _lp_improve_step(a: np.ndarray,
                     n: int,
                     conv: np.ndarray,
                     M: float,
                     t_mix: float = 0.3,
                     rel_tol: float = 1e-4) -> tuple[np.ndarray, float, bool]:
    """
    One LP improvement step.
    Returns (new_vector, new_M, improved_flag).
    """
    # active constraints: those where the convolution is close to the maximum
    tol = max(rel_tol * M, 1e-12)
    active_idx = np.where(conv >= M - tol)[0]
    if active_idx.size == 0:
        return a, M, False

    # build constraint matrix C (|active| × n)
    k = active_idx.size
    C = np.zeros((k, n), dtype=np.float64)
    for r, i in enumerate(active_idx):
        lo = max(0, i - (n - 1))
        hi = min(n - 1, i)
        # C[r, lo:hi+1] = a[i - j] for j = lo … hi
        C[r, lo:hi + 1] = a[i - np.arange(lo, hi + 1)]

    b_ub = np.full(k, M, dtype=np.float64)

    # maximise sum(b)  <=> minimise -sum(b)
    res = linprog(c=-np.ones(n),
                  A_ub=C,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs')
    if not res.success:
        return a, M, False

    b_sol = res.x
    s = b_sol.sum()
    if s <= 1.0 + 1e-12:               # no improvement possible
        return a, M, False

    g = b_sol / s                     # rescale to unit sum
    a_new = (1.0 - t_mix) * a + t_mix * g
    a_new = np.maximum(a_new, 0.0)
    a_new_sum = a_new.sum()
    if a_new_sum > 0.0:
        a_new = a_new / a_new_sum
    else:
        a_new = np.full_like(a_new, 1.0 / n)

    # recompute convolution for the new vector
    L = _next_pow2(3 * n - 2)
    Af_new = np.fft.rfft(a_new, n=L)
    conv_new_full = np.fft.irfft(Af_new * Af_new, n=L)
    conv_new = conv_new_full[: 2 * n - 1]
    M_new = conv_new.max()
    return a_new, M_new, True


# ----------------------------------------------------------------------
# Stage 3 – deterministic greedy peak‑flattening (same idea as reference)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass from the “peak” index to a “valley” index."""
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = np.fft.irfft(np.fft.rfft(a, n=Lfft) *
                        np.fft.rfft(a, n=Lfft), n=Lfft)[:L]
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_int = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in the current maximum
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # choose destination that has the smallest partner value
        partner_vals = a[max_idx - idx]
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size ε
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid numerical drift
        if it % recompute_int == 0:
            conv = np.fft.irfft(np.fft.rfft(a, n=Lfft) *
                               np.fft.rfft(a, n=Lfft), n=Lfft)[:L]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence found as a list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressive power
    # and speed.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise with a handful of deterministic shapes
    # --------------------------------------------------------------
    initial = []

    # historic construction, if available
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)
        if seq0.ndim == 1 and seq0.size > 0:
            seq0 = np.interp(np.linspace(0, 1, n),
                             np.linspace(0, 1, seq0.size),
                             seq0)
            seq0 = np.maximum(seq0, 0.0)
            seq0 = seq0 / seq0.sum()
            initial.append(seq0)
    except Exception:
        pass

    # uniform
    initial.append(np.full(n, 1.0 / n))

    # linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    initial.append(lin / lin.sum())

    # sqrt‑law
    sqrt_seq = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    initial.append(sqrt_seq / sqrt_seq.sum())

    # exponential decay
    q = 0.9995
    exp_seq = (1.0 - q) * (q ** np.arange(n, dtype=float))
    initial.append(exp_seq / exp_seq.sum())

    # a couple of random Dirichlet draws
    for _ in range(2):
        rnd = np.random.rand(n)
        rnd = rnd / rnd.sum()
        initial.append(rnd)

    best_seq = None
    best_score = float("inf")

    # quick screening of the deterministic candidates
    for cand in initial:
        if time.time() >= deadline:
            break
        short_dead = time.time() + min(5.0, deadline - time.time())
        seq, scr = _softmax_descent_short(cand, n, short_dead, start_time)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] init improve: score={scr:.6f}")

    if best_seq is None:                     # fallback (should never happen)
        best_seq = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Stage 1 – long smoothed exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase1_dead = time.time() + remaining * 0.55
        seq, scr = _softmax_descent(best_seq, n, phase1_dead, start_time,
                                    eta0=0.5,
                                    tau_init_factor=0.6,
                                    tau_decay=0.96,
                                    min_tau_factor=0.001)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] main softmax best: score={scr:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – linear‑programming improvement
    # --------------------------------------------------------------
    if time.time() < deadline:
        time_for_lp = (deadline - time.time()) * 0.25
        lp_dead = time.time() + time_for_lp
        a = best_seq.copy()
        L = _next_pow2(3 * n - 2)
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        M = conv_full.max()
        while time.time() < lp_dead:
            a, M, improved = _lp_improve_step(a, n, conv_full, M,
                                              t_mix=0.3, rel_tol=1e-4)
            if not improved:
                break
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] LP improve score={cur_score:.6f}")
            # recompute convolution for the next iteration
            Af = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
            M = conv_full.max()

    # --------------------------------------------------------------
    # Stage 3 – deterministic greedy flattening (uses the whole rest)
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_greedy_reduction(best_seq, n, deadline, start_time,
                                         best_score, best_seq,
                                         max_iters=10_000_000)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy improve: score={scr:.6f}")

    # --------------------------------------------------------------
    # Optional tiny random‑restart polishing if a few seconds remain
    # --------------------------------------------------------------
    while time.time() < deadline - 2.0:
        pert = best_seq + 0.005 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)
        total = pert.sum()
        if total > 0.0:
            pert = pert / total
        else:
            pert = np.full(n, 1.0 / n)

        short_dead = time.time() + 1.5
        seq, scr = _softmax_descent(pert, n, short_dead, start_time,
                                    eta0=0.5,
                                    tau_init_factor=0.6,
                                    tau_decay=0.96,
                                    min_tau_factor=0.001,
                                    max_iters=200_000)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑restart improve: score={scr:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (the objective is homogeneous)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    # return plain Python list
    return [float(x) for x in best_seq]
