"""
Propose a step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four phases:

1. **Broad random parametric sweep** – many inexpensive families (geometric,
   power, beta, mixtures and Dirichlet draws) are sampled with a fine
   granularity.  The best candidate is kept.

2. **Smooth descent with back‑tracking line search** – starting from the best
   candidate a soft‑max approximation of the max‑convolution is optimised.
   The line‑search guarantees that the true (non‑smoothed) objective never
   increases.  Nesterov‑style momentum is not used – a simple step‑size
   1/√t schedule together with a multiplicative temperature decay works well.

3. **Projected sub‑gradient refinement** – the exact non‑smooth objective
   is tackled with a classic sub‑gradient method (momentum, 1/√t steps)
   and projection onto the probability simplex.

4. **Directed hill‑climbing polish** – mass is moved from indices that
   contribute most to the current maximal convolution value to random
   destinations; moves are kept only when they improve the true objective.

All phases respect the global time‑budget (default 1000 s).  The final
sequence is normalised to have unit ℓ₁‑norm (the objective is scale‑invariant)
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the exact objective for a (sum≈1) sequence
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Random parametric sweep – very cheap, large coverage
# ----------------------------------------------------------------------
def _random_sweep(n: int,
                  L_fft: int,
                  rng: np.random.Generator,
                  start_time: float,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """Explore a wide variety of low‑dimensional families."""
    idx = np.arange(n)

    while time.time() < deadline:
        # pick a family at random
        family = rng.integers(5)

        if family == 0:          # pure geometric
            eps = 10 ** rng.uniform(-6, -0.5)       # eps = 1‑r
            r = 1.0 - eps
            seq = np.power(r, idx, dtype=np.float64)

        elif family == 1:        # power law
            p = rng.uniform(-5.0, 5.0)
            seq = np.power(idx + 1, p, dtype=np.float64)

        elif family == 2:        # beta‑type (two‑sided power)
            p = rng.uniform(0.0, 5.0)
            q = rng.uniform(0.0, 5.0)
            seq = np.power(idx + 1, p) * np.power(n - idx, q)

        elif family == 3:        # mixture uniform‑geometric
            eps = 10 ** rng.uniform(-6, -0.5)
            r = 1.0 - eps
            lam = rng.random()
            geo = np.power(r, idx, dtype=np.float64)
            uniform = np.full(n, 1.0 / n)
            seq = lam * uniform + (1.0 - lam) * geo

        else:                    # random Dirichlet (controlled concentration)
            concentration = rng.uniform(0.05, 10.0)
            seq = rng.gamma(concentration, size=n)

        # normalise
        seq = np.maximum(seq, 0.0)
        tot = seq.sum()
        if tot == 0.0:
            continue
        seq /= tot

        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Random sweep family={family} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smoothed gradient descent with line‑search (soft‑max of the max‑conv)
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed version of the objective.
    A back‑tracking line‑search guarantees monotone decrease of the true
    (non‑smoothed) objective.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    tau = 0.5                # initial temperature for soft‑max
    tau_min = 1e-8
    tau_decay = 0.9995       # slower decay than the reference implementation
    step0 = 1.0

    while time.time() < deadline:
        it += 1

        # --- convolution and soft‑max -------------------------------------------------
        conv = _conv_self(a, L=L_fft)               # length 2n‑1
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)      # stable soft‑max
        w = exp_vals / exp_vals.sum()               # sum(w)=1

        # --- gradient of the smoothed objective -------------------------------------
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # --- back‑tracking line search ------------------------------------------------
        step = step0 / math.sqrt(it)
        cur_score = best_score   # dummy value to enter the loop
        while True:
            a_new = _project_simplex(a - step * grad)
            new_score = _eval_fast(a_new, L_fft)
            if new_score < best_score or step < 1e-12:
                break
            step *= 0.5

        a = a_new
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth LS it={it} τ={tau:.2e} score={new_score:.6f}")

        # --- temperature decay -------------------------------------------------------
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Projected sub‑gradient refinement (identical to reference but with a larger budget)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient method on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient (active index)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climbing polish (moves mass away from the active convolution peak)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Cheap local search: mass is taken from indices that contribute most to the
    current maximal convolution value and moved elsewhere, keeping the move only
    when the true objective improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        k = int(conv.argmax())

        # contribution of each index to the current maximal entry
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        prod = a[idx] * a[k - idx]

        if prod.sum() == 0.0:
            src = rng.choice(n, p=a)
        else:
            probs = prod / prod.sum()
            src = rng.choice(idx, p=probs)          # biased towards “heavy” contributors

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # move up to 10 % of the source mass
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score

        n, L_fft, np.random.default_rng(),
        start_time, sweep_deadline,
        best_seq, best_score
    )

    # ------------------------------------------------------------------
    # Phase 2 – smooth descent with line search (≈ 55 % of the budget)
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:
        # generate a perturbed start point around the current best
        base = best_seq if np.random.rand() < 0.8 else uniform
        noise = 0.03 * np.random.randn(n) * base          # modest Gaussian perturbation
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 150.0,
                           time.time() + 12.0)          # ≤12 s per restart
        cand_seq, cand_score = _smooth_descent(start,
                                                sub_deadline,
                                                start_time,
                                                L_fft,
                                                L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after smooth restart score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient refinement (≈ 25 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sub_deadline = deadline - 80.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – directed hill‑climb polish (last ~80 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improve score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
