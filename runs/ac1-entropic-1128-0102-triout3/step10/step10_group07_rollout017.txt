    fft_len = _next_pow2(2 * n - 1)

"""
Candidate search using soft‑max weighted gradient descent (projected
gradient descent on a smoothed version of the max‑convolution).  The
algorithm keeps a pool of diverse initial vectors, picks the best and
refines it with many fast FFT‑based gradient steps; occasional random
restarts and a final deterministic greedy mass‑transfer are added.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT of length ``fft_len``."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _gradient_weighted(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of ``∑_k w_k (a*a)[k]`` w.r.t. ``a``.
    The formula is   g_i = 2 * Σ_j a_j * w_{i+j}.
    It is computed by a convolution of the reversed ``a`` with ``w``.
    """
    rev = a[::-1]                     # reversed a
    Af = np.fft.rfft(rev, n=fft_len)
    Bf = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(Af * Bf, n=fft_len)
    n = a.shape[0]
    start = n - 1                     # the slice we need
    grad = 2.0 * full[start:start + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc``."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Discrete Gaussian centred on the interval."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:

    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Optimisation core – soft‑max weighted projected gradient descent
# ----------------------------------------------------------------------


def _softmax_gd_optimize(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed version of the max‑convolution.
    The smoothing is performed by a soft‑max with a linearly increasing
    inverse temperature ``alpha``.
    """
    a = np.maximum(a0, 1e-12)          # avoid exact zeros
    a = a / a.sum()
    n = a.shape[0]

    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)      # for a_rev * w

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # current convolution and objective
        conv = _conv_self_fft(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] softmax‑gd improve iter={t:,} "
                  f"score={cur_score:.6f}")

        # ---- soft‑max weighting ------------------------------------
        # ``alpha`` grows slowly – the objective becomes closer to the real max.
        alpha = min(200.0, 5.0 + 0.005 * t)

        # shift for numerical stability (max_b is subtracted)
        shift = conv - cur_max                     # ≤ 0
        exp_term = np.exp(alpha * shift)          # at least one entry equals 1
        w = exp_term / exp_term.sum()             # non‑negative, sums to 1

        # ---- gradient of the smoothed objective --------------------
        grad = _gradient_weighted(a, w, fft_len_grad)

        # ---- simple projected gradient step -----------------------
        eta = 0.2 / math.sqrt(t)                  # decaying step size
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)          # keep non‑negative
        s = a_new.sum()
        if s > 0.0:
            a = a_new / s
        else:
            a = np.full_like(a, 1.0 / n)

    return best_a, best_score

"""
Candidate search using soft‑max weighted gradient descent.

The algorithm builds a diverse pool of initial vectors, selects the best
one, and then refines it with a long phase of projected gradient descent
on a smoothed version of the max‑convolution (soft‑max).  Short random‑
restart bursts and a final deterministic greedy mass‑transfer are added
to escape possible local minima.  All updates are performed with FFT‑based
convolutions, making each iteration cheap enough to allow thousands of
iterations within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT of length *fft_len*."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _gradient_weighted(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of ∑_k w_k (a*a)[k] w.r.t. ``a``.
    The formula is  g_i = 2·∑_j a_j·w_{i+j}.
    It is obtained by convolving the reversed a with w.
    """
    rev = a[::-1]                     # reversed a
    Af = np.fft.rfft(rev, n=fft_len)
    Bf = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(Af * Bf, n=fft_len)
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w
