def _adam_softmax_descent(start_a, deadline, start_time, L_fft, L_grad):
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    it = 0
    tau = 0.5
    tau_min = 1e-6
    tau_decay = 0.995

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        # compute softmax weights
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()
        # compute gradient via convolution of a and reversed w
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 4.0 * n * grad_full[offset - np.arange(n)]  # gradient

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = 0.1 / (math.sqrt(it))  # maybe decays = 1/sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # projection onto simplex
        a = _project_simplex(a)

        # evaluate exact score (exp may be expensive; we compute occasionally)
        if it % 5 == 0:
            cur_score = _eval_fast(a, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

"""
Propose a non‑negative step function (length 4096) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

by a three‑stage heuristic:

1. **Broad parametric sweep** – explores several analytic families
   (Beta‑type, power, symmetric geometric, etc.) including negative
   exponents that concentrate mass near the ends.  All candidates are
   normalised to the simplex and the best is kept.

2. **Annealed Adam descent** – smoothes the max by a soft‑max
   (temperature τ) and runs a projected Adam optimiser on the simplex.
   The temperature is exponentially cooled, the step‑size follows a
   1/√t schedule and after each iteration we re‑project onto the
   probability simplex.  Several random restarts are performed until
   the time budget is almost exhausted.

3. **Fine‑grained polishing** – a projected sub‑gradient descent
   (momentum) followed by a cheap deterministic hill‑climb that moves
   tiny amounts of mass between coordinates only when the true objective
   improves.

The routine respects the 1000 s wall‑clock limit, prints progress
messages and finally returns a list of non‑negative floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w≥0, Σw=1}.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation for a (approximately normalised) sequence
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climbing polish (same as reference)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                # pick proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10      # move up to 10 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Annealed Adam descent on the soft‑max smoothed objective
# ----------------------------------------------------------------------
def _adam_descent(start_a: np.ndarray,
                  deadline: float,
                  start_time: float,
                  L_fft: int,
                  L_grad: int) -> tuple[np.ndarray, float]:
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    # soft‑max temperature schedule
    tau = 0.5
    tau_min = 1e-8
    tau_decay = 0.995
    # base learning‑rate (will be scaled by 1/√t)
    lr0 = 0.2

    while time.time() < deadline:
        it += 1

        # --- convolution and soft‑max weights ---------------------------------
        conv = _conv_self(a, L=L_fft)                 # length 2n‑1
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()                 # soft‑max probabilities

        # --- gradient of the smoothed objective -----------------------------
        # (∂/∂a) 2 n τ log Σexp(conv/τ) = 4 n (a * w_rev)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 4.0 * n * grad_full[offset - np.arange(n)]

        # --- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / math.sqrt(it)                    # 1/√t decay
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # --- simplex projection ---------------------------------------------
        a = _project_simplex(a)

        # periodic exact evaluation (cheap but not every iteration)
        if it % 5 == 0:
            cur_score = _eval_fast(a, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # --- temperature decay ------------------------------------------------
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score
