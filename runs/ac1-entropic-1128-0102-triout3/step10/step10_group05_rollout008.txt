def compute_loss_grad(a, epsilon):
    n = a.shape[0]
    L = next_pow2(2*n-1)
    conv = conv_self(a, L)
    maxc = conv.max()
    # compute weights
    # To avoid underflow: subtract maxc to get max exponent 0
    diff = conv - maxc
    if epsilon <= 0: # treat as exact max
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = np.exp(diff / epsilon)
        w_sum = np.sum(w)
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

    # loss
    M = np.dot(w, conv)
    loss = 2.0 * n * M

    # gradient wrt a
    w_rev = w[::-1]
    # length for FFT
    Lg = next_pow2(n + w_rev.size - 1)
    Af = np.fft.rfft(a, n=Lg)
    Wrev = np.fft.rfft(w_rev, n=Lg)
    corr = np.fft.irfft(Af * Wrev, n=Lg)[: n + w_rev.size - 1]
    # gradient of M wrt a
    gradM = 2.0 * corr[w_rev.size-1 - np.arange(n)]
    # gradient of loss wrt a: 2*n*gradM
    grad_a = 2.0 * n * gradM

    # gradient wrt x (softmax params)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    return loss, grad_x

def run_lbfgs(a0, epsilon, maxiter, deadline):
    n = a0.shape[0]
    x0 = np.log(a0 + 1e-12)  # avoid log(0)
    def fun(x):
        a = np.exp(x)
        sum_a = np.sum(a)
        if sum_a == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a
        loss, gradx = compute_loss_grad(a, epsilon)
        return loss, gradx
    # set options
    opts = {'maxiter': maxiter, 'disp': False}
    res = opt.minimize(fun, x0, method='L-BFGS-B', jac=True, options=opts)
    # after optimization, compute a
    a_opt = np.exp(res.x)
    a_opt /= a_opt.sum()
    return a_opt, res.fun, res

class _LBFGSProblem:
    def __init__(self, n, epsilon):
        self.n = n
        self.epsilon = epsilon
        self.Lc = _next_pow2(2 * n - 1)
        # for gradient convolution length Lg maybe
        self.Lg = _next_pow2(3 * n - 2)
    def __call__(self, x):
        # compute a from x
        a = np.exp(x)
        total = a.sum()
        if total == 0:
            a = np.full(self.n, 1.0 / self.n)
        else:
            a = a / total
        # compute conv
        Af = np.fft.rfft(a, n=self.Lc)
        conv = np.fft.irfft(Af * Af, n=self.Lc)[: 2 * self.n - 1]
        maxc = conv.max()
        if self.epsilon <= 0:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            diff = conv - maxc
            w = np.exp(diff / self.epsilon)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        # loss
        M = np.dot(w, conv)
        loss = 2.0 * self.n * M
        # gradient wrt a
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=self.Lg)
        Wrev = np.fft.rfft(w_rev, n=self.Lg)
        corr = np.fft.irfft(Af * Wrev, n=self.Lg)[: self.n + w_rev.size - 1]

        grad_a = 2.0 * self.n * gradM
        # gradient wrt x using softmax derivative
        dot = np.dot(a, grad_a)

The algorithm proceeds in three phases:

    amount of mass from the most “responsible’’ coordinate to a
    “least‑responsible’’ one.  The convolution is updated incrementally
    for speed; the loop runs until the time budget expires.

All steps respect the 1000 s deadline and keep the sequence non‑negative
and normalised (Σ a_i = 1).  The final sequence is returned as a plain
list of floats.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i (r∈(0,1)), normalised to sum 1."""
    if not (0.0 < r < 1.0):
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type shape a_i ∝ (i+1)^α (n‑i)^α, normalised."""
    i = np.arange(n, dtype=float)
    a = np.power(i + 1.0, alpha, where=(i + 1.0) > 0) * np.power(
        n - i, alpha, where=(n - i) > 0
    )
    # avoid all zeros (possible for very negative α)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    conv = _conv_self(a)
    maxb = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * maxb / (s * s)


def _enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Project onto the symmetric subspace (a_i = a_{n-1-i})."""
    a = (a + a[::-1]) * 0.5
    total = a.sum()
    if total > 0.0:
        a = a / total
    else:
        a = np.full_like(a, 1.0 / a.shape[0])
    return a


# ----------------------------------------------------------------------
# Smooth L‑BFGS problem
# ----------------------------------------------------------------------
class _LBFGSProblem:
    """
    Callable object for SciPy L‑BFGS.  Uses a soft‑max parametrisation
    a_i = exp(x_i) / Σ exp(x_j) to enforce positivity and Σ a_i = 1.
    The smoothed max is a log‑sum‑exp with temperature ε.
    """

    def __init__(self, n: int, epsilon: float):
        self.n = n
        self.epsilon = epsilon
        # one FFT length large enough for both convolution and the
        # correlation with the weights (size 3·n‑2)
        self.L = _next_pow2(3 * n - 2)

    def __call__(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        # soft‑max transformation (stable)
        x_shift = x - np.max(x)
        a = np.exp(x_shift)
        a_sum = a.sum()
        if a_sum == 0.0 or np.isnan(a_sum):
            a = np.full(self.n, 1.0 / self.n)
        else:
            a = a / a_sum

        # FFT of a (used for both convolution and gradient)
        Af = np.fft.rfft(a, n=self.L)

        # ----- self‑convolution -----
        conv = np.fft.irfft(Af * Af, n=self.L)[: 2 * self.n - 1]

        # ----- smoothed max (log‑sum‑exp) -----
        maxc = conv.max()
        if self.epsilon <= 0.0:
            # hard max → one‑hot weight
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            diff = conv - maxc                     # ≤ 0
            w = np.exp(diff / self.epsilon)
            ws = w.sum()
            if ws == 0.0 or np.isnan(ws):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / ws

        # ----- loss -----
        M = np.dot(w, conv)                 # smoothed max
        loss = 2.0 * self.n * M

        # ----- gradient wrt a (derived from ∂/∂a Σ w_k (a*a)_k) -----
        w_rev = w[::-1]
        Wrev = np.fft.rfft(w_rev, n=self.L)
        corr = np.fft.irfft(Af * Wrev, n=self.L)[: self.n + w_rev.size - 1]
        grad_M = 2.0 * corr[w_rev.size - 1 - np.arange(self.n)]   # dM/da
        grad_a = 2.0 * self.n * grad_M                           # dloss/da

        # ----- chain rule to x (soft‑max) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        return float(loss), grad_x


def _run_lbfgs_stage(a_start: np.ndarray,
                     epsilon: float,
                     max_iter: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on the smoothed objective with a fixed temperature ε.
    Returns the refined sequence, the final loss value and the SciPy result.
    """
    n = a_start.shape[0]
    x0 = np.log(a_start + 1e-12)          # initialise soft‑max parameters
    prob = _LBFGSProblem(n, epsilon)

    # small wrapper for SciPy (stores time so we can abort early)
    options = {"maxiter": max_iter, "disp": False}
    # SciPy may internally call the function many times; we let it run
    # but we will stop after the deadline by raising an exception.
    def callback(_):
        if time.time() >= deadline:
            raise RuntimeError("time out")
    try:
        res = optimize.minimize(
            prob, x0, method="L-BFGS-B", jac=True, options=options, callback=callback
        )
    except RuntimeError:
        # deadline reached – return the current best (SciPy stores the last x)
        # res may be undefined; we fall back to the current x0
        # (the optimisation did not finish, but we still have a usable point)
        # Re‑evaluate at the last x stored in prob (no easy access) → just use a_start
        return a_start, float("inf")

    a_opt = np.exp(res.x - np.max(res.x))
    a_opt = a_opt / a_opt.sum()
    return a_opt, float(res.fun)


# ----------------------------------------------------------------------
# Greedy reduction (identical in spirit to the previous implementation)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          deadline: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Aggressive deterministic reduction of the peak convolution."""
    n = a0.shape[0]
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a)                     # full convolution (len 2n‑1)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second highest entry (ignore current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices participating in this maximum
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and candidate destinations
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # compute a safe step size ε
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution (exact for the modified positions) -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # periodic exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``
    under the prescribed time budget.  Returns the best sequence as a plain
    Python list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Length of the candidate (fixed to the value used in the reference)
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise – either from the previously discovered construction
    # or from a quick exponential scan.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            a0 = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, starter.size),
                starter,
            )
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            raise Exception()
    except Exception:
        # quick exponential scan for a good decay parameter
        best_score = float("inf")
        best_r = 0.99
        for r in np.linspace(0.90, 0.9999, 250):
            cand = _exponential_sequence(n, r)
            conv = _conv_self(cand)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_r = r
        a0 = _exponential_sequence(n, best_r)
        print(f"[0.00s] exponential seed r={best_r:.6f}, score={best_score:.6f}")

    best_seq = a0.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()
    print(f"[0.00s] initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – low‑dimensional beta‑shape sweep (very cheap)
    # --------------------------------------------------------------
    if time.time() < deadline:
        alphas = np.linspace(-1.0, 3.0, 200)   # coarse sweep
        for alpha in alphas:
            cand = _beta_shape(n, alpha)
            cur_score = 2.0 * n * _conv_self(cand).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] beta‑sweep α={alpha:.3f}, improve: score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth L‑BFGS refinement (continuation on ε)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # decreasing temperature schedule (coarse → fine)
        eps_schedule = [1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5]
        for eps in eps_schedule:
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            # allocate at most a few seconds per L‑BFGS stage – enough for convergence
            max_iter = min(300, int(remaining * 5))   # heuristic
            a_opt, loss_opt = _run_lbfgs_stage(best_seq, eps, max_iter, deadline)
            a_opt = _enforce_symmetry(a_opt)       # keep symmetry (helps flattening)
            cur_score = 2.0 * n * _conv_self(a_opt).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] L‑BFGS ε={eps:.1e} improve: score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – aggressive greedy reduction (dominant time consumer)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # give the greedy pass the bulk of the remaining budget
        # (max_iters is deliberately generous – the loop exits on time)
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq,
            deadline,
            best_score,
            best_seq,
            max_iters=30_000_000,
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] final greedy improve: score={score_greedy:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_obj = 2.0 * n * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
