"""
Propose a low‑objective step‑function.

The algorithm consists of three stages:

1. **Warm‑start** – import the global best shape (if available) and,
   otherwise, initialise a symmetric tent.  The vector is rescaled to lie
   on the probability simplex (∑a = 1).

2. **Smoothed gradient descent** – minimise a smooth approximation of the
   original objective
         2·n·max(conv(a,a)) / (∑a)²
   by replacing the max with a log‑sum‑exp (parameter α).  The exact
   gradient of the smooth objective is obtained via two FFT‑based
   convolutions, after which an Adam optimiser with simplex projection
   updates the variables.  The smoothing parameter is increased in stages
   (α = 5 → 10 → 20 → 40 → 80) to approach the true nonsmooth problem.

3. **Light annealing polish** – a cheap simulated‑annealing pass moves tiny
   packets of mass between coordinates to escape possible shallow minima.

All work is performed within the given time budget (default 1000 s).  The
returned list contains non‑negative floats and the sum is normalised to 1,
so the scale‑invariant evaluation function is unaffected.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv)/ (sum)² (scale‑invariant)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, Σ w = 1} (O(n log n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed objective and exact gradient (via FFT)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, alpha: float):
    """
    Returns a smooth surrogate of the original objective and its gradient.
    The surrogate replaces max(conv) by (1/α)·log Σ exp(α·conv).
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _conv_self(a)                     # length 2·n‑1
    maxc = conv.max()

    # 2) soft‑max weights p_i = exp(α·conv_i)/Σexp(...)
    #    we subtract maxc for numerical stability
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp

    # 3) gradient of Σ p_i·conv_i w.r.t. a:
    #    ∂conv_i/∂a_j = 2·a_{i‑j} (zero outside the support)
    rev_a = a[::-1]
    L = _next_pow2(len(p) + len(rev_a) - 1)
    P_fft = np.fft.rfft(p, n=L)
    A_fft = np.fft.rfft(rev_a, n=L)
    conv_pr = np.fft.irfft(P_fft * A_fft, n=L)[: len(p) + len(rev_a) - 1]
    # slice that corresponds to indices j = 0 … n‑1
    grad_slice = conv_pr[n - 1 : n - 1 + n]

    # 4) full gradient of the smooth objective (sum = 1, so denominator constant)
    #    grad = 2·n·∑_i p_i·∂conv_i/∂a = 2·n·(2·grad_slice) = 4·n·grad_slice
    grad = 4.0 * n * grad_slice

    # 5) smooth objective value
    smooth_obj = 2.0 * n * (maxc + np.log(sum_exp) / alpha)

    return smooth_obj, grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (projected)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   alpha: float,
                   deadline: float,
                   start_time: float,
                   print_every: int = 200):
    """
    Projected Adam on the smoothed objective.
    Returns the best iterate (on the true objective) seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.025                     # base learning rate (tuned empirically)

    best_a = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # smooth obj & gradient
        _, grad = _smooth_obj_and_grad(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_simplex(a)

        # occasional true‑objective evaluation
        if it % print_every == 0:
            cur = _objective(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.1f} "
                      f"iter={it:,} score={cur:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated‑annealing polish (as in the previous solution)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny random mass‑transfer annealing."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    total_steps = 30_000

    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling (fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        # pick a source proportionally to its mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")
    return best_a, best_score
