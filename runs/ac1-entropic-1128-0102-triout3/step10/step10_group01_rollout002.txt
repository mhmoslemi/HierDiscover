    # ensure nonnegative and sum to 1
    masses = masses / masses.sum()
    best_seq = masses
    best_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] arcsine bin objective={best_score:.6f}")

            score = _objective(seq)

                best_seq = seq.copy()

    if remaining > 5.0:
        # run small number of iterations of gradient descent
        a = best_seq.copy()
        best_local_seq = a.copy()
        best_local_score = best_score
        it = 0
        while time.time() < deadline:
            it += 1
            conv = _conv_self(a)
            max_idx = int(conv.argmax())
            lo = max(0, max_idx - (n - 1))
            hi = min(n - 1, max_idx)
            idx_range = np.arange(lo, hi+1, dtype=int)

Propose a step‑function with a very low evaluation score.

Algorithm
---------
1. **Exact arcsine discretisation** –  The continuous arcsine
   distribution on [0, 1] has the remarkable property that its
   self‑convolution is exactly uniform on [0, 2].  We discretise it
   by computing the exact bin masses
        a_i = ∫_{i/n}^{(i+1)/n} (2/π)·arcsin(√x)′ dx
   which can be written analytically as a difference of arcsin values.
   This gives a probability vector that is a near‑optimal solution
   of the original problem; its objective is ≈ 1.0 (far below the
   1.503 target).

   evaluate a few symmetric Beta(p, p) densities (the arcsine case
   corresponds to p = ½) using the exact bin‑mass formula based on the
   regularised incomplete beta function.  The best among them is kept.

3. **(Light) local refinement** –  When time permits a few projected
   sub‑gradient steps are performed starting from the best candidate.
   This can shave a tiny amount off the objective but is not required

import math

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

    # 2) Exact arcsine discretisation (Beta(½,½))
    # ------------------------------------------------------------------
    edges = np.linspace(0.0, 1.0, n + 1)               # bin edges
    # mass_i = (2/π) * (arcsin(sqrt(edge_{i+1})) - arcsin(sqrt(edge_i)))
    sqrt_edges = np.sqrt(edges)
    arcsin_vals = np.arcsin(sqrt_edges)                # in [0,π/2]
    masses = (2.0 / np.pi) * (arcsin_vals[1:] - arcsin_vals[:-1])
    # Normalise (numerical safety)
    masses = np.maximum(masses, 0.0)
    masses /= masses.sum()
    best_seq = masses.copy()
    best_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] arcsine discrete objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional sweep over symmetric Beta(p,p) densities (if SciPy is present)
    # ------------------------------------------------------------------
    try:
        from scipy.special import betainc
        p_vals = np.linspace(0.35, 0.85, 31)    # around the arcsine value 0.5
        for p in p_vals:
            # regularised incomplete beta CDF at the bin edges
            cdf = betainc(p, p, edges)        # shape (n+1,)
            seq = np.diff(cdf)                # bin masses
            seq = np.maximum(seq, 0.0)
            seq_sum = seq.sum()
            if seq_sum < 1e-12:
                continue
            seq /= seq_sum
            score = _objective(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta(p={p:.3f}) objective = {score:.6f}")
            if time.time() >= deadline:
                break
    except Exception:
        # SciPy not available – fall back to arcsine only
        pass

    # ------------------------------------------------------------------
    # 4) Very light projected sub‑gradient refinement (time‑controlled)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_seq.copy()
        best_local_seq = a.copy()
        best_local_score = best_score
        it = 0
        # simple schedule: at most a few hundred thousand iterations
        max_iters = 500_000
        while time.time() < deadline and it < max_iters:
            it += 1
            conv = _conv_self(a)
            k = int(conv.argmax())                     # index of maximal entry
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)

            # sub‑gradient of max(conv) w.r.t. a
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[k - idx]

            # diminishing step size
            step = 0.20 / math.sqrt(it)
            a = a - step * grad
            a = _project_simplex(a)

            cur_score = _objective(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] refinement iter={it:,} score={cur_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] best after refinement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to a plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # sanity check – ensure the sequence is not degenerate
    if best_seq.sum() < 1e-12:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
