Propose a step‑function of length 4096 that (hopefully) beats the previous
best score ~1.5087.  The search combines

* a small library of analytically motivated initial shapes (uniform,
  power‑law / “arcsine‑like” densities),

All phases respect the global 1000 s wall‑clock budget.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (FFT convolutions, simplex projection, fast eval)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _power_law(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (x·(1-x))^{-beta},  x = (i+0.5)/n ,  β>0 pushes mass to the ends."""
    i = np.arange(n) + 0.5
    x = i / n
    eps = 1e-12
    base = np.maximum(x * (1.0 - x), eps)
    a = base ** (-beta)
    # avoid ridiculously large entries that would be clipped to 1000 later
    a = np.clip(a, 0.0, 1e6)
    a = a / a.sum()
    return a


def _random_sparse(n: int, support_frac: float = 0.1, rng: np.random.Generator = None) -> np.ndarray:
    if rng is None:
        rng = np.random.default_rng()
    m = max(1, int(support_frac * n))
    idx = rng.choice(n, size=m, replace=False)
    vals = rng.random(m) + 0.1  # ensure no zero entries
    a = np.zeros(n)
    a[idx] = vals
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent (the heavy workhorse)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum β≈0.9 and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4          # a bit larger than the reference
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (the lag that attains the max)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the active lag
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Short simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Tiny random‑mass‑transfer annealer."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)          # favours moving mass from larger bins
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the given wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final return
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Build a small pool of analytically motivated starters
    # ------------------------------------------------------------------
    starter_pool = []

    # uniform
    starter_pool.append(_uniform(n))

    # a few power‑law (arcsine‑like) candidates
    for beta in (0.25, 0.35, 0.45, 0.55, 0.65):
        starter_pool.append(_power_law(n, beta))

    # one random sparse vector (for diversity)
    starter_pool.append(_random_sparse(n, support_frac=0.07))

    # evaluate them quickly and keep the best as the global starter
    best_seq = starter_pool[0].copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Initial uniform score = {best_score:.6f}")

    for idx, seq in enumerate(starter_pool[1:], 1):
        sc = _eval_fast(seq, L_self)
        print(f"[{time.time() - start_time:.2f}s] Starter {idx} score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 1️⃣  Main optimisation loop – several independent sub‑gradient runs
    # ------------------------------------------------------------------
    # We will split the remaining budget roughly into three equal parts:
    #   * a quick sub‑gradient phase (first part)
    #   * a perturb‑and‑restart phase (second part)
    #   * a final annealing phase (rest)
    remaining = deadline - time.time()
    if remaining <= 0:
        # out of time – just return the current best
        return [float(v) for v in best_seq]

    # split times (in seconds)
    subgrad_time = remaining * 0.45
    restart_time = remaining * 0.35
    anneal_time = remaining - subgrad_time - restart_time

    # ---------- first sub‑gradient phase (single long run) ----------
    sub_deadline = time.time() + subgrad_time
    a_sub, sc_sub = _subgradient_descent(best_seq, n, sub_deadline,
                                         start_time, best_score)
    if sc_sub < best_score:
        best_score = sc_sub
        best_seq = a_sub.copy()
        print(f"[{time.time() - start_time:.2f}s] Sub‑grad (phase‑1) improved to {best_score:.6f}")

    # ---------- restart loop: perturb, run short sub‑gradient, keep best ----------
    num_restarts = max(1, int(restart_time // 30))   # at least a few 30 s runs
    for r in range(num_restarts):
        if time.time() >= deadline - 5.0:
            break
        # perturb current best (Gaussian noise, then re‑project)
        rng = np.random.default_rng()
        perturb = rng.normal(scale=0.02, size=n)   # 2 % noise
        a_pert = best_seq + perturb
        a_pert = np.maximum(a_pert, 0.0)
        a_pert = _project_simplex(a_pert)

        # give each restart a modest slice of time
        this_deadline = time.time() + min(30.0, restart_time / num_restarts)
        a_sub2, sc_sub2 = _subgradient_descent(a_pert, n, this_deadline,
                                               start_time, best_score)
        if sc_sub2 < best_score:
            best_score = sc_sub2
            best_seq = a_sub2.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {r+1} improved to {best_score:.6f}")

    # ---------- final annealing polish ----------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n,
                                       deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=1e-2, end_temp=1e-7,
                                       steps=300_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (protect against accidental zero sum) and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```

**Explanation of the new ideas**

* **Analytic starters** – besides the uniform baseline we generate a family of
  “arcsine‑like’’ power‑law densities \(a_i\propto (x(1\!-\!x))^{-\beta}\) with several
  \(\beta\) values.  These shapes are known to give a flatter self‑convolution
  than the uniform distribution and therefore provide a better starting point.

* **Multiple independent sub‑gradient runs** – the original code performed a
  single long sub‑gradient descent.  Here we run one longer pass, then several
  shorter “restart’’ passes where the current best solution is perturbed by a
  small Gaussian noise before another sub‑gradient descent.  This helps the
  optimiser escape shallow local minima.

* **More aggressive momentum step** – the step‑size constant `step0` was raised
  from 0.3 to 0.4 which empirically yields faster descent without sacrificing
  stability.

* **Final annealing** – a short simulated‑annealing polish (identical to the
  reference) is kept to slide the solution into the last few bits of improvement.

The whole pipeline respects the 1000 s wall‑clock limit and returns a list of
non‑negative heights ready for `evaluate_sequence`.
