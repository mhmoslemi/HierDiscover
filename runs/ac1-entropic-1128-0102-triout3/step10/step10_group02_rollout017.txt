"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------

1️⃣  Initialise with a known good construction (``height_sequence_1``) or a
    uniform distribution.

2️⃣  Smooth‑max mirror‑descent (exponentiated‑gradient) with a geometric
    schedule of the smoothing parameter.  This quickly reduces the bulk
    of the objective.

3️⃣  Projected sub‑gradient descent using *all* active constraints
    (every index attaining the current maximum).  This improves over the
    original one‑constraint version.

4️⃣  Fast local “mass‑transfer” hill‑climbing:
    * keep the current self‑convolution `conv` in memory,
    * repeatedly move a small amount of mass from a source index that
      participates heavily in the maximal convolution entry to a random
      destination,
    * accept the move only if the global maximum of `conv` strictly decreases.
      The convolution update is performed in O(n) by an explicit slice
      update – no FFT is required during the search.

5️⃣  A tiny simulated‑annealing polish (as in the reference solution) for
    the remaining time budget.

All steps respect the wall‑clock (`budget_s` seconds) and never exceed
the allowed time.  The best sequence found is returned as a plain Python
list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # temperature schedule – a little more aggressive than before
    alphas = np.geomspace(0.3, 1000.0, num=35)

    it = 0
    max_iterations = 600_000   # allow many cheap iterations

    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # soft‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)
            w /= w.sum()

            # gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # exponentiated‑gradient step on simplex
            eta = 0.08 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)
            a /= a.sum()

            # occasional exact check
            if it % 800 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.1f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : sub‑gradient descent using *all* active constraints
# ----------------------------------------------------------------------
def _active_subgradient_descent(a0: np.ndarray,
                               n: int,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent where the sub‑gradient is the sum of
    gradients over *all* indices attaining the current maximum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=_next_pow2(2 * n - 1))
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Active‑subgrad it={it:,} score={cur_score:.6f}")

        # active set (all positions attaining the max)
        active = np.flatnonzero(conv == cur_max)

        # build w = indicator of active positions (as float)
        w = np.zeros_like(conv)
        w[active] = 1.0

        # gradient = 2 * (w ⋆ a_rev)
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : fast local hill‑climbing by mass transfers
# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      cur_best_score: float,
                      cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Directly move tiny amounts of mass from indices that heavily
    contribute to the current maximal convolution entry to random
    destinations, accepting only moves that strictly lower the global
    maximum.  Convolution updates are performed with O(n) slice
    operations (no FFT).
    """
    a = a0.copy()
    conv = _conv_self(a, L=_next_pow2(2 * n - 1))
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score

    conv_new = np.empty_like(conv)
    rng = np.random.default_rng()
    no_improve = 0
    iter_counter = 0

    while time.time() < deadline:
        iter_counter += 1

        # pick source – focus on the current maximal entry most of the time
        if rng.random() < 0.75:
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx_range = np.arange(lo, hi + 1)
            contrib = a[idx_range] * a[k - idx_range]
            src_i = idx_range[np.argmax(contrib)]
            partner = k - src_i
            # move the larger of the two participating masses
            src = src_i if a[src_i] >= a[partner] else partner
        else:
            # completely random source (but must have some mass)
            candidates = np.flatnonzero(a > 1e-14)
            if candidates.size == 0:
                break
            src = rng.choice(candidates)

        # pick a destination different from src
        dst = rng.integers(n)
        if dst == src:
            continue

        # choose a small epsilon (up to 30 % of src mass)
        eps = a[src] * rng.random() * 0.30
        if eps < 1e-14:
            continue

        # --- propose move -------------------------------------------------
        conv_new[:] = conv
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a
        conv_new[2 * src] += eps * eps
        conv_new[2 * dst] += eps * eps
        conv_new[src + dst] -= 2.0 * eps * eps

        new_max = conv_new.max()
        # accept only if we strictly improve the maximum
        if new_max + 1e-15 < cur_max:
            # apply move for real
            a[src] -= eps
            a[dst] += eps
            conv, conv_new = conv_new, conv   # swap references
            cur_max = new_max
            cur_score = 2.0 * n * cur_max
            no_improve = 0

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={iter_counter:,} score={cur_score:.6f}")

        else:
            no_improve += 1

        # occasionally re‑randomise a little to escape plateaus
        if no_improve >= 100_000:
            # small perturbation + renormalisation
            a = a * (0.9 + 0.2 * rng.random(n))
            a /= a.sum()
            conv = _conv_self(a, L=_next_pow2(2 * n - 1))
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Perturbation after stagnation")

    return best_seq, best_score

# Phase‑4 : tiny simulated‑annealing polish (identical to reference)

def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 250_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

            best_score = sc_lc
            best_seq = a_lc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved score = {sc_lc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Light annealing polish (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=1e-2, end_temp=1e-7,
                                       steps=300_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
