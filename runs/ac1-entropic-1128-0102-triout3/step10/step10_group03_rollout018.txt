"""
Simple iterative optimizer for the step‑function bound.
Uses many random restarts of a smoothed soft‑max gradient descent,
followed by a long projected sub‑gradient phase and a final
hill‑climbing polish.  Hyper‑parameters are tuned for a longer
run (≈1000 s) to push the bound below the previous 1.5088 record.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (zero‑padded to length L)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast exact evaluation of the score (assumes a.sum()≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))

def _smooth_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int,
                    step0: float = 0.30,
                    tau0: float = 0.15,
                    tau_min: float = 1e-9,
                    tau_decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """Soft‑max (temperature τ) gradient descent on the true objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # convolution (full linear)
        conv = _conv_self(a, L=L_fft)                # length 2n‑1
        maxc = conv.max()
        # stable soft‑max over all entries
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()               # sum(w)=1

        # gradient of ∑_i w_i * (a ∗ a)_i
        # (a∗a)_i = Σ_j a_j a_{i-j}
        # Hence ∂/∂a_k Σ_i w_i Σ_j a_j a_{i-j}
        #    = 2 Σ_i w_i a_{i-k}  (where index i‑k in range)
        # We compute this convolution via FFT.
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)

        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # projected step
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        # exact score
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         step0: float = 0.35,
                         beta: float = 0.95) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent (max‑conv formulation)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    momentum = np.zeros_like(a)
    it = 0
    L_grad = _next_pow2(3 * a.shape[0] - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * a.shape[0] * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} score={cur_score:.6f}")

        # active shift (first occurrence)
        k = int(conv.argmax())
        lo = max(0, k - (a.shape[0] - 1))
        hi = min(a.shape[0] - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        # derivative of a_i*a_{k-i}
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int,
                max_move_frac: float = 0.12,
                trials_per_sec: int = 800) -> tuple[np.ndarray, float]:
    """Cheap deterministic hill‑climbing: move a small fraction of mass."""
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # batch many proposals per loop to reduce overhead
        for _ in range(trials_per_sec):
            it += 1
            src = rng.choice(n, p=a)              # pick proportionally to mass
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps = a[src] * rng.random() * max_move_frac
            if eps <= 0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_score = _eval_fast(a_prop, L_fft)
            if new_score < best_score:
                a = a_prop
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_score:.6f}")
        # allow time check after a batch
        if time.time() >= deadline:
            break

    return best_a, best_score

"""
Simple iterative optimizer for the step‑function bound.

The algorithm combines:
  • a quick scan of a few deterministic families (geometric, power, beta, cosine‑V),
  • many short random‑restart soft‑max gradient descents,
  • a long projected sub‑gradient refinement,
  • and a final hill‑climbing polish.

The hyper‑parameters are chosen to make good use of the full 1000 s budget,
aiming to beat the previous 1.5088 score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact score 2·n·max(conv(a,a)) / (sum(a)**2).  Assumes sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int,
                    step0: float = 0.35,
                    tau0: float = 0.20,
                    tau_min: float = 1e-9,
                    tau_decay: float = 0.9996) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed objective
        f(a) = 2·n·softmax_τ(conv(a,a))
    The temperature τ is exponentially decayed.
    The iterate stays on the simplex via Euclidean projection.
    Returns the best sequence seen during the run.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    tau = tau0

    while time.time() < deadline:
        it += 1

        # convolution and soft‑max weights
        conv = _conv_self(a, L=L_fft)                # length 2n‑1
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()               # sum(w)=1

        # gradient via FFT
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # step + projection
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        # true objective
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (max‑conv formulation)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         step0: float = 0.40,
                         beta: float = 0.96) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    momentum = np.zeros_like(a)
    it = 0
    L_grad = _next_pow2(3 * a.shape[0] - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * a.shape[0] * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (a.shape[0] - 1))
        hi = min(a.shape[0] - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing polish
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int,
                max_move_frac: float = 0.15,
                trials_per_sec: int = 1000) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a small fraction of mass from a source
    index to a destination index; keep the move only if the true objective improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        for _ in range(trials_per_sec):
            it += 1
            src = rng.choice(n, p=a)                # pick proportional to mass
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps = a[src] * rng.random() * max_move_frac
            if eps <= 0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_score = _eval_fast(a_prop, L_fft)
            if new_score < best_score:
                a = a_prop
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb it={it:,} best={best_score:.6f}")
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.

    The routine mixes a fast parametric sweep, many soft‑max gradient
    descents (random restarts), a long projected sub‑gradient phase and a
    final hill‑climbing polish.  It respects the 1000 s budget and
    returns a list compatible with ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)       # for full convolution
    L_grad = _next_pow2(3 * n)          # for gradient convolutions

    # ------------------------------------------------------------------
    # 0️⃣  Baseline candidates
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # Optional global starter

        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)

            else:

            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score, best_seq = cur, starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Deterministic parametric families
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # geometric family
    for eps in np.logspace(-5, -1, 120):
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # power family
    for p in np.linspace(-1.5, 2.5, 81):
        if time.time() >= deadline - 30:
            break
        pow_seq = (idx + 1) ** p
        pow_seq /= pow_seq.sum()
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] power p={p:.2f} score={sc:.6f}")

    # beta‑type family (allow negative exponents)
    beta_params = [-1.0, -0.5, 0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (idx + 1) ** p * (n - idx) ** q
            if beta_seq.sum() == 0:
                continue
            beta_seq /= beta_seq.sum()
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score, best_seq = sc, beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # cosine‑V shape (high at the two ends, low in the centre)
    cos_shape = (1.0 + np.cos(2.0 * np.pi * idx / (n - 1))) / 2.0
    cos_shape /= cos_shape.sum()
    sc = _eval_fast(cos_shape, L_fft)
    if sc < best_score:
        best_score, best_seq = sc, cos_shape.copy()
        print(f"[{time.time() - start_time:.2f}s] cosine V‑shape score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent with many random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # start from the current best (with a little noise) or from uniform
        base = best_seq if np.random.rand() < 0.75 else uniform
        noise = 0.04 * np.random.randn(n) * base         # relative Gaussian noise
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        # allocate ≈20 s for each restart (or whatever is left)
        sub_deadline = min(deadline - 5.0, time.time() + 20.0)

        cand_seq, cand_score = _smooth_descent(
            start, sub_deadline, start_time, L_fft, L_grad,
            step0=0.35, tau0=0.20, tau_decay=0.9996)

        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] new global best after smooth restart score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Long projected sub‑gradient refinement (bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(
            best_seq, sub_deadline, start_time, L_fft,
            step0=0.40, beta=0.96)
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad final improve score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final deterministic hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft,
                                   max_move_frac=0.15, trials_per_sec=1000)
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improve score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return the result
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
