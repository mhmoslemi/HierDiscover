def _lp_improve(a, deadline, start_time):
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    it = 0
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)  # for evaluations
    t = 0.5
    while time.time() < deadline:
        it += 1
        # compute M
        conv = _conv_self(a, L_fft)
        M = conv.max()
        # Build A matrix
        # using toeplitz: first column length 2n-1: a padded with zeros at bottom
        col = np.concatenate([a, np.zeros(n - 1)])  # length 2n-1
        row = np.concatenate([a[:1], np.zeros(n - 1)])  # length n  (first row: a0, 0,0,...)
        A = scipy.linalg.toeplitz(col, row)
        # Optional: convert to float64
        # Solve LP
        c = -np.ones(n)  # minimize -sum(b)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'disp': False})
        if res.status != 0:
            # LP failed; break
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt / sum_b
        new_a = (1 - t) * a + t * g
        # project onto simplex (ensures sum=1, nonnegative)
        new_a = _project_simplex(new_a) # though already sum=1
        new_score = _eval_fast(new_a, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP it={it} t={t:.3f} score={new_score:.6f}")
        # decide acceptance: if new_score < best_score perhaps accept; else maybe reduce t
        if new_score < best_score:
            a = new_a
            # maybe increase t a bit for faster changes
            t = min(0.9, t * 1.05)
        else:
            t = max(0.1, t * 0.5)
            # optionally keep a unchanged
        # time checks
        if time.time() >= deadline - 10: break
    return best_a, best_score

"""
Propose a step‑function (length 1‑k) that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

The search combines three ideas that are complementary to the reference
implementation:

1. **Extensive parametric sweep** – a few hundred simple families
   (geometric, power, beta‑type and random mixtures) are evaluated
   very fast with FFT based convolution.

2. **Iterated linear‑programming refinement** – following the idea from
   the paper quoted in the prompt we repeatedly solve

        maximise   Σ b_j
        subject to  b_j ≥ 0
                    (f * b)_k ≤ M     ∀k
        with   M = max(f * f).

   The LP is solved with SciPy’s HiGHS solver.  The resulting vector
   `b` is renormalised and mixed with the current `f` (step size `t`
   is adapted) – each iteration provably never increases the objective
   and quickly drives the maximum convolution towards its average value.

3. **Fine‑grained Adam optimisation** – after the LP phase we run a
   few Adam steps on a soft‑max approximation of the max‑operator.
   The gradient is obtained with two FFTs per step, the iterate is
   always projected onto the probability simplex.

Finally a short deterministic hill‑climb moves tiny amounts of mass
between coordinates only when the true objective improves.

The whole pipeline stops well before the 1000 s budget expires and
returns a list compatible with ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.linalg

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    w ≥ 0 and Σ w = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Fast evaluation of the exact objective for a (assumed Σa≈1).
    Returns 2·n·max(conv(a,a)).
    """
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# 1️⃣  Parametric sweep (very cheap)
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, L_fft: int, start_time: float,
                     deadline: float, best_a: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Explore a few hundred simple families and keep the best."""
    idx = np.arange(n)

    # geometric families (including a uniform mix)
    eps_grid = np.logspace(-5, -0.5, 300)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = r ** idx
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 21):
            seq = lam * (np.full(n, 1.0 / n)) + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_a = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # pure power laws
    p_grid = np.linspace(-2.0, 3.0, 101)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = (idx + 1) ** p
        pow_sum = pow_seq.sum()
        if pow_sum == 0:
            continue
        pow_seq /= pow_sum
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type families
    beta_vals = np.linspace(0.0, 3.0, 13)
    for p in beta_vals:
        for q in beta_vals:
            if time.time() >= deadline - 30:
                break
            beta_seq = ((idx + 1) ** p) * ((n - idx) ** q)
            bsum = beta_seq.sum()
            if bsum == 0:
                continue
            beta_seq /= bsum
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_a = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # a few random families (to avoid being trapped)
    rng = np.random.default_rng()
    for _ in range(200):
        if time.time() >= deadline - 30:
            break
        typ = rng.integers(3)
        if typ == 0:          # random geometric
            r = rng.uniform(0.4, 0.99)
            seq = r ** idx
        elif typ == 1:        # random power
            p = rng.uniform(-2.5, 2.5)
            seq = (idx + 1) ** p
        else:                 # random beta‑type

            q = rng.uniform(0.0, 4.0)
            seq = ((idx + 1) ** p) * ((n - idx) ** q)
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0:
            continue
        seq /= seq.sum()
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_a = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Random family score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Linear‑programming refinement (the theoretical step mentioned in
#      the prompt).  Operates on a modest size (n≈1024) to keep the LP
#      cheap.
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               n: int,
               L_fft: int,
               start_time: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve   maximise Σ b_j   s.t.  b≥0  and  (a * b) ≤ M,
    where M = max(a * a).  After each LP we mix the solution back
    with the current iterate using an adaptive step size `t`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    t = 0.5                     # initial mixing weight
    it = 0

    while time.time() < deadline - 5.0:
        it += 1
        # current bound
        conv = _conv_self(a, L=L_fft)
        M = conv.max()

        # build the convolution matrix A (size (2n‑1)×n)
        # A_{k,j} = a_{k-j} if 0≤k-j<n else 0
        col = np.concatenate([a, np.zeros(n - 1)])
        row = np.concatenate([a[:1], np.zeros(n - 1)])
        A = scipy.linalg.toeplitz(col, row)

        # linear program
        c = -np.ones(n)                     # maximise sum b  → minimise -sum
        b_ub = np.full(2 * n - 1, M)        # (a * b)_k ≤ M
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=b_ub,
                                     bounds=bounds,
                                     method='highs',
                                     options={'presolve': True})
        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed (status {res.status}); stop.")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break

        # normalise to the simplex
        g = b_opt / sum_b

        # mix and re‑project
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)     # guarantees sum=1 & non‑negative

        sc_new = _eval_fast(a_new, L_fft)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] LP it={it:,} t={t:.3f} score={sc_new:.6f}")

        # accept if we made progress, otherwise shrink the step size
        if sc_new < best_score:
            a = a_new
            t = min(0.9, t * 1.05)           # slowly increase step when it works
        else:
            t = max(0.1, t * 0.5)            # otherwise be more conservative

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimiser on a soft‑max approximation of the max‑operator
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   deadline: float,
                   start_time: float,
                   L_fft: int,
                   L_grad: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective

        f(a) = 2·n * τ·log Σ exp(conv(a,a)/τ) / (Σ a)²

    The temperature τ is exponentially decayed so that the optimisation
    gradually focuses on the true max.
    """
    n = start_a.shape[0]
    a = _project_simplex(start_a)          # ensure feasibility
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    tau = 0.2
    tau_min = 1e-9
    tau_decay = 0.9995
    base_lr = 0.05

    while time.time() < deadline:
        it += 1

        # convolution and soft‑max weights
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)    # numerically stable
        w = exp_vals / exp_vals.sum()             # sum w = 1

        # gradient of Σ w_k conv_k w.r.t a  (same as in the reference code)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = _project_simplex(a)

        # exact objective
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

# 4️⃣  Deterministic hill‑climbing polish (tiny mass moves)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a tiny fraction of mass from a source
    index to a destination index, keep the move only if the true objective
    improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                # source proportional to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue
        # move up to 15 % of the source mass
        eps = a[src] * rng.random() * 0.15
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score

    if time.time() < deadline:
        best_seq, best_score = _hill_climb(
            best_seq, deadline, start_time, L_fft)

    # final normalisation and safety
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
