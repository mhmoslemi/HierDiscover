def generate_beta(n, alpha):
    i = np.arange(n)
    x = (i + 0.5) / n
    a = np.power(x, alpha-1) * np.power(1-x, alpha-1)
    # For alpha < 1 we may get infinities at edges due to zero^negative -> inf.

    a = np.power(x, alpha-1) * np.power(1-x, alpha-1)
    # Normalise to sum=1.
    a /= a.sum()
    return a

def _beta_sequence(n, alpha):
    i = np.arange(n)
    x = (i + 0.5) / n
    a = np.power(x, alpha - 1) * np.power(1 - x, alpha - 1)
    # Replace infinities and NaNs
    a[np.isinf(a)] = 0.0
    a[np.isnan(a)] = 0.0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0/n)
    else:
        a /= total
    return a

"""
Optimisation of a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

(the same quantity returned by ``evaluate_sequence`` after normalisation).

The optimiser works in five stages:

1.  A broad collection of parametric seeds (geometric, Beta(α,α),
    symmetric “U‑shape” ( (i+½)(n‑½‑i) )⁻ᵖ , and a sine shape) is generated.
    The best seed (smallest convolution peak) is kept.

2.  Adam on a smooth approximation of the max‑convolution
    (log‑sum‑exp) reduces the peak while keeping the vector on the probability
    simplex.

3.  Exponentiated‑gradient (mirror descent) directly minimises the true
    max‑convolution.

4.  Targeted simulated‑annealing with a long cooling schedule performs
    stochastic jumps that respect the simplex constraint.

5.  An aggressive greedy sweep moves a tiny amount of mass from the most
    responsible coordinate to the least responsible one, repeatedly lowering
    the peak.  A second greedy pass is performed near the deadline.

All phases respect the 1000 s budget and print progress messages.
The final (normalised) sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Full self‑convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Parametric seed generators
# ----------------------------------------------------------------------
def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i ∝ r**i (normalised)."""
    r = np.clip(r, 1e-12, 0.999999)
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise (finite sum = (1‑r**n))
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on n points.
    Uses mid‑point positions to avoid exact zeros at the borders.
    """
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n
    # avoid division by zero for very small α
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _u_shape_sequence(n: int, p: float) -> np.ndarray:
    """
    Symmetric U‑shape a_i ∝ ((i+½)*(n‑½‑i))⁻ᵖ .
    p=0 gives the uniform distribution.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-p)
    a = np.maximum(w, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _sine_sequence(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n)  (normalised)."""
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n
    a = np.sin(np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Simple scoring helper (identical to evaluate_sequence for normalised vectors)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a that already sums to 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smoothed approximation of the max‑convolution
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.6,
                        iters_per_alpha: int = 25000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    best_score = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-10
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            conv = _conv_self(a, L)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max via log‑sum‑exp -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a)
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # mirror‑descent update on the simplex
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise to simplex
        a_sum = a.sum()
        if a_sum <= 0.0 or np.isnan(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Targeted simulated annealing
# ----------------------------------------------------------------------
def _run_targeted_annealing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            cur_best_score: float,
                            cur_best_seq: np.ndarray,
                            start_temp: float = 5e-2,
                            end_temp: float = 1e-9,
                            steps: int = 500_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    score = 2.0 * n * conv.max()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # index of the current peak
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1)
        contrib = a[idx_range] * a[k - idx_range]
        if contrib.sum() > 0.0:
            src = rng.choice(idx_range, p=contrib / contrib.sum())
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop, L)
        new_score = 2.0 * n * conv_new.max()
        delta = new_score - score

        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            conv = conv_new
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal improve: "
                  f"iter={it:,}, score={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 4 – Aggressive greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually contribute to this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # vector a_k = a[max_idx - i] on the support of the max
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # choose a destination with the smallest a_k value
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partners for src and dst in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # adaptive step size
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # incremental update of the convolution (ignoring O(eps²) terms)
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # periodic exact recomputation (drift removal)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin, otherwise the caller may kill us exactly at the deadline
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size (kept equal to the previous best for fair comparison)
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 0. Initialise a pool of parametric candidates and pick the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # exponential decay (grid search, coarse)
    for r in np.linspace(0.90, 0.9999, 150):
        cand = _exponential_sequence(n, r)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            best_r = r
    print(f"[0.00s] exponential seed r={best_r:.6f}, score={best_score:.6f}")

    # Beta family
    for alpha in [0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70, 0.80, 1.00, 1.20]:
        cand = _beta_sequence(n, alpha)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"[0.00s] beta α={alpha:.2f}, score={sc:.6f}")

    # U‑shape family
    for p in [0.15, 0.20, 0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.60]:
        cand = _u_shape_sequence(n, p)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"[0.00s] U‑shape p={p:.2f}, score={sc:.6f}")

    # sine shape
    cand = _sine_sequence(n)
    sc = _score(cand)
    if sc < best_score:
        best_score, best_seq = sc, cand
        print(f"[0.00s] sine shape, score={sc:.6f}")

    # try to reuse a previously discovered construction, if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = starter / starter.sum()
            a0 = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, starter.size),
                           starter)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            sc = _score(a0)
            if sc < best_score:
                best_score, best_seq = sc, a0
                print(f"[0.00s] reused previous construction, score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phases 1–4 (identical structure to the previous optimiser but
    # with a few tuned hyper‑parameters). Each phase respects the remaining
    # time budget.
    # ------------------------------------------------------------------
    # Phase 1 – Adam smoothing
    if time.time() < deadline:
        phase1_deadline = time.time() + min(30.0, 0.12 * (deadline - time.time()))
        a_adam, sc_adam = _run_adam_smoothing(
            best_seq,
            n,
            phase1_deadline,
            start_time,
            alphas=[0.1, 0.3, 1.0, 3.0, 10.0, 30.0, 80.0],
            lr0=0.6,
            iters_per_alpha=20000,
        )
        if sc_adam < best_score:
            best_score, best_seq = sc_adam, a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 1 improve: "
                  f"score={sc_adam:.6f}")

    # Phase 2 – Exponentiated gradient descent
    if time.time() < deadline:
        phase2_deadline = time.time() + min(40.0, 0.30 * (deadline - time.time()))
        a_exp, sc_exp = _run_exp_gradient(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            eta0=1.0,
        )
        if sc_exp < best_score:
            best_score, best_seq = sc_exp, a_exp.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={sc_exp:.6f}")

    # Phase 3 – Targeted annealing
    if time.time() < deadline:
        phase3_deadline = time.time() + min(250.0, 0.35 * (deadline - time.time()))
        a_ann, sc_ann = _run_targeted_annealing(
            best_seq,
            n,
            phase3_deadline,
            start_time,
            best_score,
            best_seq,
            start_temp=2e-2,
            end_temp=1e-9,
            steps=500_000,
        )
        if sc_ann < best_score:
            best_score, best_seq = sc_ann, a_ann.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: "
                  f"score={sc_ann:.6f}")

    # Phase 4 – Aggressive greedy reduction
    if time.time() < deadline:
        phase4_deadline = deadline
        a_greedy, sc_greedy = _run_greedy_reduction(
            best_seq,
            n,
            phase4_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=5_000_000,
        )
        if sc_greedy < best_score:
            best_score, best_seq = sc_greedy, a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 4 improve: "
                  f"score={sc_greedy:.6f}")

    # ------------------------------------------------------------------
    # Optional extra greedy sweep if a few seconds remain
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        a_extra, sc_extra = _run_greedy_reduction(
            best_seq,
            n,
            deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=1_000_000,
        )
        if sc_extra < best_score:
            best_score, best_seq = sc_extra, a_extra.copy()
            print(f"[{time.time() - start_time:.2f}s] extra greedy improve: "
                  f"score={sc_extra:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_obj = _score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
