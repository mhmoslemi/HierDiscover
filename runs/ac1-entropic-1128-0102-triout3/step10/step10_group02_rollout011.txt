def _adam_smooth_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    L_self = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n-2)
    it=0
    alphas = np.geomspace(0.5, 500.0, num=40)  # increasing
    for alpha in alphas:
        while time.time() < deadline - 0.2:
            it+=1
            # compute conv
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]
            # compute softmax weights w
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)

"""
Propose a non‑negative step‑function (list of heights) that minimises
    2 * n * max(conv(a,a)) / (sum(a)**2)

The method combines three ideas that were not all used together before:

1️⃣  **Adam‑based smooth‑max mirror descent** – a smoothed approximation of the
    max‑convolution (soft‑max) is minimised on the simplex using the Adam
    optimiser.  The smoothing parameter α is increased geometrically, ending
    at a very large value (≈ 500) so that the true max is approached.

2️⃣  **Projected sub‑gradient polish** – a short momentum sub‑gradient phase
    (the same as the reference) cleans up any remaining irregularities.

3️⃣  **Greedy mass‑transfer local search** – after sub‑gradient polishing we
    repeatedly pick the pair of indices that contributes most to the current
    worst convolution value and transfer a small amount of mass to a random
    location.  This directly attacks the bottleneck terms and often yields a
    few × 10⁻⁴ improvement.

All phases respect a global wall‑clock (budget_s = 1000 s).  The final
sequence is normalised to sum = 1 (the scaling does not affect the objective)
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution, not a circular one.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (exact when sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑based smooth‑max mirror descent
# ----------------------------------------------------------------------
def _adam_smooth_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of max(conv(a,a)) using the Adam optimiser.
    The smoothing parameter α is increased geometrically.
    """
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # geometric schedule for the smoothing parameter (up to a large value)
    alphas = np.geomspace(0.5, 500.0, num=40)

    it = 0
    max_iters = 800_000          # safety cap

    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stabilisation
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient of the smooth max (2·(w⋆a_rev))
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update on the simplex (mirrored by exponential map)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = 0.05 * m_hat / (np.sqrt(v_hat) + eps)   # learning‑rate ≈ 0.05
            a = a - step
            a = _project_simplex(a)

            # occasional exact‑objective check
            if it % 750 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Momentum sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy mass‑transfer local search (new polish)
# ----------------------------------------------------------------------
def _mass_transfer(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  cur_best_score: float,
                  cur_best_seq: np.ndarray,
                  steps: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small amount of mass away from the pair that
    contributes most to the current worst convolution value.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    L_self = _next_pow2(2 * n - 1)

    for it in range(steps):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_self)
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # product a_i * a_{k-i}
        products = a[idx] * a[k - idx]
        i = idx[np.argmax(products)]
        j = k - i

        # choose a destination different from i and j
        dst = rng.integers(n)
        while dst == i or dst == j:
            dst = rng.integers(n)

        # transfer a modest fraction of the smaller of the two contributors
        eps = min(a[i], a[j]) * 0.05
        if eps < 1e-12:
            continue

        a[i] -= eps
        a[j] -= eps
        a[dst] += 2 * eps
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # evaluate
        cur_score = 2.0 * n * _conv_self(a, L_self).max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Mass‑transfer it={it:,} best={best_score:.6f}")

    return best_seq, best_score

# Light simulated‑annealing polish (identical to reference)

def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    L_self = _next_pow2(2 * n - 1)
    score = 2.0 * n * _conv_self(a, L_self).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop, L_self).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score
