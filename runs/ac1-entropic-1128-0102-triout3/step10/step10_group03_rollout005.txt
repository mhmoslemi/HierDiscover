"""
Propose a step function (non‑negative list) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm follows three stages:

1️⃣  A fast parametric sweep (geometric, power, beta families) to obtain a
    decent starting point (or the global `height_sequence_1` if provided).

2️⃣  A long projected sub‑gradient descent on the simplex.  The update uses
    a momentum term and a 1/√t step‑size; when progress stalls a tiny
    random perturbation is added to escape shallow local minima.  The loop
    runs for most of the 1000 s budget, constantly tracking the best
    sequence seen.

3️⃣  A final micro‑hill‑climb that moves very small amounts of mass between
    randomly chosen indices (ε≈10⁻⁶·a_src).  Only improving moves are kept.

All heavy work is performed with FFT‑based convolutions (O(n log n))
and the Euclidean projection onto the probability simplex.
The routine finishes well before the time limit and returns a list of
Python floats suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (long run)
# ----------------------------------------------------------------------
def _subgradient_descent_long(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Heavy‑weight sub‑gradient descent on
        f(a)=2·n·max(conv(a,a))   (sum a is projected to 1).

    Momentum (β≈0.9) and a step size α_t = α₀/√t are used.
    If no improvement occurs for a while a tiny random perturbation
    is injected to try to escape plateaus.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    momentum = np.zeros_like(a)
    beta = 0.9
    alpha0 = 0.8               # initial step magnitude
    it = 0
    last_improve = 0
    L_fft = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        it += 1
        # convolution and current score
        conv = _conv_self(a, L=L_fft)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # ---- sub‑gradient of max‑conv ----
        k = int(conv.argmax())                # active shift
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        # grad_i = 2 * a_{k-i}  (if inside the range)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = alpha0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

        # ---- escape plateau if needed ----
        if it - last_improve > 5000:
            # add a very small Gaussian perturbation
            noise = 1e-4 * rng.standard_normal(n) * a
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny hill‑climb (final polish)
# ----------------------------------------------------------------------
def _tiny_hill_climb(start_a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very small mass transfers (ε≈10⁻⁶·a_src) that are accepted only if
    the true objective improves.  This fine‑tunes the solution after the
    heavy sub‑gradient phase.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 1e-4   # extremely tiny move
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny hill‑climb score={best_score:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    evaluation function.  The search consists of:

        * a quick parametric sweep (geometric / power / beta families),
        * a long projected sub‑gradient descent on the simplex,

    # ------------------------------------------------------------------
