    best_alpha = None; best_lambda = None

    eps = 1e-30
    w_cache = {}
    for alpha in alphas:
        w = (x*(1-x) + eps)**(-alpha)
        w = w / w.sum()
        w_cache[alpha] = w
        score = _objective(w)
        if score < best_score:
            best_score = score; best_seq = w.copy()
            best_alpha = alpha; best_lambda = 1.0
    # mixture search
    lambdas = np.linspace(0.0, 1.0, 21)  # step .05
    for alpha, w in w_cache.items():
        for lam in lambdas:
            a = lam * w + (1-lam) * U
            # a already sums to 1
            score = _objective(a)
            if score < best_score:
                best_score = score; best_seq = a.copy()
                best_alpha = alpha; best_lambda = lam

    # local refinement around current best_{alpha,lambda}
    if best_alpha is not None:
        fine_alphas = np.linspace(max(0.2, best_alpha-0.01), min(1.2, best_alpha+0.01), 21)
        fine_lams = np.linspace(max(0.0, best_lambda-0.1), min(1.0, best_lambda+0.1), 21)
        for alpha in fine_alphas:
            w = (x*(1-x) + eps)**(-alpha)
            w = w / w.sum()
            for lam in fine_lams:
                a = lam * w + (1-lam) * U
                score = _objective(a)
                if score < best_score:
                    best_score = score; best_seq = a.copy()
                    best_alpha = alpha; best_lambda = lam

    # final local refinement using subgradient if time left
    remaining = deadline - time.time()
    if remaining > 5.0:
        # allocate e.g., 80% of remaining to subgradient
        sub_deadline = time.time() + 0.8 * remaining
        refined_seq, refined_score = _subgradient_descent(best_seq, sub_deadline, start_time)
        if refined_score < best_score:
            best_score = refined_score; best_seq = refined_seq.copy()

    # very light annealing (few thousand moves)
    if time.time() < deadline:
        # similar to earlier but trimmed steps
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_steps = 10000
        for it in range(total_steps):
            if time.time() >= deadline:
                break
            T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)  # same cooling
            src = rng.choice(n, p=a)

            eps_move = a[src] * rng.random() * 0.2

            best_seq = best_local.copy()
   
    # final normalisation (just in case)

    total = best_seq.sum()
    if total > 0:

    else:
        best_seq = np.full(n, 1.0/n)

    print(f"Final best score: {best_score:.6f}")
    # convert to list
    return [float(v) for v in best_seq]

"""
Propose a step‑function with a very low evaluation score.

Algorithm
---------
1. **Fine sweep of the Beta‑family** –  We restrict ourselves to
   symmetric Beta‑densities a_i ∝ (x_i·(1−x_i))^{−α} (continuous analogue
   of the arcsine law).  A dense grid for α∈[0.30, 1.00] is evaluated and
   the best α is stored.  Afterwards a tiny refinement around the best
   α (step 5·10⁻⁴) is performed – this alone already beats the previous
   best bound.

2. **Local Adam‑style sub‑gradient descent** –  Starting from the best
   candidate we run a projected Adam optimiser on the exact
   non‑smooth objective (the sub‑gradient of the active convolution entry
   is used).  The optimiser works on the probability simplex and stops
   when the wall‑time budget is exhausted.

3. **Very light simulated‑annealing polish** –  A few thousand tiny
   mass‑transfer moves are tried to squeeze out any remaining improvement.

All steps respect the 1000 s wall‑time limit; the code works without
any I/O and returns a plain list of non‑negative floats.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT‑based self‑convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the challenge (sum of seq is forced to 1).
    Returns 2·n·max(conv(seq,seq)) .
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    # seq is already normalised (sum == 1); the denominator is 1.
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser using the sub‑gradient of the maximal
    convolution entry.  Returns the best iterate seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5                # initial learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # denominator is 1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # print progress only occasionally to keep the loop fast
            if t % 50000 == 0:
                print(f"[{time.time() - start_time:.1f}s] Adam iter={t:,} "
                      f"score={cur_score:.6f}")

        # index attaining the maximal convolution entry
        k = int(conv.argmax())

        # sub‑gradient of the max‑entry w.r.t. a (see reference implementation)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        # gradient = 2 * a shifted by k (zero outside the support)
        grad[idx] = 2.0 * a[k - idx]

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (keeps sum = 1 and non‑negativity)
        a = _project_simplex(a)

    return best_a, best_score
