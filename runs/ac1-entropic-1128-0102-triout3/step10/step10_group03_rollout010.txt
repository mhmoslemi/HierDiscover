def _permute_hill_climb(a0, deadline, start_time, L_fft):

    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    n = len(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        # compute score
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Permute hill it={it} best={best_score:.6f}")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]
    return best_a, best_score

"""
Propose a non‑negative step function of length 4096 that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search combines:

1️⃣  Very dense parametric sweep (geometric, power, beta, and a 2‑exponential mix)
    with a finer ε‑grid extending to 1e‑6.

2️⃣  Smoothed gradient descent (soft‑max approximation of the max) with an
    Adam‑style momentum and a slower temperature decay.  Several short
    random‑restart runs are performed.

3️⃣  A permutation hill‑climb that swaps two entries (preserving the multiset)
    and keeps the swap only if the true objective improves.

4️⃣  A projected sub‑gradient refinement (as in the reference solution).

5️⃣  A final tiny‑mass hill‑climb polish.

All operations stay on the probability simplex, therefore the objective
is scale‑invariant.  The routine respects the 1000 s budget and prints
progress reports.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum≈1) used during the search
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max ≈ max) – Adam style momentum
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·softmax_τ(conv(a,a)) on the simplex.
    Adam‑style moment estimates give a more stable descent.
    Returns the best iterate seen.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    it = 0
    tau = 0.12
    tau_min = 5e-9
    tau_decay = 0.9995
    step0 = 0.25

    while time.time() < deadline:
        it += 1

        # convolution
        conv = _conv_self(a, L=L_fft)          # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # sum(w)=1

        # gradient = 2·A·w_rev   (see reference implementation)
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)          # base step size decays like 1/√t
        a = a - step * m_hat / (np.sqrt(v_hat) + eps_adam)

        # back to simplex
        a = _project_simplex(a)

        # exact objective (fast version)
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (identical to the reference implementation)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (the first maximal index)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (tiny mass moves)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a tiny fraction of mass from a source
    index to a destination index, keep the move only if the true objective
    improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                # pick proportional to mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Permutation hill‑climb (swap two entries)
# ----------------------------------------------------------------------
def _permute_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        L_fft: int) -> tuple[np.ndarray, float]:
    """
    Swap two entries of the vector while keeping the simplex constraint.
    Keep the swap only if the true objective improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]          # swap
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Perm‑swap it={it:,} best={best_score:.6f}")
        else:
            # revert
            a[i], a[j] = a[j], a[i]

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm mixes an extensive parametric sweep, Adam‑style
    smoothed gradient descent with random restarts, a permutation‑swap
    hill‑climb, a sub‑gradient refinement and a final tiny‑mass polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)                # for self‑convolution
    L_grad = _next_pow2(3 * n)                  # for gradient convolution

    # ------------------------------------------------------------------
    # Baseline: uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional starter taken from the global variable `height_sequence_1`
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                sc = _eval_fast(starter, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Dense parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # ----- geometric families (ε down to 1e‑6) -------------------------
    eps_grid = np.logspace(-6, -1, 300)         # finer grid, more points
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        geo_sum = geo.sum()
        if geo_sum == 0.0:
            continue
        geo /= geo_sum
        # plain geometric
        sc = _eval_fast(geo, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = geo.copy()
            print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} pure score={sc:.6f}")

        # convex combination with uniform
        for lam in np.linspace(0.0, 1.0, 13):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ----- power families ------------------------------------------------
    p_grid = np.linspace(-2.0, 2.5, 81)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        pow_sum = pow_seq.sum()
        if pow_sum == 0.0:
            continue
        pow_seq /= pow_sum
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # ----- beta‑type families --------------------------------------------
    beta_params = np.linspace(0.0, 3.0, 13)
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            beta_sum = beta_seq.sum()
            if beta_sum == 0.0:
                continue
            beta_seq /= beta_sum
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ----- double‑exponential mixtures ------------------------------------
    eps1_grid = np.logspace(-6, -2, 30)
    eps2_grid = np.logspace(-6, -2, 30)
    for eps1 in eps1_grid:
        for eps2 in eps2_grid:
            if time.time() >= deadline - 30:
                break
            r1 = 1.0 - eps1
            r2 = 1.0 - eps2
            geo1 = np.power(r1, idx, dtype=np.float64)
            geo2 = np.power(r2, idx, dtype=np.float64)
            mix = 0.5 * geo1 + 0.5 * geo2
            mix_sum = mix.sum()
            if mix_sum == 0.0:
                continue
            mix /= mix_sum
            sc = _eval_fast(mix, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = mix.copy()
                print(f"[{time.time() - start_time:.2f}s] Double‑exp eps1={eps1:.1e} eps2={eps2:.1e} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent with random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # pick a base (mostly current best, sometimes uniform)
        base = best_seq if np.random.rand() < 0.75 else uniform
        # add small relative Gaussian noise, keep on simplex
        noise = 0.03 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 20.0)   # ≤20 s per restart
        cand_seq, cand_score = _smooth_descent(start, sub_deadline,
                                               start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after smooth restart score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Permutation hill‑climb (swap two entries)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        perm_deadline = deadline - 10.0
        a_perm, sc_perm = _permute_hill_climb(best_seq, perm_deadline,
                                              start_time, L_fft)
        if sc_perm < best_score:
            best_score = sc_perm
            best_seq = a_perm.copy()
            print(f"[{time.time() - start_time:.2f}s] Permutation hill‑climb improved score={sc_perm:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient refinement (bulk of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final tiny‑mass hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Final tiny‑mass hill‑climb improve score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Normalise to a proper probability vector and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
