"""
Search for a low‑autocorrelation step function.

The algorithm proceeds in four stages:

1️⃣  A quick parametric sweep over several families (geometric,
    power, U‑shaped, Hann, etc.) to obtain a decent starting point.
2️⃣  Smoothed gradient descent (soft‑max approximation of the max) with
    random restarts.  The iterate stays on the probability simplex.
3️⃣  Projected sub‑gradient descent for a longer, more accurate refinement.
4️⃣  A simulated‑annealing polish that accepts occasional worsening moves
    and therefore can escape the local minima left by the previous steps.

All steps respect the non‑negative, sum‑to‑one normalization required by
``evaluate_sequence``.  The routine adapts its time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                      # numerical fallback
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is assumed ≈1 but we keep the exact formula)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation when a is already normalised (sum≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max)
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed objective:
        f(a) = 2 * n * softmax_τ(conv(a,a))

    The temperature τ decays exponentially; after each step the iterate is
    projected onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    tau = 0.10          # initial temperature
    tau_min = 1e-8
    tau_decay = 0.9997
    step0 = 0.25        # initial step length

    while time.time() < deadline:
        it += 1

        # convolution
        conv = _conv_self(a, L=L_fft)        # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()        # sum(w) = 1

        # gradient via FFT: grad_j = 2 * Σ_i a_i w_{i+j}
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # projected step
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        # exact score
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

# Projected sub‑gradient descent (as in the reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} score={cur_score:.6f}")

        # active index of the maximal convolution value
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap SA: move a small fraction of mass from a source index
    to a destination index; worsening moves are accepted with probability
    exp(-Δ/T) where T decays linearly to zero.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _score(a, n, L_fft)
    best_a = a.copy()
    best_score = cur_score

    # initial temperature (≈1 % of the current score)
    T0 = cur_score * 0.01
    Tmin = 1e-9

    rng = np.random.default_rng()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        T = max(Tmin, T0 * (1.0 - progress))

        # pick source (proportional to its mass) and a random destination
        src = rng.choice(n, p=a)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if dst == src:
            continue

        # move up to 10 % of the source mass
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score(a_prop, n, L_fft)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA score={new_score:.6f}")

    return best_a, best_score

        pw /= pw.sum()
        starters.append((f"pow_{p:.2f}", pw))

    # U‑shaped (mass concentrated at the ends)
    for p in np.linspace(-2.0, -0.2, 20):
        ush = ((idx + 1) * (n - idx)) ** p
        ush /= ush.sum()
        starters.append((f"ushape_{p:.2f}", ush))

    # Hann window (cosine‑bell)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    hann /= hann.sum()
    starters.append(("hann", hann))

    # Half‑sine window
    half_sin = np.sin(np.pi * (idx + 0.5) / n)
    half_sin /= half_sin.sum()
    starters.append(("half_sin", half_sin))

    best_seq = None
    best_score = float("inf")
    for name, seq in starters:
        sc = _eval_fast(seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] init {name} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent with random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 200.0:
        # pick a base (usually the current best)
        base = best_seq if np.random.rand() < 0.7 else uniform
        noise = 0.05 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(time.time() + 12.0, deadline - 50.0)
        cand_seq, cand_score = _smooth_descent(start, sub_deadline,
                                               start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] new best after smooth {cand_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient refinement (bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sub_deadline = deadline - 100.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad improved score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing polish (uses whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sa_deadline = deadline - 2.0
        a_sa, sc_sa = _simulated_annealing(best_seq, sa_deadline,
                                           start_time, L_fft)
        if sc_sa < best_score:
            best_score = sc_sa
            best_seq = a_sa.copy()
            print(f"[{time.time() - start_time:.2f}s] SA final improve score={sc_sa:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
