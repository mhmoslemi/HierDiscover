def _lp_refine(a, deadline, start_time, max_iter=10, t=0.3):
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(a, _next_pow2(2*n-1))
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _conv_self(best_a, L=_next_pow2(2*n-1))
        M = conv.max()
        # active set: indices where conv >= M * (1 - tol)
        tol = 0.02  # 2% tolerance
        active_idx = np.where(conv >= M * (1 - tol))[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(conv.argmax())])
        # Build A_ub
        K = len(active_idx)
        A_ub = np.zeros((K, n))
        for row, k in enumerate(active_idx):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # columns j from lo..hi such that i = k - j
            j = np.arange(lo, hi + 1)
            i = k - j
            A_ub[row, j] = a[i]
        b_ub = np.full(K, M)
        # Solve LP maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"LP iteration {it}: failed {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # mix
        a_new = (1 - t) * best_a + t * g_norm
        a_new = _project_simplex(a_new)
        score_new = _eval_fast(a_new, _next_pow2(2*n-1))
        if score_new < best_score:
            best_score = score_new
            best_a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP refine it={it} score={score_new:.6f}")
        else:
            # if no improvement, reduce t
            t *= 0.5
            if t < 1e-3:
                break
    return best_a, best_score

def _guided_displacement(a, deadline, start_time):
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(a, _next_pow2(2*a.shape[0]-1))
    while time.time() < deadline:
        conv = _conv_self(best_a, L=_next_pow2(2*len(best_a)-1))
        k = int(conv.argmax())
        lo = max(0, k - (len(best_a) - 1))
        hi = min(len(best_a) - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(best_a)
        grad[idx] = 2.0 * best_a[k - idx]   # subgradient of max
        # find src with largest grad, dst with smallest
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if best_a[src] < 1e-12:
            # can't move from src, break
            break
        eps = best_a[src] * 0.2  # move 20% of source mass
        a_new = best_a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, _next_pow2(2*len(a_new)-1))
        if new_score < best_score:
            best_score = new_score
            best_a = a_new
            print(f"[{time.time() - start_time:.2f}s] Guided move src={src} dst={dst} score={new_score:.6f}")
        else:
            # maybe try smaller eps
            eps *= 0.5
            if eps < 1e-12:
                break
    return best_a, best_score

def _build_gaussian_sequence(params: np.ndarray, n: int) -> np.ndarray:
    seq = np.zeros(n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    K = len(params) // 3
    for i in range(K):

"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a blend of:
* a quick parametric sweep (geometric, power, beta families),
* a long soft‑max gradient descent with Adam‑style updates,
* a low‑dimensional Gaussian‑mixture optimisation via differential evolution,
* an active‑set linear‑program refinement,
* a greedy coordinate move based on the sub‑gradient, and finally
* a tiny random hill‑climb.

The routine respects the 1000 s budget, prints progress, and finally
returns a list compatible with ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional imports – if unavailable the corresponding phase is skipped
try:
    from scipy.optimize import linprog, differential_evolution
    _has_scipy = True
except Exception:                     # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum ≈ 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑style smoothed gradient descent (soft‑max approximation of max)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """Adam optimizer on the soft‑max smoothed objective."""
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    tau = 0.05                # initial temperature for soft‑max
    tau_min = 1e-8
    tau_decay = 0.9995        # slower decay → smoother gradients for longer
    step0 = 0.15

    while time.time() < deadline:
        it += 1
        # convolution
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()
        # soft‑max weights (numerically stable)
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()

        # gradient via FFT: grad_j = 2 * Σ_i a_i w_{i+j}
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update & projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # exact score
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Low‑dimensional Gaussian‑mixture optimiser (DE)
# ----------------------------------------------------------------------
def _gaussian_sequence(params: np.ndarray, n: int) -> np.ndarray:
    """Build a non‑negative sequence of length n from a mixture of Gaussians."""
    seq = np.zeros(n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    K = len(params) // 3
    for i in range(K):
        amp = max(0.0, params[3 * i])          # non‑negative amplitude
        cen = params[3 * i + 1]
        wid = max(1e-3, params[3 * i + 2])    # avoid zero width
        seq += amp * np.exp(- ((idx - cen) ** 2) / (2.0 * wid * wid))
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)
    return seq


def _de_optimise(n: int,
                 L_fft: int,
                 deadline: float,
                 start_time: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """Run differential evolution on a Gaussian‑mixture parameterisation."""
    if not _has_scipy:                     # pragma: no cover
        return best_seq, best_score

    K = 5                                 # number of Gaussian components
    dim = 3 * K
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 10.0))        # amplitude
        bounds.append((0.0, n - 1))       # centre
        bounds.append((1.0, n / 2.0))     # width

    # objective for DE (fast evaluation)
    def de_obj(params):
        seq = _gaussian_sequence(params, n)
        return _eval_fast(seq, L_fft)

    # allocate at most ~250 s for DE; adjust maxiter accordingly
    remaining = deadline - time.time()
    maxiter = max(1, int(remaining // 10))  # roughly 10 s per generation
    maxiter = min(maxiter, 40)             # cap to keep runtime reasonable

    print(f"[{time.time() - start_time:.2f}s] Starting DE (maxiter={maxiter})")
    result = differential_evolution(de_obj,
                                    bounds,
                                    maxiter=maxiter,
                                    popsize=12,
                                    tol=1e-7,
                                    polishing=False,
                                    seed=0,
                                    workers=1,
                                    updating='deferred')
    seq_de = _gaussian_sequence(result.x, n)
    score_de = _eval_fast(seq_de, L_fft)
    if score_de < best_score:
        best_score = score_de
        best_seq = seq_de.copy()
        print(f"[{time.time() - start_time:.2f}s] DE improved score={score_de:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Active‑set linear‑program refinement (LP)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               deadline: float,
               start_time: float,
               L_fft: int,
               max_iters: int = 8,
               tol_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """Iteratively improve a by solving a small LP on the active constraints."""
    if not _has_scipy:                     # pragma: no cover
        return a0, _eval_fast(a0, L_fft)

    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, L=L_fft)
        M = conv.max()
        # active set: indices whose value is within tol_frac of the maximum
        active = np.where(conv >= M * (1.0 - tol_frac))[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])
        K = active.size

        # build dense A_ub (K x n)
        A_ub = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            j = np.arange(lo, hi + 1)
            i = k - j
            A_ub[row, j] = a[i]

        b_ub = np.full(K, M, dtype=np.float64)

        # maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})

        if not res.success:                     # pragma: no cover
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()                 # re‑normalise to sum‑1

        # convex combination (t = 0.5, gradually reduced if no improvement)
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = _project_simplex(a_new)
        new_score = _eval_fast(a_new, L_fft)
        if new_score < best_score:
            best_score = new_score
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LP refine it={it} score={new_score:.6f}")
        else:
            # no progress – stop early
            break

    return a, best_score


# ----------------------------------------------------------------------
# Greedy sub‑gradient guided mass transfer
# ----------------------------------------------------------------------
def _guided_move(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 L_fft: int,
                 max_moves: int = 5000) -> tuple[np.ndarray, float]:
    """Move mass from the highest‑gradient index to the lowest‑gradient one."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        conv = _conv_self(a, L=L_fft)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of the max‑term

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * 0.15                # move up to 15 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] GuidedMove it={it:,} score={new_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (tiny random moves, accept only improvements)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """Very cheap local search: random tiny transfers accepted only if they improve."""
    n = a0.shape[0]
    a = a0.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)          # source proportional to current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The strategy mixes a fast parametric sweep, Adam‑style smoothed
    gradient descent, a Gaussian‑mixture DE search, an active‑set LP
    refinement, a greedy sub‑gradient move, and a final hill‑climb polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)                     # for gradient convolutions

    # ------------------------------------------------------------------
    # Baseline (uniform) ------------------------------------------------
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Start from a known good global variable if available
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:                               # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric families
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # geometric family
    eps_grid = np.logspace(-4, -1, 120)   # a bit coarser to save time
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        # pure geometric (lam=0) already in the loop; also try a few mixes
        for lam in (0.0, 0.2, 0.5, 0.8, 1.0):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.1f} score={sc:.6f}")

    # power family
    p_grid = np.linspace(-1.0, 2.0, 41)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # beta‑type family
    beta_params = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            s = beta_seq.sum()
            if s == 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑style smoothed gradient descent with random restarts
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # perturb current best a bit
        base = best_seq if np.random.rand() < 0.7 else uniform
        noise = 0.04 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 5.0, time.time() + 14.0)  # ≤14 s per run
        cand_seq, cand_score = _adam_smooth_descent(start, sub_deadline,
                                                    start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSmooth new global best score={best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Gaussian‑mixture DE search (low‑dimensional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0 and best_score > 1.503:
        best_seq, best_score = _de_optimise(n, L_fft, deadline,
                                            start_time, best_seq, best_score)

    # ------------------------------------------------------------------
    # 4️⃣  Active‑set linear‑program refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0 and best_score > 1.503:
        best_seq, best_score = _lp_refine(best_seq, deadline,
                                          start_time, L_fft)

    # ------------------------------------------------------------------
    # 5️⃣  Greedy guided mass transfer
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0 and best_score > 1.503:
        best_seq, best_score = _guided_move(best_seq, deadline,
                                            start_time, L_fft)

    # ------------------------------------------------------------------
    # 6️⃣  Deterministic hill‑climb polish (final tiny tweaks)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, deadline,
                                           start_time, L_fft)

    # ------------------------------------------------------------------
    # Normalisation and final output
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
