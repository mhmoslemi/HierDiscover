"""
A probabilistic search for a step‑function with a low ‑‑
objective 2·n·max(conv)/(∑a)².

Main ingredients
----------------
* Use a fairly large canvas (n≈16384).  The continuous optimum is 1.5,
  and the discretisation error drops quickly with n.
* Warm‑start from a small family of symmetric shapes:
  – tent^p   with p∈[0.2,1.5]
  – sin(πx)  (zero at the ends)
  – a semi‑circle (√(1‑(2x‑1)²)).
* Refine the best warm‑start with two complementary
  descent methods while always projecting onto the probability simplex:
  1) a classic sub‑gradient descent on the exact non‑smooth objective
     (fast, cheap per iteration);
  2) a smoothed gradient descent where the “max” is replaced by a
     soft‑max (temperature γ).  The gradient can be computed efficiently
     with two FFT‑based convolutions, allowing many iterations.
* If time permits, a light simulated‑annealing polish moves tiny packets
  of mass between coordinates.
* All steps respect the 1000 s budget and the returned list is ready for
  ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·n‑1) using a real FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact (scale‑invariant) objective
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w≥0, Σw=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 1) Classic sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the non‑smooth objective.
    Returns the best iterate seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        k = int(conv.argmax())                # shift achieving the max
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + diminishing step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # simplex projection
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 2) Smoothed gradient descent (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed version of the objective,
    where max(conv) is replaced by a soft‑max with temperature γ.
    Uses FFT‑based convolutions to keep each iteration cheap.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam‑style accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.15                # base learning rate
    it = 0

    # temperature schedule (start smooth → become harsh)
    gamma = 1.0
    gamma_max = 250.0
    gamma_factor = 1.0015   # multiplicative increase per iteration

    while time.time() < deadline:
        it += 1
        # ---------- convolution ----------
        conv = _conv_self(a)               # length 2·n‑1

        # ---------- soft‑max weights ----------
        maxc = conv.max()
        exp_term = np.exp(gamma * (conv - maxc))
        w = exp_term / exp_term.sum()      # shape (2·n‑1,)

        # ---------- window sums (for the a·∑w term) ----------
        # prefix sums of w (zero‑padded at start)
        pref = np.concatenate(([0.0], np.cumsum(w)))
        sum_window = pref[n:] - pref[:n]   # length n

        # ---------- correlation term ----------
        # correlation a * w (valid) = sum_i a_i w_{i+j}
        w_rev = w[::-1]
        L = _next_pow2(len(a) + len(w_rev) - 1)   # >= 3·n‑2
        fft_a = np.fft.rfft(a, n=L)
        fft_wr = np.fft.rfft(w_rev, n=L)
        corr = np.fft.irfft(fft_a * fft_wr, n=L)

        start = len(w) - 1                     # = 2·n‑2
        indices = start - np.arange(n)          # decreasing indices
        corr_term = corr[indices]                # shape (n,)

        # ---------- gradient ----------
        grad = corr_term + a * sum_window

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)

        a = a - step * m_hat

        # ---------- projection ----------
        a = _project_simplex(a)

        # ---------- evaluation ----------
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth iter={it:,} "
                  f"γ={gamma:.2f} score={cur_score:.6f}")

        # ---------- temperature increase ----------
        gamma = min(gamma_max, gamma * gamma_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3) Very cheap annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers with a Metropolis rule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = 0.02 * (1e-4 / 0.02) ** (it / total_steps)

        src = rng.choice(n, p=a)           # more mass → more likely to be source
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a larger canvas yields a finer approximation.
    # Use n = 16384 when we have the full budget, otherwise fall back.
    # --------------------------------------------------------------
    n = 16384 if budget_s >= 400 else 8192

    # --------------------------------------------------------------
    # 0) Warm‑start from a small family of symmetric shapes.
    # --------------------------------------------------------------
    x = np.linspace(0.0, 1.0, n, dtype=float)

    best_seq = None
    best_score = float("inf")

    # (a) Tent‑type shapes a_i ∝ min(i,n‑1‑i)^p
    exponents = np.arange(0.2, 1.55, 0.1)          # 0.2 … 1.5
    for p in exponents:
        w = np.minimum(x, 1.0 - x) ** p
        if w.sum() == 0.0:
            continue
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] tent p={p:.2f} "
              f"score={score:.6f}")

    # (b) Sin‑shape (zero at both ends)
    w = np.sin(np.pi * x)
    if w.sum() > 0.0:
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] sin shape score={score:.6f}")

    # (c) Semi‑circle (√(1‑(2x‑1)²))
    w = np.sqrt(np.maximum(0.0, 1.0 - (2.0 * x - 1.0) ** 2))
    if w.sum() > 0.0:
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] semi‑circle score={score:.6f}")

    # (d) Global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                cur = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
    except Exception:
        pass

    # Fallback – uniform distribution (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)
        print(f"[{time.time() - start_time:.2f}s] fallback uniform score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Sub‑gradient descent (fast coarse improvement)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining * 0.55
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] After sub‑grad score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smoothed (soft‑max) gradient descent (fine‑grained)
    # --------------------------------------------------------------
    if time.time() < deadline:
        smooth_seq, smooth_score = _smooth_descent(best_seq, deadline, start_time)
        if smooth_score < best_score:
            best_score = smooth_score
            best_seq = smooth_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] After smooth descent score={smooth_score:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing polish (if any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq]
