def _grad_from_lambda(a, lam, n, Lgrad):

    a_rev = a[::-1]

    lam_fft = np.fft.rfft(lam, n=Lgrad)

    a_rev_fft = np.fft.rfft(a_rev, n=Lgrad)

    # conv length Lgrad, slice indices n-1 to n-1 + n

    grad_slice = conv[n-1: n-1 + n]

    return grad

def _run_softmax_gradient(a0, n, deadline, start_time,
                          eta0=0.2, alpha0=0.5, alpha_max=200.0, alpha_growth=1.001):
    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    Lconv = _next_pow2(2*n-1)
    Lgrad = _next_pow2(3*n-2)
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        it += 1
        # compute self-convolution conv
        conv = _conv_self(a, Lconv)   # length 2n-1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] softmax iter={it:,} score={cur_score:.6f}")

        # compute softmax lambda
        lam = np.exp(alpha * conv - np.max(alpha * conv))
        lam /= lam.sum()
        # compute gradient
        grad = _grad_from_lambda(a, lam, n, Lgrad)

        # step size
        eta = eta0 / math.sqrt(it)

        # multiplicative update (exp update)
        #paper suggests additive gradient descent for smooth objective.
        # We'll use multiplicative to maintain non-negativity.
        # ensure eta*grad not too large
        a = a * np.exp(-eta * grad)
        # renormalize to sum=1
        total = a.sum()
        if total <= 0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= total

        # increase alpha gradually
        alpha = min(alpha_max, alpha * alpha_growth)

    return best_a, best_score

"""
Propose a non‑negative step function that (approximately) minimises
evaluate_sequence().

Algorithm
---------

1.  Initialise a probability vector `a` (sum = 1).  If a previous
    construction `height_sequence_1` exists it is interpolated, otherwise a
    uniform distribution is used.

2.  **Primal‑dual soft‑max descent**  
    - At each iteration compute the self‑convolution `c = a*a`.  
    - Form a soft‑max weight vector `λ = softmax(α·c)`; the temperature α
      grows linearly from `α0` to `α_max` during the run, so the early
      iterations see a smooth approximation of the max‑convolution and the
      later ones focus on the true maximum.  
    - The gradient of the smoothed objective is
        `∇F(a) = 2·(λ ∗ a_rev)[n‑1:n‑1+n]`
      where `a_rev` is `a` reversed; it is obtained with two FFTs.  
    - Update `a` by a multiplicative mirror‑descent step
        `a ← a·exp(-η·∇F(a))` and renormalise onto the simplex.

    This is a projected mirror‑descent on a smooth approximation of
    `max(a*a)` and typically beats the plain sub‑gradient method used
    before.

3.  **Greedy peak flattening** – the final few seconds run a deterministic
    mass‑shifting routine (identical to the one from the reference code)
    that moves a tiny amount of probability from the index contributing
    most to the current peak to an index that contributes least, updating
    the convolution incrementally.

4.  Optional short random‑restart polishing if a couple of seconds remain.

The whole process respects the 1000 s budget and returns the best sequence
found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # fallback – shouldn't happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max smoothed max‑convolution
# ----------------------------------------------------------------------
def _grad_from_lambda(a: np.ndarray, lam: np.ndarray,
                      n: int, Lgrad: int) -> np.ndarray:
    """
    Gradient of the smoothed objective.
    lam : weight vector of length 2·n‑1 (soft‑max of a*a)
    Returns an array of length n.
    """
    a_rev = a[::-1]
    lam_fft = np.fft.rfft(lam, n=Lgrad)
    a_rev_fft = np.fft.rfft(a_rev, n=Lgrad)
    prod = lam_fft * a_rev_fft
    conv = np.fft.irfft(prod, n=Lgrad)
    # slice indices n‑1 … n‑1+n‑1 (inclusive)
    return 2.0 * conv[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Phase 2 – primal‑dual soft‑max mirror descent
# ----------------------------------------------------------------------
def _run_softmax_gradient(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          eta0: float = 0.8,
                          alpha0: float = 0.5,
                          alpha_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation
        F(a) = (1/α)·log Σ exp(α·(a*a))
    Returns the best vector seen and its (unsmoothed) score
        2·n·max(a*a).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")

    Lconv = _next_pow2(2 * n - 1)          # size for the self‑convolution
    Lgrad = _next_pow2(3 * n - 2)          # size for λ∗a_rev

    it = 0
    while time.time() < deadline:
        it += 1

        # 1) current self‑convolution and score
        conv = _conv_self(a, Lconv)               # length 2·n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] softmax iter={it:,} "
                  f"score={cur_score:.6f}")

        # 2) soft‑max weights (temperature grows with elapsed time)
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = alpha0 + (alpha_max - alpha0) * progress
        scaled = conv * alpha
        max_scaled = scaled.max()
        lam = np.exp(scaled - max_scaled)
        lam /= lam.sum()

        # 3) gradient of the smoothed objective
        grad = _grad_from_lambda(a, lam, n, Lgrad)

        # 4) multiplicative mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # 5) renormalise onto the simplex (exact sum = 1)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – deterministic greedy reduction (same as reference)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Move a tiny amount of mass from the index that most contributes to the
    current peak to an index that contributes least, updating the full
    convolution incrementally.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        partner_vals = a[max_idx - idx]

        # choose destination with smallest partner value (and not src)
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size (4096 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previously found construction if
    # available, otherwise fall back to uniform.
    # ------------------------------------------------------------------
    a0 = np.full(n, 1.0 / n)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # linear interpolation to the required length
            a0 = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, starter.size),
                starter,
            )
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0.0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
    except Exception:
        # keep the uniform start if anything goes wrong
        pass

    # ------------------------------------------------------------------
    # Phase 2 – soft‑max mirror descent (majority of the time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # reserve ~30 s for the greedy phase
    if remaining > 35.0:
        grad_deadline = time.time() + remaining - 30.0
    else:
        grad_deadline = deadline

    best_a, best_score = _run_softmax_gradient(
        a0, n, grad_deadline, start_time,
        eta0=0.8, alpha0=0.5, alpha_max=200.0,
    )

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak flattening (use the rest)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _run_greedy_reduction(
            best_a, n, deadline, start_time,
            best_score, best_a,
            max_iters=10_000_000,
        )

    # ------------------------------------------------------------------
    # Optional tiny random‑restart polishing if a few seconds remain.
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        pert = best_a + 0.01 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)
        pert_sum = pert.sum()
        if pert_sum > 0.0:
            pert /= pert_sum
        else:
            pert = np.full(n, 1.0 / n)

        tmp_deadline = time.time() + 2.0
        seq, scr = _run_softmax_gradient(
            pert, n, tmp_deadline, start_time,
            eta0=0.6, alpha0=0.5, alpha_max=100.0,
        )
        if scr < best_score:
            best_score = scr
            best_a = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] random‑restart improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (the objective is homogeneous)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    # return as a plain list of Python floats
    return [float(x) for x in best_a]
