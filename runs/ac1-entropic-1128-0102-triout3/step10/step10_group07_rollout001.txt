The code below searches for a non‑negative step‑function (a list of heights) that minimises

```
score(a) = 2·n·max_k (a * a)[k]          (sum(a)=1)
```

which is exactly the value returned by `evaluate_sequence`.  
The optimisation consists of three phases that complement each other:

1. **Diverse initial pool** – a variety of handcrafted, random and
   parametric shapes (uniform, Gaussian, exponential, edge‑sparse,
   Dirichlet, power‑law …) are generated, normalised and the best one is
   kept.

2. **Exponentiated‑gradient polishing** – a mirror‑descent (multiplicative)
   update that follows the sub‑gradient of the active convolution shift.
   This quickly drives the candidate towards a good local optimum.

3. **Simulated‑annealing local search** – starting from the polished
   vector we perform many small “mass‑transfer’’ moves (`δ` taken from a
   heavy index and added to a light one).  Moves are accepted if they
   improve the score or with a temperature‑controlled probability.
   The temperature slowly decays, allowing the algorithm to escape
   shallow basins and to perform a fine‑grained refinement.

A final deterministic greedy step further reduces the peak if possible.
All arithmetic is performed with FFT‑based convolutions (`O(n log n)`) so
that millions of score evaluations comfortably fit into the 1000 s budget.

The function `propose_candidate` follows the required signature and
returns the best sequence found before the deadline.

```python
python
"""
Propose a step function that (almost) minimises ``evaluate_sequence``.

Algorithm
---------
1. Build a rich pool of starter vectors (uniform, Gaussian, exponential,
   edge‑sparse, Dirichlet draws, power‑law, symmetric geometric …).
2. Keep the best starter after a cheap FFT‑based evaluation.
3. Polish the best vector with an exponentiated‑gradient (mirror‑descent)
   scheme that follows the sub‑gradient of the active convolution shift.
4. Run a simulated‑annealing local search for the remaining time:
   repeatedly move a small fraction of mass from a heavy index to a light
   one, accepting uphill moves with a decreasing temperature.
5. Finish with the deterministic greedy peak‑reduction used in the
   reference implementation.
6. Clip, renormalise and return the final vector.

All steps respect the 1000 s wall‑clock limit and only use NumPy (the
objective is evaluated analytically, not via ``evaluate_sequence``,
which saves a large amount of time).
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (FFT convolution, score evaluation, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    Returns a 1‑D array of length 2·len(a)-1.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective value (identical to `evaluate_sequence` for
    non‑negative vectors with sum≈1).  Faster than calling the external
    function because we use an FFT convolution.
    """
    conv = _conv_self(a, fft_len)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (mirror‑descent)
# ----------------------------------------------------------------------


def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on f(a)=max_k (a*a)[k].
    Returns the best vector encountered and its exact score.
    """
    a = np.maximum(a0, 1e-15)          # avoid exact zeros at start
    a = a / a.sum()                    # stay on the simplex
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # active shift producing the maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]               # sub‑gradient for the active shift

        eta = eta0 / math.sqrt(t)       # diminishing step size
        a = a * np.exp(-eta * g)        # multiplicative update
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation (cheap guard against drift)
        if t % 500 == 0:
            sc = _score(a, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction (used at the very end)
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose `m` indices, give each equal mass 1/m, zero elsewhere.
    """
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


def _edge_sparse_start(n: int, m: int) -> np.ndarray:
    """Put equal mass on the first *m* and the last *m* positions."""
    vec = np.zeros(n, dtype=float)
    if 2 * m > n:
        m = n // 2
    vec[:m] = 1.0 / (2 * m)
    vec[-m:] = 1.0 / (2 * m)
    return vec


def _powerlaw_start(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + (n-i)^{-p} (symmetric power‑law)."""
    idx = np.arange(n, dtype=float)
    w = 1.0 / ((idx + 1) ** p) + 1.0 / ((n - idx) ** p)
    return w / w.sum()


def _symmetric_geometric_start(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i + r**(n-1-i) (mirror‑symmetric geometric)."""
    idx = np.arange(n, dtype=float)
    w = np.power(r, idx) + np.power(r, n - 1 - idx)
    return w / w.sum()

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine builds a diverse pool of starters, runs an exponentiated‑gradient
    polish, then a simulated‑annealing local search and finishes with a greedy
    peak‑reduction.  The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Problem size (use the length of the known reference if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)             # type: ignore[name-defined]
    except Exception:
        n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # 1) Assemble a rich set of initial candidates
    # ------------------------------------------------------------------
    starts: List[np.ndarray] = []

        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # b) classic starters
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_gaussian_start(n))                  # bell

    starts.append(_exponential_start(n, r=0.985))     # stronger decay

    # c) Dirichlet draws with different concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # d) sparse random supports
    for m in [400, 800, 1200, 2000]:
        for _ in range(2):
            starts.append(_sparse_random_start(n, m, rng))

    # e) edge‑sparse patterns (mass concentrated at both ends)
    for m in [200, 400, 800]:
        starts.append(_edge_sparse_start(n, m))

    # f) power‑law families
    for p in [0.5, 1.0, 1.5, 2.0]:
        starts.append(_powerlaw_start(n, p))

    # g) symmetric geometric families
    for r in np.linspace(0.90, 0.9999, 12):
        starts.append(_symmetric_geometric_start(n, r))

    # ------------------------------------------------------------------
    # 2) Evaluate the raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for idx, a0 in enumerate(starts):
        a = np.clip(a0, 0.0, None)
        a = a / a.sum()
        sc = _score(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:02d} score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) Exponentiated‑gradient polishing (≈15 % of the budget)
    # ------------------------------------------------------------------
    polish_deadline = start_time + 0.15 * (deadline - start_time)
    polished_seq, polished_score = _expgrad_optimize(
        best_seq, polish_deadline, start_time, fft_len, eta0=0.8
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polish score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing local search for the remaining time
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score

    # temperature initialised as a small fraction of the current score
    temperature = max(1e-8, cur_score * 0.02)
    # multiplicative decay factor (slow enough to give thousands of accepted moves)
    temp_decay = 0.999996

    # fraction of source mass moved in each trial
    delta_frac = 0.025

    while time.time() < deadline:
        # pick source proportional to its mass (more mass ⇒ more likely to be reduced)
        src = rng.choice(n, p=cur / cur.sum())
        # pick a different destination uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        delta = delta_frac * cur[src]
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_score = _score(cand, n, fft_len)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / temperature):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA new best: score={best_score:.6f}")

        # temperature schedule
        temperature = max(1e-12, temperature * temp_decay)

        # Occasionally re‑normalise to avoid drift caused by floating‑point noise
        if (time.time() - start_time) % 30 < 0.01:
            cur = np.clip(cur, 0.0, None)
            cur = cur / cur.sum()

    # ------------------------------------------------------------------
    # 5) Final deterministic greedy peak reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Safety clipping & normalisation before returning
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # Final log using the provided evaluation routine (purely for reporting)
    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
```
