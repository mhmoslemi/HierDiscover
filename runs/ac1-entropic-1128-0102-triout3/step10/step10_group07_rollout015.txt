"""
Propose a non‑negative step function (probability vector) that (almost) minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Initial pool** – generate a diverse set of candidates (uniform,
   Gaussian, exponential decays, Dirichlet draws and sparse supports).
   Keep the best one (fast evaluation using FFT‑based self‑convolution).

2. **Smooth exponential–gradient polishing** – minimise a smooth surrogate of
   the max‑convolution:
        g(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
   The gradient of g is a correlation of the soft‑max weights with a.
   We perform mirror‑descent on the simplex with multiplicative updates
   ``a←a·exp(−η·∇g)``.  The smoothing parameter β is increased from a small
   value to a large one during the run, allowing large steps early on and a
   tight approximation of the true max later.

   The loop runs until ~85 % of the time budget is exhausted,
   constantly tracking the best exact score (``2·n·max(a*a)``).

3. **Random‑restart bursts** – while time permits we draw fresh random starts,
   run a short (≈5 s) smooth‑gradient run and keep any improvement.  This helps
   to escape possible local minima.

4. **Deterministic greedy peak reduction** – a final local move repeatedly
   transfers a fraction of the coefficient that contributes most to the current
   convolution peak towards the smallest coefficient, accepting only strict
   improvements.

5. **Safety & return** – clip to ``≥0``, renormalise to unit sum and return the
   list of floats.

The routine respects a 1000 s wall‑clock limit and prints progress
information.  All heavy work is done with FFTs, so thousands of iterations
fit comfortably into the budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    Returns the first 2·len(a)-1 samples.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _score_from_a(a: np.ndarray, fft_len_self: int) -> float:
    """Exact objective 2·n·max(a*a).  a is assumed to sum to 1."""
    conv = _conv_self(a, fft_len_self)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Gradient optimisation (smooth max of the convolution)
# ----------------------------------------------------------------------


def _smooth_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len_self: int,
    fft_len_corr: int,
    eta0: float = 1.0,
    beta_start: float = 5.0,
    beta_end: float = 250.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the smooth surrogate
    g(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
    Returns the best vector seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)  # avoid exact zeros
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _score_from_a(a, fft_len_self)

    t = 0
    while time.time() < deadline:
        t += 1
        # --- FFT of a (common for both conv and correlation) -----------------
        A = np.fft.rfft(a, n=fft_len_corr)

        # --- self‑convolution (needed for the soft‑max weights) --------------
        conv_full = np.fft.irfft(A * A, n=fft_len_corr)
        conv = conv_full[: 2 * n - 1]                # linear self‑conv
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep exact best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] SmoothGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # --- smooth max weights -------------------------------------------------
        # linear interpolation of β between start and end
        fraction = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max with numerical stability
        scaled = beta * conv
        max_scaled = np.max(scaled)
        w_unnorm = np.exp(scaled - max_scaled)
        w = w_unnorm / w_unnorm.sum()                # length 2n‑1, sums to 1

        # --- gradient of the smooth max (correlation of w and a) ----------------
        W = np.fft.rfft(w, n=fft_len_corr)
        # linear correlation via a single inverse FFT (valid part later)
        corr_full = np.fft.irfft(W * np.conj(A), n=fft_len_corr)
        grad = corr_full[n - 1 : 2 * n - 1]          # length n

        # --- multiplicative mirror‑descent update ------------------------------
        eta = eta0 / math.sqrt(t)                   # diminishing step size
        update = -eta * grad
        # clip the exponent to avoid overflow/underflow
        update = np.clip(update, -50.0, 50.0)
        a = a * np.exp(update)

        # --- re‑normalise onto the simplex ------------------------------------
        s = a.sum()
        if s > 0.0 and np.isfinite(s):
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # occasional exact evaluation (cheap guard against drift)
        if t % 500 == 0:
            sc = _score_from_a(a, fft_len_self)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] Exact re‑eval improve: score={sc:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction
# ----------------------------------------------------------------------


def _greedy_peak_reduce(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len_self: int,
    max_iters: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Local deterministic move: move a fraction of the coefficient that
    contributes most to the current convolution peak towards the smallest
    coefficient.  Accept only strict decreases of the exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _score_from_a(a, fft_len_self)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len_self)
        cur_max = conv.max()
        k_max = conv.argmax()

        # indices that participate in this maximal shift
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]

        # source = index that contributes the most to this peak
        src = i_idx[contrib.argmax()]

        # destination = index with smallest current mass (excluding src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        # move a modest fraction (tuned empirically)
        eps = 0.25 * a[src]
        if eps <= 1e-15:
            break

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score_from_a(a_prop, fft_len_self)
        if new_score < best_score - 1e-12:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: score={new_score:.6f}")

    return a, best_score

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc` (sum = 1)."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑1."""
    sigma = n / 6.0
    xs = np.arange(n, dtype=float)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.power(r, np.arange(n, dtype=float))
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Select `m` indices uniformly, give each equal mass 1/m."""
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best coefficient list found within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4800
    fft_len_self = _next_pow2(2 * n - 1)          # enough for self‑conv
    fft_len_corr = _next_pow2(3 * n - 2)          # enough for correlation

    # ------------------------------------------------------------------
    # Build a diverse set of initial candidates
    # ------------------------------------------------------------------
    starts = []

    # 1) possibly supplied previous solution
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) handcrafted starters
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_gaussian_start(n))                  # bell‑shaped
    starts.append(_exponential_start(n, r=0.9995))     # very gentle decay
    starts.append(_exponential_start(n, r=0.985))      # stronger decay

    # 3) Dirichlet draws with different concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # 4) sparse random supports – several densities
    for m in [400, 800, 1200, 2000]:
        for _ in range(2):
            starts.append(_sparse_random_start(n, m, rng))

    # 5) a pure exponential draw (continuous, not normalised yet)
    exp_vec = rng.exponential(scale=1.0, size=n)
    starts.append(exp_vec / exp_vec.sum())

    # ------------------------------------------------------------------
    # Evaluate the raw starts (fast exact score) and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = _score_from_a(a0, fft_len_self)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long smooth‑gradient polishing (≈85 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.85 * (deadline - start_time)
    polished_seq, polished_score = _smooth_gradient_optimize(
        best_seq,
        phase1_deadline,
        start_time,
        fft_len_self,
        fft_len_corr,
        eta0=1.0,
        beta_start=5.0,
        beta_end=250.0,
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polish score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – cheap random‑restart bursts (≈10 % of remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # draw a fresh start (sparse or Dirichlet)
        if rng.random() < 0.5:
            m = int(rng.integers(300, 2500))
            cand_start = _sparse_random_start(n, m, rng)
        else:
            conc = 10 ** rng.uniform(-1.5, 1.0)   # roughly 0.03 … 10
            cand_start = _dirichlet_start(n, conc=conc, rng=rng)

        # short smooth‑gradient run (~5 s)
        sub_deadline = time.time() + 5.0
        cand_seq, cand_score = _smooth_gradient_optimize(
            cand_start,
            sub_deadline,
            start_time,
            fft_len_self,
            fft_len_corr,
            eta0=0.8,
            beta_start=5.0,
            beta_end=150.0,
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from restart score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – final deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_peak_reduce(
            best_seq, deadline, start_time, fft_len_self, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # Report the score via the official evaluator (guards against clipping)
    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
