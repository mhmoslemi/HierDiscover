assistantfinal**Solution Overview**

The program searches for a non‑negative step function `a = (a₀,…,a_{n‑1})` (here `n = 4096`) that minimises  

```
2·n·max(conv(a,a)) / (sum(a)²)
```

The objective is homogeneous, therefore we can restrict the search to the probability simplex
`a_i ≥ 0 , Σ a_i = 1`.  

The search consists of three phases

1. **Adam‑based smooth‑max descent** – we minimise a smoothed version of the maximum
   `max_k conv_k(a,a)` using the log‑sum‑exp (soft‑max) approximation.
   The gradient of the smooth objective is obtained analytically in the
   frequency domain (FFT).  The parameters are the *logits* `π` with
   `a = softmax(π)`.  Adam‑style moment estimates are kept, and the smoothing
   parameter `α` is increased geometrically during the run.

2. **Projected sub‑gradient descent** – after the smooth phase we polish the
   solution with the classic sub‑gradient method used in the reference
   implementation (momentum + projection onto the simplex).

3. **Small simulated‑annealing polish** – a cheap mass‑transfer annealer
   makes the final fine improvements.

The whole routine respects the 1000 s wall‑clock limit, prints progress,
and finally returns the best sequence it has found.  It can also start from a
previously discovered good sequence (`height_sequence_1`) if such a global
variable exists.

```python
"""
minimise 2·n·max(conv(a,a)) / (sum(a)²)  for n = 4096

The algorithm:
  1. Adam‑based smooth‑max descent on the simplex (logits → soft‑max).
  2. Short projected sub‑gradient refinement (momentum + simplex projection).
  3. Tiny simulated‑annealing polish (mass‑transfer moves).

All heavy work is done with FFTs, the code never exceeds the given
time‑budget and prints intermediate scores.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT based convolutions, simplex projection, fast eval)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution, not a circular one.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear convolution of two real vectors u and v using FFT.
    Returns the full linear convolution (len(u)+len(v)-1 elements).
    """
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (reference implementation)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=_next_pow2(2 * n - 1))
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index of the current maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# Phase‑3 : tiny simulated‑annealing polish (identical to reference)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

def _smooth_adam_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth approximation
    of max(conv(a,a)).  The smoothing parameter α is increased geometrically.
    """
    # basic sizes for FFTs
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # initialise logits (π) from the start vector
    eps = 1e-12
    pi = np.log(a0 + eps)

    # Adam moments
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    lr = 0.05                     # base learning rate
    t = 0

    # scoring
    best_a = a0.copy()
    best_score = _eval_fast(best_a, L_self)

    # smoothing schedule (more aggressive than the reference)
    alphas = np.geomspace(0.5, 5000.0, num=40)

    for alpha in alphas:
        # inner loop: run until time runs out or a modest iteration cap
        while time.time() < deadline - 0.2:
            t += 1

            # ---- a = softmax(π) ----
            max_pi = np.max(pi)
            exp_pi = np.exp(pi - max_pi)
            a = exp_pi / exp_pi.sum()

            # ---- self‑convolution ----
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # ---- soft‑max weights for the smoothing ----
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)        # shift for numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # ---- gradient of the smooth max (w.r.t a) ----
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]   # shape (n,)

            # ---- convert gradient to logits (π) space ----
            avg = np.dot(grad, a)                     # ⟨grad, a⟩
            grad_pi = (grad - avg) * a                # dL/dπ

            # ---- Adam update on π ----
            m = beta1 * m + (1.0 - beta1) * grad_pi
            v = beta2 * v + (1.0 - beta2) * (grad_pi ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
            pi = pi - step

            # ---- occasional exact evaluation ----
            if t % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.2f} it={t:,} score={cur_score:.6f}")

            if time.time() >= deadline - 0.2:
                break
        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
