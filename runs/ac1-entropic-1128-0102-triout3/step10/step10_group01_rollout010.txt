"""
Propose a step‑function (non‑negative heights) with a very low
evaluate_sequence() score.

Strategy
--------
1. Choose a relatively large problem size (n = 16384).  The optimal
   continuous arcsine distribution yields an asymptotic objective of 1.5;
   using a larger n brings the discrete problem closer to this limit.

2. Sweep a simple one‑parameter family
       a_i ∝ (x_i·(1‑x_i))^{‑α},
   with α around 0.5 (the arcsine law).  For each α we normalise the
   vector to Σa_i = 1 and keep the best candidate (fast FFT‑based
   evaluation).

3. Refine the best candidate by a projected (momentum) gradient descent
   on a smooth surrogate of the true objective:
       L(a) = (2·n/λ)·log ∑_k exp(λ·(a∗a)_k) .
   The gradient can be written in closed form and computed with a
   single pair of FFTs per iteration.  We run several phases with
   increasing smoothing parameter λ = 5,10,20,40,80, each phase limited
   only by the remaining wall‑time.

4. Keep track of the true objective
       obj(a) = 2·n·max(a∗a)          (since Σa_i = 1)
   and return the best sequence ever seen.

All updates are projected onto the probability simplex
{a_i ≥ 0, Σa_i = 1}.  The whole routine respects the 1000 s budget
and never produces illegal entries.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by the judge:
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient descent (single iteration helper)
# ----------------------------------------------------------------------
def _smooth_step(a: np.ndarray,
                 lam: float,
                 beta: float,
                 momentum: np.ndarray,
                 step0: float,
                 it: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Perform one iteration of the projected momentum descent on the
    smoothed objective with parameter `lam`.

    Returns the updated vector `a` and the updated momentum.
    """
    n = a.shape[0]

    # Pad length large enough for both self‑convolution and correlation
    L = _next_pow2(3 * n)                     # >= n + (2n‑1)
    Af = np.fft.rfft(a, n=L)

    # ------------------- self‑convolution ----------------------------
    conv_full = np.fft.irfft(Af * Af, n=L)
    conv = conv_full[: 2 * n - 1]            # linear conv a*a
    max_c = conv.max()

    # ------------------- smoothed weights ---------------------------
    # w_k ∝ exp(λ·conv_k)  (shifted to avoid overflow)
    exp_shift = lam * (conv - max_c)
    w = np.exp(exp_shift)
    w_sum = w.sum()
    if w_sum == 0.0:                         # should never happen
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ------------------- correlation a * w_rev -----------------------
    w_rev = w[::-1]                          # reverse for correlation
    Bf = np.fft.rfft(w_rev, n=L)
    corr_full = np.fft.irfft(Af * Bf, n=L)   # linear a ★ w
    start = n - 1
    corr = corr_full[start:start + n]        # length n, all ≥0, sums to 1

    # ------------------- momentum / step ----------------------------
    # Gradient of the smoothed objective is (4·n/λ)·corr.
    # We absorb the constant into the step size.
    momentum = beta * momentum + (1.0 - beta) * corr
    step = step0 / math.sqrt(it)             # decreasing step
    a_new = a - step * momentum               # move opposite to gradient
    a_new = _project_simplex(a_new)           # stay on simplex

    return a_new, momentum

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.

    Returns a list of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.3

    # ------------------------------------------------------------------
    # 1) Choose problem size – larger n gives a tighter asymptotic bound.
    #    16384 works comfortably within the budget.
    # ------------------------------------------------------------------
    if budget_s >= 300:
        n = 16384
    elif budget_s >= 150:
        n = 8192
    else:
        n = 4096

    # ------------------------------------------------------------------
    # 2) Coarse sweep of the arcsine‑type family a_i ∝ (x·(1−x))^{‑α}
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions in (0,1)

    # A modest grid around the classical α = 0.5
    alphas = np.linspace(0.35, 0.65, 15)
    for alpha in alphas:
        w = (x * (1.0 - x) + 1e-30) ** (-alpha)
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # progress report
        print(f"[{time.time() - start_time:.2f}s] α={alpha:.3f}  obj={score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

                cur_score = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] Global starter obj={cur_score:.6f}")
                if cur_score < best_score:
                    best_score = cur_score

    except Exception:

    # ------------------------------------------------------------------
    # 4) Smoothed projected gradient descent with increasing λ.
    # ------------------------------------------------------------------
    if best_seq is None:                       # safety fallback

    remaining = deadline - time.time()

            # sub‑gradient of max(conv) – same as in the reference code

            lo = max(0, k - (n - 1))

    # ------------------------------------------------------------------

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()

    else:

    print(f"[{time.time() - start_time:.2f}s] Returning best objective = {best_score:.6f}")
