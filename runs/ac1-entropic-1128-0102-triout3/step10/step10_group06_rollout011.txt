"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

The method combines three ideas that differ from the reference
implementation:

1. **Diverse initialisation** – a few handcrafted seeds and a handful
   of random Dirichlet draws (different concentration parameters).
2. **LP‑based refinement** – for a current sequence a we solve the
   linear programme  

        maximise   Σ b_i
        subject to  a * b ≤ M   (pointwise)
                    b_i ≥ 0

   where ``M = max(a*a)`` and ``*`` denotes the ordinary (non‑circular)
   convolution.  The optimum b is the most “massive’’ sequence that can
   be added without raising the current peak ``M``.  After normalising
   b we move a a little towards it.  Repeating this step shrinks the
   peak steadily.
3. **Deterministic greedy polishing** – a final local search that moves
   a small fraction of the coefficient that contributes most to the
   current peak towards the smallest coefficient (strictly decreasing
   moves only).

All phases respect the supplied time budget (default 1000 s) and the
best sequence found is returned.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv

# Adam warm‑up (optional, but useful)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    A light‑weight Adam optimiser on the smoothed max‑convolution
    objective.  Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule (few steps – we only have a limited budget)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0]
    iters_per_alpha = 800

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            # soft‑max → valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 200 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: "
                        f"α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()

def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # modest fraction of the source mass
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

    starts.append(_gaussian_start(n))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))

    # 3) random Dirichlet draws with several concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # --------------------------------------------------------------
    # Pick the best raw start (exact evaluation)
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} score={sc:.6f}")

    # --------------------------------------------------------------
    # (Optional) short Adam warm‑up – uses ~5 % of the budget
    # --------------------------------------------------------------
    warmup_deadline = start_time + budget_s * 0.05
    a_warm, sc_warm = _adam_optimize(best_a, warmup_deadline, start_time, fft_len)
    if sc_warm < best_score:
        best_score = sc_warm
        best_a = a_warm.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** after Adam warm‑up best={best_score:.6f}")

    # --------------------------------------------------------------
    # LP‑based refinement (major part of the time budget)
    # --------------------------------------------------------------
    refine_deadline = start_time + budget_s * 0.85
    a_cur = best_a.copy()

    # keep a tiny floor to avoid unbounded LPs
    eps = 1e-12
    a_cur = np.maximum(a_cur, eps)
    a_cur = a_cur / a_cur.sum()

    # pre‑create CVXPY objects – this way we avoid rebuilding the model
    a_param = cp.Parameter(n, value=a_cur, nonneg=True)
    M_param = cp.Parameter(nonneg=True)
    b = cp.Variable(n, nonneg=True)
    conv_expr = cp.conv(a_param, b)                # length 2n‑1
    lp = cp.Problem(cp.Maximize(cp.sum(b)), [conv_expr <= M_param])

    iteration = 0
    while time.time() < refine_deadline:
        iteration += 1

        # current peak M
        M = np.max(np.convolve(a_cur, a_cur))
        # update CVXPY parameters
        a_param.value = a_cur
        M_param.value = M

        # solve the LP
        try:
            lp.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                     abstol=1e-8, reltol=1e-8, feastol=1e-8)
        except Exception as e:
            print(f"[{time.time() - start_time:6.2f}s] LP solve failed: {e}")
            break

        if b.value is None:
            print(f"[{time.time() - start_time:6.2f}s] LP did not return a solution.")
            break

        b_sol = np.maximum(b.value, 0.0)
        sum_b = b_sol.sum()
        if sum_b <= 0:
            print(f"[{time.time() - start_time:6.2f}s] LP yielded zero mass – stop.")
            break

        # normalise to a probability vector
        g = b_sol / sum_b

        # ---------------------------------------------------------
        # Choose a mixing factor – try a few candidates and keep the
        # one that gives the smallest exact score.
        # ---------------------------------------------------------
        candidates = [(1.0, g),          # full jump
                      (0.5, g),          # half way
                      (0.25, g),         # quarter
                      (0.125, g)]        # small step

        best_local_a = a_cur
        best_local_score = evaluate_sequence(a_cur.tolist())

        for t_mix, direction in candidates:
            a_tmp = (1.0 - t_mix) * a_cur + t_mix * direction
            a_tmp = np.maximum(a_tmp, eps)
            a_tmp = a_tmp / a_tmp.sum()
            sc_tmp = evaluate_sequence(a_tmp.tolist())
            if sc_tmp < best_local_score:
                best_local_score = sc_tmp
                best_local_a = a_tmp

        a_cur = best_local_a
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = a_cur.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP iter {iteration:5d} improve: score={best_local_score:.6f}")

        # very small safety break‑out if progress stalls
        if iteration % 50 == 0 and time.time() > deadline - 30:
            # leave a little time for the final greedy phase
            break

    # --------------------------------------------------------------
    # Final deterministic greedy polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_a, final_sc = _greedy_local_improve(best_a, deadline, start_time, max_iters=5000)
        if final_sc < best_score:
            best_score = final_sc
            best_a = final_a.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** after greedy polishing score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity checks and return
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, None)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / n)

    final_reported = evaluate_sequence(best_a.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_a.tolist()]
