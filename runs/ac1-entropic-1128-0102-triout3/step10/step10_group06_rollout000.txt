Algorithm – multi‑phase with aggressive optimisation
----------------------------------------------------

   random Dirichlet draws.  The best of them (by the exact objective)
   is kept.

   log‑sum‑exp of the convolution is minimised with Adam for the bulk

   max function.

   set** – a mirror‑descent on the exact max‑convolution objective.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)                     # length 2n‑1
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    The routine runs until ``deadline`` and returns the best vector
    (as a probability vector) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    # initialise logits to avoid zeros
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule – extended compared with the reference code
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]
    iters_per_alpha = 5000                     # a few thousand per α

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max → probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of smoothed surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score

def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] – sum of gradients of the listed
    shifts.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_polish(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    fft_len: int,
                    eta0: float = 1.0,
                    delta_frac: float = 5e-4,
                    top_k: int = 10,
                    momentum: float = 0.9,
                    jitter_interval: int = 2000,
                    jitter_scale: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    * Momentum* on the sub‑gradient and occasional random jitter are
      added to escape shallow local minima.
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    grad_mom = np.zeros_like(a)
    last_improve_iter = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_iter = t
            print(
                f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ---- active shifts -------------------------------------------------
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ---- sub‑gradient --------------------------------------------------
        g = _multi_gradient(a, active)

        # ---- momentum ------------------------------------------------------
        grad_mom = momentum * grad_mom + (1.0 - momentum) * g

        # ---- step size schedule (diminishing) ------------------------------
        eta = eta0 / math.sqrt(t)

        # ---- exponentiated update -----------------------------------------
        a = a * np.exp(-eta * (g + grad_mom))
        a_sum = a.sum()
        if a_sum == 0.0 or np.isnan(a_sum) or np.isinf(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        a = np.maximum(a, 1e-15)

        # ---- jitter when stuck --------------------------------------------
        if t - last_improve_iter >= jitter_interval:
            # add small multiplicative noise, re‑normalise
            noise = 1.0 + jitter_scale * np.random.randn(n)
            a = np.maximum(a * noise, 0.0)
            a = a / a.sum()
            last_improve_iter = t
            print(
                f"[{time.time() - start_time:6.2f}s] jitter applied at iter={t:,}"
            )

        # ---- occasional exact evaluation -----------------------------------
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

def _greedy_refine(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Deterministic local search that transfers a fraction of the
    coefficient responsible for the current convolution peak towards
    the smallest coefficient.  Several fractions are tried and the
    best improving move is accepted.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, fft_len).max()
    it = 0
    fractions = (0.10, 0.05, 0.02, 0.01)

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination – smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        best_local_a = None
        best_local_max = cur_max

        for frac in fractions:
            eps = frac * a[src]
            if eps <= 1e-14:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_max = _conv_self(a_prop, fft_len).max()
            if new_max < best_local_max:
                best_local_max = new_max
                best_local_a = a_prop

        if best_local_a is not None:
            a = best_local_a
            best_score = 2.0 * n * best_local_max
            print(
                f"[{time.time() - start_time:6.2f}s] greedy refine: iter={it:,} max={best_local_max:.6f}"
            )
        else:
            # no improving transfer – give up
            break

    return a, best_score

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _triangular_start(n: int) -> np.ndarray:
    """Linear rise to the centre and fall back – a simple triangle."""
    half = n // 2
    left = np.arange(1, half + 1)
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = np.arange(half, 0, -1)
    tri = np.concatenate([left, right])
    return tri / tri.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of a rich initialisation, an extended Adam warm‑up,
    a momentum‑enhanced exponentiated‑gradient polishing, and a refined
    greedy peak‑reduction.  The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a rich set of starting points
    # ------------------------------------------------------------------
    starts = []

    # Possibly a previously discovered good sequence
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # Hand‑crafted patterns
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_exponential_start(n, r=0.9995))    # gentle decay
    starts.append(_exponential_start(n, r=0.985))     # stronger decay
    starts.append(_gaussian_start(n))                 # bell‑shaped
    starts.append(_triangular_start(n))               # triangle
    starts.append(_dirichlet_start(n, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n, conc=0.5, rng=rng))

    # Random Dirichlet draws with various concentrations
    conc_params = [0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_params:
        for _ in range(5):
            starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # ------------------------------------------------------------------
    # Evaluate raw starts, keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 1 – Long Adam warm‑up (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.15
    cand_seq, cand_score = _adam_optimize(
        best_seq, phase1_deadline, start_time, fft_len
    )
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 2 – Momentum‑exp‑gradient polishing (≈70 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.85
    polished_seq, polished_score = _expgrad_polish(
        best_seq, phase2_deadline, start_time, fft_len,
        eta0=1.0, delta_frac=5e-4, top_k=10,
        momentum=0.9, jitter_interval=2000, jitter_scale=0.01
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polishing score={best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Phase 3 – Refined greedy peak reduction (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_refine(
            best_seq, deadline, start_time, fft_len, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy refine score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
