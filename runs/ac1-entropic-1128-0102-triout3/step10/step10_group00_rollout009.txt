"""
Candidate generator for the “step‑function” optimisation challenge.

Main ideas
----------
1. **Rich parametric warm‑start** –  a symmetric “plateau‑tent’’ shape
   a_i = min(1, (min(x_i,1-x_i)/t)**p)  (x_i = i/(n‑1)).
   The two parameters `t` (plateau width) and `p` (tail curvature) are
   swept on a modest grid; the best shape is kept.
2. **Adam‑style projected (sub‑)gradient descent** – the non‑smooth
   objective `2·n·max(conv)/(sum^2)` is approximated by the sub‑gradient
   of the *max* term.  Instead of the single most‑violating index we
   aggregate the top‑k convolution entries (k≈5) to obtain a smoother
   descent direction.  After each step the iterate is projected onto the
   probability simplex (Euclidean projection).
3. **Very light simulated‑annealing polish** – tiny random mass
   transfers are accepted with a Metropolis rule; this helps to escape
   tiny local minima that Adam may leave untouched.
4. **Scalable problem size** – the default canvas has n = 16384
   (≈2·8192).  All heavy work (FFT based convolutions) is O(n log n) and
   comfortably fits into the 1000 s budget with the chosen iteration
   limits.

The routine respects the time limit, prints progress information and
returns a plain list of non‑negative floats that will be fed to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:            # guard against divide‑by‑zero
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto { w ≥ 0, Σ w = 1 } (Wang & Carreira‑Perpinán, 2013)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent (softened max)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  top_k: int = 5,
                  step0: float = 0.5,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the exact non‑smooth objective.
    Returns the best iterate seen before ``deadline`` together with its score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    # Main loop
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)

        # Pick the top‑k convolution entries (or all if fewer)
        if top_k >= conv.shape[0]:
            idxs = np.arange(conv.shape[0])
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        # Sub‑gradient for each selected index
        grad = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx_range = np.arange(lo, hi + 1, dtype=int)
            grad[idx_range] += 2.0 * a[k - idx_range]

        # Average over the selected indices for a smoother direction
        grad /= max(1, len(idxs))

        # Adam update (step size decays like 1/√t)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto the simplex
        a = _project_simplex(a)

        # Keep the best solution so far
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (optional, runs only if time left)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap anneal that moves tiny packets of mass."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # Fixed budget; stop early if we run out of time
    total_steps = 40000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # Fast cooling schedule (geometric)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        # Choose a source index proportionally to its mass (more mass → more moves)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # Transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
