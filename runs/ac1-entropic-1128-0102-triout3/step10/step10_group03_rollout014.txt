def _targeted_hill_climb(start_a, deadline, start_time, L_fft):

    a = start_a.copy()

    best_a = a.copy() ; best_score = _eval_fast(a, L_fft)

    while time.time() < deadline:

        conv = _conv_self(a, L_fft)

        k = int(conv.argmax())

        lo = max(0, k - (n - 1))

        hi = min(n - 1, k)

        idx_range = np.arange(lo, hi + 1)

        # gradient magnitude at each idx: g[idx] = 2 * a[k - idx]

        g = np.zeros_like(a)

        g[idx_range] = 2.0 * a[k - idx_range]

        # compute src as index where g is max and a>small

        src_candidates = idx_range[g[idx_range] > 0]

        if src_candidates.size == 0:

            # fallback to random move

            src = np.random.choice(n, p=a)

        else:

            src = src_candidates[np.argmax(g[src_candidates])]

        # destination: index with smallest a

        dst = np.argmin(a)

        if src == dst:

            # choose random other

            dst = (dst + 1) % n

        if a[src] < 1e-12:

            continue

        eps = a[src] * 0.05 * np.random.rand()  # up to 5% of source mass

        a_prop = a.copy()

        a_prop[src] -= eps

        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)

        if new_score < best_score - 1e-12:

            a = a_prop

            best_score = new_score

            best_a = a.copy()

            print(f"[{time.time() - start_time:.2f}s] Targeted climb score={best_score:.6f}")

        # Also occasionally perform random hill move (like original) for diversification

        if np.random.rand() < 0.1:

            src2 = np.random.choice(n, p=a)

            dst2 = np.random.choice(n)

            if src2 != dst2 and a[src2] > 1e-14:

                eps2 = a[src2] * 0.1 * np.random.rand()

                a2 = a.copy()

                a2[src2] -= eps2

                a2[dst2] += eps2

                if s2 < best_score:

                    a = a2

                    best_score = s2

                    best_a = a.copy()

                    print(f"[{time.time() - start_time:.2f}s] Random climb score={best_score:.6f}")

"""
Search for a non‑negative step function that minimizes the evaluation
objective 2·n·max(conv(a,a)) / (sum(a)²).  The search proceeds in four
stages:

1. Fast parametric sweeps over several analytic families (geometric,
   power, Beta‑type, Gaussian, raised‑cosine) to obtain a good
   starting point.

2. Smoothed gradient descent on a soft‑max approximation of the
   ``max`` using the Adam optimiser.  The temperature of the soft‑max
   is exponentially decayed, the iterates are projected onto the
   probability simplex after each step, and several random restarts
   are performed.

3. A projected sub‑gradient descent (momentum, 1/√t step‑size) for
   fine‑grained refinement.

4. Directed hill‑climbing moves that exploit the active‑set gradient
   of the current maximal convolution entry.  Small mass transfers
   from “high‑gradient’’ to “low‑mass’’ coordinates are accepted only
   when they strictly improve the true objective.  The routine runs
   until the time budget (≈ 1000 s) is exhausted.

All helpers are top‑level functions; the returned sequence is
compatible with the supplied ``evaluate_sequence`` function.
"""
import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w ≥ 0 and sum(w) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return the exact objective 2·n·max(conv(a,a)) / (sum(a)²)
    assuming a is on the simplex (sum≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

def _adam_descent(start_a: np.ndarray,
                  deadline: float,
                  start_time: float,
                  L_fft: int,
                  L_grad: int) -> tuple[np.ndarray, float]:
    """
    Smoothed optimisation of 2·n·max(conv) using Adam on the
    soft‑max approximation of the max.  Returns the best vector seen.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # soft‑max temperature schedule
    tau = 0.5          # start relatively smooth
    tau_min = 1e-8
    tau_decay = 0.9995

    # learning‑rate (α) for Adam
    alpha = 0.12

    while time.time() < deadline:
        t += 1
        # compute convolution
        conv = _conv_self(a, L=L_fft)          # length 2n‑1
        maxc = conv.max()

        # soft‑max weights (stable)
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # ∑ w = 1

        # gradient of the smoothed objective
        # grad_j = 2 * Σ_i a_i w_{i+j}
        w_rev = w[::-1]                         # reversed weights, length 2n‑1
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step
        a = _project_simplex(a)

        # evaluate exact objective
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (the one attaining the max)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

def _targeted_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing that moves a small amount of mass from the
    coordinate with the largest active‑set gradient to the coordinate with the
    smallest mass.  Only improving moves are kept.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        # compute active‑set gradient (sub‑gradient of max)
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[k - idx_range]

        # source = index with largest gradient (and non‑zero mass)
        src_candidates = idx_range[g[idx_range] > 0]
        if src_candidates.size == 0:
            # fallback to random source weighted by mass
            src = rng.choice(n, p=a)
        else:
            src = src_candidates[np.argmax(g[src_candidates])]

        # destination = index with smallest mass (promotes spreading)
        dst = np.argmin(a)
        if src == dst or a[src] < 1e-14:
            # if the choice is degenerate, pick a random move instead
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst:
                continue

        # move a modest fraction of the source mass
        eps = a[src] * rng.random() * 0.07  # ≤ 7 % of src mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Targeted hill it={it:,} best={best_score:.6f}")

        # occasionally try a completely random small transfer for extra diversity
        if rng.random() < 0.08:
            src2 = rng.choice(n, p=a)
            dst2 = rng.integers(n)
            if src2 != dst2 and a[src2] > 1e-14:
                eps2 = a[src2] * rng.random() * 0.10
                a2 = a.copy()
                a2[src2] -= eps2
                a2[dst2] += eps2
                s2 = _eval_fast(a2, L_fft)
                if s2 < best_score:
                    a = a2
                    best_score = s2
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Random hill it={it:,} best={best_score:.6f}")

    return best_a, best_score

    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²) (a is assumed ∑a≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
# Adam‑based smoothed descent
# ----------------------------------------------------------------------
def _adam_descent(start_a: np.ndarray,
                  deadline: float,
                  start_time: float,
                  L_fft: int,
                  L_grad: int) -> tuple[np.ndarray, float]:
    """
    Smoothed optimisation of 2·n·max(conv) using Adam on the
    soft‑max approximation of the max.  Returns the best vector seen.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # soft‑max temperature schedule
    tau = 0.5
    tau_min = 1e-8
    tau_decay = 0.9995

    # learning rate (α)
    alpha = 0.12

    while time.time() < deadline:
        t += 1
        # convolution
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()

        # soft‑max weights (stable)
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # ∑ w = 1

        # gradient of the smoothed objective:
        #   grad_j = 2 * Σ_i a_i w_{i+j}
        w_rev = w[::-1]                        # reverse weights (length 2n‑1)
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step
        a = _project_simplex(a)

        # exact objective
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

# ----------------------------------------------------------------------
# Projected sub‑gradient descent (as in the reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (where the max is attained)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projection
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score

# ----------------------------------------------------------------------
# Targeted hill‑climbing polish
# ----------------------------------------------------------------------
def _targeted_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing that moves a small amount of mass from the
    coordinate with the largest active‑set gradient to the coordinate with
    the smallest mass.  Only improving moves are kept.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_fft)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_range = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of the max at the active index
        g = np.zeros_like(a)
        g[idx_range] = 2.0 * a[k - idx_range]

        # source = coordinate with largest gradient (and non‑zero mass)
        src_candidates = idx_range[g[idx_range] > 0]
        if src_candidates.size == 0:
            src = rng.choice(n, p=a)          # fallback
        else:
            src = src_candidates[np.argmax(g[src_candidates])]

        # destination = coordinate with smallest mass
        dst = np.argmin(a)
        if src == dst or a[src] < 1e-14:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst:
                continue

        # transfer a modest fraction of the source mass
        eps = a[src] * rng.random() * 0.07
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Targeted hill it={it:,} best={best_score:.6f}")

        # occasional random small move for extra diversity
        if rng.random() < 0.08:
            src2 = rng.choice(n, p=a)
            dst2 = rng.integers(n)
            if src2 != dst2 and a[src2] > 1e-14:
                eps2 = a[src2] * rng.random() * 0.10
                a2 = a.copy()
                a2[src2] -= eps2
                a2[dst2] += eps2
                s2 = _eval_fast(a2, L_fft)
                if s2 < best_score:
                    a = a2
                    best_score = s2
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Random hill it={it:,} best={best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm mixes parametric sweeps, Adam‑based smoothed descent,
    a sub‑gradient refinement and a directed hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # Uniform baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if the sandbox provides one)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Fast parametric sweep (extra analytic families added)
    # ------------------------------------------------------------------
    idx = np.arange(n)

    # geometric family
    eps_grid = np.logspace(-4, -1, 150)
    for eps in eps_grid:
        if time.time() >= deadline - 30:
            break
        r = 1.0 - eps
        geo = np.power(r, idx, dtype=np.float64)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _eval_fast(seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # power family
    p_grid = np.linspace(-1.0, 2.0, 51)
    for p in p_grid:
        if time.time() >= deadline - 30:
            break
        pow_seq = np.power(idx + 1, p, dtype=np.float64)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _eval_fast(pow_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.2f} score={sc:.6f}")

    # Beta‑type family
    beta_params = [0.0, 0.5, 1.0, 1.5, 2.0]
    for p in beta_params:
        for q in beta_params:
            if time.time() >= deadline - 30:
                break
            beta_seq = (np.power(idx + 1, p) *
                        np.power(n - idx, q))
            s = beta_seq.sum()
            if s == 0:
                continue
            beta_seq /= s
            sc = _eval_fast(beta_seq, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = beta_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta p={p:.1f} q={q:.1f} score={sc:.6f}")

    # Gaussian family (centered)
    sigma_factors = [0.05, 0.10, 0.20, 0.35]
    for sigma_factor in sigma_factors:
        if time.time() >= deadline - 30:
            break
        sigma = sigma_factor * n
        i = np.arange(n)
        gauss = np.exp(-0.5 * ((i - n // 2) / sigma) ** 2)
        s = gauss.sum()
        if s == 0:
            continue
        gauss /= s
        sc = _eval_fast(gauss, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = gauss.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma_factor:.2f} score={sc:.6f}")

    # Raised‑cosine (window) family
    width_factors = [0.25, 0.40, 0.60]
    for w_frac in width_factors:
        if time.time() >= deadline - 30:
            break
        half_width = int(w_frac * n / 2)
        centre = n // 2
        i = np.arange(n)
        dist = np.abs(i - centre)
        cosine = np.where(dist <= half_width,
                          0.5 * (1 + np.cos(np.pi * dist / half_width)),
                          0.0)
        s = cosine.sum()
        if s == 0:
            continue
        cosine /= s
        sc = _eval_fast(cosine, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cosine.copy()
            print(f"[{time.time() - start_time:.2f}s] Cosine width={w_frac:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Adam‑based smoothed descent with random restarts
    # ------------------------------------------------------------------
    min_rest_time = 12.0
    while time.time() < deadline - 200.0:
        # draw a perturbed start point
        base = best_seq if np.random.rand() < 0.75 else uniform
        noise = 0.04 * np.random.randn(n) * base
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)

        sub_deadline = min(deadline - 200.0, time.time() + min_rest_time)
        cand_seq, cand_score = _adam_descent(start, sub_deadline,
                                             start_time, L_fft, L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best after Adam restart score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Projected sub‑gradient refinement (bulk of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        sub_deadline = deadline - 60.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             sub_deadline, start_time,
                                             best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad final improve score={sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Directed hill‑climbing polish (uses active‑set information)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = _targeted_hill_climb(best_seq, deadline,
                                            start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improve score={sc_hc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
