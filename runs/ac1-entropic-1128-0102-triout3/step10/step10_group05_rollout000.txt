"""
A multi‑stage optimiser for the self‑convolution L∞ problem.

The routine works with a larger support (default 8192 points) and
combines three complementary ideas:

1️⃣  *Global shape search* – a quick sweep over several parametric
    families (geometric, power‑law, symmetric beta‑type) to obtain a
    very good seed.

2️⃣  *Continuation Adam* – minimises a smooth approximation of the
    max‑convolution via the log‑sum‑exp (soft‑max) trick.
    The smoothing parameter α is decreased gradually (continuation)
    while an Adam optimiser with simplex projection updates the
    probability vector.

3️⃣  *Local stochastic refinement* – a simple log‑space mutation
    (Gaussian noise on ``log a``) plus an occasional greedy mass‑transfer
    sweep.  Both steps respect the simplex constraints and keep the
    sequence non‑negative.

All stages are time‑controlled; the best sequence seen so far is
returned as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:   # all zero – fall back to uniform
        w = np.full_like(v, 1.0 / n)
    return w


def _score(a: np.ndarray, L: int) -> float:
    """Return the objective 2·n·max(conv) for a (assumed ∑a=1)."""
    conv = _conv_self(a, L)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Parametric families for the initialisation stage
# ----------------------------------------------------------------------
def _exp_sequence(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i (r∈(0,1))."""
    if not (0.0 < r < 1.0):
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    return a


def _power_sequence(n: int, p: float) -> np.ndarray:
    """Power law a_i ∝ (i+1)**p, p may be negative."""
    idx = np.arange(1, n + 1, dtype=float)
    a = np.power(idx, p, where=idx > 0, out=np.ones_like(idx))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _sym_beta_sequence(n: int, p: float) -> np.ndarray:
    """Symmetric beta‑type shape a_i ∝ ((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=float) + 1.0
    a = np.power(i * (n - i + 1.0), p, out=np.ones_like(i))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Continuation Adam optimiser (smooth max)
# ----------------------------------------------------------------------
def _adam_continuation(a0: np.ndarray,
                       n: int,
                       L: int,
                       deadline: float,
                       start_time: float,
                       alphas,
                       iters_per_alpha: int = 25000,
                       lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp approximation of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(best_a, L)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            it += 1
            # ----- smoothed max -------------------------------------------------
            conv = _conv_self(a, L)                       # length 2n‑1
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)               # soft‑max weights
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            w_rev = w[::-1]

            # ----- gradient of the smooth max ------------------------------------
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update ----------------------------------------------------
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it + 1.0)
            a -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto the simplex
            a = _project_simplex(a)

            # keep track of the best raw (non‑smooth) score
            cur_score = _score(a, L)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, α={alpha:.3g}, score={cur_score:.6f}")

    return best_a, best_score

# Local stochastic refinement – log‑space mutation + greedy sweep
# ----------------------------------------------------------------------
def _local_refine(a0: np.ndarray,

                  L: int,

    rng = np.random.default_rng()
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a, L)

    # pre‑allocate an array for the greedy step to avoid re‑allocations
    greedy_interval = 5000
    greedy_counter = 0

        if time.time() >= deadline:

        # 1) log‑space Gaussian mutation
        # ------------------------------------------------------------------

        log_a = np.log(a + 1e-18)

        cand = np.exp(cand_log)
        cand_sum = cand.sum()
        if cand_sum == 0.0 or np.isnan(cand_sum):

        else:
            cand /= cand_sum

        cand_score = _score(cand, L)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] mutate improve: "
                  f"it={it:,}, sigma={sigma:.3e}, score={cand_score:.6f}")
            continue  # keep the better candidate

        # ------------------------------------------------------------------
        # 2) occasional deterministic greedy reduction (cheap version)
        # ------------------------------------------------------------------
        greedy_counter += 1
        if greedy_counter >= greedy_interval:
            greedy_counter = 0
            # find current max convolution index
            conv = _conv_self(a, L)
            k_max = int(conv.argmax())
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[k_max - idx]
            src = idx[np.argmax(contrib)]

            # partner values for src in current max
            partner_src = a[k_max - src] if 0 <= k_max - src < n else 0.0

            # choose a destination with minimal partner value
            partners = a[k_max - idx] if np.all(0 <= k_max - idx) and np.all(k_max - idx < n) else np.zeros_like(idx)
            dst_candidates = idx[partners == partners.min()]
            if dst_candidates.size == 0:
                dst = rng.integers(n)
                while dst == src:
                    dst = rng.integers(n)
            else:
                dst = rng.choice(dst_candidates)
            partner_dst = a[k_max - dst] if 0 <= k_max - dst < n else 0.0

            delta = partner_src - partner_dst
            if delta <= 0.0:
                eps = min(a[src], 0.02 * a[src])
            else:
                eps_target = (conv.max() - np.partition(conv, -2)[-2]) / (2.0 * delta + 1e-18)
                eps = min(a[src], max(0.0, 0.9 * eps_target))
                if eps < 1e-12:
                    eps = min(a[src], 0.02 * a[src])

            if eps > 0.0:
                a_old = a.copy()
                a[src] -= eps
                a[dst] += eps
                a = np.maximum(a, 0.0)
                a /= a.sum()
                # recompute score after greedy move
                new_score = _score(a, L)
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                          f"it={it:,}, eps={eps:.3e}, score={new_score:.6f}")

    return best_a, best_score

    if time.time() < deadline:
        phase1_deadline = min(deadline, time.time() + max(30.0, 0.20 * (deadline - time.time())))
        alphas = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
        a_adam, score_adam = _adam_continuation(
            best_a, n, L, phase1_deadline, start_time,
            alphas=alphas,
            iters_per_alpha=30000,
            lr0=0.6,
        )
        if score_adam < best_score:
            best_score, best_a = score_adam, a_adam
            print(f"[{time.time() - start_time:.2f}s] after Adam best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – local stochastic refinement (log‑space mutation + greedy)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = deadline
        a_refined, score_refined = _local_refine(
            best_a, n, L, phase2_deadline, start_time,
            max_mut_iters=500_000,
        )
        if score_refined < best_score:
            best_score, best_a = score_refined, a_refined
            print(f"[{time.time() - start_time:.2f}s] after refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = np.full(n, 1.0 / n)

    final_obj = _score(best_a, L)
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_a]
