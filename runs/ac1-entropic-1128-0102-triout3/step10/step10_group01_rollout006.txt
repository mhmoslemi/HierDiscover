def _softmax_weights(conv, beta):
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    sum_exps = exps.sum()
    w = exps / sum_exps
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4096 # could adjust based on time

    # Precompute x positions
    x = (np.arange(n) + 0.5) / n

    # Parameter grids
    p_vals = np.linspace(0.2, 2.0, 19)  # 0.2,0.3,...2.0
    q_vals = np.linspace(0.2, 2.0, 19)
    best_seq = None
    best_score = float('inf')
    eps=1e-30

    for p in p_vals:
        for q in q_vals:
            w = (x ** (p - 1)) * ((1 - x) ** (q - 1)) + eps
            a = w / w.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start:.2f}s] p={p:.2f}, q={q:.2f} score={score:.6f}")

    # Add arcsine baseline (p=q=0.5) maybe already included.
    # Also include previously found sequence if any
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            # normalize sum to 1
            if starter.sum() > 1e-12:
                starter = starter / starter.sum()
                cur_score = _objective(starter)
                print(f"[{time.time() - start:.2f}s] global starter score={cur_score:.6f}")
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
    except Exception:
        pass

    # If best_seq is None fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # Quick check time remaining
    remaining = deadline - time.time()
    if remaining > 20.0: # proceed to gradient descent

        # parameters for gradient descent
        a = best_seq.copy()
        best_a = a.copy()
        best_obj = best_score

        # hyperparameters
        eta0 = 0.2
        beta0 = 5.0
        beta_inc = 0.005  # increase per iteration
        beta_max = 80.0
        momentum = np.zeros_like(a)

        it = 0
        while time.time() < deadline:
            it += 1
            # compute convolution
            conv = _conv_self(a)
            # compute exact objective for tracking
            cur_obj = _objective(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] GD iter={it} new best={cur_obj:.6f}")

            # compute beta
            beta = min(beta_max, beta0 + beta_inc * it)

            # compute softmax weights
            m = conv.max()
            exps = np.exp(beta * (conv - m))
            w = exps / exps.sum()

            # compute gradient via correlation
            # compute FFT length
            L = _next_pow2(2 * n - 1)
            Fa = np.fft.fft(a, n=L)
            Fw = np.fft.fft(w, n=L)
            grad_full = np.fft.ifft(Fa * np.conj(Fw))
            grad = 2.0 * np.real(grad_full[:n])

            # momentum update
            momentum = 0.9 * momentum + 0.1 * grad

            # step size decreasing
            eta = eta0 / math.sqrt(it)

            a = a - eta * momentum

            # project onto simplex
            a = _project_simplex(a)

        # after loop, compare best_a
        if best_obj < best_score:
            best_score = best_obj
            best_seq = best_a.copy()
            print(f"[{time.time() - start:.2f}s] GD final best score={best_score:.6f}")

    # final annealing small moves if time left
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        steps = int(remaining * 5000)  # approx 5000 moves per second
        for it in range(steps):
            if time.time() >= deadline:
                break
            # choose source index weighted by a
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            # propose moving eps fraction
            eps = a[src] * rng.random() * 0.1 # up to 10%
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_score = _objective(a_prop)
            if new_score < cur_score or rng.random() < 0.001:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start:.2f}s] anneal step={it} best={new_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start:.2f}s] Final anneal improve score={best_score:.6f}")

    # Ensure non-negative and normalized
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] final objective = {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. **Beta‑family sweep** –  Try densities of the form
       a_i ∝ x_i^{p‑1}·(1‑x_i)^{q‑1},
   where  x_i = (i+½)/n  and (p,q) are explored on a grid.
   The best candidate from this cheap exhaustive search is kept.

2. **Smooth‑max gradient descent** –  Starting from the best candidate,
   minimise a smooth approximation of the ‑max‑convolution objective:
       ϕβ(a)=2·n·∑_k w_k·(a*a)_k ,   w_k∝exp(β·(a*a)_k).
   The gradient is obtained via a correlation computed with FFTs.
   After each step the iterate is projected onto the probability simplex
   (non‑negative, Σa=1).  Momentum (a simple exponential moving average)
   and a decreasing step‑size are used.

3. **Very light simulated‑annealing polish** –  Tiny random mass‑transfer
   moves are accepted if they improve the exact objective or with a
   small annealing probability.

The whole search respects the 1000 s wall‑time limit.  If time runs out
early the best sequence found so far (at least the grid‑search result)
is returned.  All numbers are non‑negative and the sequence is normalised
to sum to 1, which makes the objective scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the competition harness:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑objective step sequence within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Grid search over the Beta‑family (p,q) – cheap but surprisingly
    #    effective at locating a good shape.
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑bin positions in (0,1)

    p_vals = np.linspace(0.20, 2.00, 19)      # 0.20,0.30,…,2.00
    q_vals = np.linspace(0.20, 2.00, 19)

    best_seq = None
    best_score = float("inf")
    eps = 1e-30   # prevents exact zeros when p or q < 1

    for p in p_vals:
        for q in q_vals:
            # beta‑density (unnormalised) on the grid
            w = (x ** (p - 1)) * ((1.0 - x) ** (q - 1)) + eps
            a = w / w.sum()          # normalise to sum = 1 (scale‑invariant)
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] p={p:.2f} q={q:.2f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) If a globally stored construction exists, compare it.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 1e-12:
                starter = starter / starter.sum()
                cur_score = _objective(starter)
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
    except Exception:
        pass

    if best_seq is None:
        # Safe fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3) Smooth‑max projected gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:                     # leave a few seconds for polishing
        a = best_seq.copy()
        best_a = a.copy()
        best_obj = best_score

        # hyper‑parameters
        eta0 = 0.2          # base step size
        beta0 = 5.0         # initial smooth‑max temperature
        beta_inc = 0.004    # linear increase per iteration
        beta_max = 80.0
        momentum = np.zeros_like(a)
        it = 0

        while time.time() < deadline:
            it += 1
            # ----- exact objective for bookkeeping -----
            cur_obj = _objective(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] GD iter={it:,} new best={cur_obj:.6f}")

            # ----- convolution (full) -----
            conv = _conv_self(a)

            # ----- smooth‑max weights -----
            beta = min(beta_max, beta0 + beta_inc * it)
            shift = conv.max()                     # avoid overflow
            w = np.exp(beta * (conv - shift))
            w /= w.sum()                           # normalised weights (sum = 1)

            # ----- gradient = 2 * (a ⋆ w) -----
            L = _next_pow2(2 * n - 1)
            Fa = np.fft.fft(a, n=L)
            Fw = np.fft.fft(w, n=L)
            grad_full = np.fft.ifft(Fa * np.conj(Fw))
            grad = 2.0 * np.real(grad_full[:n])

            # ----- momentum & step -----
            momentum = 0.9 * momentum + 0.1 * grad
            step = eta0 / math.sqrt(it)
            a = a - step * momentum

            # ----- projection onto the simplex -----
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # 4) Light simulated‑annealing polish (mass transfers)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy() if 'best_a' in locals() else best_seq.copy()
        cur_score = _objective(a)
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        # number of attempted moves – roughly a few thousand per second
        max_moves = int(remaining * 4500)

        for it in range(1, max_moves + 1):
            if time.time() >= deadline:
                break

            # pick a source index roughly proportional to its mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            # move a random fraction (up to 15 %) of the source mass
            eps_move = a[src] * rng.random() * 0.15
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            new_score = _objective(a_prop)
            if new_score < cur_score:
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal step={it:,} best={new_score:.6f}")
            else:
                # very low chance to accept uphill moves (helps escape tiny traps)
                if rng.random() < 1e-4:
                    a = a_prop
                    cur_score = new_score

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improvement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] final objective = {best_score:.6f}")
    return [float(v) for v in best_seq]
