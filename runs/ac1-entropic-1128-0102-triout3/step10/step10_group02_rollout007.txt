def _smooth_descent_multi(a0, n, deadline, start_time):

    best_a = a.copy()

    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    alphas = np.geomspace(0.5, 2000.0, num=40)
    it = 0

    for alpha in alphas:
        while time.time() < deadline - 0.5 and it < max_iterations:
            it += 1
            # convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

def _annealing_guided(a0, n, deadline, start_time, cur_best_score, cur_best_seq,
                      start_temp=5e-2, end_temp=1e-7, steps=400_000):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # compute convolution and active set once every few steps
        if it % 20 == 0:
            conv = _conv_self(a)
            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-12)[0]
            # compute gradient contributions for active set
            grad = np.zeros(n)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1, dtype=int)
                grad[idx] += a[k - idx]
            grad = 2.0 * grad / len(active)
            # source indices are those with large gradient (bad)
            # destination indices with small gradient (good)
            src_candidates = np.where(grad > np.percentile(grad, 90))[0]
            dst_candidates = np.where(grad < np.percentile(grad, 10))[0]
            if src_candidates.size == 0 or dst_candidates.size == 0:
                src = rng.choice(n, p=a)
                dst = rng.integers(n)
            else:
                src = rng.choice(src_candidates)
                dst = rng.choice(dst_candidates)
        else:
            # fallback random for speed
            src = rng.choice(n, p=a)
            dst = rng.integers(n)

        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.4   # up to 40% of src
        a_prop = a.copy()

"""
Improved search for the step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of

1️⃣  A randomised start (uniform or a jittered copy of the previous best).
2️⃣  Smooth‑max mirror descent with a much larger α schedule and many more
    iterations (exponentiated‑gradient / Adam‑style updates).
3️⃣  Projected sub‑gradient descent that uses **all** near‑max convolution
    indices (the whole active set) instead of a single argmax.
4️⃣  Gradient‑guided simulated‑annealing: mass is moved from the index that
    contributes most to the active constraints to the one that contributes
    least, with a geometric temperature schedule.
5️⃣  A final tiny hill‑climbing pass that moves a minute amount of mass from
    the worst to the best index and keeps only improvements.

All helpers are top‑level, the wall–clock is watched, and nothing is
written to disk.  The code returns a list of non‑negative heights (they sum
to 1, which is indifferent for the objective because the evaluation is
scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (larger α schedule)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max approximation of the maximum convolution.
    Uses exponentiated‑gradient updates.  A geometric schedule for the
    temperature α goes up to 2000 and the iteration budget is larger.
    """
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # α grows from 0.5 to 2000 (40 values, geometrically spaced)
    alphas = np.geomspace(0.5, 2000.0, num=40)

    it = 0
    max_iterations = 500_000          # safety cap

    for alpha in alphas:
        # inner loop: keep going until time runs out or we hit the iteration cap
        while time.time() < deadline - 0.3 and it < max_iterations:
            it += 1

            # 1) self‑convolution (a*a)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights w_k ∝ exp(alpha*conv_k)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)            # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            grad_full = np.fft.irfft(np.fft.rfft(w, n=L_grad) *
                                     np.fft.rfft(a_rev, n=L_grad), n=L_grad)
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step (mirror descent on the simplex)
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)

            # protect against exact zeros and re‑normalise
            a = np.maximum(a, 1e-20)
            a /= a.sum()

            # occasional true‑objective check
            if it % 1000 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.3:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (full active set)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses the whole active set (all indices whose convolution is within a tiny
    tolerance of the maximum) to obtain a better descent direction.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.4
    it = 0
    L = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # ---- active set -------------------------------------------------
        # use an absolute tolerance – the values are already tiny (~1e‑4)
        tol = 1e-12
        active = np.where(conv >= cur_max - tol)[0]
        if active.size == 0:               # fallback (should never happen)
            active = np.array([conv.argmax()], dtype=int)

        # ---- sub‑gradient = mean of gradients of the active constraints
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]

        grad = 2.0 * grad / len(active)

        # ---- momentum step + projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : gradient‑guided simulated annealing
# ----------------------------------------------------------------------
def _annealing_guided(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      cur_best_score: float,
                      cur_best_seq: np.ndarray,
                      start_temp: float = 5e-2,
                      end_temp: float = 1e-7,
                      steps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Tiny annealer that preferentially moves mass from indices that
    *increase* the active constraints to indices that *decrease* them.
    """
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    # store the current active gradient (re‑computed every 20 iterations)
    grad = np.zeros(n)

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # recompute the gradient of the active set only occasionally – cheap
        if it % 20 == 0:
            conv = _conv_self(a)
            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-12)[0]
            grad.fill(0.0)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1, dtype=int)
                grad[idx] += a[k - idx]
            grad = 2.0 * grad / len(active)

            # pick source from the 10 % worst (largest gradient)
            # and destination from the 10 % best (smallest gradient)
            src_pool = np.where(grad > np.percentile(grad, 90))[0]
            dst_pool = np.where(grad < np.percentile(grad, 10))[0]
            if src_pool.size > 0 and dst_pool.size > 0:
                src = rng.choice(src_pool)
                dst = rng.choice(dst_pool)
            else:
                src = rng.choice(n, p=a)
                dst = rng.integers(n)
        else:
            # cheap fallback – still biased towards large a entries
            src = rng.choice(n, p=a)
            dst = rng.integers(n)

        if src == dst or a[src] < 1e-14:
            continue

        # move up to 40 % of the source mass
        eps = a[src] * rng.random() * 0.40
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase‑4 : tiny hill‑climbing (deterministic mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Deterministic local optimisation: repeatedly move a very small ε from the
    index with the largest active‑set gradient to the one with the smallest.
    Stops when no improvement is possible or time runs out.
    """
    a = a0.copy()
    best_score = cur_best_score
    best_a = a.copy()
    L = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        conv = _conv_self(a, L=L)
        cur_max = conv.max()
        active = np.where(conv >= cur_max - 1e-12)[0]

        # gradient of the active set
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        grad = 2.0 * grad / len(active)

        i_max = np.argmax(grad)
        i_min = np.argmin(grad)

        if grad[i_max] <= grad[i_min] + 1e-15:
            break                     # gradients already balanced

        eps = min(a[i_max], 1e-5)      # a tiny amount
        if eps <= 0:
            break

        a_prop = a.copy()
        a_prop[i_max] -= eps
        a_prop[i_min] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        if new_score < best_score:
            best_score = new_score
            best_a = a_prop.copy()
            a = a_prop
        else:
            break                     # cannot improve further

    return best_a, best_score
