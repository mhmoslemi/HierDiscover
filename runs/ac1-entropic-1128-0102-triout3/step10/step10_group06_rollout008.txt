"""
Propose a non‑negative step function that (almost) minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from the best known construction ``height_sequence_1`
   (if present) and a small pool of random Dirichlet draws. Keep the best.
2. **LP‑based active‑set improvement** – repeatedly
   a) identify the shift(s) where the self‑convolution attains (or almost attains)
      its maximum,
   b) solve a tiny linear programme that maximises the total mass of a candidate
      vector `b` while keeping the cross‑convolution `a * b` below the current
      peak on those active shifts,
   c) mix the current vector `a` with the normalised LP solution `b̂` using a
      line‑search over the mixing weight `t`.  Accept any strict improvement.
   The LP only contains the *active* constraints (typically ≤ 5 rows), so it is
   solved in a fraction of a second with ``scipy.optimize.highs``.
3. **Fine‑grained mirror‑descent** – after the LP phase we spend the remaining
   time running the multi‑gradient exponentiated‑gradient method from the reference
   implementation (with a slightly larger active‑set) to polish the solution.
4. **Deterministic greedy peak reduction** – a final local search that moves
   a fraction of the coefficient responsible for the current peak towards the
   smallest coefficient, accepting only strict improvements.
5. **Normalisation & safety** – clip negatives, enforce a unit sum and return the
   best sequence found.

The whole routine respects the supplied 1000 s budget, prints timing and
intermediate scores and finally returns a list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_cross(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*b (length 2·len(a)‑1, assuming len(a)==len(b))."""
    Af = np.fft.rfft(a, n=fft_len)
    Bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Af * Bf, n=fft_len)
    return conv[: a.shape[0] + b.shape[0] - 1]


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Multi‑gradient exponentiated‑gradient (same as reference)
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] summed over the active shifts."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eta0: float = 0.6,
    delta_frac: float = 0.001,
    top_k: int = 5,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Multi‑ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ----- select active shifts -----
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ----- compute sub‑gradient -----
        g = _multi_gradient(a, active)

        # ----- step size schedule (diminishing) -----
        eta = eta0 / math.sqrt(t)

        # ----- exponentiated update -----
        a = a * np.exp(-eta * g)

        # ----- re‑normalise onto the simplex -----
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # protect against exact zero (helps numerical stability)
        a = np.maximum(a, 1e-15)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy local improvement (same as reference)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves."""
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # modest fraction of the source mass
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine uses an LP‑active‑set improvement, followed by a
    multi‑gradient exponentiated‑gradient polish and a final greedy pass.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # try to keep the same length as the reference solution
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)
        n = seq0.shape[0]
    except Exception:
        n = 4800
        seq0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Normalise the starting vector
    # ------------------------------------------------------------------
    seq0 = np.clip(seq0, 0.0, None)
    total = seq0.sum()
    if total > 0:
        seq0 = seq0 / total
    else:
        seq0 = np.full(n, 1.0 / n)

    best_seq = seq0.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[{time.time() - start_time:6.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Add a few random Dirichlet restarts (cheap)
    # ------------------------------------------------------------------
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            cand = _dirichlet_start(n, conc=conc, rng=rng)
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] Random start improved: {sc:.6f}")

    # ------------------------------------------------------------------
    # LP‑based active‑set improvement (major part of the budget)
    # ------------------------------------------------------------------
    try:
        import scipy.sparse as sp
        import scipy.optimize as opt
        have_scipy = True
    except Exception:
        have_scipy = False
        print("SciPy not available – skipping LP phase.")

    fft_len = _next_pow2(2 * n - 1)

    lp_deadline = start_time + budget_s * 0.65   # allocate ~65 % to LP phase
    lp_iter = 0

    while have_scipy and time.time() < lp_deadline:
        lp_iter += 1

        # current convolution and maximal peak
        conv_a2 = _conv_self(best_seq, fft_len)
        cur_max = conv_a2.max()
        # active shifts (within 0.1 % of the maximum)
        delta = max(1e-12, 0.001 * cur_max)
        active = np.where(conv_a2 >= cur_max - delta)[0]

        # keep only a handful of the very top shifts (avoid huge LP)
        if active.size > 8:
            topk = 5
            part = np.argpartition(-conv_a2[active], topk - 1)[:topk]
            active = active[part]

        # build sparse constraint matrix A_ub * b <= cur_max
        rows = []
        cols = []
        data = []
        for r, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            cols_row = np.arange(lo, hi + 1, dtype=int)
            rows.extend([r] * len(cols_row))
            cols.extend(cols_row.tolist())
            data.extend(best_seq[cols_row].tolist())
        if not rows:          # safety – should never happen
            break
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), cur_max)

        # variable bounds – a modest upper bound keeps the LP bounded
        ub_bound = 5.0
        bounds = [(0.0, ub_bound) for _ in range(n)]

        # maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n)

        # solve LP
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # a tighter bound may rescue the solve
            ub_bound = 1.0
            bounds = [(0.0, ub_bound) for _ in range(n)]
            res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                print(f"[{time.time() - start_time:6.2f}s] LP failed at iter {lp_iter}, breaking.")
                break

        b_vec = res.x
        sum_b = b_vec.sum()
        if sum_b <= 1e-12:
            print(f"[{time.time() - start_time:6.2f}s] LP gave zero mass, stopping.")
            break

        # normalise the LP solution to a probability vector
        b_norm = b_vec / sum_b

        # pre‑compute the three convolutions needed for mixing
        conv_ab = _conv_cross(best_seq, b_norm, fft_len)
        conv_b2 = _conv_self(b_norm, fft_len)

        # line‑search over mixing weight t ∈ (0, 0.5]
        improved = False
        for t in np.linspace(0.02, 0.5, 25):
            a_new = (1.0 - t) * best_seq + t * b_norm
            # combine the three pre‑computed convolutions
            mixed_conv = (
                (1.0 - t) ** 2 * conv_a2
                + 2.0 * t * (1.0 - t) * conv_ab
                + t ** 2 * conv_b2
            )
            new_max = mixed_conv.max()
            new_score = 2.0 * n * new_max
            if new_score < best_score - 1e-9:
                best_seq = a_new
                best_score = new_score
                improved = True
                print(
                    f"[{time.time() - start_time:6.2f}s] LP improve iter {lp_iter}, t={t:.3f}, score={new_score:.6f}"
                )
                break
        if not improved:
            # no improvement found – shrink the bound to keep LP bounded and retry later
            ub_bound = max(1.0, ub_bound * 0.5)

    # ------------------------------------------------------------------
    # Polishing with multi‑gradient exponentiated‑gradient (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = time.time() + (deadline - time.time()) * 0.90
        best_seq, best_score = _expgrad_multi_opt(
            best_seq,
            polish_deadline,
            start_time,
            fft_len,
            eta0=0.5,
            delta_frac=0.001,
            top_k=10,
        )
        print(f"[{time.time() - start_time:6.2f}s] After polishing score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic greedy peak reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_seq = final_seq
            best_score = final_score
            print(
                f"[{time.time() - start_time:6.2f}s] After greedy score = {final_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final safety checks and reporting
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    # return as a plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
