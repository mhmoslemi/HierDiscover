"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm overview
------------------
1. **Problem size** – use a larger support (`n = 8192`).  The bound
   improves as the discretisation gets finer.

2. **Initial candidates** – generate a few deterministic shapes
   (uniform, linear, √‑law, exponential, Beta‑law).  Keep the one with
   the smallest self‑convolution peak (computed by a fast FFT‑based
   convolution).

3. **Phase‑2: soft‑max exponentiated‑gradient** – a mirror‑descent
   optimiser on the true max‑convolution.  Instead of using only the
   worst index, the gradient is a weighted sum over the `K` largest
   convolution entries (default `K = 5`).  The step‑size follows
   `η_t = η0 / sqrt(t)`.  After each multiplicative update the vector
   is projected onto the probability simplex.

4. **Phase‑3: deterministic greedy peak‑flattening** – repeatedly move
   a small amount of mass from the index that most contributes to the
   current peak to an index that contributes the least, updating the
   full convolution incrementally (O(n) per move).

5. **Late‑stage random restarts** – whenever time permits, apply a tiny
   Gaussian perturbation to the current best candidate and run a short
   gradient burst; keep any improvement.

All optimisation is performed inside the allotted 1000 s budget.  The
final sequence is normalised to sum to 1 and returned as a plain Python
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # fallback – should not happen for a non‑empty vector
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Scoring helper (fast, homogeneous version of evaluate_sequence)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, Lfft: int) -> float:
    """
    Return the objective value for a vector a that already satisfies
    sum(a) == 1.  The objective is 2 * n * max(conv) (since sum == 1).
    """
    conv = _conv_self(a, Lfft)
    max_conv = conv.max()
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Phase‑2 – Soft‑max exponentiated‑gradient
# ----------------------------------------------------------------------
def _run_exp_gradient_multi(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            eta0: float = 0.8,
                            K: int = 5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with a soft‑max over the K largest convolution entries.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    Lfft = _next_pow2(2 * n - 1)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, Lfft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices of the K largest entries (including the absolute maximum)
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(-conv, K)[:K]

        # accumulated gradient from the selected indices
        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # derivative of conv[k] w.r.t a[idx] is 2 * a[k - idx]
            grad[idx] += 2.0 * a[k - idx]

        # learning rate
        eta = eta0 / math.sqrt(it)

        # multiplicative update with clipping to avoid overflow
        exponent = -eta * grad
        exponent = np.clip(exponent, -50.0, 50.0)
        a = a * np.exp(exponent)

        # project back onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Greedy flattening: move a tiny amount of mass from the index that most
    contributes to the current maximum to an index that contributes the
    least.  The convolution is updated incrementally.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest value (used to estimate useful step size)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # choose destination with the smallest partner value
        partner_vals = a[max_idx - idx]
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            # try to close the gap between max and second‑max
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score

    initial_candidates = []

    # Uniform
    initial_candidates.append(np.full(n, 1.0 / n))

    # Linear decreasing (a_i ∝ n‑i)
    lin = np.arange(n, 0, -1, dtype=float)
    initial_candidates.append(lin / lin.sum())

    # sqrt‑law (a_i ∝ 1/√(i+1))
    sqrt_shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    initial_candidates.append(sqrt_shape / sqrt_shape.sum())

    # Exponential decay (q close to 1)
    for q in (0.9995, 0.9998, 0.9999):
        exp_shape = (1 - q) * (q ** np.arange(n, dtype=float))
        initial_candidates.append(exp_shape / exp_shape.sum())

    # Beta‑law (a_i ∝ (i+1)^{α‑1} (n‑i)^{β‑1})
    def beta_shape(alpha, beta):
        i = np.arange(1, n + 1, dtype=float)
        shape = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
        return shape / shape.sum()
    for a, b in ((0.5, 0.5), (0.8, 0.8), (1.2, 1.2)):
        initial_candidates.append(beta_shape(a, b))

    best_seq = None
    best_score = float("inf")

    # Very short gradient runs to pick a promising start
    for cand in initial_candidates:
        if time.time() >= deadline:
            break
        # evaluate quickly
        scr = _score(cand, n, Lfft)
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] init cand score={scr:.6f}")

        # a few seconds of soft‑max gradient on each candidate
        temp_deadline = time.time() + min(3.0, (deadline - time.time()) * 0.02)
        seq, scr = _run_exp_gradient_multi(cand, n, temp_deadline,
                                           start_time, eta0=0.9, K=3)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] init grad improve score={scr:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 – long soft‑max exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + remaining * 0.60
        seq, scr = _run_exp_gradient_multi(best_seq, n, phase2_deadline,
                                           start_time, eta0=0.6, K=5)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] long exp‑grad improve score={scr:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 – deterministic greedy reduction (use all time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_greedy_reduction(best_seq, n, deadline, start_time,
                                         best_score, max_iters=8_000_000)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve score={scr:.6f}")

    # ------------------------------------------------------------------
    # Tiny random‑restart polishing if a few seconds remain.
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        pert = best_seq + 0.01 * np.random.randn(n)
        pert = np.clip(pert, 0.0, None)
        pert /= pert.sum() if pert.sum() > 0 else 1.0

        tmp_deadline = time.time() + 2.0
        seq, scr = _run_exp_gradient_multi(pert, n, tmp_deadline,
                                           start_time, eta0=0.5, K=3)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] restart improve score={scr:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (any positive scaling works, but we keep sum=1)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return a plain Python list
    return [float(x) for x in best_seq]
