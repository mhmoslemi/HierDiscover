"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Rich initial pool** – deterministic shapes (uniform, exponential,
   discrete “inverse‑sqrt” inspired by the beta(½,½) distribution) and a
   collection of random Dirichlet draws.  The best of them (by the exact
   objective) is kept.

2. **Simulated‑annealing local search** – repeatedly move a small amount
   of mass from one coordinate to another, accepting moves that improve
   the score and, with a temperature‑dependent probability, also moves
   that worsen it.  The temperature slowly cools, allowing the method to
   escape shallow local minima.  The convolution is evaluated by FFT,
   which makes each iteration very cheap (≈ 0.1 ms for n≈4800).

3. **Short exponentiated‑gradient polishing** – the final candidate is
   refined with a few iterations of the multi‑gradient exponentiated‑
   gradient scheme from the reference solution (a convex‑surrogate of the
   max‑convolution).  This drives the remaining peaks down.

The routine respects a user supplied time budget (default 1000 s) and
returns the best sequence found (normalised to sum = 1, all entries ≥ 0).
All helper functions are defined at top‑level, no closures or file I/O
are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _score_from_conv(conv: np.ndarray, sum_a: float, n: int) -> float:
    """Exact value of evaluate_sequence (without clamping)."""
    return float(2.0 * n * conv.max() / (sum_a ** 2))


def _exact_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Convenient wrapper that returns the real evaluate_sequence score."""
    conv = _conv_self(a, fft_len)
    return _score_from_conv(conv, a.sum(), n)


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def _uniform_start(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _exponential_start(n: int, r: float = 0.985) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    v = np.array([r ** i for i in range(n)], dtype=float)
    return v / v.sum()


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _inverse_sqrt_start(n: int) -> np.ndarray:
    """
    Analytic shape inspired by the beta(½,½) distribution:
    a_i ∝ 1/sqrt(i+1) + 1/sqrt(n-i).
    """
    i = np.arange(n)
    w = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(np.maximum(n - i, 1))
    return w / w.sum()


def _beta05_start(n: int) -> np.ndarray:
    """
    Discrete version of the Beta(½,½) density on [0,1]:
    a_i ∝ 1 / sqrt( (i+0.5)*(n-i-0.5) ).
    """
    i = np.arange(n) + 0.5
    w = 1.0 / np.sqrt(i * (n - i))
    return w / w.sum()


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         fft_len: int,
                         n: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple SA that transfers a fraction of mass from a source index to a
    target index.  The temperature decays linearly with the elapsed
    fraction of the allocated time.
    """
    a = a0.copy()
    conv = _conv_self(a, fft_len)
    cur_score = _score_from_conv(conv, a.sum(), n)
    best_a = a.copy()
    best_score = cur_score

    # SA hyper‑parameters
    max_transfer_frac = 0.30   # at the start we may move up to 30 % of source mass
    min_transfer_frac = 0.005  # towards the end we move only a tiny fraction

    while time.time() < deadline:
        # linearly decreasing temperature in the range [1e‑4, 5e‑2]
        frac = (time.time() - start_time) / (deadline - start_time)
        temperature = (0.05 - 0.0001) * (1.0 - frac) + 0.0001
        # gradually shrink the allowed transfer size
        transfer_frac = (max_transfer_frac - min_transfer_frac) * (1.0 - frac) + min_transfer_frac

        # propose a move
        src = rng.integers(0, n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1  # ensure src != dst

        delta = a[src] * transfer_frac * rng.random()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # keep numerical sanity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s

        conv_new = _conv_self(a_new, fft_len)
        new_score = _score_from_conv(conv_new, a_new.sum(), n)

        # acceptance test
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temperature):
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:6.2f}s] SA improve: score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (same idea as the reference code)
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] obtained by summing gradients of active shifts."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       fft_len: int,
                       n: int,
                       eta0: float = 0.6,
                       delta_frac: float = 0.001,
                       top_k: int = 10) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution
    objective, using a multi‑gradient built from the top‑K shifts that are
    close to the current maximum.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # ----- active set -----
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            # keep only the K largest values
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ----- gradient -----
        g = _multi_gradient(a, active)

        # ----- step size -----
        eta = eta0 / math.sqrt(t)

        # ----- exponentiated update -----
        a = a * np.exp(-eta * g)

        # ----- re‑normalise -----
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        a = np.maximum(a, 1e-15)

    return best_a, best_score

    """
    Search for a non‑negative step function (list of floats) minimising

    """

    start_time = time.time()
    # leave a small safety margin for the final return
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------

    try:

    except Exception:
        pass

    starts.append(_inverse_sqrt_start(n))
