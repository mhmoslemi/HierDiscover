def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096

    # Initialize a as uniform or previous best
    a = np.full(n, 1.0 / n)   # sum=1

    # ensure a positive
    # maybe add small random noise
    # use simplex projection.

    # Let target sum S = sqrt(2*n) for internal scaling
    S = math.sqrt(2.0 * n)

    def scale_to_S(x):
        return x * (S / x.sum())

    # Scale a to have sum S for LP steps; but store external representation (maybe unscaled)
    a_scaled = scale_to_S(a)

    # Define max iterations while time left
    it = 0
    best_score = evaluate_sequence(a.tolist())   # evaluate current
    best_seq = a.copy()
    # Mix parameter:
    t = 0.5  # step size maybe? Could adapt.

    while time.time() < deadline - 10:
        it += 1
        # compute M = max(conv(a_scaled, a_scaled)) (full conv)
        conv_self = np.convolve(a_scaled, a_scaled)
        M = conv_self.max()
        # Solve LP for b >=0 such that conv(a_scaled, b) <= M
        # Solve using cvxpy
        b = cp.Variable(n, nonneg=True)
        conv_ab = cp.conv(a_scaled, b)  # type: cvxpy expression
        constraints = [conv_ab <= M]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True)  # maybe set max_iters
        if b.value is None:
            print(f"Iteration {it}: LP failed, break.")
            break
        b_val = np.maximum(b.value, 0)
        sum_b = b_val.sum()
        if sum_b <= 1e-12:
            print("sum b zero")
            break
        # Scale b to have sum S = sqrt(2n)
        g = b_val * (S / sum_b)
        # mixing update f <- (1 - alpha) f + alpha g
        # choose alpha maybe decreasing with iterations
        alpha = 0.5  # could adapt: maybe 0.3
        a_scaled = (1 - alpha) * a_scaled + alpha * g
        # Evaluate objective (original scale): convert back to sum=1
        a_norm = a_scaled * (1.0 / a_scaled.sum())
        cur_score = evaluate_sequence(a_norm.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_norm.copy()
            print(f"[{time.time()-start_time:.2f}s] LP it={it} score={cur_score:.6f}")
        # optional early break if improvement small
        if it >= 50:
            # maybe adjust alpha or stop
            pass
    # final result: best_seq
    # Ensure non-negative, sum to anything (norm to sum=1).
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)
    return best_seq.tolist()

        try:

"""
Propose a step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a two‑phase optimiser:

1. **LP‑based mixing** – following the reference iteration:
   given a current sequence `f` (scaled to Σf = √(2n)),
   compute M = max(conv(f,f)).
   Solve a linear program

       maximise   Σ b_j
       subject to  b_j ≥ 0
                   conv(f, b) ≤ M   (component‑wise)

   The optimum yields a vector `b` with a larger ℓ₁‑norm.
   After scaling `g = (√(2n)/Σb)·b` we obtain a new candidate with a
   strictly smaller bound on the convolution maximum.
   A convex combination `f←(1‑α)f+αg` (α≈0.5) guarantees monotone
   improvement of the objective.

2. **Fine‑grained smoothed gradient descent** – a soft‑max
   approximation of the max‑convolution is optimised for a short
   remaining budget using FFT‑based gradients and Euclidean projection
   onto the probability simplex.

Both phases are implemented with only NumPy, SciPy and CVXPY.
The routine respects the 1000 s time budget and returns the best
sequence found (normalised to sum = 1, all entries ≥ 0).
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the exact objective for a (sum≈1) sequence
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed ≈1)."""
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max approximation of the max)
# ----------------------------------------------------------------------
def _smooth_descent(start_a: np.ndarray,
                    deadline: float,
                    start_time: float,
                    L_fft: int,
                    L_grad: int) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed objective:
        f(a) = 2 * n * softmax_tau(conv(a,a))
    The soft‑max temperature τ is exponentially decayed.
    The iterate stays on the simplex via Euclidean projection.
    Returns the best sequence found during the run.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    tau = 0.10          # initial temperature
    tau_min = 1e-8
    tau_decay = 0.9997   # decay per iteration
    step0 = 0.25         # initial step length

    while time.time() < deadline:
        it += 1

        # --- convolution and soft‑max weights ---------------------------------
        conv = _conv_self(a, L=L_fft)          # length 2n‑1
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # sum(w)=1

        # --- gradient via FFT (grad_j = 2 * Σ_i a_i w_{i+j}) ------------------
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # --- projected step ----------------------------------------------------
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_simplex(a)

        # --- exact objective evaluation ----------------------------------------
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # --- temperature decay -------------------------------------------------
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm first performs an LP‑based mixing phase (reference
    iteration) and then a short smoothed‑gradient fine‑tuning phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for final return
    deadline = start_time + budget_s - 0.5

    # -------------------------- problem size ------------------------------
    n = 4096                     # number of coefficients
    S = math.sqrt(2.0 * n)       # normalisation constant (Σa = √(2n))
    L_fft = _next_pow2(2 * n - 1)   # FFT length for convolutions
    L_grad = _next_pow2(3 * n)      # FFT length for gradient steps

    # ----------------------- initial candidate ----------------------------
    # try to use the global starter if it exists
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)

    except Exception:
        a0 = np.full(n, 1.0 / n)

    # Normalise to Σa = S (required by the LP‑mixing theory)
    a = a0 / a0.sum() * S

    # Track the best sequence seen so far (stored in sum‑1 normalisation)
    best_seq = a.copy()
    best_score = evaluate_sequence((a / a.sum()).tolist())
    print(f"[{time.time()-start_time:.2f}s] Initial objective {best_score:.6f}")

    # --------------------- LP‑based mixing loop -------------------------
    # Parameters for the mixing phase
    max_lp_iters = 15            # safety cap; will stop earlier if time runs out
    mix_alpha = 0.5              # convex combination weight

    lp_iter = 0
    while time.time() < deadline - 5.0 and lp_iter < max_lp_iters:
        lp_iter += 1

        # current bound M = max(conv(a,a))
        conv_self = _conv_self(a, L=L_fft)
        M = conv_self.max()

        # ----- build and solve the LP: maximise Σb   s.t. conv(a,b) ≤ M -----
        # variable
        b = cp.Variable(n, nonneg=True)
        # constant (current a) – use a Parameter for speed
        a_param = cp.Parameter(n, value=a, mutable=False)
        conv_ab = cp.conv(a_param, b)               # length 2n‑1
        constraints = [conv_ab <= M]                # element‑wise inequality
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        # solve – ECOS works well for dense linear programs
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except Exception as e:
            print(f"[{time.time()-start_time:.2f}s] LP solve error: {e}")
            break

        if b.value is None:
            print(f"[{time.time()-start_time:.2f}s] LP failed to produce a solution.")
            break

        b_val = np.maximum(b.value, 0.0)
        sum_b = b_val.sum()
        if sum_b <= 1e-12:
            print(f"[{time.time()-start_time:.2f}s] LP produced zero‑sum vector.")
            break

        # Scale b to have the prescribed ℓ₁‑norm S
        g = b_val * (S / sum_b)

        # Convex combination (the classic reference update)
        a_new = (1.0 - mix_alpha) * a + mix_alpha * g

        # Evaluate the true objective (convert back to Σ=1)
        a_norm = a_new / a_new.sum()
        cur_score = evaluate_sequence(a_norm.tolist())

        # Keep the best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{time.time()-start_time:.2f}s] LP it={lp_iter} improved to {cur_score:.6f}")

        # Accept the new point for the next iteration
        a = a_new

    # ------------------------ final smooth descent -----------------------
    if time.time() < deadline - 1.0:
        # start from the best point found (normalised to sum=1 for the
        # smooth‑descent routine, which assumes Σ≈1)
        start_vec = best_seq / best_seq.sum()
        sub_deadline = deadline - 0.5
        cand_seq, cand_score = _smooth_descent(start_vec,
                                                sub_deadline,
                                                start_time,
                                                L_fft,
                                                L_grad)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq * S   # re‑scale to Σ=S (any scale works later)

    # --------------------------- final normalisation --------------------
    final_seq = best_seq.copy()
    final_seq = np.maximum(final_seq, 0.0)
    total = final_seq.sum()
    if total > 0:
        final_seq /= total
    else:
        final_seq = np.full(n, 1.0 / n)

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in final_seq]
