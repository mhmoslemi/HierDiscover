"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm (≈ 1000 s)
--------------------
1. **Diverse seeding** – a handful of handcrafted vectors (uniform,
   exponential, power‑law, a small random Dirichlet) and, if present,
   the previously best known construction ``height_sequence_1``.
   The best of these seeds (by the exact objective) is kept.

2. **Smooth Adam warm‑up** – a few thousand Adam steps on the
   log‑sum‑exp surrogate of the convolution peak.  This moves the
   iterate far away from pathological regions.

3. **Two‑stage multiplicative mirror‑descent** – an exponentiated‑gradient
   (mirror‑descent) loop that directly minimises the true
   max‑convolution objective.  The first half of the time uses a *large*
   tolerance (`delta_frac=1e‑2`) and many active shifts (`top_k=20`);
   the second half tightens to `delta_frac=1e‑3` and `top_k=5`.  The
   step‑size decays as η₀/√t.

4. **Simulated‑annealing fine‑tuning** – a cheap stochastic local
   search that repeatedly moves a small amount of mass from a high‑mass
   index to a low‑mass one.  Moves are accepted if they improve the
   score or with a probability exp(−Δ/τ) where the temperature τ
   decreases linearly to 0.  This phase helps to escape the tiny
   plateau that often remains after the deterministic phase.

All phases respect the overall time budget (default 1000 s).  The
routine finally returns the best‑scoring sequence found, normalised so
that its entries sum to 1 (the evaluation is scale‑invariant).  The code
uses only NumPy and the supplied ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Adam on smoothed (log‑sum‑exp) objective
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max‑convolution objective.
    Returns the best vector seen and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # schedule of smoothing parameters (more steps than the reference)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 5000          # a bit more than the reference

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max to a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            # convolution of a with reversed w gives ∂/∂a of soft‑max
            w_rev = w[::-1]
            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the true max‑convolution objective
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] by summing the gradients of the
    shifts listed in ``active_shifts``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   fft_len: int,
                   delta_start: float,
                   delta_end: float,
                   topk_start: int,
                   topk_end: int,
                   eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) using a *two‑stage* schedule.
    The tolerance ``delta_frac`` and the number of active shifts ``top_k``
    are linearly interpolated between the start and end values.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    # pre‑compute the total duration of this phase
    phase_len = deadline - start_time
    while time.time() < deadline:
        t += 1
        elapsed = time.time() - start_time
        prog = elapsed / phase_len  # ∈[0,1]

        # interpolate parameters
        delta_frac = delta_start + prog * (delta_end - delta_start)
        top_k = int(round(topk_start + prog * (topk_end - topk_start)))
        top_k = max(1, top_k)

        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{elapsed:6.2f}s] ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ----- active set -----
        thresh = cur_max - max(delta_frac * cur_max, 1e-14)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            # keep the top‑k shifts with largest values
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ----- gradient -----
        g = _multi_gradient(a, active)

        # ----- step size -----
        eta = eta0 / math.sqrt(t)

        # ----- exponentiated update -----
        a = a * np.exp(-eta * g)

        # ----- renormalise -----
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        a = np.maximum(a, 1e-15)

        # occasional precise evaluation
        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

# Simulated annealing fine‑tuning
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         max_move_frac: float = 0.05,
                         init_temp: float = 1e-2,
                         final_temp: float = 1e-5,
                         max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search.  At each step a small amount of
    mass is transferred from a relatively heavy index to a relatively
    light one.  The move is accepted if it improves the score or with
    probability exp(−Δ/τ) where the temperature τ decays linearly.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()

    total_time = deadline - start_time
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # linear temperature schedule
        elapsed = time.time() - start_time
        tau = init_temp + (final_temp - init_temp) * (elapsed / total_time)

        # pick a source index proportional to the current mass
        src = rng.choice(n, p=a / a.sum())
        # pick a destination uniformly among the other indices
        dst = rng.choice(n - 1)
        if dst >= src:
            dst += 1

        # amount to move (fraction of source mass)
        eps = max_move_frac * a[src] * rng.random()
        if eps <= 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate the candidate
        new_score = 2.0 * n * _conv_self(a_prop, _next_pow2(2 * n - 1)).max()

        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / max(tau, 1e-12)):
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{elapsed:6.2f}s] SA improve: iter={it:,}, score={cur_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _exponential_start(n: int, r: float = 0.985) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _powerlaw_start(n: int, p: float = 1.5) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    xs = np.arange(1, n + 1)
    vec = 1.0 / (xs ** p)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of several phases (warm‑up Adam, two‑stage
    exponentiated gradient, simulated annealing).  Intermediate progress
    is printed.  The function returns the best sequence found (list of
    Python floats) before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4800                               # same size as the reference code
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a rich set of starting points
    # --------------------------------------------------------------
    starts = []

    # 1) previous best, if available
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                seq = _upsample(seq, n)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) handcrafted starts
    starts.append(np.full(n, 1.0 / n))                 # uniform
    starts.append(_exponential_start(n, r=0.9995))    # very gentle decay
    starts.append(_exponential_start(n, r=0.985))     # moderate decay
    starts.append(_exponential_start(n, r=0.95))      # stronger decay
    starts.append(_powerlaw_start(n, p=1.2))
    starts.append(_powerlaw_start(n, p=1.8))

    # 3) random Dirichlet draws with different concentrations
    conc_params = [0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_params:
        for _ in range(4):
            starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # --------------------------------------------------------------
    # Evaluate raw starts and keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 1 – Adam warm‑up (≈10 % of the budget)
    # --------------------------------------------------------------
    warmup_deadline = start_time + budget_s * 0.10
    cand_seq, cand_score = _adam_optimize(
        best_seq, warmup_deadline, start_time, fft_len
    )
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 2 – Two‑stage exponentiated‑gradient polishing (≈70 %)
    # --------------------------------------------------------------
    expgrad_deadline = start_time + budget_s * 0.80
    # first half: looser tolerance, larger top‑k
    mid_deadline = start_time + budget_s * 0.45
    # run the two‑stage routine
    polished_seq, polished_score = _expgrad_phase(
        best_seq,
        deadline=expgrad_deadline,
        start_time=start_time,
        fft_len=fft_len,
        delta_start=1e-2,    # looser tolerance at the beginning
        delta_end=1e-3,      # tighter at the end
        topk_start=20,       # consider more active shifts initially
        topk_end=5,          # then focus on the true peak
        eta0=0.8,
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after ExpGrad score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 3 – Simulated‑annealing fine‑tuning (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _simulated_annealing(
            best_seq,
            deadline=deadline,
            start_time=start_time,
            max_move_frac=0.03,   # a bit smaller moves for fine tuning
            init_temp=1e-3,
            final_temp=1e-7,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after SA score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Final normalisation / safety checks
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    # Return plain Python list
    return [float(x) for x in best_seq.tolist()]
