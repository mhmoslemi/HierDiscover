def _subgradient_descent_active(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3*n - 2)
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)   # note L_grad maybe bigger than needed; but okay
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)   # sum is 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        # active mask
        # tolerance: use absolute tolerance relative to max  (maybe 1e-12)
        tol = 1e-12
        active = np.where(conv >= max_val - tol, 1.0, 0.0)
        # if all zeros (unlikely) fallback to argmax as before
        if active.sum() == 0:
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi+1)
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[k - idx]
        else:
            a_rev = a[::-1]
            Wf = np.fft.rfft(active, n=L_grad)
            Af_rev = np.fft.rfft(a_rev, n=L_grad)

            grad = 2.0 * grad_full[n-1 : n-1 + n]
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)
        # Occasionally renormalize to avoid drift
        if it % 500 == 0:

            total = a.sum()

"""
Optimisation of a non‑negative step‑function for the objective

    2 * n * max(conv(a,a)) / (sum(a)**2)   (n = 4096)

The algorithm consists of three stages:

1️⃣  Smooth‑max mirror‑descent (exponentiated‑gradient) – a differentiable
    approximation of the max‑convolution is minimised.  A geometric schedule
    for the smoothing parameter α drives the iterates towards the true max.

2️⃣  Projected sub‑gradient descent that uses *all* active constraints
    (all indices where the convolution attains its maximum).  The sub‑gradient
    is obtained by a convolution with the indicator of the active set,
    providing a more accurate descent direction than the single‑index version.

3️⃣  A large‑scale incremental simulated‑annealing / hill‑climbing phase.
    Mass is transferred from one coordinate to another and the convolution
    is updated **incrementally** (no FFT in the inner loop).  This allows
    millions of cheap random moves, greatly expanding the search space while
    staying well inside the 1000 s budget.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
so the denominator of the objective is always 1 and the score reduces to
`2 * n * max(convolution)`.  The best sequence found is returned as a plain
Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum is 1 → exact objective)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L_self)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the smooth approximation of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_score = _eval_fast(a, L_self)

    # use a slightly broader α‑schedule than before
    alphas = np.geomspace(0.5, 500.0, num=40)

    it = 0
    max_iters = 400_000

    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) exponentiated‑gradient step on the simplex
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)     # avoid exact zeros
            a /= a.sum()

            if it % 600 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : sub‑gradient descent using *all* active constraints
# ----------------------------------------------------------------------
def _subgradient_descent_active(a0: np.ndarray,
                                n: int,
                                deadline: float,
                                start_time: float,
                                best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max(conv(a,a)).
    The sub‑gradient is the average of gradients of *all* active indices.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)          # full self‑convolution
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # ---- active‑set gradient ----
        # tolerance is absolute because conv values are ≈ 1/n at optimum
        tol = 1e-12
        active = np.where(conv >= max_val - tol, 1.0, 0.0)
        if active.sum() == 0:       # fall back to single‑index case (should not happen)
            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[k - idx]
        else:
            a_rev = a[::-1]
            Wf = np.fft.rfft(active, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # momentum + Euclidean projection onto simplex
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

        # occasional renormalisation (mitigate drift)
        if it % 500 == 0:
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : large‑scale incremental annealing / hill‑climb
# ----------------------------------------------------------------------
def _incremental_annealing(a0: np.ndarray,
                           conv0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer moves with an *incremental* update of the self‑convolution.
    No FFT is used inside the inner loop, allowing millions of cheap proposals.
    """
    a = a0.copy()
    conv = conv0.copy()                     # length = 2*n-1
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_best_score

    rng = np.random.default_rng()

    # annealing parameters
    max_iters = 5_000_000
    T0 = 5e-2
    T_end = 1e-8
    eps_start = 0.20
    eps_end = 0.02

    it = 0
    while time.time() < deadline:
        it += 1
        # progress for schedules (clamped to [0,1])
        prog = it / max_iters
        if prog > 1.0:
            prog = 1.0

        T = T0 * (T_end / T0) ** prog
        eps_factor = eps_start * (eps_end / eps_start) ** prog

        # --- propose a move -------------------------------------------------
        src = rng.choice(n, p=a)                     # heavy source
        dst = rng.integers(0, n)
        if src == dst:
            continue
        a_src = a[src]
        if a_src < 1e-15:
            continue

        eps = a_src * rng.random() * eps_factor
        if eps <= 0.0:
            continue

        # --- incremental update of the convolution -------------------------
        conv_prop = conv.copy()
        # contribution of -2*eps*a to the slice centred at src
        conv_prop[src:src + n] -= 2.0 * eps * a
        # contribution of +2*eps*a to the slice centred at dst
        conv_prop[dst:dst + n] += 2.0 * eps * a
        # quadratic corrections
        conv_prop[2 * src] += eps * eps
        conv_prop[2 * dst] += eps * eps
        conv_prop[src + dst] -= 2.0 * eps * eps

        new_max = conv_prop.max()
        new_score = 2.0 * n * new_max
        delta = new_score - cur_score

        # --- Metropolis acceptance -----------------------------------------
        accept = False
        if delta < 0.0:
            accept = True
        elif T > 0.0 and math.exp(-delta / T) > rng.random():
            accept = True

        if accept:
            a[src] -= eps
            a[dst] += eps
            conv = conv_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

        # periodic full recomputation to avoid accumulated round‑off
        if it % 500_000 == 0:
            total = a.sum()
            if total > 0:
                a /= total
            conv = _conv_self(a, L=_next_pow2(2 * n - 1))
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Re‑calc at it={it:,} best={best_score:.6f}")

    return best_a, best_score
