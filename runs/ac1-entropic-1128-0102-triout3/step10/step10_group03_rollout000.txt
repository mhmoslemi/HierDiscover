def _adam_smooth_descent(start_a, deadline, start_time, L_fft, L_grad):
    n = len(start_a)
    a = start_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.1
    it = 0
    tau = 0.2
    tau_min = 1e-7
    tau_decay = 0.9995
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()
        # compute gradient efficiently via fft as before
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / np.sqrt(it)  # or constant alpha?
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

        # evaluate exact objective
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam smooth it={it:,} tau={tau:.2e} score={cur_score:.6f}")

        # decay tau
        tau = max(tau_min, tau * tau_decay)
    return best_a, best_score

def _simulated_annealing(a0, deadline, start_time, L_fft):
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    cur_score = best_score
    n = len(a)
    rng = np.random.default_rng()
    it = 0
    T0 = 0.01
    T_min = 1e-7
    while time.time() < deadline:
        it += 1
        # linear cooling schedule
        T = max(T_min, T0 * (deadline - time.time()) / (deadline - start_time))
        # propose move
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.3  # up to 30% move
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # ensure nonnegative (already)
        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} T={T:.2e} best={best_score:.6f}")

"""
Propose a step function (length 4096) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search combines:
1. A quick parametric sweep (geometric, power‑law, beta families) and a few
   random Dirichlet starts to obtain a solid baseline.
2. Several short Adam‑based smoothed gradient descents (soft‑max
   approximation of the max) with random restarts.
3. A longer Adam descent from the current best.
4. A simulated‑annealing phase that performs mass‑transfer moves and
   occasionally accepts uphill steps.
5. A projected sub‑gradient refinement and a final hill‑climb polish.

All iterates stay on the probability simplex (non‑negative, sum = 1) via
Euclidean projection.  The routine respects the 1000 s budget and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Fast evaluation of the exact objective (assuming a.sum()≈1).
    Returns 2·n·max(conv(a,a)) / (sum(a)^2).
    """
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam‑based smoothed gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------
def _adam_smooth_descent(start_a: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int,
                         L_grad: int) -> tuple[np.ndarray, float]:
    """
    Performs Adam optimisation on the smoothed objective
        f(a) = 2·n·softmax_tau(conv(a,a)).
    The iterate is projected onto the simplex after each step.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.12          # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    tau = 0.20
    tau_min = 5e-8
    tau_decay = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and soft‑max weights
        conv = _conv_self(a, L_fft)
        maxc = conv.max()
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # ∑w = 1

        # gradient of the smoothed objective via FFT
        w_rev = w[::-1]
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # Adam update (step size slowly decays with √it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # exact objective for bookkeeping
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing with mass‑transfer moves (exact objective)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         L_fft: int) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a random fraction of mass from a source index
    (chosen proportional to its mass) to a destination index.
    Accepts uphill moves with probability exp(-Δ/T).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    cur_score = best_score
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0

    # initial temperature (tuned empirically)
    T0 = 2e-3
    T_min = 1e-7

    while time.time() < deadline:
        it += 1
        # linear cooling based on remaining time
        remaining = deadline - time.time()
        total = deadline - start_time
        T = max(T_min, T0 * remaining / total)

        # propose a mass transfer
        src = rng.choice(n, p=a)          # higher probability for larger mass
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _eval_fast(a_prop, L_fft)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} T={T:.2e} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (identical to reference, slightly tweaked)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on f(a) = 2·n·max(conv(a,a)) / (sum(a)^2)
    with momentum and Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (single worst coordinate)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing polish (exact objective)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a tiny fraction of mass between
    two coordinates, keep the move only if the true objective improves.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.choice(n, p=a)                # source weighted by current mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue
        eps = a[src] * rng.random() * 0.10      # move up to 10 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it={it:,} best={best_score:.6f}")

    return best_a, best_score

        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
