def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution, not a circular one.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear convolution of two real vectors u and v using FFT.
    Returns the full linear convolution (len(u)+len(v)-1 elements).
    """
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (unit‑sum vectors → the exact objective)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L=L)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    max_iterations: int = 300_000,
                    alpha_max: float = 400.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation of max(conv(a,a)).
    Uses exponentiated‑gradient updates, gradients via FFT.
    α is increased geometrically up to ``alpha_max``.
    """
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_score = _eval_fast(a, L_self)

    # geometric schedule of α
    alphas = np.geomspace(0.5, alpha_max, num=40)

    it = 0
    for alpha in alphas:
        while time.time() < deadline - 0.2 and it < max_iterations:
            it += 1

            # self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # soft‑max weights (smooth max)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stability shift
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # exponentiated‑gradient step (mirror descent on simplex)
            eta = 0.07 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 1e-20)
            a /= a.sum()

            # occasional objective check
            if it % 800 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=_next_pow2(2 * n - 1))
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (first maximal entry)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : tiny simulated‑annealing polish (identical to reference)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper families for diverse initialisations
# ----------------------------------------------------------------------
def _geom_dist(n: int, q: float) -> np.ndarray:
    """Geometric (left‑skewed) distribution a_i ∝ q**i, normalised."""
    if q == 1.0:
        return np.full(n, 1.0 / n)
    powers = q ** np.arange(n, dtype=np.float64)
    return powers / powers.sum()


def _sym_exp_dist(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-lam * |i-(n‑1)/2|)."""
    idx = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx))
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣  Assemble a small pool of promising starters
    # --------------------------------------------------------------
    starters = []

    # a) global best if it exists
    try:
        starter_global = np.asarray(height_sequence_1, dtype=float)
        if starter_global.ndim == 1 and starter_global.size > 0:
            starter_global = np.maximum(starter_global, 0.0)
            if starter_global.size != n:
                if starter_global.size > n:
                    starter_global = starter_global[:n]
                else:
                    starter_global = np.pad(starter_global, (0, n - starter_global.size))
            if starter_global.sum() > 0:
                starter_global /= starter_global.sum()
            else:
                starter_global = np.full(n, 1.0 / n)
            starters.append(starter_global)
    except Exception:
        pass

    # b) uniform
    starters.append(np.full(n, 1.0 / n))

    # c) a few random Dirichlet draws (moderately sparse)
    for _ in range(2):
        vec = np.random.dirichlet(np.full(n, 0.5))
        starters.append(vec)

    # d) geometric families (left‑skewed and mirrored right‑skewed)
    for q in (0.6, 0.8, 0.9):
        starters.append(_geom_dist(n, q))
        starters.append(_geom_dist(n, 1.0 / q))   # right‑skewed mirror

    # e) symmetric exponential families
    for lam in (0.001, 0.003, 0.006):
        starters.append(_sym_exp_dist(n, lam))

    # --------------------------------------------------------------
    # 1️⃣  Run a smooth‑max descent from each starter (time‑sliced)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    remaining_starters = len(starters)
    for s_idx, init in enumerate(starters):
        if time.time() >= deadline - 30.0:
            break  # keep a little budget for polishing phases

        # normalise current starter
        a = np.maximum(init, 0.0)
        total = a.sum()
        if total == 0.0:
            continue
        a /= total

        init_score = _eval_fast(a, L_self)
        if init_score < best_score:
            best_score = init_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] starter {s_idx+1}/{remaining_starters} score={init_score:.6f}")

        # allocate a slice of the remaining time for this particular run
        time_left = deadline - time.time()
        # reserve about 1/3 of the remaining time for later phases
        smooth_time = min(time_left * 0.5 / (remaining_starters - s_idx), time_left - 20.0)
        if smooth_time <= 0.0:
            continue
        smooth_deadline = time.time() + smooth_time

        a_s, sc_s = _smooth_descent(
            a,
            n,
            smooth_deadline,
            start_time,
            max_iterations=250_000,
            alpha_max=400.0,
        )
        if sc_s < best_score:
            best_score = sc_s
            best_seq = a_s.copy()
            print(f"[{time.time() - start_time:.2f}s] after smooth {s_idx+1}/{remaining_starters} score={sc_s:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Short projected sub‑gradient refinement (as reference)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sub_deadline = deadline - 5.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad polish improved score={sc_sub:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final annealing polish (uses whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        # scale number of annealing steps to the remaining wall‑clock time
        time_left = deadline - time.time()
        # roughly 200 µs per step on typical hardware → estimate step count
        est_steps = int(min(500_000, max(20_000, time_left * 2_000)))
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=5e-2, end_temp=1e-7,
                                       steps=est_steps)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improved score={sc_final:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Very short high‑α smooth finish (if any budget remains)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        final_deadline = deadline - 0.2
        a_last, sc_last = _smooth_descent(
            best_seq,
            n,
            final_deadline,
            start_time,
            max_iterations=80_000,
            alpha_max=1200.0,
        )
        if sc_last < best_score:
            best_score = sc_last
            best_seq = a_last.copy()
            print(f"[{time.time() - start_time:.2f}s] final smooth finish score={sc_last:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally and produce a plain‑Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
