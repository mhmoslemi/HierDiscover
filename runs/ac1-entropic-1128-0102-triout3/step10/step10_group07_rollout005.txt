"""
Propose a non‑negative step‑function (a list of heights) that minimises
``evaluate_sequence``.  

The algorithm mixes several well‑known ideas:

1. **Rich initial pool** – uniform, Gaussian, exponential, Dirichlet draws and
   sparse random supports.
2. **Smooth exponentiated‑gradient optimisation** – instead of using only the
   active shift of the convolution (the classic sub‑gradient), we minimise a
   smooth approximation of the max‑convolution by applying a soft‑max weighting.
   This yields a dense gradient that can escape shallow local minima.
   The temperature of the soft‑max (`lambda_`) is slowly increased (annealed)
   while the step‑size (`eta`) decays as 1/√t.
3. **Refinement with the original active‑set exp‑gradient** – after the smooth
   phase a short mirror‑descent run that uses the actual max‑shift gradient
   tightens the last bits of the objective.
4. **Deterministic greedy peak‑reduction** – a final local move that transfers
   mass from the coefficient that contributes most to the current peak towards
   the smallest coefficient, accepting only strict improvements.
5. **Safety normalisation** – the final vector is clipped to `[0,∞)`,
   renormalised to sum 1 and returned as plain Python floats.

The whole routine respects the 1000 s wall‑clock limit and prints progress
information.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# FFT based convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_full(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution x * y (length len(x)+len(y)-1) via FFT."""
    Xf = np.fft.rfft(x, n=fft_len)
    Yf = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(Xf * Yf, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Score helper (exact objective, no extra clipping)
# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, fft_len_self: int) -> float:
    """Return 2·n·max(a*a) for a normalised to sum‑1."""
    conv = _conv_self(a, fft_len_self)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient optimiser (soft‑max approximation)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     fft_len_self: int,
                     fft_len_grad: int,
                     eta0: float = 0.4,
                     lambda0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective
        F(a) = logsumexp(lambda·(a*a)),
    with projection onto the probability simplex via multiplicative updates.
    Returns the best vector seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)          # avoid exact zeros at start
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = _score(a, n, fft_len_self)

    t = 0
    lambda_ = lambda0

    while time.time() < deadline:
        t += 1
        # --- convolution a*a (full) ---
        conv = _conv_self(a, fft_len_self)               # length 2n‑1

        # --- soft‑max weights over all shifts ---
        shift = conv.max()
        # prevent overflow/underflow
        exp_arg = lambda_ * (conv - shift)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to uniform weight – should never happen
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                                   # now a probability distribution

        # --- gradient of the smoothed objective ---
        a_rev = a[::-1]
        # convolution w * a_rev (full) – length 3n‑2
        grad_full = _conv_full(w, a_rev, fft_len_grad)
        # relevant slice: indices n‑1 … n‑1+n‑1
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # --- exponentiated‑gradient update ---
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % 100 == 0:
            cur_sc = _score(a, n, fft_len_self)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] smooth improve: iter={t:,}, score={cur_sc:.6f}, λ={lambda_:.3f}")

        # anneal λ (increase, making the soft‑max sharper)
        if t % 200 == 0:
            lambda_ = min(30.0, lambda_ * 1.03)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Classic exponentiated‑gradient optimiser (active‑set gradient)
# ----------------------------------------------------------------------
def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len_self: int,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on f(a)=max_k (a*a)[k] (the original sub‑gradient method).
    Returns best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _score(a, n, fft_len_self)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] expgrad improve: iter={t:,}, score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional exact check
        if t % 500 == 0:
            sc = _score(a, n, fft_len_self)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction (local improvement)
# ----------------------------------------------------------------------
def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Transfer a fraction of the coefficient that most contributes to the current
    convolution peak towards the smallest coefficient, accepting only strict
    improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len_self = _next_pow2(2 * n - 1)
    best_score = _score(a, n, fft_len_self)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len_self)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]          # index with largest contribution

        # pick destination: the globally smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]                     # modest fraction to move
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _score(a_prop, n, fft_len_self)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: it={it}, score={new_score:.6f}")

    return a, best_score

# Random start generators

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.99) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec
