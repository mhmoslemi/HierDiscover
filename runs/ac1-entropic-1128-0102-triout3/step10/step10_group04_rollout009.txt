    import numpy as np, time, math

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # Set n
    n = 4096

    # Acquire initial sequence
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim != 1 or init.size == 0:
            raise Exception
        if init.size != n:
            # interpolate
            x_old = np.linspace(0,1,init.size)
            x_new = np.linspace(0,1,n)
            init = np.interp(x_new, x_old, init)
    except Exception:
        init = np.full(n, 1.0/n)

    # Ensure non-negative and normalize
    init = np.clip(init, 0, None)
    if init.sum() == 0:
        init = np.full(n, 1.0/n)
    else:
        init = init / init.sum()

    a = init.copy()
    # Quick evaluation function
    def compute_score(v):
        # v assumed non-negative
        total = v.sum()
        if total < 0.01:
            return np.inf
        L = _next_pow2(2*n - 1)
        conv = _conv_self(v, L)  # length 2n-1
        maxb = conv.max()
        return float(2 * n * maxb / (total**2))

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # Phase 1: exponentiated gradient descent

    # set initial eta

    Lfft = _next_pow2(2*n-1)

        cur_max = conv.max()
        cur_score = 2 * n * cur_max
        # note: sum(a) == 1 by construction
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start:.2f}s] exp‑grad iter={it} score={cur_score:.6f}")

        # gradient at max location
        k = int(conv.argmax())
        lo = max(0, k-(n-1))
        hi = min(n-1, k)

        if dst >= src:

    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s>0:
        best_a = best_a / s
    else:
        best_a = np.full(n, 1.0/n)

    # convert to python list
    return [float(x) for x in best_a]

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------

The optimizer works in three stages and is deliberately different from the
previous one:

1. **Initialisation** – use the best known construction (``height_sequence_1``)
   if it exists, otherwise fall back to a few deterministic shapes and keep the
   one with the smallest objective value.

2. **Exponentiated‑gradient descent** – mirror‑descent on the true
   ``max(a*a)`` (the peak of the self‑convolution).  The update is
   multiplicative, followed by a Euclidean projection onto the probability
   simplex.  This quickly drives the peak down and provides a good starting
   point for the next stage.

3. **Aggressive greedy flattening** – repeatedly move a tiny amount of mass
   from the index that contributes most to the current peak to an index that
   lies **outside** the contributing window and has the smallest weight.
   The move size is chosen so that the current peak would become equal to the
   second‑largest peak, i.e. it flattens the convolution as aggressively as
   possible while staying feasible.  The convolution is updated
   incrementally (O(n) per move) and fully recomputed only occasionally to
   avoid numerical drift.

If a few seconds remain after the greedy phase, a tiny random perturbation
followed by a short gradient burst is performed as a polish.

The whole routine respects the 1000 s budget, prints progress information
and finally returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    The result is a real array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v>=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # fallback (should never happen for a non‑empty vector)
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Phase‑1: exponentiated‑gradient descent on max(self‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient(a0: np.ndarray,
                  deadline: float,
                  start: float,
                  eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) optimiser that directly minimises
    ``max(conv)``.  Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # sum(a)=1 -> denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start:.2f}s] exp‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # gradient of the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # d/d a_i of conv[k]

        # mirror‑descent update (multiplicative)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex (exact sum = 1)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2: aggressive greedy flattening of the convolution peak
# ----------------------------------------------------------------------
def _greedy_flatten(a0: np.ndarray,
                    deadline: float,
                    start: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from the index that contributes
    most to the current convolution peak to an index outside the contributing
    window that has the smallest weight.  The move size is chosen so that the
    current peak would be lowered to the second‑largest peak (if possible).
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)

    conv = _conv_self(a, L)                     # length 2·n‑1
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0
    stagnation = 0
    recompute_interval = 1024

    while time.time() < deadline:
        it += 1

        # --- locate the current peak and the runner‑up ---
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second‑largest entry (ignore the current maximum)
        if conv.size > 1:
            # ``np.partition`` is O(n) and cheap
            second_max = np.partition(conv, -2)[-2]
        else:
            second_max = 0.0

        # --- indices that actually participate in this peak ---
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx_window = np.arange(lo, hi + 1)

        # partner values for every index that participates in the peak
        partner_vals = a[max_idx - idx_window]

        # source: the index whose partner value is maximal
        src = idx_window[np.argmax(partner_vals)]
        partner_src = a[max_idx - src]

        # --- choose a destination *outside* the window with minimal weight ---
        mask_out = np.ones(n, dtype=bool)
        mask_out[lo:hi + 1] = False
        if np.any(mask_out):
            dst = np.argmin(np.where(mask_out, a, np.inf))
        else:
            # fallback – any index different from src
            dst = (src + n // 2) % n

        # partner for destination (outside the window -> effectively zero)
        partner_dst = 0.0

        delta = partner_src - partner_dst    # guaranteed > 0
        if delta <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            # epsilon that would make the current peak equal the second peak
            eps_target = (max_val - second_max) / (2.0 * delta)
            if eps_target <= 0.0:
                eps = min(a[src], 0.02 * a[src])
            else:
                eps = min(a[src], max(0.0, 0.9 * eps_target))
                if eps < 1e-14:
                    eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution (O(n)) -----
        # use the *old* a for the update, then modify a
        conv[src:src + n] -= 2.0 * eps * a
        conv[dst:dst + n] += 2.0 * eps * a

        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0   # numerical safety

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, L)

        cur_score = 2.0 * n * conv.max()
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")
        else:
            stagnation += 1

        # stop early if we have been stagnant for a long time
        if stagnation > 200_000:
            break

    return best_a, best_score
