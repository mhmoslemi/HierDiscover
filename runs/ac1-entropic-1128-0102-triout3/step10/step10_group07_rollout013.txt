def _smooth_expgrad_optimize(a0, deadline, start_time, fft_len, beta0=1.0, beta_max=50.0, eta0=0.5):
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    t=0
    beta = beta0
    while time.time() < deadline:
        t += 1
        # compute FFT of a
        A = np.fft.rfft(a, n=fft_len)   # using same length for conv
        conv = np.fft.irfft(A * A, n=fft_len)[:2*len(a)-1]
        max_conv = conv.max()
        # update beta gradually
        beta = min(beta_max, beta0 + (beta_max - beta0) * t / 4000) # maybe linear schedule
        # compute softmax weights
        w = np.exp(beta * (conv - max_conv))
        w /= w.sum()
        # compute correlation via FFT
        # compute w FFT using length L = nextpow2(len(a) + len(w) -1)
        L = _next_pow2(len(a) + len(w) - 1)
        W = np.fft.rfft(w, n=L)
        # Compute correlation: a * conj(w) via FFT
        # Use A padded to length L
        A_L = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(A_L * np.conj(W), n=L)
        corr = corr[: len(a) + len(w) - 1]
        g = 2.0 * corr[2*len(a)-2 : 2*len(a)-2 + len(a)]
        # step
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum) or np.isinf(a_sum):
            a = np.full_like(a, 1.0/len(a))
        else:
            a = a / a_sum

        if t % 200 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] smooth expgrad iter={t} score={sc:.6f}")

    return best_a, best_score

"""
Propose a step function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Build a diversified pool of starting vectors (uniform, Gaussian,
  exponential, Dirichlet with various concentrations, sparse random
  supports, and the historic best if available).

* Pick the best start by a cheap exact evaluation.

* **Phase 1 – smoothed mirror descent**  
  Minimise the “soft‑max’’ of the self‑convolution:
        fβ(a)=log(∑k exp(β·(a*a)[k])) .
  The gradient of fβ is a correlation of the current vector with the
  soft‑max weights, which we compute with FFTs.  β is increased
  gradually (from 0.1 up to ~30) so that early iterations move the
  solution in a well‑behaved smooth landscape and later iterations
  focus on the true maximum.  Updates are multiplicative
  (exponentiated‑gradient) and the vector is re‑normalised onto the
  probability simplex after each step.

* **Phase 2 – fine‑grained max‑only polishing**  
  Run the original exponentiated‑gradient routine (gradient from the
  single shift attaining the current peak) for the remaining budget.

* **Phase 3 – occasional random‑restart bursts**  
  While time permits we draw fresh random starts and run a very short
  smoothed‑gradient run (≈5 s).  Any improvement updates the global best.

* **Phase 4 – deterministic greedy peak‑reduction**  
  A cheap local move transfers a fraction of the coefficient that most
  contributes to the current peak towards the smallest coefficient;
  only strict improvements are kept.

* Final clipping, normalisation and one last call to ``evaluate_sequence``.

The whole routine respects the 1000 s wall‑clock limit and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose `m` indices, give each equal mass 1/m, zero elsewhere.
    """
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             fft_len: int,
                             eta0: float = 0.5,
                             beta_start: float = 0.1,
                             beta_end: float = 30.0,
                             beta_ramp_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth surrogate
        fβ(a)=log Σ_k exp(β·(a*a)[k])
    The gradient is a correlation of `a` with the soft‑max weights.
    Returns the best vector seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)          # avoid exact zeros
    a = a / a.sum()                    # stay on the simplex
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # ----- convolution a*a -----
        A_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A_fft * A_fft, n=fft_len)[: 2 * n - 1]
        cur_max = conv.max()

        # ----- soft‑max weights for the current β -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / beta_ramp_iters)
        w = np.exp(beta * (conv - cur_max))   # shift for numerical stability
        w /= w.sum()                           # now w[k] ≥ 0, Σ w = 1

        # ----- correlation a ⋆ w  (g_i = 2 Σ_t a_t w[t+i]) -----
        # linear correlation length = n + (2n‑1) – 1 = 3n‑2
        L = _next_pow2(n + w.shape[0] - 1)
        A_full = np.fft.rfft(a, n=L)
        W_fft = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(A_full * np.conj(W_fft), n=L)
        corr = corr[: n + w.shape[0] - 1]          # keep the linear part
        # slice giving g_i = 2 * corr[2n‑2 : 2n‑2 + n]
        offset = 2 * n - 2
        g = 2.0 * corr[offset: offset + n]

        # ----- multiplicative update (mirror descent) -----
        eta = eta0 / math.sqrt(t)                 # diminishing step size
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- occasional exact evaluation -----
        if t % 200 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] smooth‑expgrad improve: iter={t:,}, score={sc:.6f}")

    return best_a, best_score

def _expgrad_optimize(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      fft_len: int,
                      eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] expgrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]          # sub‑gradient for the active shift

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 500 == 0:                 # cheap guard
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

    )
    print(
        f"[{time.time() - start_time:6.2f}s] *** after smooth phase best score = {best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Phase 2 – max‑only polishing (≈35 % of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        phase2_deadline = time.time() + 0.7 * remaining
        best_seq, polished_score = _expgrad_optimize(
            best_seq, phase2_deadline, start_time, fft_len, eta0=0.8
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = best_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** after max‑only polish best score = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – cheap random‑restart bursts (≈10 % of remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        if rng.random() < 0.5:
            m = int(rng.integers(300, 2500))
            cand_start = _sparse_random_start(n, m, rng)
        else:
            conc = 10 ** rng.uniform(-1.5, 1.0)   # 0.03 … 10
            cand_start = _dirichlet_start(n, conc=conc, rng=rng)

        sub_deadline = time.time() + 5.0
        cand_seq, cand_score = _smooth_expgrad_optimize(
            cand_start, sub_deadline, start_time, fft_len,
            eta0=0.5, beta_start=0.1, beta_end=20.0, beta_ramp_iters=1000
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** new global best from restart score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 4 – final deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** after greedy phase best score = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final safety checks and reporting
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    return [float(x) for x in best_seq.tolist()]
