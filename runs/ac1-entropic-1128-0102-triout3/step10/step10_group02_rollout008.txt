    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    iter = 0
    L_self = _next_pow2(2*n -1)
    L_grad = _next_pow2(3*n -2)

    alpha = 0.5
    alpha_target = 500.0
    alpha_steps = 10000  # steps per alpha increase maybe.

    while time.time() < deadline - 0.5:
        iter += 1

        # compute a from y:
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()

        # Compute convolution:
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]

        # compute w with current alpha
        exp_arg = alpha * conv
        max_exp = exp_arg.max()
        w = np.exp(exp_arg - max_exp)
        w_sum = w.sum()
        w /= w_sum

        # compute grad for L
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        conv_full = np.fft.irfft(Wf * Revf, n=L_grad)[:3*n-2]
        grad_a = 2.0 * conv_full[n-1:n-1+n]   # gradient of L wrt a

        # gradient wrt y
        grad_y = a * (grad_a - np.dot(grad_a, a))

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** iter)
        v_hat = v / (1 - beta2 ** iter)
        # learning_rate schedule maybe decays with sqrt(alpha)?
        eta = 0.05 / (np.sqrt(alpha) + 1e-8)  # degrade as alpha grows
        y = y - eta * m_hat / (np.sqrt(v_hat) + epsilon)

        # Evaluate exact objective periodically
        if iter % 500 == 0:
            # compute exact obj

    a0 = np.maximum(a0, 0)
    if a0.sum() == 0:

    else:

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_a = a0.copy()
    # compute initial score
    conv0 = _conv_self(a0, L_self)
    best_score = 2.0 * n * conv0.max()
    # lengths

    L_grad = _next_pow2(3 * n - 2)
    # schedule of alpha
    alpha = 0.5
    alpha_target = 300.0   # increase to high value for sharpness
    alpha_inc_steps = 2000  # increase after fixed iterations
    # iteration counter
    it = 0
    while (time.time() < deadline - 0.5) and (max_iters is None or it < max_iters):
        it += 1
        # compute a from y
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()

        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[:2 * n - 1]

        exp_arg = alpha * conv
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        w_sum = w.sum()
        w /= w_sum
        # compute grad_a

        conv_full = np.fft.irfft(Wf * Revf, n=L_grad)[:3 * n - 2]
        grad_a = 2.0 * conv_full[n - 1:n - 1 + n]  # gradient dL/da
        # gradient wrt y
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

"""
Propose a non‑negative step‑function (length 4096) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The method consists of several stages:

1️⃣  Load a starter (global `height_sequence_1` if present) and normalise it.
2️⃣  Run a few short Adam‑optimisation runs with random Dirichlet starts.
    Adam works on the *smooth* approximation of the max (log‑sum‑exp)
    using a softmax parameter α that is increased geometrically.
    The variables are the logits of a soft‑max, guaranteeing a
    non‑negative simplex‑constrained vector.
3️⃣  Continue optimisation from the best found point with a long Adam run
    (more iterations, higher α) – this usually lowers the exact objective
    substantially.
4️⃣  Refine the solution with a projected sub‑gradient descent (the same
    routine that was used in the reference solution).
5️⃣  Finish with a tiny simulated‑annealing polish (mass transfers).

The whole procedure respects the 1000 s wall‑clock budget and never
exceeds it.  The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast version, assumes sum ≈ 1)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a))   (a is assumed to be non‑negative, sum≈1)."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on the smooth (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   max_iters: int = None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth approximation
        Lα(a) = (1/α)·log Σk exp(α·convk(a))
    The variable is the softmax logits y, guaranteeing a∈Δ.
    Returns the best (exact) sequence seen and its exact score.
    """
    # initialise logits
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()
    y = np.log(a0 + 1e-30)               # logits
    m = np.zeros_like(y)                 # first moment
    v = np.zeros_like(y)                 # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # start with a small smoothness, increase to a sharp proxy
    alpha = 0.5
    alpha_target = 300.0
    alpha_inc_every = 2000   # increase α every this many iterations
    it = 0

    # record the best exact score seen so far
    conv0 = _conv_self(a0, L_self)
    best_score = float(2.0 * n * conv0.max())
    best_a = a0.copy()

    while time.time() < deadline - 0.3 and (max_iters is None or it < max_iters):
        it += 1

        # ---- a = softmax(y) -------------------------------------------------
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)
        a = exp_y / exp_y.sum()

        # ---- self‑convolution ------------------------------------------------
        Af = np.fft.rfft(a, n=L_self)
        conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

        # ---- softmax over convolution entries (smooth max) -------------------
        exp_arg = alpha * conv
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ---- gradient of the smooth max w.r.t a (∂Lα/∂a) ----------------------
        a_rev = a[::-1]
        Wf = np.fft.rfft(w, n=L_grad)
        Revf = np.fft.rfft(a_rev, n=L_grad)
        conv_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
        grad_a = 2.0 * conv_full[n - 1:n - 1 + n]    # = ∂Lα/∂a

        # ---- chain rule to obtain ∂Lα/∂y ------------------------------------
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)

        # ---- Adam update -----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        # learning‑rate decays a little with √α to stay stable for large α
        eta = 0.05 / (math.sqrt(alpha) + 1e-12)
        y = y - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional exact evaluation --------------------------------------
        if it % 500 == 0:
            cur_score = float(2.0 * n * conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # ---- increase smoothness parameter ------------------------------------
        if it % alpha_inc_every == 0 and alpha < alpha_target:
            alpha = min(alpha_target, alpha * 1.2)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – Projected sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    Momentum (β≈0.9) and a 1/√t step‑size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline - 0.2:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max / (a.sum() ** 2))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (first occurrence of the maximal value)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # sub‑gradient of max(conv)

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)                # keep a on the simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase C – Tiny simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealer (tiny random moves)."""
    a = a0.copy()
    score = float(2.0 * n * _conv_self(a).max() / (a.sum() ** 2))
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule in log‑space
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = float(2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2))
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣ Load a starter (global best if available, else uniform)
    # ------------------------------------------------------------------
    try:

            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adapt length
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:

    if starter.sum() == 0.0:

    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _exact_score(best_seq, n)
    print(f"[{time.time() - start_time:.2f}s] Starter exact score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ A few short Adam runs from random Dirichlet starts
    # ------------------------------------------------------------------
    rest_time = deadline - time.time()
    n_restarts = 3
    short_iters = 6_000                     # iterations per short run
    for r in range(n_restarts):
        if time.time() >= deadline - 30.0:
            break
        a_rand = np.random.dirichlet(np.ones(n))
        a_opt, sc_opt = _adam_optimize(a_rand, n, time.time() + rest_time / 2,
                                       start_time, max_iters=short_iters)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {r+1}/{n_restarts} improved to {sc_opt:.6f}")

    if time.time() < deadline - 50.0:
