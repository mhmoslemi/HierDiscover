```
"""
Propose a step function minimizing 2 n max(conv) / (sum a)^2.
Algorithm:
1. Choose a large even length n (default 16384) for fine discretisation.
2. Initialise with a symmetric linear tent a_i ∝ min(i, n-1-i).
3. Run a projected sub‑gradient descent that uses the average of
   the K largest convolution entries (K=5) as the sub‑gradient.
   The sub‑gradient for a convolution entry k is
       g_i = 2 a_{k-i} for i that contribute to the entry.
   The average of the top‑K gradients gives a smoother direction.
   Momentum (β=0.9) and a diminishing step 0.5/√t are used.
   After each step the vector is projected onto the probability simplex.
4. Keep the best iterate seen (lowest original objective).
5. If time permits, perform a cheap simulated‑annealing polishing that
   repeatedly moves a small random mass packet between two indices.
6. Return the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {w >= 0, Σ w = 1}."""
    # Same implementation as in the reference code.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _subgrad_topk(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent where the sub‑gradient is the
    average of the gradients of the `top_k` largest convolution entries.
    Returns (best_seq, best_score).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Momentum variables
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # Find the indices of the top_k entries.
        # np.argpartition is O(N) and gives unsorted top_k.
        if top_k >= conv.shape[0]:
            top_idx = np.arange(conv.shape[0])
        else:
            part = np.argpartition(conv, -top_k)[-top_k:]
            # optionally sort them descending for deterministic order
            top_idx = part[np.argsort(-conv[part])]

        # Build the average gradient.
        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            # contribution for entry k is 2 * a_{k - idx}
            grad[idx] += 2.0 * a[k - idx]

        # Average over the selected entries.
        grad /= float(len(top_idx))

        # Momentum + step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)

        a -= step * momentum
        # Projection onto the simplex (keeps sum = 1 and non‑negative)
        a = _project_simplex(a)

    return best_a, best_score

def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing polish that moves tiny packets of mass.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # Fixed budget of attempted moves.
    total_steps = 40_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # Fast exponential cooling.
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

"""
Propose a step function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

where conv(a,a) is the linear self‑convolution of the vector a.
The algorithm follows three stages:

1. **Warm‑start** – a symmetric linear “tent” (a_i ∝ min(i,n‑1‑i))
   normalised to the probability simplex.
2. **Projected sub‑gradient descent** – instead of using the gradient
   of a single maximal convolution entry, we average the gradients of
   the `top_k` largest entries (k≈5‑7).  For a convolution entry k the
   sub‑gradient is `g_i = 2·a_{k‑i}` on the interval where it contributes.
   The average gradient gives a smoother descent direction.
   Momentum (β=0.9) and a diminishing step 0.5/√t are used; after each step
   the iterate is projected onto the simplex.
3. **Light annealing** – if time permits we polish the best iterate by
   moving tiny random packets of mass between indices (Metropolis rule).

All helpers are top‑level functions so they can be pickled.  The total
running time never exceeds the given budget (default 1000 s).  The
returned list contains non‑negative floats and will be fed to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Identical to evaluate_sequence (without defensive checks).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent – averaging the top‑k convolution entries
# ----------------------------------------------------------------------
def _subgrad_topk(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  top_k: int = 5) -> tuple[np.ndarray, float]:
    """
    Runs a projected sub‑gradient descent where the sub‑gradient is the
    average of the gradients of the `top_k` largest convolution entries.
    Returns the best iterate seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # indices of the top‑k convolution entries (unsorted)
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(conv, -top_k)[-top_k:]
            top_idx = part[np.argsort(-conv[part])]   # sorted descending (optional)

        # build the averaged sub‑gradient
        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(top_idx))

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection onto the simplex (keeps sum = 1 and non‑negative)
        a = _project_simplex(a)

    return best_a, best_score

    rng = np.random.default_rng()
