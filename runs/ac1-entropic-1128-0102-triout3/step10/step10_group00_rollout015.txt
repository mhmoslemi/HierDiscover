import time

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self(a):
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n - 1]
    return conv

def _objective(seq):
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def _maximum_convolution_index(conv):
    return int(conv.argmax())

def _gradient_maxconv(a):
    n = a.shape[0]
    conv = _conv_self(a)
    k = conv.argmax()
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi+1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad, k, conv

def _exp_gradient_descent(a0, deadline, start_time, step0=2.0):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, k, conv = _gradient_maxconv(a)
        step = step0 / math.sqrt(it)
        # exponentiated update (mirror descent on simplex)
        factor = np.exp(-step * grad)
        a = a * factor
        # keep numerically positive
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGrad iter={it:,} score={cur_score:.6f}")
    return best_a, best_score

def _anneal(seq, deadline, start_time, total_steps=50000):
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    T0 = 0.01
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling
        T = T0 * (1e-6 / T0) ** (it / total_steps)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue
        eps = a[src] * rng.random() * 0.5  # up to 50% of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _objective(a_prop)
        delta = new_score - cur_score
        if delta < 0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")
    return best_a, best_score

"""
A fast search for a low‑objective step function.
------------------------------------------------
* Uses a larger canvas (n = 16384, a power of two – fast FFT).
* Starts from several warm‑starts:
    – a global solution if one is available,
    – symmetric “tent” shapes with several exponents,
    – a handful of random Dirichlet vectors (both sparse and uniform).
* Refines every start with exponentiated‑gradient (mirror) descent on the
  exact nonsmooth objective (the sub‑gradient of the maximal convolution
  entry).  The update stays on the probability simplex and is cheap:
  one FFT per iteration.
* The best candidate is then polished by a light simulated‑annealing
  phase that moves tiny packets of mass between bins.
* The whole routine respects the 1000 s budget and returns a plain
  Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary 1‑D arrays using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror) descent on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          step0: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using the sub‑gradient of the maximal
    convolution entry.  Returns the best iterate seen before 'deadline'.
    """
    a = a0.copy()
    n = a.shape[0]

    # keep the best we have seen
    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        k = int(conv.argmax())                     # index of the maximal entry
        cur_score = 2.0 * n * conv[k] / (a.sum() ** 2)

        # update the best solution
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGrad iter={it:,} "
                  f"score={cur_score:.6f}")

        # sub‑gradient of max(conv) w.r.t. a (only the active lag matters)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # exponentiated (mirror) update on the simplex
        step = step0 / math.sqrt(it)
        factor = np.exp(-step * grad)
        a = a * factor
        a = np.maximum(a, 1e-14)          # keep strict positivity
        a /= a.sum()                      # projection onto the simplex

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float,
            total_steps: int = 50000) -> tuple[np.ndarray, float]:
    """Very cheap annealing that moves tiny mass packets."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    T0 = 0.01                         # initial temperature
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling – fast enough for a short run
        T = T0 * (1e-6 / T0) ** (it / total_steps)

        src = rng.choice(n, p=a)                # pick a source proportionally to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 50 % of the source mass
        eps = a[src] * rng.random() * 0.5
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple “tent” shape (symmetric) generator
# ----------------------------------------------------------------------
def _tent_shape(n: int, p: float) -> np.ndarray:
    """Symmetric tent a_i ∝ min(i, n‑1‑i)^p, normalised to sum 1."""
    x = np.arange(n, dtype=float) / (n - 1)
    w = np.minimum(x, 1.0 - x) ** p
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Upsample an arbitrary 1‑D array to a new length (linear interpolation)
# ----------------------------------------------------------------------
def _resample(seq: np.ndarray, target_n: int) -> np.ndarray:
    src_n = seq.shape[0]
    src_x = np.linspace(0.0, 1.0, src_n)
    dst_x = np.linspace(0.0, 1.0, target_n)
    res = np.interp(dst_x, src_x, seq)
    res = np.maximum(res, 0.0)
    if res.sum() == 0.0:
        return np.full(target_n, 1.0 / target_n)
    return res / res.sum()


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑objective step‑function.
    The returned list contains non‑negative floats and will be fed to
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs, larger n yields a
    # finer discretisation of the continuous optimum (≈ 1.5).
    # ------------------------------------------------------------------
    n = 16384

    # ------------------------------------------------------------------
    # 1) Build a pool of warm‑starts
    # ------------------------------------------------------------------
    candidates = []

    # a) global starter if it exists (the harness may provide it)
    try:
        base = np.asarray(height_sequence_1, dtype=float)
        if base.ndim == 1 and base.size > 0:
            base = np.maximum(base, 0.0)
            if base.sum() > 0.0:
                base = base / base.sum()
                base = _resample(base, n)
                candidates.append(base)
                print(f"[{time.time() - start_time:.2f}s] Loaded global starter "
                      f"(len={base.size})")
    except Exception:
        pass

    # b) symmetric tents with several exponents
    for p in [0.6, 0.8, 1.0, 1.2, 1.4, 1.6]:
        a = _tent_shape(n, p)
        candidates.append(a)
        sc = _objective(a)
        print(f"[{time.time() - start_time:.2f}s] Tent p={p:.2f} score={sc:.6f}")

    # c) a few random Dirichlet vectors (sparse = 0.5, uniform = 1.0)
    for alpha in [0.5, 0.5, 0.5, 0.5, 1.0, 1.0]:
        a = np.random.dirichlet(np.full(n, alpha))
        candidates.append(a)

    # ------------------------------------------------------------------
    # 2) Quick exponentiated‑gradient refinement of every start.
    #    Each start gets a few seconds; we keep the best overall.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_start_deadline = time.time() + min(5.0, 0.15 * remaining)
        for idx, cand in enumerate(candidates):
            if time.time() >= deadline:
                break
            refined, sc = _exp_gradient_descent(cand,
                                                per_start_deadline,
                                                start_time)
            if sc < best_score:
                best_score = sc
                best_seq = refined.copy()
                print(f"[{time.time() - start_time:.2f}s] Candidate {idx} "
                      f"improved best to {sc:.6f}")

    # ------------------------------------------------------------------
    # 3) A longer dedicated refinement starting from the current best.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.random.dirichlet(np.full(n, 1.0))
        best_score = _objective(best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:
        long_deadline = time.time() + 0.6 * remaining
        refined, sc = _exp_gradient_descent(best_seq,
                                            long_deadline,
                                            start_time)
        if sc < best_score:
            best_score = sc
            best_seq = refined.copy()
            print(f"[{time.time() - start_time:.2f}s] Long refinement best "
                  f"{sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        annealed, sc = _anneal(best_seq, deadline, start_time)
        if sc < best_score:
            best_score = sc
            best_seq = annealed.copy()
            print(f"[{time.time() - start_time:.2f}s] Annealing final best "
                  f"{sc:.6f}")

    # ------------------------------------------------------------------
    # 5) Normalise and convert to a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Final length={n}, "
          f"objective={final_score:.6f}")

    return [float(v) for v in best_seq]
