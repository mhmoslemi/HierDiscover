"""
Multi‑start projected (sub‑)gradient descent + light annealing.

The objective to minimise is
    2·n·max(conv(a,a)) / (sum(a)^2)          (scale‑invariant)
which is the same as the provided ``evaluate_sequence``.

Ideas
-----
* Build a diverse pool of initial candidates:
    - symmetric “tent’’ shapes a_i ∝ min(i,n‑1‑i)^p
    - symmetric Gaussian shapes
    - symmetric exponential decays
    - random Dirichlet draws (α∈{0.3,0.6,1.0})
    - the optional global starter ``height_sequence_1``.
* Keep the K best initial vectors (K≈5) and improve each of them
  with a projected Adam‑style optimiser.  The sub‑gradient of
  max(conv) is the average of the gradients of all indices that
  attain the maximal value, which smooths the descent.
* When the time budget is almost exhausted run a cheap simulated‑
  annealing polishing that moves tiny mass packets.
* All iterates are projected onto the probability simplex
  (Σa_i=1, a_i≥0) – this fixes the scale and respects the
  non‑negativity requirement.
* The code respects the 1000 s budget (a small safety margin is kept).

The routine returns a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear self‑convolution (real → real)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence, but fast)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx_i = 1, x_i≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the O(n log n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerically unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Adam‑style projected optimisation of the non‑smooth objective
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   label: str = "") -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser (sub‑gradient of max‑convolution).
    Returns the best iterate seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5            # initial learning rate (will be damped)

    m = np.zeros_like(a)     # first moment
    v = np.zeros_like(a)     # second moment

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        max_val = conv.max()

        # indices attaining the maximum (the active set)
        max_idxs = np.flatnonzero(conv == max_val)

        # sub‑gradient: average of the gradients of all active indices
        grad = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(max_idxs)

        # Adam update (with a simple 1/√t decay on the raw step size)
        step = base_lr / math.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex (keeps scale fixed)
        a = _project_simplex(a)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] {label} iter={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """
    Light annealing that moves tiny packets of mass.
    Returns the best sequence seen.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # fixed budget – we break early if time runs out
    total_steps = 35_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

        # choose a source index weighted by its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move a random fraction (up to 30 %) of the source mass to dst
        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: generate a family of symmetric candidates
# ----------------------------------------------------------------------
def _make_tent(n: int, p: float) -> np.ndarray:
    x = np.arange(n, dtype=float) / (n - 1)
    w = np.minimum(x, 1.0 - x) ** p
    if w.sum() == 0.0:
        return None
    return w / w.sum()


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return None
    return w / w.sum()


def _make_exponential(n: int, c: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    w = np.exp(-c * np.abs(i - centre))
    if w.sum() == 0.0:
        return None
    return w / w.sum()


def _make_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration α."""
    # Using Gamma trick (np.random.Generator.dirichlet not always available)
    g = rng.gamma(alpha, 1.0, size=n)
    if g.sum() == 0.0:
        return None
    return g / g.sum()

    candidates = []

    # -- symmetric Gaussians (different sigmas)
    for factor in [4, 6, 8, 10, 12]:

        a = _make_gaussian(n, sigma)
        if a is not None:
            candidates.append(a)

    # -- symmetric exponentials
    for c in np.linspace(0.0003, 0.0015, 7):

        if a is not None:
            candidates.append(a)

    for alpha in [0.3, 0.6, 1.0, 2.0]:
        for _ in range(8):

            if a is not None:

    # -- optional global starter (if present in the environment)

                candidates.append(starter)
    except Exception:

    # --------------------------------------------------------------
    # 2) Evaluate all candidates, keep the few best for refinement
    # --------------------------------------------------------------
    best_seqs = []          # list of (score, seq)
    for a in candidates:

        if math.isfinite(score):

    # sort by increasing score

    # keep at most K best distinct candidates

        best_seqs = [( _objective(uniform), uniform )]

    # track overall best
    overall_best_a = best_seqs[0][1].copy()
    overall_best_score = best_seqs[0][0]
    elapsed = time.time() - start_time
    print(f"[{elapsed:.2f}s] Initial best score={overall_best_score:.6f}")

        for idx, (init_score, init_seq) in enumerate(best_seqs):
            sub_deadline = time.time() + per_candidate
            label = f"Adam#{idx}"
            refined_seq, refined_score = _adam_optimize(init_seq,
                                                        sub_deadline,
                                                        start_time,
                                                        label=label)
            if refined_score < overall_best_score:
                overall_best_score = refined_score
                overall_best_a = refined_seq.copy()
                elapsed = time.time() - start_time
