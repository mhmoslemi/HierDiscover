def propose_candidate(seed=0, budget_s=1000, **kwargs):

    import numpy as np, time, math

    np.random.seed(seed)

    start = time.time(); deadline = start + budget_s - 0.5

    candidate_n_values = [4096, 8192, 12288, 16384] (maybe stop earlier if time limited)

    best_score = float('inf'); best_seq = None

    # For each n, grid search α in [0.25, 0.55] step 0.03 (or fine)
    for n in candidate_n_values:

        x = (np.arange(n, dtype=np.float64) + 0.5)/n

        alphas = np.linspace(0.25, 0.55, 13) # 13 points

        for alpha in alphas:

            w = (x * (1.0 - x) + 1e-30) ** (-alpha)  # heavy near edges

            a = w / w.sum()

            score = _objective(a)  # exact

            if score < best_score:

                best_score = score; best_seq = a.copy(); best_n = n

                print(f"[{time.time()-start:.2f}s] init n={n} α={alpha:.3f} score={score:.6f}")

            # early break if time low

            if time.time() >= deadline: break

        if time.time() >= deadline: break

    # If best_score already < 1.503, skip refinement

    if best_score > 1.503 and time.time() < deadline - 5.0:
        # refine using Adam gradient descent on smoothed objective

        a = best_seq.copy()

        n = a.shape[0]

        # ADAM parameters

        beta1 = 0.9

        beta2 = 0.999

        eps = 1e-8

        m = np.zeros_like(a)

        v = np.zeros_like(a)

        t_iter = 0

        # gamma schedule: start at 5, increase slowly

        gamma = 5.0

        gamma_inc = 1.02  # increase factor each iteration

        step0 = 0.2

        while time.time() < deadline - 0.5:

            t_iter += 1

            # compute conv

            conv = _conv_self(a)

            maxb = conv.max()

            # compute smoothed gradient

            shift = maxb

            exp_term = np.exp(gamma * (conv - shift))

            p = exp_term / np.sum(exp_term)  # shape 2n-1

            # compute gradient

            a_rev = a[::-1]

            Lfft = _next_pow2(len(p) + len(a_rev) - 1)

            Pf = np.fft.rfft(p, n=Lfft)

            Ar = np.fft.rfft(a_rev, n=Lfft)

            conv_pa_rev = np.fft.irfft(Pf * Ar, n=Lfft)[: len(p) + len(a_rev) - 1]

            grad = 2.0 * conv_pa_rev[n-1:n-1+n]  # gradient of max approximation

            # Adam updates

            m = beta1 * m + (1 - beta1) * grad

            v = beta2 * v + (1 - beta2) * (grad ** 2)

            m_hat = m / (1 - beta1 ** t_iter)

            v_hat = v / (1 - beta2 ** t_iter)

            step = step0 / math.sqrt(t_iter)  # decaying base step

            a = a - step * m_hat / (np.sqrt(v_hat) + eps)

            # project onto simplex

            a = _project_simplex(a)

            # compute exact objective every 10 iterations

            if t_iter % 10 == 0:

                cur_score = _objective(a)

                if cur_score < best_score:

                    best_score = cur_score; best_seq = a.copy()

                    print(f"[{time.time()-start:.2f}s] refine iter={t_iter} score={cur_score:.6f}")

            # increase gamma gradually

            gamma *= gamma_inc

            # limit gamma (to avoid overflow)

            if gamma > 200: gamma = 200

    # final normalization (sum to 1)

    best_seq = np.maximum(best_seq, 0.0)

    total = best_seq.sum()

    if total > 0:

        best_seq = best_seq / total

    else:

        best_seq = np.full(best_seq.shape, 1.0 / best_seq.shape[0])

    return [float(v) for v in best_seq]

"""
Propose a step‑function with a low evaluation score.

The algorithm works in three stages:

1️⃣  **Coarse grid search** –  Try several sequence lengths (n) and several
    exponents α for the family a_i ∝ (x_i·(1−x_i))^{−α} (a discrete
    approximation of the arcsine law).  The candidate with the smallest exact
    objective is kept.

2️⃣  **Smoothed Adam optimisation** –  Starting from the best candidate,
    minimise a smooth approximation of the max‑convolution:
        Lγ(a) = (1/γ)·log Σ exp(γ·(a*a))
    The gradient of Lγ is computed via FFTs, then an Adam step is taken and
    the vector is projected onto the probability simplex.  γ is increased
    gradually, turning the smooth loss into the true max.

3️⃣  **Light annealing polish** –  A few‑ten‑thousand tiny mass‑transfer moves
    (simulated‑annealing style) are applied, accepting any move that reduces
    the exact objective (or with a small probability for uphill moves).  The
    best vector seen throughout the whole run is returned.

All operations respect the 1000 s wall‑time limit, use only NumPy and the
provided ``evaluate_sequence`` and never produce negative or overly large
entries.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Utility functions (FFT, simplex projection, objective)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation function used by the competition:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# 2. Smoothed max (soft‑max) gradient via FFT
# ----------------------------------------------------------------------
def _softmax_grad(a: np.ndarray, gamma: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(a*a) and its gradient.
    Returns:
        maxb   – the true maximal convolution entry (for monitoring)
        grad   – gradient of the smoothed max (∇ Lγ) w.r.t. a.
    """
    n = a.shape[0]

    # full convolution a*a
    conv = _conv_self(a)
    maxb = conv.max()

    # soft‑max probabilities (stable)
    shift = maxb
    exp_term = np.exp(gamma * (conv - shift))
    p = exp_term / exp_term.sum()                # probabilities, sum(p)=1

    # gradient = 2 * conv(p, a_rev) on the slice n-1 .. n-1+n-1
    a_rev = a[::-1]                              # reversed a
    L = _next_pow2(len(p) + len(a_rev) - 1)
    Pf = np.fft.rfft(p, n=L)
    Ar = np.fft.rfft(a_rev, n=L)
    conv_pa_rev = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]

    grad = 2.0 * conv_pa_rev[n - 1 : n - 1 + n]   # shape (n,)
    return maxb, grad


# ----------------------------------------------------------------------
# 3. Adam optimiser working on the smoothed loss
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best vector (and its exact objective) seen before ``deadline``.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for the smoothness parameter γ
    gamma = 5.0
    gamma_factor = 1.02          # increase every iteration
    max_gamma = 200.0

    # step‑size schedule (initial magnitude)
    step0 = 0.5

    best_a = a.copy()
    best_score = _objective(a)

    t = 0
    while time.time() < deadline:
        t += 1

        maxb, grad = _softmax_grad(a, gamma)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / math.sqrt(t)               # decaying base step
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # keep a on the simplex
        a = _project_simplex(a)

        # evaluate every few iterations (cheap compared to a full conv)
        if t % 10 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter={t:,} "
                      f"score={cur_score:.6f}")

        # ramp up the smoothness
        gamma = min(max_gamma, gamma * gamma_factor)

    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Simple simulated‑annealing polish (tiny mass transfers)
# ----------------------------------------------------------------------
def _anneal_polish(a: np.ndarray,
                   deadline: float,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Very light annealing: repeatedly move a small fraction of mass from a
    source index to a random destination.  Accept moves that improve the
    exact objective (or with a small Metropolis probability).
    """
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # decides how many attempts we can afford
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        # temperature schedule – very fast cooling
        T = 0.01 * (1e-6 / 0.01) ** (attempts / 50000.0)

        # pick a source proportional to its current mass
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # propose moving up to 20 % of the source mass
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal step={attempts:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
