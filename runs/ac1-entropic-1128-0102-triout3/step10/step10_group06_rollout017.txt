def compute_score(a):
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        return float('inf')
    a = a / total
    conv = _conv_self(a, _next_pow2(2*len(a)-1))
    max_b = conv.max()
    return 2 * len(a) * max_b

def refine_lp(a, deadline, start_time, max_iters=50, eps=1e-4):
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    while time.time() < deadline and max_iters>0:
        conv = _conv_self(a, _next_pow2(2*n-1))
        M = conv.max()
        # choose active set
        threshold = M * (1 - eps)  # maybe include k where conv >= threshold
        active = np.where(conv >= threshold)[0]
        # optionally limit size
        if len(active) > 30:
            # keep top 30 peaks
            active = active[np.argsort(-conv[active])[:30]]

        # Build constraint matrix
        A = np.zeros((len(active), n))
        for row_idx, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            # a[i] corresponds to coefficient for b_j
            A[row_idx, j] = a[i]

        # Solve LP maximize sum(b) s.t. A b <= M, b>=0
        c = -np.ones(n)  # minimize -sum(b)
        bounds = [(0, None) for _ in range(n)]

        # Use scipy.optimize.linprog
        from scipy.optimize import linprog
        # Set high iteration limit and latest method
        res = linprog(c, A_ub=A, b_ub=M * np.ones(len(active)), bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if not res.success:
            # If LP did not solve, break
            break

        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            break
        g = b / sum_b

        # Try mixing with step sizes
        t = 0.3
        improved = False
        while t > 1e-6:
            a_new = (1 - t) * a + t * g
            a_new = a_new / a_new.sum()
            new_score = compute_score(a_new)
            if new_score < best_score - 1e-9:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                print(f'[{time.time() - start_time:6.2f}s] LP improve: score={new_score:.6f}')
                break
            t *= 0.5
        if not improved:
            # No improvement with this direction; break
            break
        max_iters -= 1
    return best_a, best_score

import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self(a, fft_len):
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _compute_score(a, fft_len):
    # ensure non‑negative and normalised
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return float('inf')
    a = a / s
    conv = _conv_self(a, fft_len)
    return float(2 * a.shape[0] * conv.max())

def _make_exponential(n, r):
    idx = np.arange(n, dtype=float)
    vec = r ** idx
    return vec / vec.sum()

def _make_gaussian(n):
    sigma = n / 6.0
    xs = np.arange(n, dtype=float)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()

def _make_uniform(n):
    return np.full(n, 1.0 / n)

def _make_stretched_exponential(n, p, scale):
    i = np.arange(n, dtype=float)
    vec = np.exp(- ((i) / scale) ** p)
    return vec / vec.sum()

def _make_mixed_exponential(n, alpha, r1, r2):
    i = np.arange(n, dtype=float)
    vec = alpha * (r1 ** i) + (1.0 - alpha) * (r2 ** i)
    return vec / vec.sum()

def _build_lp_constraints(a, active):
    n = a.shape[0]
    m = active.shape[0]
    A = np.zeros((m, n), dtype=float)

        lo = max(0, k - (n - 1))

        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        # contribution a[i] * b[j] <= M
        A[row_idx, j] = a[i]
    return A

               max_iters=30, eps=1e-4, step=0.3):
    """
    One‑step iterative refinement using a tiny linear program.
    Returns the best vector found and its exact score.
    """
    from scipy.optimize import linprog

    best_a = a.copy()
    best_score = _compute_score(a, fft_len)
    n = a.shape[0]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # current convolution
        conv = _conv_self(a, fft_len)
        M = conv.max()

        # active constraints – all shifts within eps of the maximum
        thresh = M * (1.0 - eps)
        active = np.where(conv >= thresh)[0]

        # limit size to keep LP cheap
        if active.shape[0] > 30:
            # keep the 30 biggest values
            idx = np.argpartition(-conv[active], 30)[:30]
            active = active[idx]

        if active.shape[0] == 0:
            break

        # build A_ub
        A = _build_lp_constraints(a, active)

        # linear program: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n, dtype=float)
        bounds = [(0, None) for _ in range(n)]
        b_ub = M * np.ones(active.shape[0], dtype=float)

        # solve – we give a tiny time limit so that the routine
        # never blocks the overall budget
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'time_limit': 5})

            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=bounds, method='highs')

            break

        sum_b = b.sum()
        if sum_b == 0:
            break
        g = b / sum_b           # normalised candidate

        t = step
        improved = False

            break

"""
Search for a non‑negative step function that (almost) minimises
``evaluate_sequence``.  The routine works in three stages:

1. **Fast low‑dimensional families** – uniform, Gaussian, geometric,
   stretched‑exponential and mixtures of two exponentials are generated
   on a coarse grid together with a few random Dirichlet draws.  The
   best of them (by the exact objective) is kept.

2. **Linear‑programming refinement** – from the current best vector the
   convolution peaks are identified, a small LP (maximise the total mass
   of a new vector under the active peak constraints) is solved, the
   LP solution is normalised and mixed back with a short step.  This
   drives mass away from the dominant peaks and typically improves the
   bound beyond what pure gradient descent achieves.

3. **Light exponentiated‑gradient polishing** – a few thousand iterations
   of the mirror‑descent update used in the reference code are applied
   to fine‑tune the distribution.

All steps respect the time budget (default 1000 s).  The function returns
the best sequence found as a list of floats.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _compute_score(a: np.ndarray, fft_len: int) -> float:
    """
    Exact value of ``evaluate_sequence`` for a normalised non‑negative
    vector (sum == 1).  Returns 2·n·max(a*a) .
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return float("inf")
    a = a / s
    conv = _conv_self(a, fft_len)
    return float(2 * a.shape[0] * conv.max())

def _build_lp_constraints(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Build the matrix A for constraints
        (a * b)[k] = sum_i a_i * b_{k-i} ≤ M    for all k in active .
    Row ``k`` contains the coefficients of b_j in that inequality.
    """
    n = a.shape[0]
    m = active.shape[0]
    A = np.zeros((m, n), dtype=float)
    for row_idx, k in enumerate(active):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        A[row_idx, j] = a[i]          # a_i contributes to b_j
    return A


def _lp_refine(a: np.ndarray,
               fft_len: int,
               deadline: float,
               start_time: float,
               max_iters: int = 30,
               eps: float = 5e-4,
               step: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Iteratively improve ``a`` by solving a tiny LP that moves mass away
    from the currently dominant convolution peaks.
    Returns the best vector encountered and its exact score.
    """
    from scipy.optimize import linprog

    best_a = a.copy()
    best_score = _compute_score(a, fft_len)
    n = a.shape[0]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len)
        M = conv.max()

        # active shifts – those within ``eps`` of the peak
        thresh = M - max(eps * M, 1e-12)
        active = np.where(conv >= thresh)[0]

        # keep the most critical at most 30 constraints (cheap LP)
        if active.shape[0] > 30:
            idx = np.argpartition(-conv[active], 30)[:30]
            active = active[idx]

        c = -np.ones(n, dtype=float)

        try:

        sb = b.sum()

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        refined_seq, refined_score = _lp_refine(
            best_seq, fft_len, deadline, start_time,
            max_iters=30, eps=5e-4, step=0.3
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST after LP "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final light exponentiated‑gradient polish (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polished_seq, polished_score = _expgrad_polish(
            best_seq, fft_len, deadline, start_time, max_iters=5000
        )
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST after polishing "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety normalisation and reporting
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    # One last call to the official evaluator (cheap, just for the log)
    try:
        reported = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:6.2f}s] final reported score = {reported:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq.tolist()]
