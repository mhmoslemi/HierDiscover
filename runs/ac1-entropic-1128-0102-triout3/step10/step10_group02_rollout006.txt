def _adam_descent(a0, n, deadline, start_time, best_score_initial):
    a = a0.copy()
    best_a = a0.copy()
    best_score = best_score_initial
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    L = _next_pow2(3*n-2) # for gradient convolution
    while time.time() < deadline:
        t += 1
        # compute max conv and gradient
        conv = _conv_self(a, L=_next_pow2(2*n-1))
        cur_max = conv.max()
        # compute gradient of max conv (subgradient from active index)
        k = conv.argmax()
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k-idx]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps) # typical step
        a = a - step * m_hat
        # project onto simplex
        a = _project_simplex(a)
        cur_score = 2.0 * n * conv.max() / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score
```

def _adam_smooth_descent(a0, n, deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self=_next_pow2(2*n-1))
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # schedule alphas
    alphas = np.geomspace(0.5, 2000.0, num=40)  # more range
    L_self = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n-2)
    for alpha in alphas:
        while time.time() < deadline - 0.2:
            t += 1
            # compute conv
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[:2*n-1]
            # compute softmax weights
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum
            # compute gradient
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[:3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]
            # Adam updates
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr = 0.05 / (1 + t * 0.00001)  # decaying step size
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # projection onto simplex
            a = _project_simplex(a)
            # record best every 500 it
            if t % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.2f} it={t:,} score={cur_score:.6f}")
    return best_a, best_score

def _annealing(a0, n, deadline, start_time, cur_best_score, cur_best_seq):
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum()**2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    step = 0
    while time.time() < deadline:
        step += 1
        # Temperature schedule based on progress
        t_frac = (time.time() - start_time) / (deadline - start_time)
        T = max(1e-7, 0.01 * (1 - t_frac))
        # choose source based on contribution
        conv = _conv_self(a)
        k = conv.argmax()
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        # compute contributions
        contrib = a[idx] * a[k - idx]
        src = idx[contrib.argmax()]  # heavy contributor
        # choose dest random among indices not too close
        dst = rng.integers(n)
        if dst == src or a[src] < 1e-12:
            continue
        eps = a[src] * rng.random() * 0.2  # maybe 20%
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = 2.0*n*_conv_self(a_prop).max()/(a_prop.sum()**2)
        delta = new_score - score
        if delta < 0.0 or math.exp(-delta/T) > rng.random():
            a = a_prop
            score = new_score
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal step={step:,} best={best_score:.6f}")
    return best_seq, best_score

"""
Propose a step‑function (list of non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------
* Multiple restarts (global best + a few random Dirichlet starts).
* For each restart a continuation of *smooth Adam descent*:
      – a soft‑max (temperature α) is applied to the self‑convolution,
        providing a smooth approximation of the max.
      – the gradient of this smoothed max is obtained with two FFTs.
      – an Adam‑style update (with simplex projection) drives the
        distribution; α is increased geometrically so the optimisation
        gradually focuses on the true maximum.
* A short *projected sub‑gradient* phase refines the result.
* The remaining time is spent in a *targeted annealing* phase that
  repeatedly moves a small amount of mass from the pair contributing
  most to the current peak to a random location; Metropolis acceptance
  is used.
* The best sequence over all restarts is returned.

The whole routine respects the 1000 s wall‑clock budget and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear (not circular) self‑convolution a*a using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (used only for bookkeeping)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (assumed non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Adam optimisation on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation method: start with a small smoothing parameter α and
    increase it while performing Adam updates on the smoothed objective.
    """
    a = a0.copy()
    best_a = a.copy()
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # Adam accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = beta1
    beta2_pow = beta2
    t = 0

    # geometric schedule for α  (0.5 → 2000)
    alphas = np.geomspace(0.5, 2000.0, num=40)

    best_score = _eval_fast(a, L_self)

    for alpha in alphas:
        while time.time() < deadline - 0.15:
            t += 1

            # 1) self‑convolution
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights w_k ∝ exp(α·conv_k)
            exp_arg = alpha * conv
            max_exp = exp_arg.max()
            w = np.exp(exp_arg - max_exp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient of the smooth max (2·(w ⋆ a_rev))
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) Adam update + simplex projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            # a simple decaying learning rate
            lr = 0.07 / (1.0 + t * 1e-5)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a)

            # bias‑correction factors
            beta1_pow *= beta1
            beta2_pow *= beta2

            # occasional check of the true objective
            if t % 500 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.1f} it={t:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (as in the reference solution)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=_next_pow2(2 * n - 1))
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active offset (first if several)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted annealing (mass‑transfer moves guided by the current peak)
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7) -> tuple[np.ndarray, float]:
    """Tiny simulated‑annealing with moves that aim at the dominant contribution."""
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        # linear temperature schedule
        T = start_temp * (end_temp / start_temp) ** (it / 200_000)

        # recompute convolution only every few steps to save time
        if it % 10 == 0:
            conv = _conv_self(a)
            k = int(conv.argmax())
        else:
            # reuse previous k (good enough for the heuristic)
            k = int(_conv_self(a).argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # contribution of each pair to the current peak
        contrib = a[idx] * a[k - idx]
        if contrib.sum() == 0.0:
            src = rng.choice(idx)
        else:
            src = idx[np.argmax(contrib)]

        # random destination (avoid self‑move)
        dst = rng.integers(n)
        if dst == src or a[src] < 1e-14:
            continue

        # move a modest fraction of the source mass
        eps = a[src] * rng.random() * 0.25
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given wall‑clock budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a little safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Global best (if supplied) or uniform vector
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise Exception
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')

    starter = starter / max(starter.sum(), 1e-12)

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Multiple restarts – each gets a slice of the remaining budget
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0:
        # allocate roughly a third of the remaining time to this restart
        remaining = deadline - time.time()
        alloc = min(remaining * 0.33, 250.0)          # cap to keep a few restarts possible
        restart_deadline = time.time() + alloc

        if restart == 0:
            a0 = best_seq.copy()                     # start from the current best
        else:
            # random Dirichlet start (shape=1 → exponential distribution)
            a0 = np.random.gamma(1.0, size=n)
            a0 = a0 / a0.sum()

        # --------------------------------------------------------------
        # Phase 1 – smooth Adam descent (continuation)
        # --------------------------------------------------------------
        a_mid, sc_mid = _adam_smooth_descent(a0, n, restart_deadline, start_time)

        # --------------------------------------------------------------
        # Phase 2 – short sub‑gradient polish (if time permits)
        # --------------------------------------------------------------
        if time.time() < restart_deadline - 3.0:
            sub_deadline = restart_deadline - 1.0
            a_ref, sc_ref = _subgradient_descent(a_mid, n,
                                                 sub_deadline, start_time,
                                                 sc_mid)
            if sc_ref < sc_mid:
                a_mid, sc_mid = a_ref, sc_ref

        # --------------------------------------------------------------
        # Phase 3 – targeted annealing on whatever time is left
        # --------------------------------------------------------------
        if time.time() < restart_deadline:
            a_fin, sc_fin = _annealing(a_mid, n, restart_deadline,
                                       start_time, sc_mid, a_mid,
                                       start_temp=1e-2, end_temp=1e-7)
            if sc_fin < sc_mid:
                a_mid, sc_mid = a_fin, sc_fin

        # keep the overall best
        if sc_mid < best_score:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {restart} new best = {best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Final polishing with a short annealing (uses any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_seq, best_score = _annealing(best_seq, n, final_deadline,
                                          start_time, best_score,
                                          best_seq,
                                          start_temp=1e-2, end_temp=1e-7)

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
