Propose a step‐function (vector of non‑negative heights) that minimizes
``evaluate_sequence``.  

Algorithm
---------
1. **Diverse initial pool** – uniform, Gaussian, exponential, a few
   Dirichlet draws and several asymmetric “Beta‑shaped” vectors.
2. **Fast exact score** – the objective for a normalised vector is
   ``2·n·max(conv)`` where ``conv = a*a`` (full linear convolution).
   This is evaluated with an FFT implementation.
3. **Smooth exponentiated‑gradient refinement** – instead of using the
   sub‑gradient of a single active shift, we minimise a smooth
   approximation of the max (log‑sum‑exp) which yields a gradient that
   accounts for *all* shifts.  The update is multiplicative
   (mirrored descent) and the step size decays as ``1/√t``.
4. **Random‑restart bursts** – while time permits we start from fresh
   random candidates and run a short smooth‑gradient phase; any
   improvement replaces the current best.
5. **Deterministic greedy peak‑reduction** – final local moves that
   transfer a modest fraction of the coefficient responsible for the
   current peak towards the smallest coefficient (strict improvements
   only).
6. **Safety normalisation** – clip to ``[0,∞)``, renormalise to sum 1,
   evaluate once more and return the best vector found.

The whole routine respects the 1000 s wall‑clock limit and prints

import time

# FFT helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation r[j] = Σ_i w[i]·a[i+j]  (j = 0…len(a)-1).
    Implemented via FFT as convolution of w with the reversed a.
    """
    n = a.shape[0]
    L = _next_pow2(len(w) + n - 1)
    Af = np.fft.rfft(a[::-1], n=L)          # reversed a
    Wf = np.fft.rfft(w, n=L)
    conv = np.fft.irfft(Af * Wf, n=L)
    # the slice that corresponds to j = 0 … n‑1
    start = n - 1
    return conv[start : start + n]

    return np.full(n, 1.0 / n)


def _gaussian_start(n: int) -> np.ndarray:
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()

    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:

    return x / x.sum()

    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


def _beta_start(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta‑like shape: (i+1)^{a-1} (n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=float)
    # add a tiny epsilon to avoid zero when a or b < 1

    seq = (i + eps) ** (a - 1) * (n - i + eps) ** (b - 1)
    if seq.sum() == 0:

    return seq / seq.sum()

        if t % 300 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic peak‑reduction (same as before)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # modest transfer
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step function minimising ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # problem size -------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    starts = []

    # uniform and classic handcrafted starters
    starts.append(_uniform_start(n))
    starts.append(_gaussian_start(n))
    starts.append(_exponential_start(n, r=0.9995))
    starts.append(_exponential_start(n, r=0.985))

    # Dirichlet draws with various concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # Asymmetric Beta‑shaped candidates
    for _ in range(12):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        starts.append(_beta_start(n, a, b))

    # Sparse random supports
    for m in [400, 800, 1200, 2000]:
        for _ in range(2):
            starts.append(_sparse_random_start(n, m, rng))

    # --------------------------------------------------------------
    # Evaluate the raw starts and keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(starts):
        # ensure non‑negative and normalised
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0:
            cand = _uniform_start(n)
        cand = cand / cand.sum()
        # fast exact score (equivalent to evaluate_sequence for sum=1)
        conv = _conv_self(cand, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 1 – long smooth exponentiated‑gradient polishing
    # --------------------------------------------------------------
    polish_deadline = start_time + 0.80 * (deadline - start_time)
    polished_seq, polished_score = _smooth_expgrad_optimize(
        best_seq, polish_deadline, start_time, fft_len, eta0=0.8
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after smooth‑grad "
            f"score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 2 – cheap random‑restart bursts (≈10 % of remaining time)
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # draw a fresh random start (skewed Dirichlet or sparse)
        if rng.random() < 0.5:
            # sparse variant – random density
            m = int(rng.integers(300, 2500))
            cand_start = _sparse_random_start(n, m, rng)
        else:
            # Dirichlet with random concentration and a linear tilt
            conc = 10 ** rng.uniform(-1.5, 1.0)
            base = _dirichlet_start(n, conc=conc, rng=rng)
            tilt = np.linspace(1.0, 0.2, n)          # favour left side
            cand_start = base * tilt
            if cand_start.sum() == 0:
                cand_start = _uniform_start(n)
            cand_start = cand_start / cand_start.sum()

        # a short smooth‑gradient run (≈ 5 s)
        sub_deadline = time.time() + 5.0
        cand_seq, cand_score = _smooth_expgrad_optimize(
            cand_start, sub_deadline, start_time, fft_len, eta0=0.5
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from restart "
                f"score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase 3 – final deterministic greedy peak‑reduction
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy "
                f"score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    # return a plain Python list as required by the evaluator
    return [float(x) for x in best_seq.tolist()]
