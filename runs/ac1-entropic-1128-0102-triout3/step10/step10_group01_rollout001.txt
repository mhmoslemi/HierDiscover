def expgd_optimize(a0, deadline, start_time, beta_start=0.5, beta_end=10.0, eta0=0.1):

    a = a0.copy()

    best_a = a.copy()
    best_score = _objective(a)

    it = 0

    while time.time() < deadline:
        it += 1

        # compute conv
        conv = _conv_self(a)

        sum_a = a.sum()  # always 1 but may have numeric diff
        # Probably sum =1

        max_conv = conv.max()

        # actual objective (exact)
        cur_score = 2.0 * len(a) * max_conv / (sum_a ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it} score={cur_score:.6f}")

        # schedule beta
        total_frac = (time.time() - start_time) / (deadline - start_time)
        # Let's compute beta linear schedule
        beta = beta_start + (beta_end - beta_start) * total_frac

        # compute w (softmax over conv)
        # Because conv values are small, we can scale by beta with relative values.
        # Use normalized conv - max to avoid overflow.
        shifted = conv - max_conv  # max becomes 0
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / len(w)
        else:
            w = w / w_sum

        # compute gradient via correlation
        # g = 2 * correlation(a, w)
        # Compute conv_rev = convolution of w and a_rev
        # a_rev = a[::-1]
        conv_rev = _conv_fft(w, a[::-1])
        n = len(a)
        start_idx = n - 1
        g = 2.0 * conv_rev[start_idx:start_idx + n]

        # step size schedule
        eta = eta0 / math.sqrt(it)

        # update via exponentiated gradient: a_i <- a_i * exp(-eta * g_i)
        # To avoid underflow/overflow, we can clip g
        # Possibly also limit changes: using np.exp(-eta * g) may produce zero if g large.
        # We'll use np.exp(-eta * g) but clip exponent to [-50, 50]
        exp_factor = np.exp(np.clip(-eta * g, -50.0, 50.0))
        a = a * exp_factor

        # renormalize
        s = a.sum()
        if s == 0:
            # rare, fallback to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_a, best_score

"""
Propose a low‑evaluation step‑function.

Algorithm
---------
1. **Arcsine‑family initialization** – sweep a dense grid of exponents
   α for the family a_i ∝ (x_i·(1−x_i))^{−α}.  The best candidate (normalised
   to sum = 1) is kept as a start point.

2. **Smoothed max‑convolution descent** – minimise a smooth surrogate of the
   objective using an exponentiated‑gradient update on the probability simplex.
   The surrogate is the log‑sum‑exp of the self‑convolution values,
   controlled by a temperature β that is increased linearly during the run.
   The exact gradient is obtained via a single FFT‑based correlation.
   The iteration is stopped when the wall‑time budget expires.

3. **Light simulated‑annealing polish** – a few thousand random mass‑transfer
   moves (source ∝ current mass) are performed to escape tiny local traps.

The whole search respects the 1000 s budget, never returns negative numbers,
and returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (used for comparison and final reporting)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact evaluation: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s ** 2))


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on soft‑max of the convolution
# ----------------------------------------------------------------------
def _expgd_optimize(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    beta_start: float = 0.5,
                    beta_end: float = 12.0,
                    eta0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the max‑convolution objective by
    exponentiated‑gradient descent on the simplex.
    Returns the best iterate (and its exact objective value) seen before `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # exact convolution (for monitoring the true objective)
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} score={cur_score:.6f}")

        # --------------------------------------------------------------
        # 1) Temperature schedule (β moves from low to high)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        beta = beta_start + (beta_end - beta_start) * frac

        # --------------------------------------------------------------
        # 2) Soft‑max weights w_k ∝ exp(β·conv_k)
        #    (shifted by max to avoid overflow)
        # --------------------------------------------------------------
        shifted = conv - max_conv          # max becomes 0
        w = np.exp(beta * shifted)         # w_max = 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # --------------------------------------------------------------
        # 3) Gradient of the smoothed objective:
        #    ∇f_i = 2 * Σ_j w_{i+j} a_j  (cross‑correlation a ★ w)
        # --------------------------------------------------------------
        conv_rev = _conv_fft(w, a[::-1])               # length = 3n‑2
        start_idx = n - 1
        grad = 2.0 * conv_rev[start_idx:start_idx + n]

        # --------------------------------------------------------------
        # 4) Exponentiated‑gradient step on the simplex
        # --------------------------------------------------------------
        eta = eta0 / math.sqrt(it)          # decreasing step size
        # clip exponent to avoid under/overflow
        exp_factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * exp_factor

        # ensure no entry collapses to exact zero (keeps support)
        a = np.maximum(a, 1e-18)

        # projection onto the simplex = normalise (already non‑negative)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

    return best_a, best_score


# ----------------------------------------------------------------------
# Very light simulated‑annealing polishing (mass transfers)
# ----------------------------------------------------------------------
def _anneal_refine(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Perform a few thousand random mass‑transfer moves (simulated annealing)
    starting from `a0`.  The routine stops when time runs out.
    Returns the best vector seen and its exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    # total number of attempted moves – limited by time
    max_moves = 120_000

    for it in range(1, max_moves + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (very gentle)
        T = 0.02 * (1e-6 / 0.02) ** (it / max_moves)   # from 0.02 to 1e-6

        # pick source proportional to its current mass (keeps chance of moving mass)
        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # propose moving a random fraction (≤20 %) of src to dst
        eps = a[src] * rng.random() * 0.20
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or math.exp(-delta / max(T, 1e-12)) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score
