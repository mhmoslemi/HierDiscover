def propose_candidate(seed=0, budget_s=1000):

    # load previous best if exists in global variable height_sequence_1

            starter = starter / starter.sum()
    best_seq = starter.copy()

    # Phase 2: subgradient descent (more aggressive)
    if time.time() < deadline - 3:
        a_sub, sc_sub = _subgradient_descent(best_seq, n, deadline - 2, start, best_score, step0=0.6)

            best_seq, best_score = a_sub, sc_sub

    # Phase 3: targeted annealing
    if time.time() < deadline:
        a_final, sc_final = _targeted_annealing(best_seq, n, deadline, start, best_score,
                                                start_temp=5e-2, end_temp=1e-7, steps=800_000)
        if sc_final < best_score:
            best_seq, best_score = a_final, sc_final
            print(f"[{time.time()-start:.2f}s] Anneal improved {best_score:.6f}")

    # final scaling to avoid negative tiny values and ensure sum 1
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:

def _smooth_descent_adam(a0, n, deadline, start_time):
    a = a0.copy()
    x = np.log(a + 1e-20)   # log-odds
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)
    # schedule alphas
    alphas = np.geomspace(0.5, 800.0, num=40)
    total_iters = 300_000
    it_per_alpha = max(1, total_iters // len(alphas))
    for alpha in alphas:
        # for each alpha, iterate it_per_alpha steps
        for _ in range(it_per_alpha):
            if time.time() >= deadline - 0.1:
                break
            t += 1
            # compute convolution
            conv = _conv_self(a, L_self)
            # softmax weights
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum
            # compute gradient wrt a
            a_rev = a[::-1]
            grad_full = _conv_linear(w, a_rev, L_grad)
            grad_a = 2.0 * grad_full[n - 1:n - 1 + n]
            # gradient wrt x (logits)
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = 0.1 / (np.sqrt(alpha) * np.sqrt(t) + 1e-12)  # schedule
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            # convert back to simplex
            # to improve numeric stability subtract max
            x_max = np.max(x)
            a = np.exp(x - x_max)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum
            # occasional evaluation
            if t % 1000 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time()-start_time:.2f}s] Adam smooth alpha={alpha:.2f} it={t:,} score={cur_score:.6f}")
        if time.time() >= deadline - 0.1:
            break
    return best_a, best_score

def _targeted_annealing(a0, n, deadline, start_time, cur_best_score, cur_best_seq,
                        start_temp=5e-2, end_temp=1e-7, steps=500_000):
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    L_self = _next_pow2(2 * n - 1)
    score = 2.0 * n * _conv_self(a, L_self).max() / (a.sum() ** 2)
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # compute current convolution and max index periodically
        if it % 200 == 0:
            conv = _conv_self(a, L_self)
            kmax = int(conv.argmax())
        # compute contributions for indices around kmax
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        # compute contribution array
        # product for each i in lo..hi
        contrib = a[lo:hi + 1] * a[kmax - np.arange(lo, hi + 1)]
        src_rel = int(np.argmax(contrib))
        src = lo + src_rel
        # choose destination random (can be different)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.4  # up to 40% move
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop, L_self).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] Targeted anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

"""
Propose a 4096‑long non‑negative step‑function that minimises

      2 * n * max(conv(a, a)) / (sum(a)**2)

The routine combines three optimisation phases
------------------------------------------------
1️⃣  Adam‑mirrored smooth‑max descent on the soft‑max approximation of the
    max‑convolution.  The iterates stay on the probability simplex by
    exponentiating log‑weights.  A geometric schedule drives the smoothing
    parameter α from 0.5 up to 800, while an Adam step‑size schedule adapts
    per‑coordinate learning rates.

2️⃣  Projected sub‑gradient descent with momentum (the classic “max‑bound”
    optimiser).  This refines the smooth‑max solution.

3️⃣  Targeted simulated‑annealing polish.  Each move transfers a fraction of
    mass from the index that contributes most to the current maximal
    convolution coefficient to a random destination.  The Metropolis rule
    with an exponential cooling schedule enables escaping shallow minima.

The whole pipeline respects a 1000 s wall‑clock budget and returns the best
sequence found (as a plain Python list of floats).  The code is fully
self‑contained and uses only NumPy + the supplied ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast exact evaluation for vectors that already sum to 1
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).  a is non‑negative."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# 1️⃣  Adam‑mirrored smooth‑max descent
# ----------------------------------------------------------------------
def _smooth_descent_adam(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation of max(conv(a,a)).
    Updates are performed on log‑probabilities (softmax parametrisation)
    using Adam.  α is increased geometrically.
    """
    a = a0.copy()
    # initialise log‑variables safely
    x = np.log(a + 1e-20)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    # geometric schedule for the smoothing parameter (α)
    alphas = np.geomspace(0.5, 800.0, num=40)
    total_iters = 300_000
    it = 0

    for alpha in alphas:
        # inner loop runs until either the global deadline or a quota per α
        while time.time() < deadline - 0.2 and it < total_iters:
            it += 1

            # 1) self‑convolution of the current iterate
            conv = _conv_self(a, L_self)

            # 2) soft‑max weights w ∝ exp(α·conv)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # stability shift
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient of the smooth‑max w.r.t. a  (2·(w⋆rev(a)))
            a_rev = a[::-1]
            grad_full = _conv_linear(w, a_rev, L_grad)
            grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) convert gradient to the log‑space (softmax Jacobian)
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

            # bias‑corrected estimates
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # learning‑rate schedule – smaller for large α and later it
            lr = 0.12 / (np.sqrt(alpha) * np.sqrt(it) + 1e-12)

            # log‑update
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # back‑to the simplex (soft‑max)
            x_max = np.max(x)
            a = np.exp(x - x_max)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

            # occasional exact‑objective check
            if it % 1000 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Projected sub‑gradient descent (as in the reference solution)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float,
                         step0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses momentum (β≈0.9) and a 1/√t step‑size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        # active index (first occurrence of the max)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # ∂conv_k/∂a_i = 2·a_{k-i}

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted simulated‑annealing polish
# ----------------------------------------------------------------------
def _targeted_annealing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        start_temp: float = 5e-2,
                        end_temp: float = 1e-7,
                        steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer annealer that preferentially takes mass away from the
    index contributing most to the current maximal convolution coefficient.
    """
    a = a0.copy()
    L_self = _next_pow2(2 * n - 1)
    # current objective
    score = 2.0 * n * _conv_self(a, L_self).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    # cache the current max‑index for a few iterations to avoid recomputing
    kmax = int(_conv_self(a, L_self).argmax())

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        # refresh the maximal index periodically
        if it % 250 == 0:
            conv = _conv_self(a, L_self)
            kmax = int(conv.argmax())

        # indices that actually contribute to conv[kmax]
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        # contribution = a[i] * a[kmax-i]
        contrib = a[idx_range] * a[kmax - idx_range]
        src = idx_range[int(np.argmax(contrib))]   # worst‑off index

        # pick a random destination (different from src)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-16:
            continue

        # transfer a fraction of src to dst
        eps = a[src] * rng.random() * 0.35   # up‑to 35 % move
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop, L_self).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Targeted anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Load a starter (previous best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        starter = None

    if starter is None or starter.ndim != 1 or starter.size == 0:
        starter = np.full(n, 1.0 / n)
    else:
        if starter.size != n:
            if starter.size > n:
                starter = starter[:n]
            else:
                starter = np.pad(starter, (0, n - starter.size), mode='constant')
        starter = np.maximum(starter, 0.0)

    s = starter.sum()
    starter = starter / s if s > 0 else np.full(n, 1.0 / n)

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Adam‑mirrored smooth‑max descent (core optimiser)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        best_seq, best_score = _smooth_descent_adam(best_seq, n, deadline, start_time)
        print(f"[{time.time() - start_time:.2f}s] After Adam smooth best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient refinement (as in the reference)
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        a_sub, sc_sub = _subgradient_descent(best_seq, n, deadline - 2.0,
                                             start_time, best_score, step0=0.6)
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Targeted annealing polish (final refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _targeted_annealing(best_seq, n, deadline, start_time,
                                                best_score, best_seq,
                                                start_temp=5e-2, end_temp=1e-7,
                                                steps=500_000)
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain‑Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
