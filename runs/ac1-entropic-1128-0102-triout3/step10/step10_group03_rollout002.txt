"""
Exponential‑gradient descent for the max‑auto‑correlation problem.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)**2)

with a_i ≥ 0.  The ratio is invariant to scaling, so we keep
sum(a)=1 (simplex).  The algorithm starts from the known best
construction (or uniform), then repeatedly performs a soft‑max
approximation of the max, computes the gradient via FFT and updates
the coefficients with an *exponential* (multiplicative) step.
The update automatically stays on the simplex after normalisation.
Several random restarts, a final hill‑climbing polish and a short
refinement phase are combined within the 1000 s budget.

The code follows the required interface: a top‑level `propose_candidate`
function that returns a list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Zero‑padding length L is chosen automatically if omitted.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Fast exact evaluation of   2·n·max(conv(a,a))   (the sum is assumed ≈1).
    """
    conv = _conv_self(a, L=L_fft)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Exponential‑gradient descent (soft‑max smoothing)
# ----------------------------------------------------------------------


def _expgrad_descent(start_a: np.ndarray,
                     deadline: float,
                     start_time: float,
                     L_fft: int,
                     L_grad: int) -> tuple[np.ndarray, float]:
    """
    Multiplicative‑gradient descent on a smoothed version of the objective.
    The smooth max is approximated by a soft‑max with a temperature τ that
    is exponentially decayed.  The update is a_i ← a_i·exp(−η·grad_i) and
    the vector is re‑normalised to stay on the simplex.
    Returns the best sequence found during the run.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)

    it = 0
    step0 = 0.25            # initial step size
    tau = 0.10              # initial soft‑max temperature
    tau_min = 1e-8
    tau_decay = 0.9997

    while time.time() < deadline:
        it += 1

        # convolution a*a
        conv = _conv_self(a, L=L_fft)
        maxc = conv.max()

        # soft‑max weights over the 2n‑1 entries
        exp_vals = np.exp((conv - maxc) / tau)
        w = exp_vals / exp_vals.sum()            # ∑w = 1

        # gradient of the smoothed objective:
        #   ∂/∂a_j  (∑_k w_k (a⋆a)_k)  = 2 * (a * w_rev)_j
        w_rev = w[::-1]                          # reversed weights, length 2n‑1
        Af = np.fft.rfft(a, n=L_grad)
        Wf = np.fft.rfft(w_rev, n=L_grad)
        grad_full = np.fft.irfft(Af * Wf, n=L_grad)
        offset = 2 * n - 2
        grad = 2.0 * grad_full[offset - np.arange(n)]

        # multiplicative (exponential) update
        step = step0 / math.sqrt(it)
        # Clip the gradient to avoid overflow in the exponential
        grad_clipped = np.clip(grad, -40.0, 40.0)
        a = a * np.exp(-step * grad_clipped)

        # simplex normalisation (KL‑projection)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # true objective (exact max)
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGrad it={it:,} τ={tau:.2e} score={cur_score:.6f}")

        # temperature decay
        tau = max(tau_min, tau * tau_decay)

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing polish (larger moves)
# ----------------------------------------------------------------------


def _hill_climb(start_a: np.ndarray,
                deadline: float,
                start_time: float,
                L_fft: int) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a random fraction of mass from a source
    index to a destination index and keep the move only if the true
    objective improves.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_fft)
    rng = np.random.default_rng()
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move a random fraction (1 % … 30 %) of the source mass
        eps_frac = rng.random() * 0.29 + 0.01
        eps = a[src] * eps_frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the objective.
    The routine mixes:
        • random restarts with exponential‑gradient descent (soft‑max)
        • a final hill‑climbing polish
    and respects the 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 0.5

    # length of the candidate – 4096 works well for the time budget
    n = 4096
    L_fft = _next_pow2(2 * n - 1)      # for self‑convolution
    L_grad = _next_pow2(3 * n)         # for a * w_rev convolution

    # --------------------------------------------------------------
    # initialise with a known good construction or uniform
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # take the global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s == 0:
                starter = uniform.copy()
            else:
                starter = starter / s
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
            print(f"[{time.time() - start_time:.2f}s] Global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Random‑restart exponential‑gradient descent
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:        # reserve time for polishing
        remaining = deadline - time.time()
        # allocate a modest slice for this restart (max 12 s)
        per_restart = min(12.0, remaining * 0.5)

        # start point: perturb the current best (70 % chance) or uniform
        base = best_seq if np.random.rand() < 0.7 else uniform
        noise = 0.04 * np.random.randn(n) * base   # relative Gaussian noise
        start = base + noise
        start = np.maximum(start, 0.0)
        start = _project_simplex(start)            # back to the simplex

        sub_deadline = time.time() + per_restart
        cand_seq, cand_score = _expgrad_descent(start, sub_deadline,
                                                start_time, L_fft, L_grad)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best (exp‑grad) score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Final hill‑climbing polish (larger mass moves)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a_hc, sc_hc = _hill_climb(best_seq, deadline, start_time, L_fft)
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb final improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
