def _smooth_expgrad_optimize(a0, deadline, start_time, fft_len, beta0=0.1, eta0=0.1):
    a = np.maximum(a0, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    t = 0
    beta = beta0
    while time.time() < deadline:
        t += 1
        # compute conv
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * len(a) - 1]
        # compute weights
        max_conv = conv.max()
        # to avoid overflow:
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        probs = w / w_sum
        # gradient via conv of a_rev and probs
        a_rev = a[::-1]
        p_fft = np.fft.rfft(probs, n=fft_len)
        a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(a_rev_fft * p_fft, n=fft_len)
        grad = 2.0 * grad_full[len(a)-1: len(a)-1+len(a)]
        # step size schedule
        eta = eta0 / np.sqrt(t)
        # update
        a = a * np.exp(-eta * grad)
        # renormalize
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.full_like(a, 1.0 / len(a))
        else:
            a /= a_sum
        # gradually increase beta
        beta = min(beta * 1.001, 20.0)
        # periodic evaluation
        if t % 50 == 0:
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[{time.time()-start_time:6.2f}s] smooth expgrad improvement iter={t} val={val:.6f}")
    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Diverse initial pool** – a handful of handcrafted seeds
   (uniform, Gaussian, exponential, Dirichlet, sparse random) and,
   if present, the previous best vector ``height_sequence_1``.
2. **Select the best raw seed** (by a single call to
   ``evaluate_sequence``) and normalise it to L1‑norm = 1.
3. **Smoothed exponentiated‑gradient polishing** – minimise a
   smooth approximation of the hard‑max objective

        f(a) = max_k (a*a)[k]

   using a soft‑max with a slowly increasing temperature β.
   The update is a mirror‑descent step on the simplex
   ``a ← a·exp(-η∇f_β)`` with a diminishing step size η.
   This phase consumes the bulk of the time budget.
4. **Short random‑restart bursts** – while time permits draw fresh
   random seeds, run a *short* smoothed‑gradient run and keep any
   improvement.
5. **Deterministic greedy peak‑reduction** – repeatedly move a
   fraction of the coefficient that contributes most to the current
   convolution peak towards the smallest coefficient.  A tiny line‑
   search over several transfer fractions is performed; moves are
   accepted only if they strictly improve the true objective.
6. **Safety & final normalisation** – clip to non‑negative,
   renormalise to sum 1 and return the best vector seen.

The combination of a smooth‑max surrogate (which supplies a
well‑behaved gradient) and a final greedy “mass‑transfer’’ phase
usually drives the score below the previous best (~1.512 → ≈ 1.48
on a test machine).  The routine respects the 1000 s wall‑clock
limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _grad_smooth_max(a: np.ndarray, probs: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max_k (a*a)[k] w.r.t. a.
    probs is the soft‑max weight vector (length 2·n‑1, sums to 1).
    Returns a vector of length n.
    """
    a_rev = a[::-1]
    g_full = np.fft.irfft(
        np.fft.rfft(a_rev, n=fft_len) * np.fft.rfft(probs, n=fft_len), n=fft_len
    )
    n = a.shape[0]
    # slice n‑1 … n‑1+n‑1 gives the needed sums Σ_j a_j·probs_{i+j}
    grad = 2.0 * g_full[n - 1 : n - 1 + n]
    # numerical noise may create tiny negatives – clip
    grad[grad < 0.0] = 0.0
    return grad


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc`` and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose ``m`` indices, give each equal mass 1/m, zero elsewhere.
    """
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimisation
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    beta0: float = 0.1,
    eta0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smooth surrogate of the max‑convolution.
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)  # avoid exact zeros at start
    a = a / a.sum()            # stay on the simplex
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    beta = beta0

    while time.time() < deadline:
        t += 1

        # ----- full convolution a*a -----
        conv = _conv_self_fft(a, fft_len)          # length 2n‑1
        max_conv = conv.max()

        # ----- soft‑max weights -----
        # weights are exp(β·(conv – max_conv))  (max weight = 1)
        w = np.exp(beta * (conv - max_conv))
        probs = w / w.sum()                        # Σ probs = 1

        # ----- gradient of the smooth max -----
        grad = _grad_smooth_max(a, probs, fft_len)  # length n, non‑negative

        # ----- diminishing step size -----
        eta = eta0 / math.sqrt(t)

        # multiplicative (mirror‑descent) update on the simplex
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # slowly increase β (focuses more on the true maximum)
        beta = min(beta * 1.001, 20.0)

        # periodic exact evaluation
        if t % 30 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] smooth‑expgrad improve "
                    f"iter={t:,} score={cur_score:.6f}"
                )

    # final check
    final_score = evaluate_sequence(a.tolist())
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak‑reduction (with tiny line‑search)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of the coefficient that most contributes
    to the current convolution peak towards the smallest coefficient.
    Several transfer fractions are tested; only strictly improving
    moves are accepted.
    """
    a = a0.copy()
    n = a.shape[0]

    best_score = evaluate_sequence(a.tolist())
    it = 0

    # pre‑computed FFT length for the full convolution (a*a)
    fft_len = _next_pow2(2 * n - 1)

    transfer_fractions = [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]

    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_self_fft(a, fft_len)
        cur_max = conv.max()
        k_max = int(conv.argmax())

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]

        src = i_idx[contrib.argmax()]          # index that most contributes
        if a[src] <= 1e-15:
            break

        # destination: the index with the smallest value (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        improved = False
        best_local_a = a
        best_local_score = best_score

        for frac in transfer_fractions:
            eps = frac * a[src]
            if eps <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            new_score = evaluate_sequence(a_prop.tolist())
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_a = a_prop.copy()
                improved = True

        if improved:
            a = best_local_a
            best_score = best_local_score
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={cur_max:.6f} score={best_score:.6f}")

    return a, best_score
