"""
Optimiser for the step‑function upper‑bound.

The optimiser works in three stages:

1. **Two‑level exhaustive search** – the sequence is restricted to a
   simple “high‑plateau / low‑plateau’’ shape.  The maximal convolution
   of such a shape can be evaluated analytically in O(n) time, so we
   sweep a grid of plateau lengths and heights and keep the best
   candidate.

2. **Exponentiated‑gradient fine‑tuning** – starting from the best
   two‑level candidate we run a mirror‑descent (exponentiated‑gradient)
   algorithm on the true ∞‑norm of the self‑convolution.  The update
   uses the sub‑gradient given by the active convolution index and
   projects back onto the probability simplex.

3. **Deterministic greedy sweep** – a fast sweep that moves a tiny
   amount of mass from the coordinate that contributes most to the
   current peak to a coordinate that contributes the least, repeatedly
   reducing the peak.  The loop stops when the time budget is exhausted.

All phases respect the global 1000 s budget.  The final sequence is
returned as a plain Python list of floats and is already normalised
to sum to 1 (the objective is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Self‑convolution a*a using FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric distribution a_i ∝ r^i on n points (0<r<1).
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    # normalise (finite sum = (1-r^n))
    norm = 1.0 - r ** n
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


# ----------------------------------------------------------------------
# Analytic max‑convolution for a single high/low plateau
# ----------------------------------------------------------------------
def _max_conv_two_step(m: int, a: float, b: float, n: int) -> float:
    """
    Compute the maximal entry of the self‑convolution of a vector that
    consists of `m` copies of value `a` followed by `n-m` copies of `b`.
    The vector is assumed to be normalised (sum = 1), i.e. a and b satisfy
    m*a + (n-m)*b = 1.
    """
    L2 = n - m
    a2 = a * a
    b2 = b * b
    ab = a * b
    max_val = 0.0
    # total number of pairs that contribute to index k
    for k in range(2 * n - 1):
        tot = min(k, n - 1) - max(0, k - (n - 1)) + 1

        # pairs completely inside the high plateau
        if k <= 2 * m - 2:
            if k < m:
                N_aa = k + 1
            else:
                N_aa = 2 * m - 1 - k
        else:
            N_aa = 0

        # pairs completely inside the low plateau
        t = k - 2 * m
        if 0 <= t <= 2 * L2 - 2:
            if t < L2:
                N_bb = t + 1
            else:
                N_bb = 2 * L2 - 1 - t
        else:
            N_bb = 0

        N_ab = tot - N_aa - N_bb
        val = N_aa * a2 + N_ab * ab + N_bb * b2
        if val > max_val:
            max_val = val
    return max_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the infinity‑norm of the convolution.
    The step uses multiplicative updates (mirror descent on the simplex).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

        # active index and its gradient
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # step size
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # re‑project onto the simplex
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the coordinate that most
    contributes to the current peak to a coordinate that contributes the
    least.  The convolution is updated incrementally.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full convolution (length L)
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually take part in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and candidate destinations
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0

        # vector of potential partners at the same max index
        a_partner = np.zeros(n)
        a_partner[idx] = a[max_idx - idx]

        # choose destination with smallest partner value (but not src)
        dst_cands = np.where(a_partner == a_partner.min())[0]
        dst_cands = dst_cands[dst_cands != src]
        if dst_cands.size == 0:
            dst = (src + 1) % n
        else:
            dst = dst_cands[np.argmin(a_partner[dst_cands])]

        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # step size – try to close the gap to the second maximum
        if delta_a > 0.0:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            eps = max(eps, 1e-12)
        else:
            eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental convolution update -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # length of the sequence – 4096 gives a good trade‑off between quality
    # and evaluation speed.
    n = 4096

    # ------------------------------------------------------------------
    # Baseline (uniform) – just to have something to compare against.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = 2.0 * n * np.max(_conv_self(best_seq, _next_pow2(2 * n - 1)))
    print(f"[0.00s] uniform baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – fast exhaustive search over a two‑plateau family.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] phase 1 start (two‑level search)")
        # grid of plateau lengths
        m_vals = np.linspace(1, n - 1, num=200, dtype=int)
        for m in np.unique(m_vals):
            if time.time() >= deadline:
                break
            # admissible range for the high plateau height a
            a_min = 1.0 / n
            a_max = 1.0 / m
            if a_max <= a_min:
                continue
            a_grid = np.linspace(a_min, a_max, num=120)
            for a in a_grid:
                b = (1.0 - m * a) / (n - m)
                if b < 0.0:
                    continue
                max_conv = _max_conv_two_step(int(m), a, b, n)
                score = 2.0 * n * max_conv
                if score < best_score:
                    best_score = score
                    best_seq = np.concatenate(
                        [np.full(int(m), a), np.full(n - int(m), b)]
                    )
                    best_seq = np.maximum(best_seq, 0.0)
                    best_seq /= best_seq.sum()
                    print(f"[{time.time() - start_time:.2f}s] "
                          f"two‑level improve: m={m}, a={a:.8f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient fine‑tuning.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = time.time() + max(30.0, 0.30 * (deadline - time.time()))
        a_exp, score_exp = _run_exp_gradient(
            best_seq,
            n,
            phase2_deadline,
            start_time,
            eta0=1.0,
        )
        if score_exp < best_score:
            best_score = score_exp
            best_seq = a_exp.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 2 improve: "
                  f"score={score_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy reduction of the peak.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase3_deadline = deadline
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq,
            n,
            phase3_deadline,
            start_time,
            best_score,
            best_seq,
            max_iters=2_000_000,
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] phase 3 improve: "
                  f"score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and sanity checks.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_obj = 2.0 * n * np.max(_conv_self(best_seq, _next_pow2(2 * n - 1)))
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
