def _softmax_weights(conv, beta):
    maxc = conv.max()
    exps = np.exp(beta * (conv - maxc))
    w = exps / exps.sum()

def _run_adam_smooth(a0, n, deadline, start_time, beta_init=5.0, beta_max=50.0, beta_rate=0.001, lr=0.01, eps=1e-8):
    a = a0.copy()

    best_score = float("inf")
    it = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    while time.time() < deadline:
        it += 1
        # increase beta to sharpen
        beta = min(beta_max, beta_init + beta_rate * it)
        grad, conv = _grad_smooth_max(a, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _project_simplex(a_new)
        a = a_new
        cur_score = 2.0 * n * conv.max()

            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam iter={it:,} beta={beta:.2f} score={cur_score:.6f}")

"""
Propose a step‑function (non‑negative vector) minimizing ``evaluate_sequence``.

Algorithm
---------
1. Initialise from a few deterministic shapes (uniform, linear,
   sqrt‑law, exponential).  If a previous construction ``height_sequence_1``
   is available it is interpolated as well.  The best start is selected
   by a very short exponentiated‑gradient run.

2. Long exponentiated‑gradient descent (mirror‑descent on the true
   max‑convolution).  This reduces the peak of the self‑convolution
   directly.

3. Smooth‑max descent using Adam on a log‑sum‑exp surrogate of the
   max‑convolution.  The surrogate temperature (β) is increased
   gradually, turning the objective into the true ``max`` at the end.

4. Deterministic greedy “peak flattening’’ that moves a tiny amount of
   mass from the index contributing most to the current peak to an
   index contributing least, updating the convolution incrementally.

5. If time permits the sequence is up‑sampled (by a factor 2) and the
   optimisation steps 2‑4 are repeated on the larger problem.

All updates keep the vector on the probability simplex (sum = 1) and
respect non‑negativity.  The routine respects a 1000 s budget, prints
progress and finally returns the best sequence found as a plain Python
list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                               # fallback (should not happen)
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) via FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) optimiser that minimises the true
    max‑convolution.  Returns the best vector seen and its objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(2 * n - 1)
    it = 0

    while time.time() < deadline:
        it += 1

        conv = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # sub‑gradient from the active (max) convolution index
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # d/d a_i of conv[k]

        # multiplicative mirror‑descent update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # project onto the simplex (exact sum = 1)
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – smooth‑max descent via Adam (additive updates)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Softmax over convolution values with temperature 1/beta."""
    maxc = conv.max()
    exps = np.exp(beta * (conv - maxc))
    w = exps / exps.sum()
    return w


def _grad_smooth_max(a: np.ndarray, beta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient of the smooth‑max (log‑sum‑exp) of the self‑convolution.
    Returns (grad, conv) where conv is a*a.
    """
    n = a.shape[0]
    # self‑convolution
    conv = _conv_self(a)

    # softmax weights
    w = _softmax_weights(conv, beta)

    # gradient = 2 * (a_rev * w) restricted to n entries
    a_rev = a[::-1]
    L = _next_pow2(3 * n - 2)
    Af = np.fft.rfft(a_rev, n=L)
    Wf = np.fft.rfft(w, n=L)
    prod = Af * Wf
    full = np.fft.irfft(prod, n=L)            # linear convolution (length ≥ 3n‑2)
    grad = 2.0 * full[n - 1: n - 1 + n]
    return grad, conv


def _run_adam_smooth(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     beta0: float = 5.0,
                     beta_max: float = 80.0,
                     beta_rate: float = 5e-4,
                     lr: float = 0.02,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    beta grows linearly so that the surrogate approaches the true max.
    """
    a = a0.copy()
    best_a = a.copy()
    # evaluate initial objective
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    it = 0

    while time.time() < deadline:
        it += 1
        beta = min(beta_max, beta0 + beta_rate * it)

        grad, conv = _grad_smooth_max(a, beta)

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_tmp = a - step
        a = _project_simplex(a_tmp)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam iter={it:,} "
                  f"beta={beta:.2f} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – deterministic greedy reduction (same as previous code)
# ----------------------------------------------------------------------


def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the index that most
    contributes to the current peak to an index that contributes least.
    Full recomputation of the convolution is performed periodically.
    """
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self(a)                     # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second highest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices actually participating in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # pick destination with smallest partner value
        partner_vals = a[max_idx - idx]
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a + 1e-20)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0.0] = 0.0

        # occasional exact recomputation
        if it % recompute_interval == 0:
            conv = _conv_self(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the prescribed time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so we always finish before the hard deadline
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size hierarchy – start modest, then upscale if time permits.
    # ------------------------------------------------------------------
    size_hierarchy = [2048, 4096, 8192]   # increasing flexibility
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to build an initial candidate for a given length.
    # ------------------------------------------------------------------
    def make_initial(n):
        candidates = []

        # 1) previous construction, if available
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                xs_old = np.linspace(0, 1, starter.size)
                xs_new = np.linspace(0, 1, n)
                interp = np.interp(xs_new, xs_old, starter)
                interp = np.maximum(interp, 0.0)
                interp = interp / interp.sum()
                candidates.append(interp)
        except Exception:
            pass

        # 2) uniform
        candidates.append(np.full(n, 1.0 / n))

        # 3) linear decreasing
        lin = np.arange(n, 0, -1, dtype=float)
        candidates.append(lin / lin.sum())

        # 4) sqrt‑law
        sqrt_shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
        candidates.append(sqrt_shape / sqrt_shape.sum())

        # 5) exponential decay
        q = 0.9995
        exp_shape = (1 - q) * (q ** np.arange(n, dtype=float))
        candidates.append(exp_shape / exp_shape.sum())

        # pick the best after a tiny exp‑gradient burst
        local_best = None
        local_score = float("inf")
        for cand in candidates:
            # a very short run (few hundred iterations) just to gauge quality
            short_deadline = time.time() + min(2.0, (deadline - time.time()) * 0.01)
            seq, scr = _run_exp_gradient(cand, n, short_deadline, start_time, eta0=0.8)
            if scr < local_score:
                local_score = scr
                local_best = seq.copy()
        return local_best if local_best is not None else np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Iterate over increasing sizes
    # ------------------------------------------------------------------
    for n in size_hierarchy:
        if time.time() >= deadline:
            break

        # initialise for current size
        a = make_initial(n)

        # ---------------- Phase 1: long exponentiated‑gradient ----------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            phase_deadline = time.time() + min(remaining * 0.30, 30.0)
            seq, scr = _run_exp_gradient(a, n, phase_deadline, start_time, eta0=0.6)
            if scr < best_score:
                best_score = scr
                best_seq = seq.copy()
                a = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] n={n} exp‑grad best={scr:.6f}")

        # ---------------- Phase 2: Adam on smooth‑max ----------------
        if time.time() < deadline:
            remaining = deadline - time.time()
            phase_deadline = time.time() + min(remaining * 0.45, 45.0)
            seq, scr = _run_adam_smooth(a, n, phase_deadline, start_time,
                                        beta0=5.0, beta_max=80.0,
                                        beta_rate=5e-4, lr=0.02)
            if scr < best_score:
                best_score = scr
                best_seq = seq.copy()
                a = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] n={n} adam best={scr:.6f}")

        # ---------------- Phase 3: greedy peak flattening ----------------
        if time.time() < deadline:
            seq, scr = _run_greedy_reduction(a, n, deadline, start_time, best_score)
            if scr < best_score:
                best_score = scr
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] n={n} greedy final={scr:.6f}")

        # if we are not yet at the largest size, prepare for up‑sampling
        if n != size_hierarchy[-1]:
            # up‑sample to the next size (linear interpolation)
            next_n = size_hierarchy[size_hierarchy.index(n) + 1]
            xs_old = np.linspace(0, 1, n)
            xs_new = np.linspace(0, 1, next_n)
            a = np.interp(xs_new, xs_old, best_seq)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            # best_seq will be replaced in the next loop iteration

    # ------------------------------------------------------------------
    # Final normalisation (objective is homogeneous)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform of smallest size we attempted
        n = size_hierarchy[0]
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # return as plain Python list
    return [float(x) for x in best_seq]
