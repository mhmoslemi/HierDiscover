def _smooth_expgrad_optimize(a0, deadline, start_time, fft_len_self, fft_len_corr, alpha0=10.0, eta0=0.5):
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float('inf')
    t=0
    alpha = alpha0
    while time.time() < deadline:
        t+=1
        # compute conv self via FFT
        Af = np.fft.rfft(a, n=fft_len_self)
        conv_full = np.fft.irfft(Af * Af, n=fft_len_self)[:2*n-1]
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max  # since sum_a=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] smooth ExpGrad improve: iter={t},score={cur_score:.6f}")

        # compute weights
        w = np.exp(alpha * (conv_full - cur_max))

        w = w / w.sum()

"""
Propose a step function (non‑negative sequence) that (almost) minimises the
``evaluate_sequence`` objective.

Main ideas
----------
* Build a diverse pool of initial vectors (uniform, Gaussian, exponential,
  Dirichlet with many concentrations and sparse random supports).
* Keep the best raw candidate.
* **Smoothed exponentiated‑gradient descent**:
  – instead of using a sub‑gradient from the single active shift,
    a *soft‑max* (log‑sum‑exp) of all convolution values is used.
  – the resulting gradient is a weighted sum of the gradients of all shifts
    and can be obtained efficiently by a single FFT‑based correlation.
  – the update is multiplicative (mirror descent on the simplex) with a
    diminishing step size.
* Random‑restart bursts: while time permits, start from a fresh random vector
  and run a short smoothed‑EG run.
* Deterministic greedy “peak‑reduction’’ as a final polish.
* Safety‑clip, re‑normalise and return the best sequence found.

The algorithm runs completely within the 1000 s wall‑clock budget and
prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with `conc` concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised."""
    sigma = n / 6.0
    xs = np.arange(n, dtype=float)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose `m` positions, give each equal mass 1/m, zero elsewhere.
    """
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Weighted gradient (soft‑max over all shifts) via FFT correlation
# ----------------------------------------------------------------------


def _weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}   (cross‑correlation of w and a)
    using FFTs.  Returns a vector of length len(a).
    """
    n = a.shape[0]
    Af = np.fft.rfft(a, n=fft_len)
    Wf = np.fft.rfft(w, n=fft_len)
    # cross‑correlation `w ⋆ a` (linear, not circular) – see analysis above
    corr = np.fft.irfft(Wf * np.conj(Af), n=fft_len)
    # needed part: first n entries reversed (lags 0,…,‑(n‑1))
    g = 2.0 * corr[:n][::-1]
    return g


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    alpha0: float = 12.0,
    eta0: float = 0.6,
    alpha_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the *smoothed* Chebyshev
    objective  f(a)=max_k (a*a)[k] .
    The smoothing is a log‑sum‑exp with temperature 1/alpha.
    Returns the best vector observed and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0
    alpha = alpha0

    while time.time() < deadline:
        t += 1

        # --- convolution of a with itself ---
        Af = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] smooth‑EG improve: iter={t:,}, score={cur_score:.6f}"
            )

        # --- soft‑max weights over all shifts ---
        w = np.exp(alpha * (conv_full - cur_max))
        w /= w.sum()  # normalise to avoid scaling issues

        # --- weighted gradient via correlation ---
        g = _weighted_gradient(a, w, fft_len)

        # --- step size schedule (diminishing) ---
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # --- project back onto the simplex ---
        s = a.sum()
        if s <= 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --- slowly tighten the smoothing (increase alpha) ---
        if t % 500 == 0:
            alpha = min(alpha * 1.05, alpha_max)

        # --- occasional exact evaluation to guard against drift ---
        if t % 500 == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] exact eval improvement: score={exact:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy deterministic peak‑reduction (same as the previous implementation)
# ----------------------------------------------------------------------


def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Shift a modest fraction of the coefficient that contributes most to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    best_score = 2.0 * n * _conv_self(a, fft_len).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, fft_len).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

            cand_start = _sparse_random_start(n, m, rng)
        else:
            conc = 10 ** rng.uniform(-1.5, 1.0)  # ~0.03 … 10
            cand_start = _dirichlet_start(n, conc=conc, rng=rng)

        sub_deadline = time.time() + 4.0  # short burst
        cand_seq, cand_score = _smooth_expgrad_optimize(
            cand_start, sub_deadline, start_time, fft_len,
            alpha0=8.0, eta0=0.5, alpha_max=150.0
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from restart score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 3 – final deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final safety round‑trip
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    return [float(x) for x in best_seq.tolist()]
