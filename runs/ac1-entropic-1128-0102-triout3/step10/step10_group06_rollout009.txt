"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm Overview
------------------
1. **Initial pool** – a collection of handcrafted and many random Dirichlet
   draws (different concentration parameters).  The best raw candidate
   (by the exact objective) is kept.

2. **Warm‑up** – a few‑thousand Adam steps on a *log‑sum‑exp* surrogate of
   the convolution peak (the code from the reference).  This quickly
   moves the distribution away from bad regions.

3. **Adam on the true max** – the same Adam machinery, but the gradient is
   taken from the *exact* max‑convolution (sub‑gradient formed from all
   active shifts).  This is more aggressive than the multiplicative
   mirror‑descent used previously.

4. **Exponentiated‑gradient polishing** – a long “multi‑gradient’’ phase
   (mirroring the reference code) that drives the remaining peak down.

5. **Deterministic greedy peak‑reduction** – a final local search that moves a
   fraction of the coefficient responsible for the current peak towards the
   smallest coefficient, accepting only strict improvements.

The routine respects a 1000 s time budget, continuously prints progress,
and finally returns the best sequence found (normalised to sum = 1,
entries ≥ 0).  Only ``numpy`` and the provided ``evaluate_sequence`` are
used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    (Kept for compatibility – not used in the final algorithm.)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast (exact) evaluation – used inside the search to avoid the
# heavy security checks of ``evaluate_sequence``.
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """
    Returns the same objective value as ``evaluate_sequence`` but
    without the type‑checking overhead.
    """
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------
def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0.0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) + Adam warm‑up (reference implementation)
# ----------------------------------------------------------------------
def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eps_log: float = 1e-16,
) -> tuple[np.ndarray, float]:
    """
    A few Adam iterations on the smoothed max‑convolution objective.
    Returns the best vector (probability vector) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    # logits (softmax parameters)
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 3000

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = _fast_score(a)

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            # soft‑max to a valid probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if t % 250 == 0:
                sc = _fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}"
                    )
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser that works on the *exact* max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] by summing the gradients
    of the shifts listed in ``active_shifts``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _adam_max_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
    lr0: float = 0.12,
    tol_active: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation directly on the true max‑convolution objective.
    The gradient is the (multi‑)sub‑gradient of the max.
    Returns the best vector ever seen and its exact score.
    """
    # initialise logits so that softmax reproduces a0
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    y = np.log(a)

    best_a = a.copy()
    best_score = _fast_score(a)

    t = 0
    while time.time() < deadline:
        # reconstruct a from logits
        maxy = np.max(y)
        exp_y = np.exp(y - maxy)
        a = exp_y / exp_y.sum()

        # compute convolution, identify active shifts
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        # consider all shifts that are within ``tol_active`` of the current max
        active = np.where(conv >= cur_max - max(tol_active, 1e-15))[0]

        # sub‑gradient
        g = _multi_gradient(a, active)

        # gradient wrt logits (soft‑max chain rule)
        grad_y = a * (g - np.dot(g, a))

        # Adam update
        t += 1
        if t == 1:
            m = np.zeros_like(grad_y)
            v = np.zeros_like(grad_y)
        else:
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))
        y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep track of best exact score
        if t % 200 == 0:
            sc = _fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] Adam‑max improve: iter={t:,}, score={sc:.6f}"
                )
    return best_a, best_score


# ----------------------------------------------------------------------
# Multiplicative mirror‑descent with multi‑gradient (reference code)
# ----------------------------------------------------------------------
def _expgrad_multi_opt(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len: int,
    eta0: float = 0.6,
    delta_frac: float = 0.001,
    top_k: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution
    objective, using a *multi‑gradient* built from the top‑K (or all)
    shifts that are within ``delta_frac`` of the current maximum.
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] Multi‑ExpGrad improve: iter={t:,}, score={cur_score:.6f}"
            )

        # ----- active set -----
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ----- sub‑gradient -----
        g = _multi_gradient(a, active)

        # ----- step‑size schedule -----
        eta = eta0 / math.sqrt(t)

        # ----- exponentiated update -----
        a = a * np.exp(-eta * g)

        # ----- renormalise -----
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        a = np.maximum(a, 1e-15)

        if t % 500 == 0:
            sc = _fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

# Deterministic greedy peak reduction (same as the reference code)
# ----------------------------------------------------------------------
def _greedy_local_improve(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]  # modest fraction of the source mass
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine combines several optimisation phases described above and
    respects the supplied time budget (default 1000 s).
    Returns the best sequence found as a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n_target = 4800                                 # length used by the reference
    fft_len_target = _next_pow2(2 * n_target - 1)

    # --------------------------------------------------------------
    # Build a rich set of starting points (short version + upsample)
    # --------------------------------------------------------------
    starts = []

    # 1) possibly use the previously‑known best construction
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n_target:
                seq = _upsample(seq, n_target)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) handcrafted bases
    starts.append(np.full(n_target, 1.0 / n_target))           # uniform
    starts.append(_exponential_start(n_target, r=0.9995))      # very gentle decay
    starts.append(_exponential_start(n_target, r=0.985))       # stronger decay
    starts.append(_gaussian_start(n_target))                  # bell‑shaped
    starts.append(_dirichlet_start(n_target, conc=2.0, rng=rng))
    starts.append(_dirichlet_start(n_target, conc=0.5, rng=rng))

    # 3) many random Dirichlet draws with varying concentration
    conc_params = [0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_params:
        for _ in range(5):
            starts.append(_dirichlet_start(n_target, conc=conc, rng=rng))

    # --------------------------------------------------------------
    # Evaluate raw starts and keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = _fast_score(a0)
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam warm‑up on a smoothed surrogate (≈10 % of budget)
    # --------------------------------------------------------------
    warmup_deadline = start_time + budget_s * 0.10
    cand_seq, cand_score = _adam_optimize(best_seq, warmup_deadline,
                                          start_time, fft_len_target)
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam warm‑up: {best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 2 – Adam directly on the true max‑convolution (≈30 % of budget)
    # --------------------------------------------------------------
    adam_max_deadline = start_time + budget_s * 0.40
    cand_seq2, cand_score2 = _adam_max_optimize(best_seq, adam_max_deadline,
                                                start_time, fft_len_target)
    if cand_score2 < best_score:
        best_score = cand_score2
        best_seq = cand_seq2.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam‑max: {best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 3 – Multi‑gradient exponentiated‑gradient polishing (≈40 % of budget)
    # --------------------------------------------------------------
    polish_deadline = start_time + budget_s * 0.90
    polished_seq, polished_score = _expgrad_multi_opt(
        best_seq, polish_deadline, start_time, fft_len_target,
        eta0=0.6, delta_frac=0.001, top_k=5
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polishing: {best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 4 – Deterministic greedy peak reduction (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy: {best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Final normalisation / safety checks
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n_target)

    # Report the exact score using the official evaluator (just for logging)
    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    # Return plain Python list as required by the platform
    return [float(x) for x in best_seq.tolist()]
