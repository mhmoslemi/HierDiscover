"""
Propose a non‑negative step function (vector) that (approximately) minimizes
the objective used in ``evaluate_sequence``.  The optimiser works in several
stages:

1.  Small deterministic initialisations (uniform, linear, sqrt, exponential,
    optionally a known previous construction).  The best of those is kept.

2.  A short exponentiated‑gradient (mirror‑descent) run that directly reduces
    the current peak of the self‑convolution.

3.  A long “flatten‑the‑convolution’’ phase that minimises the L2 distance
    between the convolution and the perfectly flat target 1/(2 n‑1).  The
    gradient of this objective can be written as a correlation and is computed
    by FFT, giving an O(n log n) iteration cost.

4.  A deterministic greedy “peak‑flattening’’ sweep (identical to the one
    used previously) that moves a tiny amount of mass from indices that
    contribute most to the current maximum to indices that contribute least.

5.  A tiny random‑restart polishing loop that perturbs the current best
    vector and runs a brief exponentiated‑gradient burst.

All phases respect the overall 1000 s time budget.  The final vector is
projected onto the probability simplex (sum = 1) before being returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·len(a)‑1) using FFT.
    The result is truncated to the exact length.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:          # fallback – shouldn't happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Phase 1 – short exponentiated‑gradient (mirror‑descent) on the max‑conv
# ----------------------------------------------------------------------
def _run_exp_gradient(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) optimiser that directly minimises
    ``max(conv)``.  Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    # initial score
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    best_score = float(2.0 * n * conv.max())
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # --- sub‑gradient at the active (max) convolution index ---
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]          # d/d a_i of conv[k]

        # --- mirror‑descent update (multiplicative) ---
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # --- renormalise onto the simplex (exact sum = 1) ---
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2b – L2 flattening of the convolution
# ----------------------------------------------------------------------
def _gradient_l2(a: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (conv, grad) where ``conv`` = a*a (full) and ``grad`` is the
    gradient of the squared‑error loss
        L = Σ_k (conv_k - 1/(2n‑1))²
    w.r.t. a.  The gradient is 4 * correlation(a, conv - target).
    Both operations are performed via FFT for speed.
    """
    n = a.shape[0]
    # ----- convolution a*a -----
    Lc = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(Af * Af, n=Lc)[: 2 * n - 1]

    # ----- error w.r.t. flat target -----
    target = 1.0 / (2 * n - 1)
    err = conv - target

    # ----- correlation a with err (valid mode) -----
    Lcorr = _next_pow2(3 * n - 2)
    rev_a = a[::-1]
    revAf = np.fft.rfft(rev_a, n=Lcorr)
    Ef = np.fft.rfft(err, n=Lcorr)
    corr_full = np.fft.irfft(revAf * Ef, n=Lcorr)
    # valid part starts at index n‑1 and has length n
    grad = 4.0 * corr_full[n - 1:n - 1 + n]
    return conv, grad


def _run_l2_flatten(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the L2 distance between the self‑convolution and the
    perfectly flat target.  After each step the vector is projected onto the
    simplex.  Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    # ensure we start on the simplex
    a = _project_simplex(np.maximum(a, 0.0))

    # initial score
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    best_score = float(2.0 * n * conv.max())
    best_a = a.copy()
    it = 0
    eta0 = 0.6

    while time.time() < deadline:
        it += 1
        conv, grad = _gradient_l2(a)

        # simple decaying step size
        eta = eta0 / math.sqrt(it)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # evaluate new objective (no need to call evaluate_sequence)
        conv_new = _conv_self(a_new, _next_pow2(2 * n - 1))
        cur_score = float(2.0 * n * conv_new.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] L2‑flatten iter={it:,} "
                  f"score={cur_score:.6f}")

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – deterministic greedy reduction of the peak
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the index that most
    contributes to the current peak to an index that contributes least,
    updating the full convolution incrementally.  A full recomputation
    is performed every few thousand steps to remove numerical drift.
    """
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # choose destination with smallest partner value (and not src)
        partner_vals = a[max_idx - idx]
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} "
                  f"score={cur_score:.6f}")

    return best_a, best_score

    initial_candidates = []

    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore

    # 1b) uniform

    for cand in initial_candidates:

    # ------------------------------------------------------------------

            best_seq,

            start_time,

        )

            best_score = scr

            print(f"[{time.time() - start_time:.2f}s] greedy final improve: "
