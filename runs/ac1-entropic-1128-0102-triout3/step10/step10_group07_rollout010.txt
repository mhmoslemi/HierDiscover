def _smooth_expgrad_optimize(a0, deadline, start_time, L_conv, eta0=0.5, tau_factor=0.5):
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    t = 0
    while time.time() < deadline:
        t += 1
        # conv
        A_fft = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(A_fft * A_fft, n=L_conv)[:2*n-1]
        maxc = conv.max()
        # temperature
        tau = maxc * tau_factor
        # softmax weights
        # shift for numeric stability
        # conv - maxc <=0
        w_exp = np.exp((conv - maxc) / tau)
        w_sum = w_exp.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0/len(conv))
        else:
            w = w_exp / w_sum
        # gradient via correlation
        W_fft = np.fft.rfft(w, n=L_conv)
        # correlation: (w ⋆ a)[i] = sum_j w_{i+j} a_j = slice of irfft(W_fft * conj(A_fft))
        corr_full = np.fft.irfft(W_fft * np.conj(A_fft), n=L_conv)
        start = n - 1
        g = 2.0 * corr_full[start:start+n]
        # step size
        eta = eta0 / (np.sqrt(t) + 1.0)
        # update
        a = a * np.exp(-eta * g)
        # renorm
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):

"""
Smooth exponential‑gradient search for a non‑negative step function that
minimises ``evaluate_sequence``.

The algorithm
----------------
1. Build a diverse pool of initial vectors (uniform, Gaussian, exponential,
   Dirichlet draws, sparse random supports and the optional historic sequence).
2. Keep the best one from the pool.
3. Polish it with a *soft‑max* based exponentiated‑gradient descent.
   Instead of using only the single active shift (as in the previous code),
   a smooth distribution over all convolution shifts is taken – the gradient
   becomes a correlation between the weight vector and the current sequence.
4. While time permits launch many short random‑restart polish phases,
   keeping any improvement.
5. Finish with a deterministic greedy peak‑reduction that moves mass from the
   coefficient that contributes most to the current convolution peak towards
   the smallest coefficient.
6. Clip, renormalise and return the best vector found.

The soft‑max temperature is tied to the current maximal convolution value,
gradients are obtained through FFT‑based correlation, and multiplicative
updates keep the variables on the probability simplex.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution / correlation helpers (FFT based, real input)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) via FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(A * A, n=fft_len)
    return conv_full[: 2 * a.shape[0] - 1]


def _correlate_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Correlation (w ⋆ a)[i] = Σ_j w[i+j] a[j] for i = 0..len(a)-1.
    Returned array has length len(a).
    """
    W = np.fft.rfft(w, n=fft_len)
    A = np.fft.rfft(a, n=fft_len)
    corr_full = np.fft.irfft(W * np.conj(A), n=fft_len)
    start = a.shape[0] - 1
    return corr_full[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly choose `m` indices, give each equal mass 1/m, zero elsewhere.
    """
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient optimiser
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             fft_len: int,
                             eta0: float = 0.5,
                             tau_factor: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with a soft‑max weighting of all convolution shifts.
    Returns the best vector encountered and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # --- convolution a*a -------------------------------------------------
        A_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A_fft * A_fft, n=fft_len)[: 2 * n - 1]
        maxc = conv.max()

        # --- soft‑max weights -------------------------------------------------
        # temperature tied to the current peak
        tau = maxc * tau_factor if maxc > 0 else 1.0
        # exp of non‑positive numbers is safe
        w_exp = np.exp((conv - maxc) / tau)
        w_sum = w_exp.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / w_sum

        # --- gradient via correlation ----------------------------------------
        # g_i = 2 * Σ_j w[i+j] * a[j]
        corr = _correlate_fft(w, a, fft_len)
        g = 2.0 * corr

        # --- multiplicative update (mirror descent) -------------------------
        eta = eta0 / (math.sqrt(t) + 1.0)
        r = -eta * g
        # clip to avoid overflow in the exp
        r = np.clip(r, -50.0, 50.0)
        a = a * np.exp(r)

        # --- renormalise onto simplex ----------------------------------------
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --- occasional exact evaluation --------------------------------------
        if t % 200 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] smoothExpGrad improve: iter={t:,}, score={sc:.6f}"
                )

    # final check (in case the last vector is the best)
    final_sc = evaluate_sequence(a.tolist())
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy local improvement
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    # use the exact objective for the greedy phase
    conv = _conv_self_fft(a, _next_pow2(2 * n - 1))
    best_score = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self_fft(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = int(conv.argmax())

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[np.argmax(contrib)]

        # destination = smallest coefficient (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]               # move a modest fraction
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps

            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score
