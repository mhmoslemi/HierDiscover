"""
Propose a step‑function (list of non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------

* **Multi‑restart Adam optimiser on a smooth approximation of the max‑convolution**.
  The vector `a` is kept on the simplex by representing it as a soft‑max of
  logits `y`.  For a given smoothing temperature `α` we use the soft‑max
  weights `w_k ∝ exp(α·conv_k)` and compute the gradient of the smooth
  objective w.r.t. the logits.  Adam (β₁=0.9, β₂=0.999) updates the logits.
  The temperature is increased geometrically from 0.5 up to a few thousand,
  allowing the optimisation to focus gradually on the true maximum.

* **Short projected sub‑gradient polishing** (as in the reference solution).

* **Light simulated‑annealing polish** – random mass‑transfer moves to escape
  any remaining tiny local minima.

All work is timed; the routine never exceeds the supplied budget (default
1000 s).  The best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """Linear convolution (full length) using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast exact objective (vector assumed non‑negative, sum≈1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    conv = _conv_self(a, L=L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (reference implementation)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    L_grad = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L=L_grad)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Light simulated‑annealing polish
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               cur_best_score: float,
               cur_best_seq: np.ndarray,
               start_temp: float = 5e-2,
               end_temp: float = 1e-7,
               steps: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    score = 2.0 * n * _conv_self(a).max() / (a.sum() ** 2)
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    rng = np.random.default_rng()

    for it in range(1, steps + 1):
        if time.time() >= deadline:
            break
        T = start_temp * (end_temp / start_temp) ** (it / steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = 2.0 * n * _conv_self(a_prop).max() / (a_prop.sum() ** 2)
        delta = new_score - score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            score = new_score

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} best={best_score:.6f}")

    return best_seq, best_score

    """

    v = np.zeros_like(y)

    beta2 = 0.999

    L_grad = _next_pow2(3 * n - 2)

        while time.time() < deadline - 0.2:
            it += 1

            # ---- probability vector ----
            max_y = np.max(y)
            a = np.exp(y - max_y)
            a_sum = a.sum()
            a = a / a_sum    # always sums to 1

            # ---- self‑convolution (full) ----
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # ---- soft‑max over convolution entries ----
            exp_arg = alpha * conv
            max_arg = np.max(exp_arg)
            w = np.exp(exp_arg - max_arg)          # shift for stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # ---- gradient w.r.t. a (smooth max) ----
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

            # ---- gradient w.r.t. logits y (soft‑max Jacobian) ----
            avg = np.dot(grad_a, a)
            grad_y = a * (grad_a - avg)

            # ---- Adam update ----
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)

            lr = lr0 / math.sqrt(it)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---- occasional exact evaluation ----
            if it % 1500 == 0:
                # a is already normalised
                cur_max = conv.max()
                cur_score = 2.0 * n * cur_max / (a_sum ** 2)   # a_sum == 1
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam α={alpha:.1f} it={it:,} score={cur_score:.6f}")

            if time.time() >= deadline - 0.2:
                break

    # fallback – if no improvement recorded, return the final a
    if best_a is None:
        best_a = a.copy()
        best_score = 2.0 * n * _conv_self(best_a).max() / (best_a.sum() ** 2)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    L_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Load a starter vector (global best if present, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
    except Exception:
        starter = np.full(n, 1.0 / n)

    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Adam‑based smooth‑max optimisation (multiple restarts)
    # ------------------------------------------------------------------
    # we allocate about three quarters of the budget to the main optimiser
    main_deadline = deadline - 250.0
    restart_idx = 0
    while time.time() < main_deadline:
        restart_idx += 1
        # choose an initial point: first restart uses the current best,
        # later restarts use random Dirichlet draws with varying concentration
        if restart_idx == 1:
            a0 = best_seq.copy()
        else:
            # random concentration: sometimes sparse, sometimes near‑uniform
            conc = 0.4 if (restart_idx % 3 == 0) else 2.0
            a0 = np.random.dirichlet(np.full(n, conc))

        # run Adam optimiser for the remaining time slice
        slice_deadline = min(main_deadline, time.time() + 60.0)  # at most 60 s per restart
        a_opt, sc_opt = _adam_smooth(a0, n, slice_deadline, start_time)

        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart {restart_idx} improved score = {best_score:.6f}")

        # stop if we are close to the overall deadline
        if time.time() > main_deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Short projected sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        sub_deadline = deadline - 15.0
        a_sub, sc_sub = _subgradient_descent(best_seq, n, sub_deadline,
                                             start_time, best_score)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final annealing polish (uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _annealing(best_seq, n, deadline, start_time,
                                       best_score, best_seq,
                                       start_temp=2e-2, end_temp=1e-7,
                                       steps=400_000)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
