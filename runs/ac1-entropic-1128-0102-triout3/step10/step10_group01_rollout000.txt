def _adam_step(a, grad, m, v, t, lr):
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return a_new, m, v

"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1.  **Parameterized sweep** –  generate candidates from the symmetric
    Beta‑family a_i ∝ (x_i·(1−x_i))^{−α} (the “arcsine” family) and also
    convex mixtures of this shape with the uniform distribution.
    The best of these cheap candidates is kept.

2.  **Projected Adam optimisation** –  starting from the best candidate,
    minimise the maximal self‑convolution entry under the simplex
    constraint (a_i ≥ 0, Σa_i = 1).  The gradient of the max‑entry is
    obtained analytically; Adam‑style adaptive steps and a projection
    onto the probability simplex keep the iterates feasible.  The loop
    runs for the bulk of the time budget.

3.  **Light simulated‑annealing polish** –  a few thousand random
    mass‑transfer moves are attempted with a decreasing temperature to
    possibly escape a shallow local minimum.

The whole search respects the 1000 s wall‑time limit; intermediate
scores are printed for diagnostics.  The returned sequence is a plain
list of non‑negative floats whose sum equals one (the scaling does not
affect the evaluation function).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation of the competition objective:
    2·n·max(conv(seq,seq)) / (sum(seq)**2).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

#  Main entry point
# ----------------------------------------------------------------------

    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The function respects the given
    time budget (default 1000 s) and returns a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a short safety margin for finalisation
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    #  Problem size
    # ------------------------------------------------------------------
    # Larger n allows a finer approximation of the optimal density.
    # 8192 is a comfortable middle ground: good quality while still fast.
    n = 8192

    # ------------------------------------------------------------------
    #  1) Parameter sweep (arcsine family + uniform mixtures)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # grid points (mid‑bin positions) in (0,1)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    # ---------- arcsine‑type sweep ----------
    alphas = np.linspace(0.30, 0.60, 31)           # finer around 0.5
    for alpha in alphas:
        # shape a_i ∝ (x_i·(1−x_i))^{−α}
        w = (x * (1.0 - x) + 1e-30) ** (-alpha)
        a = w / w.sum()
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] arcsine α={alpha:.3f} score={score:.6f}")

    # ---------- mixture with uniform ----------
    uniform = np.full(n, 1.0 / n)
    for lam in np.linspace(0.0, 1.0, 11):   # lam = weight of arcsine
        a = lam * best_seq + (1.0 - lam) * uniform
        # already normalised because both components sum to 1
        score = _objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        print(f"[{time.time() - start_time:.2f}s] mixture λ={lam:.2f} score={score:.6f}")

    # ---------- include possible global starter ----------

                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")

    except Exception:

    # ------------------------------------------------------------------

    a = best_seq.copy()

    v = np.zeros_like(a)        # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    iter_cnt = 0
    best_a = a.copy()
    best_obj = best_score

    # allocate most of the remaining time to this stage
    while True:

        if now >= deadline - 30.0:   # leave ~30 s for polishing
            break
        iter_cnt += 1

        # compute self‑convolution and locate a maximal entry
        conv = _conv_self(a)
        max_idx = int(conv.argmax())
        # gradient of the maximal convolution entry
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        grad = np.zeros_like(a)
        # slice assignment is vectorised and cheap
        idx_range = np.arange(lo, hi + 1)
        grad[lo:hi + 1] = 2.0 * a[max_idx - idx_range]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        # learning rate decays like 1/sqrt(t) – empirical good choice
        lr = 0.5 / math.sqrt(iter_cnt)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the probability simplex (keeps Σa = 1)
        a = _project_simplex(a)

        # evaluate current objective; because Σa = 1, the formula simplifies
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{now - start_time:.2f}s] Adam iter={iter_cnt:,} score={cur_obj:.6f}")

        # occasional random jitter to help escape plateaus
        if (iter_cnt % 5000) == 0:
            jitter = 0.01 * np.random.rand(n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    #  3) Light simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_a.copy()
        cur_score = 2.0 * n * _conv_self(a).max()
        best_local = a.copy()
        best_local_score = cur_score
        rng = np.random.default_rng()
        total_steps = 40000
        for it in range(1, total_steps + 1):
            if time.time() >= deadline:
                break
            # exponential cooling (very gentle)
            T = 0.02 * (1e-6 / 0.02) ** (it / total_steps)

            # pick a source proportional to its current mass
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue

            # transfer a small fraction of src to dst
            eps_move = a[src] * rng.random() * 0.15   # ≤15 %
            a_prop = a.copy()
            a_prop[src] -= eps_move
            a_prop[dst] += eps_move

            new_score = 2.0 * n * _conv_self(a_prop).max()
            delta = new_score - cur_score

            if delta < 0.0 or math.exp(-delta / T) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_local_score:
                    best_local_score = new_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Anneal step={it:,} best={new_score:.6f}")

        if best_local_score < best_obj:
            best_obj = best_local_score
            best_a = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improved to {best_obj:.6f}")

    # ------------------------------------------------------------------
    #  4) Normalise and return as plain list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:

    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_obj:.6f}")
    return [float(v) for v in best_a]
