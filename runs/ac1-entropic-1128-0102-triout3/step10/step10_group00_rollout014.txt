"""
Propose a step‑function (non‑negative heights) that improves the
known bound ≈ 1.5087.

Algorithm
---------
1. Choose a fairly large canvas (n = 32768).  The continuous optimum
   1.5 is approached as n → ∞, so a larger n already gives a
   substantial reduction in the discretisation error.
2. Build several warm‑starts:
   * a power‑tent a_i ∝ min(i,n‑1‑i)^p  (p swept over a small grid);
   * an up‑sample of the best known construction that may be present in
     the global variable ``height_sequence_1``;
   * the uniform distribution (fallback).
   The best of these candidates w.r.t. the exact objective is kept.
3. Refine the chosen start with a **projected sub‑gradient descent**
   on the exact (non‑smooth) objective.  Momentum (β=0.9) and a
   1/√t step‑size are used; after each update the vector is projected
   onto the probability simplex (Σ a_i = 1, a_i ≥ 0).
4. If time permits a tiny **simulated‑annealing** polish is performed
   (tiny mass transfers) to escape possible shallow local minima.
5. Finally the sequence is normalised to sum 1 (the objective is
   scale‑invariant) and returned as a plain Python list.

All heavy work is done with FFT based self‑convolution, which makes a
single iteration O(n log n) and comfortably fits into the 1000 s budget
even for n = 32768.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution (linear, length 2·n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    Identical to ``evaluate_sequence`` but works on a NumPy array.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection using the O(n log n) algorithm of Wang & Carreira‑Perpínan."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     # numerically very unlikely
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent; returns the best iterate seen."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9            # momentum factor
    step0 = 0.5           # initial step‑size (tuned for typical n)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"sub‑grad iter={it:,} score={cur_score:.6f}")

        # index of the maximal convolution entry
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a (see derivation in docstring)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + diminishing step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # project onto the simplex
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Tiny random mass swaps with a fast cooling schedule."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # fast exponential cooling
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)          # more likely to move mass from big piles
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30   # move up to 30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper: up‑sample a short sequence onto a larger grid
# ----------------------------------------------------------------------
def _upsample_sequence(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``old_seq`` onto a grid of length ``new_n``."""
    old_n = old_seq.size
    if old_n == new_n:
        return old_seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, old_seq)
    new_seq = np.maximum(new_seq, 0.0)    # safeguard against tiny negative artefacts
    # normalise to sum 1 (the optimisation works on the simplex)
    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    else:
        new_seq = np.full(new_n, 1.0 / new_n)
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,

    """
    Search for a low‑objective step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so we never overshoot the time limit
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger canvas gives a finer approximation of the
    # continuous optimum (≈ 1.5). 32768 strikes a good balance between
    # quality and speed on a single CPU.
    # --------------------------------------------------------------
    n = 32768

    # --------------------------------------------------------------
    # 1) Build a pool of warm‑starts
    # --------------------------------------------------------------
    candidates = []

    # (a) power‑tent shapes with a small sweep over p
    x = np.arange(n, dtype=float) / (n - 1)          # points in [0,1]
    for p in np.linspace(0.7, 1.3, 13):             # 13 values → cheap
        w = np.minimum(x, 1.0 - x) ** p
        if w.sum() == 0.0:
            continue
        a = w / w.sum()
        candidates.append(a)

    # (b) up‑sample the globally supplied construction, if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        if old.ndim == 1 and old.size > 0:
            old = np.maximum(old, 0.0)
            up = _upsample_sequence(old, n)
            candidates.append(up)
            print(f"[{time.time() - start_time:.2f}s] Upsampled global starter.")
    except Exception:
        pass

    # (c) uniform distribution – a cheap fallback
    candidates.append(np.full(n, 1.0 / n))

    # --------------------------------------------------------------
    # 2) Pick the best warm‑start according to the exact objective
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for i, cand in enumerate(candidates, 1):
        sc = _objective(cand)
        print(f"[{time.time() - start_time:.2f}s] candidate {i}/{len(candidates)} "
              f"score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # 3) Local refinement with projected sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:                     # keep a few seconds for polishing
        sub_deadline = time.time() + remaining * 0.85
        refined_seq, refined_score = _subgradient_descent(best_seq,
                                                          sub_deadline,
                                                          start_time)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Refined score={refined_score:.6f}")

    # --------------------------------------------------------------
    # 4) Very light annealing polish (only if we still have time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_seq, anneal_score = _anneal(best_seq, deadline, start_time)
        if anneal_score < best_score:
            best_score = anneal_score
            best_seq = anneal_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Final anneal improve score={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    # Return a plain Python list as required by the harness
    return [float(v) for v in best_seq]
