def _run_softmax_exponential_gradient(a0, n, deadline, start_time, eta0=0.5, beta_start=5.0, beta_end=150.0):

    a = a0.copy()
    best_a = a.copy()
    best_score = float('inf')
    L = _next_pow2(2 * n - 1)   # for conv (full length)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv via FFT
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[:2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] softmax eg iter={it} score={cur_score:.6f}")
        # schedule beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 200000)   # increase over many iterations
        # compute weights
        max_v = conv.max()
        # shift for numerical stability
        exp_term = np.exp(beta * (conv - max_v))
        w = exp_term / exp_term.sum()
        # compute gradient via cross-correlation
        grad = _gradient_from_weights(a, w)
        # step size
        eta = eta0 / math.sqrt(it)
        # multiplicative update
        # Clip grad magnitude to avoid overflow:
        grad = np.clip(grad, -10.0, 10.0)   # maybe adjust
        a = a * np.exp(-eta * grad)
        # project onto simplex
        a = _project_simplex(a)
    return best_a, best_score

"""
Propose a non‑negative step‑function that roughly minimizes
``evaluate_sequence``.  The optimiser runs four stages:

1.  Fast evaluation of a few deterministic shapes and short soft‑max
    exponentiated‑gradient runs to pick a promising start.
2.  Long soft‑max exponentiated‑gradient descent (gradient uses a
    smooth approximation of the max, i.e. a softmax weight vector).
    The temperature (soft‑max sharpness) is increased during the run.
3.  Simulated‑annealing style random moves that occasionally accept
    uphill steps, updating the convolution incrementally.
4.  A deterministic greedy “peak‑flattening’’ pass (the same routine
    used by the reference solution) for a final polish.

All heavy work is done with FFT‑based convolutions, projection onto the
probability simplex keeps the sum equal to one, and the whole routine
fits comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                     #fallback – should never happen for non‑empty v
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]      # length 2·n‑1


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max on the convolution values."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return exps / exps.sum()


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  sum_i w_i * (a*a)[i]  w.r.t. a.
    Uses FFT to compute the cross‑correlation of rev(a) with w.
    """
    rev_a = a[::-1]
    RevA = np.fft.rfft(rev_a, n=L)
    W = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(RevA * W, n=L)
    n = a.shape[0]
    start = n - 1
    # factor 2 because d/da_i sum_j w_{i+j} a_j = 2 * Σ_j w_{i+j} a_j
    return 2.0 * corr_full[start:start + n]


# ----------------------------------------------------------------------
# Phase 2 – Soft‑max exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _run_softmax_exp_gradient(a0: np.ndarray,
                              n: int,
                              deadline: float,
                              start_time: float,
                              eta0: float = 0.5,
                              beta_start: float = 5.0,
                              beta_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) optimiser that minimises a smooth
    approximation of the max‑convolution.  The soft‑max temperature
    (beta) grows from beta_start to beta_end.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    L = _next_pow2(3 * n)                # enough for both conv and cross‑corr
    it = 0

    while time.time() < deadline:
        it += 1

        # full self‑convolution
        conv = _conv_self_fft(a, L)

        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] softmax‑grad iter={it:,}  score={cur_score:.6f}")

        # schedule beta (linear increase)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 200_000.0)

        # soft‑max weight vector
        w = _softmax_weights(conv, beta)

        # gradient of the smooth objective
        grad = _gradient_from_weights(a, w, L)

        # clip gradients to avoid extreme exponentials
        grad = np.clip(grad, -20.0, 20.0)

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # multiplicative update + simplex projection
        a = a * np.exp(-eta * grad)
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Simulated‑annealing style random moves
# ----------------------------------------------------------------------
def _run_annealing(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   cur_best_seq: np.ndarray,
                   cur_best_score: float,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Randomly moves a small amount of mass between two positions.
    Moves are accepted if they improve the score or with a
    temperature‑dependent probability (simulated annealing).
    The convolution is updated incrementally.
    """
    a = a0.copy()
    L_conv = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L_conv)          # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    cur_score = best_score

    # annealing schedule
    T0 = 0.02
    T_end = 1e-6
    total_time = deadline - time.time()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # pick a source index with non‑zero mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = np.random.choice(src_candidates)

        # pick a different destination
        dst = np.random.randint(0, n)
        while dst == src:
            dst = np.random.randint(0, n)

        # propose a tiny transfer (up to 5 % of the source mass)
        eps = a[src] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue

        # copy current state for a potential revert
        a_old = a.copy()
        conv_old = conv.copy()

        # incremental convolution update
        conv[src:src + n] -= 2.0 * eps * a_old
        conv[dst:dst + n] += 2.0 * eps * a_old

        # update the vector
        a[src] -= eps
        a[dst] += eps
        a = np.maximum(a, 0.0)

        new_score = 2.0 * n * conv.max()

        # temperature based on elapsed fraction
        elapsed = time.time() - start_time
        fraction = min(1.0, (elapsed) / (deadline - start_time))
        T = T0 * (T_end / T0) ** fraction

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis rule
            if np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={it:,}  score={new_score:.6f}")
        else:
            # reject – restore old state
            a = a_old
            conv = conv_old

        # occasional full recomputation to avoid drift
        if it % 50_000 == 0:
            conv = _conv_self_fft(a, L_conv)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Deterministic greedy peak‑flattening (same as reference)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Move a tiny amount of mass from the index that contributes most to the
    current peak to an index that contributes least, updating the convolution
    incrementally.  Full recomputation every few thousand steps removes drift.
    """
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, Lfft)               # length 2·n‑1
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # partner values for src and every other candidate
        partner_vals = a[max_idx - idx]

        # choose destination with smallest partner value (and not src)
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        # decide a small move size epsilon
        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid drift
        if it % recompute_interval == 0:
            conv = _conv_self_fft(a, Lfft)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,}  score={cur_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``
    within the given time budget.  Returns the best sequence as a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 4096 is a reliable trade‑off between quality and speed
    n = 4096

    # ------------------------------------------------------------------
    # Initialise – several deterministic shapes + a random Dirichlet start
    # ------------------------------------------------------------------
    initial_candidates = []

    # use a previously discovered construction if available
    if "height_sequence_1" in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                interp = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, starter.size),
                    starter,
                )
                interp = np.maximum(interp, 0.0)
                total = interp.sum()
                if total > 0.0:
                    interp /= total
                else:
                    interp = np.full(n, 1.0 / n)
                initial_candidates.append(interp)
        except Exception:
            pass

    # uniform
    initial_candidates.append(np.full(n, 1.0 / n))

    # linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    initial_candidates.append(lin / lin.sum())

    # sqrt‑law
    sqrt_shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    initial_candidates.append(sqrt_shape / sqrt_shape.sum())

    # exponential decay
    q = 0.9995
    exp_shape = (1 - q) * (q ** np.arange(n, dtype=float))
    initial_candidates.append(exp_shape / exp_shape.sum())

    # a random Dirichlet vector
    rand = np.random.rand(n)
    rand /= rand.sum()
    initial_candidates.append(rand)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Tiny soft‑max gradient runs to pick a promising start
    # ------------------------------------------------------------------
    for cand in initial_candidates:
        if time.time() >= deadline:
            break
        temp_deadline = time.time() + min(4.0, (deadline - time.time()) * 0.04)
        seq, scr = _run_softmax_exp_gradient(cand, n, temp_deadline,
                                             start_time,
                                             eta0=0.8,
                                             beta_start=5.0,
                                             beta_end=30.0)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] init improve: score={scr:.6f}")

    # fallback – should never be needed
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Phase 2 – long soft‑max exponentiated‑gradient descent (≈60 % time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        phase2_deadline = time.time() + remaining * 0.60
        seq, scr = _run_softmax_exp_gradient(best_seq, n, phase2_deadline,
                                             start_time,
                                             eta0=0.5,
                                             beta_start=10.0,
                                             beta_end=200.0)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] long softmax‑grad improve: "
                  f"score={scr:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing random moves (≈30 % time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_annealing(best_seq, n, deadline, start_time,
                                  best_seq, best_score,
                                  max_iters=10_000_000)

    else:
        best_seq = np.full(n, 1.0 / n)

    # return as plain Python list
    return [float(x) for x in best_seq]
