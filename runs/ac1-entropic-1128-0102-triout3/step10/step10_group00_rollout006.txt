The algorithm consists of three stages

   each iteration computes the self‑convolution, selects all indices whose

   \(\sum_k w_k\,(a*a)_k\) is obtained with a single FFT‑based convolution,
   followed by an Adam update and a projection onto the probability

"""
Propose a step‑function (non‑negative heights) with a very low
evaluate_sequence() score.

Algorithm
---------
1. Scan a few simple symmetric families (tent, quadratic, sine,
   beta‑type) for many exponents and keep the best normalised vector.
2. Refine the best vector with a projected Adam optimiser.
   At each iteration the gradient of a *smooth* approximation of the
   max‑convolution is obtained by an FFT‑based convolution with a
   weight vector that concentrates on the near‑max entries.
3. If time permits, perform a cheap simulated‑annealing polish.
4. Return the normalised vector as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·n‑1) using a real FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using a real FFT.
    Returns the full convolution (len(x)+len(y)-1).
    """
    L = _next_pow2(len(x) + len(y) - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    Identical to ``evaluate_sequence`` but without the defensive checks.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σ a_i = 1, a_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto { w ≥ 0, Σ w = 1 }.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpinán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Gradient of a smooth approximation of max(conv) using an active set
# ----------------------------------------------------------------------
def _active_gradient(a: np.ndarray, conv: np.ndarray, threshold: float) -> np.ndarray:
    """
    Computes ∇ ( Σ_k w_k (a*a)_k ) where w is uniform on the
    “active set’’ { k | conv_k ≥ max(conv)-threshold }.
    The gradient is obtained with a single FFT‑based convolution.
    """
    n = a.shape[0]
    # build weight vector (sum to 1)
    active = np.where(conv >= conv.max() - threshold)[0]
    if active.size == 0:
        active = np.array([conv.argmax()], dtype=int)
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size

    # gradient = 2 * (a * w_rev) evaluated at positions 2n-2-i, i=0..n-1
    grad_full = _conv_linear(a, w[::-1])               # length 3n-2
    idx = (2 * n - 2) - np.arange(n)                   # indices for the needed slice
    grad = 2.0 * grad_full[idx]
    return grad

# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, deadline: float, start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap annealing that moves tiny mass packets."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30_000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break
        # exponential cooling (fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)                # source proportional to its mass
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30      # move up to 30 % of the source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={new_score:.6f}")

    return best_a, best_score
