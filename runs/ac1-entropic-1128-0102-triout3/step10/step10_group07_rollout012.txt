    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = float('inf')
    t = 0
    while time.time() < deadline:
        t += 1
        # convolution of a with itself
        conv = _conv_self(a, fft_len_conv)
        max_conv = conv.max()
        cur_score = 2.0 * len(a) * max_conv  # approximate (since sum normalized to 1).
        # evaluate exact objective periodically
        if t % 400 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Smooth exact improve: iter={t:,}, score={sc:.6f}")

        # compute weighted gradient
        # compute weight vector w using softmax on conv with temperature
        # linear schedule for beta
        total_smooth_time = deadline - start_time
        elapsed = time.time() - start_time
        frac = min(1.0, (elapsed) / total_smooth_time)
        beta = beta_initial + frac * (beta_final - beta_initial)

        # compute softmax weights
        # subtract max to avoid overflow
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # compute gradient via convolution of reversed a and w
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w, fft_len_grad)
        # slice indices n-1 to n-1+n
        n = len(a)
        grad = 2.0 * conv_rev[n-1 : n-1 + n]

        # step size
        eta = eta0 / math.sqrt(t)

        # mirror descent update (multiplicative)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
    # final evaluate
    final_sc = evaluate_sequence(a.tolist())
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
        print(f"[{time.time() - start_time:6.2f}s] Smooth final improve: score={final_sc:.6f}")
    return best_a, best_score

"""
Search for a step function minimizing evaluate_sequence.

The algorithm builds a diverse pool of random starts, keeps the best
and then refines it with a two‑phase gradient descent:
  1. Smooth optimisation of a log‑sum‑exp surrogate of the max
     convolution using a temperature schedule (soft‑max weights) and
     mirror‑descent updates.
  2. A classic exponentiated‑gradient polish that uses the exact
     active shift (the true max) as sub‑gradient.
A final deterministic greedy peak‑reduction step moves a fraction of
mass from the worst‑contributing coefficient towards the smallest one.
All updates keep the vector on the probability simplex, guaranteeing
valid inputs for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution x*y via FFT."""
    Xf = np.fft.rfft(x, n=fft_len)
    Yf = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(Xf * Yf, n=fft_len)
    return conv[: x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
# Random start generators
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``conc`` and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Discrete Gaussian centred on the interval, normalised."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose ``m`` indices, give each equal mass, zero elsewhere."""
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


# ----------------------------------------------------------------------
# Smooth optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    fft_len_conv: int,
    fft_len_grad: int,
    beta_initial: float = 0.3,
    beta_final: float = 6.0,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector ever observed and the exact score.

    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    total_phase = deadline - start_time

    while time.time() < deadline:
        t += 1

        # --- exact objective (quick approximation) ---
        # because a is normalised, the exact objective equals 2*n*max(conv)
        conv = _conv_self(a, fft_len_conv)

        cur_score = 2.0 * n * cur_max

            if exact < best_score:
                best_score = exact

        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution x*y via FFT."""
    Xf = np.fft.rfft(x, n=fft_len)
    Yf = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(Xf * Yf, n=fft_len)
    return conv[: x.shape[0] + y.shape[0] - 1]

def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet vector with concentration ``conc``."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian, normalised."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:

    # Evaluate raw starts, keep the best
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(starts):
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] start #{idx+1:2d} raw score = {sc:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 1 – smooth log‑sum‑exp gradient descent (≈55 % of budget)
    # --------------------------------------------------------------
    smooth_deadline = start_time + 0.55 * (deadline - start_time)
    smooth_seq, smooth_score = _smooth_optimize(
        best_seq,
        smooth_deadline,
        start_time,
        fft_len_conv,
        fft_len_grad,
        beta_initial=0.3,
        beta_final=6.0,
        eta0=0.6,
    )
    if smooth_score < best_score:
        best_score = smooth_score
        best_seq = smooth_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after smooth score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient polish (≈30 % of budget)
    # --------------------------------------------------------------
    polish_deadline = start_time + 0.85 * (deadline - start_time)
    polished_seq, polished_score = _expgrad_optimize(
        best_seq, polish_deadline, start_time, fft_len_conv, eta0=0.8
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(
            f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after polish score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Phase 3 – cheap random‑restart bursts (≈10 % of remaining time)
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        if rng.random() < 0.5:
            m = int(rng.integers(300, 2500))
            cand_start = _sparse_random_start(n, m, rng)
        else:
            conc = 10 ** rng.uniform(-1.5, 1.0)
            cand_start = _dirichlet_start(n, conc=conc, rng=rng)

        sub_deadline = time.time() + 4.0
        cand_seq, cand_score = _smooth_optimize(
            cand_start,
            sub_deadline,
            start_time,
            fft_len_conv,
            fft_len_grad,
            beta_initial=0.3,
            beta_final=2.0,
            eta0=0.5,
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from restart score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Phase 4 – final deterministic greedy peak‑reduction
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Safety – clip, re‑normalise and final reporting
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(
        f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}"
    )

    return [float(x) for x in best_seq.tolist()]
