"""
Propose a low‑score step function.

Algorithm
---------
1. **Fine‑grained arcsine family sweep** –  Build symmetric families
   a_i ∝ (x_i·(1−x_i)+ε)^{−α} with a dense grid of exponents
   α∈[0.30,0.49] (step ≈0.004) and a few tiny regularisers ε.
   The best candidate is kept.

2. **Projected sub‑gradient descent** –  Starting from the best
   candidate we run a long, momentum‑based sub‑gradient descent on the
   exact non‑smooth objective
       2·n·max(conv(a,a))/‖a‖₁² .
   Iterations stop when the wall‑time budget is almost exhausted.

3. **Very light annealing polish** –  A few‑thousand random
   mass‑transfer moves are attempted; a move is kept if it improves the
   objective (or is accepted with a tiny Metropolis probability).

All steps respect the 1000 s wall‑time limit and never produce negative
or overly large entries; the final vector is normalised to unit sum.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based self‑convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the verifier:
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed to be a 1‑D NumPy array.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Projection onto the simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (nonsmooth objective)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a simple projected sub‑gradient descent on the exact
    (non‑smooth) objective.  Returns the best iterate seen before
    `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9          # momentum coefficient
    step0 = 0.25        # initial step size factor
    it = 0

    while time.time() < deadline:
        it += 1

        # compute convolution and current score
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] "
                f"sub‑grad iter={it:,} score={cur_score:.6f}"
            )

        # index where the maximum is attained
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a (see reference implementation)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + decreasing step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum

        # keep a on the simplex
        a = _project_simplex(a)

    return best_a, best_score
