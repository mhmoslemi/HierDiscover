"""
Propose a step‑function (non‑negative heights) with a very low
objective 2·n·max(conv)/(∑a)².

Main ideas
----------
* Use a symmetric “tent” as warm‑start, sweeping a fine grid of exponents.
* Run a *smooth* descent where the non‑smooth max is replaced by a
  temperature‑controlled soft‑max.  This gives a dense gradient that
  can move the solution out of poor local minima.
* Finish with a few thousand iterations of the classic projected
  sub‑gradient method (the same scheme that gave the previous record)
  to polish the result.
* An optional light annealing step may improve the final score if time
  permits.
* All iterates are projected onto the probability simplex and
  symmetrised (a_i = a_{n‑1‑i}) – the optimal solution is known to be
  symmetric.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(∑seq)² (scale‑invariant)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i} while keeping the sum unchanged."""
    return 0.5 * (a + a[::-1])


# ----------------------------------------------------------------------
# Smooth (soft‑max) descent
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a soft‑max approximation of max(conv).
    Returns the best iterate seen before ``deadline`` together with its score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    # pre‑compute FFT length for the correlation needed in the gradient
    L_corr = _next_pow2(3 * n - 2)

    # schedule parameters
    max_iter = 30000               # safety bound; actual loops stop at deadline
    step0 = 0.25                   # base step size
    tau_start = 0.5                # relative to current max conv
    tau_end = 0.02                 # relative to current max conv

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # convolution a*a
        conv = _conv_self(a)
        cur_score = _objective(a)

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth iter={it:,} "
                  f"score={cur_score:.6f}")

        # temperature schedule (geometric decay)
        tau_rel = tau_start * (tau_end / tau_start) ** (it / max_iter)
        tau = max(conv.max() * tau_rel, 1e-12)

        # soft‑max weights (stable formulation)
        w = np.exp((conv - conv.max()) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * (w ⋆ a)  (cross‑correlation)
        Wf = np.fft.rfft(w, n=L_corr)
        Af = np.fft.rfft(a, n=L_corr)
        corr = np.fft.irfft(Wf * np.conj(Af), n=L_corr)
        grad = 2.0 * corr[(n - 1):(n - 1 + n)]

        # step‑size (1/√t)
        step = step0 / math.sqrt(it)
        a -= step * grad

        # project onto simplex and enforce symmetry
        a = _project_simplex(a)
        a = _symmetrize(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (as used previously)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient descent on the exact max‑objective."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad iter={it:,} "
                  f"score={cur_score:.6f}")

        # index where the maximum is attained
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # momentum + step‑size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a -= step * momentum

        # projection and symmetrisation
        a = _project_simplex(a)
        a = _symmetrize(a)

    return best_a, best_score

# Light simulated‑annealing polish (optional)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray,
            deadline: float,
            start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap annealing that moves tiny mass packets."""
    a = seq.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total_steps = 30000
    for it in range(1, total_steps + 1):
        if time.time() >= deadline:
            break

        # exponential cooling (fast)
        T = 0.01 * (1e-6 / 0.01) ** (it / total_steps)

        src = rng.choice(n, p=a)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        eps = a[src] * rng.random() * 0.30
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective(a_prop)
        delta = new_score - cur_score

        if delta < 0.0 or math.exp(-delta / T) > rng.random():
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} "
                      f"best={new_score:.6f}")

    return best_a, best_score
