    # Build A_ub (2n-1 constraints)
    # For each k from 0 to 2n-2:
    #   Define row data: for i in [max(0, k-(n-1)), min(n-1, k)]:
    #       j = i
    #       A[k, i] = f[i] ??? Actually convf_g[k] = sum_{i} f[i] * g[k-i], where g index j = k - i
    # So coefficient for g[j] = f[i] if 0 <= i <= n-1, 0 <= j <= n-1
    # Equivalent: A[k, j] = f[k - j] if 0 <= k - j < n else 0.
    # For each row k and column j, if 0 <= k - j < n then A[k,j] = f[k - j].
    #
    # We'll generate data arrays - using loops for each k maybe heavy but okay for n=2048.
    import numpy as np
    import scipy.sparse as sp
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ g <= M * np.ones(2*n-1)
    # bounds: (0, None)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = sp.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    g0 = res.x
    return g0

            perturb = rng.normal(scale=0.1, size=K)  # relative perturb
            cand = w0 * (1 + perturb)
            cand = np.maximum(cand, 0.0)
            init_pop.append(cand)
        init_pop = np.array(init_pop)
        bounds = [(0.0, 1.0)] * K

        def block_obj(w):
            # if any negative, treat as zero
            w = np.maximum(w, 0.0)
            a_seq = np.empty(n, dtype=float)
            for i in range(K):
                start = i*seg_len
                end = (i+1)*seg_len if i < K-1 else n
                a_seq[start:end] = w[i]
            if a_seq.sum() == 0:
                return np.inf
            a_seq = a_seq / a_seq.sum()
            conv = _conv_self(a_seq, L)
            return 2.0 * n * conv.max()

        from scipy.optimize import differential_evolution
        result_de = differential_evolution(block_obj,
                                           bounds,

                                           maxiter=max_iter,
                                           polish=False,
                                           updating='deferred',
                                           disp=False,

            a_de[start:end] = result_de.x[i]
        if a_de.sum() == 0:
            pass
        else:
            a_de = a_de / a_de.sum()
            score_de = compute_obj(a_de)
            if score_de < best_score:
                best_score = score_de
                best_seq = a_de.copy()
                print(f"[{time.time() - start_time:.2f}s] DE improve: score={score_de:.6f}")

    # Phase 3: Greedy reduction (more iterations)
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(remaining * 5000)  # estimate 5000 iters per second?
        # but ensure not too large, clamp
        max_iters = min(max_iters, 10_000_000)
        a_greedy, score_greedy = _run_greedy_reduction(best_seq, n, deadline, start_time, best_score, best_seq, max_iters=max_iters)
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] Greedy improve final: score={score_greedy:.6f}")

    # Phase 4: Simulated annealing fine tuning
    if time.time() < deadline:
        a = best_seq.copy()
        L = _next_pow2(2*n - 1)
        conv = _conv_self(a, L)
        cur_score = 2.0 * n * conv.max() / (a.sum()**2)
        best_local = a.copy()
        best_local_score = cur_score
        T0 = 0.05
        T_end = 1e-9
        total_steps = 300_000
        rng = np.random.default_rng(seed+42)
        start = time.time()
        it = 0
        while time.time() < deadline and it < total_steps:
            it += 1
            # exponential cooling
            T = T0 * (T_end / T0) ** (it / total_steps)
            # choose src weighted by contribution to current max conv index
            k = int(conv.argmax())
            lo = max(0, k - (n-1))

            contrib = a[idx] * a[k - idx]
            if contrib.sum() > 0:
                src = rng.choice(idx, p=contrib / contrib.sum())
            else:
                src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:

            eps = a[src] * rng.random() * 0.1
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            conv_new = _conv_self(a_prop, L)

                conv = conv_new

                    best_local_score = cur_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal fine improve: iter={it:,}, score={cur_score:.6f}")
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={best_local_score:.6f}")

    # final normalization to sum=1
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:

    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]

"""
Propose a step‑function with a lower evaluation score.

The search consists of several complementary phases

1.  Fine‑tune a pure exponential seed by bounded scalar optimisation.
2.  Run Adam on the smooth log‑sum‑exp approximation of the max‑convolution.
3.  Refine the shape with a cheap black‑box optimiser (Differential Evolution)
    on a block‑wise constant parametrisation (≈30 blocks).
4.  Deterministic greedy reduction of the peak (the same routine that was
    used in the reference implementation, but run until the time budget
    expires).
5.  A short targeted simulated‑annealing sweep to escape possible local
    minima.

All phases keep track of the best sequence seen so far; the final sequence
is normalised to unit sum and returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy import optimize
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full (linear) self‑convolution using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Normalised geometric series a_i ∝ r**i (0 < r < 1).
    Returned vector sums to 1.
    """
    if r <= 0.0 or r >= 1.0:
        r = 0.999
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * r
    norm = 1.0 - r ** n          # (1‑r**n)  = sum_i r**i  after factor (1‑r)
    a = a * (1.0 - r) / norm
    a[a < 0.0] = 0.0
    return a


def _objective(a: np.ndarray, n: int, L: int) -> float:
    """Exact value of evaluate_sequence for a given vector (sum may be !=1)."""
    conv = _conv_self(a, L)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _run_adam_smoothing(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        alphas,
                        lr0: float = 0.4,
                        iters_per_alpha: int = 50000) -> tuple[np.ndarray, float]:
    """Adam optimisation on the log‑sum‑exp (smooth max) surrogate."""
    a = a0.copy()
    best_a = a.copy()
    L = _next_pow2(2 * n - 1)

    # initial score (exact, not smoothed)
    best_score = _objective(a, n, L)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            it += 1

            conv = _conv_self(a, L)
            cur_score = _objective(a, n, L)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] adam improve: "
                      f"iter={it:,}, score={cur_score:.6f}")

            # ----- smooth max (log‑sum‑exp) -----
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)          # stability
            w /= w.sum()
            w_rev = w[::-1]

            # ----- gradient of the smoothed max -----
            Af = np.fft.rfft(a, n=L)
            Wrev = np.fft.rfft(w_rev, n=L)
            corr = np.fft.irfft(Af * Wrev, n=L)[: n + len(w_rev) - 1]
            grad = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # ----- Adam update -----
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            lr = lr0 / math.sqrt(it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto the simplex
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Greedy reduction of the peak (deterministic)
# ----------------------------------------------------------------------
def _run_greedy_reduction(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          cur_best_score: float,
                          cur_best_seq: np.ndarray,
                          max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Same routine as in the reference implementation, but the iteration
    budget is passed from the caller."""
    a = a0.copy()
    L = 2 * n - 1
    Lfft = _next_pow2(L)
    conv = _conv_self(a, Lfft)               # full self‑convolution
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 1000

    while time.time() < deadline and it < max_iters:
        it += 1
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]

        # build vector a_k[i] = a[max_idx - i] (zero outside)
        a_k = np.zeros(n)
        a_k[idx] = a[max_idx - idx]

        # destination: a_k minimal (but different from src)
        dst_candidates = np.where(a_k == a_k.min())[0]
        dst_candidates = dst_candidates[dst_candidates != src]
        if dst_candidates.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_candidates)

        # partner values for src and dst in the current max
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental update of the convolution -----
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps

        a[a < 0] = 0.0

        # periodic exact recomputation (removes drift)
        if it % recompute_interval == 0:
            conv = _conv_self(a, Lfft)

        cur_score = _objective(a, n, Lfft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy improve: "
                  f"iter={it:,}, score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    The routine combines a fine‑tuned exponential seed, Adam on a
    smooth surrogate, a short differential‑evolution refinement on
    block‑wise parameters, deterministic greedy reduction and a final
    simulated‑annealing sweep.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    L = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Phase 0 – fine‑tune a pure exponential seed
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] start exponential fine‑tune")
    def exp_obj(r):
        a = _exponential_sequence(n, r)
        return _objective(a, n, L)

    res = optimize.minimize_scalar(
        exp_obj,
        bounds=(0.90, 0.99999),
        method='bounded',
        options={'xatol': 1e-7, 'maxiter': 200}
    )
    best_r = res.x
    a = _exponential_sequence(n, best_r)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = _objective(a, n, L)
    print(f"[{time.time() - start_time:.2f}s] exponential r={best_r:.8f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smoothed max (≈2 minutes)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase_deadline = min(deadline, time.time() + 120.0)
        a_adam, score_adam = _run_adam_smoothing(
            best_seq,
            n,
            phase_deadline,
            start_time,
            alphas=[0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0],
            lr0=0.4,
            iters_per_alpha=50000
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam improve: score={score_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Block‑wise Differential Evolution (≈30 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # number of blocks – a compromise between flexibility and speed
        K = 32
        seg_len = n // K
        # baseline (exponential) block values – used as centre of the population
        baseline = _exponential_sequence(n, best_r)
        w0 = np.array([baseline[i * seg_len:(i + 1) * seg_len].sum()
                       for i in range(K)])

        rng = np.random.default_rng(seed + 777)
        popsize = 15
        init_pop = []
        for _ in range(popsize):
            perturb = rng.normal(scale=0.15, size=K)
            cand = w0 * (1.0 + perturb)
            cand = np.maximum(cand, 0.0)
            init_pop.append(cand)
        init_pop = np.array(init_pop)

        bounds = [(0.0, 1.0)] * K

        def block_obj(w):
            w = np.maximum(w, 0.0)
            a_vec = np.empty(n, dtype=float)
            for i in range(K):
                start = i * seg_len
                end = (i + 1) * seg_len if i < K - 1 else n
                a_vec[start:end] = w[i]
            if a_vec.sum() == 0.0:
                return np.inf
            a_vec /= a_vec.sum()
            return _objective(a_vec, n, L)

        # a modest DE run – enough to move away from the pure exponential
        de_res = differential_evolution(
            block_obj,
            bounds,
            init=init_pop,
            maxiter=30,
            popsize=popsize,
            strategy='best1bin',
            polish=False,
            updating='deferred',
            disp=False,
            seed=seed + 555,
        )
        a_de = np.empty(n, dtype=float)
        for i in range(K):
            start = i * seg_len
            end = (i + 1) * seg_len if i < K - 1 else n
            a_de[start:end] = de_res.x[i]
        if a_de.sum() > 0.0:
            a_de /= a_de.sum()
            score_de = _objective(a_de, n, L)
            if score_de < best_score:
                best_score = score_de
                best_seq = a_de.copy()
                print(f"[{time.time() - start_time:.2f}s] DE improve: score={score_de:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Aggressive greedy reduction (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # a rough estimate: 5 000 iterations per second are cheap on this hardware
        max_iters = int(remaining * 5000)
        max_iters = min(max_iters, 10_000_000)
        a_greedy, score_greedy = _run_greedy_reduction(
            best_seq, n, deadline, start_time, best_score, best_seq, max_iters=max_iters
        )
        if score_greedy < best_score:
            best_score = score_greedy
            best_seq = a_greedy.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy final improve: score={score_greedy:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Short simulated‑annealing fine‑tune
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        conv = _conv_self(a, L)
        cur_score = _objective(a, n, L)
        best_local = a.copy()
        best_local_score = cur_score

        T0 = 5e-2
        T_end = 1e-9
        total_steps = 300_000
        it = 0
        rng = np.random.default_rng(seed + 1234)

        while time.time() < deadline and it < total_steps:
            it += 1
            T = T0 * (T_end / T0) ** (it / total_steps)

            k = int(conv.argmax())
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.sum() > 0.0:
                src = rng.choice(idx, p=contrib / contrib.sum())
            else:
                src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-14:
                continue
            eps = a[src] * rng.random() * 0.1
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            conv_new = _conv_self(a_prop, L)
            new_score = _objective(a_prop, n, L)
            delta = new_score - cur_score

            if delta < 0 or math.exp(-delta / T) > rng.random():
                a = a_prop
                conv = conv_new
                cur_score = new_score
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] anneal improve: iter={it:,}, score={cur_score:.6f}")

        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local.copy()
            print(f"[{time.time() - start_time:.2f}s] final anneal improve: score={best_local_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise to a probability vector and report the final objective
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_obj = _objective(best_seq, n, L)
    print(f"[{time.time() - start_time:.2f}s] final objective = {final_obj:.6f}")

    return [float(x) for x in best_seq]
