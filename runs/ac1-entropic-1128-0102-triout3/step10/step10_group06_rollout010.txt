from scipy.optimize import linprog
import scipy.sparse as sp

def lp_step(f, M, t):
    n = len(f)
    # Build A: shape (2n-1, n)
    offsets = np.arange(n)   # row-col offsets
    data = np.vstack([np.full(n, f[d]) for d in offsets])   # shape (n, n)
    # Build sparse matrix using sp.diags
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csc')
    # Define constraints: A @ b <= M
    # Convert to inequality constraints: A_ub = A, b_ub = np.full(2n-1, M)
    # Objective: minimize -np.ones(n) @ b
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    # Scale to sum 1:
    b = b / b.sum()
    # Mix
    new_f = (1 - t) * f + t * b
    # Normalize sum to 1:
    new_f = new_f / new_f.sum()
    return new_f
```

"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from a handful of handcrafted / random
   vectors (uniform, exponential, Gaussian, Dirichlet draws) and keep the
   best one (exact score is measured with ``evaluate_sequence``).

2. **Rough smoothing** – a few hundred exponentiated‑gradient (mirror
   descent) iterations driven by the true max‑convolution.  This moves the
   distribution away from the uniform triangular peak obtained by the
   analytical solution.

3. **LP polishing** – given the current vector *f* we solve a linear
   program

        maximise   Σ b_j
        subject to (f * b)[k] ≤ M   for all “active’’ shifts k,
                   b_j ≥ 0

   where *M* = max (f*f) and the active set consists of the shifts whose
   convolution value is within a tiny relative tolerance of *M*.  The LP
   finds a new distribution *g* with the largest possible total mass
   while never exceeding the current worst convolution value *M*.
   Afterwards we mix the old and the new vector

        f ← (1‑t)·f + t·g   (t≃0.3)

   and repeat.  Because the active set is very small the LP is tiny
   (a few hundred constraints) and can be solved quickly with SciPy’s
   HiGHS interior‑point solver.

4. **Final greedy sweep** – a deterministic local improvement that moves a
   small fraction of the coefficient that contributes most to the current
   peak towards the smallest coefficient (accept only strict improvements).

The whole routine respects the supplied time budget (default 1000 s) and
returns the best vector found (normalised to sum = 1, all entries ≥ 0).
"""

import time
import math
import numpy as np

# optional SciPy imports – used for the LP polishing step
try:                      # pragma: no‑cover
    import scipy.sparse as _sp
    from scipy.optimize import linprog as _linprog
    _HAS_SCIPY = True
except Exception:        # pragma: no‑cover
    _sp = None
    _linprog = None
    _HAS_SCIPY = False


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * n - 1]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------


def _expgrad_step(a: np.ndarray, eta: float, top_k: int,
                  delta_frac: float) -> np.ndarray:
    """
    One exponentiated‑gradient step w.r.t. the true max‑convolution.
    Returns the updated vector (already renormalised to sum = 1).
    """
    n = a.shape[0]
    conv = _conv_self_fft(a)
    cur_max = conv.max()

    # active shifts (top‑k or within δ of the maximum)
    thresh = cur_max - max(delta_frac * cur_max, 1e-12)
    active = np.where(conv >= thresh)[0]
    if active.size > top_k:
        # keep only the largest *top_k* shifts
        part = np.argpartition(-conv[active], top_k - 1)[:top_k]
        active = active[part]

    # sub‑gradient of the max‑function
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]

    # exponentiated‑gradient update
    a = a * np.exp(-eta * g)
    a = np.maximum(a, 1e-15)                # keep away from exact zero
    a /= a.sum()
    return a


def _run_expgrad(a0: np.ndarray,
                 deadline: float,
                 start_time: float,
                 max_iters: int = 50000,
                 eta0: float = 0.6,
                 delta_frac: float = 0.001,
                 top_k: int = 7) -> tuple[np.ndarray, float]:
    """Perform many exp‑grad steps until time runs out."""
    a = a0.copy()
    best_a = a.copy()
    best_score = float("inf")
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        eta = eta0 / math.sqrt(t)
        a = _expgrad_step(a, eta, top_k, delta_frac)
        if t % 250 == 0 or t == max_iters:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"exp‑grad improve iter={t:,} score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming polishing step (requires SciPy)
# ----------------------------------------------------------------------


def _lp_polish(f: np.ndarray,
               t_mix: float,
               epsilon: float = 1e-3,
               max_constraints: int = 120) -> tuple[np.ndarray, float]:
    """
    One LP‑based polishing iteration.

    Parameters
    ----------
    f : np.ndarray
        Current probability vector (sum = 1).
    t_mix : float
        Mixing coefficient when forming the new iterate
        f_new = (1‑t_mix)·f + t_mix·g.
    epsilon : float
        Relative tolerance for selecting the active constraint set.
    max_constraints : int
        Upper bound on the number of linear constraints that are fed to the
        LP solver (keeps the problem tiny).

    Returns
    -------
    f_new, new_score
        Updated vector and its exact score (via ``evaluate_sequence``).
        If the LP fails, ``f_new`` is the untouched ``f`` and the score is
        ``float('inf')``.
    """
    n = f.shape[0]
    conv = _conv_self_fft(f)
    M = conv.max()

    # active constraints – those shifts whose convolution value is close to M
    active = np.where(conv >= M - epsilon * M)[0]
    if active.size > max_constraints:
        # keep the *max_constraints* largest values only
        order = np.argpartition(-conv[active], max_constraints - 1)[:max_constraints]
        active = active[order]

    m = active.size
    # build sparse matrix A (m × n) for the constraints (f * b)[k] ≤ M
    rows, cols, vals = [], [], []
    for r, k in enumerate(active):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1, dtype=int)          # indices of b
        i = k - j                                      # corresponding indices of f
        rows.extend([r] * len(j))
        cols.extend(j.tolist())
        vals.extend(f[i].tolist())
    A = _sp.csr_matrix((vals, (rows, cols)), shape=(m, n))

    # linear programme: maximise Σ b_j  ⇔ minimise –Σ b_j
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(m, M)

    # solve
    try:
        res = _linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'presolve': True})
    except Exception:          # pragma: no‑cover
        return f, float('inf')

    if not res.success:
        return f, float('inf')

    g = res.x
    if g.sum() == 0.0:
        return f, float('inf')
    g = g / g.sum()                     # normalise to sum = 1

    # mix the old and the new vector
    f_new = (1.0 - t_mix) * f + t_mix * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s == 0.0:
        f_new = np.full_like(f_new, 1.0 / n)
    else:
        f_new /= s

    new_score = float(evaluate_sequence(f_new.tolist()))
    return f_new, new_score

def _greedy_cleanup(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Transfer a modest fraction of the coefficient that most contributes
    to the current convolution peak towards the smallest coefficient.
    Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = float(evaluate_sequence(a.tolist()))
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        # destination – the index with the smallest weight (different from src)
        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]          # move a quarter of the source mass
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = float(evaluate_sequence(a_prop.tolist()))
        if new_score < best_score:
            a = a_prop
            best_score = new_score
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: score={new_score:.6f}")
    return a, best_score

                                                    eta0=0.6, delta_frac=0.001, top_k=7)
                # tighten mixing a bit for the next round
                t_mix = max(0.10, t_mix * 0.92)
            else:
                # no improvement – shrink the mixing factor to make smaller steps
                t_mix = max(0.05, t_mix * 0.85)
                if t_mix <= 0.06:
                    break    # further polishing unlikely to help
    else:
        # fallback – run a longer exp‑grad phase if SciPy is missing
        exp_deadline = start_time + budget_s * 0.80
        best_seq, best_score = _run_expgrad(best_seq, exp_deadline, start_time,
                                            max_iters=25000, eta0=0.5,
                                            delta_frac=0.001, top_k=11)

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic greedy cleanup (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_cleanup(best_seq, deadline,
                                                 start_time, max_iters=6000)
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq
            print(f"[{time.time() - start_time:6.2f}s] *** GLOBAL BEST after greedy: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation / safety checks
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0.0:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq /= best_seq.sum()

    final_reported = float(evaluate_sequence(best_seq.tolist()))
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    # Return a plain Python list – the evaluator expects this format.
    return [float(v) for v in best_seq.tolist()]
