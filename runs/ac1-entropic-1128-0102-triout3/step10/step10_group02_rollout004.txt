"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

for n = 4096.

Algorithm
---------
1. Load the best known sequence (``height_sequence_1``) or start from uniform.
2. **Smooth‑max mirror descent** – exponentiated‑gradient updates on a
   smooth approximation of the max‑convolution.  The smoothing parameter
   ``alpha`` is increased geometrically, a small momentum term is added and
   the step‑size decays as 1/√t.  This quickly shapes the distribution.
3. **Projected sub‑gradient descent** – a classic sub‑gradient method on the
   exact objective with a 1/√t step‑size and Euclidean projection onto the
   probability simplex.  We stop early once improvements become rare.
4. **Greedy hill‑climb** – repeatedly transfer a tiny amount of mass from a
   source index (chosen proportionally to its current height) to a random
   destination and keep the move only if it improves the objective.  This
   final polish explores the neighbourhood of the best point found so far.
All phases respect a hard wall‑clock deadline (default 1000 s) and log
progress.  The returned list is normalised to have sum = 1, guaranteeing
that ``evaluate_sequence`` sees a valid candidate.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full linear convolution.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray, L: int = None) -> np.ndarray:
    """
    Linear convolution of two real vectors u and v using FFT.
    Returns the full linear convolution (len(u)+len(v)-1 elements).
    """
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Returns w with w≥0 and sum(w)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum of a is assumed to be 1)
# ----------------------------------------------------------------------
def _eval_fast(a: np.ndarray, L_self: int) -> float:
    """Exact objective for a (non‑negative, sum≈1)."""
    conv = _conv_self(a, L_self)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Phase‑1 : smooth‑max mirror descent (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    L_self: int,
                    L_grad: int,
                    deadline: float,
                    start_time: float,
                    max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation of the maximum convolution.
    Uses exponentiated‑gradient updates with a momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _eval_fast(a, L_self)

    # schedule for the smoothing parameter alpha
    alphas = np.geomspace(0.5, 800.0, num=40)

    # momentum buffer on the (log‑domain) gradient
    m = np.zeros_like(a)

    it = 0
    beta = 0.9

    for alpha in alphas:
        # inner loop – stop when time or iteration budget exhausted
        while time.time() < deadline - 0.2 and it < max_iters:
            it += 1

            # 1) self‑convolution (a * a)
            Af = np.fft.rfft(a, n=L_self)
            conv = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

            # 2) soft‑max weights w_k ∝ exp(alpha * conv_k)
            exp_arg = alpha * conv
            max_arg = exp_arg.max()
            w = np.exp(exp_arg - max_arg)          # numerical stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum

            # 3) gradient = 2 * (w ⋆ a_rev)
            a_rev = a[::-1]
            Wf = np.fft.rfft(w, n=L_grad)
            Revf = np.fft.rfft(a_rev, n=L_grad)
            grad_full = np.fft.irfft(Wf * Revf, n=L_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # 4) momentum + exponentiated‑gradient (mirror‑descent on simplex)
            m = beta * m + (1.0 - beta) * grad
            eta = 0.06 / (math.sqrt(it) * max(1.0, math.sqrt(alpha)))
            a = a * np.exp(-eta * m)          # stay strictly positive
            a = np.maximum(a, 1e-30)
            a /= a.sum()

            # occasional true‑objective check
            if it % 800 == 0:
                cur_score = _eval_fast(a, L_self)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.2f} it={it:,} score={cur_score:.6f}")

        if time.time() >= deadline - 0.2:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : projected sub‑gradient descent (classic)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         L_self: int,
                         deadline: float,
                         start_time: float,
                         init_best_score: float,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Uses a 1/√t step‑size and Euclidean projection onto the simplex.
    Stops early if no improvement is seen for a while.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_best_score

    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    stagnation = 0          # counts iterations without improvement

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, L_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")
        else:
            stagnation += 1

        # stop early if stagnating for 20000 successive iterations
        if stagnation >= 20_000:
            break

        # active indices (within a tolerance)
        tol = 1e-12
        active = np.where(np.abs(conv - cur_max) <= tol)[0]

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # momentum + projected step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        a = a - step * momentum
        a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : greedy hill‑climbing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_self: int,
                deadline: float,
                start_time: float,
                cur_best_seq: np.ndarray,
                cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: move a random fraction of mass from a
    source index (chosen with probability proportional to height) to a
    random destination; keep the move only if it improves the exact
    objective.
    """
    a = a0.copy()
    best_seq = cur_best_seq.copy()
    best_score = cur_best_score
    cur_score = cur_best_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.choice(n, p=a)               # source weighted by current mass
        dst = rng.integers(0, n)               # uniform destination
        if src == dst or a[src] < 1e-15:
            continue

        # transfer up to 30 % of the source mass
        eps = a[src] * rng.random() * 0.30
        if eps <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv = _conv_self(a_prop, L_self)
        new_score = 2.0 * n * conv.max() / (a_prop.sum() ** 2)

        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best={best_score:.6f}")

    return best_seq, best_score

# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the external harness
    deadline = start_time + budget_s - 0.5

    n = 4096
    L_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution
    L_grad = _next_pow2(3 * n - 2)          # length needed for the smooth‑max grad

    # ------------------------------------------------------------------
    # Load a starter vector (global best if available, else uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
    except Exception:
        starter = np.full(n, 1.0 / n)

    # adapt length if necessary
    if starter.size != n:
        if starter.size > n:
            starter = starter[:n]
        else:
            starter = np.pad(starter, (0, n - starter.size), mode='constant')
    starter = np.maximum(starter, 0.0)
    if starter.sum() == 0.0:
        starter = np.full(n, 1.0 / n)
    else:
        starter = starter / starter.sum()

    best_seq = starter.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] Starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑1 : smooth‑max mirror descent (core shaping phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:          # keep some time for later phases
        smooth_deadline = deadline - 20.0
        max_iters_smooth = 300_000            # generous cap
        best_seq, best_score = _smooth_descent(best_seq, n,
                                                L_self, L_grad,
                                                smooth_deadline,
                                                start_time,
                                                max_iters_smooth)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : projected sub‑gradient refinement (fast descent)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        max_iters_sub = 500_000
        a_sub, sc_sub = _subgradient_descent(best_seq, n,
                                             L_self, sub_deadline,
                                             start_time, best_score,
                                             max_iters_sub)
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad polish improved score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 : greedy hill‑climbing polish (lots of tiny moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = _hill_climb(best_seq, n,
                                        L_self, deadline,
                                        start_time,
                                        best_seq, best_score)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved score = {sc_final:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
