"""
Hybrid optimizer for ``evaluate_sequence``

The optimizer works in three stages:

1.  Smoothed exponentiated‑gradient descent.
    A soft‑max approximation of the maximum convolution coefficient
    (parameter ``alpha``) provides a dense gradient, allowing the
    procedure to escape the “single‑active‑index” sub‑gradient stuck
    in the reference implementation.

2.  Deterministic greedy “peak‑flattening”.
    The current maximum entry of the self‑convolution is reduced by
    moving a small amount of mass from the most responsible index
    to a low‑impact destination.  The convolution is updated
    incrementally (O(n) per move) and recomputed exactly every few
    thousand steps.

3.  Simulated‑annealing local search.
    Random mass transfers are accepted with a Metropolis criterion,
    enabling a fine‑grained polishing of the last few digits of the
    objective.

All phases respect the overall 1000 s budget; the best sequence found
at any moment is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of two real vectors using FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    L = _next_pow2(len(a) + len(b) - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)
    return conv[: len(a) + len(b) - 1]


def _self_conv(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    return _fft_convolve(a, a)


def _score_from_vec(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for a vector with sum(a)=1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _self_conv(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Phase 1 – Smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _run_exp_grad_smoothing(a0: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            eta0: float = 0.5,
                            alpha0: float = 1.0,
                            alpha_max: float = 200.0,
                            alpha_growth: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Mirrors a “soft‑max” gradient descent.
    The gradient is the derivative of
        2·n·(log‑sum‑exp(alpha·conv))/alpha.
    The update is multiplicative (exponentiated gradient) and the vector
    is kept on the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_from_vec(a)

    L1 = _next_pow2(2 * n - 1)          # size for a*a
    L2 = _next_pow2(3 * n - 1)          # size for a_rev * softmax
    alpha = alpha0
    it = 0

    while time.time() < deadline:
        it += 1

        # --- self‑convolution (full) ---------------------------------
        Af = np.fft.rfft(a, n=L1)
        conv = np.fft.irfft(Af * Af, n=L1)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad‑smooth iter={it:,} "
                  f"score={cur_score:.6f}  alpha={alpha:.2f}")

        # --- soft‑max weights ----------------------------------------
        # numerical stability: shift by the max
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum                         # shape (2n‑1,)

        # --- gradient via convolution with the weight vector ----------
        a_rev = a[::-1]                           # reversed a
        Af_rev = np.fft.rfft(a_rev, n=L2)
        Bf = np.fft.rfft(p, n=L2)
        grad_full = np.fft.irfft(Af_rev * Bf, n=L2)[: 3 * n - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]   # length n

        # --- step size (classic 1/√t schedule) -----------------------
        eta = eta0 / math.sqrt(it)

        # --- multiplicative update + renormalisation -----------------
        # clipping the exponent to avoid overflow
        expo = -eta * grad
        expo = np.clip(expo, -700, 700)           # np.exp overflow guard
        a = a * np.exp(expo)
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # --- increase the soft‑max sharpness -------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_growth)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Deterministic greedy peak flattening
# ----------------------------------------------------------------------
def _run_greedy_flatten(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        cur_best_score: float,
                        cur_best_seq: np.ndarray,
                        max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a tiny amount of mass from the index that contributes
    most to the current maximum convolution entry to an index that contributes
    least, updating the convolution incrementally (O(n) per move).
    """
    a = a0.copy()
    Lfft = _next_pow2(2 * n - 1)
    conv = _self_conv(a)[:2 * n - 1]   # full self‑convolution
    best_a = a.copy()
    best_score = cur_best_score
    it = 0
    recompute_interval = 500

    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- locate the peak ----------------------------------------
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # second largest entry (ignore the current maximum)
        mask = np.ones_like(conv, dtype=bool)
        mask[max_idx] = False
        second_max = conv[mask].max()

        # indices that actually participate in this max
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        contrib = a[idx] * a[max_idx - idx]
        src = idx[np.argmax(contrib)]                     # worst contributor

        # pick destination with smallest partner value
        partner_vals = a[max_idx - idx]
        dst_cand = idx[partner_vals == partner_vals.min()]
        dst_cand = dst_cand[dst_cand != src]
        if dst_cand.size == 0:
            dst = np.random.randint(0, n)
            while dst == src:
                dst = np.random.randint(0, n)
        else:
            dst = np.random.choice(dst_cand)

        # how much mass can be moved profitably?
        partner_src = a[max_idx - src] if 0 <= max_idx - src < n else 0.0
        partner_dst = a[max_idx - dst] if 0 <= max_idx - dst < n else 0.0
        delta_a = partner_src - partner_dst

        if delta_a <= 0.0:
            eps = min(a[src], 0.02 * a[src])
        else:
            eps_target = (max_val - second_max) / (2.0 * delta_a)
            eps = min(a[src], max(0.0, 0.9 * eps_target))
            if eps < 1e-12:
                eps = min(a[src], 0.02 * a[src])

        if eps <= 0.0:
            continue

        # ----- incremental convolution update -------------------------
        a_old = a.copy()
        conv[dst:dst + n] += 2.0 * eps * a_old
        conv[src:src + n] -= 2.0 * eps * a_old
        a[src] -= eps
        a[dst] += eps
        a[a < 0] = 0.0

        # occasional exact recomputation to avoid floating‑point drift
        if it % recompute_interval == 0:
            conv = _self_conv(a)[:2 * n - 1]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy iter={it:,} score={cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Simulated‑annealing polishing
# ----------------------------------------------------------------------
def _run_simulated_annealing(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             cur_best_score: float,
                             cur_best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Randomly transfers a small amount of probability mass between two
    indices and accepts the move with a Metropolis rule.  The temperature
    decays linearly from an initial value to a tiny final value.
    """
    a = a0.copy()
    cur_score = _score_from_vec(a)
    best_a = cur_best_seq.copy()
    best_score = cur_best_score

    L1 = _next_pow2(2 * n - 1)

    # temperature schedule
    start_temp = max(1e-3, cur_score * 0.02)   # ~2 % of current score
    end_temp = cur_score * 1e-6
    total_steps_est = 100_000
    step = 0

    while time.time() < deadline and step < total_steps_est:
        step += 1
        # ----- propose a move -----------------------------------------
        src = np.random.choice(n, p=a)                 # bias towards heavy bins
        dst = np.random.randint(0, n - 1)
        if dst >= src:
            dst += 1
        max_move = min(a[src], 0.05 * a[src])           # up to 5 % of src
        eps = np.random.rand() * max_move
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # ----- evaluate candidate --------------------------------------
        Af = np.fft.rfft(a_new, n=L1)
        conv = np.fft.irfft(Af * Af, n=L1)[: 2 * n - 1]
        new_score = 2.0 * n * conv.max()

        # ----- Metropolis acceptance ----------------------------------
        temp = start_temp + (end_temp - start_temp) * step / total_steps_est
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / temp):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal iter={step:,} "
                      f"score={new_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative step‑function (list of floats) that
    minimizes ``evaluate_sequence`` within the given time budget.
    The algorithm combines smoothed exponentiated‑gradient descent,
    deterministic greedy peak‑flattening and a short simulated‑annealing
    polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin (the caller will kill the process after budget_s)
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise – try a handful of deterministic shapes.
    # --------------------------------------------------------------
    init_candidates = []

    # a) use the supplied global construction if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            interp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, starter.size),
                               starter)
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            init_candidates.append(interp)
    except Exception:
        pass

    # b) uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # c) linearly decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    init_candidates.append(lin / lin.sum())

    # d) sqrt‑law
    sqrt_shape = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    init_candidates.append(sqrt_shape / sqrt_shape.sum())

    # e) exponential with various bases
    for q in (0.9999, 0.9995, 0.999, 0.99):
        exp_shape = (1 - q) * (q ** np.arange(n, dtype=float))
        init_candidates.append(exp_shape / exp_shape.sum())

    # pick the best of the above using the fast score estimator
    best_seq = init_candidates[0].copy()
    best_score = _score_from_vec(best_seq)

    for cand in init_candidates[1:]:
        scr = _score_from_vec(cand)
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] init candidate improved: score={scr:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        phase1_deadline = time.time() + 0.55 * (deadline - time.time())
        seq, scr = _run_exp_grad_smoothing(best_seq, n,
                                           phase1_deadline,
                                           start_time,
                                           eta0=0.6,
                                           alpha0=1.0,
                                           alpha_max=150.0,
                                           alpha_growth=1.0015)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑1 improve: score={scr:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deterministic greedy reduction of the peak
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_greedy_flatten(best_seq, n,
                                       deadline,
                                       start_time,
                                       best_score,
                                       best_seq,
                                       max_iters=8_000_000)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑2 improve: score={scr:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing (uses remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, scr = _run_simulated_annealing(best_seq, n,
                                            deadline,
                                            start_time,
                                            best_score,
                                            best_seq)
        if scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] phase‑3 improve: score={scr:.6f}")

    # --------------------------------------------------------------
    # Final normalisation (the objective is homogeneous)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    # return as a plain Python list
    return [float(x) for x in best_seq]
