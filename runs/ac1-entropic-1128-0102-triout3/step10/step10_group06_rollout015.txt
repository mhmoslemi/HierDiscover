def _lp_optimize_step(f, M):
    n = f.shape[0]
    K = 2 * n - 1
    # Build A matrix as dense np.ndarray
    A = np.zeros((K, n), dtype=float)
    for k in range(K):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i ranges i_min..i_max
        # j = k - i
        # Set A[k, j] = f[i]
        # j ranging from k-i_max to k-i_min
        # We'll fill row by row:
        for i in range(i_min, i_max + 1):
            j = k - i
            A[k, j] = f[i]
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b = np.full(K, M, dtype=float)
    # bounds (0, None) for each variable
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception as e:
        print("LP exception:", e)
        return None
    if not res.success:
        # if infeasible maybe due to rounding, return None
        return None
    g = res.x
    # ensure non-negative
    g = np.maximum(g, 0.0)
    return g

"""
Propose‑candidate for the “step‑function attack”

Strategy
--------
1. **Small‑scale linear‑programming polishing**  
   Work on a reduced problem (n≈300).  For a fixed vector f we
   compute M = max(conv(f,f)).  Then we solve the linear program  

        maximise   Σ g_j
        subject to conv(f,g)[k] ≤ M   (k = 0 … 2n‑2)
                  g_j ≥ 0

   The LP gives an extreme point g that pushes mass to indices that
   are “safe” w.r.t. the current peak M.  We mix f←(1‑η)f+η g
   (η decreasing) and repeat a few hundred times.  This is the exact
   analogue of the LP step described in the paper, but carried out
   on a tiny problem where the LP is cheap.

2. **Upsample to the full size (n=4800)** – linear interpolation
   preserves the total mass.

3. **Adam warm‑up on a smooth log‑sum‑exp surrogate** – a short
   few‑thousand‑step Adam optimisation to get away from bad plateaus.

4. **Multiplicative mirror‑descent (exponentiated‑gradient) with a
   richer multi‑gradient** – we now use the true max‑convolution
   objective, collect the top‑K active shifts (K=10) and perform
   exponentiated‑gradient updates with a diminishing step size.

5. **Deterministic greedy peak‑reduction** – the same local move as
   before (transfer a fraction of the coefficient contributing most to
   the current peak towards the smallest coefficient).

6. **Optional random hill‑climbing** – if time permits we perform cheap
   random pairwise transfers accepting strictly improving moves.

All phases respect the global 1000 s budget.  The routine returns the
best sequence found (normalised to sum = 1, non‑negative).  The code
has no closures, uses only `numpy`, `scipy.optimize.linprog`,
`time`, and `math`, and prints progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # LP solver for the small‑scale phase


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation preserving the total sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full_like(up, 1.0 / new_n)
    return up


# ----------------------------------------------------------------------
# Random start generators (small problem)
# ----------------------------------------------------------------------


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _exponential_start(n: int, r: float = 0.99) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


# ----------------------------------------------------------------------
# Small‑scale LP optimisation (exact step)
# ----------------------------------------------------------------------


def _lp_optimize_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  maximise Σ g_j   s.t.  conv(f,g)[k] ≤ M  (all k),  g ≥ 0.
    Returns the optimal g (unnormalised).  If the LP fails, returns None.
    """
    n = f.shape[0]
    K = 2 * n - 1

    # Build the constraint matrix A (K × n) where
    #   conv(f,g)[k] = Σ_i f_i * g_{k-i}
    #   ⇒ row k has coefficients f_i placed at column j = k-i.
    A = np.zeros((K, n), dtype=float)
    for k in range(K):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i runs from i_min … i_max, column j = k‑i
        for i in range(i_min, i_max + 1):
            j = k - i
            A[k, j] = f[i]

    c = -np.ones(n)                     # maximise Σ g  ⇔ minimise –Σ g
    b = np.full(K, M, dtype=float)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:  # pragma: no cover – safeguard
        return None

    if not res.success:
        return None

    g = np.maximum(res.x, 0.0)
    return g


# ----------------------------------------------------------------------
# Smooth surrogate + Adam (warm‑up)
# ----------------------------------------------------------------------


def _smooth_objective(a: np.ndarray, alpha: float, fft_len: int):
    """Soft‑max (log‑sum‑exp) approximation of max(conv)."""
    conv = _conv_self(a, fft_len)
    scaled = alpha * conv
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return lse / alpha, conv


def _adam_optimize(a0: np.ndarray,
                   global_deadline: float,
                   start_time: float,
                   fft_len: int,
                   eps_log: float = 1e-16) -> tuple[np.ndarray, float]:
    """
    A few Adam iterations on the smoothed max‑convolution objective.
    Returns the best vector (probability vector) and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    y = np.log(np.maximum(a, eps_log))

    # smoothing schedule (increasingly sharp)
    alpha_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0]
    iters_per_alpha = 2500

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))

    for alpha in alpha_schedule:
        for _ in range(iters_per_alpha):
            if time.time() >= global_deadline:
                return best_a, best_score

            # soft‑max → probability vector
            maxy = np.max(y)
            exp_y = np.exp(y - maxy)
            a = exp_y / exp_y.sum()

            f_val, conv = _smooth_objective(a, alpha, fft_len)

            # gradient of the smooth surrogate w.r.t. a
            scaled = alpha * conv
            max_scaled = scaled.max()
            w = np.exp(scaled - max_scaled)
            w /= w.sum()
            w_rev = w[::-1]

            Af = np.fft.rfft(a, n=fft_len)
            Wrev = np.fft.rfft(w_rev, n=fft_len)
            corr = np.fft.irfft(Af * Wrev, n=fft_len)[: n + len(w_rev) - 1]
            grad_a = 2.0 * corr[len(w_rev) - 1 - np.arange(n)]

            # back‑prop through the soft‑max
            grad_y = a * (grad_a - np.dot(grad_a, a))

            # Adam update on logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr_t = lr0 * (np.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t))

            y = y - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)

            if t % 250 == 0:
                sc = evaluate_sequence(a.tolist())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] Adam improve: α={alpha:.1f}, score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Multiplicative mirror‑descent (main optimisation)
# ----------------------------------------------------------------------


def _multi_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] obtained by summing the gradients
    of the listed active shifts.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def _expgrad_multi_opt(a0: np.ndarray,
                       deadline: float,
                       start_time: float,
                       fft_len: int,
                       eta0: float = 0.6,
                       delta_frac: float = 0.001,
                       top_k: int = 10) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the true max‑convolution objective,
    using a multi‑gradient built from the top‑K active shifts.
    Returns the best vector ever seen and its exact score.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Multi‑ExpGrad improve: iter={t:,}, score={cur_score:.6f}")

        # ----- active shifts -----
        thresh = cur_max - max(delta_frac * cur_max, 1e-12)
        active = np.where(conv >= thresh)[0]

        if active.size > top_k:
            part = np.argpartition(-conv[active], top_k - 1)[:top_k]
            active = active[part]

        # ----- sub‑gradient -----
        g = _multi_gradient(a, active)

        # ----- step size -----
        eta = eta0 / math.sqrt(t)

        # ----- exponentiated update -----
        a = a * np.exp(-eta * g)

        # ----- re‑normalise -----
        s = a.sum()
        if s == 0.0 or np.isnan(s) or np.isinf(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        a = np.maximum(a, 1e-15)

        if t % 500 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic greedy peak reduction
# ----------------------------------------------------------------------


def _greedy_local_improve(a0: np.ndarray,
                          deadline: float,
                          start_time: float,
                          max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Shift a fraction of the coefficient that most contributes to the
    current convolution peak towards the smallest coefficient.
    Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]          # modest fraction of the source mass
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
# Random hill‑climbing (very cheap moves)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                deadline: float,
                start_time: float,
                max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Simple random pairwise transfers.  Accept only strict improvements.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        src, dst = rng.choice(n, size=2, replace=False)
        if a[src] < 1e-12:
            continue
        eps = a[src] * rng.random() * 0.1   # up to 10 % of src
        a_tmp = a.copy()
        a_tmp[src] -= eps
        a_tmp[dst] += eps
        a_tmp = np.maximum(a_tmp, 0.0)
        a_tmp /= a_tmp.sum()
        conv_tmp = _conv_self(a_tmp, _next_pow2(2 * n - 1))
        score_tmp = 2.0 * n * conv_tmp.max()
        if score_tmp < best_score:
            best_score = score_tmp
            best_a = a_tmp.copy()
            a = a_tmp
            conv = conv_tmp
            cur_score = score_tmp
            print(f"[{time.time() - start_time:6.2f}s] hill‑climb iter={it:,} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine consists of:
        1. a cheap LP‑based polishing on a reduced problem,
        2. up‑sampling,
        3. Adam warm‑up,
        4. a refined exponentiated‑gradient phase,
        5. deterministic greedy improvement,
        6. optional random hill‑climbing.
    All phases respect the supplied time budget.
    Returns the best sequence (list of floats) found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Sizes
    # ------------------------------------------------------------------
    n_big = 4800                                   # target length
    n_small = 300                                  # LP works comfortably here
    fft_len_big = _next_pow2(2 * n_big - 1)
    fft_len_small = _next_pow2(2 * n_small - 1)

    # ------------------------------------------------------------------
    # 0) Build a collection of small‑scale start vectors
    # ------------------------------------------------------------------
    starts_small = []

    # uniform
    starts_small.append(np.full(n_small, 1.0 / n_small))

    # linear decreasing
    lin = np.arange(n_small, 0, -1, dtype=float)
    starts_small.append(lin / lin.sum())

    # exponential decays
    starts_small.append(_exponential_start(n_small, r=0.99))
    starts_small.append(_exponential_start(n_small, r=0.97))

    # Dirichlet draws with several concentrations
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            starts_small.append(_dirichlet_start(n_small, conc, rng))

    # if a previous best is available, up‑sample it
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            seq_small = _upsample(seq, n_small)
            seq_small = np.maximum(seq_small, 1e-12)
            seq_small /= seq_small.sum()
            starts_small.append(seq_small)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) Pick the best among the raw small starts (objective on the tiny problem)
    # ------------------------------------------------------------------
    best_small = None
    best_small_score = float("inf")
    for idx, a0 in enumerate(starts_small):
        conv = _conv_self(a0, fft_len_small)
        score = 2.0 * n_small * conv.max()          # exact objective for sum=1
        if score < best_small_score:
            best_small_score = score
            best_small = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] small start #{idx+1:2d} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑based iterative polishing (≈15 % of the budget)
    # ------------------------------------------------------------------
    lp_deadline = start_time + budget_s * 0.15
    f = best_small.copy()
    best_f = f.copy()
    best_f_score = best_small_score
    lp_iter = 0

    while time.time() < lp_deadline and lp_iter < 250:
        lp_iter += 1
        conv_f = _conv_self(f, fft_len_small)
        M = conv_f.max()
        g = _lp_optimize_step(f, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        if g.sum() == 0.0:
            break
        g_norm = g / g.sum()

        # diminishing mixing coefficient
        eta = 0.3 / math.sqrt(lp_iter)
        f_new = (1.0 - eta) * f + eta * g_norm
        f_new = np.maximum(f_new, 1e-15)
        f_new /= f_new.sum()

        conv_new = _conv_self(f_new, fft_len_small)
        score_new = 2.0 * n_small * conv_new.max()
        if score_new < best_f_score:
            best_f_score = score_new
            best_f = f_new.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve iter={lp_iter} score={score_new:.6f}")

        f = f_new

    # ------------------------------------------------------------------
    # 2b) A few cheap random perturbations (still within the LP window)
    # ------------------------------------------------------------------
    perturb_deadline = start_time + budget_s * 0.20
    while time.time() < perturb_deadline:
        i, j = rng.integers(0, n_small, size=2, endpoint=False)
        if i == j or f[i] < 1e-12:
            continue
        delta = f[i] * rng.random() * 0.2
        f_tmp = f.copy()
        f_tmp[i] -= delta
        f_tmp[j] += delta
        f_tmp = np.maximum(f_tmp, 0.0)
        f_tmp /= f_tmp.sum()
        conv_tmp = _conv_self(f_tmp, fft_len_small)
        score_tmp = 2.0 * n_small * conv_tmp.max()
        if score_tmp < best_f_score:
            best_f_score = score_tmp
            best_f = f_tmp.copy()
            f = f_tmp
            print(f"[{time.time() - start_time:6.2f}s] random perturb improve score={score_tmp:.6f}")

    # ------------------------------------------------------------------
    # 3) Upsample to the full length
    # ------------------------------------------------------------------
    f_big = _upsample(best_f, n_big)
    f_big = np.maximum(f_big, 0.0)
    f_big /= f_big.sum()

    # ------------------------------------------------------------------
    # 4) Adam warm‑up (≈10 % of the remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_deadline = time.time() + remaining * 0.10
    a_adam, score_adam = _adam_optimize(f_big, adam_deadline, start_time, fft_len_big)
    if score_adam < 2.0 * n_big * _conv_self(f_big, fft_len_big).max():
        best_seq = a_adam
        best_score = score_adam
        print(f"[{time.time() - start_time:6.2f}s] Adam new global best {best_score:.6f}")
    else:
        best_seq = f_big
        best_score = 2.0 * n_big * _conv_self(f_big, fft_len_big).max()

    # ------------------------------------------------------------------
    # 5) Exponentiated‑gradient polishing (≈70 % of the total budget)
    # ------------------------------------------------------------------
    polish_deadline = deadline - 30.0                # keep ~30 s for final tweaks
    a_polish, score_polish = _expgrad_multi_opt(
        best_seq, polish_deadline, start_time, fft_len_big,
        eta0=0.6, delta_frac=0.001, top_k=10
    )
    if score_polish < best_score:
        best_seq = a_polish
        best_score = score_polish
        print(f"[{time.time() - start_time:6.2f}s] ExpGrad new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Deterministic greedy refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_greedy, score_greedy = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if score_greedy < best_score:
            best_seq = a_greedy
            best_score = score_greedy
            print(f"[{time.time() - start_time:6.2f}s] Greedy new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Optional random hill‑climbing in any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_hc, score_hc = _hill_climb(best_seq, deadline, start_time, max_iters=10000)
        if score_hc < best_score:
            best_seq = a_hc
            best_score = score_hc
            print(f"[{time.time() - start_time:6.2f}s] Hill‑climb new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety normalisation and reporting
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n_big)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
