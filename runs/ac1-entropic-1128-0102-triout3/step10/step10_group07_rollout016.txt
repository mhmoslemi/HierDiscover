"""
Propose a step function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Build a diverse pool of initial candidates (uniform, Gaussian,
  exponential, Dirichlet draws, sparse random supports, the previous
  best if available) and keep the best raw one.
* Run a *smooth‑max* mirror‑descent (Adam on the log‑variables) for the
  majority of the time budget.  The smooth max  ``S_γ(a)=
  (1/γ)log Σ exp(γ·(a*a)[k])`` approximates the true max and yields a
  differentiable objective.  Its gradient is a correlation of the
  current vector with the soft‑max weights and is evaluated by FFTs.
* Perform a few short Adam restarts from fresh random starts to escape
  local basins.
* Finish with the deterministic greedy “peak‑reduction’’ step used in
  the earlier code.
* Clip, re‑normalise and return the best sequence found.

import time

# Helper utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _gaussian_start(n: int) -> np.ndarray:
    """Centered discrete Gaussian (σ≈n/6) normalised to sum‑one."""
    sigma = n / 6.0
    xs = np.arange(n)
    g = np.exp(-0.5 * ((xs - (n - 1) / 2) / sigma) ** 2)
    return g / g.sum()


def _dirichlet_start(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration and normalise."""
    x = rng.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _exponential_start(n: int, r: float = 0.9995) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised."""
    vec = np.array([r ** i for i in range(n)], dtype=float)
    return vec / vec.sum()


def _sparse_random_start(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly choose `m` indices, give each equal mass 1/m, zero elsewhere."""
    idx = rng.choice(n, size=m, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / m
    return vec


def _greedy_local_improve(a0: np.ndarray,

    contributes most to the current convolution peak towards the smallest
    coefficient.  Accept only strictly decreasing moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = 2.0 * n * _conv_self(a, _next_pow2(2 * n - 1)).max()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a, _next_pow2(2 * n - 1))
        cur_max = conv.max()
        k_max = conv.argmax()

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        contrib = a[i_idx] * a[j_idx]
        src = i_idx[contrib.argmax()]

        dst_candidates = np.delete(np.arange(n), src)
        dst = dst_candidates[a[dst_candidates].argmin()]

        eps = 0.25 * a[src]
        if eps <= 1e-14:
            break
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_max = _conv_self(a_prop, _next_pow2(2 * n - 1)).max()
        if new_max < cur_max:
            a = a_prop
            best_score = 2.0 * n * new_max
            print(f"[{time.time() - start_time:6.2f}s] greedy improve: max={new_max:.6f}")

    return a, best_score


# --------------------------------------------------------------
# Smooth‑max Adam optimiser
# --------------------------------------------------------------


def _adam_optimize(start_a: np.ndarray,
                   deadline: float,
                   start_time: float,
                   n: int,
                   L: int,
                   fft_len_self: int,
                   fft_len_grad: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence (as a plain ndarray) and its exact score.
    """
    # initialise log‑variables (to keep positivity)
    a = np.maximum(start_a, 1e-15)
    a = a / a.sum()
    x = np.log(a)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    t = 0

    while time.time() < deadline:
        t += 1

        # reconstruct a from log‑variables and enforce simplex
        a = np.exp(x)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # ----- exact max‑conv (for reporting & eventual early stop) -----
        conv_self = _conv_self(a, fft_len_self)
        cur_max = conv_self.max()
        cur_score = 2.0 * n * cur_max  # denominator = 1 because sum(a)=1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] Adam improve: iter={t:,}, score={cur_score:.6f}")

        # ----- smooth‑max weights -----
        # increase the smoothing parameter gradually
        gamma = min(30.0, 0.02 * t)  # starts small, grows up to ≈30
        # stabilise numerically
        w_raw = np.exp(gamma * (conv_self - cur_max))
        w = w_raw / w_raw.sum()          # shape (L,)

        # ----- gradient of smooth‑max (correlation a ⋆ w) -----
        w_rev = w[::-1]                  # length L
        Af = np.fft.rfft(a, n=fft_len_grad)
        Wf = np.fft.rfft(w_rev, n=fft_len_grad)
        corr = np.fft.irfft(Af * Wf, n=fft_len_grad)   # length n+L-1
        grad_a = 2.0 * corr[L - 1: L - 1 + n]          # take the valid part

        # ----- Adam step on log‑variables -----
        grad_x = a * grad_a
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # simple decay of the base learning rate
        lr = 0.05 / (math.sqrt(t) + 1.0)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation (every ~300 iterations)
        if t % 300 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Adam exact eval: iter={t}, score={sc:.6f}")

    # final exact score of the best seen vector
    final_score = evaluate_sequence(best_a.tolist()) if best_a is not None else float("inf")
    return best_a, final_score


# --------------------------------------------------------------
# Main driver
# --------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns a list of floats (the sequence).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # problem size (use the same length as the previous best)
    # ------------------------------------------------------------------
    n = 4800
    L = 2 * n - 1
    fft_len_self = _next_pow2(L)                # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)        # for correlation a ⋆ w

    # ------------------------------------------------------------------
    # Build a pool of diverse initial candidates
    # ------------------------------------------------------------------
    starts = []

    # 1) previous best (if the global variable exists)
    try:
        seq = np.asarray(height_sequence_1, dtype=float)
        if seq.ndim == 1 and seq.size > 0:
            if seq.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                seq = np.interp(new_x, old_x, seq)
            starts.append(seq / seq.sum())
    except Exception:
        pass

    # 2) handcrafted starters
    starts.append(np.full(n, 1.0 / n))               # uniform
    starts.append(_gaussian_start(n))                # bell
    starts.append(_exponential_start(n, r=0.9999))   # very gentle decay
    starts.append(_exponential_start(n, r=0.998))    # stronger decay

    # 3) Dirichlet draws with various concentrations
    for conc in [0.1, 0.3, 0.5, 1.0, 2.0]:
        starts.append(_dirichlet_start(n, conc=conc, rng=rng))

    # 4) sparse random supports
    for m in [300, 600, 1200, 2400]:
        for _ in range(2):
            starts.append(_sparse_random_start(n, m, rng))

    # ------------------------------------------------------------------
    # Evaluate the raw starts and keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, a0 in enumerate(starts):
        sc = evaluate_sequence(a0.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a0.copy()
        print(f"[{time.time() - start_time:6.2f}s] start #{idx + 1:2d} raw score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation (≈ 70 % of total budget)
    # ------------------------------------------------------------------
    main_deadline = start_time + 0.70 * (deadline - start_time)
    polished_seq, polished_score = _adam_optimize(
        best_seq, main_deadline, start_time, n, L,
        fft_len_self, fft_len_grad
    )
    if polished_score < best_score:
        best_score = polished_score
        best_seq = polished_seq.copy()
        print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after Adam score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short Adam bursts from fresh random starts (≈ 20 % of time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:
        # choose a fresh start (Dirichlet with random concentration)
        conc = 10 ** rng.uniform(-2.0, 1.0)          # roughly 0.01 … 10
        cand_start = _dirichlet_start(n, conc=conc, rng=rng)

        sub_deadline = time.time() + 6.5            # a few seconds per run
        cand_seq, cand_score = _adam_optimize(
            cand_start, sub_deadline, start_time, n, L,
            fft_len_self, fft_len_grad
        )
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST from restart score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy peak‑reduction
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_seq, final_score = _greedy_local_improve(
            best_seq, deadline, start_time, max_iters=5000
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] *** NEW GLOBAL BEST after greedy score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety round‑trip
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, None)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_reported = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:6.2f}s] final reported score = {final_reported:.6f}")

    return [float(x) for x in best_seq.tolist()]
