"""
Improved search for a non‑negative step function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

We use a four‑stage pipeline:

1️⃣  A dense parametric sweep over two families
    a_i ∝ (i+1)^p·exp(-λ·i)      (geometric × power)
    a_i ∝ exp(-c·i²)            (Gaussian)
    on a larger problem size (n = 32768).  The best candidate becomes the
    starting point for the subsequent optimisation phases.

2️⃣  Projected Adam optimisation on the exact L∞‑objective.
    The Adam updates use the sub‑gradient of the active convolution entry,
    Euclidean projection onto the probability simplex and a decaying step‑size.
    Adam’s adaptive moments give a more aggressive search than vanilla
    sub‑gradient descent.

3️⃣  Multiplicative (exponentiated‑gradient) fine‑tuning – identical to the
    previous approach but run for a shorter budget.

4️⃣  Final random hill‑climb that moves tiny masses between random indices
    and accepts only improving moves.

All phases respect the 1000 s time budget, print timing information and
return a plain Python list (sum = 1) that can be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_a_w_fft(a: np.ndarray, w: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Convolution of a (length n) with w (length 2·n‑1). Returns the first n
    entries which correspond to g_i = Σ_j a_j·w_{i+j}.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(n + w.shape[0] - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)
    conv = np.fft.irfft(Af * Wf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (objective) – works for arbitrary sum
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return normalised a_i ∝ (i+1)^p·exp(-λ·i)."""
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)           # for numerical stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Return normalised a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used by Adam)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the index where the maximum is attained.
    For each i involved in the max sum (i+j=k) we get 2·a_j.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    np.add.at(g, i, 2.0 * a[j])
    return g


# ----------------------------------------------------------------------
# Adam optimiser (projected) – stage 2
# ----------------------------------------------------------------------


def _adam_descent(a0: np.ndarray,
                  L_fft: int,
                  deadline: float,
                  start_time: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the exact max‑convolution objective.
    Uses the sub‑gradient at the active index, keeps the iterate on the
    probability simplex and applies bias‑corrected first/second moments.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.5      # base step‑size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve = start_time

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)

        # decreasing learning rate
        alpha = alpha0 / (1.0 + 0.0004 * it)
        step = -alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = a + step
        cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"score={cand_score:.6f}")
        else:
            # occasional jitter to shake off plateaus
            if it % 800 == 0:
                eps_j = rng.random(a.shape) * a * 3e-4
                a = np.maximum(a + eps_j, 0.0)
                a = _project_simplex(a)

        # restart from best if we have been stuck for a while
        if (time.time() - last_improve > 30.0) and (it % 2500 == 0):
            eps_j = rng.random(best.shape) * best * 5e-4
            a = np.maximum(best + eps_j, 0.0)
            a = _project_simplex(a)
            last_improve = time.time()

    return best, best_score

def _expgrad_phase(a0: np.ndarray,

                   rng: np.random.Generator) -> tuple[np.ndarray, float]:

    a = a0.copy()

    best_score = _eval_fast(best, L_fft)
    it = 0

        if it % 2500 == 0:
            eps = rng.random(a.shape) * a * 5e-5
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score

def _hill_climb(a0: np.ndarray,

            cand = cand / cand.sum()
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum → 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 32768 provides a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 32768
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – resample if needed.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # linear interpolation to match our chosen length
                old_idx = np.arange(starter.size, dtype=float)
                new_idx = np.linspace(0, starter.size - 1, n)
                starter = np.interp(new_idx, old_idx, starter)
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0.0:
                starter = starter / total
            else:
                starter = uniform.copy()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fine parametric sweep (≈80 s)
    # ------------------------------------------------------------------
    sweep_end = start_time + 80.0
    i_arr = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_arr + 1.0)

    # geometric × power family
    lam_grid = np.linspace(1e-6, 3e-3, 180)   # ~180 points
    p_grid = np.linspace(-8.0, 5.0, 111)      # step ≈0.12

    for lam in lam_grid:
        if time.time() >= sweep_end:
            break
        base = -lam * i_arr
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.2e} p={p:.2f} score={sc:.6f}")

    # pure Gaussian family (exp(-c·i²))
    c_grid = np.logspace(-9.0, -5.5, 40)      # 40 points
    for c in c_grid:
        if time.time() >= sweep_end:
            break
        log_shape = -c * (i_arr * i_arr)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gaussian c={c:.2e} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Projected Adam optimisation – allocate ~50 % of remaining budget.
    # ------------------------------------------------------------------
    adam_end = start_time + 0.5 * budget_s
    if time.time() < adam_end:
        best_seq, best_score = _adam_descent(best_seq, L_fft,
                                             adam_end, start_time,
                                             rng)

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient fine‑tuning – allocate ~25 % of remaining budget.
    # ------------------------------------------------------------------
    exp_end = start_time + 0.75 * budget_s
    if time.time() < exp_end:
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_end, start_time,
                                              rng)

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb – use whatever time is left.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_fft,
                                           deadline, start_time,
                                           rng)

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Report final bound using the provided evaluate_sequence (optional).
    # ------------------------------------------------------------------
    try:
        from __main__ import evaluate_sequence
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
