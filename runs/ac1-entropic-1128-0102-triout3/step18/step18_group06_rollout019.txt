    a = a0.copy()

"""
Search for a non‑negative step function minimising

    2 * n * max(convolve(a, a)) / (sum(a)**2)

The routine consists of four phases:

1️⃣  Fast parametric sweeps (geometric, power, exponential, two‑block and a
    Beta‑shape family) to obtain strong starting points.

2️⃣  Global exploration by many random Dirichlet draws – both dense and
    sparse regimes.

3️⃣  Smoothed mirror‑descent on the true objective.
    The hard max is replaced by a softmax (log‑sum‑exp) with a temperature
    that grows during optimisation; the exact gradient of the smooth
    surrogate is obtained via a correlation.  After a multiplicative
    update the iterate is projected back onto the simplex.

4️⃣  Simple hill‑climbing / simulated‑annealing refinements that move a
    small amount of mass between two coordinates or mix in a fresh Dirichlet
    sample.  All phases respect a global 1000 s CPU budget and the best
    sequence found is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based self‑convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta_shape(n, best_seq, best_score, deadline):
    """
    a_i ∝ (i+½)^α * (n‑i‑½)^β  (Beta‑type shape, both α and β in [‑2,2]).
    """
    idx = np.arange(n, dtype=np.float64) + 0.5               # avoid zero
    rev = n - idx
    for α in np.linspace(-2.0, 2.0, 41):
        if time.time() >= deadline:
            break
        for β in np.linspace(-2.0, 2.0, 41):
            if time.time() >= deadline:
                break
            seq = np.power(idx, α) * np.power(rev, β)
            # protect against infinities when α or β are negative
            seq = np.where(np.isfinite(seq), seq, 0.0)
            tot = seq.sum()
            if tot == 0:
                continue
            seq /= tot
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={α:.2f} β={β:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (cheap global exploration)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter: log‑uniform between 10⁻² and 10⁰·⁷
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Smoothed mirror‑descent (soft‑max surrogate of the max‑convolution)
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray, deadline: float,
                        best_seq: np.ndarray, best_score: float):
    """
    Mirror‑descent on a smooth approximation of max(conv(a,a)).
    Soft‑max temperature γ grows slowly; step size η_k = η0 / sqrt(k).
    """
    try:
        import scipy.signal as signal
    except Exception:  # pragma: no cover
        signal = None

    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5
    gamma = 5.0               # start with a modest smoothing

    while time.time() < deadline:
        it += 1
        # --- compute soft‑max weights over the convolution ----------
        conv = _conv_self(a)            # length 2·n‑1
        maxc = conv.max()
        # numerical stability: shift by maxc
        expw = np.exp(gamma * (conv - maxc))
        w = expw / expw.sum()           # probabilities, sum to 1

        # --- gradient of the smooth max (via correlation)  ----------
        if signal is not None:
            # correlate(w, a) with 'valid' returns length n = desired gradient size
            grad = 2.0 * signal.correlate(w, a, mode='valid')
        else:
            # fallback – explicit but slower O(n²)
            grad = np.empty(n, dtype=np.float64)
            for i in range(n):
                grad[i] = 2.0 * np.dot(w[i:i + n], a)
        # --- multiplicative mirror‑descent update --------------------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = _project_simplex(a)          # stay on the simplex

        # --- assess improvement ------------------------------------
        conv_new = _conv_self(a)
        sc = float(2.0 * n * conv_new.max())
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothDesc it={it:,} score={sc:.6f}")

        # slowly increase the sharpening parameter
        if it % 1000 == 0:
            gamma = min(gamma * 1.2, 30.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣ Simple hill‑climbing / simulated‑annealing refinement
# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float):
    """
    Move a small δ of mass from one coordinate to another (or mix with a
    fresh Dirichlet draw) and keep improvements.  An annealing rule
    occasionally accepts a worse move to escape shallow plateaux.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        if rng.random() < 0.85:
            # Local transfer of mass between two positions
            i, j = rng.choice(n, size=2, replace=False)
            # move at most 10 % of the source mass
            max_move = min(0.1 * a[i], a[i])
            eps = rng.uniform(0.0, max_move)
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps
        else:
            # Global mixing with a fresh sparse Dirichlet sample
            t = 0.02
            conc = rng.uniform(0.01, 5.0)
            g = rng.gamma(shape=conc, scale=1.0, size=n)
            g_sum = g.sum()
            if g_sum == 0:
                continue
            g /= g_sum
            a_new = (1.0 - t) * a + t * g

        # Numerical safety
        a_new = np.maximum(a_new, 0.0)
        tot = a_new.sum()
        if tot > 0:
            a_new /= tot

        sc = _objective_score(a_new)
        if sc < best_score:
            best_score, best_seq, a = sc, a_new.copy(), a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] HillClimb it={it:,} new best {sc:.6f}")
        else:
            # Simulated‑annealing acceptance
            remaining = deadline - time.time()
            total = deadline - start_time
            T = max(0.0, remaining / total) * 1e-2
            if T > 0 and rng.random() < np.exp(-(sc - best_score) / T):
                a = a_new.copy()

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises the
    evaluation function.  The routine runs several quick parametric sweeps,
    a large Dirichlet exploration, a smoothed mirror‑descent optimisation and
    a final hill‑climbing phase.  The best sequence (normalised to sum = 1) is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) if provided by the runner
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Parametric sweeps (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = min(deadline - 700.0, start_time + 150.0)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_beta_shape(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈300 s)
    # --------------------------------------------------------------
    phase_deadline = min(deadline - 350.0, start_time + 450.0)
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Smoothed mirror‑descent (≈350 s)
    # --------------------------------------------------------------
    phase_deadline = min(deadline - 200.0, start_time + 800.0)
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_max_descent(best_seq, phase_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End of smooth descent, best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Local hill‑climbing / annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting local hill‑climb")
        best_seq, best_score = _local_hill_climb(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End of hill‑climb, best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
