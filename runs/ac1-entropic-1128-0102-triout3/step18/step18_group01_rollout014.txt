"""
Improved search for the step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm uses a *smooth* approximation of the max‑convolution
and runs a (L‑BFGS) quasi‑Newton optimisation on the simplex
by re‑parameterising the probabilities with a soft‑max (logits u).
A continuation on the smoothing parameter γ (small → large) drives the
solution towards the true non‑smooth objective.
After the quasi‑Newton phase a cheap random hill‑climb is applied.

The whole search respects the 1000 s budget and prints progress
information.  The final vector is returned as a plain Python list of
floats (sum normalised to 1, all entries ≥0).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Exact evaluation of the original objective (used for monitoring)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = np.fft.irfft(np.fft.rfft(a, n=L_fft) ** 2, n=L_fft)[: 2 * a.shape[0] - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smooth max‑convolution objective and gradient (log‑softmax re‑param.)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(
    u: np.ndarray,
    n: int,
    L_big: int,
    gamma: float,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_u) where
        f   = 2·n·smooth_max(conv(a,a))
        grad_u = ∂f/∂u   (u are logits, a = softmax(u))
    The smooth max is the log‑sum‑exp of the linear convolution,
    with smoothing parameter γ.
    """
    # ----- logits → probabilities (soft‑max) -----
    max_u = np.max(u)
    exp_u = np.exp(u - max_u)
    a = exp_u / exp_u.sum()                     # a >= 0, sum(a)=1

    # ----- FFT of a (single transform, reused) -----
    A_fft = np.fft.rfft(a, n=L_big)

    # ----- linear self‑convolution a*a (full length 2·n‑1) -----
    conv_full = np.fft.irfft(A_fft * A_fft, n=L_big)[: 2 * n - 1]

    # ----- smooth maximum of the convolution -----
    max_c = conv_full.max()
    # stable log‑sum‑exp
    exp_weights = np.exp(gamma * (conv_full - max_c))
    sum_exp = exp_weights.sum()
    smooth_max = max_c + np.log(sum_exp) / gamma
    f = 2.0 * n * smooth_max

    # ----- normalised weights w = softmax(γ·conv) -----
    w = exp_weights / sum_exp                     # length 2·n‑1, sums to 1

    # ----- cross‑correlation a ⋆ w (needed for the gradient) -----
    W_fft = np.fft.rfft(w, n=L_big)
    # correlation = IFFT(conj(FFT(a)) * FFT(w))
    corr_full = np.fft.irfft(A_fft.conj() * W_fft, n=L_big)

    # zero‑lag index for the correlation (len(w)-1)
    offset = 2 * n - 2
    corr_seg = corr_full[offset : offset + n]    # length n

    # ----- gradient w.r.t. a (df/da) -----
    # d smooth_max / da = 2 * (a ⋆ w)
    # therefore df/da = 2·n * d smooth_max/da = 4·n * (a ⋆ w)
    grad_a = 4.0 * n * corr_seg                  # shape (n,)

    # ----- chain rule to get gradient w.r.t. logits u -----
    avg = np.dot(a, grad_a)                      # scalar = a·grad_a
    grad_u = a * (grad_a - avg)                  # shape (n,)

    return f, grad_u


# ----------------------------------------------------------------------
# Simple random hill‑climb (fine tuning after the main optimisation)
# ----------------------------------------------------------------------


def _hill_climb(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with a deterministic acceptance rule."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={sc:.6f}")

        # occasional bigger exploratory jump
        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={sc:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 16384 gives a good balance of quality vs speed.
    # ------------------------------------------------------------------
    n = 16384
    L_fft = _next_pow2(2 * n - 1)               # for the *exact* objective
    L_big = _next_pow2(3 * n - 2)               # for the smooth‑max gradient

    # ------------------------------------------------------------------
    # Starting points (uniform + a few random Dirichlet draws)
    # ------------------------------------------------------------------
    init_vectors = []

    # 1 – uniform
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2 – random Dirichlet, moderately spiky (α = 0.5)
    init_vectors.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # 3 – random Dirichlet, smoother (α = 2.0)
    init_vectors.append(rng.dirichlet(alpha=np.full(n, 2.0)))

    # If a global starter exists, up‑sample / pad it
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = None
            else:
                # pad / truncate to length n
                if starter.size < n:
                    pad = np.zeros(n - starter.size, dtype=starter.dtype)
                    starter = np.concatenate([starter, pad])
                else:
                    starter = starter[:n]
                starter = starter / starter.sum()
                init_vectors.append(starter)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Helper to evaluate the *exact* objective quickly
    # ------------------------------------------------------------------
    def exact_score(vec: np.ndarray) -> float:
        return _eval_fast(vec, L_fft)

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation: smooth‑max L‑BFGS with a continuation on γ
    # ------------------------------------------------------------------
    gamma_schedule = [5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]

    # try each initial vector (few restarts)
    for idx, a0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break

        # initialise logits u = log(a)  (add tiny eps for safety)
        eps = 1e-12
        u = np.log(a0 + eps)

        print(
            f"[{time.time() - start_time:.2f}s] "
            f"restart {idx + 1}/{len(init_vectors)} – initial exact score {exact_score(a0):.6f}"
        )

        for gamma in gamma_schedule:
            if time.time() >= deadline:
                break

            # allocate a modest number of LBFGS iterations (time‑aware)
            remain = deadline - time.time()
            max_iter = max(30, min(250, int(remain // 0.03)))  # rough 30 ms / iter

            # ------------------------------------------------------------------
            # Try to use SciPy's L‑BFGS‑B; fall back to a very simple GD if missing.
            # ------------------------------------------------------------------
            try:
                from scipy.optimize import minimize

                res = minimize(
                    fun=lambda uu: _smooth_obj_and_grad(uu, n, L_big, gamma),
                    x0=u,
                    jac=True,
                    method="L-BFGS-B",
                    options={"maxiter": max_iter, "disp": False, "gtol": 1e-12},
                )
                u = res.x
            except Exception:
                # Very simple projected‑gradient descent on the logits
                step0 = 0.6
                for it in range(max_iter):
                    f_val, grad_u = _smooth_obj_and_grad(u, n, L_big, gamma)
                    step = step0 / math.sqrt(it + 1)
                    u = u - step * grad_u

            # evaluate the *true* (non‑smoothed) objective
            max_u = np.max(u)
            a = np.exp(u - max_u)
            a = a / a.sum()
            sc = exact_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] "
                    f"gamma {gamma:.0f} – new best exact score {sc:.6f}"
                )

    # ------------------------------------------------------------------
    # If nothing succeeded (should not happen), fall back to uniform.
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = exact_score(best_a)

    # ------------------------------------------------------------------
    # Final random hill‑climb – uses whatever time is left.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _hill_climb(
            best_a, L_fft, deadline, start_time, rng
        )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as plain Python list.
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Report final bound using the supplied evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}"
        )
    except Exception:
        print(
            f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}"
        )

    return [float(v) for v in best_a]
