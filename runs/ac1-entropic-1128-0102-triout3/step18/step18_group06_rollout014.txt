def _random_start(n, rng):
    choice = rng.integers(0,4)
    if choice == 0: # Dirichlet
        logc = rng.uniform(-3, 1)
        c = 10 ** logc
        a = rng.gamma(c, size=n)
        a /= a.sum()
        return a
    elif choice == 1: # two blocks
        L_frac = rng.uniform(0.01, 0.2)
        L = max(1, int(L_frac * n))
        w = rng.uniform(0.1, 0.9)
        seq = np.zeros(n)
        seq[:L] = w / L
        seq[-L:] = (1 - w) / L
        return seq
    elif choice == 2: # geometric
        eps = rng.uniform(1e-6, 0.5)
        r = 1 - eps
        idx = np.arange(n)
        geo = r ** idx
        geo_sum = geo.sum()
        geo /= geo_sum
        lam = rng.uniform(0.0, 1.0)
        seq = lam * np.full(n, 1.0/n) + (1-lam) * geo
        seq /= seq.sum()
        return seq
    else: # Weibull shape
        p = rng.uniform(0.5, 5.0)
        sigma = rng.uniform(0.1, 1.0) * n
        idx = np.arange(n) / (n-1)
        seq = np.exp(- (idx / sigma) ** p)
        seq /= seq.sum()
        return seq

    eps = 1e-12
    w = np.log(seq + eps)
    cur_seq = seq.copy()
    cur_score = _objective_score(cur_seq)
    rng = np.random.default_rng()

    init_sigma = 0.5   # initial std of perturbation (in log-space)
    init_temp = 0.01
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total
        sigma = init_sigma * (1 - frac) + 1e-4
        temp = init_temp * (1 - frac) + 1e-6

        maxw = proposal.max()
        p = np.exp(proposal - maxw)  # to avoid overflow

    strong candidates quickly.

2️⃣  Global exploration with random Dirichlet draws.

3️⃣  Many short sub‑gradient (mirror‑descent) runs from random restarts.
    The previous implementation used a single long run – we instead
    perform dozens of independent runs (each a few thousand iterations)
    to escape local minima.

4️⃣  A final simulated‑annealing phase on the log‑probabilities.  The
    update is multiplicative (keeps the iterate on the simplex) and the
    temperature/step‑size decay is linear in the remaining time.

All work is done with FFT based convolutions, which are fast enough to
allow thousands of objective evaluations within the 1000 s budget.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.special import gammaln  # for binomial‑like families (optional)

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_edge_power(n, best_seq, best_score, deadline):
    """a_i ∝ min(i+1,n-i)^p  (p may be negative or positive)."""
    idx = np.arange(n, dtype=np.float64)
    d = np.minimum(idx + 1, n - idx)  # distance from nearest edge (1‑based)
    for p in np.linspace(-2.5, 2.5, 101):
        if time.time() >= deadline:
            break
        seq = np.power(d, p)
        s = seq.sum()
        if s == 0 or np.isnan(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] EdgePower p={p:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_gaussian(n, best_seq, best_score, deadline):
    """Symmetric Gaussian centred at the middle."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma_grid = np.logspace(np.log10(0.02 * n), np.log10(0.4 * n), 90)
    for sigma in sigma_grid:
        if time.time() >= deadline:
            break
        seq = np.exp(-((idx - mu) ** 2) / (2.0 * sigma ** 2))
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Gaussian σ={sigma:.1f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta(n, best_seq, best_score, deadline):
    """Beta‑like shape: a_i ∝ (i+1)^α (n-i)^β."""
    idx = np.arange(n, dtype=np.float64)
    a_grid = np.linspace(-2.0, 2.0, 41)
    b_grid = np.linspace(-2.0, 2.0, 41)
    for a in a_grid:
        if time.time() >= deadline:
            break
        p1 = np.power(idx + 1.0, a)
        for b in b_grid:
            if time.time() >= deadline:
                break
            seq = p1 * np.power(n - idx, b)
            s = seq.sum()
            if s == 0 or np.isnan(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta a={a:.2f} b={b:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws (cheap global search)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        logc = rng.uniform(-2.0, 0.7)          # c ∈ [10⁻²,5]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Sub‑gradient descent with many restarts
# ----------------------------------------------------------------------
def _subgradient_restart(a0: np.ndarray, max_iters: int,
                         stop_time: float, best_seq: np.ndarray,
                         best_score: float):
    """
    Mirror‑descent from a0 for at most max_iters iterations or until
    stop_time, updating best_seq / best_score when an improvement occurs.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5
    while it < max_iters and time.time() < stop_time:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad restart it={it:,} obj={obj:.6f}")

        # active set (tolerance relative to max)
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]
        m = len(active)
        if m == 0:
            break

        grad = np.zeros_like(a)
        # compute sub‑gradient for each active index
        for i in active:
            lo = max(0, i - (n - 1))
            hi = min(i, n - 1)
            # j runs from lo … hi, k = i‑j
            # we add a[k] to grad[j]
            j = np.arange(lo, hi + 1)
            k = i - j
            grad[lo:hi + 1] += a[k]
        grad = 2.0 * grad / m   # average over active constraints

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
    return best_seq, best_score


def _random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random start (Dirichlet, two‑block, geometric or Weibull)."""
    choice = rng.integers(0, 4)
    if choice == 0:  # Dirichlet
        logc = rng.uniform(-3.0, 1.0)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        return a
    if choice == 1:  # Two blocks
        L_frac = rng.uniform(0.01, 0.2)
        L = max(1, int(L_frac * n))
        w = rng.uniform(0.1, 0.9)
        seq = np.zeros(n)
        seq[:L] = w / L
        seq[-L:] = (1.0 - w) / L
        return seq
    if choice == 2:  # Geometric mixture
        eps = rng.uniform(1e-6, 0.4)
        r = 1.0 - eps
        idx = np.arange(n, dtype=np.float64)
        geo = np.power(r, idx)
        geo /= geo.sum()
        lam = rng.uniform(0.0, 1.0)
        seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
        seq /= seq.sum()
        return seq
    # Weibull‑like shape
    p = rng.uniform(0.5, 5.0)
    scale = rng.uniform(0.05, 0.4) * n
    idx = np.arange(n, dtype=np.float64) / (n - 1)
    seq = np.exp(- (idx / scale) ** p)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# 4️⃣  Simulated annealing on log‑probabilities
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float):
    """
    Multiplicative (log‑space) simulated annealing.
    """
    n = start_seq.shape[0]
    rng = np.random.default_rng()
    eps = 1e-12
    w = np.log(start_seq + eps)          # log‑weights
    cur_seq = start_seq.copy()
    cur_score = _objective_score(cur_seq)

    # annealing schedule parameters (tuned heuristically)
    init_sigma = 0.6
    init_temp = 0.02

    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        sigma = init_sigma * (1.0 - frac) + 1e-4
        temp = init_temp * (1.0 - frac) + 1e-6

        # propose a new log‑vector
        proposal = w + sigma * rng.normal(size=n)
        # convert to a probability vector (softmax, numerically stable)
        maxp = proposal.max()
        p = np.exp(proposal - maxp)
        p /= p.sum()

        sc = _objective_score(p)

        # Metropolis acceptance
        if sc < cur_score or rng.random() < np.exp(-(sc - cur_score) / temp):
            w = proposal
            cur_seq = p
            cur_score = sc
            if sc < best_score:
                best_score, best_seq = sc, p.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it:,} score={sc:.6f}")

        # occasional tiny re‑heat to avoid getting totally stuck
        if it % 10000 == 0 and rng.random() < 0.1:
            w = np.log(best_seq + eps) + sigma * rng.normal(size=n)
            cur_seq = best_seq.copy()
            cur_score = best_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises the evaluation function.  The routine runs a series of
    parametric sweeps, a large Dirichlet exploration, many short
    sub‑gradient runs from random restarts and a final simulated
    annealing phase.  The best sequence (normalised to sum = 1) is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweeps (≈200 s)
    # --------------------------------------------------------------
    phase1_deadline = deadline - 600.0  # reserve ample time for later phases
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase1_deadline)
    # extra families
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_edge_power(n, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_gaussian(n, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_beta(n, best_seq, best_score, phase1_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws (≈150 s)
    # --------------------------------------------------------------
    phase2_deadline = deadline - 450.0
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase2_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Multiple sub‑gradient restarts (≈250 s)
    # --------------------------------------------------------------
    phase3_deadline = deadline - 180.0
    rng = np.random.default_rng()
    restart_max_iters = 12000
    while time.time() < phase3_deadline:
        # give a tiny budget per restart so that many restarts are possible
        time_left = phase3_deadline - time.time()
        if time_left < 5.0:
            break
        # generate a diverse start
        a0 = _random_start(n, rng)
        best_seq, best_score = _subgradient_restart(a0, restart_max_iters,
                                                    phase3_deadline,
                                                    best_seq, best_score)

    # --------------------------------------------------------------
    # 4️⃣  Simulated annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Annealing finished, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
