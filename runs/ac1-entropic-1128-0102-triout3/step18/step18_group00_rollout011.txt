def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)
    return conv[:n + m - 1]
```

def _smooth_descent_segment(w0, seg_len, L_fft, deadline, start_time, rng):
    w = w0.copy()
    best_w = w.copy()
    best_score = float('inf')
    n = seg_len * w.shape[0]
    it = 0
    step0 = 0.2
    beta = 5.0
    # schedule boosting of beta
    while time.time() < deadline:
        it += 1
        # construct a from w
        a = np.repeat(w / seg_len, seg_len)
        # exact score evaluation
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_w = w.copy()
            print(f"[{time.time() - start_time:.2f}s] seg descent it={it} beta={beta:.2f} score={score:.6f}")
        # compute smooth max and gradient
        smooth_max, grad_a = _smooth_max_and_grad(a, beta, L_fft)
        # map to w gradient
        grad_w = grad_a.reshape((-1, seg_len)).sum(axis=1) / seg_len
        # step size schedule
        step = step0 / (it ** 0.55)
        # update
        w_candidate = w - step * grad_w
        w_candidate = _project_simplex(w_candidate)
        w = w_candidate
        # increase beta gradually

        # jitter occasionally
        if rng.random() < 0.01:
            jitter = 1e-4 * rng.random(w.shape[0]) * w
            w = np.maximum(w + jitter, 0.0)
            w = _project_simplex(w)
    return best_w, best_score

"""
Hybrid optimisation with extended parametric sweep and
smooth‐gradient descent on a low‑dimensional piecewise‑constant
representation.

The search proceeds in three phases (all respecting the overall
budget of 1000 s).

1️⃣  **Extended parametric sweep** – explores three families of
    symmetric shapes (sum, max and monotone) on a fine
    grid of decay λ and polynomial exponent p.  The best candidate
    (vector length n=4096) is kept.

2️⃣  **Low‑dimensional smooth descent** – the vector is forced to be
    piecewise constant on K=256 equal segments.  The segment weights
    (a probability vector w,  Σw=1) are optimised with a smoothed
    surrogate of the max‑convolution (soft‑max with temperature β).
    The true objective ``2·n·max(conv) / (∑a)²`` (scale‑invariant) is
    evaluated after each iteration; the best weights are remembered.
    The surrogate temperature β is annealed from 5 up to ~200,
    and Euclidean projection onto the simplex keeps w feasible.

3️⃣  **Short projected sub‑gradient refinement** – starting from the
    best full‑resolution vector obtained in phase 2, a few
    sub‑gradient steps (as in the original baseline) are performed
    to polish the solution.

The final sequence is normalised to sum = 1 and returned as a plain
list of floats.  All steps print timing and score information to aid
debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b via FFT."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)
    return conv[: n + m - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact fast objective (the same as evaluate_sequence for sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used in the final polishing phase)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Smoothed max‑convolution and its gradient (used on the segmented model)
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, beta: float, L_fft: int) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(conv(a,a)) via the
    soft‑max (temperature β) and return the value together with its
    gradient w.r.t. a.
    """
    conv = _conv_self_fft(a, L_fft)                 # length 2·n‑1
    max_conv = conv.max()
    # soft‑max weights (avoid overflow by subtracting the max)
    exp_vals = np.exp(beta * (conv - max_conv))
    prob = exp_vals / exp_vals.sum()                # length 2·n‑1, sums to 1

    # gradient w.r.t. a:
    #   ∂/∂a  Σ_i prob_i * (a*a)_i  =  2 * (a * prob_rev)  restricted to n entries
    prob_rev = prob[::-1]                           # reversed version
    conv_full = _conv_fft(a, prob_rev)              # length n + (2n‑1) – 1 = 3n‑2
    L = conv.shape[0]                               # = 2·n‑1
    grad_a = 2.0 * conv_full[L - 1 : L - 1 - a.shape[0] : -1]
    # smooth approximation of the max (expected value under prob)
    smooth_val = np.dot(prob, conv)
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Segment‑wise smooth descent
# ----------------------------------------------------------------------


def _smooth_descent_segment(
    w0: np.ndarray,
    seg_len: int,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Optimise a piecewise‑constant distribution.
    w stores the mass of each segment (Σ w = 1); the actual vector is
    a_i = w_{segment(i)} / seg_len.
    """
    w = w0.copy()
    best_w = w.copy()
    best_score = float("inf")
    n = w.shape[0] * seg_len
    it = 0
    step0 = 0.2
    beta = 5.0                     # start temperature for the smooth max
    while time.time() < deadline:
        it += 1
        # rebuild the full vector from w
        a = np.repeat(w / seg_len, seg_len)

        # exact objective (scale‑invariant, sum(a)=1)
        conv = _conv_self_fft(a, L_fft)
        max_conv = conv.max()
        score = float(2.0 * n * max_conv)   # denominator = 1
        if score < best_score:
            best_score = score
            best_w = w.copy()
            print(
                f"[{time.time() - start_time:.2f}s] seg‑descent it={it:,} "
                f"beta={beta:.2f} score={score:.6f}"
            )

        # smooth surrogate and its gradient
        smooth_val, grad_a = _smooth_max_and_grad(a, beta, L_fft)

        # aggregate the gradient onto the segment weights
        grad_w = grad_a.reshape((-1, seg_len)).sum(axis=1) / seg_len

        # decaying step size
        step = step0 / (it ** 0.55)

        w_candidate = w - step * grad_w
        w_candidate = _project_simplex(w_candidate)
        w = w_candidate

        # anneal β slowly
        if it % 1000 == 0:
            beta = min(beta * 1.1, 200.0)

        # occasional jitter to escape tiny plateaus
        if rng.random() < 0.01:
            jitter = 1e-4 * rng.random(w.shape[0]) * w
            w = np.maximum(w + jitter, 0.0)
            w = _project_simplex(w)

    return best_w, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        step = step0 / (it ** 0.55)

        improved = False
        for _ in range(8):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(
                    f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                    f"step={step:.5e} score={cand_score:.6f}"
                )
                break
            step *= 0.5
        if not improved:
            # allow occasional uphill move to get out of plateaus
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

            sym = left + left[::-1]
            sym /= sym.sum()
            sc_sym = _eval_fast(sym, L_fft)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep sum λ={lam:.6e} "
                    f"p={p:.2f} score={sc_sym:.6f}"
                )

            # ------- symmetric max shape (plateau) -------
            plat = np.maximum(left, left[::-1])
            plat /= plat.sum()
            sc_plat = _eval_fast(plat, L_fft)
            if sc_plat < best_score:
                best_score = sc_plat
                best_seq = plat.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep max λ={lam:.6e} "
                    f"p={p:.2f} score={sc_plat:.6f}"
                )

    # ------------------------------------------------------------------
    # 2️⃣  Low‑dimensional smooth descent on a piecewise‑constant model.
    # ------------------------------------------------------------------
    # Piecewise‑constant representation: K equal segments.
    K = 256
    seg_len = n // K
    # collapse the current best sequence onto segment masses
    w0 = best_seq.reshape(K, seg_len).sum(axis=1)
    w0 = _project_simplex(w0)  # guard against numerical drift

    seg_deadline = deadline - 80.0   # reserve ~80 s for the final polishing
    if time.time() < seg_deadline:
        best_w, best_score = _smooth_descent_segment(
            w0, seg_len, L_fft, seg_deadline, start_time, rng
        )
        best_seq = np.repeat(best_w / seg_len, seg_len)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq /= best_seq.sum()
        print(
            f"[{time.time() - start_time:.2f}s] after segment descent score≈{best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 3️⃣  Final short projected sub‑gradient refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline, start_time, best_seq, best_score, rng
        )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```

This implementation expands the original search space, introduces a smooth surrogate optimisation on a compact segment representation, and adds a final sub‑gradient polishing stage, all within the 1000 s time budget. The printed progress messages help monitor convergence toward a lower bound (target ≈ 1.503).
