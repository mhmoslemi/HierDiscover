def _smooth_gradient_descent(a0, L_corr, deadline, start_time, rng):
    a = a0.copy()
    n = a.shape[0]
    # Precompute parameter
    factor = 4.0 * n
    # Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # smoothing param alpha schedule
    alpha = 5.0
    alpha_max = 200.0
    # Determine number of iterations approx: (deadline - time.time())/approx_time_per_iter ~ maybe 500k
    # We'll loop until time.
    while time.time() < deadline:
        t += 1
        # compute FFT of a with L_corr
        A = np.fft.rfft(a, n=L_corr)
        # self-convolution: conv = irfft(A * np.conj(A), n=L_corr)[:2*n-1]
        conv = np.fft.irfft(A * np.conj(A), n=L_corr)[:2*n-1]
        # true score
        conv_max = conv.max()
        true_score = 2.0 * n * conv_max # sum =1
        # track best
        if true_score < best_score: update best
        # compute softmax
        # subtract max to avoid overflow
        exp_arg = alpha * (conv - conv_max)
        # maybe clip exp_arg to avoid overflow (if too large negative exponent)
        # compute exp safely: w = np.exp(exp_arg)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # compute correlation of a with w
        B = np.fft.rfft(w, n=L_corr)
        corr = np.fft.irfft(A * np.conj(B), n=L_corr)[: n + w.size - 1]
        conv_w_a = corr[w.size - 1 : w.size - 1 + n]  # length n
        grad = factor * conv_w_a
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # update a
        a = a - step
        # non-negative and renorm
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)
        # optionally enforce symmetry:
        # a = (a + a[::-1]) * 0.5

        # update alpha gradually
        if alpha < alpha_max:
            # increase by fraction each iteration or time. Use linear schedule with total it_target.
            alpha = min(alpha_max, alpha + (alpha_max - alpha) * 0.001) # adapt
        # occasional jitter
        if t % 5000 == 0:
            eps_noise = 1e-5 * rng.random(n) * a
            a = np.maximum(a + eps_noise, 0)
            a /= a.sum()
    return best_seq, best_score

def _adam_smooth_descent(a0, n, L, deadline, start_time, best_seq, best_score, rng):
    a = a0.copy()
    # parameters
    lr = 2e-4          # base learning rate for Adam
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    factor = 4.0 * n
    # smoothing alpha
    alpha = 2.0
    alpha_max = 150.0
    # schedule for alpha: geometric growth
    # precompute length of w
    w_len = 2 * n - 1
    # loop
    while time.time() < deadline:
        t += 1
        # FFT of a
        A = np.fft.rfft(a, n=L)
        # self-convolution
        conv_full = np.fft.irfft(A * np.conj(A), n=L)
        conv = conv_full[: w_len]   # length 2n-1
        conv_max = conv.max()
        # true objective score (sum is 1)
        true_score = 2.0 * n * conv_max
        if true_score < best_score:
            best_score = true_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam it={t} score={true_score:.6f} α={alpha:.2f}")

"""
Hybrid optimiser for the L∞‑convolution bound.

The routine consists of three stages:

1️⃣ Warm‑start (optional random sweep) – a quick random scan over the
   “geo‑power’’ family to obtain a decent initial vector.

2️⃣ Smooth Adam descent – the max‑convolution is approximated by a
   soft‑max (log‑sum‑exp) with a slowly increasing temperature α.
   The gradient of the smooth objective is computed by a single FFT
   pair (self‑convolution + correlation).  Adam updates with a
   projection onto the probability simplex drive the vector towards a
   lower true max‑convolution.

3️⃣ Final simulated‑annealing refinement – tiny mass‑transfer moves
   (src → dst) are accepted with the usual Metropolis rule while the
   temperature is cooled linearly.  The best vector seen throughout is
   returned.

All work is done inside the 1000 s budget; the code falls back to the
current best if time runs out.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a))/(sum(a)²) – assumes a is 1‑D."""
    n = a.shape[0]
    A = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(A * np.conj(A), n=L_fft)
    conv = conv_full[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

    Returns a plain Python list of floats (L1‑normalised).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_conv = _next_pow2(2 * n - 1)                     # for self‑convolution
    L_corr = _next_pow2(3 * n - 2)                     # for correlation with the soft‑max weights

    # ------------------------------------------------------------------
    # Warm‑start: try to reuse a global construction if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # Resize by linear interpolation if the length differs.
            if starter.size != n:
                x_old = np.linspace(0, 1, starter.size)
                x_new = np.linspace(0, 1, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter /= s
            else:

            a0 = starter.copy()
        else:

    except Exception:

# ----------------------------------------------------------------------
# 2️⃣ Smooth Adam descent
# ----------------------------------------------------------------------


def _adam_smooth_descent(
    a0: np.ndarray,
    n: int,
    L: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
):
    """
    Adam optimisation on a soft‑max approximation of max(conv).

    The smoothing parameter α starts small and is slowly increased.
    After each update the vector is projected onto the probability simplex.
    """
    a = a0.copy()
    # Adam hyper‑parameters
    lr = 2e-4            # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    factor = 4.0 * n                     # gradient multiplier for the true objective
    w_len = 2 * n - 1                    # length of the self‑convolution vector
    offset = w_len - 1                    # index where correlation slice starts

    # smoothing schedule – start with a gentle soft‑max and cool down
    alpha = 2.0
    alpha_max = 150.0

    t = 0
    while time.time() < deadline:
        t += 1

        # ------------------- FFT of the current vector -----------------
        A = np.fft.rfft(a, n=L)

        # ------------------- self‑convolution (full) -------------------
        conv_full = np.fft.irfft(A * np.conj(A), n=L)
        conv = conv_full[:w_len]                      # length 2·n‑1
        conv_max = conv.max()

        # true objective (sum(a)=1)
        true_score = 2.0 * n * conv_max
        if true_score < best_score:
            best_score = true_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={t:,} score={true_score:.6f} α={alpha:.2f}"
            )

        # ------------------- soft‑max weights -------------------------
        # w_i ∝ exp(α·conv_i); we shift by conv_max for numerical safety.
        exp_arg = alpha * (conv - conv_max)          # max exp_arg = 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w_len)
        else:
            w /= w_sum

        # ------------------- correlation a * w ------------------------
        B = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(A * np.conj(B), n=L)
        # slice that corresponds to Σ_k a_k·w_{i+k}
        conv_w_a = corr_full[offset : offset + n]    # length n

        # ------------------- gradient of the *smooth* objective ------
        # ∇ smooth = 2·Σ_k w_k·a_{k‑i}  ⇒  grad_raw = 2·conv_w_a
        # True objective gradient = 2·n·∇ smooth = factor·conv_w_a
        grad = factor * conv_w_a

        # ------------------- Adam update -------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)                     # keep a ≥0 and Σa=1

        # ------------------- increase smoothing ------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.003)

        # ------------------- occasional jitter -------------------------
        if t % 5000 == 0:
            jitter = 1e-5 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    n: int,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer hill‑climb with linear cooling."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score

    T0 = 1.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * 5e-3            # up to 0.5 % move
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        # occasional random jitter
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = cur / cur.sum()
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score
