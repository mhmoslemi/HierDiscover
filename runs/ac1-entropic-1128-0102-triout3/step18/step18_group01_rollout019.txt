import time
import numpy as np

# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    u_max = np.max(u)
    e = np.exp(u - u_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT of length L (L ≥ 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _weighted_conv_grad(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Compute  Σ_k w_k * (2 * a_{k‑i})  for all i, i.e. the gradient of
    Σ_k w_k * conv_k w.r.t. a (up to the factor 2).

    Result = 2 * Σ_k w_k * a_{k‑i}  (i = 0..n‑1)
    """
    # reverse a to obtain the correct cross‑correlation
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=L)
    W = np.fft.rfft(w, n=L)
    conv = np.fft.irfft(A_rev * W, n=L)[: 3 * a.shape[0] - 2]
    # slice that corresponds to indices i = 0..n‑1
    # conv index (n‑1) contains the sum for i=0
    start = a.shape[0] - 1
    end = start + a.shape[0]
    weighted = conv[start:end]          # Σ_k w_k * a_{k‑i}
    return weighted


def _eval_exact(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a))  (a sums to 1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) with a
    low value of the evaluation function.  Returns the vector as a Python
    list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.07     # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 provides a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)      # for the exact convolution
    L_grad = _next_pow2(3 * n - 2)     # for the weighted gradient convolution

    # ------------------------------------------------------------------
    # Starting point – use the global starter if it exists, otherwise uniform.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined globally by the evaluation harness
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = None
            else:
                starter = starter / starter.sum()
    except Exception:
        starter = None

    if starter is not None and starter.size == n:
        a0 = starter.copy()
        print(f"[{time.time() - start_time:.2f}s] Using global starter.")
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print(f"[{time.time() - start_time:.2f}s] Using uniform start (n={n}).")

    # logits for the soft‑max parametrisation
    u = np.log(a0 + 1e-32)                     # avoid log(0)
    # Adam optimiser state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2                                   # learning rate (tuned empirically)

    # smooth‑max temperature (β).  Start small, anneal up.
    beta_smooth = 1.0
    beta_target = 500.0                       # final sharpness
    beta_inc = (beta_target / beta_smooth) ** (1.0 / 1500)   # grow per iteration

    # track the best exact solution found so far
    best_a = a0.copy()
    best_score = _eval_exact(best_a, L_fft)

    it = 0
    last_report = start_time

    while time.time() < deadline:
        it += 1

        # --- probability vector -------------------------------------------------
        a = _softmax(u)                     # guarantees non‑negative, sum = 1

        # --- exact convolution (for occasional true‑score evaluation) ----------
        conv = _conv_self_fft(a, L_fft)
        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] it={it:,} "
                  f"exact score={cur_score:.6f}")

        # --- smooth‑max weights -------------------------------------------------
        # w_k ∝ exp(β·conv_k)  (numerically stable)
        w = np.exp(beta_smooth * (conv - max_conv))
        w /= w.sum()                        # ∑_k w_k = 1

        # --- gradient of the smooth objective w.r.t. a -------------------------
        # ∂f/∂a = 4·n· Σ_k w_k·a_{k‑i}    (see analysis in the doc‑string)
        weighted_sum = _weighted_conv_grad(a, w, L_grad)
        grad_a = 4.0 * n * weighted_sum

        # --- map gradient onto the log‑it space (soft‑max Jacobian) ------------
        grad_u = a * (grad_a - np.dot(a, grad_a))

        # --- Adam update -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- anneal the smoothness parameter -----------------------------------
        if it % 100 == 0 and beta_smooth < beta_target:
            beta_smooth = min(beta_target, beta_smooth * beta_inc)

        # --- occasional random jitter to help escape plateaus ---------------
        if it % 500 == 0:
            jitter = rng.normal(scale=1e-5, size=n)
            u += jitter

        # --- periodic verbose output -----------------------------------------
        if time.time() - last_report > 30.0:
            last_report = time.time()
            print(f"[{last_report - start_time:.2f}s] it={it:,} "
                  f"β={beta_smooth:.1f}  cur_score={cur_score:.6f}  "
                  f"best_score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final normalisation (sum → 1) and return as a plain list.
    # ----------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0.0:
        best_a = best_a / s
    else:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # Report the final bound using evaluate_sequence (optional)
    try:
        from __main__ import evaluate_sequence  # may already be imported
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_a]
