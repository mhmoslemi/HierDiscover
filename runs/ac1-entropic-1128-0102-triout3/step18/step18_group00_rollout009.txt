"""
Hybrid optimiser for the L∞‑convolution bound.

The search proceeds in four stages:

1️⃣  **Parametric sweep** over two simple families (monotone “geo‑power” and
    symmetric “U‑shape”).  The best candidate from ≈30 k points is kept.

2️⃣  **Smooth‑max gradient descent**.
    Instead of the pure sub‑gradient of the max, we minimise a
    log‑sum‑exp surrogate of the self‑convolution:
        φβ(a)=β⁻¹ log Σ exp(β·(a*a)) .
    Its gradient is a weighted sum of the per‑shift gradients,
    which is obtained cheaply with FFTs.  We start with a small β
    (very smooth) and increase it exponentially; a decaying step‑size
    together with Euclidean projection onto the simplex produces
    monotone improvement.

3️⃣  **Projected sub‑gradient descent** (as in the reference code) –
    a final “hard‑max” polishing phase.

4️⃣  **Simulated‑annealing hill‑climb** for tiny mass‑transfer moves and
    occasional jitter to escape any remaining plateaus.

All phases respect a total wall‑time budget of 1000 s.  The final
vector is returned as a plain Python list of non‑negative floats whose
sum is (numerically) 1.0.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection, …
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a))/(sum(a)²) using FFT."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families for the initial sweep
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(‑lam·i)  (L1‑normalised)."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape” a_i ∝ (i+1)^p·exp(‑lam·i) + (n‑i)^p·exp(‑lam·(n‑1‑i))."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    right = left[::-1]
    shape = left + right
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of the (hard) max‑convolution
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Stage 2 – smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def _smooth_max_descent(
    a0: np.ndarray,
    L_self: int,
    L_full: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate of max(conv) with projected
    gradient descent.  β (smoothness) starts small and grows
    exponentially; step size decays as 1/√t and is normalised by
    ‖∇‖.
    """
    a = a0.copy()
    n = a.shape[0]
    indices = 2 * n - 2 - np.arange(n)      # used to pick the right slice
    beta = 0.5                               # initial smoothness
    step0 = 0.5
    it = 0

    # current score (sum of a is 1 after projection)
    cur_conv = np.fft.irfft(np.fft.rfft(a, n=L_full) ** 2, n=L_full)[: 2 * n - 1]
    cur_score = 2.0 * n * cur_conv.max()
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    while time.time() < deadline:
        it += 1

        # ----- FFT of the current a (re‑used for both convolutions) -----
        A = np.fft.rfft(a, n=L_full)

        # ----- self‑convolution (needed for the objective) -----
        conv_self = np.fft.irfft(A * A, n=L_full)[: 2 * n - 1]
        max_conv = conv_self.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth‑grad it={it:,} β={beta:.3f} score={cur_score:.6f}"
            )

        # ----- soft‑max weights over the 2n‑1 convolution entries -----
        expo = beta * conv_self
        expo_max = expo.max()
        w = np.exp(expo - expo_max)          # shift for stability
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        w_rev = w[::-1]                       # reverse to obtain the proper cross‑terms
        B = np.fft.rfft(w_rev, n=L_full)
        conv_grad_full = np.fft.irfft(A * B, n=L_full)[: 3 * n - 2]
        g = 2.0 * conv_grad_full[indices]    # gradient vector (length n)

        # ----- step size (scaled by ‖∇‖) and back‑tracking line search -----
        norm_g = np.linalg.norm(g) + 1e-12
        step = step0 / (np.sqrt(it) * norm_g)

        improved = False
        step_tmp = step
        for _ in range(6):
            cand = a - step_tmp * g
            cand = _project_simplex(cand)     # stays on the simplex
            # fast objective (sum(cand)≈1)
            A_cand = np.fft.rfft(cand, n=L_full)
            conv_cand = np.fft.irfft(A_cand * A_cand, n=L_full)[: 2 * n - 1]
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                break
            step_tmp *= 0.5

        # if no improvement, keep a unchanged (the jitter below may move it)

        # ----- increase smoothing parameter β slowly -----
        if it % 200 == 0:
            beta = min(beta * 1.07, 2000.0)

        # ----- occasional random jitter to avoid stagnation -----
        if rng.random() < 0.001:
            eps = rng.random(n) * a * 1e-4
            cand = np.maximum(a + eps, 0.0)
            cand = _project_simplex(cand)
            A_cand = np.fft.rfft(cand, n=L_full)
            conv_cand = np.fft.irfft(A_cand * A_cand, n=L_full)[: 2 * n - 1]
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – projected sub‑gradient descent (hard max)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    L_self: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = np.fft.irfft(np.fft.rfft(a, n=L_self) ** 2, n=L_self)[: 2 * n - 1]
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)

        improved = False
        step_tmp = step
        for _ in range(8):
            cand = a - step_tmp * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_self)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(
                    f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step_tmp:.5e} score={cand_score:.6f}"
                )
                break
            step_tmp *= 0.5

        if not improved:
            # tiny uphill move with a decaying probability
            prob = np.exp(-max(0, cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 4 – simulated‑annealing final hill‑climb
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_self: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with a linearly decreasing temperature."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]
    T0 = 1.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * 5e-3
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_self)
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = _project_simplex(cur)
            cur_score = _eval_fast(cur, L_self)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score

    try:

    # Normalise (sum → 1) and return a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
