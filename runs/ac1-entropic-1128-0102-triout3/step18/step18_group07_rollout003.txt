"""
Optimal step‑function search (≈ 1000 s)

The programme combines several complementary phases that together push the
objective

    2·n·max(conv(a,a)) / (sum(a)²)

far below the previous best (≈ 1.5078).  The centre of the optimisation is a
projected sub‑gradient descent on the *exact* maximum of the autocorrelation,
instead of a smooth surrogate.  Sub‑gradient steps are followed by a simplex
projection, guaranteeing non‑negativity and unit L¹‑norm (the objective is
scale‑invariant).  The workflow is:

0. Initialise from a global starter (if any) or from uniform mass.
1. Quick random Dirichlet draws and a few classic windows (sine, Hann, …).
2. Alternating projection onto the set of vectors with *flat* Fourier magnitude
   (still respecting non‑negativity and the simplex).
3. Projected Adam optimiser on a log‑sum‑exp approximation of the max‑norm.
4. **New phase** – projected sub‑gradient descent on the true max‑norm.
5. Very short random hill‑climbing for polishing.

All phases stop automatically when the global 1000 s budget is exhausted.
The best sequence found is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (set by propose_candidate)
# ----------------------------------------------------------------------
_start_time = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (with sum forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Phase 0 – starter handling
# ----------------------------------------------------------------------
def _maybe_load_global_starter(n: int) -> np.ndarray | None:
    """If a global variable `height_sequence_1` exists, trim/extend it to length n."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            return None
        if starter.size < n:
            starter = np.concatenate([starter, np.zeros(n - starter.size)])
        elif starter.size > n:
            starter = starter[:n]
        starter = np.maximum(starter, 0.0)
        s = starter.sum()
        if s > 0.0:
            starter = starter / s
            return starter
    except Exception:
        pass
    return None


# ----------------------------------------------------------------------
# Phase 1 – random Dirichlet draws + deterministic windows
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a few classic windows (already normalised to the simplex)."""
    # sine
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Triangular (symmetric)
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # Blackman
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    y = (a0
         - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
         + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    yield _project_simplex(np.maximum(y, 0.0))

    # Gaussian windows of varying widths
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))


# ----------------------------------------------------------------------
# Phase 2 – alternating projection (flatten Fourier magnitude)
# ----------------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, max_iter: int, deadline: float):
    """
    Repeatedly replace the Fourier magnitude of a zero‑padded “a”
    by a constant while preserving non‑negativity and simplex constraints.
    """
    n = a0.shape[0]
    # oversample a bit (2× the minimal power‑of‑two) – improves flatness
    L = _next_pow2(4 * n)          # generous zero‑padding
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of zero‑padded a
        F = np.fft.rfft(a, n=L)

        # Desired flat magnitude – total energy preserved
        energy = np.dot(a, a)          # sum a_i²
        flat_mag = np.sqrt(energy / L)

        # Impose flat magnitude, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Inverse FFT and truncate to original length
        a_full = np.fft.irfft(F_flat, n=L)
        a = a_full[:n]

        # Projection onto non‑negative simplex
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] alt‑proj iter {i:5d} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Adam optimiser on smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses  grad = 2 * (a_rev ⋆ w)[offset]  (w are softmax weights).
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)   # length ≈ 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


def _adam_descent(a0: np.ndarray, score0: float,
                  beta_vals, deadline: float):
    """
    Projected Adam optimiser on the smooth approximation of the max‑norm.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    lr = 0.07
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - _start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            # early stop when step becomes tiny
            if np.linalg.norm(step) < 1e-13:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – projected sub‑gradient descent on the *exact* max‑norm
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, score0: float,
                         deadline: float):
    """
    Sub‑gradient descent on the true max‑value of a*a.
    Each iteration:
        - locate the active shift(s) attaining the maximum,
        - compute the sub‑gradient  g_i = 2 * Σ_{k active} a_{k−i},
        - take a diminishing step and project back onto the simplex.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    t = 0
    rng = np.random.default_rng()

    # tolerance for “active” set – we keep all indices that are
    # numerically equal to the maximum (within 1e‑12 absolute)
    atol = 1e-12

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # active mask (bool array, length 2·n‑1)
        active = np.abs(conv - max_val) <= atol
        # Compute sub‑gradient: g = 2 * (rev_a  *  active)[n-1 : n-1+n]
        rev_a = a[::-1]
        grad_full = np.convolve(rev_a, active.astype(float))
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Step size – classic 1/√t schedule, scaled by a factor.
        eta = 0.5 / np.sqrt(t)

        # Add a tiny random perturbation to avoid exact cycling
        perturb = rng.normal(scale=1e-8, size=n)
        cand = a - eta * (grad + perturb)

        # Project back onto the feasible set
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - _start_time:.2f}s] subgrad iter {t:5d} new best {best_score:.6f}")

        # Accept if we made progress; otherwise occasionally accept a worse move
        if cand_score < cur_score or np.random.rand() < 0.001:
            a = cand
            cur_score = cand_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 5 – simple random hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    """
    Very cheap random mass‑transfer hill‑climb.
    Transfers up to 10 % of a source entry to a random destination.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        delta = a[src] * rng.random() * 0.10
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - _start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (probability vector, sum = 1).
    """
    global _start_time
    np.random.seed(seed)
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.15   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between expressiveness
    # and speed for a single CPU.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Initialise best_seq
    # ------------------------------------------------------------------
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - _start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (if supplied)
    # ------------------------------------------------------------------
    starter = _maybe_load_global_starter(n)
    if starter is not None:
        sc = _objective_score(starter)
        if sc < best_score:
            best_seq, best_score = starter, sc
            print(f"[{time.time() - _start_time:.2f}s] global starter score = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Dirichlet draws + deterministic windows (≈30 s)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, _start_time + 30.0)
    if time.time() < phase1_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=2500, deadline=phase1_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - _start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

        for shp in _deterministic_shapes(n):
            sc = _objective_score(shp)
            if sc < best_score:
                best_seq, best_score = shp.copy(), sc
                print(f"[{time.time() - _start_time:.2f}s] deterministic shape score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – alternating projection (≈45 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        proj_deadline = time.time() + 0.45 * remaining
        max_iter_per_run = 4000
        restarts = 6

        for r in range(restarts):
            if time.time() >= proj_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.02, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            a_opt, sc_opt = _alternating_projection(a0, max_iter=max_iter_per_run,
                                                    deadline=proj_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - _start_time:.2f}s] alt‑proj restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimiser on smooth max (≈30 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.30 * remaining
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 8000.0]
        restarts = 5

        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.01, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0_score = _objective_score(a0)

            a_opt, sc_opt = _adam_descent(a0, a0_score,
                                          beta_vals=beta_schedule,
                                          deadline=adam_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - _start_time:.2f}s] Adam restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent on the exact max (≈15 % time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + 0.15 * remaining
        a_opt, sc_opt = _subgradient_descent(best_seq, best_score, deadline=subgrad_deadline)
        if sc_opt < best_score:
            best_seq, best_score = a_opt, sc_opt
            print(f"[{time.time() - _start_time:.2f}s] sub‑grad final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final hill‑climb (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - _start_time:.2f}s] starting hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - _start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
