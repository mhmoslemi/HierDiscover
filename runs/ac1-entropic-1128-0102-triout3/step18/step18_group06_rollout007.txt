def _random_local_search(a, best_seq, best_score, deadline):
    a_current = a.copy()
    n = len(a)
    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        # choose i,j distinct
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # choose delta: some random fraction of a_i
        # ensure a[i] > epsilon
        if a_current[i] <= 1e-12:
            continue
        delta_fraction = rng.uniform(0.001, 0.1)  # at most 10% of mass at i
        delta = delta_fraction * a_current[i]

```
    max_beta = 30.0

    while time.time() < deadline:
        it += 1
        beta = min(max_beta, 1.0 + (it / 50000.0)*(max_beta-1.0))
        conv = _conv_self(a)
        max_val = conv.max()
        # compute smoothed weights
        exp_term = np.exp(beta * (conv - max_val))
        w = exp_term / exp_term.sum()
        # gradient

        # step size
        eta = eta0 / np.sqrt(it)
        # multiplicative update
        a = a * np.exp(-eta * grad)
```

assistantfinalBelow is a self‑contained implementation of `propose_candidate`.  
It extends the previous search pipeline with two extra families of
parametric shapes, a *smooth* (log‑sum‑exp) gradient descent that can
break the flat plateaus of the pure sub‑gradient method, and a short
random local‑search phase.  All phases respect the 1000 s time‑budget
and keep the sequence on the probability simplex.

```python
"""
Propose a non‑negative step‑function (list of floats) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The search consists of:

1️⃣  Fast parametric sweeps (geometric, power, exponential, two‑block,
    three‑block, triangular) to obtain strong starting points.

2️⃣  Random Dirichlet draws – cheap global exploration of the simplex.

3️⃣  Two complementary descent phases:
    • Sub‑gradient mirror descent (as before) for a quick coarse improvement.
    • Smooth‑gradient mirror descent (log‑sum‑exp smoothing) that can
      escape the flat plateaus of the pure sub‑gradient method.
      The smoothing parameter β is increased gradually (continuation).

4️⃣  A brief random local‑search (mass transfer between two coordinates)
    that can capture tiny remaining improvements.

The routine always respects the 1000 s CPU budget and returns the best
sequence found (normalised to sum = 1).
"""

import time
import numpy as np
from scipy.signal import correlate

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (evaluate_sequence multiplies by 1/(sum(seq)**2) but the expression
    is homogeneous, so fixing sum=1 is sufficient.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

def _subgradient_descent(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float):
    """
    Projected mirror‑descent using the exact sub‑gradient of max(conv(a,a)).
    Step size η_k = η0 / sqrt(k).
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5  # initial step‑size
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={it:,} obj={obj:.6f} η={eta0/np.sqrt(it):.2e}")
        # active constraints (allow tiny tolerance)
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]
        m = len(active)
        if m == 0:
            break
        grad = np.zeros_like(a)
        for i in active:
            lo = max(0, i - (n - 1))
            hi = min(i, n - 1)
            j = np.arange(lo, hi + 1)
            k = i - j
            grad[lo:hi + 1] += a[k]
        grad = 2.0 * grad / m
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣‑B Smooth (log‑sum‑exp) gradient descent (continuation)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             best_seq: np.ndarray, best_score: float):
    """
    Mirror‑descent using a smooth approximation of max(conv):
        max_β(x) = (1/β) log sum_i exp(β x_i)
    β is increased gradually (continuation) to escape plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.3          # initial step‑size
    max_beta = 30.0     # final smoothing strength
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        # continuation schedule for β
        beta = 1.0 + (it / 50000.0) * (max_beta - 1.0)
        if beta > max_beta:
            beta = max_beta
        conv = _conv_self(a)
        max_val = conv.max()
        # smooth weights (log‑sum‑exp trick)
        exp_term = np.exp(beta * (conv - max_val))
        w = exp_term / exp_term.sum()
        # gradient of the smooth objective (without the constant 2n)
        grad = 2.0 * correlate(w, a, mode='valid')
        eta = eta0 / np.sqrt(it)
        # multiplicative update on the simplex (mirror descent)
        a = a * np.exp(-eta * grad)

        # occasional jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = (rng.random(n) - 0.5) * 0.01
            a = a * (1.0 + jitter)
            a = np.maximum(a, 0.0)

        total = a.sum()
        if total <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        sc = 2.0 * n * max_val  # exact objective (sum(a)=1)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGrad it={it:,} β={beta:.2f} score={sc:.6f} η={eta:.2e}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣ Local random mass‑transfer hill‑climbing
# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float):
    """
    Tiny random moves: transfer a small fraction of mass from one coordinate
    to another.  Accept only improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # move at most 5 % of the source mass
        delta_frac = rng.uniform(0.001, 0.05)
        delta = delta_frac * a[i]
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:                # numerical safety
            continue
        conv = _conv_self(new_a)
        sc = 2.0 * n * conv.max()
        if sc < best_score:
            best_score, best_seq = sc, new_a.copy()
            a = new_a
            print(f"[{time.time() - start_time:.2f}s] LocalSearch it={it} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine runs a quick parametric sweep,
    a large random Dirichlet exploration, a deterministic sub‑gradient
    descent, a smooth‑gradient continuation phase and a final local
    hill‑climbing refinement.  The best sequence (normalised to sum = 1)
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweeps (~450 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0   # keep ~300 s for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_three_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_triangle(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0   # reserve time for descent phases
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Exact sub‑gradient mirror descent (≈100 s)
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        subgrad_deadline = deadline - 60.0
        best_seq, best_score = _subgradient_descent(best_seq, subgrad_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End sub‑gradient descent, best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣‑B Smooth‑gradient continuation (≈40 s)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑gradient descent")
        smooth_deadline = deadline - 10.0
        best_seq, best_score = _smooth_gradient_descent(best_seq, smooth_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End smooth‑gradient, best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Local random hill‑climbing (final few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting local hill‑climb")
        best_seq, best_score = _local_hill_climb(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End local search, best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
```
