"""
Propose a non‑negative step‑function (probability vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm (≈1000 s total)

1️⃣  Quick parametric sweep over several analytic families
    (geometric×power, Gaussian, Hann window) and a few random
    “spiky” Dirichlet‑like vectors → obtain a decent starting shape.

2️⃣  Continuation + Adam optimisation on a smooth approximation of the
    max‑convolution (log‑sum‑exp) – the “β‑homotopy”.  Each iteration
    computes the exact self‑convolution (via FFT), forms softmax
    weights over the shift axis, obtains the gradient by a second FFT,
    and updates the vector with Adam while projecting onto the simplex.
    β is increased from a small value (≈5) to a large one (≈250) so the
    optimisation first sees a smooth landscape and finally the true
    L∞ objective.

3️⃣  Final random hill‑climb (tiny mass transfers) to polish the result.

All steps respect the 1000 s budget, print progress and finally return
the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via real FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Exact objective (fast, assumes a is already a probability vector)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p·exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_hann(n: int) -> np.ndarray:
    """a_i ∝ sin²(π·i/(n‑1)) (the Hann window), normalised."""
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * i / (n - 1)) ** 2
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max + gradient via FFT
# ----------------------------------------------------------------------


def _smooth_max_and_grad(
    a: np.ndarray, beta: float, L_self: int, L_grad: int
) -> tuple[float, np.ndarray]:
    """
    Compute
        max_conv   = max(conv(a,a))
        grad       = ∇_a  ( 1/β log Σ_k exp(β·conv_k) )
    using FFTs.
    """
    n = a.shape[0]

    # ---- self‑convolution (full length 2n‑1) -------------------------
    Af = np.fft.rfft(a, n=L_self)
    conv_full = np.fft.irfft(Af * Af, n=L_self)[: 2 * n - 1]

    # ---- smooth max (log‑sum‑exp) -----------------------------------
    max_c = conv_full.max()
    # shift by max to avoid overflow
    exps = np.exp(beta * (conv_full - max_c))
    w = exps / exps.sum()                     # soft‑max weights, Σ w = 1

    # ---- gradient of the smooth max ---------------------------------
    # g = Σ_k w_k * ∂ conv_k / ∂a ; ∂ conv_k / ∂a_i = 2·a_{k‑i}
    rev_a = a[::-1]                          # reversed a
    # convolution w * rev_a  (length 3n‑2) via FFT
    Wf = np.fft.rfft(w, n=L_grad)
    RevAf = np.fft.rfft(rev_a, n=L_grad)
    conv_wr_full = np.fft.irfft(Wf * RevAf, n=L_grad)
    start = n - 1
    g = 2.0 * conv_wr_full[start : start + n]  # slice length n

    # the exact max‑convolution (needed for the real objective)
    max_conv = conv_full.max()
    return max_conv, g


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max
# ----------------------------------------------------------------------


def _adam_opt(
    a0: np.ndarray,
    L_self: int,
    L_grad: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max objective, with a β‑homotopy (β grows from
    a small to a large value).  The iterate is kept on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # β schedule
    beta_start = 5.0
    beta_end = 250.0

    best = a.copy()
    best_score = _eval_fast(best, L_self)

    it = 0
    while time.time() < deadline:
        it += 1

        # linearly increase β according to elapsed fraction
        progress = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + (beta_end - beta_start) * progress
        beta = max(beta_start, min(beta, beta_end))

        # smooth‑max value and gradient
        max_conv, g = _smooth_max_and_grad(a, beta, L_self, L_grad)

        # Adam update (step size tuned for n≈3·10⁴)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.15 / (np.sqrt(v_hat) + eps)   # base learning rate

        a = a - step * m_hat
        a = _project_simplex(a)               # stay on probability simplex

        # occasional exact evaluation (every ~50 iterations)
        if it % 50 == 0:
            sc = _eval_fast(a, L_self)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Adam it={it:,} β={beta:.1f} score={sc:.6f}"
                )

        # small random jitter to keep escaping plateaus
        if it % 1200 == 0:
            eps_j = rng.random(n) * a * 1e-4
            a = np.maximum(a + eps_j, 0.0)
            a = a / a.sum()

        # restart‑like jitter if stuck for a while
        if it % 4000 == 0:
            eps_j = rng.random(n) * a * 5e-4
            a = np.maximum(best + eps_j, 0.0)
            a = a / a.sum()

    return best, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def _hill_climb(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb: transfer a tiny fraction of one coordinate
    to another and accept only if the exact objective improves.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand /= cand.sum()

        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={sc:.6f}")

        # occasional larger exploratory jitter
        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={sc:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises 2·n·max(conv(a,a))/(sum(a)²).
    Returns the best vector found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # safety margin

    # --------------------------------------------------------------
    # Problem size – a compromise between quality and speed.
    # --------------------------------------------------------------
    n = 32768
    L_self = _next_pow2(2 * n - 1)          # for a*a
    L_grad = _next_pow2(3 * n - 2)          # for w * rev(a)
    L_fft = L_self                         # used for exact evaluations

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter exists, try it.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep over a few analytic families (≈70 s)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    i_arr = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_arr + 1.0)

    # geometric × power family
    lam_grid = np.linspace(1e-06, 3e-03, 260)          # 260 values
    p_grid = np.linspace(-9.0, 6.0, 151)               # 151 values
    stop_outer = False
    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            stop_outer = True
            break
        base = -lam * i_arr
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} p={p:.2f} score={sc:.6f}")

    if not stop_outer:
        # pure Gaussian family
        c_grid = np.logspace(-9.0, -5.0, 36)
        for c in c_grid:
            if time.time() >= sweep_deadline:
                break
            log_shape = -c * (i_arr * i_arr)
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] gaussian c={c:.2e} score={sc:.6f}")

        # Hann window
        if time.time() < sweep_deadline:
            cand = _make_hann(n)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Hann window score={sc:.6f}")

        # Random “spiky’’ vectors (few non‑zero entries)
        for _ in range(250):
            if time.time() >= sweep_deadline:
                break
            k = rng.integers(2, 12)          # number of spikes
            idx = rng.choice(n, size=k, replace=False)
            vals = rng.random(k) + 1e-6
            cand = np.zeros(n, dtype=np.float64)
            cand[idx] = vals
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] spiky k={k} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam optimisation on a smooth max‑convolution (≈60 % of budget)
    # --------------------------------------------------------------
    adam_deadline = start_time + 0.60 * budget_s
    if time.time() < adam_deadline:
        best_seq, best_score = _adam_opt(
            best_seq, L_self, L_grad, adam_deadline, start_time, rng
        )
        print(f"[{time.time() - start_time:.2f}s] Adam finished best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_fft, deadline, start_time, rng)
        print(f"[{time.time() - start_time:.2f}s] Hill‑climb finished best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (guard against tiny negative rounding) and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Report the final bound with the provided evaluate_sequence (optional)
    # ------------------------------------------------------------------
    try:
        from __main__ import evaluate_sequence  # might already be imported
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
