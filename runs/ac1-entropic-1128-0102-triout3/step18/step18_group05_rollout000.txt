def propose_candidate(seed: int = 0, budget_s: int = 1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.05

    rng = np.random.default_rng(seed)

    # Set n
    n = 4096

    # Load starter
    try:
        f0 = np.asarray(height_sequence_1, dtype=float)
        if f0.ndim != 1 or f0.size == 0:
            raise ValueError
        if f0.size != n:
            # if size different, adjust by padding/truncating
            if f0.size < n:
                f = np.concatenate([f0, np.full(n-f0.size, 0.0)])
            else:
                f = f0[:n]
        else:
            f = f0.copy()
    except Exception:
        f = np.full(n, 1.0/n)

    f = np.maximum(f, 0.0)
    total = f.sum()
    if total > 0.0:
        f = f / total
    else:
        f = np.full(n, 1.0/n)

    cur_f = f.copy()
    cur_score = _objective_score(cur_f)
    best_f = cur_f.copy()
    best_score = cur_score

    print(f"[0.00s] initial score = {cur_score:.6f}")

    # Simulated annealing
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1

        # Temperature schedule based on iteration fraction
        # We compute fraction of time elapsed relative to total time budget
        now = time.time()
        elapsed = now - start_time
        frac = elapsed / budget_s
        # schedule: exponential decay
        T0 = 0.005
        T_end = 1e-7
        T = T0 * (T_end / T0) ** frac

        # max move fraction decreases from 0.2 to 0.001
        max_move_frac = 0.2 * (1 - frac) + 0.001

        src = rng.integers(n)
        if cur_f[src] <= 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # choose delta
        delta = cur_f[src] * rng.random() * max_move_frac
        cand_f = cur_f.copy()
        cand_f[src] -= delta
        cand_f[dst] += delta

        cand_score = _objective_score(cand_f)

        # acceptance
        if cand_score < cur_score:
            accept = True
        else:
            # compute acceptance probability
            diff = cand_score - cur_score
            # avoid overflow
            if diff / T > 50:
                accept = False
            else:
                accept = rng.random() < math.exp(-diff / T)

        if accept:
            cur_f = cand_f
            cur_score = cand_score

            if cur_score < best_score - 1e-12:
                best_f = cur_f.copy()
                best_score = cur_score
                print(f"[{elapsed:.2f}s] new best = {best_score:.6f} (iter {iter_count})")

        # occasional global random restart from best or uniform
        if rng.random() < 0.0005:
            # random perturb full vector (Dirichlet)
            new_f = rng.dirichlet(np.ones(n))
            cur_f = new_f
            cur_score = _objective_score(cur_f)
            # keep best
            if cur_score < best_score:
                best_f = cur_f.copy()
                best_score = cur_score
                print(f"[{elapsed:.2f}s] restart gave new best {best_score:.6f}")

    # final polishing: do small local transfers (like earlier)
    # we can loop a few passes of deterministic small moves improving
    # Use greedy local improvements: for each pair (i,j), try moving small mass if improves.
    # But that is O(n^2); too heavy. Instead use again random moves but only accept improvements, until time runs out.
    final_deadline = deadline
    while time.time() < final_deadline:
        src = rng.integers(n)
        if best_f[src] <= 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = best_f[src] * rng.random() * 0.001  # tiny
        cand = best_f.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_f = cand
            best_score = cand_score
            now = time.time()
            print(f"[{now - start_time:.2f}s] polishing improved to {best_score:.6f}")

    # ensure non-negative and renorm
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0.0:
        best_f = best_f / total
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    print(f"[{time.time() - start_time:.2f}s] finished best objective ≈ {best_score:.6f}")

    return [float(v) for v in best_f.tolist()]

"""
Simulated annealing for the step‑function bound.

We search for a non‑negative vector f (∑f = 1) that minimizes

        F(f) = 2·n·max(conv(f,f))

because the supplied “evaluate_sequence” is scale‑invariant.
The algorithm starts from the best known construction (global variable
``height_sequence_1``) or a uniform vector, then performs a long
simulated‑annealing run in which a random amount of mass is moved from
one index to another.  Moves that improve the objective are always
accepted; uphill moves are accepted with probability exp(−Δ/T) where the
temperature T decays exponentially.  The mass‑transfer size also shrinks
with time, allowing coarse exploration early and fine polishing later.
Occasional full‑mass moves create sparsity, and rare global restarts
inject fresh diversity.  After the annealing phase a final greedy
polish (only improving moves) runs until the time budget is exhausted.

The code uses only NumPy and the standard library, runs in < 1000 s on a
single CPU and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())

def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))

# ----------------------------------------------------------------------
# Simulated annealing search
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes the evaluation
    bound.  Returns a list of floats (the best sequence found).
    """
    start_time = time.time()
    # safety margin so we finish before the outer timeout
    deadline = start_time + budget_s - 0.05

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size & initial vector
    # --------------------------------------------------------------
    n = 4096                     # default length (fits the previous best)
    # try to start from the global best if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        if starter.size < n:
            f = np.concatenate([starter, np.zeros(n - starter.size)])
        else:
            f = starter[:n].copy()
        print(f"[{0.00:.2f}s] Loaded height_sequence_1 (size={starter.size})")
    except Exception:
        f = np.full(n, 1.0 / n, dtype=float)
        print(f"[{0.00:.2f}s] No starter – using uniform vector (n={n})")

    # Normalise to ℓ₁‑norm 1 and enforce non‑negativity
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 0.0:
        f /= s
    else:
        f = np.full(n, 1.0 / n)

    cur_f = f.copy()
    cur_score = _objective_score(cur_f)
    best_f = cur_f.copy()
    best_score = cur_score

    print(f"[{time.time() - start_time:.2f}s] Initial objective = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing core
    # --------------------------------------------------------------
    iter_cnt = 0
    # temperature schedule parameters (exponential decay)
    T0 = 0.02
    T_end = 1e-6

    # main annealing loop
    while time.time() < deadline:
        iter_cnt += 1
        now = time.time()
        elapsed = now - start_time
        frac = elapsed / budget_s          # 0 → 1

        # temperature
        T = T0 * (T_end / T0) ** frac

        # move size (percentage of source mass)
        max_move_frac = 0.20 * (1.0 - frac) + 0.001  # shrinks to ~0.001

        # pick source and destination indices
        src = rng.integers(n)
        if cur_f[src] <= 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # sometimes move *all* mass from src (to promote sparsity)
        if rng.random() < 0.01:
            delta = cur_f[src]
        else:
            delta = cur_f[src] * rng.random() * max_move_frac
            # ensure delta does not empty src completely (except the 1% case)
            delta = min(delta, cur_f[src] * 0.999)

        cand_f = cur_f.copy()
        cand_f[src] -= delta
        cand_f[dst] += delta

        cand_score = _objective_score(cand_f)

        # acceptance test
        if cand_score < cur_score:
            accept = True
        else:
            diff = cand_score - cur_score
            # avoid overflow in exp
            if diff / T > 50:
                accept = False
            else:
                accept = rng.random() < math.exp(-diff / T)

        if accept:
            cur_f = cand_f
            cur_score = cand_score
            # record improvement over the best ever found
            if cur_score < best_score - 1e-12:
                best_f = cur_f.copy()
                best_score = cur_score
                print(f"[{elapsed:.2f}s] New best = {best_score:.6f} (iter {iter_cnt})")

        # occasional global restart from a fresh Dirichlet sample
        if rng.random() < 0.0005:
            new_f = rng.dirichlet(np.ones(n))
            new_score = _objective_score(new_f)
            cur_f = new_f
            cur_score = new_score
            print(f"[{time.time() - start_time:.2f}s] Random restart, score {new_score:.6f}")
            if new_score < best_score:
                best_f = new_f.copy()
                best_score = new_score
                print(f"[{time.time() - start_time:.2f}s] Restart gave new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Greedy polishing (only accept strict improvements)
    # --------------------------------------------------------------
    # Use a very small move size and run until the absolute deadline.
    while time.time() < deadline:
        src = rng.integers(n)
        if best_f[src] <= 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = best_f[src] * rng.random() * 0.001   # ≤0.1 % of source mass
        cand = best_f.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score - 1e-12:
            best_f = cand
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] Polishing improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0.0:
        best_f /= total
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")

    # convert to plain Python list of floats (required by the evaluator)
    return [float(v) for v in best_f.tolist()]
