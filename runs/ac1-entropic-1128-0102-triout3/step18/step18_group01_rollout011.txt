"""
Propose a non‑negative step function (probability vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1️⃣  Initialise from the global starter `height_sequence_1` if present.
    The starter is up‑sampled (by repeating entries) to a target length
    `n = 32768`.  If that is impossible we fall back to the uniform vector.

2️⃣  **Smoothed‑max gradient descent** – a novel phase.
    We minimise the smooth approximation
           Jα(a) = 2·n·log‑sum‑exp(α·(a*a)) / α   (sum(a)=1)
    The gradient of the smooth max can be written in closed form using a
    convolution, so each iteration only needs a few FFTs.
    α starts modest (≈2 000) and is increased slowly; the step size decays
    like 1/√t.  After every improvement the current vector is projected onto
    the probability simplex.

3️⃣  **Projected sub‑gradient descent** (the previous robust phase) to
    polish the solution.

4️⃣  **Final random hill‑climb** for tiny mass transfers.

All three phases respect the global 1000 s budget.  The best vector found
is normalised to sum = 1, printed and returned as a plain list of Python
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – works for arbitrary sum
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = np.fft.irfft(np.fft.rfft(a, n=L_fft) * np.fft.rfft(a, n=L_fft),
                       n=L_fft)[: 2 * a.shape[0] - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (new phase)
# ----------------------------------------------------------------------


def _smooth_max_grad_descent(a0: np.ndarray,
                             n: int,
                             L_big: int,
                             L_self: int,
                             deadline: float,
                             start_time: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of `max(conv(a,a))` using gradient descent.
    The smoothing parameter α is increased slowly; the step size decays.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_self)

    # hyper‑parameters
    step0 = 0.02          # initial learning rate
    alpha0 = 2000.0       # starting smoothing factor
    alpha_max = 1e5
    it = 0
    last_improve = start_time

    while time.time() < deadline:
        it += 1
        # FFT of the current vector (re‑used twice)
        Af = np.fft.rfft(a, n=L_big)

        # ---------- self‑convolution ----------
        conv_self = np.fft.irfft(Af * Af, n=L_big)[: 2 * n - 1]
        max_conv = conv_self.max()

        # ---------- soft‑max weights ----------
        # increase α slowly (log‑linear schedule)
        alpha = min(alpha_max, alpha0 * (1.0 + it * 1e-3))
        # stable soft‑max: shift by max
        w = np.exp(alpha * (conv_self - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:               # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the smoothed max ----------
        # convolution of w with a (valid part gives length n)
        Wf = np.fft.rfft(w, n=L_big)
        conv_valid = np.fft.irfft(Af * Wf, n=L_big)[n - 1:2 * n - 1]
        # gradient of J = 4·n·conv_valid  (sum(a)=1 after projection)
        grad = 4.0 * n * conv_valid

        # ---------- update ----------
        step = step0 / (1.0 + 0.001 * it)          # decreasing step size
        cand = a - step * grad
        cand = _project_simplex(cand)             # keep it a probability vector
        cand_score = _eval_fast(cand, L_self)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] grad it={it:,} "
                  f"step={step:.3e} α={alpha:.1f} score={cand_score:.6f}")
        else:
            # occasional jitter to escape tiny plateaux
            if it % 1200 == 0:
                eps = rng.random(a.shape) * a * 5e-5
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

        # restart from the best if stuck for a while
        if time.time() - last_improve > 30.0 and it % 2000 == 0:
            eps = rng.random(best.shape) * best * 2e-4
            a = np.maximum(best + eps, 0.0)
            a = _project_simplex(a)
            last_improve = time.time()

    return best, best_score

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (the lag where the max occurs)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    np.add.at(g, i, 2.0 * a[j])
    return g


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:

    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    step0 = 0.45
    it = 0
    last_improve = start_time

    while time.time() < deadline:
        it += 1
        conv = np.fft.irfft(np.fft.rfft(a, n=L_fft) *
                            np.fft.rfft(a, n=L_fft), n=L_fft)[: 2 * a.shape[0] - 1]
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)

        cand = a - step * g
        cand = _project_simplex(cand)
        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "

        else:
            if it % 1200 == 0:
                eps = rng.random(a.shape) * a * 8e-5
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer hill‑climb with a greedy acceptance rule."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand /= cand.sum()
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cand_score:.6f}")

        if rng.random() < 0.001:   # occasional exploratory jitter
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand /= cand.sum()
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # safety margin

    # --------------------------------------------------------------
    # Problem size – 32768 gives a finer canvas while still fast enough.
    # --------------------------------------------------------------
    n = 32768
    L_self = _next_pow2(2 * n - 1)      # for self‑convolution
    L_big = _next_pow2(3 * n - 2)       # for w*a convolution (gradient phase)

    # --------------------------------------------------------------
    # Initialise from the global starter if possible.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            # up‑sample by integer repetition if lengths fit
            if n % starter.size == 0:
                factor = n // starter.size
                a0 = np.repeat(starter, factor)
                a0 = a0 / a0.sum()
                print(f"[{time.time() - start_time:.2f}s] up‑sampled global starter "
                      f"(factor={factor})")
            else:
                # fall back to uniform if up‑sampling is impossible
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
                print(f"[{time.time() - start_time:.2f}s] global starter length "
                      f"{starter.size} does not divide {n}; using uniform")
        else:
            raise ValueError
    except Exception:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        print(f"[{time.time() - start_time:.2f}s] no usable global starter; using uniform")

    # --------------------------------------------------------------
    # Baseline evaluation.
    # --------------------------------------------------------------
    best_seq = a0.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Smoothed‑max gradient descent – main improvement phase.
    # --------------------------------------------------------------
    grad_end = start_time + 0.65 * budget_s   # ~65 % of the budget
    if time.time() < grad_end:
        best_seq, best_score = _smooth_max_grad_descent(
            best_seq, n, L_big, L_self, grad_end, start_time, rng)

    # --------------------------------------------------------------
    # 2️⃣ Projected sub‑gradient descent – polish.
    # --------------------------------------------------------------
    sub_end = start_time + 0.90 * budget_s   # another ~25 % of the budget
    if time.time() < sub_end:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_self, sub_end, start_time, rng)

    # --------------------------------------------------------------
    # 3️⃣ Final hill‑climb – use whatever time is left.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(
            best_seq, L_self, deadline, start_time, rng)

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Report final bound using the provided `evaluate_sequence`.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
