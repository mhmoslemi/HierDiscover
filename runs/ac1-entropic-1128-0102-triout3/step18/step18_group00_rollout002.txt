    total = best_seq.sum()

    return [float(v) for v in best_seq]

"""
Hybrid optimizer with smooth‑max gradient (Adam) + SA.

The search proceeds in three phases:

1️⃣  A dense parametric sweep over two‑parameter families
    (monotone and symmetric “U‑shapes”) to obtain a decent starter
    vector (length n = 4096).

2️⃣  Smooth‑max optimisation:
    - Replace max(conv) by the differentiable soft‑max
      fβ(a)= (1/β)·log Σk exp(β·convk(a)).
    - Compute the gradient via FFT (O(n log n)).
    - Perform Adam updates with Euclidean projection onto the
      probability simplex after each step.
    - β is increased gradually so the approximation becomes tighter.

3️⃣  A short simulated‑annealing hill‑climb that makes tiny mass‑transfer
    moves (src→dst) and accepts uphill moves with a linearly cooling
    temperature.  The best vector seen throughout is returned.

All work respects the 1000 s budget; intermediate progress and scores
are printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _smooth_max_grad(a: np.ndarray,
                     beta: float,
                     L_conv: int,
                     L_grad: int) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Returns (conv, soft‑max weights w, gradient of the smooth max).

    conv : full self‑convolution (length 2n‑1)
    w    : soft‑max probabilities over the conv entries
    grad : ∇a  (1/β)·log Σ exp(β·conv)  = 2·(w * a_rev) restricted to the n coordinates.
    """
    n = a.shape[0]

    # 1) full self‑convolution
    conv = _conv_self_fft(a, L_conv)

    # 2) soft‑max over conv entries (stable by subtracting the max)
    shift = conv.max()
    exp_w = np.exp(beta * (conv - shift))
    w = exp_w / exp_w.sum()

    # 3) gradient via FFT (correlation of w with a reversed)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L_grad)
    A_rev = np.fft.rfft(a_rev, n=L_grad)
    v = np.fft.irfft(W * A_rev, n=L_grad)          # length L_grad, real‑valued
    # the useful slice corresponds to indices n‑1 … n‑1+n‑1
    start = n - 1
    end = start + n
    grad = 2.0 * v[start:end]                      # real numbers only
    return conv, w, grad
