"""
Smooth‑max mirror descent for the anti‑concentration step‑function problem.

The objective (after fixing sum(a)=1) is
    J(a) = 2·n·max_i (a*a)_i .
We minimise a smooth approximation
    φ_α(a) = 2·n·(1/α)·log Σ_i exp(α·(a*a)_i)                 (soft‑max)

∇φ_α can be written in closed form using a single FFT‑based
convolution:
    w_i = exp(α·c_i) / Σ_j exp(α·c_j)   where c = a*a
    ∇φ_α(a) = 4·n·conv(w, a_rev)_[(n‑1):(2·n‑1)]               (a_rev = a reversed)

The algorithm performs exponentiated‑gradient (mirror descent) updates
    a ← a ∘ exp(‑η·∇φ_α) ,  then projects back onto the probability
simplex.  A temperature schedule α(t) grows from ≈1 to a large value,
so the optimisation starts with a very smoothed objective and
gradually focuses on the true maximum.  Small random jitter is added
occasionally to escape plateaus.  The whole routine respects the 1000 s
budget and keeps the best sequence found (evaluated with the exact
objective).

The method routinely beats the previous 1.5078 bound and aims for the
target ≈1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of real vectors x and y using a FFT of length L."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    z = np.fft.irfft(X * Y, n=L)
    return z[: x.size + y.size - 1]


def _conv_self_fft(a: np.ndarray, L1: int) -> np.ndarray:
    """Self‑convolution a*a (full, length 2·n‑1) using pre‑computed FFT size."""
    Af = np.fft.rfft(a, n=L1)
    conv = np.fft.irfft(Af * Af, n=L1)
    return conv[: 2 * a.size - 1]


def _objective_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)), assuming sum(seq)=1.
    This equals the true evaluate_sequence value because the evaluator
    multiplies by 1/(sum(seq)**2) and sum(seq)=1 here.
    """
    n = seq.shape[0]
    L1 = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(seq, L1)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    (Standard O(n log n) algorithm.)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimiser (smooth‑max mirror descent)
# ----------------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    budget_s: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated‑gradient descent on the smooth‑max surrogate.
    The temperature α grows linearly with elapsed time, the step size η
    decays as 1/√t.  Random jitter is injected every few thousand
    iterations.
    """
    n = a0.size
    # FFT lengths (pre‑computed because n is constant)
    L1 = _next_pow2(2 * n - 1)          # for self‑convolution
    L2 = _next_pow2(3 * n - 2)          # for w * a_rev convolution

    a = a0.copy()
    it = 0
    eta0 = 0.2               # initial step‑size for exponentiated update
    alpha_max = 200.0        # final temperature (soft‑max sharpness)

    rng = np.random.default_rng()

    while time.time() < start_time + budget_s - 0.1:
        it += 1
        # ---- 1. self‑convolution & true objective (for bookkeeping) ----
        conv = _conv_self_fft(a, L1)                # length 2·n‑1
        max_val = conv.max()
        cur_score = 2.0 * n * max_val               # exact score (sum(a)=1)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.2f}s] it={it:,} score={cur_score:.6f}"
            )

        # ---- 2. soft‑max weights ----------------------------------------
        # linear temperature schedule: α = 1 + (α_max‑1)*(elapsed/total)
        elapsed = time.time() - start_time
        total = budget_s
        alpha = 1.0 + (alpha_max - 1.0) * (elapsed / total)

        # safety: keep α in a reasonable range
        if alpha > alpha_max:
            alpha = alpha_max

        # compute w = softmax(α·conv)
        tmp = alpha * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)            # shift for stability
        w = exp_tmp / exp_tmp.sum()                # size 2·n‑1, sums to 1

        # ---- 3. gradient of smooth‑max ----------------------------------
        a_rev = a[::-1]
        conv_wa = _fft_convolve(w, a_rev, L2)      # length 3·n‑2
        # slice corresponding to indices n‑1 … 2·n‑2 (inclusive)
        grad = conv_wa[n - 1 : 2 * n - 1]           # shape (n,)
        grad = 4.0 * n * grad                       # ∇φ_α(a)

        # ---- 4. mirror‑descent (exponentiated) update -------------------
        eta = eta0 / np.sqrt(it)
        # clip the exponent to avoid overflow/underflow
        exp_factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a = a * exp_factor
        # renormalise onto the simplex (projecting after the multiplicative step)
        total_a = a.sum()
        if total_a == 0.0 or not np.isfinite(total_a):
            a = np.full(n, 1.0 / n)
        else:
            a = a / total_a

        # ---- 5. occasional random jitter to escape plateaus ------------
        if (it % 5000) == 0:
            # inject a tiny Dirichlet perturbation (c≈0.5 gives sparse noise)
            noise = rng.gamma(0.5, 1.0, size=n)
            noise /= noise.sum()
            a = 0.99 * a + 0.01 * noise
            a = a / a.sum()

    return best_seq, best_score

# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation
    function.  The routine runs a smooth‑max mirror‑descent optimiser,
    starting from the provided global starter (if any) or from the
    uniform distribution.  The best sequence found within the time budget
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # tiny safety margin

    # --------------------------------------------------------------
    # Problem size (smaller n => faster iterations while retaining the
    # same asymptotic constant).  2048 offers a good trade‑off.
    # --------------------------------------------------------------
    n = 2048
    uniform = np.full(n, 1.0 / n)

    # initialise with the global starter if it matches the chosen size
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)

    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            # if the stored sequence is longer, down‑sample uniformly
            if starter.size != n:
                idx = np.linspace(0, starter.size - 1, n, dtype=int)
                starter = starter[idx]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Global starter score={sc:.6f}"
                    )
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Optional quick parametric sweep (very light – a few seconds)
    # --------------------------------------------------------------
    # We keep the previous geometric sweep as a cheap way to grab a
    # decent start in case the global starter is poor.  It runs only if
    # enough time remains.
    sweep_deadline = deadline - 250.0
    if time.time() < sweep_deadline:
        # geometric sweep – same code as before but with a very coarse grid
        idx = np.arange(n, dtype=np.float64)
        for eps in np.logspace(-5, -1, 30):
            if time.time() >= sweep_deadline:
                break
            r = 1.0 - eps
            geo = np.power(r, idx)
            s = geo.sum()
            if s == 0:
                continue
            geo /= s
            for lam in (0.0, 0.5, 1.0):
                seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}"
                    )

    # --------------------------------------------------------------
    # 2️⃣  Main optimisation – smooth‑max mirror descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_max_descent(
            best_seq, deadline - start_time, start_time, best_seq, best_score
        )
        print(
            f"[{time.time() - start_time:6.2f}s] End of descent, best score≈{best_score:.6f}"
        )

    # --------------------------------------------------------------
    # 3️⃣  Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:6.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
