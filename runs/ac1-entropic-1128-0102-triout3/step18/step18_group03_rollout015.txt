def upsample(vec_small, n_large):
    m = len(vec_small)
    xs = np.linspace(0, 1, m)
    xt = np.linspace(0, 1, n_large)
    large = np.interp(xt, xs, vec_small)
    # ensure nonnegative
    large = np.maximum(large, 0.0)
    # renorm sum to 1
    total = large.sum()
    if total > 0:
        large = large / total
    else:
        large = np.full(n_large, 1.0/n_large)
    return large
```

"""
Propose a high‑quality non‑negative step‑function (probability vector) that
minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in two stages:

1. **Low‑dimensional optimisation**  
   – Optimise a compact prototype of length `m` (here 256) using many
     Adam steps on a smooth‑max surrogate of the L∞‑objective.  
   – The prototype is searched with a rich set of parametric seeds
     (geometric×power shapes, Dirichlet draws, uniform mixtures).  
   – Because `m << n` the optimisation can afford many more iterations
     than a direct full‑size run.

2. **Upsampling & polishing**  
   – The best prototype is interpolated to the target size `n` (4096).  
   – A short exponentiated‑gradient phase, projected sub‑gradient descent,
     and a final random hill‑climb further reduce the exact objective.

All operations respect the required constraints (non‑negative entries,
normalisation, clipping) and the total running time is limited to the
given budget (≈1000 s).  The returned list is a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2 * Σ_j a_j p_{i+j}.
    Implemented as a convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Fast exact evaluation (used for cheap benchmarking and final scoring)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation (size m)
# ----------------------------------------------------------------------


def _make_general_shape(n: int, r: float = 0.9, p: float = 0.0, q: float = 0.0) -> np.ndarray:
    """
    Shape a_i ∝ r**i · (i+1)**p · (n-i+1)**q, normalised to sum = 1.
    r∈(0,1) gives a geometric decay,
    p controls a power at the left boundary,
    q controls a power at the right boundary.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)                # prevent overflow
    shape = np.exp(log_shape)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative (exponential) updates on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score

            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")

            eps = 1e-4 * rng.random(n) * a

def _upsample(vec_small: np.ndarray, n_large: int) -> np.ndarray:

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for beta in beta_schedule:
        if time.time() >= deadline - 200:
            break
        for idx in range(len(pool)):
            a = pool[idx]
            m_vec = np.zeros_like(a)
            v_vec = np.zeros_like(a)
            cur_score = pool_scores[idx]
            t = 0
            # allow many cheap steps because m is small
            max_iter = 8000
            while time.time() < deadline - 150:
                t += 1
                if t > max_iter:
                    break
                conv = _conv_self_fft(a, L_fft_m)
                p_weights = _softmax_weights(conv, beta)
                grad = _grad_softmax_new(a, p_weights, L_grad_m)

                # Adam update
                m_vec = beta1 * m_vec + (1.0 - beta1) * grad
                v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
                m_hat = m_vec / (1.0 - beta1 ** t)
                v_hat = v_vec / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)

                a_candidate = a - step
                a_candidate = _project_simplex(a_candidate)

                cand_score = _eval_fast(a_candidate, L_fft_m)

                # Accept improvements outright; occasional uphill moves help escape plateaus.
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    prob = np.exp(-(cand_score - cur_score) * 3.0 / (np.sqrt(t) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score

                # Update pool and global best
                if cand_score < pool_scores[idx]:
                    pool[idx] = a_candidate.copy()
                    pool_scores[idx] = cand_score
                if cand_score < best_score_m:
                    best_score_m = cand_score
                    best_m = a_candidate.copy()
                    print(f"[{time.time() - start:.2f}s] Adam β={beta} it={t:,} score_m={cand_score:.6f}")

                # Small random jitter to promote exploration
                if t % 2000 == 0:
                    eps_move = 1e-4 * rng.random(m) * a
                    a = np.maximum(a + eps_move, 0.0)
                    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Upsample best prototype to the final size n.
    # ------------------------------------------------------------------
    best_n = _upsample(best_m, n)
    L_fft_n = _next_pow2(2 * n - 1)
    L_grad_n = _next_pow2(3 * n - 2)

    best_score_n = _eval_fast(best_n, L_fft_n)
    print(f"[{time.time() - start:.2f}s] after up‑sampling score (n) = {best_score_n:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Exponentiated‑gradient polishing (full size).
    # ------------------------------------------------------------------
    if time.time() < deadline - 100:
        best_n, best_score_n = _expgrad_phase(
            best_n, L_fft_n, deadline - 100, start,
            best_n, best_score_n, rng)

    # ------------------------------------------------------------------
    # 3️⃣ Projected sub‑gradient descent (classic polishing).
    # ------------------------------------------------------------------
    if time.time() < deadline - 50:
        best_n, best_score_n = _subgradient_descent(
            best_n, L_fft_n, deadline - 50, start,
            best_n, best_score_n, rng)

    # ------------------------------------------------------------------
    # 4️⃣ Final random hill‑climb (tiny mass transfers).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_n[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps_move = best_n[src] * rng.random() * 5e-3   # ≤0.5 % of source mass
        prop = best_n.copy()
        prop[src] -= eps_move
        prop[dst] += eps_move
        new_score = _eval_fast(prop, L_fft_n)
        if new_score < best_score_n:
            best_n = prop
            best_score_n = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_n = np.maximum(best_n, 0.0)
    total = best_n.sum()
    if total > 0.0:
        best_n = best_n / total
    else:
        best_n = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score_n:.6f}")
    return [float(v) for v in best_n]
