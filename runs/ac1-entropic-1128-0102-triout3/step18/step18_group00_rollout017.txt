def _adam_optimize(a0, L_corr, L_conv, deadline, start_time, best_seq, best_score, rng):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.1   # hyperparameter
    beta0 = 0.5
    beta_max = 8.0
    it = 0
    while time.time() < deadline:
        it += 1
        # compute A_fft
        A_fft = np.fft.rfft(a, n=L_corr)
        # conv
        conv = np.fft.irfft(A_fft * A_fft, n=L_corr)[: 2*n - 1]
        max_conv = conv.max()
        # compute softmax weights
        fraction = (time.time() - start_time) / (deadline - start_time + 1e-12)
        beta = beta0 + fraction * (beta_max - beta0)
        # stable softmax
        # shift by max to avoid overflow
        exps = np.exp(beta * (conv - max_conv))
        w = exps / exps.sum()
        # compute correlation via FFT
        W_fft = np.fft.rfft(w, n=L_corr)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), n=L_corr)[: n]   # first n entries
        grad = 4.0 * n * corr   # gradient of objective (approx)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = a - step
        # projection onto simplex
        a_new = _project_simplex(a_new)
        # evaluate actual objective (fast)
        score = _eval_fast(a_new, L_conv)
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} beta={beta:.2f} score={score:.6f}")
        # occasional jitter
        if it % 3000 == 0:
            eps = 1e-5 * rng.random(n) * a_new
            a_new = np.maximum(a_new + eps, 0.0)
            a_new = a_new / a_new.sum()
        a = a_new
    return best_seq, best_score

"""Improved search for low L∞ self‑convolution bound.

The algorithm uses:
1. A dense parametric sweep over monotone & symmetric “geo‑power” families.
2. A smooth‑approximation of the max‑convolution objective (soft‑max) and
   Adam‑style projected gradient descent on the probability simplex.
   The smoothing parameter β is increased linearly so the optimisation first
   sees a smooth landscape and later focuses on the true max.
3. A short simulated‑annealing hill‑climb to polish the final vector.

All steps respect the 1000 s budget and keep the vector non‑negative with
unit ‑L¹ norm.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _cross_corr_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation (a ⋆ w)[k] = Σ_i a_i w_{i+k} for k≥0.
    Returns the first len(a) entries (k = 0 … n‑1).
    """
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(A * np.conj(W), n=L)[: a.shape[0] + w.shape[0] - 1]
    return corr_full[: a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²) (fast via FFT)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Candidate families – same as the previous code (monotone & symmetric)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(-lam·i) (L1‑normalised)."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric U‑shape a_i ∝ (i+1)^p·exp(-lam·i) + reversed part."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    right = left[::-1]
    shape = left + right
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Adam optimisation with a soft‑max approximation of max‑convolution
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    L_corr: int,
    L_conv: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed objective
        O_β(a) = 2·n·(1/β)·log Σ_k exp(β·(a*a)_k).

    β is increased linearly from β0 to β_max during the run.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12  # base learning rate – tuned empirically
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    beta0 = 0.5      # start smoothing
    beta_max = 9.0   # final smoothing (large enough to approximate max)
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- FFT of a (big enough for both conv and correlation) ----
        A_fft = np.fft.rfft(a, n=L_corr)

        # ---- self‑convolution (for soft‑max) ----
        conv = np.fft.irfft(A_fft * A_fft, n=L_corr)[: 2 * n - 1]
        max_conv = conv.max()

        # ---- soft‑max weights ----
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        beta = beta0 + frac * (beta_max - beta0)

        # stable soft‑max (shift by max)
        exps = np.exp(beta * (conv - max_conv))
        w = exps / exps.sum()          # Σ w = 1

        # ---- correlation a ⋆ w (first n entries) ----
        W_fft = np.fft.rfft(w, n=L_corr)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), n=L_corr)[: n]   # k = 0 … n‑1

        # Gradient of the smoothed objective: 4·n·corr
        grad = 4.0 * n * corr

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

"""Improved search for low L∞ self‑convolution bound.

The algorithm uses three stages:

1️⃣  A dense parametric sweep over monotone and symmetric “geo‑power”
    families (the same families used previously but explored a bit longer).

2️⃣  A smooth‑approximation of the max‑convolution objective
    (soft‑max) combined with a projected Adam optimiser.
    The smoothing parameter β is increased linearly, so the optimiser
    first sees a smooth landscape and later concentrates on the true
    maximum.

3️⃣  A short simulated‑annealing hill‑climb to polish the final vector.

All stages respect the 1000 s budget, keep the vector non‑negative,
and maintain unit L¹‑norm.  The returned list contains the heights of the
step function.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _cross_corr_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation (a ⋆ w)[k] = Σ_i a_i w_{i+k} for k ≥ 0.
    Returns the first len(a) entries (k = 0 … n‑1).
    """
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(A * np.conj(W), n=L)[: a.shape[0] + w.shape[0] - 1]
    return corr_full[: a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²) (fast via FFT)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Candidate families – monotone and symmetric “geo‑power”
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(-lam·i) (L1‑normalised)."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape” a_i ∝ (i+1)^p·exp(-lam·i) + reversed term."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    right = left[::-1]
    shape = left + right
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Adam optimisation using a soft‑max approximation of the max
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    L_corr: int,
    L_conv: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smoothed objective

        O_β(a) = 2·n·(1/β)·log Σ_k exp(β·(a*a)_k).

    The smoothing parameter β is linearly increased from β0 to β_max.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12  # base learning rate (empirically good)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    beta0 = 0.5      # start of smoothing
    beta_max = 9.0   # final smoothing (approximates the max)

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- FFT of a (large enough for both conv & correlation) -----
        A_fft = np.fft.rfft(a, n=L_corr)

        # ----- self‑convolution (for the soft‑max) -----
        conv = np.fft.irfft(A_fft * A_fft, n=L_corr)[: 2 * n - 1]
        max_conv = conv.max()

        # ----- smooth weights (soft‑max) -----
        fraction = (time.time() - start_time) / (deadline - start_time + 1e-12)
        beta = beta0 + fraction * (beta_max - beta0)

        # stable soft‑max: shift by the maximum to avoid overflow
        exps = np.exp(beta * (conv - max_conv))
        w = exps / exps.sum()            # Σ w = 1

        # ----- correlation a ⋆ w (first n entries) -----
        W_fft = np.fft.rfft(w, n=L_corr)
        corr = np.fft.irfft(A_fft * np.conj(W_fft), n=L_corr)[: n]

        # Gradient of the smoothed objective: 4·n·corr
        grad = 4.0 * n * corr

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step

        # ----- exact (non‑smooth) objective for bookkeeping -----
        score = _eval_fast(a_candidate, L_conv)
        if score < best_score:

            best_seq = a_candidate.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                f"beta={beta:.2f} score={score:.6f}"

            jitter = 5e-5 * rng.random(n) * a_candidate
            a_candidate = np.maximum(a_candidate + jitter, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate

    return best_seq, best_score
