"""
Gradient‑guided stochastic hill‑climbing with occasional smoothing
and progressive up‑sampling.

The evaluation function is scale‑invariant: after normalising
∑a = 1 the objective becomes 2·n·max(conv(a,a)).  We therefore
keep the sequence normalised at every step.

Strategy
--------
1.  Initialise from the global best vector ``height_sequence_1`` if it
    exists, otherwise from a uniform distribution.
2.  Repeatedly improve the current vector with a hill‑climbing loop:
    *   Compute the full self‑convolution and locate the index k where
        the maximum occurs.
    *   The contribution of entry i to this peak is a[i]·a[k‑i] (zero if
        k‑i is out of range).  We take a source index with a large
        contribution and a destination index with a small contribution,
        and move a small fraction of mass from the source to the
        destination.  The move is accepted only if it strictly reduces
        the objective.
    *   With a modest probability we perform a *smoothing* move:
        pick a short interval and flatten the mass inside it (preserving
        the total).  This can break symmetry and lower the peak.
    *   The step size is large at the beginning of the search and
        decays linearly with the remaining time, allowing coarse
        exploration followed by fine‑grained polishing.
3.  When the allotted time for the current resolution is exhausted,
    the vector is *upsampled* (each entry is split into two halves).
    Upsampling does not change the objective but doubles the number of
    degrees of freedom, enabling a lower constant C in
    max(conv)≈C/n.
4.  The process repeats until the global time budget (≈ 1000 s) is
    exhausted or the target bound 1.503 is reached.
5.  The best vector found is returned (non‑negative, normalised).

The algorithm is deliberately different from the LP‑based approach
used previously and focuses on cheap, gradient‑informed local moves,
which are fast enough to explore many thousands of candidates
within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _contributions_to_peak(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given peak index k, return the vector
    c[i] = a[i] * a[k-i] (zero if k-i out of range).
    This measures how much entry i participates in the maximal
    convolution value.
    """
    n = a.shape[0]
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[mask] * a[idx[mask]]
    return contrib


def _upsample(a: np.ndarray) -> np.ndarray:
    """
    Double the resolution by splitting each entry into two halves.
    The total mass is preserved.
    """
    return np.repeat(a, 2) * 0.5


# ----------------------------------------------------------------------
# Hill‑climbing with gradient‑guided transfers and smoothing moves
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                cur_score: float,
                deadline: float,
                rng: np.random.Generator,
                start_time: float,
                total_budget: float):
    """
    Perform stochastic local optimisation on vector a until `deadline`.
    Returns the best vector found and its score.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score

    cur = a.copy()
    cur_score_local = cur_score

    # parameters
    smoothing_prob = 0.20      # chance to try a smoothing move
    min_step_frac = 0.001      # smallest fraction of source mass to move
    max_step_frac = 0.02       # largest fraction at the very start

    while time.time() < deadline:
        # ----- compute current peak and contributions -----
        conv = _conv_self(cur)
        k_max = int(np.argmax(conv))
        # contributions of each entry to this peak
        contrib = _contributions_to_peak(cur, k_max)

        # ----- randomly decide move type -----
        if rng.random() < smoothing_prob:
            # ---- Smoothing move: flatten a random short window ----
            centre = rng.integers(n)
            radius = rng.integers(1, 6)          # window half‑width 1…5
            lo = max(0, centre - radius)
            hi = min(n - 1, centre + radius)
            win_len = hi - lo + 1
            if win_len <= 1:
                continue
            sum_window = cur[lo:hi + 1].sum()
            cand = cur.copy()
            cand[lo:hi + 1] = sum_window / win_len   # uniform redistribution
            cand_score = _objective_score(cand)
            if cand_score + 1e-12 < cur_score_local:
                cur = cand
                cur_score_local = cand_score
                if cand_score < best_score - 1e-12:
                    best = cand.copy()
                    best_score = cand_score
                    print(f"[{time.time() - start_time:.2f}s] smoothing improved to {best_score:.6f}")
            continue

        # ----- Transfer move (gradient‑guided) -----
        # source: pick an index with large contribution and positive mass
        src_mask = (cur > 1e-15) & (contrib > 0)
        if not src_mask.any():
            # nothing movable – exit
            break
        src_indices = np.where(src_mask)[0]
        src_weights = contrib[src_mask]
        src_probs = src_weights / src_weights.sum()
        src = rng.choice(src_indices, p=src_probs)

        # destination: pick an index with low contribution (preferably different from src)
        low_thr = np.median(contrib)
        dst_candidates = np.where(contrib <= low_thr)[0]
        # ensure we have at least one candidate not equal to src
        if dst_candidates.size == 0:
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)
        else:
            if dst_candidates.size == 1 and dst_candidates[0] == src:
                dst = rng.integers(n)
                while dst == src:
                    dst = rng.integers(n)
            else:
                # optionally remove src from the pool
                if src in dst_candidates:
                    dst_candidates = dst_candidates[dst_candidates != src]
                if dst_candidates.size == 0:
                    dst = rng.integers(n)
                    while dst == src:
                        dst = rng.integers(n)
                else:
                    dst = rng.choice(dst_candidates)

        # step size – decays linearly with remaining time
        remaining_frac = max(0.0, deadline - time.time()) / total_budget
        step_frac = min_step_frac + (max_step_frac - min_step_frac) * remaining_frac
        delta = cur[src] * step_frac
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        # evaluate
        cand_score = _objective_score(cand)
        if cand_score + 1e-12 < cur_score_local:
            cur = cand
            cur_score_local = cand_score
            if cand_score < best_score - 1e-12:
                best = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start_time:.2f}s] hill improved to {best_score:.6f}")

    return best, best_score

    # --------------------------------------------------------------
    max_n = 16384                 # we will not exceed this length
    total_budget = budget_s

    while time.time() < deadline - 0.5:
        n = f.shape[0]
        time_left = deadline - time.time()
        # how many up‑sampling steps could still be performed?
        remaining_levels = 0
        tmp = n
        while tmp < max_n:
            remaining_levels += 1
            tmp <<= 1
        # allocate a fair share of the remaining time to the current level
        if remaining_levels == 0:
            alloc = time_left
        else:
            alloc = time_left / (remaining_levels + 1)

        # ----- hill‑climbing phase on current resolution -----
        f, cur_score = _hill_climb(f,
                                   cur_score,
                                   time.time() + alloc,
                                   rng,
                                   start_time,
                                   total_budget)

        # keep the globally best vector seen so far
        if cur_score < best_score - 1e-12:
            best_f = f.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] New global best = {best_score:.6f}")

        # early exit if target achieved
        if best_score <= 1.503:
            print(f"[{time.time() - start_time:.2f}s] Target bound reached!")
            break

        # ----- up‑sample if we still have room and time -----
        if f.shape[0] < max_n and (deadline - time.time() > 0.5):
            f = _upsample(f)
            # renormalise (numerical safety)
            total = f.sum()
            if total > 0:
                f = f / total
            cur_score = _objective_score(f)
            print(f"[{time.time() - start_time:.2f}s] Upsampled to n={f.shape[0]}, score={cur_score:.6f}")
            if cur_score < best_score - 1e-12:
                best_f = f.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] Updated global best after up‑sample = {best_score:.6f}")
        else:
            # cannot up‑sample any further – finish
            break

    # --------------------------------------------------------------
    # 4. Final normalisation and return
    # --------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    tot = best_f.sum()
    if tot > 0.0:
        best_f = best_f / tot
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_f.tolist()]
