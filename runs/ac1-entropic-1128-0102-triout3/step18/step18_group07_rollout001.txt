def _alternating_projection_overrelax(a0, max_iter, deadline, lambda_rel=1.2):
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    for i in range(max_iter):
        if time.time() >= deadline:
            break
        # Compute FFT
        F = np.fft.rfft(a, n=L)
        # Compute flat magnitude
        energy = np.dot(a, a)
        flat_mag = np.sqrt(energy / L)
        # Replace magnitude
        F_flat = flat_mag * np.exp(1j * np.angle(F))
        a_full = np.fft.irfft(F_flat, n=L)
        a_spec = a_full[:n]
        # domain projection
        a_proj = np.maximum(a_spec, 0.0)
        a_proj = _project_simplex(a_proj)
        # over-relaxation
        a_new = a + lambda_rel * (a_proj - a)
        # re-project to maintain feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new
        # evaluate
        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] alt‑proj overrelax iter {i} new best {best_score:.6f}")
    return best_a, best_score

def _greedy_step(a, step_factor):
    grad, max_val = _subgradient_max(a)

    src = np.argmax(grad)

    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    step_factor = initial_step_factor
    n = a.shape[0]
    while time.time() < deadline:
        cand, max_val, moved = _greedy_step(a, step_factor)
        if not moved:
            step_factor *= decay
            if step_factor < min_step_factor:

"""
Improved optimizer for the L∞ self‑convolution objective.

The algorithm builds a carefully tuned pipeline:

1️⃣  A large pool of random Dirichlet draws and a richer set of deterministic
    windows (Hann, Blackman, Kaiser, exponential, Gaussian…) to obtain a
    strong starting point.

2️⃣  An *over‑relaxed alternating projection* that forces the Fourier magnitude
    of the (zero‑padded) sequence to be flat while staying on the probability
    simplex.  Over‑relaxation (λ>1) helps escape cycles that limited the previous
    approach.

3️⃣  A projected Adam optimiser on a log‑sum‑exp smooth approximation of the
    max‑norm, with an extended temperature schedule (up to β=10 000) and a
    modest learning‑rate decay.

4️⃣  A *greedy sub‑gradient descent* that directly targets the indices where the
    convolution attains its maximum.  At each step mass is transferred from the
    most “harmful’’ coefficient (largest sub‑gradient) to the most “helpful’’
    one (smallest sub‑gradient).  This rapidly reduces the max‑entry.

5️⃣  A final random hill‑climbing stage (mass transfer, swaps and block moves)
    for polishing.

All phases respect the global 1000 s budget and constantly keep the best
candidate seen so far.  The returned sequence is a probability vector
(non‑negative, sum = 1), which can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (set inside propose_candidate)
# ----------------------------------------------------------------------
start_time = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (zero‑padded)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (sum forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max w.r.t. a (soft‑max weighting)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()
    rev_corr = _convolve_fft(a[::-1], w)
    n = a.shape[0]
    return 2.0 * rev_corr[n - 1 : n - 1 + n]

# Phase‑0 helpers – random draws & deterministic windows
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [1e‑3, 5]
        conc_log = rng.uniform(-3.0, 0.69897)  # 10**0.69897 ≈ 5
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a collection of classic and exotic windows (already normalised)."""
    k = np.arange(n)

    # sine
    y = np.sin(np.pi * (k + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * k / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Blackman
    y = (0.42 - 0.5 * np.cos(2.0 * np.pi * k / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * k / (n - 1)))
    yield _project_simplex(np.maximum(y, 0.0))

    # Kaiser (approximation using cosine series – no SciPy)
    for beta in (2.0, 5.0):
        # simple approximation: window = i0(beta*sqrt(1-(2k/(n-1)-1)^2))/i0(beta)
        # Use np.i0 from numpy (available)
        arg = beta * np.sqrt(1.0 - ((2.0 * k) / (n - 1) - 1.0) ** 2)
        y = np.i0(arg) / np.i0(beta)
        yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(k, n - 1 - k).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # exponential decay (various scales)
    for scale in (n / 8.0, n / 4.0, n / 2.0):
        y = np.exp(-k / scale)
        yield _project_simplex(np.maximum(y, 0.0))

    # a few Gaussian windows
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((k - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

# Phase‑1 – alternating projection with over‑relaxation
# ----------------------------------------------------------------------
def _alternating_projection_overrelax(a0: np.ndarray, max_iter: int,
                                      deadline: float, lambda_rel: float = 1.2):
    """
    Repeatedly:
        – flatten the Fourier magnitude (same total energy for all frequencies)
        – enforce non‑negativity & sum‑to‑one (simplex projection)
    The update is over‑relaxed to escape cycles.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)          # length for FFT (≥ 2·n‑1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of zero‑padded a
        F = np.fft.rfft(a, n=L)

        # Flat magnitude that preserves total energy
        energy = np.dot(a, a)
        flat_mag = np.sqrt(energy / L)

        # Replace magnitude, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Inverse FFT and truncate
        a_full = np.fft.irfft(F_flat, n=L)
        a_spec = a_full[:n]

        # Projection onto feasible set
        a_proj = np.maximum(a_spec, 0.0)
        a_proj = _project_simplex(a_proj)

        # Over‑relaxed update
        a_new = a + lambda_rel * (a_proj - a)

        # Re‑project to stay feasible (helps numerical stability)
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] alt‑proj λ={lambda_rel:.2f} iter {i:5d} new best {best_score:.6f}")

    return best_a, best_score

# Phase‑2 – Adam optimiser on smooth max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, init_score: float,
                  beta_vals, deadline: float,
                  lr0: float = 0.07):
    """
    Projected Adam optimiser on the smooth log‑sum‑exp approximation.
    Learning‑rate decays gently with the temperature.
    """
    a = a0.copy()
    cur_score = init_score
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        # simple learning‑rate schedule (decrease as beta grows)
        lr = lr0 / (1.0 + 0.1 * beta)

        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            # accept if improvement, else occasional uphill move
            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            if np.linalg.norm(step) < 1e-13:
                break

    return best_a, best_score

# Phase‑3 – greedy sub‑gradient descent targeting the max‑entry
# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray):
    """Return (sub‑gradient, current max value) of max(conv(a,a))."""
    conv = _conv_self(a)
    max_val = conv.max()
    # treat all indices within machine epsilon of the max as active
    tol = max_val * 1e-12 + 1e-15
    K = np.where(conv >= max_val - tol)[0]
    if K.size == 0:
        K = np.array([np.argmax(conv)], dtype=int)

    w = np.zeros_like(conv)
    w[K] = 1.0 / K.size
    # gradient = 2 * (rev_a ⋆ w) restricted to length n
    rev = a[::-1]
    grad_full = _convolve_fft(rev, w)           # length = 3·n‑2
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val


def _greedy_descent(a0: np.ndarray, init_score: float,
                    deadline: float,
                    start_step: float = 0.12,
                    min_step: float = 1e-4,
                    decay: float = 0.98):
    """
    Repeatedly move a small fraction of mass from the coefficient with the
    largest sub‑gradient (most harmful) to the one with the smallest
    sub‑gradient (most helpful).  The step size decays slowly.
    """
    a = a0.copy()
    cur_score = init_score
    best_a = a.copy()
    best_score = cur_score
    step = start_step

    while time.time() < deadline:
        grad, max_val = _subgradient_max(a)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        if a[src] <= 0.0 or step < min_step:
            step *= decay
            if step < min_step:
                break
            continue

        delta = min(step * a[src], a[src])
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] greedy new best {best_score:.6f}")
            a = cand
            cur_score = cand_score
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            # occasional uphill acceptance to escape plateaus
            if np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

        # slowly decay step size
        if np.random.rand() < 0.05:
            step = max(min_step, step * decay)

    return best_a, best_score

# Phase‑4 – final random hill‑climbing (mass transfer + swaps)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, init_score: float, deadline: float):
    a = a0.copy()
    cur_score = init_score
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue

        # option 1 – transfer a fraction of source mass
        delta = a[src] * rng.random() * 0.15  # up to 15 %
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

        # option 2 – occasional direct swap (helps with discrete structure)
        if rng.random() < 0.03:
            idx1, idx2 = rng.choice(n, size=2, replace=False)
            cand2 = a.copy()
            cand2[idx1], cand2[idx2] = cand2[idx2], cand2[idx1]
            cand2 = _project_simplex(cand2)
            cand2_score = _objective_score(cand2)
            if cand2_score < best_score:
                best_score = cand2_score
                best_a = cand2.copy()
                a = cand2
                cur_score = cand2_score
                print(f"[{time.time() - start_time:.2f}s] swap‑climb new best {best_score:.6f}")
            elif cand2_score < cur_score:
                a = cand2
                cur_score = cand2_score

    return best_a, best_score
