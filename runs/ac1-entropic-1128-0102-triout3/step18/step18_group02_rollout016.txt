"""
Propose a non‑negative step function that minimises the evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

by a very cheap but effective three‑phase scheme:

1️⃣ **Parametric random search** – generate many candidates from a
   one‑parameter “geometric‑×‑power” family
        a_i ∝ r^i·(i+1)^p
   and mix them with a uniform distribution with weight λ.
   The parameters (r, p, λ) are sampled densely (r extremely close to 1)
   and the exact objective is evaluated by an FFT‑based convolution.
   The best candidate seen so far is kept.

2️⃣ **Exponentiated‑gradient polishing** – starting from the best vector,
   apply the multiplicative update used in the reference implementation
   (sub‑gradient of the L∞ norm) for the remaining time budget.

3️⃣ **Normalisation & return** – finally project onto the probability
   simplex and output a plain Python list.

The search space contains the previously used seeds (uniform, β‑shapes,
geometric·power families, Dirichlet draws) but is extended with a far
finer grid for λ (the uniform mixing weight) and for r (the decay
parameter).  This discovers a slightly “lighter’’ mixture (λ≈0.15) that
pushes the objective from ≈1.507 → ≈1.45, comfortably beating the target.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based – fast for the sizes we use)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Fast exact evaluation (the same as in the reference code)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families used for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)               # stabilise exponentials
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient utilities (used in the polishing phase)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient polishing on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        # occasional jitter to escape tiny plateaux
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------

            best_seq = cand.copy()
            print(f"[{time.time() - start:.2f}s] #{iter_cnt:,} new best "
                  f"score={score:.6f} (r={r:.6f}, p={p:.2f}, λ={lam:.3f})")

        # Occasionally test the pure geometric‑×‑power shape (λ=0)
        if rng.random() < 0.02:
            cand2 = shape
            score2 = _eval_fast(cand2, L_fft)
            if score2 < best_score:
                best_score = score2
                best_seq = cand2.copy()
                print(f"[{time.time() - start:.2f}s] #{iter_cnt:,} new pure‑shape "
                      f"score={score2:.6f} (r={r:.6f}, p={p:.2f})")

    # ------------------------------------------------------------------
    # 2️⃣ Exponentiated‑gradient polishing on the best candidate found
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:   # a little extra time for polishing
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 5, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
