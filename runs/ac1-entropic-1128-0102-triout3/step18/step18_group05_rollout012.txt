def simplex_projection(v):
   """Project onto the probability simplex."""
   # uses algorithm of Wang & Carreira-Perpiñán.
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   ind = np.arange(len(v)) + 1
   cond = u - cssv / ind > 0
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

def simplex_projection(v):
    """Project a vector v onto the probability simplex (non-negative, sum=1)."""
    # Based on algorithm from https://doi.org/10.1007/978-3-540-85969-5_1
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero? return uniform?
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid gradient‑descent / random‑swap optimizer for the max‑convolution bound.

The objective is  F(a)=2·n·max(conv(a,a))/(∑a)².
It is scale‑invariant, so we keep the vector on the probability simplex
( non‑negative entries summing to 1 ).
For a given a we compute the full linear self‑convolution with an FFT
(O(n log n)).  The maximum is attained at a few shift indices; the
sub‑gradient of the max‑convolution with respect to a can be written in
closed form:

    ∂max conv / ∂a_i = Σ_{k∈A} (a_i + a_{k−i}) ,

where A is the set of active shifts (those attaining the maximum).
A gradient step a←a−η∇ pushes mass away from the positions that
contribute most to the peak.  After the step we project back onto the
simplex (standard Euclidean projection).  When a step does not improve
the objective we fall back to a cheap random pairwise mass transfer,
accepting only strict improvements.  The whole search runs for the
allocated time (default 1000 s) and finally returns the best sequence
found.

The algorithm works with a larger problem size (n≈8192) than the
reference implementation; the extra degrees of freedom give the
optimizer more chance to lower the bound below the previous best
≈ 1.5078.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – maps any vector onto {x≥0, Σx = 1}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):               # numerical safety net
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Objective (scale‑invariant) – assumes Σa = 1
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    max_conv = _max_conv(a)
    return float(2.0 * n * max_conv)

# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (active‑shift version)
# ----------------------------------------------------------------------
def _max_conv_active_shifts(conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Indices of shifts whose value is within eps of the maximum."""
    max_val = conv.max()
    return np.where(conv >= max_val - eps)[0]

def _gradient_active(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of max(conv(a,a)) with respect to a,
    using only the active shift indices.
    """
    n = a.shape[0]
    L = len(active_shifts)
    grad = L * a                     # the “+a_i” part for each active shift
    for k in active_shifts:
        # range of i for which a_i participates in conv_k
        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k - idx                # the paired index
        grad[idx] += a[jdx]          # the a_{k-i} part
    return grad

# ----------------------------------------------------------------------
# Random local refinement (pairwise mass transfers)
# ----------------------------------------------------------------------
def _random_swap_refine(a: np.ndarray, cur_score: float,
                        deadline: float, rng: np.random.Generator) -> tuple:
    """Very cheap polishing: move a tiny random fraction of mass, keep improvements."""
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or best[src] <= 1e-15:
            continue
        # move at most 0.5 % of the source mass
        delta = best[src] * rng.random() * 0.005
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] random‑swap improved to {best_score:.6f}")
    return best, best_score

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05       # tiny safety margin

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n gives more freedom for improvement.
    # --------------------------------------------------------------
    n = 8192                       # feel free to change (must fit in memory)

    # --------------------------------------------------------------
    # Initialise a.  Prefer the globally‑available starter if it fits.
    # --------------------------------------------------------------
    try:

            if starter.size > n:
                # truncate if the starter is longer than our chosen n
                f = starter[:n].copy()
            else:

                f[:starter.size] = np.maximum(starter, 0.0)
            total = f.sum()
            if total > 0.0:

            else:

        else:

    except Exception:

    # --------------------------------------------------------------
    # Gradient‑descent phase driven by active‑shift sub‑gradient
    # --------------------------------------------------------------
    iter_cnt = 0
    no_improve_steps = 0
    max_no_improve = 250          # after many dead steps we trigger random swaps
    while time.time() < deadline - 1.0:
        conv = _conv_self(f)
        max_val = conv.max()
        active = _max_conv_active_shifts(conv, eps=1e-12)

        # compute sub‑gradient for the current active set
        grad = _gradient_active(f, active)

        # try a few step sizes (backtracking line search)
        improved = False

        else:
            no_improve_steps += 1
            # occasional random pairwise swap if gradient step stalls
            if no_improve_steps >= max_no_improve:
                src = rng.integers(n)
                dst = rng.integers(n)
                if src != dst and f[src] > 1e-15:
                    delta = f[src] * rng.random() * 0.02  # up‑to 2 % move
                    cand = f.copy()
                    cand[src] -= delta
                    cand[dst] += delta
                    cand_score = _objective_score(cand)
                    if cand_score < best_score:
                        f = cand
                        best_f = cand.copy()
                        best_score = cand_score
                        print(f"[{time.time() - start_time:.2f}s] Random swap reset improved to {best_score:.6f}")
                no_improve_steps = 0

        iter_cnt += 1

    # --------------------------------------------------------------
    # Final cheap stochastic polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 0.2:
        print(f"[{time.time() - start_time:.2f}s] Starting final random‑swap polishing")
        best_f, best_score = _random_swap_refine(best_f, best_score, deadline, rng)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0.0:

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
