def _smooth_expgrad_phase(a0, L_conv, L_corr, deadline, start_time, rng):
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_conv)
    # define schedule; total intended max iterations maybe 20000 but break on deadline.
    alphas = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0]   # ascend
    for alpha in alphas:
        # step size (eta) schedule within each block
        it = 0
        while time.time() < deadline:
            it += 1
            if it > 5000: break  # move to next alpha
            # compute conv
            conv = _conv_self_fft(a, L_conv)  # length 2n-1
            # compute w
            # avoid overflow: subtract max conv
            max_c = conv.max()
            exp_vals = np.exp(alpha * (conv - max_c))
            w = exp_vals / exp_vals.sum()
            # correlation w⋆a
            a_rev = a[::-1]
            # compute convolution of w and a_rev using FFT
            W = np.fft.rfft(w, n=L_corr)
            Ar = np.fft.rfft(a_rev, n=L_corr)
            corr_full = np.fft.irfft(W * Ar, n=L_corr)[: (len(w) + len(a) - 1)]
            start = len(a) - 1
            corr = corr_full[start:start + len(a)]
            # update multiplicatively
            eta = 0.3 / (1.0 + 1e-4 * it)  # decaying
            mult = np.exp(-eta * corr)
            a = a * mult
            # renorm to simplex
            total = a.sum()
            if total == 0.0 or not np.isfinite(total):
                a = np.full_like(a, 1.0 / len(a))
            else:
                a = a / total
            # occasional jitter
            if it % 2000 == 0:
                eps = rng.random(a.shape) * a * 5e-5
                a = np.maximum(a + eps, 0.0)
                a = a / a.sum()
            # evaluate score
            sc = _eval_fast(a, L_conv)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] smooth_expgrad (α={alpha}) it={it} score={sc:.6f}")
        # after each alpha block, we can jitter around best
        eps = rng.random(best.shape) * best * 1e-4
        a = np.maximum(best + eps, 0.0)
        a = a / a.sum()
    return best, best_score

"""
Propose a non‑negative step function (probability vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The search proceeds in five stages:

1️⃣  A fine‐grained parametric sweep over two families (geometric×power and
    pure Gaussian) to obtain a decent starter.

2️⃣  Classical projected sub‑gradient descent on the exact L∞ objective
    (as in the reference solution).

3️⃣  **New** smooth‑exponential‑gradient phase.
    The L∞ term is approximated by a soft‑max:
        smax_α(conv) = (1/α)·log Σ_i exp(α·conv_i) .
    Its gradient w.r.t. a can be written using a cross‑correlation.
    We perform multiplicative (exponential) updates
        a_i ← a_i·exp(‑η·corr_i)
    with a schedule of increasing smoothing parameter α and a decaying
    learning rate η, projecting onto the probability simplex after each step.

4️⃣  A short exponentiated‑gradient run (hard max sub‑gradient) – useful as
    a final polish.

5️⃣  Random hill‑climb with tiny mass transfers for the remaining time.

All stages respect the 1000 s budget and keep the sequence non‑negative.
The final vector is normalised to sum = 1 and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – works for arbitrary sum
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes (used in the sweep)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return normalized a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)           # stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Return normalized a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (hard max version)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at index k (where the maximum is attained).
    For each i involved in the max sum (i+j=k) we get 2·a_j.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    np.add.at(g, i, 2.0 * a[j])
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (phase 2)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact L∞ objective."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    step0 = 0.45
    it = 0
    last_improve = start_time

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        # decreasing step size
        step = step0 / (it ** 0.55)

        cand = a - step * g
        cand = _project_simplex(cand)
        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.3e} "
                  f"score={cand_score:.6f}")
        else:
            # occasional jitter to escape plateaus
            if it % 1200 == 0:
                eps = rng.random(a.shape) * a * 8e-5
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

        # restart from best with jitter if stuck too long
        if time.time() - last_improve > 30.0 and it % 2000 == 0:
            eps = rng.random(best.shape) * best * 2e-4
            a = np.maximum(best + eps, 0.0)
            a = _project_simplex(a)
            last_improve = time.time()

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑exponential‑gradient phase (new – phase 3)
# ----------------------------------------------------------------------


def _smooth_expgrad_phase(a0: np.ndarray,
                          L_conv: int,
                          L_corr: int,
                          deadline: float,
                          start_time: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation to the L∞ objective:
        f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)),
    using multiplicative updates a_i ← a_i·exp(‑η·corr_i) where
    corr_i = Σ_j w_{j+i}·a_j with w the soft‑max weights.
    α is increased in stages; η decays within each stage.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_conv)
    n = a.shape[0]

    # schedule of smoothing parameters (α) – start very smooth, finish sharp
    alphas = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    for alpha in alphas:
        it = 0
        # each α gets its own time slice; we break when deadline is near
        while time.time() < deadline:
            it += 1
            # --- compute self‑convolution -------------------------------------------------
            conv = _conv_self_fft(a, L_conv)                # length 2n‑1

            # --- soft‑max weights w -------------------------------------------------------
            max_c = conv.max()
            # subtract max to avoid overflow
            w = np.exp(alpha * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w = w / w_sum                               # shape (2n‑1,)

            # --- correlation w⋆a (linear, not circular) ---------------------------------
            # correlation (w ⋆ a)[k] = Σ_j w_{j+k}·a_j, k=0..n‑1.
            # Using convolution of w with reversed a:
            a_rev = a[::-1]                                 # length n
            # length needed for linear convolution
            total_len = w.shape[0] + a.shape[0] - 1        # = 3n‑2
            # FFT length (power of two) – supplied from outside
            W = np.fft.rfft(w, n=L_corr)
            Ar = np.fft.rfft(a_rev, n=L_corr)
            conv_full = np.fft.irfft(W * Ar, n=L_corr)[:total_len]
            start = n - 1
            corr = conv_full[start:start + n]               # length n, non‑negative

            # --- multiplicative update ---------------------------------------------------
            # decaying learning rate η
            eta0 = 0.8
            eta = eta0 / (1.0 + 1e-4 * it)

            mult = np.exp(-eta * corr)                      # keep positivity
            a = a * mult

            # --- project back onto the simplex -----------------------------------------
            total = a.sum()
            if total <= 0.0 or not np.isfinite(total):
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / total

            # occasional jitter to avoid stagnation
            if it % 2500 == 0:
                eps = rng.random(a.shape) * a * 5e-5
                a = np.maximum(a + eps, 0.0)
                a = a / a.sum()

            # evaluate exact objective (no smoothing)
            sc = _eval_fast(a, L_conv)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                print(f"[{time.time() - start_time:.2f}s] smooth_expgrad α={alpha:.2f} "
                      f"it={it} score={sc:.6f}")

            # safety break for each α (max 5000 iters per α)
            if it >= 5000:
                break

        # after finishing an α‑stage, re‑initialise a with a small jitter around the best
        eps = rng.random(best.shape) * best * 1e-4
        a = np.maximum(best + eps, 0.0)
        a = a / a.sum()

        # stop early if time is exhausted
        if time.time() >= deadline:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (hard max version – optional fine‑tuning)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates based on the hard max sub‑gradient."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    it = 0
    eta0 = 0.8

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / total

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] expgrad it={it:,} score={score:.6f}")

        if it % 2500 == 0:
            eps = rng.random(a.shape) * a * 5e-5
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score


# ----------------------------------------------------------------------
# Final random hill‑climb (phase 4)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer hill climb with simple acceptance."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3          # up to 0.2 % transfer
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cand_score:.6f}")

        # occasional exploratory jump
        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand = cand / cand.sum()
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # safety margin

    # --------------------------------------------------------------
    # Problem size – 16384 offers a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 16384
    L_fft = _next_pow2(2 * n - 1)          # for self‑convolution
    L_corr = _next_pow2(3 * n - 2)         # for correlation (w ⋆ a)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter exists (height_sequence_1), use it.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (≈70 s)
    # --------------------------------------------------------------
    sweep_end = start_time + 70.0
    i_array = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_array + 1.0)

    # geometric × power family
    lam_grid = np.linspace(1e-05, 2.5e-03, 350)   # dense grid
    p_grid = np.linspace(-8.0, 5.0, 131)          # step ≈0.1

    for lam in lam_grid:
        if time.time() >= sweep_end:
            break
        base = -lam * i_array
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} p={p:.2f} "
                      f"score={sc:.6f}")

    # pure Gaussian family
    c_grid = np.logspace(-8.0, -5.0, 48)
    for c in c_grid:
        if time.time() >= sweep_end:
            break
        log_shape = -c * (i_array * i_array)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gaussian c={c:.2e} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent (≈35 % of budget)
    # --------------------------------------------------------------
    sub_end = start_time + 0.35 * budget_s
    if time.time() < sub_end:
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_end, start_time, rng)

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑exponential‑gradient phase (new – ≈30 % of budget)
    # --------------------------------------------------------------
    smooth_end = start_time + 0.65 * budget_s
    if time.time() < smooth_end:
        best_seq, best_score = _smooth_expgrad_phase(best_seq, L_fft, L_corr,
                                                    smooth_end, start_time, rng)

    # --------------------------------------------------------------
    # 4️⃣  Hard‑max exponentiated‑gradient fine‑tune (≈15 % of budget)
    # --------------------------------------------------------------
    exp_end = start_time + 0.80 * budget_s
    if time.time() < exp_end:
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_end, start_time, rng)

    # --------------------------------------------------------------
    # 5️⃣  Final hill‑climb (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_fft,
                                           deadline, start_time, rng)

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # optional report via the provided evaluate_sequence
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
