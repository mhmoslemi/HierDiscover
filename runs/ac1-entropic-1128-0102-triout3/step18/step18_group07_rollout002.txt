"""
Improved optimiser for 2·n·max(conv(a,a))/(sum(a)²).

The algorithm combines several complementary stages:

* **Stage 0 – initialisation** – use a global starter if provided, otherwise a uniform
  vector.

* **Stage 1 – random Dirichlet / classic windows** – a quick broad search to obtain a
  decent initial point.

* **Stage 2 – alternating projection** – flatten the Fourier magnitude while staying
  on the probability simplex (same idea as the reference, but with fewer restarts).

* **Stage 3 – long Adam optimisation** – minimise a smooth log‑sum‑exp approximation
  of the max‑convolution.  The temperature schedule is finer and the inner loop
  runs longer than before.

* **Stage 4 – exact sub‑gradient descent** – after the smooth phase we switch to an
  exact sub‑gradient of the true max‑convolution (weights = 1 on the active set).
  A diminishing step‑size (1/√t) together with projection onto the simplex drives the
  solution further down.

* **Stage 5 – final random hill‑climb** – a cheap mass‑transfer local search that
  polishes the last few digits.

The whole search respects a 1000 s budget; intermediate best scores are printed.
The returned list is a non‑negative vector (its sum is irrelevant – the objective is
scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used by helpers for progress messages)
# ----------------------------------------------------------------------
_start_time = None


def _elapsed():
    return time.time() - _start_time if _start_time else 0.0


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution via real FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (sum is forced to 1 → denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth approximation of max and its gradient (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses  grad = 2 * (a_rev ⋆ w)[offset]  (w are softmax weights).
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)   # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad

# Phase 1 – random Dirichlet draws + deterministic windows
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{_elapsed():.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a few classic windows (already normalised to the simplex)."""
    # Sine window
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))


# ----------------------------------------------------------------------
# Phase 2 – alternating projection (flatten Fourier magnitude)
# ----------------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, max_iter: int, deadline: float):
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break
        # FFT of zero‑padded a
        F = np.fft.rfft(a, n=L)

        # Desired flat magnitude (same total energy for all frequencies)
        energy = np.dot(a, a)                # sum a_i^2
        flat_mag = np.sqrt(energy / L)

        # Replace magnitude, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Inverse FFT and truncate
        a_full = np.fft.irfft(F_flat, n=L)
        a = a_full[:n]

        # Projection onto feasible set
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{_elapsed():.2f}s] alt‑proj iter {i:5d} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – long Adam optimiser on smooth max
# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray, score0: float,
                     beta_schedule, deadline: float):
    """
    Projected Adam optimiser on the smooth log‑sum‑exp approximation.
    The temperature β follows the supplied schedule; each β is kept for as long
    as time permits.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters (tuned for this problem)
    lr0 = 0.25
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_schedule:
        # keep a moderate number of iterations per β – controlled by time only
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # Adam update (step size adapts with β)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate scaling: higher β → smaller steps
            lr = lr0 / (1.0 + beta * 0.001)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{_elapsed():.2f}s] Adam β={beta:.0f} new best {best_score:.6f}")

            # accept improvement (or very occasional uphill move)
            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            # early stopping if step becomes tiny
            if np.linalg.norm(step) < 1e-12:
                break

    return best_a, best_score

# Phase 4 – exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, score0: float, deadline: float):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    At each iteration we form the active set {k | conv[k] ≈ max},
    compute the (sub)gradient 2*J_k a summed over the active set,
    and take a diminishing step.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    # Pre‑allocate a weight vector for the active set
    w = np.zeros(2 * n - 1, dtype=float)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        M = conv.max()
        # active mask: either exact max or within a tiny tolerance
        # tolerance is chosen relative to M to tolerate floating noise
        tol = 1e-12 * max(1.0, M)
        active = (conv >= M - tol)
        active_count = np.count_nonzero(active)
        if active_count == 0:
            break
        w.fill(0.0)
        w[active] = 1.0 / active_count   # average over active indices

        # gradient = 2 * (a_rev ⋆ w)[n-1 : n-1+n]
        rev_corr = _convolve_fft(a[::-1], w)
        grad = 2.0 * rev_corr[n - 1 : n - 1 + n]

        # step‑size 1/sqrt(t) – classic diminishing schedule
        step_size = 0.5 / np.sqrt(it)

        cand = a - step_size * grad
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{_elapsed():.2f}s] sub‑grad it {it:6d} new best {best_score:.6f}")

        # accept if improves; otherwise keep current a (no uphill moves)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score

        # occasional random jitter to escape plateaus (every ~5000 iters)
        if it % 5000 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.01, size=n)
            a = _project_simplex(np.maximum(a + jitter, 0.0))
            # recompute current score after jitter
            cur_score = _objective_score(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 5 – simple random hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer up to 5 % of source mass
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{_elapsed():.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

    return best_a, best_score
