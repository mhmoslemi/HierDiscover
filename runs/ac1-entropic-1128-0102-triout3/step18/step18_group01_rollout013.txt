"""
Propose a non‑negative step function (probability vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

1️⃣ Global parametric sweep (geometric·power, Gaussian, and a
   symmetric Beta‑like family a_i ∝ (i+1)^α·(n−i)^β).  The best
   candidate becomes the start point for the next phase.

2️⃣ Smoothed “max‑convolution’’ descent.
   * We minimise the Lp‑norm of the self‑convolution with p gradually
     increasing from 2 up to ≈64.
   * The gradient of (∑c_k^p)^{1/p} w.r.t. a is
        g = 2·corr(a, w)   with   w_k ∝ c_k^{p‑1}.
     It is computed by a single FFT‑based correlation.
   * Updates follow the Adam optimiser and the iterate is projected
     onto the probability simplex each step.

3️⃣ Simulated‑annealing hill‑climb.
   Tiny mass‑transfer moves are accepted with the usual Metropolis
   rule, allowing occasional uphill steps.  Temperature decays
   linearly until the deadline.

All phases respect the total time budget (default 1000 s).  The final
vector is normalised to sum = 1 and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full linear self‑convolution a*a via FFT (length 2·len(a)‑1).
    Returns a real‑valued array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – works for arbitrary sum
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)           # stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Return a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Return a_i ∝ (i+1)^α · (n−i)^β (Beta‑like family)."""
    i = np.arange(n, dtype=np.float64)
    # log‑space for numerical stability
    log_shape = alpha * np.log(i + 1.0) + beta * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smoothed gradient of the Lp‑norm of the self‑convolution
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     p: float,
                     L_corr: int) -> np.ndarray:
    """
    Gradient of (∑ conv_k^p)^{1/p} w.r.t. a.
    Uses w_k ∝ conv_k^{p‑1} and a correlation via FFT.
    Returns a vector of length len(a).
    """
    # protect against exact zeros
    conv_pos = conv + 1e-30
    w = conv_pos ** (p - 1.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # correlation a ⋆ w (no flip of a, flip of w)
    rev_w = w[::-1]
    Af = np.fft.rfft(a, n=L_corr)
    Wf = np.fft.rfft(rev_w, n=L_corr)
    corr = np.fft.irfft(Af * Wf, n=L_corr)
    n = a.shape[0]
    # slice the part that corresponds to indices 0 … n‑1
    grad = 2.0 * corr[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def _adam_update(a: np.ndarray,
                 grad: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float = 0.12,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One Adam step followed by Euclidean projection onto the simplex."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = _project_simplex(a_new)
    return a_new, m, v


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _simulated_hill_climb(a0: np.ndarray,
                          L_fft: int,
                          start: float,
                          end: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves with Metropolis acceptance.
    Temperature decays linearly from T0 to ~0.
    """
    a = a0.copy()
    best = a.copy()
    cur_score = best_score = _eval_fast(best, L_fft)
    n = a.shape[0]
    T0 = 0.02  # initial temperature

    while time.time() < end:
        now = time.time()
        # linear cooling
        frac = (now - start) / (end - start)
        T = max(T0 * (1.0 - frac), 1e-9)

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = a[src] * rng.random() * 5e-3   # up to 0.5 % transfer
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand /= cand.sum()                    # renormalise

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[{now - start:.2f}s] SA new best {best_score:.6f}")

        # occasional larger jump to shake the system
        if rng.random() < 0.001:
            idx = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand2 = best.copy()
            cand2[idx] = max(cand2[idx] + jitter, 0.0)
            cand2 /= cand2.sum()
            cand2_score = _eval_fast(cand2, L_fft)
            if cand2_score < best_score:
                best_score = cand2_score
                best = cand2.copy()
                a = cand2
                cur_score = cand2_score
                print(f"[{now - start:.2f}s] SA random jitter {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.02  # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a compromise between quality and speed.
    # --------------------------------------------------------------
    n = 8192                    # power of two, moderate memory‑footprint
    L_self = _next_pow2(2 * n - 1)          # for self‑convolution
    L_corr = _next_pow2(3 * n - 2)          # for correlation (gradient)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Use a globally supplied starter if present (height_sequence_1).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_self)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Global parametric sweep (≈10 % of the budget).
    # --------------------------------------------------------------
    sweep_deadline = start_time + 0.10 * budget_s
    i = np.arange(n, dtype=np.float64)
    log_i = np.log(i + 1.0)
    log_ni = np.log(n - i)

    # geometric × power family
    lam_grid = np.linspace(1e-06, 4e-03, 300)
    p_grid = np.linspace(-8.0, 5.0, 121)
    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * i
        for p in p_grid:
            log_shape = base + p * log_i
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.2e} p={p:.2f} score={sc:.6f}")

    # pure Gaussian family
    c_grid = np.logspace(-8.0, -5.0, 44)
    for c in c_grid:
        if time.time() >= sweep_deadline:
            break
        log_shape = -c * (i * i)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_self)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gaussian c={c:.2e} score={sc:.6f}")

    # Beta‑like family (α,β) – 25 × 25 grid
    alpha_grid = np.linspace(-6.0, 6.0, 25)
    beta_grid = np.linspace(-6.0, 6.0, 25)
    for alpha in alpha_grid:
        if time.time() >= sweep_deadline:
            break
        for beta in beta_grid:
            log_shape = alpha * log_i + beta * log_ni
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] beta α={alpha:.2f} β={beta:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Smoothed Adam descent (≈70 % of the budget).
    # --------------------------------------------------------------
    descent_deadline = start_time + 0.80 * budget_s
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    # parameters for the p‑schedule
    p_max = 64.0
    while time.time() < descent_deadline:
        it += 1
        # ---------------- conv & p‑schedule ----------------
        conv = _conv_self_fft(a, L_self)               # length 2·n‑1
        p = min(p_max, 2.0 + it / 500.0)               # slowly increase p
        # ---------------- gradient ----------------
        grad = _smooth_gradient(a, conv, p, L_corr)
        # ---------------- Adam update ----------------
        a, m, v = _adam_update(a, grad, m, v, it, lr=0.12)
        # ---------------- evaluate ----------------
        cur_score = _eval_fast(a, L_self)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam it={it:,} p={p:.1f} score={cur_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if it % 3000 == 0:
            jitter = rng.random(a.shape) * a * 7e-5
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

        # restart if stagnation lasts > 25 s
        if it % 5000 == 0 and time.time() - start_time > 100.0:
            jitter = rng.random(a.shape) * a * 2e-4
            a = np.maximum(best_seq + jitter, 0.0)
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # 3️⃣ Simulated‑annealing hill‑climb (remaining time).
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_hill_climb(best_seq, L_self,
                                                    start_time, deadline, rng)

    # --------------------------------------------------------------
    # Final normalisation and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Report final bound using the provided evaluate_sequence (optional)
    try:
        from __main__ import evaluate_sequence
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    # Return a plain Python list of floats
    return [float(v) for v in best_seq]
