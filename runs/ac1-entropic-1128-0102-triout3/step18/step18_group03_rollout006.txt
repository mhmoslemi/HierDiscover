            beta = min(beta * 2, max_beta)
            iter_since_beta = 0
            print(f"[{time.time() - start:.2f}s] increase beta to {beta}")

    # Final polishing: use exponentiated gradient and random hill climbing

    total = best.sum()
    if total > 0:
        best /= total
    else:
        best = np.full(n, 1.0/n)

"""
Propose a high‑quality step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

* Initialise from the global seed ``height_sequence_1`` (if present) – a
  previously discovered good candidate – otherwise fall back to the uniform
  distribution.  The vector is normalised to sum 1.

* Run a long “Adam on a smooth‑max surrogate” phase.
  The surrogate is the soft‑max of the self‑convolution:
          fβ(a) = Σk w_k(β)·conv_k(a) ,   w_k ∝ exp(β·conv_k) .
  Its gradient can be written as a single convolution, which we evaluate with
  FFTs.  β starts at 0.5 and is doubled every *beta_change* iterations up to
  ``max_beta`` (≈ 8192).  The Adam update is performed with an additive step,
  followed by a cheap non‑negative‐renormalisation (clip + renorm).  Whenever a
  new vector improves the exact objective we keep it as the current global best.
  An occasional uphill move (Metropolis acceptance) helps escape plateaus.

* After the Adam phase a lightweight random hill‑climb is performed for the
  remaining time: a tiny amount of mass is transferred between two random
  positions; the move is kept if it improves the exact objective.

The whole routine respects the 1000 s budget, prints progress timestamps,
and finally returns the best vector found as a plain Python list (sum normalised
to 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights w_i ∝ exp(beta·conv_i)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax_new(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2 * Σ_j a_j·w_{i+j}.
    Implemented as a convolution of a_rev with w.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(w, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Fast exact objective (used for bookkeeping)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs for at most `budget_s` seconds and returns a
    plain Python list (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.05          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)   # length for self‑convolution FFTs

    # ------------------------------------------------------------------
    # Initialise a from the global starter if available
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)
        if a0.ndim != 1 or a0.size == 0:
            raise ValueError
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            raise ValueError
        a = a0 / a0.sum()
    except Exception:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = a / a.sum()

    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    print(f"[{time.time() - start:.2f}s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam on the smooth‑max surrogate
    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Schedule for the smooth‑max temperature β
    beta = 0.5

    iter_since_beta = 0

    # Reserve some seconds for the final hill‑climb

        grad = _grad_softmax_new(a, w, L_fft)

        prop = np.maximum(prop, 0.0)

        if prop_sum > 0.0:

    # ------------------------------------------------------------------
    best = np.maximum(best, 0.0)

    else:

    print(f"[{time.time() - start:.2f}s] finished – best score ≈ {best_score:.6f}")
