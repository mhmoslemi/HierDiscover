The code below searches for a non‑negative step‑function (a probability vector) that minimizes  

\[
\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}\; .
\]

The search consists of four phases that fit into the 1000 s budget.

1. **Broad parametric sweep** – evaluates three flexible families of
   distributions (geometric × power, pure Gaussian and symmetric Beta).  
   The best candidate from the sweep is kept.

2. **Smooth‑max gradient descent** – replaces the non‑smooth `max` by a
   temperature‑controlled soft‑max (log‑sum‑exp) and performs projected
   gradient descent on the resulting smooth objective.  
   The gradient is obtained analytically using FFT‑based convolutions,
   which makes each iteration cheap (≈ 0.003 s for n = 16384).

3. **Exponentiated‑gradient fine‑tuning** – a classic multiplicative update
   on the simplex, useful for polishing the solution after the smooth phase.

4. **Random hill‑climb** – tiny mass‑transfer moves with occasional
   larger jitter to escape tiny plateaus.

All phases respect the time budget, print progress information and finally
return a plain Python list of floats (the sequence is normalised to sum = 1,
so the evaluation function’s internal clipping never harms the result).

```python
"""
Step‑function optimisation for the L∞ self‑convolution bound.

Phases
-------
1. Parametric sweep over three families:
   * (i+1)^p·exp(-λ·i)                     – geometric × power
   * exp(-c·i²)                             – pure Gaussian
   * x^{α‑1}(1‑x)^{α‑1} on [0,1] (Beta)      – symmetric Beta (U‑shaped to bell)

2. Smooth‑max gradient descent.
   The hard max is replaced by a soft‑max (log‑sum‑exp) with a temperature β.
   The gradient of the smoothed objective is computed analytically:
       ∇ = (2n / s²)·g₁ – (4n·⟨w,y⟩ / s³)·a,
   where g₁ is a convolution of the soft‑max weights w with a reversed.
   Each step is projected onto the probability simplex.

3. Exponentiated‑gradient (multiplicative) updates – annealed learning rate.

4. Final random hill‑climb with tiny mass transfers.

All operations use FFT‑based convolutions for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of x and y using FFT (output length x+y‑1)."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: x.shape[0] + y.shape[0] - 1]


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    return _conv_fft(a, a, L)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast, works for arbitrary sum)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Normalized a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Normalized a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """
    Normalized a_i ∝ x^{α‑1} (1‑x)^{α‑1},   x = i/(n‑1).
    α>0 (α≈0.5 gives the arcsine shape, α>1 gives a bell).
    """
    eps = 1e-15
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    # log‑density, kept stable by subtraction of the maximum later
    log_shape = (alpha - 1.0) * (np.log(x + eps) + np.log(1.0 - x + eps))
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (Phase 2)
# ----------------------------------------------------------------------


def _smooth_max_gradient_descent(a0: np.ndarray,
                                 L_self: int,
                                 L_full: int,
                                 deadline: float,
                                 start_time: float,
                                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a soft‑max approximation of max(conv).
    The temperature β is increased gradually, turning the smooth problem
    into the original non‑smooth one.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _eval_fast(best, L_self)

    it = 0
    beta = 5.0               # start with a gentle smoothing
    beta_max = 200.0
    step0 = 0.30             # initial step size
    last_improve = time.time()

    while time.time() < deadline:
        it += 1

        # ---- 1) current self‑convolution
        conv = _conv_self_fft(a, L_self)          # length 2n‑1
        y = conv                                   # alias for readability

        # ---- 2) soft‑max weights w = exp(β·y) / Σexp(β·y)
        zb = beta * y
        max_zb = zb.max()
        exp_w = np.exp(zb - max_zb)               # stability shift
        w = exp_w / exp_w.sum()

        # ---- 3) weighted average ⟨w, y⟩  (smooth approximation of max)
        wy = np.dot(w, y)

        # ---- 4) gradient of Σ w_k y_k  (see derivation in the analysis)
        #   g1 = 2 * conv(w, a_rev)[n‑1 : 2n‑1]
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev, L_full)               # length 3n‑2
        g1 = 2.0 * conv_w_rev[n - 1: 2 * n - 1]

        s = a.sum()
        # full gradient of the smooth objective
        grad = (2.0 * n / (s * s)) * g1 - (4.0 * n * wy / (s * s * s)) * a

        # ---- 5) step size (simple 1/√t decay)
        step = step0 / math.sqrt(it)

        cand = a - step * grad
        cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_self)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] smooth‑gd it={it:,} "
                  f"β={beta:.1f} step={step:.3e} score={cand_score:.6f}")
        else:
            # occasional jitter to avoid stagnation
            if it % 700 == 0:
                eps = rng.random(a.shape) * a * 4e-5
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

        # increase temperature slowly (more faithful to the true max)
        beta = min(beta_max, beta * 1.005)

        # restart from best + jitter if we have been idle for a while
        if time.time() - last_improve > 25.0 and it % 1500 == 0:
            eps = rng.random(best.shape) * best * 2e-4
            a = _project_simplex(np.maximum(best + eps, 0.0))
            last_improve = time.time()

    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (Phase 3)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_self: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates on the simplex with annealed learning rate."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_self)
    it = 0
    eta0 = 0.85

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_self)
        k = int(conv.argmax())
        # sub‑gradient of max at the active index
        g = np.zeros_like(a)
        i_start = max(0, k - (a.shape[0] - 1))
        i_end = min(k, a.shape[0] - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        np.add.at(g, i, 2.0 * a[j])

        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / total

        score = _eval_fast(a, L_self)
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] expgrad it={it:,} score={score:.6f}")

        if it % 2600 == 0:
            eps = rng.random(a.shape) * a * 4e-5
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score


# ----------------------------------------------------------------------
# Final random hill‑climb (Phase 4)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_self: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with a simple acceptance rule."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_self)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_self)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cand_score:.6f}")

        # occasional larger exploratory jump
        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand = cand / cand.sum()
            cand_score = _eval_fast(cand, L_self)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # small safety margin

    # --------------------------------------------------------------
    # Problem size – 16384 gives a good trade‑off between resolution and speed.
    # --------------------------------------------------------------
    n = 16384
    L_self = _next_pow2(2 * n - 1)          # for a*a
    L_full = _next_pow2(3 * n - 2)          # for w * a_rev in the smooth phase

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global starter (if the platform provided one).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_self)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (≈150 s)
    # --------------------------------------------------------------
    sweep_end = start_time + 150.0
    i_arr = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_arr + 1.0)

    # a) geometric × power
    lam_grid = np.linspace(1e-05, 0.0100, 400)    # extended range
    p_grid = np.linspace(-12.0, 6.0, 193)        # finer steps
    for lam in lam_grid:
        if time.time() >= sweep_end:
            break
        base = -lam * i_arr
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep gp λ={lam:.6e} p={p:.2f} score={sc:.6f}")

    # b) pure Gaussian
    c_grid = np.logspace(-8.0, -4.0, 60)   # denser range
    for c in c_grid:
        if time.time() >= sweep_end:
            break
        log_shape = -c * (i_arr * i_arr)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_self)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] sweep gauss c={c:.2e} score={sc:.6f}")

    # c) symmetric Beta family
    alpha_grid = np.concatenate((
        np.linspace(0.06, 0.4, 12),    # strongly U‑shaped
        np.linspace(0.4, 1.5, 15),    # arcsine‑like to mild bell
        np.linspace(1.5, 5.0, 15)     # increasingly bell‑shaped
    ))
    for alpha in alpha_grid:
        if time.time() >= sweep_end:
            break
        cand = _make_beta_symmetric(n, alpha)
        sc = _eval_fast(cand, L_self)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] sweep beta α={alpha:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Smooth‑max gradient descent (≈400 s)
    # --------------------------------------------------------------
    gd_end = start_time + 0.55 * budget_s   # ~550 s total for gd & later phases
    if time.time() < gd_end:
        best_seq, best_score = _smooth_max_gradient_descent(
            best_seq, L_self, L_full, gd_end, start_time, rng)

    # --------------------------------------------------------------
    # 3️⃣ Exponentiated‑gradient phase (≈250 s)
    # --------------------------------------------------------------
    exp_end = start_time + 0.80 * budget_s
    if time.time() < exp_end:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_self, exp_end, start_time, rng)

    # --------------------------------------------------------------
    # 4️⃣ Final hill‑climb (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(
            best_seq, L_self, deadline, start_time, rng)

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Report final bound using the provided evaluate_sequence (optional)
    try:
        from __main__ import evaluate_sequence
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
```
