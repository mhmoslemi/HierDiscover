def anneal(a, score, deadline):
    n = a.shape[0]; rng = np.random.default_rng()
    cur_score = score
    cur_a = a.copy()
    best_local = a.copy()
    best_local_score = score
    T0 = max(cur_score - 1.0, 0.5) # initial temperature, guess? maybe (cur_score - 1)/10.
    T_end = 1e-6
    schedule_len = 200000 (or based on remaining time)
    start = time.time()
    iteration = 0
    while time.time() < deadline:
        # compute temperature
        # linear schedule in log scale
        t = (time.time() - start) / (deadline - start)
        T = T0 * (1 - t) + T_end * t   # linear decreasing or exponential
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or cur_a[i] == 0:
            continue
        delta = cur_a[i] * rng.random() * 0.5  # up to 0.5 of source
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # Already sum remains 1. If negative due to rounding, skip.
        # No need for projection because sum unaffected and positivity maintained (ensured by delta <= cur_a[i]).
        cand_score = eval_obj(cand)
        if cand_score < best_local_score:
            best_local_score = cand_score
            best_local = cand.copy()
        # acceptance
        if cand_score < cur_score or np.exp(-(cand_score - cur_score) / T) > rng.random():
            cur_a = cand
            cur_score = cand_score
        iteration += 1
    return best_local, best_local_score

"""
Proposer – a multi‑stage optimizer for the “max self‑convolution” objective.

The objective for a non‑negative vector a (∑a = 1) is
    J(a) = 2·n·max_k (a * a)[k]          (n = len(a))

Goal: find a with J(a) as low as possible (target < 1.503 for n≈4k).

The search consists of the following stages (time‑aware):
1. Random Dirichlet draws and random “constant‑magnitude FFT” seeds.
2. Alternating projection that forces a flat Fourier magnitude while staying
   on the probability simplex.
3. Long Adam optimisation on a smooth (log‑sum‑exp) surrogate of the max.
4. Projected gradient descent on the exact max (steepest‑ascent removal).
5. Simulated‑annealing mass‑transfer hill‑climbing for final polishing.

All stages respect a global 1000 s budget, keep the best sequence encountered,
and return it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv.real  # discard tiny imag part caused by rounding


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    (v ≥ 0, sum(v) = 1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (a already sums to 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max w.r.t. a (see description in the docstring)."""
    n = a.shape[0]
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()
    # rev_corr = a_rev ☆ w   (full convolution, length 3n‑2)
    rev_corr = np.fft.irfft(
        np.fft.rfft(a[::-1], n=_next_pow2(3 * n - 2))
        * np.fft.rfft(w, n=_next_pow2(3 * n - 2)),
        n=_next_pow2(3 * n - 2),
    )
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Phase‑1 helpers – random draws
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration parameter log‑uniform on [10⁻⁴, 10⁰]
        conc = 10 ** rng.uniform(-4.0, 0.0)
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={conc:.4g} → {sc:.6f}")
    return best_seq, best_score


def _random_constant_mag_seeds(n: int, trials: int, deadline: float):
    """Generate vectors whose FFT has (approximately) constant magnitude."""
    rng = np.random.default_rng()
    L = _next_pow2(2 * n - 1)
    best_seq = None
    best_score = float("inf")
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # random phases for the real FFT (zero‑frequency stays real)
        phases = rng.uniform(0.0, 2.0 * np.pi, size=L // 2 + 1)
        # keep the DC and Nyquist components real (phase = 0)
        phases[0] = 0.0
        if L % 2 == 0:
            phases[-1] = 0.0
        F = np.exp(1j * phases)                # constant magnitude = 1
        a_full = np.fft.irfft(F, n=L)[:n]
        a = np.maximum(a_full, 0.0)
        a = _project_simplex(a)
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] const‑mag seed → {sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a few classic window functions (already normalised)."""
    # sine
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # a few Gaussian windows
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 30.0, n / 3.0, 4):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

# Phase‑2 – alternating projection (flat Fourier magnitude)
# ----------------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, max_iter: int, deadline: float):
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break
        # FFT of padded a
        F = np.fft.rfft(a, n=L)

        # Current total energy (sum of squares)
        energy = np.dot(a, a)
        # Desired flat magnitude
        flat_mag = np.sqrt(energy / L)

        # Replace magnitude, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Inverse FFT
        a_full = np.fft.irfft(F_flat, n=L)[:n]

        # Project onto feasible set
        a = np.maximum(a_full, 0.0)
        a = _project_simplex(a)

        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] alt‑proj iter {i:5d} → {sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 – Adam on smooth max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, score0: float,
                  beta_vals, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # hyper‑parameters
    lr = 0.09
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam β={beta:5.1f} → {best_score:.6f}")

            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            if np.linalg.norm(step) < 1e-13:
                break
    return best_a, best_score

# Phase‑4 – projected gradient descent on the exact max
# ----------------------------------------------------------------------
def _max_grad_descent(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    step_factor = 0.5               # initial step size (relative to gradient norm)

    while time.time() < deadline:
        conv = _conv_self(a)
        k = int(np.argmax(conv))

        # gradient of the *single* maximal entry
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        norm = np.linalg.norm(grad)
        if norm < 1e-15:
            break
        step = (step_factor * grad) / norm
        cand = a - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] max‑grad new best {best_score:.6f}")

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score

        # slowly shrink the step factor
        step_factor = max(0.01, step_factor * 0.9995)

    return best_a, best_score

def _simulated_annealing(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    # temperature schedule – start relatively hot, cool to near‑zero
    T0 = max(best_score - 1.0, 0.5)
    T_end = 1e-7
    start = time.time()

        t = (time.time() - start) / (deadline - start)

        i = rng.integers(n)

        if i == j or a[i] == 0.0:

            a = cand

            jitter = rng.normal(scale=0.008, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0_score = _objective_score(a0)

            a_opt, sc_opt = _adam_descent(a0, a0_score,
                                          beta_vals=beta_schedule,
                                          deadline=adam_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] Adam restart {r} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected gradient descent on the exact max (≈10 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        grad_deadline = time.time() + 0.10 * remaining
        a_opt, sc_opt = _max_grad_descent(best_seq, best_score, grad_deadline)
        if sc_opt < best_score:
            best_seq, best_score = a_opt, sc_opt
            print(f"[{time.time() - start_time:.2f}s] max‑grad final → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated‑annealing hill‑climb (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting SA polishing")
        best_seq, best_score = _simulated_annealing(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise and return (guard against accidental zero sum)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best J ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
