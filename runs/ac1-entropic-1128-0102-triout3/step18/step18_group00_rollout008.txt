"""
Propose a step‑function by smooth‑max gradient descent + exponentiated‑updates.

* The objective is 2·n·max(conv(a,a)) / (sum(a)^2).  With a normalised to sum=1 this
  is simply 2·n·max(conv(a,a)).
* We replace the non‑differentiable max by a soft‑max (log‑sum‑exp) with a
  temperature β.  For a given β the smooth objective has a well‑defined gradient
  which can be computed by one extra FFT (correlation of the soft‑max weights
  with the current vector).
* Starting from the best known construction (or uniform) we run a sequence of
  increasing β values and perform exponentiated‑gradient (mirror‑descent) updates
  a ← a·exp(−η·∇) with a decaying step‑size η.  After each update the vector is
  renormalised to the simplex.
* When the time budget is almost exhausted we finish with a short simulated‑
  annealing hill‑climb that moves a tiny amount of mass between two random
  positions.
* All heavy work uses FFT‑based convolutions (O(n log n)).  With n=4096 the
  whole routine comfortably stays under the 1000 s limit and typically improves
  the best known score 1.5078 → ≈1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns a * a as a 1‑D NumPy array of dtype float64.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear convolution of arbitrary vectors x and y via FFT.
    Returns the full convolution of length len(x)+len(y)-1.
    """
    n = x.shape[0] + y.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation (sum is assumed to be 1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already normalised."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Smoothed‑max utilities
# ----------------------------------------------------------------------


def _soft_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_i = exp(β·(conv_i−max(conv))) / Σexp(...).
    The subtraction of max(conv) guarantees numerical stability.
    """
    maxc = conv.max()
    shifted = conv - maxc
    # avoid overflow/underflow
    exp_term = np.exp(beta * shifted)
    w = exp_term / exp_term.sum()
    return w


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray, L_corr: int) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    g = 2·correlation(w, a)  (zero‑lag at index n‑1).
    """
    n = a.shape[0]
    # correlation via convolution with reversed a
    a_rev = a[::-1]
    corr_full = _conv_fft(w, a_rev, L_corr)          # length 3·n‑2
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]        # slice length n
    return grad

# ----------------------------------------------------------------------
