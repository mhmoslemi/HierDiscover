                    print(f"[{time.time() - start_time:.2f}s] random tiny move src={src} dst={dst} delta={delta:.2e} score={new_score:.6f}")
        return best_seq, best_score

    if time.time() < deadline - 50:
        best_seq, best_score = _targeted_hill_climb(best_seq, L_fft,
                                                    deadline - 50, start,
                                                    best_seq, best_score, rng)

    # Phase 5: final random hill climb (tiny moves)
    while time.time() < deadline:

            continue
        dst = rng.integers(n)

            print(f"[{time.time() - start:.2f}s] final random tiny move score={new_score:.6f}")

    total = best_seq.sum()
    if total > 0:

    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
search for a non‑negative step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm builds a large diverse pool of seed vectors (geometric‑power
shapes, low‑α Dirichlet draws, random low‑frequency Fourier‐based shapes,
and the arcsine family), keeps the best few, and refines them with several
phases:

1. Adam optimisation on a smooth‑max surrogate (log‑sum‑exp of the
   convolution). The temperature β is increased gradually up to 4096.
2. Multiplicative (exponential) gradient polishing.
3. Projected sub‑gradient descent on the true L‑∞ objective.
4. Targeted hill‑climbing that moves a tiny amount of mass from indices
   that contribute most to the current maximal convolution term towards
   indices that contribute least (using the exact sub‑gradient).
5. A final random tiny‑mass hill‑climb for any remaining improvement.

All operations are performed with FFT‑based convolutions for speed and the
candidate with the lowest objective found within the time budget is
returned as a plain Python list (normalised to sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2 * Σ_j a_j p_{i+j}.
    Implemented as a convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast objective (exact, used for speed inside the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_general_shape(n: int, r: float = 0.9, p: float = 0.0, q: float = 0.0) -> np.ndarray:
    """
    Shape a_i ∝ r**i · (i+1)**p · (n-i+1)**q, normalised to sum = 1.
    r∈(0,1] gives a geometric decay,
    p controls a left‑hand power,
    q controls a right‑hand power.
    """
    i = np.arange(n, dtype=np.float64)
    # avoid log(0) – the smallest argument is 1
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)                # prevent overflow
    shape = np.exp(log_shape)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient helpers (used in polishing phases)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (the true non‑smooth gradient)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative (exponential) updates on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the L∞ objective with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb using exact sub‑gradient information
# ----------------------------------------------------------------------


def _targeted_hill_climb(a: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy moves that transfer a tiny amount of mass from indices that
    contribute most to the current maximal convolution entry towards indices
    that contribute least.
    """
    n = a.shape[0]
    while time.time() < deadline:
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)                    # shape (n,)
        # contribution of each i to the maximal entry
        contrib = a * (g * 0.5)                        # because g_i = 2*a_{k-i}
        # pick few source and destination candidates
        src_idx = np.argpartition(-contrib, min(5, n - 1))[:5]
        dst_idx = np.argpartition(contrib, min(5, n - 1))[:5]
        improved = False
        for src in src_idx:
            if a[src] <= 1e-14:
                continue
            for dst in dst_idx:
                if src == dst:
                    continue
                delta = a[src] * rng.uniform(1e-5, 2e-3)   # ≤0.2 % of source mass
                a_new = a.copy()
                a_new[src] -= delta
                a_new[dst] += delta
                new_score = _eval_fast(a_new, L_fft)
                if new_score < best_score:
                    a = a_new
                    best_seq = a_new.copy()
                    best_score = new_score
                    improved = True
                    print(f"[{time.time() - start_time:.2f}s] target move src={src} dst={dst} δ={delta:.2e} score={new_score:.6f}")
                    break
            if improved:
                break
        if not improved:
            # fallback tiny random move
            src = rng.integers(n)
            if a[src] < 1e-14:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            delta = a[src] * rng.uniform(1e-6, 5e-4)       # ≤0.05 %
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            new_score = _eval_fast(a_new, L_fft)
            if new_score < best_score:
                a = a_new
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time() - start_time:.2f}s] random tiny move src={src} dst={dst} δ={delta:.2e} score={new_score:.6f}")
    return best_seq, best_score

        seeds.append(_mix_with_uniform(arc_shape, lam))

    # Low‑α Dirichlet draws (very sparse vectors).
    for alpha in [0.01, 0.02, 0.05, 0.1]:
        for _ in range(8):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # Random low‑frequency Fourier‑based shapes.
    def _random_lowfreq_shape(num_freq=6):
        x = np.arange(n, dtype=np.float64)
        a = np.zeros(n, dtype=np.float64)
        for _ in range(num_freq):
            freq = rng.integers(1, n // 2)
            phase = rng.random() * 2 * np.pi
            amp = rng.random()
            a += amp * np.cos(2 * np.pi * freq * x / n + phase)
        a -= a.min()                     # make non‑negative
        a += 1e-12
        a /= a.sum()
        return a

    for _ in range(50):
        seeds.append(_random_lowfreq_shape(num_freq=6))

    # --------------------------------------------------------------
    # Keep the K best seeds after a cheap evaluation.
    # --------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 12
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam on the smooth‑max surrogate (β grows gradually).
    # --------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128,
                     256, 512, 1024, 2048, 4096]
    adam_lr = 0.70
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    phase1_deadline = deadline - 300   # keep ~5 min for later polishing
    for beta in beta_schedule:
        if time.time() >= phase1_deadline:
            break
        max_iter_per_vec = 3000
        for idx in range(len(pool)):
            a = pool[idx]
            m = np.zeros_like(a)
            v_vec = np.zeros_like(a)
            cur_score = pool_scores[idx]
            t = 0
            while time.time() < phase1_deadline:
                t += 1
                conv = _conv_self_fft(a, L_fft)
                p_weights = _softmax_weights(conv, beta)
                g = _grad_softmax_new(a, p_weights, L_grad)

                # Adam update (mirrored descent)
                m = beta1 * m + (1.0 - beta1) * g
                v_vec = beta2 * v_vec + (1.0 - beta2) * (g * g)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v_vec / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)

                a_candidate = a - step
                a_candidate = _project_simplex(a_candidate)

                cand_score = _eval_fast(a_candidate, L_fft)

                # Accept improvements outright; occasional uphill moves help escape plateaus.
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    prob = np.exp(-(cand_score - cur_score) * 3.0 / (np.sqrt(t) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score

                # Update global best and pool entry.
                if cand_score < pool_scores[idx]:
                    pool_scores[idx] = cand_score
                    pool[idx] = a_candidate.copy()
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start:.2f}s] Adam beta={beta} it={t} score={cand_score:.6f}")

                if t >= max_iter_per_vec:
                    break

    # --------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing.
    # --------------------------------------------------------------
    if time.time() < deadline - 200:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 200, start,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent (classic polishing).
    # --------------------------------------------------------------
    if time.time() < deadline - 150:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 150, start,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 5️⃣  Targeted hill‑climb using exact sub‑gradient information.
    # --------------------------------------------------------------
    if time.time() < deadline - 50:
        best_seq, best_score = _targeted_hill_climb(
            best_seq, L_fft, deadline - 50, start,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 6️⃣  Final random hill‑climb (tiny mass transfers).
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = best_seq[src] * rng.random() * 1e-3   # ≤0.1 % of source mass
        prop = best_seq.copy()
        prop[src] -= delta
        prop[dst] += delta
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final random tiny move score={new_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
