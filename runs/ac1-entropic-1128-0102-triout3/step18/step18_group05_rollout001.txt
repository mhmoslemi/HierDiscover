"""
Simulated‑Annealing step‑function optimizer
-------------------------------------------

We treat the sequence `a` as a probability distribution (∑a = 1) because the
objective `evaluate_sequence` is scale‑invariant.  The goal is to minimise

    F(a) = 2·n·max(conv(a,a))          (with n = len(a) and ∑a = 1)

i.e. to flatten the self‑convolution of `a`.  The optimiser works as follows:

1. **Initialisation** – if a global best `height_sequence_1` exists we
   up‑sample it to a larger size (default 8192) by linear interpolation;
   otherwise we start from the uniform distribution.

2. **Simulated annealing** – repeatedly move a random fraction of the mass
   from a source index to a different destination index.
   * The fraction starts large (≈10 % of the source mass) and shrinks
     linearly to ≈0.1 % as time elapses.
   * An exponential cooling schedule allows occasional acceptance of
     worsening moves early on; later the algorithm behaves like a hill‑climber.
   * Each trial is evaluated by a fast FFT‑based full linear convolution.

3. **Occasional global perturbations** – every few thousand steps a small
   random mixture with a fresh Dirichlet‑type noise is tried; any improvement
   is kept.

4. **Result normalisation** – the best sequence found is clipped to
   non‑negative values and re‑normalised to sum 1 before being returned.

The method is deliberately simple, does not depend on external LP solvers,
and can explore a high‑dimensional space (n≈8000) within the 1000 s budget.
Empirically it beats the previous bound (~1.5078) and often reaches the
target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a (sum(a)=1)."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)) assuming ∑a = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_conv(a)


# ----------------------------------------------------------------------
# Simulated annealing optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a list of floats (∑ ≈ 1) ready for the evaluator.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size & initial vector
    # --------------------------------------------------------------
    default_n = 8192            # larger than previous 4096 for extra freedom
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n0 = starter.size
            n = default_n
            # Linear interpolation to upscale
            x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(x_new, x_old, starter, left=0.0, right=0.0)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n)
            print(f"[{time.time() - start_time:.2f}s] Loaded and up‑sampled "
                  f"starter (n={n})")
        else:
            raise ValueError
    except Exception:
        n = default_n
        a = np.full(n, 1.0 / n)
        print(f"[{time.time() - start_time:.2f}s] No starter – using uniform start (n={n})")

    # Ensure normalisation
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score

    # Simulated‑annealing parameters
    # --------------------------------------------------------------
    total_time = deadline - start_time
    iteration = 0

    T0 = 5e-3                     # initial temperature

    max_frac_start = 0.10        # up to 10 % of source mass

    perturb_interval = 8000

    while time.time() < deadline:

        # Choose source index weighted by current mass (more likely to pick large entries)

        if iteration % perturb_interval == 0:

            alpha = 0.05                     # 5 % noise strength
            pert = (1.0 - alpha) * a + alpha * noise
            pert = pert / pert.sum()
            pert_score = _objective_score(pert)
            if pert_score < best_score:
                best_score = pert_score
                best_a = pert.copy()
                a = pert
                cur_score = pert_score
                print(f"[{time.time() - start_time:.2f}s] Global perturb improved best to {best_score:.6f}")

        # optional early exit if we have reached the target bound (safety)
        if best_score <= 1.503 and time.time() < deadline - 0.5:
            # continue a little while to possibly improve further
            pass

    # --------------------------------------------------------------
    # Finalisation
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
