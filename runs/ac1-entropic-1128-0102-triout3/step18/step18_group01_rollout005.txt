"""
propose_candidate – deterministic‑annealing optimisation of the
step‑function bound

The search proceeds in five stages (all respecting a 1000 s wall‑clock
budget):

1️⃣  A fine parametric sweep over two families of monotone shapes:
    (i+1)^p·exp(‑λ·i)   and   exp(‑c·i²).  The best vector from this sweep
    seeds the subsequent optimisation.

2️⃣  Projected sub‑gradient descent on the exact L∞ objective (the same
    routine as in the reference implementation).

3️⃣  Classical exponentiated‑gradient descent on the max‑convolution
    sub‑gradient.

4️⃣  **Deterministic‑annealing phase** – a smooth approximation of the
    max‑convolution is obtained by a soft‑max (temperature β).  The
    resulting weighted gradient is applied with an exponentiated‑gradient
    update while β is slowly increased (annealing).  This phase refines
    the distribution far beyond the plain sub‑gradient approach.

5️⃣  Final random hill‑climb for the last remaining seconds.

The algorithm works on a probability vector (sum = 1) of length
n = 16384, uses FFT‑based convolutions for O(n log n) speed and never
produces negative entries.  The best sequence found is returned as a
plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – works for arbitrary sum
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return normalized a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)           # stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Return normalized a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used by classic phases)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the index where the maximum is attained.
    For each i involved in the max sum (i+j=k) we get 2·a_j.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    np.add.at(g, i, 2.0 * a[j])
    return g


# ----------------------------------------------------------------------
# Classical projected sub‑gradient descent (phase 2)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    step0 = 0.45
    it = 0
    last_improve = start_time

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        step = step0 / (it ** 0.55)
        cand = a - step * g
        cand = _project_simplex(cand)
        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.3e} score={cand_score:.6f}")
        else:
            if it % 1200 == 0:
                eps = rng.random(a.shape) * a * 8e-5
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

        if time.time() - last_improve > 30.0 and it % 2000 == 0:
            eps = rng.random(best.shape) * best * 2e-4
            a = np.maximum(best + eps, 0.0)
            a = _project_simplex(a)
            last_improve = time.time()

    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (phase 3)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    it = 0
    eta0 = 0.8

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        eta = eta0 / (1.0 + 0.001 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / total

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] expgrad it={it:,} "
                  f"score={score:.6f}")

        if it % 2500 == 0:
            eps = rng.random(a.shape) * a * 5e-5
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score


# ----------------------------------------------------------------------
# Deterministic‑annealing smooth‑gradient phase (new – phase 4)
# ----------------------------------------------------------------------


def _smooth_expgrad_phase(a0: np.ndarray,
                          L_fft: int,
                          deadline: float,
                          start_time: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Smooth approximation of the max‑convolution via a soft‑max with
    temperature β (annealed).  The weighted gradient is applied with an
    exponentiated‑gradient update while β grows from a small value to a
    large one, allowing the optimiser to escape plateaus and then focus
    on the true max‑objective.
    """
    n = a0.shape[0]
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)

    it = 0
    beta = 0.5               # start temperature (softening)
    beta_max = 400.0
    beta_inc = (beta_max - beta) / 2500.0  # linear schedule

    eta0 = 0.6               # initial learning rate for exponentiated update

    # FFT size for convolution w * a_rev (length 3n‑2)
    L2 = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1

        # current objective
        conv = _conv_self_fft(a, L_fft)
        max_conv = conv.max()
        score = float(2.0 * n * max_conv / (a.sum() ** 2))

        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} "
                  f"β={beta:.2f} score={score:.6f}")

        # ----- smooth weights (soft‑max) -----
        # w_i ∝ exp(β·(conv_i – max_conv))
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:      # extreme under‑flow – fall back to one‑hot
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- weighted gradient via convolution w * a_rev -----
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=L2)
        A_rev = np.fft.rfft(a_rev, n=L2)
        grad_full = np.fft.irfft(W * A_rev, n=L2)[: 3 * n - 2]
        # relevant slice corresponds to indices n‑1 … 2n‑2
        g = 2.0 * grad_full[n - 1: 2 * n - 1]

        # ----- exponentiated‑gradient update -----
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # anneal temperature
        beta = min(beta_max, beta + beta_inc)

        # occasional jitter to avoid exact plateaus
        if it % 800 == 0:
            eps = rng.random(a.shape) * a * 1e-4
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score


# ----------------------------------------------------------------------
# Final random hill‑climb (phase 5)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cand_score:.6f}")

        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand = cand / cand.sum()
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation
    function.  The routine follows the five‑stage pipeline described
    above and respects the given time budget.
    Returns a plain Python list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # safety margin

    # problem size – 16384 balances quality and speed
    n = 16384
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  uniform baseline
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0.5️⃣ optional global starter
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  fine parametric sweep (≈70 s)
    # ------------------------------------------------------------------
    sweep_end = start_time + 70.0
    i_arr = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_arr + 1.0)

    # geometric × power family
    lam_grid = np.linspace(1e-06, 2.5e-03, 400)   # a bit denser
    p_grid = np.linspace(-12.0, 6.0, 151)        # extended range
    for lam in lam_grid:
        if time.time() >= sweep_end:
            break
        base = -lam * i_arr
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} p={p:.2f} score={sc:.6f}")

    # pure Gaussian family
    c_grid = np.logspace(-9.0, -5.0, 60)
    for c in c_grid:
        if time.time() >= sweep_end:
            break
        log_shape = -c * (i_arr * i_arr)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gaussian c={c:.2e} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  projected sub‑gradient descent
    # ------------------------------------------------------------------
    sub_end = start_time + 0.45 * budget_s
    if time.time() < sub_end:
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_end, start_time, rng)

    # ------------------------------------------------------------------
    # 3️⃣  exponentiated‑gradient phase
    # ------------------------------------------------------------------
    exp_end = start_time + 0.70 * budget_s
    if time.time() < exp_end:
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_end, start_time, rng)

    # ------------------------------------------------------------------
    # 4️⃣  deterministic‑annealing smooth‑gradient phase (new)
    # ------------------------------------------------------------------
    smooth_end = start_time + 0.92 * budget_s
    if time.time() < smooth_end:
        best_seq, best_score = _smooth_expgrad_phase(best_seq, L_fft,
                                                     smooth_end, start_time, rng)

    # ------------------------------------------------------------------
    # 5️⃣  final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_fft,
                                           deadline, start_time, rng)

    # ------------------------------------------------------------------
    # Normalise finally and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # optional reporting using the provided evaluate_sequence
    try:
        from __main__ import evaluate_sequence
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
