def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 8192
    L_fft = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n-2)

    # uniform baseline
    best_seq = np.full(n, 1.0 / n)
    best_score = _eval_fast(best_seq, L_fft)

    # generate seeds
    seeds = [best_seq]
    # parametric shapes random draws
    for _ in range(1200):
        log_r = rng.uniform(np.log(0.5), np.log(0.999999))
        r = np.exp(log_r)
        p = rng.uniform(-8, 8)
        q = rng.uniform(-8, 8)
        shape = _make_general_shape(n, r=r, p=p, q=q)
        lam = rng.random()
        seeds.append(_mix_with_uniform(shape, lam))

    # Low-alpha Dirichlet
    for alpha in [0.01, 0.02, 0.05, 0.1]:
        for _ in range(400):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # compute scores, keep top K=20
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 20
    top_idx = np.argsort(scores)[:K]
    best_seq = seeds[top_idx[0]].copy()
    best_score = scores[top_idx[0]]
    print(f"initial best score {best_score}")

    # Mirror descent with increasing beta
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]
    # for each beta, run certain number of iterations or until time
    for beta in beta_schedule:
        if time.time() > deadline - 5: break
        # step schedule: maybe 2000 steps
        max_iter = 4000
        eta0 = 0.5
        a = best_seq.copy()
        t = 0
        while t < max_iter and time.time() < deadline - 2:
            t += 1
            conv = _conv_self_fft(a, L_fft)
            p_weights = _softmax_weights(conv, beta)
            g = _grad_softmax_new(a, p_weights, L_grad)
            # multiplicative update
            eta = eta0 / np.sqrt(t)
            a = a * np.exp(-eta * g)
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum
            else:
                a = np.full_like(a, 1.0 / n)
            cur_score = _eval_fast(a, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start:.2f}s] Beta {beta} itr {t} best {best_score:.6f}")

    # final hill climb: small swaps
    while time.time() < deadline:
        # random tiny mass movement between random indices
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 1e-3  # smaller step maybe 0.1% mass
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"hill step best {best_score:.6f}")

    # final normalise
    best_seq = np.maximum(best_seq, 0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total

    # print final
    print(f"Finished best score {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Propose a high‑quality step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works as follows:

1.  Choose a relatively large problem size (n = 8192).  A larger n gives
    more degrees of freedom for the shape of the step‑function, which
    typically lowers the ‑∞ autocorrelation bound.

2.  Build a modest pool of *seed* vectors:
    – the uniform vector,
    – the global starter `height_sequence_1` (if supplied),
    – a large collection of parametric shapes
      `a_i ∝ r**i·(i+1)^p·(n‑i)^q` with random parameters,
    – low‑α Dirichlet draws.
    The pool is kept to the K best seeds (K≈20) using a cheap exact
    evaluation of the objective.

3.  Starting from the best seed run a long *mirror‑descent* (exponential‑
    gradient) optimisation on the smooth‑max surrogate
        L_β(a) = (1/β)·log Σ_k exp(β·conv_k(a,a)).
    β is increased geometrically (1,2,4,…,8192) and a decreasing step‑size
    η_t = η₀/√t is used.  The gradient of L_β is obtained by a single FFT‑
    based convolution.  The update is multiplicative
        a_i ← a_i·exp(‑η·∇_i L_β)  followed by normalisation onto the simplex.
    The current best vector (lowest true objective) is recorded continuously.

4.  After the smooth‑max phase a short *sub‑gradient* polishing on the
    true ‑∞ objective is performed (identical to the exponential‑gradient
    step but using the sub‑gradient at the maximal convolution index).

5.  Finally a lightweight random hill‑climb (tiny mass transfers
    between random positions) is run until the time budget expires.

All heavy work uses FFT‑based convolutions, the Euclidean simplex
projection is avoided by explicit renormalisation, and the routine
stays well within the 1000 s budget.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a (length 2·len(a)‑1) using FFT.
    L must be a power of two ≥ 2·len(a)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2·Σ_j a_j·p_{i+j}.
    Implemented as a convolution of a_rev with p (full length 3·n‑2),
    then the central n entries are extracted.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv_full = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv_full[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries (numerically stable)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact objective (used for selection and final reporting)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_general_shape(n: int, r: float = 0.9, p: float = 0.0, q: float = 0.0) -> np.ndarray:
    """
    Shape a_i ∝ r**i · (i+1)**p · (n-i+1)**q,
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    # log‑space construction to avoid overflow/underflow
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient helpers (used in polishing phases)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

                print(f"[{time.time() - start:.2f}s] subgrad it={it:,} score={cur:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final random hill‑climb (tiny mass transfers) until time expires.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move at most 0.3 % of the source mass
        eps_move = best_seq[src] * rng.random() * 3e-3
        prop = best_seq.copy()
        prop[src] -= eps_move
        prop[dst] += eps_move
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
