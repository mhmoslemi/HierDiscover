Projection onto simplex algorithm (O(n log n)):

```
def project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def line_search(a, grad, step_size, best_score):

    for factor in [1.0, 0.5, 0.2, 0.1, 0.05]:

Implementation details of projecting onto simplex:

```
def project_simplex(v):
    # v is 1d array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero or negative
        return np.zeros_like(v)
    rho = ind[cond][-1]  # index (1-indexed)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def random_refine(a, best_score, deadline, rng):
    best = a.copy()
    best_score = best_score
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst: continue
        if best[src] < 1e-12: continue
        delta = best[src] * rng.random() * 0.005
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        # (positive)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        # optional print each improvement
    return best, best_score

"""
Projected‑gradient + optional LP refinement + random polishing
================================================================

This solver searches for a non‑negative step‑function (a probability vector)
that minimises

    F(a) = 2·n·max(conv(a,a)) / (∑ a)²   (scale invariant)

The algorithm proceeds as follows

1. **Initialisation** – start from the global best `height_sequence_1`
   (if available) or from a uniform vector.  The vector is normalised to
   sum to 1.

2. **Projected gradient descent** on the L∞‑norm of the self‑convolution.
   * Compute the full linear convolution by FFT.
   * Locate all indices where the convolution attains its maximum.
   * The sub‑gradient of `max(conv)` is the sum of the gradients of the
     active indices; for an active index `k` the gradient component `i` is
     `2·a_{k‑i}` (zero if out of range).
   * Move a small step opposite to the sub‑gradient and project back onto
     the probability simplex (non‑negative entries, sum = 1).
   * An adaptive step size with simple back‑tracking line‑search is used.
   * If no improvement is seen for many iterations a tiny random
     perturbation (Dirichlet sample) is injected to escape possible plateaus.

3. **LP improvement (optional)** – if `cvxpy` is available and time permits,
   a few alternating‑LP steps are performed exactly as described in the
   reference.  They increase the ℓ₁‑norm of a feasible vector while keeping
   the convolution bound unchanged, after which a short line‑search finds the
   best convex combination.

4. **Stochastic polishing** – cheap pairwise mass‑transfer moves are tried
   until the time budget expires; only strict improvements are kept.

5. Return the best sequence found (as a list of floats).  All numbers are
   non‑negative and the sum is normalised to 1, satisfying the security
   checks of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – clip to zero.
    if conv.min() < -1e-15:
        conv = np.clip(conv, 0.0, None)
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_conv(a)


# ----------------------------------------------------------------------
# Simplex projection (Michelot algorithm)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                     # the last true index (1‑based)
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to avoid tiny floating‑point drift
    w_sum = w.sum()
    if w_sum <= 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# LP improvement step (identical to the reference implementation)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, M: float, n: int):
    """
    Solve
        maximise   Σ g_j
        subject to conv(f,g) ≤ M   (elementwise)
                   g ≥ 0
    Returns a scaled version of the optimal g with Σ g = 1 (same total mass as f)
    or ``None`` when the LP cannot increase the ℓ₁‑norm.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    conv_fg = cp.conv(f_param, g)          # length 2·n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_fg <= M, g >= 0])

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=20000,
                   abstol=1e-9, reltol=1e-9, verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_opt = np.asarray(g.value).flatten()
    if g_opt is None:
        return None
    g_opt = np.maximum(g_opt, 0.0)
    sum_g = g_opt.sum()
    sum_f = f.sum()          # should be 1.0

    if sum_g <= sum_f + 1e-12:
        return None            # no ℓ₁‑norm improvement

    # Scale back to original total mass (=1)
    g_opt *= (sum_f / sum_g)
    return g_opt


# ----------------------------------------------------------------------
# Random local refinement (pairwise mass transfers)
# ----------------------------------------------------------------------
def _random_local_refine(a: np.ndarray, cur_score: float,
                         deadline: float, rng: np.random.Generator):
    """Cheap polishing: move a tiny random fraction of mass and keep strict improvements."""
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or best[src] <= 1e-15:
            continue
        # move at most 0.5 % of the source mass
        delta = best[src] * rng.random() * 0.005
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
    return best, best_score
