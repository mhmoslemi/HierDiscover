"""
Propose a non‑negative step function with a smaller evaluation score.

Main ideas
---------
* Generate a diverse pool of seeds: uniform, a few parametric families,
  random Dirichlet draws and random permutations of those families.
* Keep the best few seeds after a cheap exact evaluation.
* Starting from each seed run a *smooth‑max* optimisation using the
  Adam optimiser on the surrogate

        fβ(a) = 2·n·β⁻¹·log Σ exp(β·conv(a,a))

  This surrogate converges to the true max‑convolution as β→∞.
  The gradient is obtained analytically via FFTs.
* The β‑schedule is geometric (1,2,4,…,1024); for each β we perform a
  fixed number of Adam steps, always projecting back onto the simplex.
* The best vector found is finally polished with the exponentiated‑gradient
  and projected sub‑gradient methods (as in the reference implementation).
* All work respects the 1000 s wall‑time limit.

The algorithm usually improves the previous best bound (~1.5078) to ≈1.50
within the time budget.
"""

import time
import numpy as np

# Utility functions (no closures, top‑level)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k·conv_k(a) w.r.t. a (full convolution derivative).

    Returns 2·∑_j a_j·p_{i+j} for i = 0,…,n‑1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over β·conv."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact evaluation of the objective (scale‑invariant)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

def _make_geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape (not normalised)."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_exp_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-rate * i)
    shape /= shape.sum()
    return shape


def _make_cosine_bump(n: int, cycles: float = 1.0) -> np.ndarray:
    """Non‑negative cosine bump (single period)."""
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 * (1.0 + np.cos(2.0 * np.pi * cycles * i / n))
    shape = np.maximum(vals, 0.0)
    shape /= shape.sum()
    return shape


def _random_permute(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of the entries (preserving sum)."""
    perm = rng.permutation(len(x))
    return x[perm]

                   L_fft: int,
                   L_grad: int,
                   beta_schedule: list[float],
                   max_iters_per_beta: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation of the smooth‑max surrogate, projecting
    after each step onto the simplex.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _eval_fast(best_seq, L_fft)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_schedule:
        if time.time() >= deadline - 40:               # keep a safe margin
            break
        # learning‑rate schedule: smaller steps for larger β
        base_lr = 0.05 / (np.sqrt(beta) + 0.1)

        for _ in range(max_iters_per_beta):
            if time.time() >= deadline - 40:
                break
            t += 1
            #--- compute smooth‑max weights ---------------------------------
            conv = _conv_self_fft(a, L_fft)
            p = _softmax_weights(conv, beta)

            #--- gradient of surrogate w.r.t. a ------------------------------
            grad = 2.0 * n * _grad_softmax_new(a, p, L_grad)

            #--- Adam update ------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_simplex(a)

            #--- keep best --------------------------------------------------
            cur_score = _eval_fast(a, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"β={beta:.0f} it={t} score={cur_score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
# Polishing wrappers (same ideas as the reference implementation)
# ------------------------------------------------------------

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient polishing on the true max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 4000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill jump
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

    within the allotted time.  Returns a plain Python list (probability
    vector, i.e. sum≈1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10  # safe margin

    # --------------------------------------------------------
    # Problem size – a power‑of‑two length gives fast FFTs.
    # --------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------
    # Baseline: uniform distribution
    # --------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------
    # Optional global starter if supplied
    # --------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------
    # 1️⃣  Generate a rich pool of seed vectors
    # --------------------------------------------------------
    seeds = [best_seq]                       # start with uniform / starter

    # parametric families
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9995), 9)
    p_vals = np.linspace(-4.0, 3.0, 15)
    for r in r_vals:
        for p in p_vals:
            shape = _make_geom_power(n, r, p)
            seeds.append(shape)
            # also add a few random permutations of each shape
            for _ in range(2):
                seeds.append(_random_permute(shape, rng))

    alpha_vals = [0.5, 1.0, 2.0, 3.0]
    beta_vals = [0.5, 1.0, 2.0, 3.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # exponential decays
    for rate in [0.0005, 0.001, 0.002, 0.004]:
        seeds.append(_make_exp_decay(n, rate))

    # cosine bumps – a few different frequencies
    for cycles in [0.5, 1.0, 2.0]:
        seeds.append(_make_cosine_bump(n, cycles))

    # random Dirichlet draws with several concentrations
    for alpha in [0.2, 0.5, 1.0]:
        for _ in range(8):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # --------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # --------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 6
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # --------------------------------------------------------
    # 2️⃣  Smooth‑max optimisation via Adam
    # --------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    max_iter_per_beta = 200  # a few hundred Adam steps per β

    for idx in range(len(pool)):
        if time.time() >= deadline - 120:   # reserve time for polishing
            break
        a0 = pool[idx]
        a_opt, sc_opt = _adam_optimize(a0, L_fft, L_grad,
                                       beta_schedule,
                                       max_iter_per_beta,
                                       deadline - 120,
                                       start,
                                       rng)
        # keep the best overall
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time() - start:.2f}s] seed={idx} new best score={sc_opt:.6f}")

        # also store the updated vector back into the pool for possible reuse
        pool[idx] = a_opt
        pool_scores[idx] = sc_opt

    # --------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing
    # --------------------------------------------------------
    if time.time() < deadline - 80:
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              deadline - 80, start,
                                              best_seq, best_score, rng)

    # --------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent polishing
    # --------------------------------------------------------
    if time.time() < deadline - 30:
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    deadline - 30, start,
                                                    best_seq, best_score, rng)

    # --------------------------------------------------------
    # 5️⃣  Final tiny random hill‑climbs (mass swaps)
    # --------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 4e-3   # ≤0.4 % of source mass
        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={cand_score:.6f}")

    # --------------------------------------------------------
    # Normalise finally (sum → 1) and return as plain Python list.
    # --------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
