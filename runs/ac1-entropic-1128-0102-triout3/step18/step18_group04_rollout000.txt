"""
Sub‑gradient descent on the exact ℓ∞‑convolution objective.

We minimise
       F(a) = 2·n·max_k (a * a)[k]          (∑a = 1, a_i ≥ 0)

where “*” is the linear self‑convolution.
The function is non‑smooth because of the max, but its sub‑gradient at a
maximising shift k can be written in closed form:
    ∂(a*a)[k]/∂a_j = a_{k‑j}   (indices out of range are ignored).

The algorithm:

1. Initialise a with the best known global construction (``height_sequence_1``)
   or, if unavailable, with the uniform vector.
2. Repeatedly
   – compute the full convolution via FFT,
   – locate the (few) shift(s) attaining the maximum,
   – build the exact sub‑gradient from those shifts,
   – take a projected gradient step on the probability simplex,
   – adapt the step‑size (increase on improvement, shrink otherwise).
3. When progress stalls, inject a tiny random mass transfer to escape flat
   plateaux.
4. After the time budget a cheap random hill‑climb is performed for a final
   polish.

The method works directly on the true nonsmooth objective (no smooth‑max
approximation) and therefore can push the bound a little further than the
previous smoothed continuation approach.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Timer utilities
# ----------------------------------------------------------------------
_start_time = None  # will be set in propose_candidate


def _now() -> float:
    """Seconds elapsed since start."""
    return time.time() - _start_time


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (v_i ≥ 0, Σv_i = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # fallback (should never happen)
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear convolution
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Return the full linear convolution a*a (length 2·len(a)‑1).
    ``L`` must be a power of two ≥ 2·len(a)‑1.
    """
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (unsmoothed)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    L = _next_pow2(2 * n - 1)
    conv = _conv_full(a, L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution term
# ----------------------------------------------------------------------
def _subgradient_of_max(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of  max_k (a*a)[k]  with respect to a.
    ``active`` are the shift indices where the maximum is attained.
    """
    n = a.shape[0]
    g = np.zeros(n, dtype=float)
    for k in active:
        if k < n:
            # overlap length = k+1, contributions reversed
            g[: k + 1] += a[: k + 1][::-1]
        else:
            start = k - n + 1
            g[start:] += a[start:][::-1]
    return g


# ----------------------------------------------------------------------
# Random Dirichlet starter (cheap)
# ----------------------------------------------------------------------
def _dirichlet_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector from a Dirichlet with random concentration."""
    c = 10 ** rng.uniform(-3.0, 0.7)      # log‑uniform concentration
    a = rng.gamma(shape=c, scale=1.0, size=n)
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    a /= a.sum()
    return a

# Main optimisation routine
# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs a sub‑gradient descent on the exact ℓ∞‑convolution
    objective, followed by a tiny random hill‑climb.
    """
    global _start_time
    np.random.seed(seed)
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.2   # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep the same size as the reference run.
    # --------------------------------------------------------------
    n = 4096
    L = _next_pow2(2 * n - 1)                # length for FFTs

    # --------------------------------------------------------------

    try:

                step = max(step, 1e-5)      # give a little room after perturb
            no_improve = 0

        # safety break if we are extremely close to the time limit
        if time.time() >= deadline - 0.5:

    # --------------------------------------------------------------
    # Final cheap random local refinement (pure hill‑climb)
    # --------------------------------------------------------------
    a = best_seq.copy()

            best_seq = cand.copy()

    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)

    if total > 0:

    else:

    print(f"[{_now():.2f}s] Finished – final objective ≈ {best_score:.6f}")
