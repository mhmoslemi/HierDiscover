def _run_descent(a0, max_iters, time_limit, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    L_conv = _next_pow2(2*n-1)  # for _conv_self_fft
    L_cross = _next_pow2(3*n-2) # for gradient
    beta = 1.0
    beta_max = 3000.0
    beta_mul = 1.005
    eta0 = 1.0
    best_seq_local = best_seq
    best_score_local = best_score
    for it in range(1, max_iters+1):
        if time.time() >= time_limit:
            break
        # compute convolution self
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[: 2*n - 1]
        # compute softmax weights
        scaled = beta * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()
        # compute gradient via cross-correlation
        w_rev = w[::-1]
        Af2 = np.fft.rfft(a, n=L_cross)
        Wf_rev = np.fft.rfft(w_rev, n=L_cross)
        conv_full = np.fft.irfft(Af2 * Wf_rev, n=L_cross)
        # relevant part
        conv_valid = conv_full[: n + len(w_rev) - 1]
        idx = len(w_rev) - 1
        grad = 2.0 * conv_valid[idx - np.arange(n)]
        # multiplicative update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # avoid zeros
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0/n)
            sum_a = 1.0
        a = a / sum_a
        # evaluate objective
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score_local - 1e-12:
            best_score_local = score
            best_seq_local = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Descent it={it:,} score={score:.6f} beta={beta:.2f}")
        # update beta
        beta = min(beta * beta_mul, beta_max)
    return best_seq_local, best_score_local

Propose a non‑negative step‑function minimizing
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm uses an annealed soft‑max approximation of the
max‑convolution and a mirror‑descent (multiplicative) update on
the probability simplex.  Starting from the best known sequence
(if any) as well as many random Dirichlet seeds, it performs a
large number of fast FFT‑based iterations.  The soft‑max

the true max‑objective, while the step size decays as 1/√k.
The best sequence found before the time budget expires is returned.

    return 1 << ((x - 1).bit_length())

def _objective_score(sequence: np.ndarray) -> float:
    """Exact objective for a sequence normalised to sum==1."""
    n = sequence.shape[0]
    # full self‑convolution via FFT
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(sequence, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv)

def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over the convolution values."""
    scaled = beta * conv
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    return exp_scaled / exp_scaled.sum()

def _gradient_from_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a))
    with respect to a (multiplicative mirror‑descent version).
    Implements 2 * cross‑correlation(a, w) .
    """
    n = a.shape[0]
    w_rev = w[::-1]                         # length 2n‑1
    L = _next_pow2(n + len(w_rev) - 1)      # size for the cross‑correlation
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w_rev, n=L)
    full = np.fft.irfft(Af * Wf, n=L)
    # keep the valid part (length 3n‑2)
    valid = full[: n + len(w_rev) - 1]      # shape 3n‑2
    idx = len(w_rev) - 1                    # = 2n‑2
    # grad_i = 2 * valid[idx - i]  for i=0…n‑1
    grad = 2.0 * valid[idx - np.arange(n)]
    return grad

def _run_annealed_descent(a0: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          start_time: float,
                          max_iters: int = 10**9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation.
    beta starts at 1 and is multiplied by beta_mul each iteration,
    up to beta_max.  Step size η = η0 / sqrt(k).
    Returns the best sequence observed.
    """
    a = a0.copy()
    n = a.shape[0]

    L_conv = _next_pow2(2 * n - 1)          # for a*a
    L_cross = _next_pow2(3 * n - 2)         # for cross‑correlation

    beta = 1.0
    beta_max = 3000.0
    beta_mul = 1.003                        # slow annealing
    eta0 = 1.0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- self‑convolution (needed for the true objective) ----------
        Af = np.fft.rfft(a, n=L_conv)
        conv = np.fft.irfft(Af * Af, n=L_conv)[: 2 * n - 1]

        # --- soft‑max weights -----------------------------------------
        w = _softmax_weights(conv, beta)

        # --- gradient w.r.t. a (smooth) --------------------------------
        grad = _gradient_from_softmax(a, w)

        # --- multiplicative mirror‑descent update ----------------------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # --- renormalise onto the simplex -------------------------------
        total = a.sum()
        if total <= 0.0 or np.isnan(total):
            a = np.full_like(a, 1.0 / n)
            total = 1.0
        a /= total

        # --- evaluate the true objective (not the smooth proxy) -------
        max_conv = conv.max()
        score = float(2.0 * n * max_conv)

        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Iter {it:,}  score={score:.6f}  beta={beta:.2f}")

        # --- anneal beta ------------------------------------------------
        beta = min(beta * beta_mul, beta_max)

        # occasional random jitter to leave flat plateaus
        if it % 5000 == 0:
            eps = 1e-4 * np.random.randn(n)
            a = np.clip(a + eps, 0.0, None)
            a /= a.sum()

    return best_seq, best_score
