def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (v_i ≥ 0, Σv_i = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
