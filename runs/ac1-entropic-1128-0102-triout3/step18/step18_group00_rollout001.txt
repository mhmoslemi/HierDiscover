"""
Hybrid optimiser with smoothed‐max gradient descent.

The routine consists of four phases (all respecting the 1000 s budget):

1️⃣  Fast parametric sweep over two families (monotone “geo‑power” and symmetric
    “U‑shape”) to obtain a good starting point.

2️⃣  Smoothed‑max gradient descent.
    – The true objective contains a max‑convolution term.
    – We replace max by a softmax (temperature γ) so that the gradient is a
      weighted sum of all convolution sub‑gradients.
    – The weighted gradient can be computed by a single cross‑correlation via FFT,
      i.e. in O(n log n) time.
    – The step size decays as 1/√t and the solution is projected onto the
      probability simplex after each step.
    – γ is slowly increased (up to ≈30) so the procedure gradually focuses on the
      true max.

3️⃣  A short simulated‑annealing hill‑climb with larger mass‑transfer moves to
    escape possible plateaus.

4️⃣  Final cleanup: project onto the simplex, normalise and return the sequence.

The search works for n = 4096 (the original size) but the code can easily be
scaled to larger n (e.g. 8192) by adjusting the `n` variable – the optimiser
remains within the 1000 s budget thanks to the FFT‑based fast convolutions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear convolution a*b via FFT (output length len(a)+len(b)-1)."""
    n, m = a.shape[0], b.shape[0]
    if L is None:
        L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _cross_corr(a: np.ndarray, b: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ b)[i] = Σ_j a[j]·b[i+j] for i=0..len(a)-1.
    Computed via FFT using the conjugate of the second transform.
    Returns the first len(a) entries.
    """
    n, m = a.shape[0], b.shape[0]
    if L is None:
        L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(Af * np.conj(Bf), n=L)[: n]  # linear part
    return corr


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective (fast version) – assumes sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families for the parametric sweep
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(-lam·i) (L1‑normalised)."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape” (sum of left and mirrored right parts, L1‑normed)."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    left = np.exp(log_shape)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent
# ----------------------------------------------------------------------


def _smooth_max_gradient_descent(
    a0: np.ndarray,
    L_self: int,
    L_corr: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of max(conv).
    The temperature γ is slowly increased, turning the objective into the true
    max as the optimisation proceeds.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.5
    gamma = 5.0               # start with a smooth max
    gamma_max = 30.0
    while time.time() < deadline:
        it += 1

        # self‑convolution
        conv = _conv_self_fft(a, L_self)

        # soft‑max weights (stable)
        conv_max = conv.max()
        w = np.exp(gamma * (conv - conv_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient = weighted sum of sub‑gradients = cross‑correlation(a, w)
        grad = _cross_corr(a, w, L_corr)  # length n

        # step‑size schedule ≈ 1/√t
        step = step0 / np.sqrt(it)

        cand = a - step * grad
        cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_self)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] smooth‑GD it={it:,} "
                f"γ={gamma:.2f} step={step:.5e} score={cand_score:.6f}"
            )
            a = cand  # accept improvement immediately

        else:
            # occasional random acceptance to escape tiny plateaus
            if rng.random() < 1e-4:
                a = cand

        # slowly raise the temperature
        if it % 2000 == 0 and gamma < gamma_max:
            gamma = min(gamma_max, gamma * 1.05)

        # small jitter to keep diversity
        if it % 5000 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_self: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Simple SA with linearly cooling temperature."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    T0 = 1.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        # linear cooling based on progress
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer up to 0.5 % of source mass (a bit larger than baseline)
        eps = cur[src] * rng.random() * 5e-3
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_self)
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        # occasional random jitter
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = _project_simplex(cur)
            cur_score = _eval_fast(cur, L_self)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score

    lam_grid = np.linspace(min_lambda, max_lambda, 300)   # 300 points
    p_grid = np.linspace(-6.0, 3.0, 91)                  # step ≈0.1

    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)   # pre‑computed log(i+1)

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            # monotone shape
            mono = _make_geo_power(n, lam, p, log_i)
            sc = _eval_fast(mono, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = mono.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep mono λ={lam:.6e} "
                    f"p={p:.2f} score={sc:.6f}"
                )

            # symmetric “U‑shape”
            sym = _make_geo_power_symmetric(n, lam, p, log_i)
            sc_sym = _eval_fast(sym, L_self)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep sym λ={lam:.6e} "
                    f"p={p:.2f} score={sc_sym:.6f}"
                )
        # occasionally inject a few random Dirichlet starts for diversity
        if rng.random() < 0.03:
            rand = rng.random(n)
            rand = _project_simplex(rand)
            sc_rand = _eval_fast(rand, L_self)
            if sc_rand < best_score:
                best_score = sc_rand
                best_seq = rand.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] random start from sweep score={sc_rand:.6f}"
                )

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed‑max gradient descent (major phase).
    # ------------------------------------------------------------------
    gd_deadline = deadline - 200.0   # leave ~200 s for the final SA
    if time.time() < gd_deadline:
        best_seq, best_score = _smooth_max_gradient_descent(
            best_seq,
            L_self,
            L_corr,
            gd_deadline,
            start_time,
            best_seq,
            best_score,
            rng,
        )

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing final hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq,
            best_score,
            L_self,
            deadline,
            start_time,
            rng,
        )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
