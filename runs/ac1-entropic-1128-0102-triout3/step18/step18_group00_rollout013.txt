from scipy.optimize import linprog

def solve_lp(f, M, target_sum):
    n = f.shape[0]
    # Build sparse matrix A_ub
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)  # 0,1,...,n-1
    A = sp.diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
    # add sum constraint
    # we can stack matrix: 
    A_ub = sp.vstack([A, np.ones((1, n), dtype=np.float64)], format='csr')
    b_ub = np.concatenate([np.full(2*n-1, M, dtype=np.float64), np.array([target_sum])])
    c = -np.ones(n, dtype=np.float64) # maximize sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        # fallback: return current f maybe
        g = f.copy()
    return g, res
```

"""
Advanced hybrid optimiser for the L∞‑convolution bound.

The pipeline consists of four phases:

1️⃣  Parameter sweep (mono‑/symmetric “geo‑power” families) to obtain a
    good basin of attraction.

2️⃣  LP‑smoothing:  for the current vector `f` we solve the linear programme
    max ∑ g_j  s.t.  f * g ≤ M  (M = max(f*f)) and g ≥ 0, ∑ g = √(2n).
    The solution `g` is mixed with `f` (t = 0.5) which provably does not
    increase the peak of the self‑convolution.  The step is repeated a
    few times – this often yields a noticeable drop in the objective.

3️⃣  Projected sub‑gradient descent with a *batch* sub‑gradient (averaged
    over all indices attaining the maximal convolution value).  A simple
    back‑tracking line‑search and occasional jitter help to escape tiny
    plateaus.

4️⃣  Simulated‑annealing hill‑climb (tiny mass‑transfer moves) to polish
    the final candidate.

All phases respect a total wall‑time budget of ≈ 1000 s.  The output is a
plain list of non‑negative floats whose sum is `sqrt(2·n)`;  feeding the
list to `evaluate_sequence` yields the desired upper bound.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast version) – assumes the sum equals √(2·n).
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return the objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate vectors (used in the sweep)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape: a_i ∝ (i+1)^p·exp(‑lam·i); L1‑normalised."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)          # stabilise exp
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape”: left + mirrored right; L1‑normalised."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)                      # decreasing part
    right = left[::-1]                       # mirror
    shape = left + right
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient utilities (batch version)
# ----------------------------------------------------------------------


def _subgrad_max_conv_batch(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) averaged over all arg‑max indices.
    Returns a vector of length n.
    """
    n = a.shape[0]
    max_val = conv.max()
    # indices where the max is attained (tolerance for floating errors)
    eps = 1e-12
    ks = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * a[j]
    # average over the number of active constraints
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Linear‑programming smoothing phase
# ----------------------------------------------------------------------


def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1)×n matrix A with entries A[k,j]=f_{k‑j} for k≥j,
    otherwise 0.  Implemented via `scipy.sparse.diags`.
    """
    n = f.shape[0]
    # each diagonal offset d∈[0,n‑1] consists of the constant f[d]
    diags = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int32)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_step(f: np.ndarray, M: float, target_sum: float, rng: np.random.Generator):
    """
    Solve   max 1ᵀg   s.t.   A·g ≤ M·1,   ∑g = target_sum,   g ≥ 0.
    Returns the feasible vector g (may be unchanged if LP fails).
    """
    n = f.shape[0]
    A = _build_constraint_matrix(f)
    # add the global sum constraint (∑g ≤ target_sum)
    one_row = sp.csr_matrix(np.ones((1, n), dtype=np.float64))
    A_ub = sp.vstack([A, one_row], format="csr")
    b_ub = np.concatenate([np.full(2 * n - 1, M, dtype=np.float64),
                           np.array([target_sum], dtype=np.float64)])

    c = -np.ones(n, dtype=np.float64)               # maximise sum(g)
    bounds = [(0.0, None)] * n

    # HiGHS interior‑point solver – fast for our moderate size.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method="highs", options={"presolve": True})

    if not res.success:
        # fall back to the current vector – the iteration will simply continue
        return f.copy(), False
    g = res.x
    # numerical clean‑up
    g[g < 0] = 0.0
    s = g.sum()
    if s > 0:
        g = g * (target_sum / s)   # renormalise exactly
    else:
        g = np.full_like(g, target_sum / n)
    return g, True


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (using batch sub‑gradient)
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _subgrad_max_conv_batch(a, conv)

        # step decays like 1/√t
        step = step0 / (it ** 0.55)

        improved = False
        for _ in range(8):
            cand = a - step * g
            cand = _project_simplex(cand) * np.sqrt(2 * n)  # re‑normalise sum
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(
                    f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                    f"step={step:.5e} score={cand_score:.6f}"
                )
                break
            step *= 0.5
        if not improved:
            # tiny uphill attempt
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a) * np.sqrt(2 * n)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with a decreasing temperature."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score

    n = cur.shape[0]
    T0 = 1.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * 5e-3
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = cur / cur.sum() * np.sqrt(2 * n)
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )
    return best, best_score

# Main entry point
# ----------------------------------------------------------------------

      2. LP‑based smoothing / mixing
      3. Projected sub‑gradient descent with batch sub‑gradients
      4. Simulated‑annealing polishing

    """

    start_time = time.time()

    # ------------------------------------------------------------------

    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    uniform = np.full(n, sum_target / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter * (sum_target / starter.sum())
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (≈70 s). Two families: monotone & symmetric.
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    min_lambda = 4e-05
    max_lambda = 1.5e-03
    lam_grid = np.linspace(min_lambda, max_lambda, 300)   # 300 points
    p_grid = np.linspace(-6.0, 3.0, 91)                  # step ≈0.1

    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)   # pre‑computed log(i+1)

            break
        base = -lam * i_arr
        for p in p_grid:
            # ---------- monotone shape ----------
            log_shape = base + p * log_i
            log_shape -= np.max(log_shape)
            mono = np.exp(log_shape)
            mono = mono * (sum_target / mono.sum())
            sc = _eval_fast(mono, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = mono.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep mono λ={lam:.6e} p={p:.2f} score={sc:.6f}"
                )

            # ---------- symmetric shape ----------
            left = mono                     # reuse monotone component
            sym = left + left[::-1]
            sym = sym * (sum_target / sym.sum())
            sc_sym = _eval_fast(sym, L_fft)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep sym λ={lam:.6e} p={p:.2f} score={sc_sym:.6f}"
                )

    # ------------------------------------------------------------------
    # 2️⃣  LP‑based smoothing / mixing (a few iterations).
    # ------------------------------------------------------------------
    lp_deadline = deadline - 350.0
    lp_iter = 0
    while time.time() < lp_deadline:
        lp_iter += 1
        conv = _conv_self_fft(best_seq, L_fft)
        M = conv.max()
        g, ok = _lp_step(best_seq, M, sum_target, rng)
        if not ok:
            break
        # Mix with t = 0.5 (convex combination keeps the bound ≤ M)
        t = 0.5
        cand = (1.0 - t) * best_seq + t * g
        cand = cand * (sum_target / cand.sum())          # re‑normalise
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter} improved score={cand_score:.6f}")
        else:
            # if no improvement, try a smaller mixing weight
            for t_small in [0.3, 0.1]:
                cand = (1.0 - t_small) * best_seq + t_small * g
                cand = cand * (sum_target / cand.sum())
                cand_score = _eval_fast(cand, L_fft)
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] LP iter {lp_iter} t={t_small:.2f} improved score={cand_score:.6f}"
                    )
                    break
        # occasional random jitter to keep diversity
        if rng.random() < 0.02:
            jitter = 1e-4 * rng.random(n) * best_seq
            best_seq = np.maximum(best_seq + jitter, 0.0)
            best_seq = best_seq * (sum_target / best_seq.sum())
    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient descent (more intensive).
    # ------------------------------------------------------------------
    sub_deadline = deadline - 150.0
    if time.time() < sub_deadline:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, sub_deadline, start_time, best_seq, best_score, rng
        )

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing final hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, L_fft, deadline, start_time, rng
        )

    # ------------------------------------------------------------------
    # Final normalisation (sum → √(2·n)) and conversion to pure python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq * (sum_target / total)
    else:
        best_seq = np.full(n, sum_target / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
