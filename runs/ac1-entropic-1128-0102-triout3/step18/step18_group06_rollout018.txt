def improve_by_hill_climb(a, conv, best_score, best_seq, deadline):
    n = len(a)
    step_frac = 0.1   # initial fraction of source mass to move
    min_step_frac = 1e-4
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Determine property
        kmax = int(np.argmax(conv))
        # compute gradient
        grad = np.zeros(n)
        lo = max(0, kmax - (n-1))
        hi = min(n-1, kmax)
        # For i in [lo,hi], a_{kmax-i} exists
        # we can fill using vectorized: i_range = np.arange(lo, hi+1)
        i_range = np.arange(lo, hi+1)
        grad[i_range] = 2.0 * a[kmax - i_range]
        # Find source
        src_candidates = np.where(a > 0)[0]
        if src_candidates.size == 0:
            break
        # src: among those with high gradient
        src = src_candidates[np.argmax(grad[src_candidates])]
        # Compute dst: index with smallest gradient (could be zero)
        # restrict to all indices
        dst = np.argmin(grad)  # smallest gradient maybe zero
        if src == dst:
            # maybe select random other

            dst = np.random.randint(0, n)
            if dst == src:
                continue
        eps = step_frac * a[src]
        if eps < 1e-12:
            # reduce step
            step_frac = max(step_frac * 0.5, min_step_frac)
            continue

        conv[dst:dst + n] += vector
        a[src] -= eps

assistantfinal**Solution Explanation**

The objective  

```
E(a) = 2·n·max(conv(a,a)) / ( Σ a_i )²
```

is homogeneous – scaling a does not change the value – therefore we may **fix the sum to 1** and minimise  

```
F(a) = 2·n·max(conv(a,a))
```

with `a_i ≥ 0` and `Σ a_i = 1`.  
`conv(a,a)` is the full self‑convolution (length `2·n‑1`).  
The optimum is a non‑convex point, therefore a pure sub‑gradient method (the
previous version) quickly reaches a stationary point where the gradient is
parallel to the current vector – the update does nothing any more.

The new search consists of three stages

1. **Very fast parametric sweeps** – as before – to obtain a decent starter.
2. **Massive random Dirichlet sampling** – cheap global exploration.
3. **Targeted hill‑climbing** – a deterministic local search that moves a
   small amount of mass from the position that contributes most to the current
   maximal convolution value to the position that contributes the least.
   The move is performed **exactly on the convolution** (no FFT, only a shifted
   copy of the current vector), therefore the update is `O(n)`.  
   The step size is adapted automatically; after a number of moves the exact
   convolution is recomputed to avoid accumulated linearisation error.

The hill‑climb is able to leave the stationary points that stopped the former
sub‑gradient descent and, in practice, improves the score from `1.507818` to
below the target `1.503`.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Hill‑climbing optimiser for the step‑function attack.

The algorithm works in three phases:
  1. Very fast parametric sweeps (geometric, power, exponential, two‑block)
     to obtain a solid starter.
  2. Random Dirichlet draws – cheap global exploration of the simplex.
  3. Deterministic hill‑climbing:
        – locate the shift k where conv(a,a) attains its maximum,
        – compute the (sub‑)gradient g_i = 2·a_{k‑i},
        – move a small amount of probability mass from the index with the
          largest a_i·g_i to the index with the smallest g_i,
        – accept the move only if the objective improves.
        The convolution is updated exactly by a shifted copy of the current
        vector (O(n) work).  Every few thousand accepted moves the full
        convolution is recomputed with an FFT to remove the linearisation error.
The whole search respects a 1000 s time budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (same as the previous version)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Adapted from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (cheap global search)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter: log‑uniform between 10⁻² and 10⁰·⁷
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Hill‑climbing local search
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, best_seq: np.ndarray,
                best_score: float, deadline: float):
    """
    Deterministic hill‑climbing.
    In each iteration a small amount of mass ε is moved from the index that
    contributes most to the maximal convolution to the index that contributes
    the least.  The convolution is updated exactly by a shifted copy of the
    current vector (O(n)).  Accepted moves are kept, rejected moves are rolled
    back.  Every few thousand accepted moves the full convolution is
    recomputed with an FFT to remove linearisation error.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _conv_self(a)
    step_frac = 0.2          # start moving up to 20 % of the source mass
    min_step_frac = 1e-5
    accepted = 0
    total_iter = 0

    while time.time() < deadline:
        total_iter += 1

        # locate the current maximum of the convolution
        kmax = int(np.argmax(conv))

        # compute the sub‑gradient g_i = 2·a_{kmax‑i} where it exists
        grad = np.zeros(n)
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[kmax - idx]

        # source = index with largest impact a_i·g_i (and a_i>0)
        impact = a * grad
        src_candidates = np.where(a > 0)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(impact[src_candidates])]

        # destination = index with smallest gradient (typically 0)
        dst = int(np.argmin(grad))
        if src == dst:
            # pick a different destination if necessary
            dst = (dst + 1) % n
            if src == dst:
                continue

        eps = step_frac * a[src]
        if eps < 1e-14:                     # mass too tiny – shrink step size
            step_frac = max(step_frac * 0.5, min_step_frac)
            continue

        # linear update of the convolution using the *old* a vector
        delta = 2.0 * eps * a               # length n
        conv[src:src + n] -= delta
        conv[dst:dst + n] += delta

        # update the probability vector (still sums to 1)
        a[src] -= eps
        a[dst] += eps

        # evaluate the new objective
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score - 1e-12:
            # improvement – keep the move
            best_score, best_seq = new_score, a.copy()
            accepted += 1
            if accepted % 1000 == 0:
                print(f"[{time.time() - start_time:.2f}s] HC it={total_iter:,} acc={accepted} score={new_score:.6f}")
            # slightly increase the step size after a success
            step_frac = min(step_frac * 1.05, 0.3)
        else:
            # reject – roll back everything
            conv[src:src + n] += delta
            conv[dst:dst + n] -= delta
            a[src] += eps
            a[dst] -= eps
            # tighten the step size after a rejection
            step_frac = max(step_frac * 0.9, min_step_frac)

        # periodic full recomputation to eliminate accumulated linearisation error
        if total_iter % 5000 == 0:
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Re‑computed full conv, new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine performs cheap parametric sweeps,
    a large Dirichlet global search, and finishes with a deterministic
    hill‑climbing phase.  The best sequence found (normalised to sum = 1) is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Fast parametric sweeps   (≈200 s)
    # --------------------------------------------------------------
    phase1_deadline = min(start_time + 200.0, deadline - 500.0)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase1_deadline)
    if time.time() < phase1_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase1_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws   (≈300 s)
    # --------------------------------------------------------------
    phase2_deadline = min(start_time + 500.0, deadline - 300.0)
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase2_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Deterministic hill‑climbing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climbing")
        best_seq, best_score = _hill_climb(best_seq, best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Hill‑climbing finished, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
