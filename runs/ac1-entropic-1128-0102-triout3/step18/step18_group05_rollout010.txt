```
def proj_simplex(v):
    # project v onto non-negative simplex sum=1
    # returns v_proj
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
```

The score of a sequence `a` of length `n` is  

\[
F(a)=\frac{2n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the *full* linear convolution of `a` with itself.
Because the score is scale‑invariant we may (and do) keep the
vector on the probability simplex (`sum(a)=1`, `a_i≥0`).  
The remaining task is to minimise the peak of the convolution.

`max(a*a)` is the maximum of a family of linear functions of `a`.  
Hence it is a convex, piece‑wise‑linear function and its sub‑gradient at
a point is a convex combination of the gradients of the *active*
constraints (the convolution entries attaining the maximum).  
For a fixed shift `k` the gradient of the entry `c_k = (a*a)[k]` is  

\[
\frac{\partial c_k}{\partial a_i}=2\,a_{k-i}\qquad
(0\le i<n,\;0\le k-i<n),
\]

otherwise it is zero.  Using this sub‑gradient we perform a *projected
sub‑gradient descent*:

1. start from the best known sequence (`height_sequence_1` if it exists,
   otherwise uniform);
2. at each iteration compute the convolution, locate the active indices,
   build the sub‑gradient, take a step `a ← a – η·grad`,
   and project back onto the simplex;
3. use a simple back‑tracking line‑search on the step‑size multiplier
   `η` and keep the move only if the score improves;
4. after the sub‑gradient phase run a cheap random local polishing
   (moving a tiny amount of mass between two positions).

All expensive operations are FFT‑based convolutions (`O(n log n)`), the
gradient itself is `O(|active|·n)` and usually `|active|=1`.  The whole
procedure easily fits into the 1000 s budget and, starting from the
previous best construction, typically drives the score below the target
`1.503`.

```python
"""
Projected sub‑gradient descent for minimising the peak self‑convolution
of a non‑negative step function.

The algorithm:
    * initialise from the global best vector (height_sequence_1) or uniform;
    * repeat:
        - compute full convolution (FFT);
        - find the active indices where the maximum is attained;
        - build the sub‑gradient of the max‑convolution;
        - back‑track line‑search on a projected sub‑gradient step;
        - keep the best vector seen;
    * finish with a cheap random pairwise mass‑transfer polish.
The code respects the 1000 s time budget and returns a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, max_val: float, active: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of M(a)=max(a*a) at the current point.
    active – array of convolution indices where the maximum is attained.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        # overlap region i ∈ [max(0,k-n+1), min(k,n-1)]
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)          # inclusive
        # slice of a needed for this k
        # a_{k-i} for i = i_start … i_end is a reversed slice
        if k < n:
            # a_start = a[0:k+1] reversed
            grad[: k + 1] += 2.0 * a[k :: -1]   # a[k],…,a[0]
        else:
            start = k - n + 1
            # a[start:n] reversed
            grad[start:] += 2.0 * a[start:][::-1]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low score."""
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # a tiny safety margin

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Initial vector (normalised to sum = 1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a = np.maximum(starter, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(starter.size, 1.0 / starter.size)
            n = a.shape[0]
            print(f"[{time.time() - start_time:.2f}s] Loaded global starter (n={n})")
        else:
            raise ValueError
    except Exception:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=float)
        print(f"[{time.time() - start_time:.2f}s] No starter – using uniform (n={n})")

    best_f = a.copy()
    best_score = _objective_score(best_f)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Projected sub‑gradient descent
    # --------------------------------------------------------------
    max_iter = 200_000          # upper bound – will stop on time
    eta_initial = 0.5           # initial step multiplier
    min_eta = 1e-6
    eps_rel = 1e-9              # tolerance for the active set

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        # current convolution and maximum
        conv = _conv_self(a)
        M = conv.max()

        # keep the best solution seen so far
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_f = a.copy()
            best_score = cur_score
            print(f"[{time.time() - start_time:.2f}s] iter {iteration} new best = {best_score:.6f}")

        # indices where the maximum is (almost) attained
        tol = max(eps_rel * M, 1e-12)
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max‑convolution
        grad = _subgradient(a, M, active)

        # step‑size schedule (diminishing)
        eta = eta_initial / np.sqrt(iteration)

        # back‑tracking line search – try decreasing multipliers
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            step = eta * factor
            cand = a - step * grad
            cand = _project_simplex(cand)
            cand_score = _objective_score(cand)
            if cand_score < best_score - 1e-12:
                a = cand
                best_f = cand
                best_score = cand_score
                improved = True
                print(f"[{time.time() - start_time:.2f}s] iter {iteration} "
                      f"factor {factor:.4g} → score {best_score:.6f}")
                break

        if not improved:
            # occasional random shake‑up to escape flat regions
            if rng.random() < 0.01:
                i, j = rng.integers(n, size=2)
                if i != j and a[i] > 1e-15:
                    delta = a[i] * rng.random() * 0.1   # up to 10 % of source mass
                    tmp = a.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_score = _objective_score(tmp)
                    if tmp_score < best_score:
                        a = tmp
                        best_f = tmp
                        best_score = tmp_score
                        print(f"[{time.time() - start_time:.2f}s] random shake‑up "
                              f"→ score {best_score:.6f}")
                        continue
            # if nothing helped, shrink the global step factor
            eta_initial *= 0.9
            if eta_initial < min_eta:
                print(f"[{time.time() - start_time:.2f}s] step size exhausted – stopping")
                break

    # --------------------------------------------------------------
    # Final cheap stochastic polishing (pairwise mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print(f"[{time.time() - start_time:.2f}s] Starting final refinement")
        cur = best_f.copy()
        cur_score = best_score
        while time.time() < deadline:
            i, j = rng.integers(n, size=2)
            if i == j or cur[i] < 1e-15:
                continue
            delta = cur[i] * rng.random() * 0.01      # ≤1 % of source mass
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand_score = _objective_score(cand)
            if cand_score < cur_score - 1e-12:
                cur = cand
                cur_score = cand_score
                print(f"[{time.time() - start_time:.2f}s] refinement improved to {cur_score:.6f}")
        best_f = cur
        best_score = cur_score

    # --------------------------------------------------------------
    # Normalisation and return
    # --------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0.0:
        best_f = best_f / total
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_f.tolist()]
```
