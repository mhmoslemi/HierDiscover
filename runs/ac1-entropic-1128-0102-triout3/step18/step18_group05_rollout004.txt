"""
Gradient‑driven local optimisation for the step‑function bound.

The evaluation function is
    F(a) = 2·n·max(conv(a,a)) / (∑a)² .
We keep the vector normalised (∑a = 1) so that the score is
    2·n·max(conv(a,a)) .

Algorithm
---------
1. Initialise the sequence from the global variable ``height_sequence_1`` if it
   exists, otherwise from a uniform distribution.  Normalise to sum = 1.
2. Repeatedly perform **gradient‑driven moves**:
      * Compute the full self‑convolution by an FFT (O(n log n)).
      * Let k be an index where the convolution attains its maximum.
      * The sub‑gradient of the maximal entry w.r.t. a_i is 2·a_{k‑i}.
        Move a fraction of the mass from an index with a large sub‑gradient
        (source) to an index with a small sub‑gradient (target).
      * Several step‑sizes are tried; the best improving move is taken.
      * The step‑size decays from 0.5 to 0.01 during the run.
3. Occasionally (5 % of iterations) try a tiny random transfer between two
   positions – this helps to escape plateaus.
4. After the main loop finish a cheap random‑pair refinement (as used in the
   previous solution) to polish the last bits.
5. Clip to non‑negative, re‑normalise and return the list of floats.

The whole procedure is bounded by the ``budget_s`` argument (default 1000 s).
It needs only ``numpy`` and ``time``; ``cvxpy`` is not required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global start‑time (used for all sub‑routines)
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# FFT based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)
    return conv_full[: 2 * n - 1]


def _max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self_fft(a).max())


def _objective_score_fft(a: np.ndarray) -> float:
    """
    Score 2·n·max(conv(a,a)) assuming the vector is normalised
    (∑a == 1).  The function is used during the search; the final
    sequence will be re‑normalised before returning.
    """
    n = a.shape[0]
    return 2.0 * n * _max_conv_fft(a)


# ----------------------------------------------------------------------
# Random local refinement (same idea as the previous baseline)
# ----------------------------------------------------------------------
def _random_local_refine(a: np.ndarray,
                         cur_score: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap polishing: move a tiny random fraction of mass and keep strict improvements."""
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or best[src] <= 1e-15:
            continue
        delta = best[src] * rng.random() * 0.005  # ≤0.5 % of source mass
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score_fft(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] local refine improved to {best_score:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
# Gradient‑driven optimisation
# ----------------------------------------------------------------------
def _gradient_descent_optimize(a: np.ndarray,
                               deadline: float,
                               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform many small transfers guided by the sub‑gradient of the maximal
    convolution entry.  Returns the best vector found and its score.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective_score_fft(cur)
    best = cur.copy()
    best_score = cur_score

    # fractions to try (large to small)
    fractions = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    # main loop
    while time.time() < deadline:
        # current convolution & maximal index
        conv = _conv_self_fft(cur)
        M = conv.max()
        k_max = int(conv.argmax())

        # sub‑gradient g_i = 2 * a_{k_max-i} (factor 2 irrelevant for ordering)
        i_arr = np.arange(n)
        j_arr = k_max - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(cur)
        grad[mask] = cur[j_arr[mask]]

        # viable sources: positive mass and positive gradient
        src_mask = (grad > 0) & (cur > 1e-15)
        src_candidates = np.where(src_mask)[0]
        if src_candidates.size == 0:
            break

        source = src_candidates[np.argmax(grad[src_candidates])]

        # target: smallest gradient, but not the same index as source
        all_except_src = np.arange(n) != source
        target = np.argmin(grad[all_except_src])
        # argmin works on the compressed array; need to map back
        target = np.arange(n)[all_except_src][target]

        improved = False
        best_local = None
        best_local_score = cur_score

        # try the predefined fractions (largest first)
        for f in fractions:
            delta = cur[source] * f
            if delta <= 0.0:
                continue
            cand = cur.copy()
            cand[source] -= delta
            cand[target] += delta
            cand_score = _objective_score_fft(cand)
            if cand_score < best_local_score - 1e-12:
                best_local = cand
                best_local_score = cand_score
                improved = True
                break  # accept the first successful (largest) move

        if improved and best_local is not None:
            cur = best_local
            cur_score = best_local_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] GD improved to {best_score:.6f}")
            continue

        # If no improvement with directed move, try a tiny random move (5 % chance)
        if rng.random() < 0.05:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src != dst and cur[src] > 1e-15:
                delta = cur[src] * rng.random() * 0.01  # up to 1 % of src mass
                cand = cur.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand_score = _objective_score_fft(cand)
                if cand_score < cur_score - 1e-12:
                    cur = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best = cur.copy()
                        best_score = cur_score
                        print(f"[{time.time() - start_time:.2f}s] GD random improved to {best_score:.6f}")
                    continue

        # No improvement found – shrink step size implicitly by continuing
        # The loop will keep trying until time runs out.
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the given
    evaluation.  The routine runs for at most ``budget_s`` seconds and
    returns the best sequence found (as a list of floats).
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Initialise the problem size and starting vector
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
            a = np.maximum(starter, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n)
            print(f"[{time.time() - start_time:.2f}s] Loaded global starter (n={n})")
        else:
            raise ValueError
    except Exception:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=float)
        print(f"[{time.time() - start_time:.2f}s] Using uniform start (n={n})")

    # initial score
    cur_score = _objective_score_fft(a)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Gradient‑driven optimisation phase
    # --------------------------------------------------------------
    best_vec, best_score = _gradient_descent_optimize(a, deadline, rng)

    # --------------------------------------------------------------
    # Final cheap stochastic polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print(f"[{time.time() - start_time:.2f}s] Starting final local refinement")
        best_vec, best_score = _random_local_refine(best_vec, best_score, deadline, rng)

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0.0:
        best_vec = best_vec / total
    else:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_vec.tolist()]
