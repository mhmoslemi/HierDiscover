def _conv_self(a):
    n = a.shape[0]
    L = _next_pow2(2*n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]

    rho = ind[cond][-1]

"""
Exponentiated‑gradient descent on a soft‑max approximation of the max‑convolution.
The algorithm works on the probability simplex (∑a = 1, a_i ≥ 0) and
adaptively reduces a smoothness parameter γ and a learning rate η during the
allocated time.  At each step we compute a
    w_k ∝ exp(γ·(conv(a,a)_k – max conv))
which is a soft‑max over the convolution values.  The gradient of the
soft‑max objective is
    ∇_i = 2·∑_k w_k·a_{k−i},
which we obtain efficiently by one FFT‑based convolution.
An exponentiated‑gradient update
    a_i ← a_i·exp(–η·∇_i)
preserves non‑negativity; we re‑normalise to stay on the simplex.
Only strict improvements of the true objective
    F(a)=2·n·max(conv(a,a))
are accepted.
Occasional tiny random mass transfers are tried when progress stalls,
providing a cheap stochastic polish.
The best vector seen during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors via FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv

# ----------------------------------------------------------------------
# Simplex projection (ℓ₁‑norm = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Objective (true evaluation function, scale‑invariant)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step function minimizing the given evaluation.
    Uses exponentiated‑gradient descent on a soft‑max smoothed max‑convolution.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Initialise from a known good construction if possible,
    # otherwise fall back to a uniform vector.

        n = starter.size
        a = np.maximum(starter, 0.0)

        if total > 0.0:

    except Exception:

        a = np.full(n, 1.0 / n, dtype=float)
        print(f"[{time.time() - start_time:.2f}s] No starter – using uniform, n={n}")

    best_a = a.copy()
    best_score = _objective_score(best_a)
    print(f"[{time.time() - start_time:.2f}s] Initial objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameter schedule (soft‑max temperature γ and learning rate η)
    # ------------------------------------------------------------------
    gamma_start, gamma_end = 5.0, 50.0      # start & final smoothness
    eta_start, eta_end = 0.5, 0.01        # start & final step size

    it = 0
    stagnation_counter = 0

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------

        cur_score = 2.0 * n * max_conv

        # keep global best

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        t = min(1.0, elapsed / budget_s)          # in [0,1]

    # ------------------------------------------------------------------
    while time.time() < deadline:
