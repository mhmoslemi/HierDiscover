"""
Propose a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

by a three‑phase scheme:

1️⃣ **Rich seeds** – uniform, geometric·power families, beta‑shapes,
   linear/power families and many Dirichlet draws.
   Keep the best few after a cheap evaluation.

2️⃣ **Smooth‑max L‑BFGS continuation** – optimise the differentiable surrogate

        fβ(a)=2·n·β⁻¹·log Σ exp(β·conv(a,a))

   for an increasing schedule of β (up to 256).  The probability vector
   a is represented by a soft‑max of free logits; gradients are obtained
   analytically via FFTs.

3️⃣ **Intensive local search** – a temperature‑controlled simulated‑annealing
   that moves mass from indices that most contribute to the current maximal
   convolution value to indices that contribute least.  The annealer is
   followed by exponentiated‑gradient and projected sub‑gradient polishing.
   All steps respect a 1000 s budget and the best sequence (normalised to sum = 1)
   is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b)   # sum(a) is 1 by construction


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type shape, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_linear_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Monotonically decreasing linear/power shape."""
    i = np.arange(n, dtype=np.float64)
    shape = (n - i) ** power
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max surrogate (L‑BFGS) class
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """Wrapper for the smooth‑max objective (soft‑max re‑parameterisation)."""

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        conv = _conv_self_fft(a, self.L_fft)
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates (exponentiated‑gradient) on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10  # safety margin

    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (if provided).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Generate a diverse pool of seeds.
    # ------------------------------------------------------------------
    seeds = [best_seq]  # start with uniform / starter already present

    # geometric × power sweeps
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9999), 12)
    p_vals = np.linspace(-6.0, 6.0, 25)
    lam_vals = np.linspace(0.0, 0.95, 6)
    for r in r_vals:
        for p in p_vals:
            base = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(base, lam))

    # beta‑type shapes (including strong U‑shapes)
    alpha_vals = np.linspace(0.2, 2.0, 10)
    beta_vals = np.linspace(0.2, 2.0, 10)
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # linear / power decreasing families
    for power in [0.5, 1.0, 2.0]:
        shape = _make_linear_shape(n, power)
        for lam in lam_vals:
            seeds.append(_mix_with_uniform(shape, lam))

    # random Dirichlet draws with varied concentration
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(8):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 6
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Smooth‑max L‑BFGS continuation.
    # ------------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256]
    max_iter_per_beta = 80

    for beta in beta_schedule:
        if time.time() >= deadline - 300:  # reserve time for polishing
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for i in range(len(pool)):
            if time.time() >= deadline - 300:
                break
            a0 = pool[i]
            eps = 1e-12
            x0 = np.log(a0 + eps)
            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={i} score={cand_score:.6f}")

            pool[i] = a_candidate
            pool_scores[i] = cand_score

    # ------------------------------------------------------------------
    # 3️⃣ Simulated‑annealing local search.
    # ------------------------------------------------------------------
    sa_deadline = deadline - 200  # leave ~200 s for polishing
    a_current = best_seq.copy()
    cur_score = best_score
    init_T = 0.05
    sa_start = time.time()

    while time.time() < sa_deadline:
        # current convolution & max index
        conv = _conv_self_fft(a_current, L_fft)
        max_idx = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[max_idx]

        # sub‑gradient at max index
        g = _subgrad_max_conv(a_current, max_idx)

        # choose source (high contribution) and destination (low contribution)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1

        # temperature schedule (linear decay)
        elapsed = time.time() - sa_start
        fraction = elapsed / (sa_deadline - sa_start)
        T = init_T * max(0.0, 1.0 - fraction)

        # step size proportional to temperature
        step_factor = max(1e-4, T)

        # proposal: move a small amount of mass from src to dst
        delta = step_factor * a_current[src] * rng.random()
        delta = min(delta, a_current[src])  # stay non‑negative
        a_prop = a_current.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        # evaluate proposal
        conv_prop = _conv_self_fft(a_prop, L_fft)
        prop_score = 2.0 * n * conv_prop.max()

        # Metropolis acceptance
        accept = False
        if prop_score < cur_score:
            accept = True
        else:
            if T > 1e-12:
                prob = np.exp(-(prop_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a_current = a_prop
            cur_score = prop_score
            if prop_score < best_score:
                best_score = prop_score
                best_seq = a_prop.copy()
                print(f"[{time.time() - start:.2f}s] SA update score={prop_score:.6f}")

        # occasional random jitter to escape plateaus
        if int(elapsed * 10) % 100 == 0:
            eps = 1e-4 * rng.random(n) * a_current
            if rng.random() < 0.5:
                a_current = np.clip(a_current + eps, 0.0, None)
            else:
                a_current = np.clip(a_current - eps, 0.0, None)
            a_current /= a_current.sum()

    # ------------------------------------------------------------------
    # 4️⃣ Exponentiated‑gradient polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 120, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣ Projected sub‑gradient descent polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 70, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 6️⃣ Final tiny hill‑climb.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 4e-3
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
