"""
Propose a candidate step‑function using a **projected sub‑gradient descent**
on the exact (non‑smoothed) objective

    F(a) = 2·n·max_k (a * a)[k]      with   Σ a_i = 1 ,  a_i ≥ 0 .

For a given shift k* where the convolution attains its maximum the gradient
of max_k (a*a)[k] w.r.t. a is simply

    ∇_a  (a*a)[k*] = 2·a shifted by k*    (g_j = 2·a_{k*‑j} if in range).

Thus each iteration:
    1. compute the full self‑convolution via an FFT,
    2. locate the maximal entry,
    3. form the sub‑gradient,
    4. take a step opposite to the gradient,
    5. project back onto the probability simplex.

The step size is adapted: we enlarge it after a successful improvement
and shrink it otherwise.  When no improvement is seen for many iterations
we restart from a fresh sparse Dirichlet sample to escape possible
local minima.  Finally a cheap random hill‑climb is performed until the
time budget expires.

The routine returns a list of non‑negative heights that (empirically)
improves the previous best bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer helpers
# ----------------------------------------------------------------------
_start_time = None  # set in propose_candidate


def _now() -> float:
    """Seconds elapsed since the start of the search."""
    return time.time() - _start_time


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 } .
    Reference: Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # fallback (should not happen)
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """
    Exact unsmoothed objective for a probability vector a (Σ a = 1):
        2·n·max_k (a*a)[k] .
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    a_pad = np.zeros(L, dtype=float)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            src, dst = rng.integers(n, size=2)
            if src != dst and a[src] > 1e-12:
                delta = a[src] * rng.random() * 0.005
                cand = a.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand = _project_simplex(np.maximum(cand, 0.0))
                cand_score = _objective_score(cand)
                if cand_score < best_score - improve_eps:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand
                    step = min(step * 1.1, 1.0)
                    no_improve = 0
                    print(f"[{_now():.2f}s] Light hill‑climb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final cheap random local refinement (pure hill‑climb)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    while time.time() < deadline:
        src, dst = rng.integers(n, size=2)
        if src == dst or a[src] <= 1e-15:
            continue
        delta = a[src] * rng.random() * 0.003
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[{_now():.2f}s] Final refinement improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (just in case) and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{_now():.2f}s] Finished – final objective ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
