def _smooth_max_grad(a, beta):
    n = a.shape[0]
    conv = _conv_self_fft(a, L_fft)  # length 2n-1
    max_c = conv.max()
    # compute weight = exp(beta*(conv - max_c))   (avoid overflow)
    exp_vals = np.exp(beta * (conv - max_c))
    weight = exp_vals / exp_vals.sum()
    # compute gradient via convolution of weight with a_rev
    a_rev = a[::-1]
    L2 = _next_pow2(len(weight) + n - 1)

def _adam_optimize(a0, max_time, start_time, best_seq, best_score, rng):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    iter = 0
    beta = 10.0
    while time.time() < max_time:
        iter += 1
        # maybe increase beta gradually:
        beta = min(500.0, 10.0 + iter * (500.0-10.0)/20000) # schedule
        # compute gradient
        grad = _smooth_max_grad(a, beta)
        # scale for O: e.g., grad = 2*n * grad
        # but we can incorporate scaling factor later.
        # apply Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**iter)
        v_hat = v / (1 - beta2**iter)
        # learning rate schedule: lr0 = 0.1 maybe
        lr = 0.2 / (np.sqrt(iter) + 1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_simplex(a)
        # evaluate objective using fast _eval_fast
        score = _eval_fast(a, L_fft)   # includes factor 2n * max conv
        if score < best_score:

            # small random jitter
            a = np.maximum(a + 1e-4 * rng.random(a.shape), 0)

def _upsample(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0)
    up /= up.sum()
    return up

def _smooth_weighted_grad(a, beta, L_fft):
    n = a.shape[0]
    conv = _conv_self_fft(a, L_fft)
    max_c = conv.max()
    # compute smooth weights normalized to sum 1
    # use exp(beta*(conv - max_c))
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # Compute gradient via convolution of w and reversed a
    a_rev = a[::-1]
    L2 = _next_pow2(len(w) + n - 1)
    W_fft = np.fft.rfft(w, n=L2)
    A_rev_fft = np.fft.rfft(a_rev, n=L2)
    conv_grad = np.fft.irfft(W_fft * A_rev_fft, n=L2)[: len(w) + n - 1]
    # gradient = 2 * conv_grad[n-1:n-1+n]
    grad = 2.0 * conv_grad[n-1:n-1+n]
    return grad
```

def _adam_smooth_opt(a0, L_fft, n, deadline, start_time, best_seq, best_score, rng):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    # initial beta parameter
    beta = 5.0
    # schedule: increase beta slowly
    while time.time() < deadline:
        it += 1
        # Increase beta linearly to maxBeta
        maxBeta = 200.0
        beta = min(maxBeta, beta + (maxBeta - beta) * 0.001) # can adapt
        # Compute gradient
        grad = _smooth_weighted_grad(a, beta, L_fft)
        # multiply by factor 2*n maybe:
        grad = grad * (2.0 * n)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        # Learning rate schedule
        lr0 = 0.01
        lr = lr0 / (np.sqrt(it) + 1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # Project onto simplex
        a = _project_simplex(a)
        # Evaluate
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} beta={beta:.2f} score={score:.6f}")
        # occasional jitter
        if it % 3000 == 0:
            # random small jitter
            jitter = rng.random(n) * 1e-4 * a
            a = np.maximum(a + jitter, 0)
            a = _project_simplex(a)
    return best_seq, best_score
```

def _smooth_max_grad(a, beta, L_fft):
    n = a.shape[0]
    conv = _conv_self_fft(a, L_fft)  # length 2n-1
    # compute weights as exp(beta * conv)
    # to avoid overflow, subtract max before exp
    maxc = conv.max()
    exp_vals = np.exp(beta * (conv - maxc))
    w = exp_vals / exp_vals.sum()
    # compute gradient = 2 * (conv of w and reversed a) at appropriate indices
    a_rev = a[::-1]
    L2 = _next_pow2(len(w) + n - 1)
    # Compute convolution using FFT
    W_fft = np.fft.rfft(w, n=L2)
    A_rev_fft = np.fft.rfft(a_rev, n=L2)
    full = np.fft.irfft(W_fft * A_rev_fft, n=L2)[:len(w) + n - 1]
    grad = 2.0 * full[n-1:n-1+n]
    return grad
```

"""
Hybrid optimisation with smooth max‑convolution gradient & aggressive SA.

The search is performed in four phases (all respecting the 1000 s budget):

1️⃣  Fast parametric sweep over two analytic families (monotone Geo‑Power
    and symmetric “U‑shape”).  A dense grid of (λ,p) is explored and the
    best vector (L1‑normalised) is kept.

2️⃣  Smooth‑max gradient descent (Adam).  The non‑smooth max‑convolution
    is replaced by a softmax (parameter β) which is annealed from a
    small value to a large one.  The gradient of the smoothed objective
    is computed via two FFT‑based convolutions, an Adam step is taken,
    and the iterate is projected onto the probability simplex.

3️⃣  Optional mixing with the reverse of the current best vector – this
    reduces the dominant inner‑product 〈a,rev(a)〉 that often limits
    the objective.

4️⃣  Simulated‑annealing hill‑climb with larger mass‑transfer moves and
    occasional random jitter to escape plateaus.

The routine finally returns the best sequence found (sum normalised to 1)
as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (assumes a is already L1‑normalised)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(‑lam·i).  L1‑normalised."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape” a_i ∝ (i+1)^p·exp(‑lam·i) + (n‑i)^p·exp(‑lam·(n‑1‑i))."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _make_two_exp(
    n: int, lam1: float, lam2: float, w: float
) -> np.ndarray:
    """
    Two‑exponential mixture:
        a_i ∝ w·exp(‑lam1·i) + (1‑w)·exp(‑lam2·i)
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    left = w * np.exp(-lam1 * i)
    right = (1.0 - w) * np.exp(-lam2 * i)
    shape = left + right
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max over all convolution lags)
# ----------------------------------------------------------------------


def _smooth_max_grad(a: np.ndarray, beta: float, L_fft: int) -> np.ndarray:
    """
    Gradient of the β‑smooth approximation of max(conv(a,a)).
    Returns a vector of length n.
    """
    n = a.shape[0]
    conv = _conv_self_fft(a, L_fft)               # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - max_c))
    w /= w.sum()                                   # shape (2n‑1,)

    # gradient = 2·(w * a_rev) convolved, extracted at appropriate offsets
    a_rev = a[::-1]
    L2 = _next_pow2(len(w) + n - 1)
    W_fft = np.fft.rfft(w, n=L2)
    A_rev_fft = np.fft.rfft(a_rev, n=L2)
    full = np.fft.irfft(W_fft * A_rev_fft, n=L2)[: len(w) + n - 1]
    grad = 2.0 * full[n - 1 : n - 1 + n]           # length n
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max objective
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    L_fft: int,
    n: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max approximation.
    β is annealed from 1 up to ~200.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0
    beta = 1.0
    beta_target = 200.0

    while time.time() < deadline:
        it += 1

        # anneal β (slowly increase)
        beta = min(beta_target, beta + (beta_target - beta) * 0.0005)

        # smooth‑max gradient
        g = _smooth_max_grad(a, beta, L_fft)          # ∂F/∂a
        # scale to the original objective (2·n·F) – scaling absorbed in step size
        g = g * (2.0 * n)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule
        lr0 = 0.05
        lr = lr0 / (np.sqrt(it) + 1.0)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step
        a = _project_simplex(a)

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} β={beta:.1f} "
                f"score={score:.6f}"
            )

        # occasional jitter to avoid stagnation
        if it % 2500 == 0:
            epsj = 5e-5 * rng.random(n) * a
            a = np.maximum(a + epsj, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (aggressive mass transfer)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny + medium mass‑transfer moves with a linear cooling schedule."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    T0 = 2e-3
    Tf = 1e-7

    while time.time() < deadline:
        # linear temperature schedule
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        # pick two distinct indices
        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer up to 1 % of src mass (larger than baseline)
        eps = cur[src] * rng.random() * 1e-2
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        # occasional random jitter (global)
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = _project_simplex(cur)
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score


# ----------------------------------------------------------------------
# Mixing step – combine with reverse to reduce the dominant inner product
# ----------------------------------------------------------------------


def _mix_with_reverse(a: np.ndarray, n: int,
                      L_fft: int,
                      best_seq: np.ndarray,
                      best_score: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Explore convex combinations a←(1‑α)a+α·rev(a) for α∈[0,0.5]."""
    rev = a[::-1]
    for alpha in np.linspace(0.0, 0.5, 21):
        cand = (1.0 - alpha) * a + alpha * rev
        cand = _project_simplex(cand)
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[mix] α={alpha:.3f} improved score={sc:.6f}"
            )
    return best_seq, best_score
