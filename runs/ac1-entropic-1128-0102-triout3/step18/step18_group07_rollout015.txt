def _subgradient_descent(a0, score0, deadline):
    a = a0.copy()
    best_a, best_score = a.copy(), score0
    t = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # potential early exit if best_score already near global lower bound?
        indices = np.where(conv >= max_val - 1e-12)[0]
        # choose one index at random among them for gradient direction
        k = np.random.choice(indices)
        # compute gradient for that diagonal
        grad = np.zeros_like(a)
        lo = max(0, k - (len(a)-1))
        hi = min(len(a)-1, k)
        idx_i = np.arange(lo, hi+1, dtype=int)
        idx_j = k - idx_i
        grad[idx_i] = 2.0 * a[idx_j]   # as derived
        # step schedule
        t += 1
        step = 0.1 / np.sqrt(t)  # adapt
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()

        if cand_score < score0:
            a = cand

"""
Optimises a non‑negative step‑function to minimise the evaluation metric
`evaluate_sequence`.  The search is performed within a fixed time budget
(default 1000 s) and combines several complementary heuristics:

1️⃣  Random Dirichlet draws and a rich library of classic windows (sine,
    Hann, Hamming, Blackman, Kaiser) to obtain diverse initial candidates.

2️⃣  Alternating projection that enforces a flat Fourier magnitude while
    keeping the vector non‑negative and on the probability simplex.

3️⃣  Projected Adam optimisation on a smooth (log‑sum‑exp) approximation of
    the max‑convolution.

4️⃣  Direct sub‑gradient descent on the exact max‑convolution (the true
    objective) with a simple decreasing step‑size schedule.

5️⃣  A short random hill‑climbing phase (mass‑transfer moves) for final
    polishing.

All phases respect the overall deadline; the best sequence found is
returned as a plain list of floats (sum = 1, non‑negative).  The code
uses only NumPy and SciPy (for the Kaiser window) and runs on a single
CPU.
"""

import time
import numpy as np
from scipy.special import i0  # needed for the Kaiser window (via np.kaiser)

# ----------------------------------------------------------------------
# Global timer (set in propose_candidate)
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (sum forced to 1 → denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) used by Adam
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Implements  grad = 2 * (a_rev ⋆ w)[offset]  (w are softmax weights).
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)   # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad

# Deterministic windows (already normalised to the simplex)
# ----------------------------------------------------------------------
def _deterministic_shapes(n: int):
    """Yield a variety of classic windows."""
    # sine (half‑period of a sine, shifted to be non‑negative)
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hamming
    y = 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Blackman
    y = (0.42 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / (n - 1)))
    yield _project_simplex(np.maximum(y, 0.0))

    # Kaiser with several beta values
    for beta in [2.0, 4.0, 8.0, 14.0]:
        y = np.kaiser(n, beta)
        yield _project_simplex(np.maximum(y, 0.0))

    # Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

# Random Dirichlet trials
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = np.inf
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Alternating projection (flatten Fourier magnitude)
# ----------------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, max_iter: int, deadline: float):
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of padded a
        F = np.fft.rfft(a, n=L)

        # Desired flat magnitude: keep total energy constant
        energy = np.dot(a, a)                     # sum a_i^2
        flat_mag = np.sqrt(energy / L)

        # Replace magnitude, keep phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Inverse FFT and truncate to length n
        a_full = np.fft.irfft(F_flat, n=L)
        a = a_full[:n]

        # Projection onto feasible set
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] alt‑proj iter {i:5d} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on smooth max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, score0: float,
                  beta_vals, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    lr = 0.07
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            # accept if improves (or with a tiny random acceptance)
            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            # stop early if steps become negligible
            if np.linalg.norm(step) < 1e-13:
                break

    return best_a, best_score

# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # identify (all) maximal indices; pick one at random
        max_inds = np.flatnonzero(conv >= max_val - 1e-15)
        k = rng.choice(max_inds)

        # gradient of conv_k (the max diagonal) w.r.t. a:
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_i = np.arange(lo, hi + 1, dtype=int)
        idx_j = k - idx_i
        grad[idx_i] = 2.0 * a[idx_j]

        # simple decreasing step‑size
        t += 1
        step = 0.5 / (1.0 + t * 1e-3)   # starts ~0.5 and decays slowly

        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad iter {t:6d} new best {best_score:.6f}")

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        # occasional random jitter to escape flat regions
        if t % 2000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

# Simple random hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer up to 10 % of source mass
        delta = a[src] * rng.random() * 0.10
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # allow occasional uphill moves
            a = cand
            cur_score = cand_score

    return best_a, best_score
