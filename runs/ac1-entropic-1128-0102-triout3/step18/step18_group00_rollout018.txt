"""
Hybrid optimiser with a soft‑max Adam phase.

The search consists of three stages:

1️⃣  Quick parametric sweep (geometric‑power, symmetric and cosine families)
    to obtain a decent starting vector.

2️⃣  Adam optimisation on a smooth approximation of the max‑convolution:
    – a is represented as a soft‑max of unconstrained parameters θ,
    – the objective is the log‑sum‑exp of the self‑convolution (α controls
      the smoothness, α is increased slowly),
    – gradients are obtained via a linear correlation (FFT‑based, O(n log n)),
    – Adam with momentum accelerates convergence and escapes shallow plateaus.

3️⃣  A short simulated‑annealing hill‑climb (tiny mass transfers) to polish
    the final vector.

All stages respect the 1000 s budget.  The routine finally returns a plain
Python list of non‑negative floats (sum≈1) which can be fed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation: 2·n·max(conv(a,a)) / (sum(a)²).  Assumes a ≥ 0."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate vectors for the initial sweep
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(-lam·i), L1‑normalised."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape”: left part + mirrored right part, L1‑normalised."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    left = np.exp(log_shape)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _make_cosine_shape(n: int, c: float) -> np.ndarray:
    """Cosine dip/bump: a_i ∝ max(0, 1 + c·cos(π·i/(n‑1)))."""
    i = np.arange(n, dtype=np.float64)
    base = 1.0 + c * np.cos(np.pi * i / (n - 1))
    base = np.maximum(base, 0.0)
    if base.sum() == 0.0:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        base /= base.sum()
    return base


# ----------------------------------------------------------------------
# Adam optimiser over the soft‑max parametrisation
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    L_fft: int,
    Lc: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Optimise θ (soft‑max of a) with Adam on the smoothed max‑convolution.
    Returns the best vector seen and its (exact) score.
    """
    n = a0.shape[0]
    len_w = 2 * n - 1
    offset = len_w - 1

    # initialise θ from the starting vector (log‑softmax)
    theta = np.log(a0 + 1e-30)

    m = np.zeros_like(theta)
    v = np.zeros_like(theta)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.15

    # smoothing parameter α
    alpha = 5.0
    alpha_max = 1500.0
    alpha_inc = 1.001

    best_seq = a0.copy()
    best_score = _eval_fast(best_seq, L_fft)

    it = 0
    beta1_pow = 1.0
    beta2_pow = 1.0

    # buffers to avoid re‑allocation
    a_pad = np.zeros(Lc, dtype=np.float64)
    w_pad = np.zeros(Lc, dtype=np.float64)

    while time.time() < deadline:
        it += 1

        # ----- a = soft‑max(θ) -----
        a_exp = np.exp(theta)
        a = a_exp / a_exp.sum()               # L1‑normalised
        a_pad[:n] = a
        a_pad[n:] = 0.0

        # ----- convolution a*a (FFT length Lc) -----
        Fa = np.fft.rfft(a_pad, n=Lc)
        conv_full = np.fft.irfft(Fa * Fa, n=Lc)
        conv = conv_full[: 2 * n - 1]          # linear self‑convolution

        max_conv = conv.max()
        cur_score = float(2.0 * n * max_conv)  # sum(a)=1 ⇒ denominator =1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                f"score={cur_score:.6f}"
            )

        # ----- soft‑max over the convolution entries -----
        # w_i ≈ exp(α·conv_i)   (log‑sum‑exp approximation of max)
        shift = max_conv                       # for numerical stability
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        w /= w_sum

        # ----- linear correlation (a ⋆ w) -----
        w_pad[:len_w] = w
        w_pad[len_w:] = 0.0
        Fw = np.fft.rfft(w_pad, n=Lc)
        corr_full = np.fft.irfft(Fa * np.conj(Fw), n=Lc)
        grad_a = 2.0 * corr_full[offset : offset + n]   # length n

        # ----- gradient w.r.t. θ (soft‑max Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_theta = a * (grad_a - dot)

        # ----- Adam update -----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- increase smoothness (α) slowly -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)

        # occasional jitter to shake out of flat plateaus
        if it % 2000 == 0:
            jitter = 1e-4 * rng.standard_normal(theta.shape)
            theta += jitter

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny mass‑transfer simulated annealing (same as benchmark)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with a linear cooling schedule."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score

    n = cur.shape[0]
    T0 = 1.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        # linear cooling based on fraction of budget used
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 0.5 % of the source mass
        eps = cur[src] * rng.random() * 5e-3
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        # occasional random jitter
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur /= cur.sum()
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score

        for p in p_grid:
            # monotone
            log_shape = base + p * log_i
            log_shape -= np.max(log_shape)
            mono = np.exp(log_shape)
            mono /= mono.sum()
            sc = _eval_fast(mono, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = mono.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep mono λ={lam:.6e} "
                    f"p={p:.2f} score={sc:.6f}"
                )
            # symmetric
            sym = mono + mono[::-1]
            sym /= sym.sum()
            sc_sym = _eval_fast(sym, L_fft)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep sym λ={lam:.6e} "
                    f"p={p:.2f} score={sc_sym:.6f}"
                )

    # cosine family
    c_grid = np.linspace(-0.95, 0.95, 41)
    for c in c_grid:
        if time.time() >= sweep_deadline:
            break
        cos_seq = _make_cosine_shape(n, c)
        sc = _eval_fast(cos_seq, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cos_seq.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sweep cosine c={c:.2f} score={sc:.6f}"
            )

    # --------------------------------------------------------------
    # 2️⃣ Adam optimisation on a soft‑max approximation of the max‑conv.
    # --------------------------------------------------------------
    adam_deadline = deadline - 150.0
    if time.time() < adam_deadline:
        best_seq, best_score = _adam_optimize(
            best_seq, L_fft, Lc, adam_deadline, start_time, rng
        )

    # --------------------------------------------------------------
    # 3️⃣ Final simulated‑annealing hill‑climb.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, L_fft, deadline, start_time, rng
        )

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
