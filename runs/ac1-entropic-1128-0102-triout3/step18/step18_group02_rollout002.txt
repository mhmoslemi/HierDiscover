"""
Improved search for a non‑negative step function minimizing

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm:

1️⃣  Initialise from the provided global solution or from a few random
    Dirichlet / beta‑shape seeds (all normalised to sum = 1).

2️⃣  Run a continuation of L‑BFGS on the smooth‑max surrogate
    f_β(a)=2·n·β⁻¹·log∑exp(β·conv(a,a)).
    The schedule uses β∈{1,2,4,…,1024} and a generous number of
    iterations per β for a deep local optimisation.

3️⃣  Perform a long projected sub‑gradient descent on the exact
    L∞‑objective, with a back‑tracking line search and occasional
    random mass‑transfer moves (simulated‑annealing‑style).

All phases respect the 1000 s time budget; the best sequence found is
returned as a plain Python list (sum → 1)."""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a∗a via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k·conv_k(a) w.r.t. a.
    Returns 2·∑_j a_j·p_{i+j} for i=0,…,n‑1.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families (used for a few extra seeds)
# ----------------------------------------------------------------------


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: (i+1)^{α‑1}·(n‑i)^{β‑1}, normalised.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used for L‑BFGS)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable object evaluating the smooth‑max surrogate

        fβ(a)=2·n·β⁻¹·log Σ_k exp(β·conv_k(a))

    and its gradient w.r.t. the unconstrained logits x,
    where a = softmax(x).
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Safe soft‑max conversion."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        # convolution a*a
        conv = _conv_self_fft(a, self.L_fft)

        # soft‑max weights over β·conv
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a of the smooth‑max term
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule for logits
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact, used in polishing)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 5
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS continuation (deep optimisation).
    # ------------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    max_iter_per_beta = 200   # fairly large, still cheap

    for beta in beta_schedule:
        if time.time() >= deadline - 60:          # reserve ~1 min for final polishing
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, float(beta))
        for idx in range(len(pool)):
            if time.time() >= deadline - 60:
                break
            a_start = pool[idx]
            eps = 1e-12
            x0 = np.log(a_start + eps)

            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            a_new = np.exp(res.x - np.max(res.x))
            a_new /= a_new.sum()
            cand_score = _eval_fast(a_new, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} "
                      f"score={cand_score:.6f}")
            pool[idx] = a_new

    # ------------------------------------------------------------------
    # 3️⃣  Detailed local refinement (sub‑gradient + random moves).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # back‑tracking line‑search on the exact objective
        step = 1.0 / np.sqrt(it)          # decreasing step size
        improved = False
        for _ in range(8):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5

        if not improved:
            # occasional random mass‑transfer move (simulated‑annealing flavour)
            if rng.random() < 0.5:
                src = rng.integers(n)
                if a[src] < 1e-15:
                    continue
                dst = rng.integers(n)
                if src == dst:
                    continue
                eps = a[src] * rng.random() * 0.02   # ≤2 % of source mass
                prop = a.copy()
                prop[src] -= eps
                prop[dst] += eps
                prop = _project_simplex(prop)
                prop_score = _eval_fast(prop, L_fft)
                if prop_score < best_score:
                    best_score = prop_score
                    best_seq = prop.copy()
                    a = prop
                    print(f"[{time.time() - start:.2f}s] random move it={it:,} "
                          f"score={prop_score:.6f}")

        # small jitter every few thousand iterations to shake stagnation
        if it % 2000 == 0:
            jitter = rng.random(n) * 1e-4 * a
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
