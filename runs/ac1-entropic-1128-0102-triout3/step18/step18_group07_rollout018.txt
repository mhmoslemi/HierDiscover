def _subgradient_descent(a0, score0, deadline):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    iter = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # tolerance: small relative
        mask = conv >= max_val * (1 - 1e-8)
        # Compute gradient as average of subgradients for each active k
        grad = np.zeros_like(a)
        ks = np.where(mask)[0]
        for k in ks:
            # gradient of conv_k: g_i = 2 * a_{k - i}
            # vectorized: produce shifted version
            idx = np.arange(len(a))
            partner = k - idx
            valid = (partner >= 0) & (partner < len(a))
            grad[valid] += 2 * a[partner[valid]]
        # average
        if ks.size > 0:
            grad /= ks.size
        # step size schedule
        step_size = 0.5 / np.sqrt(iter + 1)
        a_new = a - step_size * grad
        # project onto simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        # evaluate
        new_score = _objective_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] Subgrad iter {iter} new best {best_score:.6f}")
        a = a_new
        cur_score = new_score
        iter += 1
        # early stop if step size small or gradient small
        if np.linalg.norm(grad) < 1e-12:
            break
    return best_a, best_score
```

    if not mask.any(): return np.zeros_like(a)

"""
Optimiser for 2·n·max(conv(a,a))/(sum(a)²).

Main ideas
---------
* Use a larger problem size (n = 8192) – the theoretical lower bound on the
  objective is 2·n/(2·n‑1) ≈ 1, so a larger n gives more “room” for improvement.
* Add a dedicated random‑binary‑support search (half the entries are
  1/m, the rest are 0).  For a support of size ≈ n/2 the expected maximum
  of the self‑convolution is close to the average 1/(2n), yielding an
  objective near 1.0.
* Keep the previous pipeline (Dirichlet draws, deterministic windows,
  alternating‑projection, Adam on a smooth max) and extend it with a
  simple sub‑gradient descent that directly minimises the true
  ∞‑norm of the convolution.
* All phases respect the global 1000 s budget; time is split
  adaptively according to the remaining budget.

The function ``propose_candidate`` returns a plain list of non‑negative
floats.  The sequence is normalised to sum to 1 (the objective is
scale‑invariant) and is compatible with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used for progress messages)
# ----------------------------------------------------------------------
_start_time = None

def _now():
    return time.time() - _start_time

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Objective (sum is forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))

# ----------------------------------------------------------------------
# Smooth max (used by Adam)
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta

def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max w.r.t. a (softmax weighting)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()                     # shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)     # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad

# ----------------------------------------------------------------------
# Phase 0 – random Dirichlet draws
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{_now():.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score

# ----------------------------------------------------------------------
# Phase 0 – random binary support trials
# ----------------------------------------------------------------------
def _random_binary_trials(n: int, density: float, max_trials: int, deadline: float):
    """
    Choose a random subset of indices of size round(density*n) and
    assign weight 1/m to the selected positions (so the vector sums to 1).
    """
    rng = np.random.default_rng()
    m = max(1, int(round(density * n)))
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / m
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{_now():.2f}s] Binary density={density:.2f} score={sc:.6f}")
    return best_seq, best_score

# ----------------------------------------------------------------------
# Deterministic windows (used as starters)
# ----------------------------------------------------------------------
def _deterministic_shapes(n: int):
    """Yield a few classic windows (already normalised to the simplex)."""
    # sine
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # a few Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

    # Blackman window
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    t = np.arange(n)
    y = a0 - a1 * np.cos(2 * np.pi * t / (n - 1)) + a2 * np.cos(4 * np.pi * t / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

# ----------------------------------------------------------------------
# Phase 2 – alternating projection (flatten Fourier magnitude)
# ----------------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, max_iter: int, deadline: float):
    """
    Replace the magnitude of the FFT of a by a constant while keeping a
    non‑negative and on the probability simplex.
    """
    n = a0.shape[0]
    L = 2 * n - 1                         # full convolution length (no power‑of‑2 padding)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of zero‑padded a
        F = np.fft.rfft(a, n=L)

        # Flat magnitude (same total energy for all frequencies)
        energy = np.dot(a, a)            # sum of squares
        flat_mag = np.sqrt(energy / L)

        # Replace magnitude, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Back to time domain and truncate
        a_full = np.fft.irfft(F_flat, n=L)
        a = a_full[:n]

        # Feasibility projection
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{_now():.2f}s] alt‑proj iter {i:5d} new best {best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Phase 3 – Adam optimiser on smooth max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, score0: float,
                  beta_vals, deadline: float):
    """Projected Adam optimiser on the smooth max of the convolution."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    lr = 0.07
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{_now():.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            # early stop when step is tiny
            if np.linalg.norm(step) < 1e-13:
                break
    return best_a, best_score

# ----------------------------------------------------------------------
# Phase 4 – sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, score0: float, deadline: float):
    """
    Direct sub‑gradient method on  max(conv(a,a)).
    Uses the exact sub‑gradient (average over all active indices) and a
    decaying step‑size.  The iterate is projected onto the simplex after
    each step.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # indices attaining (almost) the max – a small relative tolerance
        mask = conv >= max_val * (1.0 - 1e-9)
        if not np.any(mask):
            break
        # uniform weighting over active indices gives a valid sub‑gradient
        w = np.zeros_like(conv)
        w[mask] = 1.0 / mask.sum()
        rev_corr = _convolve_fft(a[::-1], w)
        grad = 2.0 * rev_corr[n - 1 : n - 1 + n]

        # step‑size schedule  η_t = η0 / sqrt(t+1)
        eta0 = 0.3
        step = eta0 / np.sqrt(it + 1.0) * grad

        cand = a - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{_now():.2f}s] sub‑grad iter {it:5d} new best {best_score:.6f}")

        a = cand
        cur_score = cand_score
        it += 1

        if np.linalg.norm(step) < 1e-14:
            break

    return best_a, best_score

# ----------------------------------------------------------------------
# Phase 5 – simple random hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer up to 10 % of source mass
        delta = a[src] * rng.random() * 0.10
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{_now():.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

    return best_a, best_score

    print(f"[{_now():.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
