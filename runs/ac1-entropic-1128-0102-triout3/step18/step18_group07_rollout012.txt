def _subgradient_descent(a0, score0, deadline):
    a = a0.copy()
    best = a.copy()
    best_score = score0
    t = 0
    eta0 = 0.5 # base step size maybe
    # Momentum
    m = np.zeros_like(a)
    beta_m = 0.9
    while time.time() < deadline:
        conv = _conv_self(a)
        k_max = conv.argmax()
        max_val = conv[k_max]
        # compute gradient (shift)
        idx = k_max - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        # momentum update
        m = beta_m * m + (1 - beta_m) * grad
        # step size schedule
        t += 1
        eta = eta0 / np.sqrt(t)
        # additive update
        cand = a - eta * m
        # clip negatives and project
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        # evaluate
        cand_score = _objective_score(cand)
        # multiplicative update candidate
        # compute exponentials
        exp_factor = np.exp(-eta * grad)
        cand2 = a * exp_factor
        cand2 = _project_simplex(cand2)  # still sum constraint; but need to renormalize: after multiplication, we can renorm:
        # actually multiply then renorm:
        cand2 = cand2 / cand2.sum()
        cand2_score = _objective_score(cand2)
        # choose best of current candidates:
        if cand2_score < cand_score:
            cand, cand_score = cand2, cand2_score
        # optionally accept if improves, else maybe accept probabilistically
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] subgrad new best {best_score:.6f}")
        # acceptance criteria: accept if cand_score < current_score else maybe random
        # to avoid stuck we can accept if improvement else with prob 0.001
        if cand_score < score0 or np.random.rand() < 0.001:
            a = cand
            score0 = cand_score
        # early break if gradient magnitude small (norm)
        if np.linalg.norm(m) * eta < 1e-12:
            break
    return best, best_score

def _max_conv_and_grad(a):
    conv = _conv_self(a)
    idx = conv.argmax()
    # compute gradient shift: g = np.zeros_like(a)
    # use vectorized index
    n = a.shape[0]
    i = np.arange(n)
    # positions in a that correspond to idx - i:
    rev_idx = idx - i
    mask = (rev_idx >= 0) & (rev_idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[rev_idx[mask]]
    return conv, idx, grad

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    # size
    n = 4096   # maybe use config param for n from kwargs if present.

    # baseline uniform
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective_score(best_seq)

    # Phase 1 random Dirichlet (increase trials) for up to min(30s, time left)
    phase1_deadline = min(deadline, start_time + 30)
    seq, sc = _random_dirichlet_trials(n, max_trials=5000, deadline=phase1_deadline)
    if seq is not None and sc < best_score:
        best_seq, best_score = seq, sc

    for shp in _deterministic_shapes(n):
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc

    # Phase 2 subgradient descent (70% of remaining time)
    if time.time() < deadline:
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + 0.7 * remaining
        # number of restarts
        restarts = 5
        for r in range(restarts):
            if time.time() >= subgrad_deadline:
                break
            # jitter around best_seq
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.01, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0)
            a0 = _project_simplex(a0)
            a_opt, sc_opt = _subgradient_descent(a0, _objective_score(a0), subgrad_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time()-start_time:.2f}s] subgrad restart {r} new best {best_score:.6f}")

    # Phase 3 hill climbing final  (remaining time)
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # Normalize again
    best_seq = np.maximum(best_seq, 0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0/n)

    print(f"[{time.time()-start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]

"""
Propose a non‑negative step‑function (vector) that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The optimiser combines three simple phases:

1️⃣  Quick Dirichlet / deterministic window sweeps to obtain a good
    starting point (uniform, sine, Hann, triangular, Gaussian).

2️⃣  A large‑scale *exact* sub‑gradient descent on the true max‑norm.
    The gradient of max(conv) is just a shifted copy of the current vector,
    so each iteration costs a single FFT (to locate the maximal shift) and a
    O(n) update.  An additive + multiplicative (mirror‑descent) update with
    Euclidean projection onto the simplex is used, together with a
    diminishing step‑size and a momentum term.  Many restarts with small
    random jitter are performed while time permits.

3️⃣  A short random mass‑transfer hill‑climb to polish the final answer.

All work is limited to the supplied 1000 s budget.  The returned list
contains non‑negative floats that sum to 1 (the scale is irrelevant for
the evaluation function).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used by helpers for progress messages)
# ----------------------------------------------------------------------
_start_time = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective (the denominator is forced to 1, so we minimise 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Phase 1 – random Dirichlet draws + deterministic windows
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Search a few Dirichlet‑distributed vectors."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        conc_log = rng.uniform(-3.0, 0.7)          # log10 of concentration
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a few classic windows (already normalised to the simplex)."""
    # sine
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # a few Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))


# ----------------------------------------------------------------------
# Phase 2 – exact sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         init_score: float,
                         deadline: float,
                         max_restarts: int = 6):
    """
    Perform many sub‑gradient steps on the true max‑norm.
    Uses a simple additive update (with momentum) and a multiplicative
    mirror‑descent update; the better of the two is kept.
    """
    n = a0.shape[0]
    best_a = a0.copy()
    best_score = init_score

    for restart in range(max_restarts):
        if time.time() >= deadline:
            break

        # jitter the start of each restart
        rng = np.random.default_rng()
        jitter = rng.normal(scale=0.01, size=n)
        a = best_a + jitter
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective_score(a)
        momentum = np.zeros_like(a)
        beta_m = 0.9          # momentum factor
        eta0 = 0.6            # initial step size
        t = 0

        while time.time() < deadline:
            # one full convolution to locate the worst shift
            conv = _conv_self(a)
            k_max = int(conv.argmax())

            # gradient = 2 * shifted copy of a (zero outside the interval)
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[idx[mask]]

            # momentum update
            momentum = beta_m * momentum + (1.0 - beta_m) * grad
            t += 1
            eta = eta0 / np.sqrt(t)          # diminishing step size

            # ---------- additive candidate ----------
            cand_add = a - eta * momentum
            cand_add = np.maximum(cand_add, 0.0)
            cand_add = _project_simplex(cand_add)
            cand_add_score = _objective_score(cand_add)

            # ---------- multiplicative (mirror‑descent) candidate ----------
            # a_i <- a_i * exp(-eta * grad_i) , then renormalise
            exp_factor = np.exp(-eta * grad)
            cand_mul = a * exp_factor
            cand_mul_sum = cand_mul.sum()
            if cand_mul_sum > 0:
                cand_mul = cand_mul / cand_mul_sum
            else:
                cand_mul = a.copy()
            cand_mul_score = _objective_score(cand_mul)

            # pick the better candidate of the two
            if cand_mul_score < cand_add_score:
                cand, cand_score = cand_mul, cand_mul_score
            else:
                cand, cand_score = cand_add, cand_add_score

            # record global best
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - _start_time:.2f}s] subgrad restart {restart} new best {best_score:.6f}")

            # simple acceptance rule (always keep improvement, occasional uphill)
            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            # stop if updates become vanishingly small
            if np.linalg.norm(eta * momentum) < 1e-14:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – simple random hill‑climbing (mass transfer) for polishing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    """Tiny random mass‑transfer hill‑climb."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        delta = a[src] * rng.random() * 0.10   # move up to 10 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - _start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

    return best_a, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising `evaluate_sequence`.
    Returns a plain Python list of floats (probability vector).
    """
    global _start_time
    np.random.seed(seed)
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size – a compromise between richness and speed
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - _start_time:.2f}s] uniform start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (if the runner provides one)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - _start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random Dirichlet draws + deterministic windows (≈30 s)
    # --------------------------------------------------------------
    phase1_deadline = min(deadline, _start_time + 30.0)
    if time.time() < phase1_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=4000, deadline=phase1_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - _start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

        for shp in _deterministic_shapes(n):
            sc = _objective_score(shp)
            if sc < best_score:
                best_seq, best_score = shp.copy(), sc
                print(f"[{time.time() - _start_time:.2f}s] deterministic shape score {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent on max(conv) (≈55 % of remaining)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + 0.55 * remaining
        best_seq, best_score = _subgradient_descent(
            best_seq,
            _objective_score(best_seq),
            subgrad_deadline,
            max_restarts=8,
        )
        print(f"[{time.time() - _start_time:.2f}s] after sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final hill‑climb polishing (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - _start_time:.2f}s] starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - _start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
