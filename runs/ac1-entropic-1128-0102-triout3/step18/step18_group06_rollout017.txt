```
def _gradient_softmax(a: np.ndarray, alpha: float) -> np.ndarray:
    n = a.shape[0]
    conv = _conv_self(a)  # length 2n-1
    # scale for numeric stability
    max_conv = conv.max()
    exp_term = np.exp(alpha * (conv - max_conv))
    sum_exp = exp_term.sum()
    # softmax weights
    p = exp_term / sum_exp
    # conv1 = sum_i p_i * a_{i - j}:
    # Using convolution: conv_rev = np.convolve(p[::-1], a)
    # conv1_j = conv_rev[(2*n-2) - j]

    conv_rev = np.convolve(p_rev, a)
    conv1 = conv_rev[(2*n-2) - np.arange(n)]
    # gradient of softmax term: 2 * conv1 (since ∂conv_i/∂a = 2*a_{i-j})
    grad = 2.0 * conv1

    # So final gradient = 4*n * conv1
```

The program searches a non‑negative step function `a = (a₀,…,a_{n‑1})` that minimises  

```
2·n·max(conv(a,a)) / (∑ a)²                (1)
```

The objective is homogeneous, therefore we may work with sequences normalised to the
probability simplex `∑ a = 1`.  The problem is convex – the term `max_i (a*a)_i` is a
maximum of convex quadratics – and a good solution can be found by a combination of

* **Parametric sweeps** – cheap families (geometric, power, exponential, two‑blocks,
  symmetric Beta‑shapes, dual Gaussians) covering a wide variety of shapes.
* **Global random sampling** – Dirichlet draws with a log‑uniform concentration
  and random multi‑block constructions.
* **Deterministic optimisation** – two mirror‑descent style methods  
  – (i) a classical sub‑gradient descent that uses only the *worst* convolution index,
  – (ii) a *smoothed* Nesterov‑style descent where `max` is replaced by a
  log‑sum‑exp (soft‑max) with a continuation on the smoothing parameter `α`.
  Both updates keep the iterates on the simplex (multiplicative‑exponential
  update followed by normalisation).

All phases respect a global time‑budget (`budget_s` seconds, default 1000 s).  
The best sequence seen at any time is returned.

The search is entirely self‑contained, uses only `numpy`, and never writes to disk.  
It can be called as `propose_candidate(seed=…, budget_s=…)` – the evaluation platform
will invoke it and pass the imported `evaluate_sequence` when needed.

```python
"""
Propose a non‑negative step‑function minimising
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of:

1️⃣  Fast parametric sweeps (geometric, power, exponential, two‑block,
    symmetric Beta‑shape, dual‑Gaussian) → good initial points.

2️⃣  Global random search:
      * Dirichlet draws with a wide range of concentration parameters.
      * Random multi‑block constructions.

3️⃣  Deterministic optimisation:
      * Sub‑gradient descent that updates using the *single* worst convolution index.
      * Smoothed (log‑sum‑exp) gradient descent with a continuation on the
        smoothing parameter α, using an exponentiated‑gradient (mirror‑descent)
        update and a decreasing step‑size η≈η₀/√t.

All phases stop automatically when the overall time budget (default 1000 s) is
exhausted.  The best sequence seen is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    #   Duchi et al., “Efficient Projections onto the L1-Ball for
    #   Learning in High Dimensions”, ICML 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta(n, best_seq, best_score, deadline):
    """Symmetric Beta‑type shape: a_i ∝ ((i+1)*(n-i))^p."""
    idx = np.arange(n, dtype=np.float64)
    eps = 1e-12
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        a = ((idx + 1) * (n - idx) + eps) ** p
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Beta p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_dual_gaussian(n, best_seq, best_score, deadline):
    """Two symmetric Gaussians placed at the ends."""
    idx = np.arange(n, dtype=np.float64)
    for sigma in np.logspace(0, 2, 60):  # sigma from 1 to 100
        if time.time() >= deadline:
            break
        left = np.exp(- (idx ** 2) / (2 * sigma ** 2))
        right = np.exp(- ((idx - (n - 1)) ** 2) / (2 * sigma ** 2))
        a = left + right
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] DualGauss σ={sigma:.1f} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter: log‑uniform between 10⁻² and 10⁰·⁷
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _random_block_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        K = rng.integers(2, 6)          # 2 … 5 blocks
        seq = np.zeros(n)
        total_weight = 0.0
        for _ in range(K):
            start = rng.integers(0, n)
            max_len = max(1, n // K)
            length = rng.integers(1, max_len + 1)
            end = min(start + length, n)
            if end <= start:
                continue
            w = rng.random()
            seq[start:end] += w / (end - start)
            total_weight += w
        if total_weight == 0.0:
            continue
        s = seq.sum()
        if s == 0.0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] RandomBlocks K={K} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣ Deterministic optimisation
# ----------------------------------------------------------------------
def _subgradient_one_max(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float):
    """
    Mirror‑descent using the sub‑gradient belonging to the *single* worst
    convolution index (instead of averaging over the whole active set).
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5  # base step‑size (tuned empirically)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        i_max = int(conv.argmax())
        max_val = conv[i_max]

        # exact sub‑gradient for the worst index
        grad = np.zeros_like(a)
        lo = max(0, i_max - (n - 1))
        hi = min(i_max, n - 1)
        # grad[j] = 2 * a[i_max - j] for j in [lo, hi]
        j_range = np.arange(lo, hi + 1, dtype=np.int64)
        grad[j_range] = 2.0 * a[i_max - j_range]

        # mirror‑descent (exponential / multiplicative) update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGradOneMax it={it:,} "
                  f"score={sc:.6f} η={eta:.2e}")

    return best_seq, best_score


def _gradient_softmax(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
          f_α(a) = 2·n· (1/α)·log Σ_i exp(α·conv_i(a))
    with respect to a (computed under the simplex normalisation).
    The returned vector already contains the factor 4·n (see derivation).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2·n‑1
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    p = exp_term / exp_term.sum()            # soft‑max weights, Σ p = 1

    # conv1_j = Σ_i p_i·a_{i‑j} (correlation of p and a)
    # compute via convolution with reversed p
    p_rev = p[::-1]
    conv_rev = np.convolve(p_rev, a)         # length 3·n‑2
    conv1 = conv_rev[(2 * n - 2) - np.arange(n)]

    # gradient of f_α : 4·n·conv1
    grad = 4.0 * n * conv1
    return grad


def _smooth_continuation(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float):
    """
    Continuation on the smoothing parameter α.
    We start with a small α (very smooth) and increase it, each stage performing
    a few exponentiated‑gradient steps on the smoothed objective.
    """
    a = a0.copy()
    n = a.shape[0]
    # schedule of smoothing parameters (geometric progression)
    alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    it = 0
    eta0 = 5.0  # base step for the exponential update (works on the normalised grad)

    for alpha in alphas:
        # give each α a limited budget; break when we run out of time
        while time.time() < deadline:
            it += 1
            grad = _gradient_softmax(a, alpha)   # contains factor 4·n
            # normalise the gradient to avoid huge exponentials
            grad_norm = grad / (np.max(np.abs(grad)) + 1e-12)

            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad_norm)
            total = a.sum()
            if total == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / total

            sc = _objective_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth α={alpha:.1f} it={it} "
                      f"score={sc:.6f} η={eta:.2e}")

            # simple early‑stop: if no improvement for 2000 iterations, move on
            if it % 2000 == 0:
                break
        if time.time() >= deadline:
            break
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    evaluation function.  The routine runs several cheap sweeps, a broad
    stochastic search and two deterministic refinements.  The best sequence
    (normalised to sum = 1) is returned.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Parametric sweeps (≈200 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 600.0        # reserve time for later stages
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_dual_gaussian(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 400.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Random block constructions (≈100 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_block_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 4️⃣ Sub‑gradient descent on the *worst* index (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 140.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient (worst‑index) descent")
        best_seq, best_score = _subgradient_one_max(best_seq,
                                                    phase_deadline,
                                                    best_seq,

    # --------------------------------------------------------------
    # 5️⃣ Smoothed continuation (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed continuation")
        best_seq, best_score = _smooth_continuation(best_seq, deadline,
                                                    best_seq,
                                                    best_score)
        print(f"[{time.time() - start_time:.2f}s] End of smoothed descent, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
```
