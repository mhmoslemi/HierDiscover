"""
Propose a candidate step‑function sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv is the full linear convolution.  The optimizer combines:

1️⃣  A fast random‑Dirichlet + a few classic window initialisers.
2️⃣  An extended alternating‑projection phase that forces a flat
    Fourier magnitude while staying on the probability simplex.
3️⃣  A projected Adam run on the smooth log‑sum‑exp approximation
    of the max‑norm (as in the reference code) – but with fewer restarts.
4️⃣  A dedicated **sub‑gradient descent** on the exact max‑convolution.
    The sub‑gradient is computed analytically, a simple diminishing‑step
    schedule is used, and the iterate is always Euclidean‑projected onto
    the simplex.
5️⃣  A short random hill‑climbing polish.

All phases respect a total wall‑time budget (default 1000 s).  The
result is a plain Python list of non‑negative floats that sum to 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer
# ----------------------------------------------------------------------
_start_time = None


def _elapsed() -> float:
    """seconds since start of optimisation."""
    return time.time() - _start_time


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
    v ≥ 0 , sum(v) = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (sum forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Phase 1 – random Dirichlet draws + deterministic windows
# ----------------------------------------------------------------------
def _random_dirichlet_trials(
    n: int, max_trials: int, deadline: float
):
    """Return the best Dirichlet‐sampled sequence seen before `deadline`."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        c = 10 ** rng.uniform(-3.0, 0.7)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{_elapsed():.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a few classic windows, already normalised to the simplex."""
    # sine window
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # a few Gaussian windows
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))


# ----------------------------------------------------------------------
# Phase 2 – alternating projection (flatten Fourier magnitude)
# ----------------------------------------------------------------------
def _alternating_projection(
    a0: np.ndarray, max_iter: int, deadline: float
):
    """Iteratively enforce flat magnitude + simplex constraints."""
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        F = np.fft.rfft(a, n=L)

        energy = np.dot(a, a)  # sum a_i^2
        flat_mag = np.sqrt(energy / L)

        # replace magnitude, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        a_full = np.fft.irfft(F_flat, n=L)[:n]

        a = np.maximum(a_full, 0.0)
        a = _project_simplex(a)

        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{_elapsed():.2f}s] alt‑proj iter {i:5d} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Adam optimiser on smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(
    a: np.ndarray, conv: np.ndarray, beta: float
) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    (see original source – equivalent to a convolution with softmax weights)
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()
    rev_corr = _convolve_fft(a[::-1], w)
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


def _adam_descent(
    a0: np.ndarray,
    score0: float,
    beta_vals,
    deadline: float,
):
    """Projected Adam optimiser on the smooth approximation."""
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    lr = 0.07
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            t += 1
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad ** 2)
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = _project_simplex(np.maximum(cand, 0.0))

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{_elapsed():.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            if np.linalg.norm(step) < 1e-13:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    score0: float,
    deadline: float,
):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses the analytical sub‑gradient:
        grad = 2 * (a_rev ⋆ w)_{slice}
    where w puts equal mass on all maximising lags.
    """
    n = a0.shape[0]
    avg_conv = 1.0 / (2 * n - 1)           # average of conv (∑conv = 1)
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    iteration = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        slack = max_val - avg_conv
        # early stop if gap already negligible
        if slack < 1e-13:
            break

        # mask of maximal entries (tolerance avoids missing ties due to fp noise)
        mask = conv >= max_val - 1e-15
        cnt = np.count_nonzero(mask)
        if cnt == 0:  # should never happen
            cnt = 1
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt

        # sub‑gradient
        rev = _convolve_fft(a[::-1], w)
        grad = 2.0 * rev[n - 1 : n - 1 + n]

        # diminishing step size
        eta = 0.5 / np.sqrt(iteration + 1.0)

        cand = a - eta * grad
        cand = _project_simplex(np.maximum(cand, 0.0))

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(
                f"[{_elapsed():.2f}s] subgrad iter {iteration:5d} new best {best_score:.6f}"
            )

        # simple acceptance rule (mostly greedy, occasional uphill)
        if cand_score < cur_score or np.random.rand() < 0.001:
            a = cand
            cur_score = cand_score

        iteration += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 5 – simple random hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(
    a0: np.ndarray,
    score0: float,
    deadline: float,
):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer up to 10 % of source mass
        delta = a[src] * rng.random() * 0.10
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{_elapsed():.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a plain list of floats.
    """
    np.random.seed(seed)
    global _start_time
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.1  # safety margin

    # problem size – a balance between expressiveness and speed
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{_elapsed():.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if provided by the surrounding harness)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq = starter.copy()
                    best_score = sc
                    print(f"[{_elapsed():.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – Random Dirichlet draws + deterministic windows (≈30 s)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, _start_time + 30.0)
    if time.time() < phase1_deadline:
        seq, sc = _random_dirichlet_trials(
            n, max_trials=5000, deadline=phase1_deadline
        )
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{_elapsed():.2f}s] after Dirichlet best = {best_score:.6f}")

        for shp in _deterministic_shapes(n):
            sc = _objective_score(shp)
            if sc < best_score:
                best_seq, best_score = shp.copy(), sc
                print(f"[{_elapsed():.2f}s] deterministic shape score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – alternating projection (≈45 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        proj_deadline = time.time() + 0.45 * remaining
        max_iter_per_run = 3000
        restarts = 8

        for r in range(restarts):
            if time.time() >= proj_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.02, size=n)
            a0 = best_seq + jitter
            a0 = _project_simplex(np.maximum(a0, 0.0))

            a_opt, sc_opt = _alternating_projection(
                a0, max_iter=max_iter_per_run, deadline=proj_deadline
            )
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(
                    f"[{_elapsed():.2f}s] alt‑proj restart {r} new best {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimiser on smooth max (≈25 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.25 * remaining
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
        restarts = 3

        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.01, size=n)
            a0 = best_seq + jitter
            a0 = _project_simplex(np.maximum(a0, 0.0))
            a0_score = _objective_score(a0)

            a_opt, sc_opt = _adam_descent(
                a0,
                a0_score,
                beta_vals=beta_schedule,
                deadline=adam_deadline,
            )
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(
                    f"[{_elapsed():.2f}s] Adam restart {r} new best {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Phase 4 – exact sub‑gradient descent (≈20 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + 0.20 * remaining
        a_opt, sc_opt = _subgradient_descent(
            best_seq, best_score, deadline=subgrad_deadline
        )
        if sc_opt < best_score:
            best_seq, best_score = a_opt, sc_opt
            print(
                f"[{_elapsed():.2f}s] sub‑gradient phase new best {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 5 – final hill‑climb (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{_elapsed():.2f}s] starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{_elapsed():.2f}s] finished – best objective ≈ {best_score:.6f}")

    return [float(x) for x in best_seq]
