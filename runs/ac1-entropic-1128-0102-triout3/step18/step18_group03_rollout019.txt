"""
Search for a low‑value step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The approach is deliberately different from the reference implementation:

1. **Parametric sweep** – we generate a large collection of vectors from simple
   one‑parameter families (pure geometric, pure power, and a product of both)
   and from low‑α Dirichlet draws.  Every candidate is evaluated with a fast
   FFT‑based exact objective; the K best are kept.

2. **Focused exponential‑gradient polishing** – starting from the best seed we
   apply many iterations of a multiplicative (exponential) sub‑gradient update
   on the true L∞ objective (the max of the self‑convolution).  The update is
   cheap (O(n) to compute the sub‑gradient) and quickly reduces the peak
   convolution value.

3. **Final random hill‑climb** – very small mass transfers between random
   positions are tried until the time budget expires.

The whole routine runs for at most `budget_s` seconds (default 1000) and returns
the best vector found as a plain Python list (already normalised to sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Fast exact objective (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _geometric(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(i * np.log(r))
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    """Power law a_i ∝ (i+1)**p, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a /= a.sum()
    return a


def _geometric_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined shape a_i ∝ r**i · (i+1)**p, normalised."""
    i = np.arange(n, dtype=np.float64)
    log_a = i * np.log(r) + p * np.log(i + 1.0)
    a = np.exp(log_a - np.max(log_a))  # prevent overflow
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient helpers (used in polishing)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Main entry point – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs for at most `budget_s` seconds and returns
    a plain Python list (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10          # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 offers a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep – cheap but diverse seeds.
    # --------------------------------------------------------------
    candidates = [best_seq]

    # (a) Pure geometric shapes.
    for r in np.linspace(0.85, 0.9999, 150):
        cand = _geometric(n, r)
        candidates.append(cand)

    # (b) Pure power‑law shapes.
    for p in np.linspace(-2.0, 3.0, 121):
        cand = _power(n, p)
        candidates.append(cand)

    # (c) Geometric × power.
    for r in np.linspace(0.90, 0.9999, 80):
        for p in np.linspace(-1.5, 2.5, 81):
            cand = _geometric_power(n, r, p)
            candidates.append(cand)

    # (d) Low‑α Dirichlet draws – produce very sparse vectors.
    for alpha in [0.02, 0.05, 0.1, 0.2]:
        for _ in range(8):
            cand = rng.dirichlet(np.full(n, alpha))
            candidates.append(cand)

    # (e) Random mixtures of geometric and its reverse (helps break symmetry).
    for _ in range(30):
        r = rng.uniform(0.85, 0.9999)
        lam = rng.random()
        g = _geometric(n, r)
        rev = g[::-1]
        mix = lam * g + (1.0 - lam) * rev
        # mixture already sums to 1
        candidates.append(mix)

    # --------------------------------------------------------------
    # Quick evaluation and keep the K best seeds.
    # --------------------------------------------------------------
    K = 12
    scores = [_eval_fast(c, L_fft) for c in candidates]
    best_indices = np.argsort(scores)[:K]
    pool = [candidates[i].copy() for i in best_indices]
    pool_scores = [scores[i] for i in best_indices]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] after sweep – best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Exponential‑gradient polishing (focus on max‑conv).
    # --------------------------------------------------------------
    # We'll devote the remaining time to a long exp‑grad run starting from the
    # current best vector.
    def _expgrad(a0: np.ndarray, deadline: float, start_time: float,
                 best_seq: np.ndarray, best_score: float):
        """Multiplicative updates on the true L∞ objective."""
        a = a0.copy()
        n = a.shape[0]
        it = 0
        eta0 = 1.0
        while time.time() < deadline:
            it += 1
            conv = _conv_self_fft(a, L_fft)
            k = int(np.argmax(conv))
            g = _subgrad_max_conv(a, k)
            eta = eta0 / (1.0 + 0.001 * it)
            mult = np.exp(-eta * g)
            a = a * mult
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = uniform.copy()
            score = _eval_fast(a, L_fft)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"exp‑grad it={it:,} score={score:.6f}")
            # occasional jitter to escape plateaus
            if it % 4000 == 0:
                eps = 1e-4 * rng.random(n) * a
                a = np.maximum(a + eps, 0.0)
                a = a / a.sum()
        return best_seq, best_score

    # Allocate most of the budget to exponentiated‑gradient polishing.
    expgrad_deadline = deadline - 60.0   # leave ~1 min for final hill‑climb
    if time.time() < expgrad_deadline:
        best_seq, best_score = _expgrad(best_seq, expgrad_deadline, start,
                                       best_seq, best_score)

    # --------------------------------------------------------------
    # 3️⃣  Final random hill‑climb (tiny mass transfers).
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps_move = best_seq[src] * rng.random() * 5e-4   # ≤0.05 % of source mass
        prop = best_seq.copy()
        prop[src] -= eps_move
        prop[dst] += eps_move
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] hill‑climb score={new_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq /= tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
