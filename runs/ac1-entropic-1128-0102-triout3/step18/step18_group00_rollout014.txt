def _smooth_gradient_descent(a0, L_fft, deadline, start_time, best_seq, best_score, rng):
    a = a0.copy()
    n = len(a)
    it = 0
    step0 = 0.5
    beta = 0.5
    # We'll increase beta gradually each iteration: maybe beta = min(10, 0.5 + it* (10-0.5)/max_iters)
    # or exponential schedule: beta = 0.5 * (1.05**it) clipped at 10.
    max_beta = 15.0
    while time.time() < deadline:
        it += 1
        # compute smoothed conv and gradient
        conv = _conv_self_fft(a, L_fft)  # length 2n-1
        # Smooth with beta
        # To avoid overflow, subtract max before exp.
        max_conv = np.max(conv)
        scaled = conv - max_conv  # for numeric stability
        w = np.exp(beta * scaled)
        w_sum = w.sum()
        prob = w / w_sum
        # compute gradient
        prob_rev = prob[::-1]
        # Convolution a with prob_rev using FFT length L2 = next_pow2(n + (2*n-1) - 1) = next_pow2(3*n-2)
        L2 = _next_pow2(3 * n - 2)
        Af = np.fft.rfft(a, n=L2)
        Bf = np.fft.rfft(prob_rev, n=L2)
        conv2 = np.fft.irfft(Af * Bf, n=L2)[: 3 * n - 2]  # full convolution length n+(2n-1)-1 = 3n-2
        # gradient = 2 * conv2_rev[:n] where conv2_rev = conv2[::-1]
        grad = 2.0 * conv2[::-1][:n]
        # This gradient is derivative of smooth_max w.r.t a.
        # Now step
        step = step0 / (np.sqrt(it) + 1.0)  # decayed
        cand = a - step * grad
        cand = _project_simplex(cand)
        # Evaluate real objective
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score - 1e-10:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] smoothGD it={it:,} beta={beta:.2f} score={cand_score:.6f}")
            a = cand
        else:
            # optionally accept with some probability decreasing
            if rng.random() < 0.01:
                a = cand
        # increase beta gradually
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)  # slow increase

        # jitter occasionally
        if it % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

"""
Hybrid random‑parameter sweep + smoothed gradient descent + aggressive
simulated‑annealing.  The goal is to push the value of

    2·n·max(conv(a,a)) / (∑ a)²

below the previously reported 1.5078.  The search works with a single
CPU, respects the 1000 s budget and returns a plain Python list of floats.
All vectors are kept on the probability simplex (non‑negative, sum = 1).

Algorithm outline
-----------------
1️⃣ Random parametric sweep (≈10–20 % of the budget)
    – families: monotone geo‑power, symmetric U‑shape,
      beta‑shape ( (i+1)^α·(n‑i)^β ),
      discrete Gaussian, mixture of two exponentials.
    – keep the best candidate (fast FFT based evaluation).

2️⃣ Smoothed gradient descent (≈40 % of the budget)
    – minimise a smooth approximation of the max‑convolution
      ϕβ(a)= (1/β)·log∑exp(β·conv(a,a)).
    – gradient is obtained via a second FFT; step size decays,
      Euclidean projection onto the simplex after each step.
    – β is increased slowly (homotopy) until a large value (≈15),
      so the smoothed objective converges to the true max.

3️⃣ Aggressive simulated‑annealing hill‑climb (≈40 % of the budget)
    – tiny mass‑transfer moves (src→dst) with a linearly cooling
      temperature; both the maximal move size and the temperature are
      annealed.
    – occasional jitter keeps the search diverse.
    – the best sequence encountered is remembered.

The final vector is renormalised to sum = 1 and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full linear self‑convolution a * a via FFT (length 2·len(a)‑1).
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear convolution a * b via FFT (full length len(a)+len(b)‑1).
    """
    na = a.shape[0]
    nb = b.shape[0]
    if L is None:
        L = _next_pow2(na + nb - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: na + nb - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1; otherwise the denominator is taken into account)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Families of candidate vectors
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """
    Monotone shape: a_i ∝ (i+1)^p·exp(‑lam·i)
    Returned vector is L1‑normalised.
    """
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(n: int, lam: float, p: float,
                              log_i: np.ndarray) -> np.ndarray:
    """
    Symmetric “U‑shape”: a_i ∝ (i+1)^p·exp(‑lam·i) + (n‑i)^p·exp(‑lam·(n‑1‑i))
    Normalised to unit L1‑norm.
    """
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)  # decreasing part
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float, beta: float,
                     log_i: np.ndarray, log_ni: np.ndarray) -> np.ndarray:
    """
    Beta‑type shape: a_i ∝ (i+1)^α·(n‑i)^β .
    """
    log_shape = alpha * log_i + beta * log_ni
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian_shape(n: int, mu: float, sigma: float) -> np.ndarray:
    """
    Discrete Gaussian: a_i ∝ exp( -(i‑mu)² / (2·sigma²) ).
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = -((i - mu) ** 2) / (2.0 * sigma * sigma)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_mixture_exp(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    """
    Mixture of two exponentials:
        a_i ∝ w·exp(-lam1·i) + (1‑w)·exp(-lam2·(n‑1‑i))
    """
    i = np.arange(n, dtype=np.float64)
    left = w * np.exp(-lam1 * i)
    right = (1.0 - w) * np.exp(-lam2 * (n - 1 - i))
    shape = left + right
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp smoothing of the max)
# ----------------------------------------------------------------------


def _smooth_grad_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the max‑convolution.
    The smoothing parameter β is increased gradually (homotopy).
    """
    a = a0.copy()
    n = a.shape[0]
    L2 = _next_pow2(3 * n - 2)                 # length for a * prob_rev
    it = 0
    step0 = 0.5
    beta = 0.5
    max_beta = 15.0

    while time.time() < deadline:
        it += 1
        # ----- smoothed max -----
        conv = _conv_self_fft(a, L_fft)        # length 2n‑1
        # stability: shift by max before exp
        shift = conv.max()
        scaled = conv - shift
        w = np.exp(beta * scaled)
        w_sum = w.sum()
        prob = w / w_sum                       # softmax weights, length 2n‑1

        # ----- gradient of the smoothed objective -----
        prob_rev = prob[::-1]                  # reversed probability vector
        grad_full = _conv_fft(a, prob_rev, L2)  # length 3n‑2
        # gradient_i = 2 * conv_rev[(2n‑2)‑i]  == 2 * grad_full_rev[i]
        grad = 2.0 * grad_full[::-1][:n]

        # ----- simple step + simplex projection -----
        step = step0 / (np.sqrt(it) + 1.0)
        cand = a - step * grad
        cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth‑GD it={it:,} "
                  f"β={beta:.2f} score={cand_score:.6f}")
            a = cand

        # mild random escape (rare)
        if rng.random() < 0.005:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

        # increase β slowly, never exceed max_beta
        if beta < max_beta:
            beta = min(max_beta, beta * 1.001)

    return best_seq, best_score

def _simulated_annealing(a_start: np.ndarray,
                         score_start: float,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves with a linearly decreasing temperature.
    The maximal transfer size also shrinks with time.
    """
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    # temperature schedule
    T0 = 5.0e-3          # initial temperature (more permissive than before)
    Tf = 1.0e-7          # final temperature

    # move‑size schedule
    mv0 = 5.0e-2         # up to 5 % of source mass initially
    mvf = 5.0e-4         # down to 0.05 % at the end

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = (elapsed) / (deadline - start_time + 1e-12)  # 0 → 1
        T = T0 * (1.0 - frac) + Tf * frac
        max_move = mv0 * (1.0 - frac) + mvf * frac

        src = rng.integers(n)
        if cur[src] < 1e-18:          # ignore almost‑empty sources
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * max_move
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            # accept (down‑hill always, uphill with prob.)
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}")

        # occasional jitter to keep diversity
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = cur / cur.sum()
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}")

    return best, best_score

            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Random parametric sweep (≈10 % of the total budget)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + max(30.0, 0.12 * budget_s)   # at least 30 s
    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)               # log(i+1)
    log_ni = np.log(np.maximum(n - i_arr, 1.0))  # log(n‑i)  (avoid log(0))

    families = 5
    while time.time() < sweep_deadline:
        fam = rng.integers(families)

        if fam == 0:            # monotone geo‑power
            lam = np.exp(rng.uniform(np.log(1e-5), np.log(1e-2)))
            p = rng.uniform(-6.0, 5.0)
            cand = _make_geo_power(n, lam, p, log_i)

        elif fam == 1:          # symmetric U‑shape
            lam = np.exp(rng.uniform(np.log(1e-5), np.log(1e-2)))
            p = rng.uniform(-6.0, 5.0)
            cand = _make_geo_power_symmetric(n, lam, p, log_i)

        elif fam == 2:          # beta‑type shape
            alpha = rng.uniform(-5.0, 5.0)
            beta = rng.uniform(-5.0, 5.0)
            cand = _make_beta_shape(n, alpha, beta, log_i, log_ni)

        elif fam == 3:          # discrete Gaussian
            mu = rng.uniform(0.0, n - 1.0)
            sigma = rng.uniform(n / 200.0, n / 5.0)   # from ~0.5 % to 20 % of n
            cand = _make_gaussian_shape(n, mu, sigma)

        else:                   # mixture of two exponentials
            w = rng.random()
            lam1 = np.exp(rng.uniform(np.log(1e-5), np.log(1e-2)))
            lam2 = np.exp(rng.uniform(np.log(1e-5), np.log(1e-2)))
            cand = _make_mixture_exp(n, w, lam1, lam2)

        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] sweep fam={fam} "
                  f"score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Smoothed gradient descent (≈40 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 150.0:            # ensure we have enough time left
        grad_deadline = time.time() + 0.4 * remaining
        best_seq, best_score = _smooth_grad_descent(
            best_seq, L_fft, grad_deadline, start_time,
            best_seq, best_score, rng
        )

    # ------------------------------------------------------------------
    # 3️⃣ Aggressive simulated‑annealing hill‑climb (rest of the time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, L_fft, deadline, start_time, rng
        )

    # ------------------------------------------------------------------
    # Final normalisation (sum → 1) and return.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
