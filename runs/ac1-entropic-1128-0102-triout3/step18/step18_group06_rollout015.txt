```python
def _frank_wolfe_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # objective = 2*n*max_val (since sum=1)
        obj = 2.0 * n * max_val
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"...")
        # compute active set
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for i in active:
            lo = max(0, i - (n-1))
            hi = min(i, n-1)
            j = np.arange(lo, hi+1)
            k = i - j
            grad[lo:hi+1] += a[k]
        grad = 2.0 * grad / len(active)
        # select indices
        i_min = np.argmin(grad)
        # step size: diminishing
        gamma = 2.0 / (it + 2.0)  # typical FW step
        # ensure we keep sum=1 (FW move to vertex)
        a = (1.0 - gamma) * a
        a[i_min] += gamma
        # enforce symmetry: average with reversed copy
        a = (a + a[::-1]) * 0.5
        # renormalize sum to 1 (due to symmetry may cause slight drift)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0/n)
    return best_seq, best_score
```

def _local_random_swaps(a, deadline, best_seq, best_score):

        frac = rng.uniform(0.0, 0.2)  # up to 20% of mass at i

"""
Propose a non‑negative step‑function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines several complementary search stages:

1️⃣  Fast parametric sweeps (geometric, power, exponential, two‑block,
    Gaussian) to obtain good seeds.

2️⃣  Global exploration by random Dirichlet draws.

3️⃣  A convex Frank‑Wolfe style descent on the true objective.
    The sub‑gradient of   max_i (a*a)_i   is obtained via a correlation,
    and the linear sub‑problem is solved analytically (move mass to the
    coordinate with smallest gradient).  A modest diminishing step size
    and an optional symmetry projection are used.

4️⃣  A short simulated‑annealing / random‑swap refinement that fine‑tunes
    the best candidate.

All stages respect a 1000 s CPU budget and the best sequence found is
returned (normalised to sum = 1).  The implementation avoids closures,
uses only NumPy (no file‑IO) and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Assumes sum(seq) == 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the max‑convolution term (via correlation)
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Returns a sub‑gradient of  max(conv(a,a))  at a.
    The active set is taken as all indices within 1e-12 relative tolerance.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]
    if active.size == 0:
        return np.zeros_like(a)
    mask = np.zeros(conv.shape, dtype=np.float64)
    mask[active] = 1.0
    # correlation of mask with a gives exactly   sum_{l} a[l] * mask[l+k]
    grad = 2.0 * np.correlate(mask, a, mode="valid")
    grad /= active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_gaussian(n, best_seq, best_score, deadline):
    """Gaussian bell centred in the middle; also mixed with uniform."""
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    for alpha in np.logspace(-6, -1, 50):
        if time.time() >= deadline:
            break
        seq = np.exp(-alpha * ((idx - mid) ** 2))
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        for lam in (0.0, 0.5, 1.0):
            mixed = lam * np.full(n, 1.0 / n) + (1.0 - lam) * seq
            mixed /= mixed.sum()
            sc = _objective_score(mixed)
            if sc < best_score:
                best_score, best_seq = sc, mixed.copy()
                print(f"[{time.time() - start_time:.2f}s] Gaussian α={alpha:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (global exploration)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        logc = rng.uniform(-2.0, 0.7)          # concentration ∈ [10⁻²,5]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        # enforce symmetry (helps the later descent)
        a = (a + a[::-1]) * 0.5
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Frank‑Wolfe style descent on the simplex
# ----------------------------------------------------------------------
def _frank_wolfe_descent(a0: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float):
    """
    Convex descent: at each iteration move a fraction γ towards the coordinate
    with the smallest sub‑gradient component.  A simple symmetry projection
    (a ↔ a[::-1]) keeps the candidate even, which is known to be beneficial.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] FW it={it:,} obj={obj:.6f}")

        # sub‑gradient of the max‑convolution term
        grad = _subgradient(a, conv)

        # coordinate with smallest gradient (most promising descent direction)
        i_min = int(np.argmin(grad))

        # diminishing step size – never larger than 0.2 and shrinks over time
        gamma = min(0.2, 2.0 / (it + 10.0))

        # FW update towards the unit vector e_i_min
        a = (1.0 - gamma) * a
        a[i_min] += gamma

        # optional symmetry projection (helps to avoid asymmetric artefacts)
        a = (a + a[::-1]) * 0.5

        # renormalise to stay on the simplex
        tot = a.sum()
        if tot > 0:
            a /= tot
        else:
            a = np.full(n, 1.0 / n)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣ Local random swaps (simulated‑annealing style fine‑tuning)
# ----------------------------------------------------------------------
def _local_random_swaps(a0: np.ndarray, deadline: float,
                       best_seq: np.ndarray, best_score: float):
    """
    Simple stochastic refinement:
    repeatedly move a small random chunk of probability mass from a random
    source index to a random target index, accept any improvement and also
    occasionally accept worsening moves according to a slowly cooling temperature.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    # initial temperature – a small fraction of the current score
    T0 = max(1e-4, best_score * 0.01)

    while time.time() < deadline:
        # pick source and target (different)
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0:
            continue
        # transfer up to 15 % of the source mass
        delta = rng.uniform(0.0, 0.15) * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # enforce symmetry and re‑normalise
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new /= a_new.sum()

        conv = _conv_self(a_new)
        obj = 2.0 * n * conv.max()
        Δ = obj - best_score
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start_time))
        accept = Δ < 0 or rng.random() < np.exp(-Δ / (T + 1e-12))
        if accept:
            a = a_new
            if obj < best_score:
                best_score, best_seq = obj, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Swap it={it} obj={obj:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The function follows the four‑stage strategy described above
    and returns the best sequence (normalised to sum = 1) as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Determine problem size n (use the provided starter if possible)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        starter = np.asarray(globals()["height_sequence_1"], dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter /= starter.sum()
        else:
            starter = None
    else:
        starter = None
        n = 4096

    # fallback uniform if no valid starter
    if starter is None:
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq = starter.copy()

    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweeps (≈15 % of the budget)
    # --------------------------------------------------------------
    phase_deadline = start_time + 0.15 * budget_s
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_gaussian(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈40 % of the budget)
    # --------------------------------------------------------------
    phase_deadline = start_time + 0.55 * budget_s
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Frank‑Wolfe descent (≈30 % of the budget)
    # --------------------------------------------------------------
    phase_deadline = start_time + 0.85 * budget_s
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Frank‑Wolfe descent")
        best_seq, best_score = _frank_wolfe_descent(best_seq, phase_deadline, best_seq, best_score)

    # --------------------------------------------------------------
    # 4️⃣ Local random swaps (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting local random‑swap refinement")
        best_seq, best_score = _local_random_swaps(best_seq, deadline, best_seq, best_score)

    # --------------------------------------------------------------
    # Normalise and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
