def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 8192
    # param sweep Beta symmetrical
    best_seq = np.full(n, 1.0/n)
    best_score = _objective_score(best_seq)
    # define x positions as midpoints
    x = (np.arange(n) + 0.5)/n

    # symmetrical sweep
    for a_param in np.logspace(-1.5, 0.5, 40):
        if time.time() >= deadline: break
        p = np.power(x, a_param - 1) * np.power(1 - x, a_param - 1)
        if p.sum() == 0: continue
        p /= p.sum()
        score = _objective_score(p)
        if score < best_score:
            best_score, best_seq = score, p.copy()
            print(f"[{time.time() - start:.2f}s] Beta a={a_param:.4f} sym score={score:.6f}")

    # asymmetric sweep
    a_vals = np.linspace(0.2, 1.2, 12)
    b_vals = np.linspace(0.2, 1.2, 12)
    for a_param in a_vals:
        for b_param in b_vals:
            if time.time() >= deadline: break
            p = np.power(x, a_param - 1) * np.power(1 - x, b_param - 1)
            if p.sum() == 0: continue
            p /= p.sum()
            score = _objective_score(p)
            if score < best_score:
                best_score, best_seq = score, p.copy()
                print(f"[{time.time() - start:.2f}s] Beta a={a_param:.3f} b={b_param:.3f} score={score:.6f}")

    # random Dirichlet exploration for remaining time before gradient
    phase_mc_end = deadline - 150
    while time.time() < phase_mc_end:
        # concentration c drawn log-uniform
        c = 10**np.random.uniform(-2, 0.7)
        a = np.random.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        score = _objective_score(a)
        if score < best_score:
            best_score, best_seq = score, a.copy()
            print(f"[{time.time() - start:.2f}s] Dirichlet c={c:.3f} score={score:.6f}")

    # Adam gradient descent with smoothing
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    iter = 0
    gamma = 1.0
    gamma_max = 30.0

    while time.time() < deadline:
        iter += 1
        # compute conv and objective
        conv = _conv_self(a)
        obj = 2.0 * n * conv.max()
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start:.2f}s] Adam iter={iter} obj={obj:.6f} gamma={gamma:.2f}")

        # adapt gamma gradually
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma * 1.001)  # increase slowly

        # compute softmax weights w (stable)
        maxc = conv.max()
        exp_vals = np.exp(gamma * (conv - maxc))
        w = exp_vals / exp_vals.sum()

        # term1 via cross-correlation
        L = _next_pow2(len(a) + len(w) - 1)
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w, n=L)
        term1 = np.fft.irfft(A * np.conj(W), n=L)[:len(a)]

        # term2 sliding sum of w of length n
        cum = np.cumsum(w)
        # term2_k = cum[k+n-1] - cum[k-1] (including bounds)
        term2 = cum[n-1: 2*n-1] - np.concatenate(([0.0], cum[:n-1]))

        grad = term1 + a * term2

        # Adam update
        lr = 0.2 / np.sqrt(iter)  # decreasing learning rate
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iter)
        v_hat = v / (1 - beta2 ** iter)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_simplex(a)

        # occasional random perturbation to escape stagnation
        if iter % 5000 == 0:
            # add small Gaussian noise then renorm
            noise = np.random.normal(scale=0.01, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # final projection and return
    a = np.maximum(best_seq, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    else:
        a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished best score≈{best_score:.6f}")
    return [float(x) for x in a]

"""
Propose a non‑negative step‑function (probability vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three ideas:

1️⃣  A fast parametric sweep over symmetric and asymmetric Beta kernels
    (a_i ∝ x^{α‑1}(1‑x)^{β‑1}) on the simplex.  These families capture the
    edge‑heavy shapes that are known to flatten the self‑convolution.

2️⃣  A cheap global search with Dirichlet draws to explore the simplex
    beyond the parametric families.

3️⃣  A deterministic Adam‑style mirror descent on the *smooth*
    approximation  f(a)=log‑sum‑exp(γ·conv(a,a))/γ  of the max‑convolution.
    The gradient of this smooth objective is computed analytically by a
    single FFT‑based cross‑correlation, giving cheap updates even for
    n≈8000.  γ is increased slowly so the algorithm starts with a smooth
    landscape and finishes by focusing on the true max.

All candidates are projected onto the probability simplex (non‑negative,
unit‑sum).  The best sequence found before the time budget expires is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute ∇_a  ( 1/γ log Σ_i exp(γ·conv_i(a)) )
    using a single FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2·n‑1
    # stable softmax
    maxc = conv.max()
    w = np.exp(gamma * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum                               # shape (2·n‑1,)

    # term1 = Σ_i w_i  a_{i‑k}   (cross‑correlation a ⋆ w)
    L = _next_pow2(len(a) + len(w) - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(w, n=L)
    term1 = np.fft.irfft(Af * np.conj(Wf), n=L)[:n]

    # term2 = Σ_{i=k}^{k+n‑1} w_i   (sliding window sum of length n)
    cum = np.cumsum(w)
    term2 = cum[n - 1 : 2 * n - 1] - np.concatenate(([0.0], cum[: n - 1]))

    grad = term1 + a * term2
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine runs a quick Beta‑kernel sweep,
    a large random Dirichlet exploration and a deterministic Adam‑style
    descent on a smoothed version of the true objective.  The best
    sequence (normalised to sum = 1) is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 8192                         # a larger n gives a finer discretisation
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Beta‑kernel sweep (symmetric & asymmetric)
    # --------------------------------------------------------------
    x = (np.arange(n) + 0.5) / n          # mid‑point positions in (0,1)

    # symmetric Beta(α,α) – α ∈ [0.5, 3] (α<0.5 would blow up the convolution)
    for a in np.linspace(0.5, 3.0, 36):
        if time.time() >= deadline:
            break
        p = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
        s = p.sum()
        if s == 0.0:
            continue
        p /= s
        sc = _objective_score(p)
        if sc < best_score:
            best_score, best_seq = sc, p.copy()
            print(f"[{time.time() - start_time:.2f}s] Beta sym α={a:.3f} score={sc:.6f}")

    # asymmetric Beta(α,β)
    a_vals = np.linspace(0.5, 2.5, 12)
    b_vals = np.linspace(0.5, 2.5, 12)
    for a in a_vals:
        for b in b_vals:
            if time.time() >= deadline:
                break
            p = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
            s = p.sum()
            if s == 0.0:
                continue
            p /= s
            sc = _objective_score(p)
            if sc < best_score:
                best_score, best_seq = sc, p.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta a={a:.3f} b={b:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet draws (global exploration)
    # --------------------------------------------------------------
    # run Dirichlet while we still have a generous chunk of time left
    dirichlet_deadline = deadline - 150.0
    while time.time() < dirichlet_deadline:
        # concentration: log‑uniform between 10⁻² and 5
        logc = np.random.uniform(-2.0, 0.7)
        c = 10.0 ** logc
        a = np.random.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Adam‑style mirror descent on smoothed max‑convolution
    # --------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    gamma = 1.0                # start smooth, increase gradually
    gamma_max = 30.0

    while time.time() < deadline:
        it += 1

        # current (true) objective – for bookkeeping only
        conv = _conv_self(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score, best_seq = cur_obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} obj={cur_obj:.6f} gamma={gamma:.2f}")

        # increase smoothing parameter slowly
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma * 1.001)

        # gradient of smooth approximation
        grad = _smooth_gradient(a, gamma)

        # Adam update (learning‑rate decays like 1/√t)
        lr = 0.2 / np.sqrt(it)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # enforce simplex constraints (non‑negative + unit‑sum)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = uniform.copy()
        else:
            a = a / s

        # occasional random jitter to help escape flat regions
        if it % 5000 == 0:
            noise = np.random.normal(scale=0.01, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = uniform.copy()
            else:
                a = a / s

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
