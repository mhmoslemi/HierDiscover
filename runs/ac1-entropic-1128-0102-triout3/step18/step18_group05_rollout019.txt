"""
Annealing + targeted hill‑climbing for the max–self‑convolution problem.

We keep a non‑negative vector a of length n with Σa = 1.
The objective is 2·n·max(conv(a,a)).  Because the problem is scale‑invariant we
always maintain the ℓ₁‑norm equal to 1, so the objective reduces to 2·n·M where
M = max_k Σ_i a_i·a_{k‑i}.

The algorithm consists of three stages:

1. **Initialisation** – start from the global best `height_sequence_1` if it
   exists, otherwise from a uniform distribution.

2. **Targeted descent** – repeatedly locate the shift `k_max` that attains the
   current maximum M, find the pair (i,j) with i+j = k_max that contributes
   most to this maximum, and move a small fraction of mass from the larger of
   the two indices to a random destination.  The convolution is updated
   incrementally (O(n)) using the identity  
   conv(a+Δ,a+Δ) = conv(a,a) + 2·conv(a,Δ) + conv(Δ,Δ) with Δ having only two
   non‑zero entries.  An improvement is accepted if the new maximum does not
   increase.

3. **Exploratory random moves** – with a complementary probability we perform
   a small random mass transfer between two random positions; these moves are
   also accepted only when they do not raise the current maximum.

Both phases are run until the time budget (≈ 1000 s) expires.  The best
vector found is finally re‑normalised (to guard against rounding errors) and
returned as a plain Python list.

The incremental update avoids recomputing the full convolution with an FFT at
every iteration, allowing many thousands of cheap improvements within the
time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT for the initial full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Simple helper: project onto the probability simplex (used only for safety)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}. """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback
        return np.maximum(v, 0.0) / np.maximum(v.sum(), 1e-12)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Run a fast annealing / hill‑climbing search for a low‑bound step function.

    Returns a list of non‑negative floats (the sequence of heights).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Load starter if available, otherwise uniform
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a = np.maximum(starter, 0.0)
            n = a.size
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n)
            print(f"[{time.time() - start_time:.2f}s] Loaded global starter (n={n})")
        else:
            raise ValueError
    except Exception:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=float)
        print(f"[{time.time() - start_time:.2f}s] Using uniform start (n={n})")

    # --------------------------------------------------------------
    # Initial full convolution and objective
    # --------------------------------------------------------------
    b = _conv_self(a)                # length 2n‑1
    cur_max = b.max()
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score
    best_max = cur_max

    print(f"[{time.time() - start_time:.2f}s] Initial objective = {cur_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for the stochastic search
    # --------------------------------------------------------------
    move_frac_start = 0.12           # max fraction of a[src] moved initially
    move_frac_end = 0.001            # at the end of the run
    prob_targeted = 0.75             # probability to perform a targeted move
    iter_count = 0
    max_iter = 10_000_000            # hard limit – real stop is time

    while time.time() < deadline and iter_count < max_iter:
        iter_count += 1
        # linearly anneal the move size
        elapsed = time.time() - start_time
        t_frac = elapsed / budget_s
        move_frac = move_frac_start * (1.0 - t_frac) + move_frac_end * t_frac

        # ----------------------------------------------------------
        # Choose the type of move
        # ----------------------------------------------------------
        if rng.random() < prob_targeted:
            # ----- targeted move -------------------------------------------------
            # locate the shift attaining the current maximum
            k_max = int(np.argmax(b))

            # indices that participate in this max shift
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx = np.arange(lo, hi + 1)

            # product a[i] * a[k_max - i] for every contributing pair
            prod = a[idx] * a[k_max - idx]
            pair = idx[np.argmax(prod)]                 # i with largest contribution
            j_pair = k_max - pair                        # its partner j = k_max - i

            # pick the larger of the two as source
            if a[pair] >= a[j_pair]:
                src = pair
            else:
                src = j_pair

            # destination: a random index different from src
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)
        else:
            # ----- random move ---------------------------------------------------
            src = rng.integers(n)
            if a[src] <= 1e-16:
                continue
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

        # amount of mass to move – never exceed the source amount
        delta = a[src] * move_frac * rng.random()
        if delta <= 1e-16:
            continue

        # ----------------------------------------------------------
        # Incremental update of a and its convolution
        # ----------------------------------------------------------
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        b_new = b.copy()

        # linear part 2·conv(a,Δ)
        # Δ has +δ at dst and –δ at src
        # conv(a, +δ·e_dst) adds  δ·a shifted by dst
        # conv(a, –δ·e_src) subtracts δ·a shifted by src
        # factor 2 appears because conv is symmetric
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[dst:dst + n] += 2.0 * delta * a

        # quadratic part conv(Δ,Δ)  (only four positions are affected)
        idx2src = 2 * src
        idx2dst = 2 * dst
        idx_sum = src + dst
        b_new[idx2src] += delta * delta
        b_new[idx2dst] += delta * delta
        b_new[idx_sum] -= 2.0 * delta * delta

        # enforce non‑negativity (tiny negative values can appear due to rounding)
        # this does not affect the objective because the evaluation clips negatives.
        # b_new = np.maximum(b_new, 0.0)

        new_max = b_new.max()
        new_score = 2.0 * n * new_max

        # accept only if the maximal convolution does not increase
        if new_max <= cur_max + 1e-15:
            # update current state
            a = a_new
            b = b_new
            cur_max = new_max
            cur_score = new_score

            # keep the best ever seen
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                best_max = new_max
                print(
                    f"[{time.time() - start_time:.2f}s] iter {iter_count} "
                    f"improved best to {best_score:.6f}"
                )

    # --------------------------------------------------------------
    # Final clean‑up: ensure non‑negativity and re‑normalise
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    print(
        f"[{time.time() - start_time:.2f}s] Finished after {iter_count} iterations; "
        f"final objective ≈ {best_score:.6f}"
    )
    return [float(v) for v in best_a.tolist()]
