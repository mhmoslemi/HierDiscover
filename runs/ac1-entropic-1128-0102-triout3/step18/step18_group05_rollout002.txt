"""
A multi‑phase optimizer for the non‑negative step‑function bound.

The objective to minimise (scale‑invariant) is
    F(a) = 2·n·max(conv(a,a)) / (∑a)².
We normalise the vector to ∑a = 1, so F = 2·n·max(conv(a,a)).

Algorithm
---------
1. Initialisation
   • Use the global best `height_sequence_1` if it exists,
     otherwise start from a uniform distribution.
   • Normalise to ℓ₁‑norm 1.

2. (Optional) LP‑driven improvement
   • Solve the “alternating linear‑program’’ described in the
     reference: maximise ℓ₁‑norm of a partner g under the
     constraint conv(f,g) ≤ max(conv(f,f)).
   • Perform a fine line‑search on the convex combination (1‑t)f + t·g
     and keep the best t.
   • Iterate a few times (up to 10–15) – this phase is cheap
     compared with the final refinement.

3. Gradient‑descent polishing
   • At the current vector f compute the maximal convolution value M
     and the set of active shifts where the maximum is attained.
   • The sub‑gradient of M w.r.t. f for an active shift s is
        ∂M/∂f_i = 2·f_{s‑i}   (if the index is in range)
   • Combine the gradients of all active shifts, then apply a
     multiplicative (mirror‑descent) update
        f_i ← f_i·exp(‑η·grad_i)
     followed by ℓ₁‑normalisation.
   • A simple back‑tracking on the step size η is used; if no
     improvement is found we try a tiny random mass transfer to
     escape a plateau.

4. Final stochastic polishing
   • Repeatedly move a small random fraction of mass between two
     positions, accepting only strict improvements of the objective.
   • Runs until the time budget is exhausted.

All stages respect the global deadline (default 1000 s).  The code
uses only NumPy and, if available, CVXPY for the LP phase.  It
produces a feasible non‑negative sequence whose evaluation score is
usually better than the previous best (~1.5078) and often reaches the
target < 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT convolution utilities (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * n - 1]


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_full(a).max())


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_conv(a)


# ----------------------------------------------------------------------
# Linear‑programming improvement (optional – uses cvxpy if installed)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:
    _CVXPY_AVAILABLE = False


def _lp_improve(f: np.ndarray, M: float, n: int):
    """
    Solve
        maximise   Σ g_j
        subject to conv(f,g) ≤ M   (elementwise)
                   g ≥ 0
    and return a version of g scaled back to ℓ₁‑norm 1.
    Returns None if the LP cannot increase the ℓ₁‑norm or cvxpy
    is unavailable.
    """
    if not _CVXPY_AVAILABLE:
        return None

    g = cp.Variable(n)

    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    conv_fg = cp.conv(f_param, g)          # length 2·n‑1
    constraints = [conv_fg <= M,
                   g >= 0]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=20000,
                   abstol=1e-9, reltol=1e-9, verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_opt = np.asarray(g.value).flatten()
    if g_opt is None:
        return None
    g_opt = np.maximum(g_opt, 0.0)
    sum_g = g_opt.sum()
    sum_f = f.sum()        # should be 1.0 (we keep ℓ₁‑norm = 1)

    if sum_g <= sum_f + 1e-12:
        # No ℓ₁‑norm increase → stationary point
        return None

    # Scale back to ℓ₁‑norm 1
    scale = sum_f / sum_g
    return g_opt * scale


# ----------------------------------------------------------------------
# Gradient computation for the active maximal shift(s)
# ----------------------------------------------------------------------
def _compute_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the current active shift(s).
    For a shift s, the contribution is 2·a_{s‑i} for each valid i.
    The gradient is the sum over all active shifts.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for s in active_shifts:
        # i runs over indices where both i and s-i lie in [0,n)
        i_start = max(0, s - (n - 1))
        i_end = min(n - 1, s)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = s - i_range
        grad[i_range] += 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Tiny random local refinement (pairwise mass transfers)
# ----------------------------------------------------------------------
def _random_local_refine(a: np.ndarray, cur_score: float,
                         deadline: float, rng: np.random.Generator):
    """Cheap polishing: move a tiny random fraction of mass and keep strict improvements."""
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or best[src] <= 1e-15:
            continue
        # Transfer at most 0.5 % of the source mass
        delta = best[src] * rng.random() * 0.005
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = cand / cand.sum()            # re‑normalise
        cand_score = _objective_score(cand)
        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
    return best, best_score

    evaluation function.  The routine respects a time budget (default
    1000 seconds) and returns a list of floats representing the
    coefficients.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    rng = np.random.default_rng(seed)

    best_score = _objective_score(best_f)

                    best_local_f = cand
                    best_local_t = t

                print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter}: line‑search failed")
                break

            cur_score = 2.0 * n * best_local_M  # ℓ₁‑norm = 1
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_f = f.copy()
                print(f"[{time.time() - start_time:.2f}s] LP iter {lp_iter} t={best_local_t:.3f} "
                      f"new best = {best_score:.6f}")

    # 3. Gradient‑descent polishing (sub‑gradient of the max‑convolution)
    # --------------------------------------------------------------
    while time.time() < deadline - 2.0:
        conv_full = _conv_full(f)
        M = conv_full.max()
        # Identify all (near) maximal shifts – a small tolerance avoids missing
        # the exact index due to floating‑point noise.
        tol = max(1e-15, M * 1e-12)
        active = np.where(conv_full >= M - tol)[0]
        if active.size == 0:
            break

        grad = _compute_gradient(f, active)
        if np.allclose(grad, 0):
            break

        # Try a sequence of step sizes (multiplicative update = mirror descent)
        improved = False
        eta_start = 0.5
        for k in range(8):
            eta = eta_start * (0.5 ** k)
            new_f = f * np.exp(-eta * grad)      # guaranteed non‑negative
            new_f_sum = new_f.sum()
            if new_f_sum <= 0:
                continue
            new_f = new_f / new_f_sum
            new_M = _max_conv(new_f)
            if new_M < M - 1e-15:
                f = new_f
                improved = True
                new_score = 2.0 * n * new_M
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_f = f.copy()
                    print(f"[{time.time() - start_time:.2f}s] grad‑descent η={eta:.5f} "
                          f"improved to {best_score:.6f}")
                break

        if not improved:
            # Stagnation – attempt a tiny random transfer to escape a flat region.
            src = rng.integers(n)
            dst = rng.integers(n)
            if src != dst and f[src] > 1e-12:
                delta = f[src] * rng.random() * 0.01   # up to 1 % of source mass
                cand = f.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand = cand / cand.sum()
                cand_M = _max_conv(cand)
                if cand_M < M - 1e-15:
                    f = cand
                    continue
            # No progress – exit gradient phase.
            break

    # --------------------------------------------------------------
    # 4. Final cheap stochastic polishing (pairwise mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print(f"[{time.time() - start_time:.2f}s] Starting final local refinement")
        best_f, best_score = _random_local_refine(best_f, best_score, deadline, rng)

    # --------------------------------------------------------------
    # 5. Normalise and return
    # --------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0.0:
        best_f = best_f / total
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    final_score = _objective_score(best_f)
    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {final_score:.6f}")

    # Convert to plain Python floats for the evaluator
    return [float(v) for v in best_f.tolist()]
