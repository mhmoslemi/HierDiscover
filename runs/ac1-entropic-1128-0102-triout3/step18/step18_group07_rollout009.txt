"""
Propose a non‑negative step‑function (a probability vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1️⃣  Initialise from simple analytic windows (Gaussian, Hann, etc.).
2️⃣  A long alternating‑projection phase that forces the Fourier magnitude of
    the zero‑padded sequence to be flat while keeping the vector non‑negative
    and normalised (this is a phase‑retrieval style “Gerchberg‑Saxton” loop).
3️⃣  Exponentiated‑gradient descent on a smooth log‑sum‑exp approximation of
    the max‑convolution, with a schedule of increasing β (temperature) and a
    decreasing step‑size.  The gradient is computed analytically via a
    convolution with the soft‑max weights.
4️⃣  A short random mass‑transfer hill‑climb for final polishing.

All phases respect the global 1000 s budget.  The best vector found is
returned as a plain list of Python floats (non‑negative, summed to 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used only for printing)
# ----------------------------------------------------------------------
_start_time = None

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using the FFT.
    The result length is len(x)+len(y)-1.
    """
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {v ≥ 0, sum(v)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (sum forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max utilities (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Implements  grad = 2 * (a_rev ⋆ w)[offset],
    where w are the soft‑max weights.
    """
    n = a.shape[0]
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # shape (2·n‑1,)
    # Convolution of reversed a with the soft‑max weights
    rev_corr = _convolve_fft(a[::-1], w)   # length n + (2n‑1) -1 = 3n‑2
    # Extract the n‑length segment that corresponds to the gradient
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Deterministic window shapes (already normalised to the simplex)
# ----------------------------------------------------------------------
def _deterministic_shapes(n: int):
    """Yield a few classic windows (sine, Hann, triangular, Gaussian)."""
    # sine window
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # a few Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))


# ----------------------------------------------------------------------
# Alternating projection: flatten Fourier magnitude while staying on the simplex
# ----------------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, deadline: float):
    """
    Run a long Gerchberg–Saxton style projection:
        * enforce constant magnitude in the frequency domain
        * enforce non‑negativity and sum‑to‑one in the time domain
    Returns the best vector (lowest objective) found during the run.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)  # length for zero‑padded FFT
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    it = 0
    while time.time() < deadline:
        # FFT of zero‑padded a
        F = np.fft.rfft(a, n=L)

        # Preserve total energy, enforce constant magnitude
        energy = np.dot(a, a)          # sum a_i^2
        flat_mag = np.sqrt(energy / L)
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Back to time domain and truncate
        a_full = np.fft.irfft(F_flat, n=L)
        a = a_full[:n]

        # Projection onto the feasible set (non‑negative + simplex)
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # Track best
        if (it & 511) == 0:   # report every 512 iterations
            sc = _objective_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - _start_time:.2f}s] alt‑proj iter {it:5d} new best {best_score:.6f}")
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the smooth max objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float):
    """
    Projected exponentiated‑gradient descent on the smoothed max.
    A schedule of increasing β (temperature) and decreasing step size η is used.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    # Schedule: start warm (β small) and finish very cold (β large)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
                     100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
    base_eta = 0.12  # base step size (will be scaled by sqrt(beta))

    for beta in beta_schedule:
        eta = base_eta / np.sqrt(beta)   # decreasing step for larger β
        while time.time() < deadline:
            conv = _conv_self(a)
            # soft‑max weights
            max_val = conv.max()
            exps = np.exp(beta * (conv - max_val))
            w = exps / exps.sum()

            # gradient of smooth max
            rev_corr = _convolve_fft(a[::-1], w)   # length 3n‑2
            grad = 2.0 * rev_corr[n - 1 : n - 1 + n]  # n‑vector

            # exponentiated (multiplicative) update, staying non‑negative
            factor = np.exp(-eta * grad)
            a = a * factor

            # Avoid the whole vector collapsing to zero because of underflow
            eps = 1e-15
            a = np.maximum(a, eps)

            # Renormalise to the simplex (sum = 1)
            a_sum = a.sum()
            a = a / a_sum

            # Evaluate objective
            sc = _objective_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - _start_time:.2f}s] EG β={beta:.0f} new best {best_score:.6f}")

            # Simple termination condition for this β‑phase
            if np.linalg.norm(eta * grad) < 1e-13:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climbing (mass transfer) for polishing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, deadline: float):
    rng = np.random.default_rng()
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer up to 10 % of the source mass
        delta = a[src] * rng.random() * 0.10
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # project onto simplex (non‑negativity + sum‑to‑1)
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        sc = _objective_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand
            print(f"[{time.time() - _start_time:.2f}s] hill‑climb new best {best_score:.6f}")

        elif sc < _objective_score(a):
            # allow small downhill moves
            a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises `evaluate_sequence`.
    Returns the best sequence found (as a plain list of floats).
    """
    global _start_time
    np.random.seed(seed)
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.1   # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressiveness and speed
    # ------------------------------------------------------------------
    n = 8192                     # more degrees of freedom than the reference code
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - _start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter sequence is available, incorporate it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            starter_sum = starter.sum()
            if starter_sum > 0.0:
                starter = starter / starter_sum
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - _start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – deterministic windows (≈20 s)
    # ------------------------------------------------------------------
    for shape in _deterministic_shapes(n):
        sc = _objective_score(shape)
        if sc < best_score:
            best_seq, best_score = shape.copy(), sc
            print(f"[{time.time() - _start_time:.2f}s] deterministic shape score = {sc:.6f}")
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – long alternating projection (≈30 % of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        proj_deadline = time.time() + 0.30 * remaining
        # initialise the alternating‑projection run from the current best
        a_proj, sc_proj = _alternating_projection(best_seq, proj_deadline)
        if sc_proj < best_score:
            best_seq, best_score = a_proj, sc_proj
            print(f"[{time.time() - _start_time:.2f}s] after alt‑proj best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent on smooth max (≈55 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        eg_deadline = time.time() + 0.55 * remaining
        a_eg, sc_eg = _exp_gradient_descent(best_seq, eg_deadline)
        if sc_eg < best_score:
            best_seq, best_score = a_eg, sc_eg
            print(f"[{time.time() - _start_time:.2f}s] after EG best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final hill‑climb (remaining seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - _start_time:.2f}s] starting hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)

    # ------------------------------------------------------------------
    # Normalise one last time and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - _start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
