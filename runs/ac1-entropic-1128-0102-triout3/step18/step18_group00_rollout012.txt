    beta2 = 0.999

        # compute softmax weights

"""
Hybrid optimisation for the L∞‑convolution bound.

The routine works in three stages:

1️⃣  **Initialisation** – start from a uniform vector, optionally
    replace it by the global starter ``height_sequence_1`` if it exists.

2️⃣  **Parametric sweep** – explore the 3‑parameter monotone family

       a_i ∝ (i+1)^p · exp(‑λ·i^q) ,   λ>0 , p∈ℝ , q∈{1,1.5,2}

    on a moderately large problem size (default 8192).  The best
    candidate found by the fast FFT‑based objective
    ``2·n·max(conv(a,a))/sum(a)^2`` is kept.

3️⃣  **Smoothed‑max Adam optimisation** – from the sweep result we
    minimise a smooth approximation of the L∞‑objective

       M_α(a) = (1/α)·log∑_k exp(α·(a*a)_k)

    using Adam updates, a projection onto the probability simplex and
    an annealing schedule for the smoothing parameter α (starting from
    5 and slowly growing to ≈200).  Every few hundred iterations a tiny
    random jitter is added to avoid stagnation.

The algorithm respects a 1000 s wall‑time budget; the best vector
encountered is returned as a plain Python list (ℓ₁‑norm = 1, all
entries ≥ 0).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a))/(sum(a)^2) – fast version."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _adam_optimize(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Smoothed‑max Adam optimisation (see docstring)."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 5.0                    # smoothing for the max
    alpha_max = 200.0
    t = 0
    while time.time() < deadline:
        t += 1
        # --- convolution of the current iterate -----------------
        conv = _conv_self_fft(a, L_fft)

        # --- soft‑max weights for the smoothed max ---------------
        cmax = conv.max()
        # stabilise exponentials
        w = np.exp(alpha * (conv - cmax))
        w = w / w.sum()

        # --- gradient of the smoothed max ------------------------
        # g_j = 2 * Σ_i w_{i+j} * a_i   (np.correlate does exactly this)
        g = 2.0 * np.correlate(w, a, mode="valid")

        # --- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # a simple decay of the step size (helps after many iterations)
        lr = 0.05 / (np.sqrt(t) + 1.0)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- project back onto the simplex (non‑negative, sum=1) --
        a = _project_simplex(a)

        # --- occasional progress report ---------------------------
        if t % 50 == 0:
            cur_score = _eval_fast(a, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Adam it={t:,} "
                    f"α={alpha:.2f} score={cur_score:.6f}"
                )

        # --- raise the smoothing parameter slowly (annealing) -----
        if t % 100 == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.02)

        # --- random jitter to shake out of tiny plateaus ----------
        if t % 2000 == 0:
            jitter = 5e-5 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _adam_optimize(
            best_seq,
            L_fft,
            deadline,
            start_time,
            best_seq,
            best_score,
            rng,
        )

    # ------------------------------------------------------------------
    # Final normalisation (guard against numerical drift) and return.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
