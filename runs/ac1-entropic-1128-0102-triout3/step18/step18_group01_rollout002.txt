"""
Propose a non‑negative step function (probability vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm outline
-----------------
1️⃣  Simple seed generation:
    - uniform distribution,
    - the global starter `height_sequence_1` (if present),
    - a coarse parametric sweep over geometric×power and Gaussian families,
    - a few random Dirichlet‑type vectors.
   The best of these seeds is kept.

2️⃣  Smoothed‑max gradient descent:
    * Minimise a smooth approximation of the max‑convolution:
          Lβ(a) = (1/β)·log Σ_i exp(β·(a*a)_i)
      for an increasing sequence of β (continuation method).
    * The gradient of Lβ is
          ∇Lβ = 2·(w ⋆ a)   with   w_i ∝ exp(β·(a*a)_i)  (soft‑max weights)
    * Updates are performed with Adam (first‑order momentum) and a Euclidean
      projection onto the probability simplex.
    * Small random jitter is injected occasionally to escape plateaus.

3️⃣  Final local refinement:
    Tiny mass‑transfer hill‑climb (as in the previous baseline).

All steps respect the 1000 s time budget and keep the best vector found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full linear self‑convolution a⋆a (length 2·len(a)‑1) via FFT.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (for quick internal ranking)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """
    Exact evaluation: 2·n·max(conv(a,a)) / (sum(a)**2).
    Sum is assumed to be > 0.
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple parametric families – used for seeding
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Normalized a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Normalized a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i ∝ exp(beta·conv_i) in a numerically stable way.
    """
    max_c = conv.max()
    # subtract the max to avoid overflow
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()
    return w


def _gradient_softmax(a: np.ndarray, w: np.ndarray,
                      L_corr: int) -> np.ndarray:
    """
    Gradient of the smoothed max w.r.t a.
    g = 2·(w ⋆ a) where ⋆ denotes (non‑reversed) cross‑correlation.
    Implemented via FFT as convolution of w with reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    # convolution length = len(w)+len(a_rev)-1 = (2n-1)+n-1 = 3n-2
    Wf = np.fft.rfft(w, n=L_corr)
    Af = np.fft.rfft(a_rev, n=L_corr)
    corr_full = np.fft.irfft(Wf * Af, n=L_corr)
    # keep the central n entries: indices n-1 … 2n-2 (inclusive)
    start = n - 1
    g = 2.0 * corr_full[start:start + n]
    return g


def _smooth_max_grad(a: np.ndarray, beta: float,
                     L_conv: int, L_corr: int):
    """
    Compute (conv, soft‑max weights, gradient) for a given beta.
    Returns:
        conv   – full self‑convolution (length 2n‑1)
        w      – soft‑max probabilities (same length)
        grad   – ∇_a ( (1/β)·log Σ exp(β·conv) )
    """
    conv = _conv_self_fft(a, L_conv)
    w = _softmax_weights(conv, beta)
    grad = _gradient_softmax(a, w, L_corr)
    return conv, w, grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def _adam_step_simplex(a: np.ndarray,
                       grad: np.ndarray,
                       m: np.ndarray,
                       v: np.ndarray,
                       t: int,
                       eta: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam update followed by projection onto the probability simplex.
    Returns the new a, updated first moment m and second moment v.
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = eta * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = _project_simplex(a_new)
    return a_new, m, v

def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:

    Tiny mass‑transfer moves with a simple acceptance rule.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:

        src = rng.integers(n)
        if a[src] < 1e-14:
            continue

        if src == dst:

        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_fft)

            best_score = cand_score
            best = cand.copy()
            a = cand

            cand = cand / cand.sum()
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation function.
    Returns a plain Python list of floats (sum → 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12      # safety margin

    # --------------------------------------------------------------
    # Problem size – 16384 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 16384
    L_conv = _next_pow2(2 * n - 1)               # for self‑convolution
    L_corr = _next_pow2(3 * n - 2)               # for gradient correlation

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_conv)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Try globally supplied starter if present.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # up‑sample / down‑sample by simple linear interpolation
                x_src = np.linspace(0, 1, starter.size)
                x_tgt = np.linspace(0, 1, n)
                starter = np.interp(x_tgt, x_src, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_conv)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep (≈20 s)
    # --------------------------------------------------------------
    sweep_end = start_time + 20.0
    i_arr = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_arr + 1.0)

    # geometric × power
    lam_grid = np.linspace(1e-05, 2.5e-03, 120)         # fewer points for speed
    p_grid = np.linspace(-8.0, 5.0, 36)                 # step ≈0.35
    for lam in lam_grid:
        if time.time() >= sweep_end:
            break
        base = -lam * i_arr
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_conv)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} p={p:.2f} score={sc:.6f}")

    # pure Gaussian family
    c_grid = np.logspace(-8.0, -5.0, 30)
    for c in c_grid:
        if time.time() >= sweep_end:
            break
        log_shape = -c * (i_arr * i_arr)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_conv)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gaussian c={c:.2e} score={sc:.6f}")

    # --------------------------------------------------------------
    # 1.5️⃣ Random Dirichlet‑type seeds (≈10 s)
    # --------------------------------------------------------------
    rand_end = start_time + 30.0
    while time.time() < rand_end:
        # draw exponential variables and normalise → Dirichlet(1)
        rnd = rng.random(n)
        cand = rnd / rnd.sum()
        sc = _eval_fast(cand, L_conv)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] random seed score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smoothed‑max (log‑sum‑exp) gradient descent with Adam.
    # --------------------------------------------------------------
    # Continuation on β: start small, increase multiplicatively.
    beta = 1.0
    beta_max = 3000.0
    beta_factor = 1.4
    # Adam state
    m = np.zeros_like(best_seq)
    v = np.zeros_like(best_seq)
    it = 0
    # Base learning rate – empirically good for n≈16000
    eta0 = 0.12

    # Allocate the bulk of the time budget to this phase.
    grad_end = start_time + 0.70 * budget_s
    while time.time() < grad_end and beta <= beta_max:
        # Number of inner Adam steps per β – adapt to remaining time.
        inner_budget = min(200, int( (grad_end - time.time()) / 0.004 ))  # rough 4 ms/step
        for _ in range(inner_budget):
            it += 1
            conv, w, grad = _smooth_max_grad(best_seq, beta, L_conv, L_corr)

            # Adam update (projected onto simplex)
            best_seq, m, v = _adam_step_simplex(best_seq, grad, m, v, it, eta0)

            # Occasionally inject a tiny jitter to avoid exact plateaus
            if it % 1100 == 0:
                jitter = rng.random(best_seq.shape) * best_seq * 5e-5
                best_seq = _project_simplex(best_seq + jitter)

            # Internal exact score (fast) – keep track of best
            cur_score = _eval_fast(best_seq, L_conv)
            if cur_score < best_score:
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] smooth‑grad β={beta:.0f} it={it} score={cur_score:.6f}")

        # increase β (sharpen the approximation)
        beta = min(beta * beta_factor, beta_max)

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climb – use whatever time remains.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_conv, deadline, start_time, rng)

    # --------------------------------------------------------------
    # Normalisation & final reporting
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Optional final check via the provided evaluate_sequence (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
