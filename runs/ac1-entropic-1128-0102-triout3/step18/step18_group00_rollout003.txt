def _simulated_annealing_smart(a_start, score_start, L_fft, deadline, start_time, rng):
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    # Temperature schedule
    T0 = 1e-3
    Tf = 1e-8

    while time.time() < deadline:
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1 - frac) + Tf * frac

        # propose move
        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n-1)
        if dst >= src:
            dst += 1  # ensure distinct
        # amount up to 0.01 proportion of source (1%)
        eps = cur[src] * rng.random() * 0.01
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{time.time()-start_time:.2f}s] SA improved score={cand_score:.6f}")
        # occasional random jitter
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = _project_simplex(cur)
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time()-start_time:.2f}s] SA jitter improved score={cur_score:.6f}")

    return best, best_score

"""
Hybrid search for the step‑function that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases:

1️⃣  A fast parametric sweep over monotone “geo‑power’’ families
    (a_i ∝ (i+1)^p·e^{‑λ i}) and their symmetric U‑shapes.
    The best candidate (in the exact objective) is kept.

2️⃣  A smooth‑max optimisation:
    * The non‑smooth max is replaced by a smooth log‑sum‑exp
      (α controls the approximation quality).
    * Adam (adaptive‑momentum) updates are applied on the simplex,
      using the exact gradient of the smooth approximation
      (derived from the cross‑correlation a ⋆ w with w the soft‑max
       weights of the convolution).
    * α is increased from 1 to 200 during the run, allowing the method
      to first explore a wide region and then focus on the true max.

3️⃣  A final simulated‑annealing hill‑climb with tiny mass‑transfer
    moves, a linearly cooling temperature and occasional jitter,
    to polish the result.

The whole routine respects the 1000 s budget, works with any
sequence length (default 8192, which is a good compromise between
quality and speed), and returns a plain Python list of non‑negative
floats whose sum equals 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full linear self‑convolution a*a via FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of two vectors (full length) via FFT."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: x.shape[0] + y.shape[0] - 1]


def _cross_corr_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ w) of length L (zero‑padded).
    Returns an array of length L containing the correlation for shifts
    k = -(len(w)-1) … len(a)-1.
    """
    FA = np.fft.rfft(a, n=L)
    FW = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(FA * np.conj(FW), n=L)
    return corr


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast version) – sum assumed to be ≈1
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families used in the parametric sweep
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape: a_i ∝ (i+1)^p·e^{‑lam·i}, L1‑normalised."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape’’ (left + mirrored right), L1‑normalised."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _adam_smooth_max(
    a0: np.ndarray,
    best_score0: float,
    L_fft: int,
    L_corr: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth approximation
        S_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
    with a projecting onto the simplex after each update.
    α is ramped from 1 to α_max = 200.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2
    # schedule for α
    alpha = 1.0
    alpha_max = 200.0
    # we target roughly 6000 iterations for the full increase
    alpha_step = (alpha_max - alpha) / 6000.0

    it = 0
    best_seq = a.copy()
    best_score = best_score0

    offset = 2 * n - 2  # len(w)-1 for w of length 2n-1

    while time.time() < deadline:
        it += 1

        # ----- convolution of a with itself -----
        conv = _conv_self_fft(a, L_fft)

        # ----- smooth‑max weights -----
        conv_max = conv.max()
        # stable exponentiation
        w = np.exp(alpha * (conv - conv_max))
        w_sum = w.sum()
        w_norm = w / w_sum  # this is the soft‑max over convolution entries

        # ----- gradient = 2·(a ⋆ w_norm) -----
        corr = _cross_corr_fft(a, w_norm, L_corr)
        g = 2.0 * corr[offset : offset + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # occasional jitter to provoke escape from flat plateaus
        if it % 1200 == 0:
            eps_j = 5e-5 * rng.random(n) * a_new
            a_new = np.maximum(a_new + eps_j, 0.0)
            a_new = _project_simplex(a_new)

        # ----- exact evaluation -----
        exact_score = _eval_fast(a_new, L_fft)
        if exact_score < best_score:
            best_score = exact_score
            best_seq = a_new.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} α={alpha:.1f} "
                f"score={exact_score:.6f}"
            )

        a = a_new

        # ----- raise α gradually -----
        alpha = min(alpha_max, alpha + alpha_step)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (tiny mass transfers)
# ----------------------------------------------------------------------


def _simulated_annealing_smart(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Final hill‑climb with decreasing temperature."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    T0 = 1.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        # linear cooling based on elapsed fraction
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # transfer up to 0.5 % of the source mass (more aggressive than baseline)
        eps = cur[src] * rng.random() * 5e-3
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        # occasional random jitter to keep diversity
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = _project_simplex(cur)
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

    The routine follows three stages:
        1️⃣ parametric sweep,
        2️⃣ Adam on a smooth‑max surrogate,
        3️⃣ simulated‑annealing polishing.
    Returns a python list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # small safety margin

    # --------------------------------------------------------------
    # Problem size – 8192 gives a modest quality gain over 4096.
    # --------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)      # for cross‑correlation (a ⋆ w)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global starter (if provided by the surrounding environment).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
                if starter.shape[0] != n:
                    # interpolate / truncate to the current length
                    idx = np.linspace(0, starter.shape[0] - 1, n)
                    starter = np.interp(idx, np.arange(starter.shape[0]), starter)
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (≈30 s). Two families + a few random mixes.
    # --------------------------------------------------------------
    sweep_deadline = start_time + 30.0
    # grids – slightly broader than in the previous run to give the
    # optimizer fresh directions when n is larger.
    lam_grid = np.linspace(2e-05, 2e-03, 250)      # 250 values
    p_grid = np.linspace(-8.0, 5.0, 140)          # 140 values
    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * i_arr
        for p in p_grid:
            # monotone shape
            log_shape = base + p * log_i
            log_shape -= np.max(log_shape)
            mono = np.exp(log_shape)
            mono /= mono.sum()
            sc = _eval_fast(mono, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = mono.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep mono λ={lam:.6e} p={p:.2f} score={sc:.6f}"
                )

            # symmetric U‑shape
            sym = mono + mono[::-1]
            sym /= sym.sum()
            sc_sym = _eval_fast(sym, L_fft)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep sym λ={lam:.6e} p={p:.2f} score={sc_sym:.6f}"
                )

            # occasional random mixture of two families (adds diversity)
            if rng.random() < 0.001:
                # draw a second pair (lam2,p2)
                lam2 = rng.uniform(2e-05, 2e-03)
                p2 = rng.uniform(-8.0, 5.0)
                log_shape2 = -lam2 * i_arr + p2 * log_i
                log_shape2 -= np.max(log_shape2)
                comp2 = np.exp(log_shape2)
                comp2 /= comp2.sum()
                # blend
                w = rng.random()
                mix = w * mono + (1 - w) * comp2
                mix /= mix.sum()
                sc_mix = _eval_fast(mix, L_fft)
                if sc_mix < best_score:
                    best_score = sc_mix
                    best_seq = mix.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] sweep mix score={sc_mix:.6f}"
                    )

    print(f"[{time.time() - start_time:.2f}s] after sweep best_score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max Adam optimisation (≈ half the remaining budget)
    # --------------------------------------------------------------
    adam_deadline = deadline - 250.0   # reserve ~250 s for final SA
    if time.time() < adam_deadline:
        best_seq, best_score = _adam_smooth_max(
            best_seq,
            best_score,
            L_fft,
            L_corr,
            adam_deadline,
            start_time,
            rng,
        )
        print(f"[{time.time() - start_time:.2f}s] after Adam best_score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polishing (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing_smart(
            best_seq, best_score, L_fft, deadline, start_time, rng
        )
        print(f"[{time.time() - start_time:.2f}s] after SA best_score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation and conversion to plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
