"""
Hybrid optimisation with expanded parametric families and Adam‑projected
sub‑gradient descent.

Main ideas
----------
1️⃣  A faster parametric sweep that, besides the original “geo‑power’’ and
    symmetric U‑shape families, also explores a three‑parameter
    family   a_i ∝ (i+1)^p · (n‑i)^q · exp(‑λ·min(i,n‑1‑i)).
    Random draws from this family are performed for a few minutes.

2️⃣  A projected Adam optimiser (momentum + adaptive learning rate) that
    works on the exact sub‑gradient of the *max* convolution (the
    non‑smooth part).  Adam’s adaptivity often pushes the solution past
    plateaus that stalled the plain sub‑gradient descent used before.

3️⃣  Several short random restarts of the Adam optimiser (Dirichlet
    initialisations) to increase the chance of escaping local minima.

4️⃣  A final simulated‑annealing hill‑climb (identical to the reference
    implementation) to polish the best vector found.

The routine respects the 1000 s budget, prints progress information and
returns a plain Python list whose elements are non‑negative and sum to 1.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ------------------------------------------------------------
# Exact (fast) objective – assumes sum≈1.
# ------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ------------------------------------------------------------
# Adam‑projected sub‑gradient descent
# ------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the exact sub‑gradient of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.08
    it = 0
    C = 2.0 * n

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)               # sub‑gradient of max(conv)
        s = a.sum()
        maxc = conv[k]

        # gradient of the full objective  O = C*maxc / s^2
        grad_obj = C * (g / (s * s) - 2.0 * maxc * a / (s * s * s))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple decay of the base learning rate
        lr = lr0 / (1.0 + 1e-4 * it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}"
            )

        # occasional tiny jitter to keep exploration alive
        if it % 2500 == 0:
            jitter = 5e-5 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ------------------------------------------------------------
# Simulated‑annealing hill‑climb (unchanged)
# ------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with a decreasing temperature."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score

    n = cur.shape[0]
    T0 = 1.0e-3
    Tf = 5.0e-7  # final temperature

    while time.time() < deadline:
        # linear cooling based on elapsed fraction
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # up to 0.5 % of the source mass
        eps = cur[src] * rng.random() * 5e-3
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        # occasional random jitter
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = cur / cur.sum()
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score


# ------------------------------------------------------------
# Helper for random Dirichlet initialisation
# ------------------------------------------------------------


def _random_simplex_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the probability simplex (exponential trick)."""
    x = rng.exponential(1.0, size=n)
    x /= x.sum()
    return x


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

    Returns a plain Python list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 offers a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Pre‑compute index‑based helpers
    # ------------------------------------------------------------------
    i_arr = np.arange(n, dtype=np.float64)                     # 0 … n‑1
    log_i1 = np.log(i_arr + 1.0)                               # log(i+1)
    log_i2 = np.log(np.maximum(n - i_arr, 1.0))                # log(n‑i)
    i_dist = np.minimum(i_arr, n - 1 - i_arr)                  # distance to nearer end

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Attempt to use a globally supplied starter (if any).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (≈70 s). Two classic families + random draws
    #   from a richer three‑parameter family.
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    min_lambda = 4e-05
    max_lambda = 1.5e-03
    lam_grid = np.linspace(min_lambda, max_lambda, 300)   # 300 points
    p_grid = np.linspace(-6.0, 3.0, 91)                   # step ≈0.1

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * i_arr
        for p in p_grid:
            # ---------- monotone (geo‑power) ----------
            log_shape = base + p * log_i1
            log_shape -= np.max(log_shape)
            mono = np.exp(log_shape)
            mono /= mono.sum()
            sc = _eval_fast(mono, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = mono.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep mono λ={lam:.6e} "
                    f"p={p:.2f} score={sc:.6f}"
                )

            # ---------- symmetric “U‑shape’’ ----------
            left = mono
            sym = left + left[::-1]
            sym /= sym.sum()
            sc_sym = _eval_fast(sym, L_fft)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep sym λ={lam:.6e} "
                    f"p={p:.2f} score={sc_sym:.6f}"
                )

    # ------------------------------------------------------------------
    # 1b️⃣ Random search in the three‑parameter family (≈150 s).
    #   a_i ∝ (i+1)^p (n‑i)^q exp(‑λ·min(i,n‑1‑i))
    # ------------------------------------------------------------------
    random_deadline = start_time + 220.0
    while time.time() < random_deadline:
        lam = rng.uniform(min_lambda, max_lambda)
        p = rng.uniform(-8.0, 5.0)
        q = rng.uniform(-8.0, 5.0)

        log_shape = -lam * i_dist + p * log_i1 + q * log_i2
        log_shape -= np.max(log_shape)
        a = np.exp(log_shape)
        a /= a.sum()
        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] rand3‑param λ={lam:.6e} "
                f"p={p:.2f} q={q:.2f} score={sc:.6f}"
            )

        # also try the asymmetric “left+right’’ version (two halves may differ)
        left = np.exp(p * log_i1 - lam * i_arr)
        right = np.exp(q * log_i2 - lam * (n - 1 - i_arr))
        a2 = left + right
        a2 /= a2.sum()
        sc2 = _eval_fast(a2, L_fft)
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()
            print(
                f"[{time.time() - start_time:.2f}s] rand asymmetric λ={lam:.6e} "
                f"p={p:.2f} q={q:.2f} score={sc2:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑based projected sub‑gradient descent (intensive).
    # ------------------------------------------------------------------
    adam_deadline = deadline - 200.0
    if time.time() < adam_deadline:
        best_seq, best_score = _adam_optimize(
            best_seq, L_fft, adam_deadline, start_time, best_seq, best_score, rng
        )

    # ------------------------------------------------------------------
    # 3️⃣  Random‑restart Adam runs (short bursts) – useful for escaping
    #     local minima.  Each run gets a few thousand iterations.
    # ------------------------------------------------------------------
    restart_deadline = deadline - 120.0
    while time.time() < restart_deadline:
        # initialise from a random Dirichlet point
        a0 = _random_simplex_vector(n, rng)
        # allocate a modest slice of time for this restart (≈4 s)
        sub_deadline = time.time() + 4.0
        best_seq, best_score = _adam_optimize(
            a0, L_fft, sub_deadline, start_time, best_seq, best_score, rng
        )
        # tiny jitter before next restart to diversify the start point
        jitter = 1e-4 * rng.random(n) * best_seq
        best_seq = np.maximum(best_seq + jitter, 0.0)
        best_seq = _project_simplex(best_seq)

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing final hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq, best_score, L_fft, deadline, start_time, rng
        )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```

This implementation expands the search space with a richer three‑parameter family, adds a robust Adam optimiser (which typically out‑performs plain sub‑gradient descent on plateaus), performs multiple random restarts, and finally refines the best candidate with simulated annealing—all within the allotted 1000 s budget.
