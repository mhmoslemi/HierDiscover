def _subgradient_descent(a0, score0, deadline):
   a = a0.copy()
   cur_score = score0
   best_a = a.copy()
   best_score = cur_score
   step0 = 0.1   # initial step magnitude per unit gradient
   t = 0
   while time.time() < deadline:
      t += 1
      conv = _conv_self(a)
      max_idx = int(np.argmax(conv))
      # compute gradient
      idx = max_idx - np.arange(len(a))
      mask = (idx >= 0) & (idx < len(a))
      grad = np.zeros_like(a)
      grad[mask] = 2.0 * a[idx[mask]]
      # step size schedule
      step = step0 / np.sqrt(t)
      # direction opposite gradient
      cand = a - step * grad / (np.linalg.norm(grad) + 1e-12)
      cand = np.maximum(cand, 0.0)
      cand = _project_simplex(cand)
      cand_score = _objective_score(cand)
      if cand_score < best_score:
          best_score = cand_score
          best_a = cand.copy()
          print(f"[{time.time() - start_time:.2f}s] subgrad iter {t} new best {best_score:.6f}")
      # accept if improved
      if cand_score < cur_score:
          a = cand
          cur_score = cand_score
      else:
          # occasional uphill acceptance
          if np.random.rand() < 0.001:
              a = cand
              cur_score = cand_score
      # early stop if step gets tiny
      if step * np.linalg.norm(grad) < 1e-12:
          break
   return best_a, best_score

def _annealing_hill_climb(a0, score0, deadline):
   a = a0.copy()
   cur_score = score0
   best_a = a.copy()
   best_score = cur_score
   n = a.shape[0]
   rng = np.random.default_rng()
   # start temperature as maybe 0.01 * cur_score
   T0 = 0.01 * cur_score
   t0 = time.time()
   while time.time() < deadline:
       # linear cooling schedule based on elapsed fraction
       elapsed = time.time() - t0
       total = deadline - t0
       T = T0 * max(0.0, 1.0 - elapsed / total)
       # choose random subset size 2-10
       k = rng.integers(2, 7)  # 2..6
       idxs = rng.choice(n, size=k, replace=False)
       mass = a[idxs].sum()
       if mass == 0:
            continue
       # redistribute mass according to Dirichlet
       new_weights = rng.gamma(shape=1.0, scale=1.0, size=k)
       new_weights = new_weights / new_weights.sum() * mass
       cand = a.copy()
       cand[idxs] = new_weights
       # already sum=1 due to conservation, ensure nonneg
       cand_score = _objective_score(cand)
       if cand_score < best_score:
           best_score = cand_score
           best_a = cand.copy()
           a = cand
           cur_score = cand_score
           print(f"[{time.time() - start_time:.2f}s] anneal new best {best_score:.6f}")
       elif cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
           a = cand
           cur_score = cand_score
   return best_a, best_score

"""
A multi‑phase optimiser for the “max self‑convolution” objective.

The algorithm works with a fixed length `n` (default 4096).  It proceeds
in four stages, each using the remaining time budget:

1️⃣  *Diverse initialisation*: many classical windows (sine, Hann,
    triangular, Gaussian, exponential, power‑law, logistic) together
    with random Dirichlet draws.  The best start is retained.

2️⃣  *Sub‑gradient descent*: the objective is `max(conv(a,a))`.  The
    sub‑gradient of the active convolution index is cheap to compute:
    `grad_i = 2·a_{k−i}` for the index `k` where the convolution attains
    its maximum.  A diminishing step size and projection onto the
    probability simplex drives the sequence downward.

3️⃣  *Projected Adam on a smooth max*: a log‑sum‑exp approximation with an
    annealed temperature `β` yields a smooth surrogate whose gradient is
    computed via a soft‑max weighting of the convolution.  The Adam update
    respects the simplex constraints.

4️⃣  *Simulated‑annealing hill‑climbing*: randomly redistributes the mass
    inside a small random subset of coordinates (Dirichlet‑style) and
    accepts worsening moves with a linearly cooling temperature.

All phases respect the global 1000 s deadline; the current best sequence
(and its objective value) is printed whenever an improvement is found.
The returned list is a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used in many helpers)
# ----------------------------------------------------------------------
_start_time = None


def _now() -> float:
    return time.time() - _start_time

# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (zero‑padding to a power of two)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (the denominator is 1 because we stay on the simplex)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient – used by the Adam phase
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max w.r.t. a (see the code in the prompt)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # soft‑max weights, length 2·n‑1
    rev = a[::-1]
    # convolution rev ⋆ w (full) – the relevant slice is n‑1 … 2·n‑2
    tmp = _convolve_fft(rev, w)        # length 3·n‑2
    n = a.shape[0]
    grad = 2.0 * tmp[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Phase 1 – various random / deterministic starter sequences
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻⁴, 2] (covers very spiky to smooth)
        conc_log = rng.uniform(-4.0, 0.3)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{_now():.2f}s] Dirichlet c={c:.5f} score={sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes_extended(n: int):
    """Yield a collection of classic windows, plus exponential / power‑law."""
    # Sine
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # Gaussian family
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

    # Exponential decay (scale = n / s where s∈{4,8,16})
    for s in (4, 8, 16):
        scale = n / s
        y = np.exp(-np.arange(n) / scale)
        yield _project_simplex(np.maximum(y, 0.0))

    # Power‑law (p∈{0.5,1,1.5})
    for p in (0.5, 1.0, 1.5):
        y = 1.0 / (np.arange(1, n + 1) ** p)
        yield _project_simplex(np.maximum(y, 0.0))

    # Logistic (steepness)
    for steep in (0.01, 0.03, 0.06):
        y = 1.0 / (1.0 + np.exp((np.arange(n) - n / 2) / (steep * n)))
        yield _project_simplex(np.maximum(y, 0.0))


# ----------------------------------------------------------------------
# Phase 2 – Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    step0 = 0.12                # raw step magnitude (will be normalised)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        k = int(np.argmax(conv))               # active convolution index
        # build gradient = 2 * a shifted according to k
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # normalise gradient direction
        norm = np.linalg.norm(grad) + 1e-20
        step = step0 / np.sqrt(it) * (grad / norm)

        cand = a - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{_now():.2f}s] Sub‑grad it {it:5d} new best {best_score:.6f}")

        # accept strictly improving moves; allow a tiny uphill chance
        if cand_score < cur_score or np.random.rand() < 0.001:
            a = cand
            cur_score = cand_score

        # early stop when updates become negligible
        if np.linalg.norm(step) < 1e-14:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Adam optimiser on a smooth approximation of max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, score0: float, beta_vals, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    lr = 0.04
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        # each β‑phase runs until the global deadline expires
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{_now():.2f}s] Adam β={beta:.0f} new best {best_score:.6f}")

            # deterministic acceptance when better; occasional uphill moves
            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            # stop early if the update is vanishingly small
            if np.linalg.norm(step) < 1e-13:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Simulated‑annealing hill‑climbing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _annealing_hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    t0 = time.time()
    T0 = 0.02 * cur_score          # initial temperature (relative to score)

    while time.time() < deadline:
        elapsed = time.time() - t0
        frac = elapsed / (deadline - t0)
        T = T0 * max(0.0, 1.0 - frac)   # linear cooling

        # pick a random subset (size 2‑6) and reshuffle its mass
        k = rng.integers(2, 7)          # subset size
        idxs = rng.choice(n, size=k, replace=False)
        mass = a[idxs].sum()
        if mass == 0.0:
            continue
        # Dirichlet‑style redistribution of the gathered mass
        new_weights = rng.gamma(shape=1.0, scale=1.0, size=k)
        new_weights = new_weights / new_weights.sum() * mass

        cand = a.copy()
        cand[idxs] = new_weights
        cand_score = _objective_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{_now():.2f}s] Anneal new best {best_score:.6f}")
        elif cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            a = cand
            cur_score = cand_score

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a plain ``list[float]``.
    """
    np.random.seed(seed)
    global _start_time
    _start_time = time.time()
    deadline = _start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a sweet spot between expressive power and speed
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{_now():.2f}s] start – uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter is supplied, incorporate it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter, sc
                    print(f"[{_now():.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – rich collection of starters (≈25 s)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, _start_time + 25.0)
    if time.time() < phase1_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=3000,
                                          deadline=phase1_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{_now():.2f}s] after Dirichlet best = {best_score:.6f}")

        for shp in _deterministic_shapes_extended(n):
            sc = _objective_score(shp)
            if sc < best_score:
                best_seq, best_score = shp.copy(), sc
                print(f"[{_now():.2f}s] deterministic shape score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent (≈35 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + 0.35 * remaining
        a_opt, sc_opt = _subgradient_descent(best_seq, best_score,
                                             deadline=subgrad_deadline)
        if sc_opt < best_score:
            best_seq, best_score = a_opt, sc_opt
            print(f"[{_now():.2f}s] sub‑grad new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on smooth max (≈30 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.30 * remaining
        # a richer β schedule reaching up to 2·10⁴
        beta_schedule = [0.5, 2, 5, 20, 80, 200, 800, 2000, 8000, 20000]
        a_opt, sc_opt = _adam_descent(best_seq, best_score,
                                      beta_vals=beta_schedule,
                                      deadline=adam_deadline)
        if sc_opt < best_score:
            best_seq, best_score = a_opt, sc_opt
            print(f"[{_now():.2f}s] Adam final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (the final few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{_now():.2f}s] starting annealing hill‑climb")
        best_seq, best_score = _annealing_hill_climb(best_seq, best_score,
                                                     deadline=deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{_now():.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
