def _random_local_refine(a, cur_score, deadline, rng):
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or best[src] <= 1e-15:
            continue
        delta = best[src] * rng.random() * 0.005  # <=0.5% of source mass
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] local refine improved to {best_score:.6f}")
    return best, best_score

"""
Gradient–Mirror Descent for the step‑function bound

We minimise
        F(a) = 2·n·max(conv(a,a)) / (∑a)²
under a_i ≥ 0.  By normalising ∑a = 1 the objective becomes
        2·n·max(conv(a,a)).

The algorithm consists of three phases

1. **Smoothed‑max mirror descent** – we replace the nonsmooth max by a
   log‑sum‑exp ( temperature α ), compute the gradient of this smooth
   surrogate (which is a correlation of the weight vector with the current
   distribution) and perform an exponentiated‑gradient (KL‑mirror) update.
   The temperature α is increased from a low value (smooth) to a high value
   (close to the true max) while the step size η is annealed.

2. **Monotone acceptance** – a new iterate is kept only if the true
   (non‑smoothed) objective does not increase, guaranteeing a non‑increasing
   sequence of scores.

3. **Cheap random polishing** – after the descent phase we move tiny random
   fractions of mass between two positions and keep strict improvements.
   This usually extracts the final few hundredths of the bound.

The code works entirely on the probability simplex, never violates the
non‑negativity or sum‑to‑one constraints and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT based convolutions)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_general_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using an FFT of power‑of‑2 length."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    return _conv_general_fft(a, a)


def _objective_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Random local polishing (pairwise mass transfers)
# ----------------------------------------------------------------------
def _random_local_refine(a: np.ndarray, cur_score: float,
                         deadline: float, rng: np.random.Generator):
    """
    Cheap stochastic polishing: repeatedly move a tiny random fraction of mass
    from one position to another and keep strict improvements.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or best[src] <= 1e-15:
            continue
        # move at most 0.5 % of the source mass
        delta = best[src] * rng.random() * 0.005
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] local refine improved to {best_score:.6f}")

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
