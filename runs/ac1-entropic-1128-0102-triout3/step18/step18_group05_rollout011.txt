Projection to simplex:

Given new vector v (may have negative entries) we need to project onto simplex: the algorithm:

def project_to_simplex(v):
    """Project onto the probability simplex {x >=0, sum x = 1}."""
    # Use sorting method.
    # v - we want x = max(v - tau, 0) such that sum x = 1.
    # Compute sorted v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho = max i s.t. u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function (non‑negative sequence) that minimises

        F(a) = 2·n·max(conv(a,a)) / (∑a)²

The algorithm consists of three stages:

1. **Initialisation** – load the global best `height_sequence_1` if it exists,
   otherwise start from a uniform distribution.  The vector is normalised to
   Σa = 1 (the objective is scale‑invariant).

2. **Sub‑gradient descent on the active convolution offset** – at each iteration
   we locate the index `k*` where the self‑convolution attains its maximum.
   The (sub‑)gradient of `max(conv)` with respect to the entries of `a` is
   `g_i = 2·a_{k*‑i}` for all valid `i`.  A projected gradient step is taken
   with a decreasing step‑size (∝ 1/√iter) and the result is projected back onto
   the probability simplex (non‑negative, sum = 1).  If a step improves the
   objective it is kept; otherwise a small random mass transfer is tried.
   After a long streak without improvement a random perturbation (convex blend
   with a fresh Dirichlet sample) is injected to escape possible plateaus.

3. **Fine stochastic polishing** – for the remaining time a cheap local move
   (transfer a tiny fraction of mass between two positions) is repeatedly tried,
   accepting only strict improvements.  This mirrors the final polishing stage
   of the reference code but works on the much improved starting point.

The whole routine runs for the allotted budget (default 1000 s) and returns the
best sequence found.  All heavy work uses FFT‑based convolution (`O(n log n)`);
the simplex projection is performed by the standard sorting‑based algorithm.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """
    Return the evaluation score for a vector that already satisfies Σa = 1.
    This is exactly the value returned by ``evaluate_sequence`` for such a.
    """
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x ≥ 0, Σx = 1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Tiny random local refinement (pairwise mass transfers)
# ----------------------------------------------------------------------
def _random_local_refine(a: np.ndarray, cur_score: float,
                         deadline: float, rng: np.random.Generator):
    """Cheap polishing: move a tiny random fraction of mass and keep strict improvements."""
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or best[src] <= 1e-15:
            continue
        delta = best[src] * rng.random() * 0.005   # ≤0.5 % of source mass
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(cand)  # keep feasibility
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] local refine improved to {best_score:.6f}")

    return best, best_score
