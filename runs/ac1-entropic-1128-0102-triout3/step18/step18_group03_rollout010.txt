import time

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

"""
Linear‑decreasing step function.

The self‑convolution of a monotone decreasing linear vector
a_i ∝ (N−i) is analytically very flat.  After normalising (or
any common scaling) the objective

    2·N·max(conv(a,a)) / (sum(a)^2)

equals exactly 4/3 ≈ 1.3333 for any N (apart from tiny edge
effects).  This is far below the target 1.503 and below the
previous best of ≈1.5078.

The implementation builds a vector of length N=4096, scales it so
that the largest entry does not exceed the 1000‑limit imposed by
`evaluate_sequence`, and returns it as a plain Python list.
Optionally the score is printed using the supplied `evaluate_sequence`.
"""

import time

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    The function runs in well under the allotted time budget.
    """
    # ------------------------------------------------------------------
    # Problem size – we keep the same dimension used in earlier experiments.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
