    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # If we have global height_sequence_1, use as initial, else uniform.
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim !=1 or init.size==0:
            raise ValueError
        init = np.maximum(init,0)
        s = init.sum()
        if s>0: init = init / s
        else: init = None
    except Exception:
        init = None

    # Choose length n for sequence. If init present, use its length else set n=4096.
    n = len(init) if init is not None else 4096

    # If init not present, start with uniform
    if init is None:
        seq = np.full(n, 1.0/n, dtype=float)
    else:
        seq = init.copy()
    # compute best score
    best_seq = seq.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time()-start:.2f}s] Initial score={best_score:.6f} n={n}")

    # Phase 1: LP improvement iterations
    # Do as many LP iterations as time allows (with min time per solve).
    iteration = 0
    while time.time() < deadline - 5.0:   # leave time for local search
        iteration += 1
        M = _conv_self(seq).max()
        # Solve LP
        g = _lp_max_sum_given_bound(seq, M)

        total = g.sum()

            break
        # Normalize

        if new_score < best_score:
            best_score = new_score
            best_seq = g_norm.copy()

"""
Simple two‑phase optimizer for the step‑function evaluation problem.

Phase 1 – Linear‑programming (LP) improvement:
    Starting from a feasible sequence f (sum = 1) we compute M = max(conv(f,f)).
    We then solve the LP
        maximise   Σ g_j
        subject to g_j ≥ 0,
                   (f * g)_i ≤ M  for all i,
    i.e. we add as much mass as possible without exceeding the current peak M.
    The solution g is renormalised to a probability vector and mixed with f
    (convex combination).  Repeating this step monotonically reduces the peak
    of the self‑convolution.

Phase 2 – Local Dirichlet perturbations:
    After the LP phase we perform a fast stochastic hill‑climb.
    New candidates are drawn from Dirichlet distributions centred at the current
    best sequence (high concentration → small moves, low concentration → large
    exploratory jumps).  Improvements are kept.

The algorithm respects the 1000 s budget, prints timing information and finally
returns the best sequence as a plain Python list of non‑negative floats.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using a real FFT.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return the evaluation value 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    This is the exact value returned by ``evaluate_sequence`` after the
    normalisation.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Linear‑programming improvement (Phase 1)
# ----------------------------------------------------------------------
def _lp_step(f_seq: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f_seq * g)_k ≤ M   for all k.
    Returns the optimal g (as a NumPy array) or ``None`` on failure.
    """
    n = f_seq.shape[0]

    # Constant copy of the current sequence
    f_const = cp.Constant(f_seq)

    # Decision variable
    g = cp.Variable(n, nonneg=True)

    # Convolution constraint (vector inequality)
    conv_fg = cp.conv(f_const, g)          # length 2·n‑1
    constraints = [conv_fg <= M]

    # Linear objective
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # Try ECOS first (fast interior‑point), fall back to SCS if needed
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=3000)
        except Exception:
            return None

    if g.value is None:
        return None
    return np.asarray(g.value, dtype=float)


# ----------------------------------------------------------------------
# Local random Dirichlet search (Phase 2)
# ----------------------------------------------------------------------
def _local_dirichlet_search(best_seq: np.ndarray,
                            best_score: float,
                            rng: np.random.Generator,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a fast hill‑climb around ``best_seq`` using Dirichlet proposals.
    The concentration parameter is sampled adaptively so that both small
    (exploratory) and large (local) moves are attempted.
    """
    n = best_seq.shape[0]
    while time.time() < deadline:
        # Choose concentration: 80 % local, 20 % global
        if rng.random() < 0.8:
            conc = rng.uniform(50.0, 200.0)   # fairly tight
        else:
            conc = rng.uniform(0.5, 5.0)      # exploratory

        alpha = best_seq * conc + 1e-8          # avoid exact zeros
        cand = rng.dirichlet(alpha)
        score = _objective_score(cand)

        if score < best_score:
            best_seq, best_score = cand, score
            print(f"[{time.time() - start_time:.2f}s] Dirichlet improvement → {score:.6f}")

    return best_seq, best_score

    # --------------------------------------------------------------

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished. Best score≈{final_score:.6f}")

    return [float(x) for x in best_seq]
