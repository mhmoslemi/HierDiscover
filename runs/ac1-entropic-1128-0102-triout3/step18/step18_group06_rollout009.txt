"""
Propose a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages:

1️⃣  **Rich parametric sweeps** – geometric, power, exponential,
    two‑block and a flexible Beta family.  These cheap scans give
    decent starting points.

2️⃣  **Global Dirichlet search** – many random draws from a Dirichlet
    distribution (different concentration parameters) to explore
    the simplex broadly.

3️⃣  **Smoothed mirror descent** – we minimise a smooth
    approximation of the “max‑convolution” using a soft‑max (log‑sum‑exp)
    with a schedule of increasing smoothness (α).  The exact gradient
    of the smoothed objective is obtained analytically; a multiplicative
    (exponentiated‑gradient) update keeps the iterates on the simplex.

4️⃣  **Local stochastic refinement** – small random mass transfers
    (hill‑climbing / simulated‑annealing style) are tried and accepted
    only when they improve the true objective.

All stages stop automatically when the 1000 s budget is exhausted,
always keeping the best sequence seen so far.  The final sequence is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution (len(x)+len(y)-1) via FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _objective_score(seq: np.ndarray) -> float:
    """Score under the assumption sum(seq)=1 (the true objective is scaling‑invariant)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation from (Duchi et al., 2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score

            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta(n, best_seq, best_score, deadline):
    """
    Beta‑family: a_i ∝ x_i^{α-1} (1-x_i)^{β-1}
    with x_i = (i+0.5)/n to avoid exact zeros.
    """
    idx = (np.arange(n, dtype=np.float64) + 0.5) / n
    alphas = np.logspace(-1, 2, 30)   # 0.1 … 100
    betas = np.logspace(-1, 2, 30)
    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                return best_seq, best_score
            # use log‑space to stay safe from under/overflow
            log_seq = (a - 1) * np.log(idx) + (b - 1) * np.log(1.0 - idx)
            seq = np.exp(log_seq - np.max(log_seq))   # shift for numerical stability
            total = seq.sum()
            if total == 0:
                continue
            seq /= total
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta a={a:.3g} b={b:.3g} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws (cheap global search)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        logc = rng.uniform(-2.0, 0.7)       # concentration ∈ [10⁻²,~5]
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smoothed mirror‑descent on a soft‑max approximation of max(conv)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of   f_α(a) = 1/α·log Σ_k exp(α·(a*a)_k) .
    Returns (grad, max_conv) where grad has the same shape as a.
    """
    conv = _conv_self(a)                     # length 2n‑1
    max_conv = conv.max()
    # soft‑max weights (stable)
    shifted = conv - max_conv                # ≤0
    exp_shift = np.exp(alpha * shifted)      # may under‑flow to 0 – fine
    w = exp_shift / exp_shift.sum()          # length 2n‑1, sums to 1

    # gradient = (w_rev ⋆ a) + a   (see derivation in analysis)
    w_rev = w[::-1]
    conv_rev = _conv_full_fft(w_rev, a)      # length 3n‑2
    L = len(w)                               # =2n‑1
    grad = conv_rev[L - 1 - np.arange(a.shape[0])] + a
    return grad, max_conv


def _smoothed_mirror_descent(a0: np.ndarray,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float):
    """
    Run mirror‑descent with a schedule of increasing smoothness.
    """
    a = a0.copy()
    n = a.shape[0]

    # schedule of α – start smooth, end sharp
    alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0]
    total_iters = 0
    for alpha in alphas:
        it = 0
        eta0 = 0.8            # initial step‑size (tuned empirically)
        while time.time() < deadline:
            it += 1
            total_iters += 1
            eta = eta0 / np.sqrt(it)
            grad, max_conv = _smooth_gradient(a, alpha)

            # exponentiated‑gradient update (mirror descent on simplex)
            a = a * np.exp(-eta * grad)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

            # occasional true‑objective check
            if it % 30 == 0:
                sc = _objective_score(a)   # sum == 1, so this equals the true objective
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] α={alpha:.1f} it={total_iters:,} score={sc:.6f}")

        # move to the next α (keep the current a as start point)
        if time.time() >= deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Stochastic local refinement (hill‑climbing)
# ----------------------------------------------------------------------
def _local_refinement(a0: np.ndarray,
                      best_seq: np.ndarray,
                      best_score: float,
                      deadline: float):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick two distinct positions
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # amount to move: a random fraction (up to 20 %) of the donor's mass
        eps = rng.uniform(0.0, 0.2) * a[i]
        if eps <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # keep in simplex
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        sc = _objective_score(a_new)
        if sc < best_score:
            best_score, best_seq = sc, a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] Perturb i={i} j={j} eps={eps:.6g} score={sc:.6f}")
        # (optional) occasional acceptance of worse moves could be added,
        # but the simple greedy policy works well when we are already close.
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the provided evaluation function.  The routine follows the four stages
    described above and returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – start from the supplied global construction if any
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0:
                raise ValueError
            starter = starter / starter.sum()
        else:
            raise ValueError
    except Exception:
        starter = None

    if starter is None:
        n = 4096
        a = np.full(n, 1.0 / n)          # uniform baseline
    else:
        a = starter
        n = a.shape[0]

    best_seq = a.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # 1️⃣ Parametric sweeps (≈150 s total)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_beta(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet exploration (≈200 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Smoothed mirror‑descent (majority of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed mirror‑descent")
        best_seq, best_score = _smoothed_mirror_descent(best_seq, best_seq, best_score,
                                                        deadline - 30.0)   # leave a little room
        print(f"[{time.time() - start_time:.2f}s] End of mirror‑descent, best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Local stochastic refinement (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting local stochastic refinement")
        best_seq, best_score = _local_refinement(best_seq, best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # Final normalisation and output
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{final_score:.6f}")

    # Return as plain Python list
    return [float(x) for x in best_seq]
