def _momentum_subgradient(a0, best_score, L_fft, deadline, start_time, rng):
    a = a0.copy()
    v = np.zeros_like(a)
    beta = 0.9
    it = 0
    step0 = 0.5
    best_seq = a.copy()
    best = best_score
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.5)
        v = beta * v - step * g
        cand = a + v
        cand = _project_simplex(cand)
        # Optionally symmetrize
        #cand = (cand + cand[::-1]) / 2.0
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best:
            best = cand_score
            best_seq = cand.copy()
            # reset momentum to zero to avoid overshoot
            v = np.zeros_like(a)
            a = cand
            print(f"[{time.time()-start_time:.2f}s] momentum subgrad it={it:,} score={cand_score:.6f}")
            continue
        # If not better, still update a to cand to continue
        a = cand
        # occasional jitter
        if it % 1000 == 0:
            eps = rng.random(a.shape[0]) * 1e-4 * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best

def _simulated_annealing_large(a_start, score_start, L_fft, deadline, start_time, rng):
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]
    T0 = 5e-3   # start temperature (increase)
    Tf = 1e-7
    # set max transfer fraction initial 0.02 (2%)
    while time.time() < deadline:
        fraction = (time.time() - start_time) / (deadline - start_time)

    return best, best_score

"""
Hybrid optimiser for the L∞‑convolution bound.

The algorithm works in five stages:
1️⃣  Random search over the two‑parameter family
    a_i ∝ (i+1)^p·exp(‑λ·i)  (monotone or symmetric).
2️⃣  Local refinement of the best (λ,p) by Gaussian steps.
3️⃣  Momentum projected sub‑gradient descent on the full vector.
4️⃣  Exponentiated‑gradient (mirror‑descent) fine tuning.
5️⃣  Aggressive simulated‑annealing hill‑climb with larger mass moves.

All stages respect a global 1000 s budget; the best sequence found is
returned (normalised to sum = 1) as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """
    Monotone shape: a_i ∝ (i+1)^p·exp(-lam·i).  Returned vector is L1‑normalised.
    """
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)          # stabilise exp
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """
    Symmetric “U‑shape”: a_i ∝ (i+1)^p·exp(-lam·i) + (n‑i)^p·exp(-lam·(n‑1‑i)).
    Normalised to unit L1‑norm.
    """
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)                      # decreasing part
    right = left[::-1]                       # mirror
    shape = left + right
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Momentum projected sub‑gradient descent
# ----------------------------------------------------------------------


def _momentum_subgradient(
    a0: np.ndarray,
    best_score: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum."""
    a = a0.copy()
    v = np.zeros_like(a)
    beta = 0.9
    it = 0
    step0 = 0.5
    best_seq = a.copy()
    best = best_score

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        step = step0 / (it ** 0.5)
        v = beta * v - step * g
        cand = a + v
        cand = _project_simplex(cand)

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            eps = 4e-5 * rng.random(cand.shape[0]) * cand
            cand = np.maximum(cand + eps, 0.0)
            cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best:
            best = cand_score
            best_seq = cand.copy()
            # reset momentum on improvement
            v = np.zeros_like(a)
            a = cand
            print(f"[{time.time()-start_time:.2f}s] momentum subgrad it={it:,} score={cand_score:.6f}")
        else:
            a = cand

    return best_seq, best


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) phase
# ----------------------------------------------------------------------


def _expgrad_phase(
    a0: np.ndarray,
    best_score: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Multiplicative updates a ← a·exp(‑η g) with annealed η."""
    a = a0.copy()
    best_seq = a.copy()
    best = best_score
    it = 0
    eta0 = 0.8

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        eta = eta0 / (1.0 + 0.0008 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        a = a / a.sum()

        score = _eval_fast(a, L_fft)
        if score < best:
            best = score
            best_seq = a.copy()
            print(f"[{time.time()-start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")

        if it % 3000 == 0:
            eps = 5e-5 * rng.random(a.shape[0]) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best_seq, best


# ----------------------------------------------------------------------
# Aggressive simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def _simulated_annealing_large(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny to medium mass‑transfer moves with a slowly decreasing temperature."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    T0 = 5e-3          # start temperature (more permissive)
    Tf = 1e-7          # final temperature

    while time.time() < deadline:
        # linear cooling based on elapsed fraction
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # up to 2 % of the source mass (bigger jumps than the baseline)
        eps = cur[src] * rng.random() * 2e-2
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{time.time()-start_time:.2f}s] SA improved score={cand_score:.6f}")

        # occasional global jitter (smooths the profile)
        if rng.random() < 0.002:
            jitter = 5e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = _project_simplex(cur)
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time()-start_time:.2f}s] SA jitter improved score={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

    The routine follows a multi‑stage scheme described in the module docstring.
    Returns a plain Python list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try the global starter if it exists and matches our size.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                cur_score = _eval_fast(starter, L_fft)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (≈30 s).
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 30.0)
    best_family = "mono"
    best_lam = 0.0
    best_p = 0.0

    while time.time() < phase1_deadline:
        lam = rng.uniform(0.0, 0.002)        # λ ∈ [0, 2·10⁻³]
        p = rng.uniform(-6.0, 3.0)          # p ∈ [‑6, 3]
        family = "sym" if rng.random() < 0.5 else "mono"
        if family == "mono":
            a = _make_geo_power(n, lam, p, log_i)
        else:
            a = _make_geo_power_symmetric(n, lam, p, log_i)
        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_family = family
            best_lam = lam
            best_p = p
            print(
                f"[{time.time() - start_time:.2f}s] random {family} λ={lam:.6e} p={p:.3f} score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Local refinement of (λ,p) (≈20 s).
    # ------------------------------------------------------------------
    phase2_deadline = min(deadline, start_time + 50.0)
    sigma_lam = 2e-4
    sigma_p = 0.2

    while time.time() < phase2_deadline:
        lam_cand = max(0.0, best_lam + rng.normal() * sigma_lam)
        p_cand = best_p + rng.normal() * sigma_p
        if best_family == "mono":
            a = _make_geo_power(n, lam_cand, p_cand, log_i)
        else:
            a = _make_geo_power_symmetric(n, lam_cand, p_cand, log_i)
        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_lam = lam_cand
            best_p = p_cand
            print(
                f"[{time.time() - start_time:.2f}s] refined {best_family} λ={best_lam:.6e} p={best_p:.3f} score={sc:.6f}"
            )
        sigma_lam *= 0.9995
        sigma_p *= 0.9995

    # ------------------------------------------------------------------
    # 3️⃣  Momentum projected sub‑gradient descent (≈400 s).
    # ------------------------------------------------------------------
    sub_deadline = deadline - 400.0
    if time.time() < sub_deadline:
        best_seq, best_score = _momentum_subgradient(
            best_seq, best_score, L_fft, sub_deadline, start_time, rng
        )

    # ------------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient phase (≈200 s).
    # ------------------------------------------------------------------
    exp_deadline = deadline - 200.0
    if time.time() < exp_deadline:
        best_seq, best_score = _expgrad_phase(
            best_seq, best_score, L_fft, exp_deadline, start_time, rng
        )

    # ------------------------------------------------------------------
    # 5️⃣  Aggressive simulated‑annealing final hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing_large(
            best_seq, best_score, L_fft, deadline, start_time, rng
        )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
