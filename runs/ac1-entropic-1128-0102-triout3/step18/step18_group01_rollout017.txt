"""
Improved candidate generator for the “max‑convolution / ℓ1²” problem.

The algorithm works in three stages:

1️⃣  Quick random exploration of the geometric × power family
    (a_i ∝ (i+1)^p·exp(−λ·i)).  A few hundred random (λ,p) pairs are
    evaluated and the best one becomes the starting point.

2️⃣  Projected gradient descent on the *exact* objective.
    For a fixed sum (the simplex) the gradient of
        F(a) = 2·n·max(conv(a,a)) / (∑a)²
    can be written in closed form:
        ∇F = 2·n·(g – 2·M·1)   where
            M   = max(conv(a,a))   (scalar),
            g   = sub‑gradient of the maximal convolution entry,
            1   = vector of ones.
    After each gradient step we project back onto the probability simplex.
    An adaptive step size (scaled by the gradient norm) together with a
    simple back‑tracking line‑search yields reliable decrease.
    Stagnation triggers a small random jitter and occasional restarts
    from the current best.

3️⃣  Final random “hill‑climb”: move tiny amounts of mass between
    random indices and accept only improvements.  This cleans up
    remaining plateaus while the remaining budget lasts.

All three phases respect the overall time budget (default 1000 s) and
use only NumPy and the supplied ``evaluate_sequence`` for the final
report.  The returned list is normalised (sum = 1) and contains only
non‑negative numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear convolution a*b via FFT."""
    n, m = a.shape[0], b.shape[0]
    if L is None:
        L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the index where the maximum is attained.
    For each i involved in the max sum (i+j=k) we get 2·a_j.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    np.add.at(g, i, 2.0 * a[j])
    return g


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return normalised a_i ∝ (i+1)^p · exp(−λ·i)."""
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)           # for stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Phase 1 – random parametric sweep (very cheap)
# ----------------------------------------------------------------------
def _random_sweep(n: int, rng: np.random.Generator,
                  L_fft: int, deadline: float, start_time: float,
                  cur_best_seq: np.ndarray, cur_best_score: float):
    """Explore a few hundred random (λ,p) pairs of the geometric×power family."""
    best_seq, best_score = cur_best_seq, cur_best_score
    # we limit the sweep to at most ~5 s or 200 trials, whichever comes first
    max_trials = 200
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        lam = 10 ** rng.uniform(-7, -3)          # 1e‑7 … 1e‑3
        p = rng.uniform(-10.0, 10.0)              # wide power range
        cand = _make_geo_power(n, lam, p)
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.2e} p={p:.2f} score={sc:.6f}")
    return best_seq, best_score

# Phase 2 – projected gradient descent on the exact objective
# ----------------------------------------------------------------------
def _gradient_descent(a0: np.ndarray,
                      L_fft: int,
                      deadline: float,
                      start_time: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on F(a)=2·n·max(conv)/sum².
    Uses the analytical gradient (including the denominator term) and
    a simple back‑tracking line‑search.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _eval_fast(best, L_fft)

    step_base = 0.3            # controls the absolute step magnitude
    it = 0
    last_improve = start_time

    while time.time() < deadline:
        it += 1
        # compute convolution, locate the maximal entry
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        M = conv[k]

        # sub‑gradient of the maximal entry
        g = _subgrad_max_conv(a, k)

        # exact gradient of the full objective (sum(a) == 1)
        grad = 2.0 * n * (g - 2.0 * M)

        # adaptive step size (scaled by gradient norm)
        norm = np.linalg.norm(grad) + 1e-16
        step = step_base / norm

        improved = False
        # back‑tracking: try a few reductions of the step size
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            cand = _project_simplex(a - step * factor * grad)
            sc = _eval_fast(cand, L_fft)
            if sc < best_score - 1e-12:
                best, best_score = cand.copy(), sc
                a = cand
                improved = True
                last_improve = time.time()
                print(f"[{time.time() - start_time:.2f}s] grad it={it:,} step={step*factor:.3e} score={sc:.6f}")
                break

        if not improved:
            # occasional jitter to escape tiny plateaus
            if it % 400 == 0:
                eps = rng.random(a.shape) * a * 8e-5
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

        # restart from the incumbent best if we have been stuck for a while
        if time.time() - last_improve > 30.0 and it % 2000 == 0:
            eps = rng.random(best.shape) * best * 2e-4
            a = np.maximum(best + eps, 0.0)
            a = _project_simplex(a)
            last_improve = time.time()

    return best, best_score


# ----------------------------------------------------------------------
# Phase 3 – simple random hill‑climb for final polishing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves with a simple acceptance rule.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3          # up to 0.2 % move
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best, best_score = cand.copy(), sc
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={sc:.6f}")

        # occasional larger exploratory jump
        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand = cand / cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best, best_score = cand.copy(), sc
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={sc:.6f}")

    return best, best_score

# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns the sequence as a plain Python list (∑a_i = 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12    # tiny safety margin

    # Problem size: 8192 gives a good trade‑off between quality and speed.
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ---- baseline: uniform distribution ---------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ---- optional global starter -----------------------------------------
    try:
        # height_sequence_1 may have a different length; adapt if possible
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # resample by simple slicing / padding to match length n
                if starter.size > n:
                    starter = starter[:n]
                else:
                    # pad with zeros at the end
                    pad = np.zeros(n - starter.size, dtype=float)
                    starter = np.concatenate([starter, pad])
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_fft)
            if sc < best_score:
                best_seq, best_score = starter.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # ---- Phase 1: cheap random sweep ------------------------------------
    sweep_deadline = start_time + min(8.0, budget_s * 0.01)   # at most 8 s or 1 % of budget
    best_seq, best_score = _random_sweep(n, rng, L_fft,
                                         sweep_deadline, start_time,
                                         best_seq, best_score)

    # ---- Phase 2: projected gradient descent ------------------------------
    gd_deadline = start_time + 0.70 * budget_s    # 70 % of the budget for GD
    if time.time() < gd_deadline:
        best_seq, best_score = _gradient_descent(best_seq, L_fft,
                                                gd_deadline, start_time, rng)

    # ---- Phase 3: final hill‑climb ---------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_fft,
                                           deadline, start_time, rng)

    # ---- Normalisation & final reporting ---------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Optional final score using the provided evaluate_sequence
    try:
        from __main__ import evaluate_sequence
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
