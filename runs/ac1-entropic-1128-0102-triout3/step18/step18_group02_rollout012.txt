"""
Propose a non‑negative step function minimizing
    2 * n * max(conv(a,a)) / (sum(a)**2)

Main ideas:
- Use a larger support (n=8192) to gain finer resolution.
- Generate a rich set of parametric seeds (geometric·power, Beta,
  Gaussian, triangular, sinc, random Dirichlet).
- Warm‑start a smooth‑max L‑BFGS continuation with a long β‑schedule
  up to β=1024, 150 iterations per β.
- Follow with aggressive exponentiated‑gradient and projected sub‑gradient
  polishing, finally a targeted random hill‑climb that moves tiny mass
  from the indices that dominate the current worst convolution entry.
All work respects the 1000 s budget and returns the best sequence (sum⇒1).
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type shape (i+1)^{α-1}·(n-i)^{β-1}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred on the middle, normalised."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    shape /= shape.sum()
    return shape


def _make_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape (linearly rising then falling)."""
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = np.arange(half - 1, 0, -1, dtype=np.float64)
    shape = np.concatenate([inc, dec])
    shape /= shape.sum()
    return shape


def _make_sinc(n: int, width: float) -> np.ndarray:
    """Discrete |sinc| shape centred, normalised."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64) - centre
    # avoid division by zero
    eps = 1e-12
    arg = np.pi * i / width
    shape = np.where(np.abs(arg) < eps, 1.0, np.abs(np.sin(arg) / arg))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max objective wrapped for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable for the smooth‑max surrogate
        fβ(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k(a))

    Variables are unconstrained logits `x`; a = softmax(x).
    Provides both value and gradient w.r.t. x.
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable conversion from logits to probability vector."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        # convolution a*a (full linear)
        conv = _conv_self_fft(a, self.L_fft)

        # smooth‑max weights (soft‑max over β·conv)
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a of the smooth‑max term (unnormalised)
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule for logits (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # smooth‑max surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Polishing subroutines
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    # contribution 2*a[j] (including diagonal case automatically)
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated‑gradient) updates on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 4000 == 0:
            # small random jitter to avoid stalls
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.35
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        cand_score = None
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move based on a simple temperature schedule
            if cand_score is not None:
                prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
                if rng.random() < prob:
                    a = cand
        if it % 3000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


def _targeted_hill_climb(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small‑mass transfers focused on the indices that dominate the current
    worst convolution entry.  The move is always accepted when it improves
    the objective.
    """
    a = a0.copy()
    n = a.shape[0]
    while time.time() < deadline:
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        # indices (i, j) that contribute to conv[k]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        contrib = a[i_vals] * a[j_vals]
        # pick a pair with probability proportional to its contribution
        if contrib.sum() == 0:
            # fallback to a random move
            src = rng.integers(n)
            dst = rng.integers(n)
        else:
            probs = contrib / contrib.sum()
            idx = rng.choice(i_vals.shape[0], p=probs)
            src = i_vals[idx]
            # move to a random destination that is not src
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1
        # transfer a tiny fraction of src to dst
        eps = a[src] * rng.random() * 3e-3  # up to 0.3 % of src mass
        if eps <= 0:
            continue
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_seq = prop.copy()
            a = prop
            print(f"[{time.time() - start_time:.2f}s] targeted hill‑climb score={new_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function minimising the evaluation
    function within the given time budget.  Returns a list of floats
    (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 points gives a finer approximation.
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)          # 16384 for n=8192
    L_grad = _next_pow2(3 * n - 2)         # 32768

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Generate a diverse pool of seeds.
    # ------------------------------------------------------------------
    seeds = [best_seq]  # already contains uniform / starter

    # Geometric×power family (wide sweep)
    r_vals = np.logspace(np.log10(0.80), np.log10(0.9999), 12)
    p_vals = np.linspace(-6.0, 6.0, 25)
    lam_vals = [0.0, 0.1, 0.3, 0.6, 0.9]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Discrete Beta shapes (dense grid)
    alpha_vals = np.linspace(0.2, 5.0, 12)
    beta_vals = np.linspace(0.2, 5.0, 12)
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # Gaussian families
    sigma_vals = np.linspace(n / 20.0, n / 4.0, 8)
    for sigma in sigma_vals:
        seeds.append(_make_gaussian(n, sigma))

    # Triangular shape
    seeds.append(_make_triangular(n))

    # Sinc family (different widths)
    width_vals = np.linspace(n / 30.0, n / 5.0, 6)
    for w in width_vals:
        seeds.append(_make_sinc(n, w))

    # Random Dirichlet draws
    for alpha in [0.1, 0.3, 0.6, 1.0, 2.0]:
        for _ in range(6):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 10
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]

"""
Propose a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

by a multi‑stage search:

1️⃣  Build a large and diverse pool of parametric seeds (geometric×power,
    Beta, Gaussian, triangular, sinc, Dirichlet draws) on a fine grid.
2️⃣  Warm‑start a smooth‑max surrogate and run L‑BFGS‑B for a long β‑schedule
    up to β=1024 (120 iterations per β).  The surrogate is
        fβ(a)=2·n·β⁻¹·log Σ exp(β·conv(a,a))
    and its gradient is obtained analytically via FFTs.  Each seed is
    refined successively for all β values.
3️⃣  Aggressive exponentiated‑gradient polishing on the true L∞‑objective.
4️⃣  Projected sub‑gradient descent with a decreasing step size.
5️⃣  Targeted hill‑climb that moves a tiny mass from the indices that
    dominate the current worst convolution entry, followed by a final
    random fine‑grained hill‑climb.

All phases respect the 1000 s budget; the best sequence (normalised to sum = 1)
is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type shape (i+1)^{α-1}·(n-i)^{β-1}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred on the middle, normalised."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    shape /= shape.sum()
    return shape


def _make_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape (linear up then down)."""
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = np.arange(half - 1, 0, -1, dtype=np.float64)
    shape = np.concatenate([inc, dec])
    shape /= shape.sum()
    return shape


def _make_sinc(n: int, width: float) -> np.ndarray:
    """Absolute sinc shape centred, normalised."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64) - centre
    eps = 1e-12
    arg = np.pi * i / width
    shape = np.where(np.abs(arg) < eps, 1.0, np.abs(np.sin(arg) / arg))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max objective wrapped for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable for the smooth‑max surrogate

        fβ(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k(a))

    Variables are logits x; a = softmax(x).
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        conv = _conv_self_fft(a, self.L_fft)

        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,

        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)

def _subgradient_descent(a0: np.ndarray,

def _targeted_hill_climb(a0: np.ndarray,

        k = int(np.argmax(conv))
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        contrib = a[i_vals] * a[j_vals]
        if contrib.sum() == 0:
            src = rng.integers(n)
            dst = rng.integers(n)
        else:
            probs = contrib / contrib.sum()
            idx = rng.choice(i_vals.shape[0], p=probs)
            src = i_vals[idx]

        prop[src] -= eps

            best_seq = prop.copy()
            a = prop
