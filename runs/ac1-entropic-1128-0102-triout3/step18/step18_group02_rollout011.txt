"""
Improved search for a non‑negative step function minimizing

        2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm follows a three‑phase scheme, but with a larger problem size
(n = 8192) and more aggressive refinement:

1️⃣  Diverse parametric seeds (geometric·power, beta, exponential, Dirichlet)
    are generated and the best K are kept after a cheap exact evaluation.

2️⃣  A continuation on the smooth‑max surrogate
        fβ(a)=2·n·β⁻¹·log Σ exp(β·conv(a,a))
    is performed with L‑BFGS‑B for β = 1,2,…,1024 and many iterations per
    β, starting from each seed.

3️⃣  Fine‑tuning of the best candidate with a high‑β Adam optimiser
    (β up to 4096) directly on the simplex via the soft‑max
    re‑parameterisation a = softmax(x).  After that we apply
    exponentiated‑gradient and projected sub‑gradient polishing,
    finishing with a tiny random hill‑climb.

All steps respect the 1000 s time budget; the best sequence found is
returned as a plain Python list (already normalised to sum = 1).
"""

import time
import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k·conv_k(a) w.r.t. a.
    Returns 2·∑_j a_j·p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_exp_shape(n: int, c: float) -> np.ndarray:
    """Pure exponential shape exp(-c·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: (i+1)^{α-1}·(n-i)^{β-1}, normalised.
    Allows symmetric or skewed U‑/∧‑shapes.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable for L‑BFGS‑B optimisation of the smooth‑max surrogate.
    The variables are logits x (a = softmax(x)).
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        conv = _conv_self_fft(a, self.L_fft)
        max_conv = conv.max()
        # log‑sum‑exp (stable)
        shift = max_conv
        exps = np.exp(self.beta * (conv - shift))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule to logits
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # smooth‑max value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Polishing phases (same as the reference implementation)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient refinement of the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10       # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 gives a tighter bound at still modest cost.
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1 if present).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(globals().get("height_sequence_1"), dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Generate a rich pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [best_seq]   # includes uniform / starter

    # Geometric × power sweep (more aggressive sampling)
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9999), 12)
    p_vals = np.linspace(-5.0, 4.0, 20)
    lam_vals = [0.0, 0.1, 0.3, 0.6, 0.9]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Pure exponential family
    c_vals = np.logspace(np.log10(0.0005), np.log10(0.01), 8)
    for c in c_vals:
        shape = _make_exp_shape(n, c)
        seeds.append(shape)

    # Discrete Beta‑type shapes (expanded parameter range)
    alpha_vals = [0.2, 0.5, 0.8, 1.0, 2.0, 4.0]
    beta_vals = [0.2, 0.5, 0.8, 1.0, 2.0, 4.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # Random Dirichlet draws with various concentrations
    for alpha in [0.1, 0.2, 0.5, 1.0]:
        for _ in range(6):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 6
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Continuation with L‑BFGS on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    max_iter_per_beta = 150     # more iterations per β for deeper optimisation

    for beta in beta_schedule:
        # reserve ~300 s for later polishing phases
        if time.time() >= deadline - 300:
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for idx in range(len(pool)):
            if time.time() >= deadline - 300:
                break
            a0 = pool[idx]
            eps = 1e-12
            x0 = np.log(a0 + eps)

            # L‑BFGS‑B optimisation
            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # ------------------------------------------------------------------
    # 3️⃣  High‑β Adam refinement (smooth‑max with β up to 4096).
    # ------------------------------------------------------------------
    if time.time() < deadline - 200:
        # initialise Adam on logits
        x = np.log(best_seq + 1e-12)
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1 = 0.9
        beta2 = 0.999
        lr0 = 0.2
        t = 0
        beta_ad = 256      # start value (already explored by L‑BFGS)
        while time.time() < deadline - 200:
            t += 1
            # optional β schedule: double every 4000 iterations up to 4096
            if t % 4000 == 0 and beta_ad < 4096:
                beta_ad = min(beta_ad * 2, 4096)
                print(f"[{time.time() - start:.2f}s] Adam β increased to {beta_ad}")

            # compute a = softmax(x)
            shift = np.max(x)
            exps = np.exp(x - shift)
            a = exps / exps.sum()

            # compute convolution and smooth‑max weights
            conv = _conv_self_fft(a, L_fft)
            p = _softmax_weights(conv, beta_ad)

            # gradient w.r.t a
            grad_a = 2.0 * n * _grad_softmax_new(a, p, L_grad)

            # gradient w.r.t logits (softmax chain rule)
            dot = np.dot(a, grad_a)
            grad_x = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (1.0 + 0.0005 * t)   # slowly decaying learning rate
            x -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # occasional true‑objective evaluation
            if t % 250 == 0:
                a_curr = np.exp(x - np.max(x))
                a_curr /= a_curr.sum()
                cur_score = _eval_fast(a_curr, L_fft)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_curr.copy()
                    print(f"[{time.time() - start:.2f}s] Adam it={t:,} β={beta_ad} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 120, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Projected sub‑gradient descent polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 60:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 60, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 6️⃣  Final random hill‑climb (tiny mass transfers).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 4e-3  # ≤0.4 % of source mass
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
