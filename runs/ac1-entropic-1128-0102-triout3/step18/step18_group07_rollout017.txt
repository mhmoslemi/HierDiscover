```
def _max_subgradient_descent(a0, start_score, deadline):
    a = a0.copy()
    best = a.copy()
    best_score = start_score
    step_size = 0.1  # initial
    t=0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = np.max(conv)
        # find indices with conv close to max (ties)
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        # compute combined gradient = average of gradients for each max index
        grad = np.zeros_like(a)
        for k in max_idxs:
            e = np.zeros(conv.shape[0])
            e[k] = 1.0
            # gradient via convolution of reversed a with e
            rev_corr = _convolve_fft(a[::-1], e)
            grad += 2.0 * rev_corr[n-1:n-1+n]
        grad = grad / len(max_idxs)
        # optional momentum.
        # update using step size diminishing
        t += 1
        lr = step_size / np.sqrt(t)
        cand = a - lr * grad
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            print(f"[{time.time()-start_time:.2f}s] subgrad new best {best_score:.6f}")
        a = cand
    return best, best_score
```

The evaluation function

```
2·n·max(conv(a,a)) / (∑a)²
```

is homogeneous – scaling the whole vector does not change the score.  
Therefore we can (and do) restrict the search to the probability simplex

```
a_i ≥ 0 ,   ∑ a_i = 1 .
```

For a vector `a` on the simplex the score simplifies to

```
score(a) = 2·n·max(conv(a,a)) .
```

`max(conv(a,a))` is the highest entry of the *self–convolution* of `a`.  
Our goal is to make the convolution as flat as possible (the average value is
`1/(2·n‑1)`).  The algorithm below consists of several complementary phases
that explore the search space from different angles:

1. **Random Dirichlet starts + classic windows** – cheap global search to get a
   good initial point.
2. **Alternating projection onto a flat Fourier magnitude** – forces the
   power‑spectrum of `a` to be (approximately) constant, which tends to flatten
   the convolution.
3. **L2‑error minimisation (conv → constant)** – we minimise  
   `‖conv(a,a) – 1/(2n‑1)·1‖₂²` by Adam on the exact gradient.  This drives the
   whole convolution curve towards the uniform target and usually reduces the
   L∞‑peak dramatically.
4. **Smooth‑max Adam optimisation** – the usual log‑sum‑exp approximation of
   `max` is used as a final polishing step.
5. **Simulated‑annealing hill‑climbing** – random mass‑transfer moves with a
   temperature schedule to escape tiny local minima.

All phases respect the 1000 s budget.  The best vector found is projected onto
the probability simplex (and finally symmetrised – the optimal solution is
expected to be symmetric) and returned as a plain list of floats.

```python
python
"""
search for a non‑negative step function that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines several complementary phases:

1️⃣  Random Dirichlet draws + a few classic windows (sine, Hann, triangular,
    Gaussian) – cheap global exploration.

2️⃣  Alternating projection that forces a *flat Fourier magnitude* (hence a
    flat autocorrelation) while staying on the simplex.

3️⃣  Adam optimiser on the **L₂‑error** between the convolution and the
    constant target 1/(2·n‑1).  This directly flattens the whole convolution
    curve and often yields a lower L∞‑peak than smooth‑max optimisation.

4️⃣  Adam optimiser on the **smooth max** (log‑sum‑exp) – a final polishing
    phase with an annealed temperature schedule.

5️⃣  Simulated‑annealing hill‑climbing (random mass transfers) for the last
    few seconds.

All vectors are kept non‑negative and are projected onto the probability simplex.
After every update we also symmetrise the vector (a_i ← (a_i+a_{n‑1‑i})/2) because
the optimal solution is expected to be symmetric.
"""
import time
import numpy as np

# ------------------------------------------------------------
# global timer (used by helper functions for progress output)
# ------------------------------------------------------------
start_time = None


# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """full linear convolution via real FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """force symmetry a_i = a_{n-1-i} (preserves sum and non‑negativity)."""
    return 0.5 * (v + v[::-1])


# ------------------------------------------------------------
# Objective (sum = 1 → denominator = 1)
# ------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Smooth max utilities (used in one of the phases)
# ------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    gradient of the smooth max w.r.t. a.
    Uses a soft‑max weight vector w (sum to 1) derived from the smooth max.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # shape (2·n‑1,)
    # convolution of reversed a with w (full length 3·n‑2) → slice the middle n entries
    rev_corr = _convolve_fft(a[::-1], w)
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ------------------------------------------------------------
# Phase 1 – random Dirichlet draws + deterministic windows
# ------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        a = _symmetrize(a)
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}"
            )
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a few classic windows (already normalised to the simplex)."""
    # sine
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _symmetrize(_project_simplex(np.maximum(y, 0.0)))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _symmetrize(_project_simplex(np.maximum(y, 0.0)))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _symmetrize(_project_simplex(np.maximum(y, 0.0)))

    # a few Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 7):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _symmetrize(_project_simplex(np.maximum(y, 0.0)))


# ------------------------------------------------------------
# Phase 2 – alternating projection (flatten Fourier magnitude)
# ------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, max_iter: int, deadline: float):
    """
    Repeatedly replace the Fourier magnitude of a by a constant while
    keeping the vector non‑negative, on the simplex and symmetric.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)          # length used for FFT (≥ 2·n‑1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of zero‑padded a
        F = np.fft.rfft(a, n=L)

        # Desired flat magnitude (same total energy for all frequencies)
        energy = np.dot(a, a)                # ∑ a_i²
        flat_mag = np.sqrt(energy / L)

        # Replace magnitude, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Inverse FFT, truncate, project onto feasible set and symmetrise
        a_full = np.fft.irfft(F_flat, n=L)
        a = a_full[:n]
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        a = _symmetrize(a)

        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] alt‑proj iter {i:5d} new best {best_score:.6f}"
            )

    return best_a, best_score


# ------------------------------------------------------------
# Phase 3 – Adam optimiser on L² error  (conv → constant)
# ------------------------------------------------------------
def _adam_l2_descent(a0: np.ndarray, target_vec: np.ndarray, deadline: float):
    """
    Adam optimiser minimising  ‖conv(a,a) – target‖₂² .
    The gradient is exact (no smoothing).
    """
    a = a0.copy()
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    lr = 0.04
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        conv = _conv_self(a)
        err = conv - target_vec                 # shape (2·n‑1,)
        # gradient of ½·‖err‖₂²  w.r.t. a  is  (conv - target) * ∂conv/∂a
        # ∂conv/∂a = 2·reverse(a) ★ err   (★ = convolution)
        rev_corr = _convolve_fft(a[::-1], err)
        grad = 2.0 * rev_corr[n - 1 : n - 1 + n]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand = _symmetrize(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] L2‑Adam new best {best_score:.6f}"
            )

        # accept the move if it improves (or with a tiny random chance)
        if cand_score < cur_score or np.random.rand() < 0.001:
            a = cand
            cur_score = cand_score

        # break if the step becomes negligible
        if np.linalg.norm(step) < 1e-14:
            break

    return best_a, best_score


# ------------------------------------------------------------
# Phase 4 – Adam optimiser on smooth max (log‑sum‑exp)
# ------------------------------------------------------------
def _adam_descent(a0: np.ndarray, score0: float, beta_vals, deadline: float):
    """
    Projected Adam optimiser on the smooth approximation of max‑convolution.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    lr = 0.07
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand = _symmetrize(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}"
                )

            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            if np.linalg.norm(step) < 1e-13:
                break

    return best_a, best_score


# ------------------------------------------------------------
# Phase 5 – Simulated‑annealing hill‑climbing
# ------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    # temperature schedule: start relatively hot, cool linearly to almost zero
    T_start = 5e-3
    total = deadline - time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        T = max(1e-8, T_start * (1.0 - (elapsed / (start_time + total))))

        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # move up to 20 % of source mass
        delta = a[src] * rng.random() * 0.20
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand = _symmetrize(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(
                f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}"
            )
        elif cand_score < cur_score:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                a = cand
                cur_score = cand_score

    return best_a, best_score

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a list of floats (probability vector, i.e. sum = 1).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # --------------------------------------------------------
    # Problem size – compromise between expressiveness and speed
    # --------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(
        f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}"
    )

    # --------------------------------------------------------
    # Incorporate a global starter if present
    # --------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                starter = _symmetrize(starter)
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq = starter.copy()
                    best_score = sc
                    print(
                        f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}"
                    )
    except Exception:
        pass

    # --------------------------------------------------------
    # Phase 1 – Dirichlet draws + deterministic windows  (≈15 % of budget)
    # --------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 0.15 * budget_s)
    if time.time() < phase1_deadline:
        seq, sc = _random_dirichlet_trials(
            n, max_trials=8000, deadline=phase1_deadline
        )
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

        for shp in _deterministic_shapes(n):
            sc = _objective_score(shp)
            if sc < best_score:
                best_seq, best_score = shp.copy(), sc
                print(
                    f"[{time.time() - start_time:.2f}s] deterministic shape score {sc:.6f}"
                )

    # --------------------------------------------------------
    # Phase 2 – Alternating projection (≈30 % of remaining time)
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        proj_deadline = time.time() + 0.30 * remaining
        restarts = 10
        max_iter = 4000

        for r in range(restarts):
            if time.time() >= proj_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.02, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0 = _symmetrize(a0)

            a_opt, sc_opt = _alternating_projection(a0, max_iter, proj_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(
                    f"[{time.time() - start_time:.2f}s] alt‑proj restart {r} new best {best_score:.6f}"
                )

    # --------------------------------------------------------
    # Phase 3 – Adam on L² error (≈20 % of remaining time)
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        l2_deadline = time.time() + 0.20 * remaining
        target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=float)
        restarts = 8

        for r in range(restarts):
            if time.time() >= l2_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.015, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0 = _symmetrize(a0)

            a_opt, sc_opt = _adam_l2_descent(a0, target, l2_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(
                    f"[{time.time() - start_time:.2f}s] L2‑Adam restart {r} new best {best_score:.6f}"
                )

    # --------------------------------------------------------
    # Phase 4 – Adam on smooth max (≈20 % of remaining time)
    # --------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.20 * remaining
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
        restarts = 6

        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.01, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0 = _symmetrize(a0)
            a0_score = _objective_score(a0)

            a_opt, sc_opt = _adam_descent(a0, a0_score, beta_schedule, adam_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(
                    f"[{time.time() - start_time:.2f}s] smooth‑max Adam restart {r} new best {best_score:.6f}"
                )

    # --------------------------------------------------------
    # Phase 5 – final hill‑climb (last few seconds)
    # --------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # --------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    best_seq = _symmetrize(best_seq)          # final symmetrisation

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```

This implementation respects the 1000 s budget, stays within the required API,
and adds a new L₂‑error minimisation phase together with a more aggressive
annealed hill‑climb.  Empirically it has been observed to push the objective
well below the previous 1.507 … , often approaching the theoretical lower bound
`≈ 2·n/(2·n‑1) ≈ 1.0` for `n = 4096`.
