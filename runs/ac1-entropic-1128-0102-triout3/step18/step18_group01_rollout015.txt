"""
Propose a non‑negative step function that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1️⃣  A fast parametric sweep over several low‑dimensional families:
        – (i+1)^p·exp(-λ·i)      (geometric×power, λ∈[0,5·10⁻³], p∈[-8,5])
        – Gaussian                (exp(-c·i²))
        – Beta‑like               (x^α·(1‑x)^β on the grid x=i/(n‑1))
        – Sinusoidal power        (sin(π·(i+0.5)/n)^γ)
   The best candidate of the sweep is kept.

2️⃣  Refine the candidate with a *smooth‑max* surrogate of the
    L∞‑objective and Adam‑style projected gradient descent.
    The surrogate is
        smooth_maxγ(conv) = max(conv) + log‑sum‑exp(γ·(conv-max))/γ .
    For growing γ the surrogate approaches the true max.
    Gradients are obtained analytically using the convolution
    structure and each step is projected onto the probability simplex.

3️⃣  Final random hill‑climb (tiny mass transfers) for the last few seconds.

The whole search respects the 1000 s budget and returns the best sequence
found as a plain list of floats (sum normalised to 1).  The implementation
contains many print statements to trace progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – works for arbitrary sum
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return normalized a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)           # stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Return normalized a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape on the grid x=i/(n‑1): a_i ∝ x^α·(1‑x)^β."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1.0)
    # avoid log(0)
    eps = 1e-16
    log_shape = alpha * np.log(np.maximum(x, eps)) + beta * np.log(np.maximum(1.0 - x, eps))
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_sinusoid(n: int, gamma: float) -> np.ndarray:
    """Sinusoidal power: a_i ∝ sin(π·(i+0.5)/n)^γ."""
    i = np.arange(n, dtype=np.float64)
    arg = np.pi * (i + 0.5) / n
    base = np.sin(arg)
    # protect against zeros (first/last entry)
    base = np.maximum(base, 1e-12)
    log_shape = gamma * np.log(base)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max surrogate utilities
# ----------------------------------------------------------------------


def _smooth_max_and_weights(conv: np.ndarray, gamma: float):
    """
    Compute a smooth approximation of max(conv) using log‑sum‑exp
    and return the weights w_i = exp(gamma·conv_i) / Σ exp(...).
    """
    max_c = conv.max()
    shifted = conv - max_c                # improve stability
    exp_vals = np.exp(gamma * shifted)
    sum_exp = exp_vals.sum()
    smooth = max_c + math.log(sum_exp) / gamma
    weights = exp_vals / sum_exp
    return smooth, weights


def _grad_smooth_max(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of smooth_max(conv(a,a)) w.r.t. a.
    Using the identity ∂conv_k/∂a_i = 2·a_{k-i} (when index in range).
    The result is 2·(a *_{cross} weights)[i] where the cross‑correlation is
    performed via a convolution with the reversed weights vector.
    """
    n = a.shape[0]
    # weights length = 2·n‑1
    rev_weights = weights[::-1]
    # full convolution a * rev_weights  (length 3·n‑2)
    conv_rev = np.convolve(a, rev_weights, mode='full')
    # extract the needed slice: indices (2·n‑2 - i)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_rev[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max Adam‑style projected gradient descent
# ----------------------------------------------------------------------


def _smooth_adam_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimizer on the smooth‑max surrogate.
    The iterates are projected onto the probability simplex after each step.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_alpha = 0.05          # base learning rate (tuned empirically)
    t = 0

    # start with a modest smoothing
    gamma = 5.0

    while time.time() < deadline:
        t += 1
        # current convolution
        conv = _conv_self_fft(a, L_fft)
        smooth, weights = _smooth_max_and_weights(conv, gamma)

        # gradient of the smoothed objective (note sum(a)=1 ⇒ scaling factor 2·n)
        grad_smooth = _grad_smooth_max(a, weights)          # d(smooth)/da
        grad = 2.0 * a.shape[0] * grad_smooth                # d(objective)/da

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step_alpha / (np.sqrt(v_hat) + eps)

        cand = a - step * m_hat
        cand = _project_simplex(cand)
        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] adam iter {t:,} score={cand_score:.6f}")
        else:
            # occasional jitter to escape plateaus
            if rng.random() < 0.01:
                eps_j = rng.random(a.shape) * a * 5e-4
                a = np.maximum(a + eps_j, 0.0)
                a = _project_simplex(a)

        # increase smoothing gradually – the optimisation gets more faithful to the true max
        gamma = min(gamma * 1.02, 60.0)

    return best, best_score

# Final random hill‑climb
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer hill‑climb with simple acceptance.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cand_score:.6f}")

        # occasional larger exploratory jump
        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)
            cand = cand / cand.sum()
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand_score:.6f}")

    return best, best_score

        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
