def _subgradient(a):
    n = a.shape[0]
    b = _conv_self(a)
    max_b = b.max()
    # tolerance
    tol = 1e-12 * max_b if max_b>0 else 1e-12
    idx = np.where(b >= max_b - tol)[0]
    w = np.zeros_like(b)
    w[idx] = 1.0 / len(idx)
    # reverse w
    w_rev = w[::-1]
    # correlation
    conv = _fft_convolve(a, w_rev)
    # slice
    start = n-1
    end = 2*n-1
    corr = conv[start:end]  # length n
    grad = 4.0 * n * corr
    return grad, max_b

def _simulated_annealing(a, cur_score, deadline, rng):
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    a_cur = a.copy()
    score_cur = cur_score
    # initialize temperature based on typical drop observed
    # let's estimate initial temperature as 1% of current score
    temp = max(1e-3, 0.01 * cur_score)
    # cooling rate
    cooling = 0.9995
    # parameters
    max_move_factor = 0.02  # 2% of source mass each move
    steps = 0
    while time.time() < deadline:
        steps += 1
        # random move
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a_cur[src] < 1e-15:
            continue
        delta = a_cur[src] * rng.random() * max_move_factor
        cand = a_cur.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < score_cur:
            a_cur = cand
            score_cur = cand_score
            # maybe accept as best
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal improved to {best_score:.6f}")
        else:
            # accept with probability exp(-(cand_score - score_cur)/temp)
            prob = np.exp(-(cand_score - score_cur) / temp)
            if rng.random() < prob:
                a_cur = cand
                score_cur = cand_score
        # cool down
        temp *= cooling
        # optionally stop after many steps
        if steps % 10000 == 0:
            # re-evaluate best if many steps
            pass
        # we can break when temperature extremely low
        if temp < 1e-12:
            break
    return best_a, best_score

"""
Hybrid optimizer for the convolution‑max objective.

We treat the sequence a (non‑negative, sum = 1) and minimise
        F(a) = 2·n·max(conv(a,a)) .
The algorithm consists of three stages:

1. Initialisation – start from the global best if it exists,
   otherwise from a uniform vector or a few random Dirichlet draws.

2. Sub‑gradient descent – an exponentiated‑gradient update that follows
   the sub‑gradient of the max‑convolution (exactly the average of the
   active constraints).  The update is
        a ← a * exp(-η·g) ,   a ← a / Σa
   which stays on the probability simplex.

3. Simulated‑annealing polishing – repeatedly move a tiny random
   fraction of mass from one index to another, accepting both
   improvements and occasional worsening moves according to a cooling
   schedule.  Finally a cheap local refinement (pairwise mass transfer)
   is applied.

The whole routine respects the 1000 s budget and keeps the best
sequence found over all restarts and over several problem sizes.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using an FFT of length
    a power of two.  Result length = len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    L = _next_pow2(n)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·n‑1)."""
    return _fft_convolve(a, a)


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv(a,a)) assuming Σ a = 1.
    The function does **not** perform any clipping – that is done by
    evaluate_sequence when the answer is scored.
    """
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))


# ----------------------------------------------------------------------
# Sub‑gradient descent (exponentiated‑gradient on the simplex)
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray):
    """
    Return the sub‑gradient of max(conv(a,a)) and the current maximum.
    The sub‑gradient is the average of the gradients of all active
    constraints (those attaining the max).
    """
    n = a.shape[0]
    b = _conv_self(a)
    max_b = b.max()
    # indices where the maximum is attained (within a tiny tolerance)
    tol = max(1e-15, 1e-12 * max_b)
    act = np.where(b >= max_b - tol)[0]
    w = np.zeros_like(b)
    w[act] = 1.0 / len(act)   # uniform weight over active indices

    # correlation w ⋆ a  (length 3·n‑2, we need the slice n‑1 … 2·n‑2)
    w_rev = w[::-1]
    conv = _fft_convolve(a, w_rev)          # length = 3·n‑2
    corr = conv[n - 1: 2 * n - 1]            # size n
    grad = 4.0 * n * corr                    # ∂F/∂a (since F = 2·n·max)

    return grad, max_b


def _subgradient_descent(a: np.ndarray, max_iters: int, deadline: float,
                         rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the probability simplex.
    Returns the best sequence encountered and its score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(best_a)

    # base step size – will be scaled by the maximum component of the gradient
    base_eta = 0.5

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        grad, _ = _subgradient(a)

        # scale step so that the largest update is ≈ base_eta
        max_g = grad.max()
        if max_g == 0:
            break
        eta = base_eta / max_g

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = _objective_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter {it:4d}  score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_score: float, deadline: float,
                         rng: np.random.Generator):
    """
    Random pairwise mass transfers with Metropolis acceptance.
    Returns the best sequence seen during the run.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_val = cur_score

    # initial temperature – 1 % of the current objective (heuristic)
    temp = max(1e-6, 0.01 * cur_score)
    cooling = 0.9995
    max_move_frac = 0.02            # move at most 2 % of source mass per step
    steps = 0

    while time.time() < deadline:
        steps += 1
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or cur_a[src] <= 1e-15:
            continue
        delta = cur_a[src] * rng.random() * max_move_frac
        cand = cur_a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)

        if cand_score < cur_val:
            cur_a = cand
            cur_val = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal step {steps:6d}  best={best_score:.6f}")
        else:
            # Metropolis acceptance
            accept_prob = np.exp(-(cand_score - cur_val) / max(temp, 1e-12))
            if rng.random() < accept_prob:
                cur_a = cand
                cur_val = cand_score

        temp *= cooling
        if temp < 1e-12:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Very cheap pairwise refinement (pure improvement)
# ----------------------------------------------------------------------
def _tiny_refine(a: np.ndarray, cur_score: float, deadline: float,
                 rng: np.random.Generator):
    """Move a tiny random fraction of mass and keep only strict improvements."""
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or best[i] <= 1e-15:
            continue
        delta = best[i] * rng.random() * 0.005
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] tiny refine improved to {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Core optimisation driver for a given length n
# ----------------------------------------------------------------------
def _optimise_n(n: int, init_vec: np.ndarray,
                deadline: float, rng: np.random.Generator):
    """
    Optimise a sequence of length n starting from init_vec (will be
    normalised).  The routine performs:
        – sub‑gradient descent,
        – simulated annealing,
        – tiny refinements.
    Returns the best sequence and its objective.
    """
    # initialise and normalise
    a = init_vec.copy()
    if a.shape[0] != n:
        if a.shape[0] > n:
            a = a[:n]
        else:
            a = np.concatenate([a, np.zeros(n - a.shape[0])])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n)

    cur_score = _objective_score(a)
    best_seq = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] start n={n}  score={cur_score:.6f}")

    # --------------------------------------------------------------
    # 1) sub‑gradient descent
    # --------------------------------------------------------------
    sub_max_iters = 3000
    a, sub_best_score = _subgradient_descent(a, sub_max_iters, deadline, rng)
    if sub_best_score < best_score:
        best_seq = a.copy()
        best_score = sub_best_score

    # --------------------------------------------------------------
    # 2) simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a, anneal_best_score = _simulated_annealing(best_seq, best_score,
                                                    deadline, rng)
        if anneal_best_score < best_score:
            best_seq = a.copy()
            best_score = anneal_best_score

    # --------------------------------------------------------------
    # 3) final cheap refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 0.1:
        best_seq, best_score = _tiny_refine(best_seq, best_score, deadline, rng)

    # final normalisation (just to be safe)
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"[{time.time() - start_time:.2f}s] finished n={n}  final score={best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the given
    evaluation.  The method combines exponentiated‑gradient sub‑gradient
    descent, simulated‑annealing polishing and a final cheap local
    refinement.  Several problem sizes are tried and the best result
    is returned.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine a list of lengths to explore (larger n gives more freedom)
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]   # can be trimmed automatically if out of time
    overall_best_seq = None
    overall_best_score = np.inf

    # ------------------------------------------------------------------
    # Base initial vectors (global best if available, otherwise uniform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # may be defined globally
        starter = np.maximum(starter, 0.0)
        if starter.sum() == 0:
            starter = None
    except Exception:
        starter = None

    # ------------------------------------------------------------------
    # Main optimisation loop over chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # use the global starter when its length matches, otherwise uniform
        if starter is not None and starter.size == n:
            init_vec = starter.copy()
        else:
            init_vec = np.full(n, 1.0 / n, dtype=float)

        # run the optimiser for the current length
        seq, score = _optimise_n(n, init_vec, deadline, rng)

        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq.copy()
            print(f"[{time.time() - start_time:.2f}s] *** new overall best (n={n}) score={overall_best_score:.6f}")

        # occasional random restart with a Dirichlet draw (if time permits)
        if time.time() < deadline - 20.0:
            # generate a random dense start (Dirichlet with α=0.5 encourages sparsity)
            rand_start = rng.gamma(shape=0.5, scale=1.0, size=n)
            rand_start = np.maximum(rand_start, 0.0)
            if rand_start.sum() == 0:
                rand_start = np.full(n, 1.0 / n)
            else:
                rand_start /= rand_start.sum()
            seq2, score2 = _optimise_n(n, rand_start, deadline, rng)
            if score2 < overall_best_score:
                overall_best_score = score2
                overall_best_seq = seq2.copy()
                print(f"[{time.time() - start_time:.2f}s] *** new overall best from random start (n={n}) score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # fallback – uniform vector of moderate size
        overall_best_seq = np.full(2048, 1.0 / 2048, dtype=float)

    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    total = overall_best_seq.sum()
    if total > 0:
        overall_best_seq = overall_best_seq / total
    else:
        overall_best_seq = np.full_like(overall_best_seq, 1.0 / overall_best_seq.size)

    print(f"[{time.time() - start_time:.2f}s] DONE – best objective≈{overall_best_score:.6f}")
    return [float(v) for v in overall_best_seq.tolist()]
