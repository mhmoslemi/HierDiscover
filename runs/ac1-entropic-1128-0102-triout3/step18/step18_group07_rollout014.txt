def _random_phase_trials(a, num_trials, deadline):
    L = _next_pow2(2 * len(a) - 1)
    n = len(a)
    flat_mag = 1.0 # let's use 1
    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng()
    for i in range(num_trials):
        if time.time() >= deadline:
            break
        # generate random phases
        # half-spectrum
        # Use rfft representation:
        # length of rfft => L // 2 + 1
        phases = rng.uniform(0, 2*np.pi, size=(L//2+1,))
        # set DC and Nyquist phases to zero
        phases[0] = 0.0
        if L % 2 == 0:
            phases[-1] = 0.0
        F = flat_mag * np.exp(1j * phases)
        a_full = np.fft.irfft(F, n=L)[:n]
        a_candidate = np.maximum(a_full, 0.0)
        a_candidate = _project_simplex(a_candidate)
        sc = _objective_score(a_candidate)
        if sc < best_score:
            best_score = sc
            best_seq = a_candidate.copy()
            print(f"[{time.time() - start_time:.2f}s] random phase trial {i} score={sc:.6f}")
    return best_seq, best_score

"""
Propose a non‑negative step function (a probability vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The algorithm builds a good
starting point with Dirichlet draws / classic windows, then explores the
space with a random‑phase constant‑magnitude construction (the Fourier
magnitude is forced to be flat), follows up with a corrected alternating
projection that truly preserves the total energy, refines the best
candidate with a projected Adam optimiser on a smooth max, and finishes
with a short stochastic hill‑climb.  All phases respect a total wall‑time
budget of 1000 s.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used by helper functions for progress output)
# ----------------------------------------------------------------------
_start_time = None  # will be set inside propose_candidate


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v >= 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    B = A * A
    conv = np.fft.irfft(B, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))

def _random_dirichlet_trials(rng, n: int, max_trials: int, deadline: float):
    best_seq = None
    best_score = float("inf")
    for i in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Dirichlet trial {i:05d} c={c:.4g} score={sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a few classic window shapes, already normalised to the simplex."""
    # sine window
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann window
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # Gaussian windows of various widths
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

# Phase 1 – random constant‑magnitude Fourier constructions
# ----------------------------------------------------------------------
def _random_phase_constant_mag(rng, n: int, max_trials: int, deadline: float):
    """
    Draw random phases, enforce a *flat* Fourier magnitude (energy = 1),
    invert, then project onto the simplex.
    """
    L = _next_pow2(2 * n - 1)            # FFT length (power of two)
    half = L // 2 + 1                    # length of rfft output
    flat_mag = 1.0                       # any positive constant works (projection rescales)
    best_seq = None
    best_score = float("inf")
    for i in range(max_trials):
        if time.time() >= deadline:
            break
        # random phases for all non‑DC / Nyquist bins
        phases = rng.uniform(0.0, 2.0 * np.pi, size=half)
        phases[0] = 0.0                    # keep DC real (does not affect shape after projection)
        if L % 2 == 0:
            phases[-1] = 0.0               # Nyquist component must be real
        F = flat_mag * np.exp(1j * phases)
        a_full = np.fft.irfft(F, n=L)[:n]          # keep only first n samples
        a_candidate = np.maximum(a_full, 0.0)     # enforce non‑negativity
        a_candidate = _project_simplex(a_candidate)
        sc = _objective_score(a_candidate)
        if sc < best_score:
            best_score = sc
            best_seq = a_candidate.copy()
            print(f"[{time.time() - _start_time:.2f}s] Rand‑phase trial {i:05d} score={sc:.6f}")
    return best_seq, best_score

# Phase 2 – corrected alternating projection (true energy preserving)
# ----------------------------------------------------------------------
def _alternating_projection_corrected(a0: np.ndarray, max_iter: int, deadline: float):
    """
    Repeatedly replace the FFT magnitude by a constant that exactly
    preserves the L2‑energy of the signal, while keeping the vector
    non‑negative and on the simplex.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        # zero‑pad and FFT
        a_pad = np.zeros(L)
        a_pad[:n] = a
        F = np.fft.fft(a_pad)

        # constant magnitude that preserves total energy
        total_energy = np.dot(a, a)           # = sum_i a_i^2
        flat_mag = np.sqrt(total_energy)

        # enforce the flat spectrum, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # inverse FFT and truncate
        a_new = np.fft.ifft(F_flat).real[:n]

        # project onto feasible set
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Alt‑proj iter {i:05d} new best {best_score:.6f}")

    return best_a, best_score

# Phase 3 – Adam optimiser on the smooth max (log‑sum‑exp)

def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses grad = 2 * (a_rev ⋆ w)[segment] (w are softmax weights).
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # shape (2·n‑1,)
    rev_corr = np.fft.ifft(np.fft.fft(a[::-1], n=3 * a.shape[0] - 2) *
                           np.fft.fft(w, n=3 * a.shape[0] - 2)).real
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


def _adam_descent(a0: np.ndarray, score0: float,
                  beta_vals, deadline: float):
    """
    Projected Adam optimiser on the log‑sum‑exp smooth approximation of the
    max‑convolution.  The temperature β follows the supplied schedule.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters (a little more conservative)
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - _start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            # accept if improvement or with a tiny chance of uphill move
            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            if np.linalg.norm(step) < 1e-13:
                break

    return best_a, best_score

def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # move up to 10 % of the source mass
        delta = a[src] * rng.random() * 0.10
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - _start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

    return best_a, best_score
