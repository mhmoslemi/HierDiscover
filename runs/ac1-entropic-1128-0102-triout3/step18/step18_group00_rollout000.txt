   maxv = np.max(alpha * conv)
   exp_term = np.exp(alpha * conv - maxv)
   w = exp_term / exp_term.sum()

"""
Gradient‑Smoothing optimiser for the L∞ self‑convolution bound.

The algorithm works with a soft‑max parametrisation a = softmax(theta) so that
the coefficients are non‑negative and sum to one.  A smooth approximation of
the “max” in the objective is built by a soft‑max (log‑sum‑exp) with a
temperature α that is slowly increased from a gentle value (α≈5) to a strong
value (α≈80).  The resulting differentiable objective is minimised with the
Adam optimiser (adaptive step size, momentum).  Every few thousand steps the
exact non‑smooth objective 2·n·max(conv(a,a))/‖a‖₁² is evaluated and the best
sequence is remembered.  After the smoothing phase a short projected
sub‑gradient descent (the same routine used in the baseline) is performed to
polish the solution.

The routine respects the 1000 s budget, prints progress information and
returns the best sequence found as a plain list of Python floats
(normalised to sum = 1).  It can start from the global variable
`height_sequence_1` (if present) or from a random Dirichlet sample.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full linear self‑convolution a*a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²) (fast via FFT)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of conv(a,a) at the index attaining the maximum."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact (non‑smooth) objective.
    Simple back‑tracking line‑search is employed; periodic jitter helps escape
    tiny plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # step decays roughly as 1 / t^0.55
        step = step0 / (it ** 0.55)

        improved = False
        cand = None
        cand_score = None
        for _ in range(8):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(
                    f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                    f"step={step:.5e} score={cand_score:.6f}"
                )
                break
            step *= 0.5

        if not improved:
            # occasional uphill move (tiny probability)
            if cand_score is not None and np.isfinite(cand_score):
                prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
                if rng.random() < prob:
                    a = cand

        # jitter every few thousand iterations
        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

    # ------------------------------------------------------------------

    v = np.zeros_like(theta)

    alpha_start = 5.0

        dot = np.dot(grad_obj, a)

                print(f"[{time.time() - start_time:.2f}s] it={it:,}  exact score={score:.6f}")

    # ------------------------------------------------------------------
    # Polishing phase – a short projected sub‑gradient descent (as in the baseline)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining - 5.0
        best_seq, best_score = _subgradient_descent(
            best_seq, L_conv, sub_deadline, start_time, best_seq, best_score, rng
        )

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best exact score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
