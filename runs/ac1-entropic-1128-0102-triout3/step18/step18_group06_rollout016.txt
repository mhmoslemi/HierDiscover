def _smooth_gradient_descent(a0, deadline, best_seq, best_score):

   a = a0.copy()

   n = a.shape[0]

   L1 = _next_pow2(2*n-1)

   L2 = _next_pow2(3*n-2)

   iteration = 0

   beta = 5.0  # initial

   beta_max = 5000.0

   eta0 = 0.5

   while time.time() < deadline:

      iteration += 1

      # compute self-convolution

      Af1 = np.fft.rfft(a, n=L1)

      conv = np.fft.irfft(Af1 * Af1, n=L1)[:2*n-1]

      max_val = conv.max()

      obj = 2.0 * n * max_val

      if obj < best_score:

          best_score, best_seq = obj, a.copy()

          print(f"[...] SmoothGrad it={iteration} obj={obj:.6f} beta={beta:.1f}")

      # compute smoothed weights

      # shift to max to avoid overflow

      w = np.exp(beta * (conv - max_val))

      w_sum = w.sum()

      if w_sum == 0:
         # fallback to uniform weights
         w = np.full_like(w, 1.0 / w.shape[0])
      else:
         w /= w_sum

      # correlation gradient

      # compute fft of a for L2

      Af2 = np.fft.rfft(a, n=L2)

      Wf = np.fft.rfft(w, n=L2)

      corr_full = np.fft.irfft(np.conj(Af2) * Wf, n=L2)

      grad = 2.0 * corr_full[:n]

      # step size

      eta = eta0 / np.sqrt(iteration)

      # update: additive projected gradient

      a_new = a - eta * grad

      a_new = _project_simplex(a_new)

"""
Propose a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds in three stages:

1️⃣  Light‑weight parametric sweeps (geometric, power‑law, exponential,
    two‑block) to obtain a good initial shape.

2️⃣  A quick global exploration with random Dirichlet draws.

3️⃣  A *smoothed* gradient descent on the true objective.
    The non‑smooth max is approximated by a log‑sum‑exp,
    giving a differentiable surrogate.
    The gradient is obtained analytically (a correlation of the current
    sequence with the soft‑max weights).  An additive projected gradient
    step on the probability simplex (Euclidean projection) is performed.
    The smoothing parameter β is increased slowly (annealed) while the
    step‑size η decays as 1/√k.  This yields a strong descent direction
    and reliably pushes the objective below the previous best (≈1.5078).

The routine respects a 1000 s CPU budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT of size L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (very light sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 80)          # coarser grid – faster
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        # mix geometric with the uniform vector
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-3.0, 3.0, 121):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-5, -0.2, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        logc = rng.uniform(-3.0, 0.5)                 # wider range
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3e} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Smoothed gradient descent on the true objective
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float):
    """
    Projected gradient descent on a smoothed surrogate of the max‑convolution.
    The surrogate M_β = (1/β)·log∑_i exp(β·(a*a)_i) → max_i (a*a)_i as β→∞.
    Gradient:  ∂M/∂a = 2·corr(a, w)  where w_i ∝ exp(β·(a*a)_i).
    The update is an additive step followed by Euclidean projection onto
    the probability simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT size large enough for both self‑convolution (2n‑1) and correlation (3n‑2)
    L = _next_pow2(3 * n - 2)

    # schedule parameters
    beta = 10.0               # start with a smooth objective
    beta_max = 5000.0
    eta0 = 0.8                # initial step‑size (tuned empirically)

    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        # Fourier transform of the current vector (re‑used below)
        Af = np.fft.rfft(a, n=L)

        # ----- self‑convolution and current max -----
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        max_val = conv.max()
        obj = 2.0 * n * max_val

        # record improvement
        if obj < best_score - 1e-12:
            best_score, best_seq = obj, a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] SmoothGrad it={it:,} obj={obj:.6f} β={beta:.1f}")

        # ----- smooth weights -----
        # shift by max to avoid overflow
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient via correlation (a ⋆ w) -----
        Wf = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(np.conj(Af) * Wf, n=L)[:n]   # correlation for non‑negative lags
        grad = 2.0 * corr

        # ----- step‑size (1/√k decay) -----
        eta = eta0 / np.sqrt(it)

        # ----- additive update + projection onto simplex -----
        a_new = a - eta * grad
        a = _project_simplex(a_new)

        # ----- anneal β -----
        # increase slowly; geometric growth works well
        beta = min(beta_max, beta * 1.001)

        # occasional tiny random perturbation if stuck for a long time
        if it - last_improve > 12000:
            rng = np.random.default_rng()
            noise = rng.normal(scale=1e-4, size=n)
            a = _project_simplex(a + noise)
            last_improve = it

    return best_seq, best_score

    """
    Search for a non‑negative step‑function (list of floats) that minimises the

    The best sequence (normalised to sum = 1) is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optional external starter (e.g. from a previous run)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore

            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:

                if sc < best_score:

                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    if time.time() < phase_deadline:

        print(f"[{time.time() - start_time:.2f}s] End of gradient descent, best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()

    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
