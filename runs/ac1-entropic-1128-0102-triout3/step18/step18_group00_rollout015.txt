def _smooth_adam(a0, L_fft, L_grad, deadline, start_time):
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    alpha = 0.001 # smoothing
    alpha_min = 1e-7
    lr0 = 0.1
    while time.time() < deadline:
        t += 1
        # reduce smoothing
        alpha = max(alpha * 0.9999, alpha_min)
        conv = _conv_self_fft(a, L_fft)
        # compute gradient
        w = np.exp(conv / alpha)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        a_rev = a[::-1]
        # compute correlation conv of w and a_rev
        g_full = np.fft.irfft(np.fft.rfft(w, L_grad) * np.fft.rfft(a_rev, L_grad))
        grad = 2.0 * g_full[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        # project onto simplex
        a_candidate = _project_simplex(a_candidate)
        # evaluate
        score = _eval_fast(a_candidate, L_fft)
        if score < best_score:
            best_score = score
            best = a_candidate.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam it={t} score={score:.6f} alpha={alpha:.2e}")
        # optionally accept moving even if not improving? We rely on Adam.
        a = a_candidate
        # occasional jitter or restart?
        if t % 2000 == 0:
            # jitter
            epsj = 5e-5 * np.random.rand(n) * a
            a = np.maximum(a + epsj, 0.0)
            a = _project_simplex(a)
    return best, best_score

"""
Hybrid optimizer for the L∞‑convolution bound.

The search consists of four stages:

1️⃣  Parameter sweep over simple analytic families (geometric‑power,
    symmetric “U‑shape”, classic window functions).  The best seed is kept.

2️⃣  Smoothed Adam descent: we minimise a smooth approximation of
    max(conv(a,a)) obtained by a soft‑max with a gradually shrinking
    temperature α.  Gradients are computed via FFT‑based cross‑correlation
    and the iterate is projected onto the probability simplex (Euclidean
    projection).  This phase yields a strong non‑convex improvement.

3️⃣  Permutation annealing: the multiset of amplitudes is kept fixed,
    but indices are swapped (simulated‑annealing with a linear cooling).
    Since the self‑convolution depends on the ordering of the values,
    a good permutation can substantially lower the peak of the
    convolution.

4️⃣  Optional fine‑tuning with the original projected sub‑gradient /
    exponentiated‑gradient / simulated‑annealing pipeline (run only if
    time permits).

The routine respects a 1000 s budget, prints progress, and finally
returns a list of non‑negative heights (sum normalised to 1).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²).  Assumes a is already 1‑normed."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate vectors (parameter sweep)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(‑lam·i). Returned vector is L1‑normalised."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape”.  Returned vector is L1‑normalised."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _window_family(n: int, kind: str, param: float | None = None) -> np.ndarray:
    """Generate classic window shapes (Hann, Hamming, Blackman, Kaiser)."""
    if kind == "hann":
        w = np.hanning(n)
    elif kind == "hamming":
        w = np.hamming(n)
    elif kind == "blackman":
        w = np.blackman(n)
    elif kind == "kaiser":
        beta = float(param) if param is not None else 0.0
        w = np.kaiser(n, beta)
    else:
        raise ValueError("unknown window")
    # ensure non‑negative (windows are already ≥0) and normalise to 1‑norm
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w.sum()
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used in the original baseline)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _subgradient_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max‑convolution (fallback)."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(8):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(
                    f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                    f"step={step:.5e} score={cand_score:.6f}"
                )
                break
            step *= 0.5
        if not improved:
            # gentle uphill move
            prob = np.exp(-(cand_score - best_score) * 6.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2000 == 0:
            eps = 4e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed Adam (new phase)
# ----------------------------------------------------------------------


def _adam_smoothing(
    a0: np.ndarray,
    L_fft: int,
    L_grad: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth approximation
        α·log Σ exp(conv/α)
    with a decreasing temperature α.  The iterate is kept on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    # start temperature (smoothing) – a fraction of the typical conv magnitude
    conv0 = _conv_self_fft(a, L_fft)
    alpha = max(0.001, conv0.max() * 0.5)  # initial α
    alpha_min = 1e-7
    lr = 0.25  # base learning rate
    while time.time() < deadline:
        t += 1
        # geometric cooling of α
        alpha = max(alpha * 0.99995, alpha_min)

        conv = _conv_self_fft(a, L_fft)                     # length 2n‑1
        # soft‑max weights (stable)
        scaled = conv / alpha
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()                   # length 2n‑1, sums to 1

        # gradient: 2·cross‑correlation of w with reversed a
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, L_grad)
        a_rev_fft = np.fft.rfft(a_rev, L_grad)
        g_full = np.fft.irfft(w_fft * a_rev_fft, n=L_grad)
        grad = 2.0 * g_full[n - 1 : n - 1 + n]               # length n

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam step (projected)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={t:,} "
                f"score={cand_score:.6f} α={alpha:.2e}"
            )
        a = cand

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = 5e-5 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Permutation annealing (re‑ordering of a fixed multiset)
# ----------------------------------------------------------------------


def _perm_anneal(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the ordering of the entries.
    Moves consist of swapping two positions.
    """
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    T0 = 1.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        cand = cur.copy()
        cand[i], cand[j] = cur[j], cur[i]

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] perm‑anneal improved score={cand_score:.6f}"
                )

        # occasional full shuffle to re‑seed the search
        if rng.random() < 0.001:
            rng.shuffle(cur)
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] perm‑shuffle improved score={cur_score:.6f}"
                )

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
        2·n·max(conv(a,a)) / (sum(a)²).

    Returns a plain list of floats (L1‑normalised).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 gives a good trade‑off between expressivity
    # and speed for a 1000 s budget.
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (if available from a previous run).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}"
                )
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parameter sweep (≈150 s).  Families: geometric‑power, symmetric,
    #     and classic windows (Hann, Hamming, Blackman, Kaiser).
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 150.0
    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)

    # geometric‑power grid (coarser than the original run)
    lam_grid = np.geomspace(4e-05, 1.5e-03, 120)   # 120 points
    p_grid = np.linspace(-5.5, 2.5, 70)            # 70 points

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * i_arr
        for p in p_grid:
            # monotone
            log_shape = base + p * log_i
            log_shape -= np.max(log_shape)
            mono = np.exp(log_shape)
            mono /= mono.sum()
            sc = _eval_fast(mono, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = mono.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep mono λ={lam:.6e} "
                    f"p={p:.2f} score={sc:.6f}"
                )
            # symmetric “U‑shape”
            sym = mono + mono[::-1]
            sym /= sym.sum()
            sc_sym = _eval_fast(sym, L_fft)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep sym λ={lam:.6e} "
                    f"p={p:.2f} score={sc_sym:.6f}"
                )

    # window functions
    window_kinds = [("hann", None), ("hamming", None), ("blackman", None)]
    # a few Kaiser betas
    for beta in [0.0, 2.0, 5.0, 10.0, 15.0, 20.0]:
        window_kinds.append(("kaiser", beta))

    for kind, param in window_kinds:
        if time.time() >= sweep_deadline:
            break
        w = _window_family(n, kind, param)
        sc = _eval_fast(w, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = w.copy()
            desc = f"kaiser β={param}" if kind == "kaiser" else kind
            print(
                f"[{time.time() - start_time:.2f}s] sweep window {desc} score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed Adam descent (≈300 s)
    # ------------------------------------------------------------------
    adam_deadline = deadline - 400.0  # reserve time for later phases
    if time.time() < adam_deadline:
        best_seq, best_score = _adam_smoothing(
            best_seq,
            L_fft,
            L_grad,
            adam_deadline,
            start_time,
            best_seq,
            best_score,
            rng,
        )

    # ------------------------------------------------------------------
    # 3️⃣  Permutation annealing (≈300 s)
    # ------------------------------------------------------------------
    perm_deadline = deadline - 150.0
    if time.time() < perm_deadline:
        best_seq, best_score = _perm_anneal(
            best_seq, best_score, L_fft, perm_deadline, start_time, rng
        )

    # ------------------------------------------------------------------
    # 4️⃣  Optional fine‑tuning with the original pipeline (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = deadline - 100.0
        if time.time() < sub_deadline:
            best_seq, best_score = _subgradient_descent(
                best_seq,
                L_fft,
                sub_deadline,
                start_time,
                best_seq,
                best_score,
                rng,
            )
        # final simulated‑annealing hill‑climb
        if time.time() < deadline:
            # reuse the previous SA implementation (tiny mass‑transfer moves)
            # – small wrapper that respects the remaining time.
            def _final_sa(a_start, score_start, L_fft, deadline, start_time, rng):
                cur = a_start.copy()
                cur_score = score_start
                best = cur.copy()
                best_score = cur_score
                n = cur.shape[0]
                T0 = 1.0e-3
                Tf = 5.0e-7
                while time.time() < deadline:
                    frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
                    T = T0 * (1.0 - frac) + Tf * frac
                    src = rng.integers(n)
                    if cur[src] < 1e-16:
                        continue
                    dst = rng.integers(n)
                    if src == dst:
                        continue
                    eps = cur[src] * rng.random() * 5e-3
                    cand = cur.copy()
                    cand[src] -= eps
                    cand[dst] += eps
                    cand_score = _eval_fast(cand, L_fft)
                    delta = cand_score - cur_score
                    if delta < 0 or rng.random() < np.exp(-delta / T):
                        cur = cand
                        cur_score = cand_score
                        if cand_score < best_score:
                            best = cand.copy()
                            best_score = cand_score
                            print(
                                f"[{time.time() - start_time:.2f}s] final SA improved score={cand_score:.6f}"
                            )
                    if rng.random() < 0.001:
                        jitter = 1e-4 * rng.random(n) * cur
                        cur = np.maximum(cur + jitter, 0.0)
                        cur = cur / cur.sum()
                        cur_score = _eval_fast(cur, L_fft)
                        if cur_score < best_score:
                            best = cur.copy()
                            best_score = cur_score
                            print(
                                f"[{time.time() - start_time:.2f}s] final SA jitter improved score={cur_score:.6f}"
                            )
                return best, best_score

            best_seq, best_score = _final_sa(
                best_seq, best_score, L_fft, deadline, start_time, rng
            )

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
