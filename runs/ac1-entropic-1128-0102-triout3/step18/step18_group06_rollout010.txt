def _smooth_objective_and_grad(z, alpha, n):
    # a = softmax(z)
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z)

    # compute convolution a*a via FFT
    L = _next_pow2(2 * n - 1)
    A_f = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A_f * A_f, n=L)[: 2 * n - 1]   # shape 2n-1

    # compute softmax of alpha*conv
    # to avoid overflow:
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp  # length 2n-1
    # weighted avg (smooth max)
    smooth_max = np.dot(w, conv)

    # Compute gradient wrt a:
    # Compute v = sum_i w_i a_{i - k} for each k: using convolution of w with a_rev
    a_rev = a[::-1]
    L2 = _next_pow2((2 * n - 1) + n - 1)  # maybe bigger length
    # Actually use same L2 = _next_pow2(3*n - 2)
    L2 = _next_pow2(3 * n - 2)
    W_f = np.fft.rfft(w, n=L2)
    A_rev_f = np.fft.rfft(a_rev, n=L2)
    v_full = np.fft.irfft(W_f * A_rev_f, n=L2)[: (2 * n - 1) + n - 1]
    # slice:
    start = n - 1
    v = v_full[start:start + n]  # shape n

    # compute sliding window sum W_k:
    # cumulative sum of w:
    cum_w = np.concatenate(([0.0], np.cumsum(w)))  # length 2n

    index_start = np.arange(n)

    grad_a = 2.0 * n * (v + a * W)

    # chain rule to grad_z:
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

"""
Optimiser for the “step‑function” problem.

The search follows three stages:

1️⃣ Very cheap parametric sweeps (geometric, power, exponential, two‑block)
   to obtain a good starting point.

2️⃣ A global Dirichlet‑sampling phase that quickly explores the simplex.

3️⃣ A smooth‑max gradient descent on the simplex.
   • Parameters are represented as logits `z` and turned into a probability
     vector `a = softmax(z)`.
   • The non‑smooth max‑convolution is approximated by a log‑sum‑exp
     (soft‑max) with a slowly increasing temperature `α`.  The exact
     gradient of the smooth objective is derived analytically and applied
     with Adam (first‑order) updates.  As `α` grows the smooth objective
     approaches the true objective, allowing the optimiser to escape
     plateaus and then finely tighten the max‑convolution peak.
   • The descent runs until the time budget is exhausted, constantly
     tracking the true (unsmoothed) objective and keeping the best
     sequence seen.

The routine works for any length; we use `n = 8192`, which gives a
large enough search space while keeping each FFT cheap (≈ 0.001 s).

All numbers are non‑negative; the final sequence is normalised to sum = 1,
so the evaluation function is scale–invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (The real evaluate_sequence multiplies by 1/sum² – the ratio is
    homogeneous, therefore fixing sum=1 is sufficient for comparison.)
    """
    n = seq.shape[0]
    conv = _conv_self_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 80)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time()-start_time:.2f}s] Geom eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time()-start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time()-start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time()-start_time:.2f}s] Two‑block L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (cheap global search)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        logc = rng.uniform(-2.0, 0.7)          # concentration ∈ [10⁻², 5]
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time()-start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Smooth‑max gradient descent (Adam on logits)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, alpha: float, n: int):
    """
    Returns:
        obj_smooth   – 2·n·(soft‑max weighted average of conv)
        grad_z       – gradient w.r.t. logits `z`
        a            – soft‑max probabilities (sum = 1)
    """
    # ----- a = softmax(z) -------------------------------------------------
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    a = exp_z / np.sum(exp_z)

    # ----- conv = a * a (full linear convolution) -------------------------
    Lc = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(Af * Af, n=Lc)[: 2 * n - 1]          # length 2n‑1

    # ----- soft‑max weights on conv --------------------------------------
    scaled = alpha * conv
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / np.sum(exp_s)                               # length 2n‑1
    smooth_max = np.dot(w, conv)                            # soft‑max estimate

    # ----- gradient w.r.t. a ---------------------------------------------
    # term 1 : v_k = Σ_i w_i a_{i‑k}  (cross‑correlation)
    a_rev = a[::-1]
    Lv = _next_pow2(3 * n - 2)                              # enough for full conv
    Wf = np.fft.rfft(w, n=Lv)
    Af_rev = np.fft.rfft(a_rev, n=Lv)
    v_full = np.fft.irfft(Wf * Af_rev, n=Lv)[: (2 * n - 1) + n - 1]
    start = n - 1
    v = v_full[start:start + n]                             # length n

    # term 2 : sliding window sum of w (length‑n windows)
    cum_w = np.concatenate(([0.0], np.cumsum(w)))          # length 2n
    idx = np.arange(n)
    W = cum_w[idx + n] - cum_w[idx]                         # Σ_{i=k}^{k+n-1} w_i

    # assemble ∂obj/∂a
    grad_a = 2.0 * n * (v + a * W)

    # ----- back‑prop through soft‑max (z‑gradient) ------------------------
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    obj = 2.0 * n * smooth_max
    return obj, grad_z, a


def _smooth_descent(start_seq: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float):
    """
    Adam optimiser on the smooth‑max surrogate.
    `alpha` (temperature) is increased gradually so that the optimisation
    starts from a smooth landscape and ends up approximating the true max.
    """
    n = start_seq.shape[0]
    # initialise logits
    eps = 1e-12
    z = np.log(np.maximum(start_seq, eps))

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1

        # temperature schedule (slowly increase)
        if it < 2000:
            alpha = 5.0
        elif it < 5000:
            alpha = 30.0
        elif it < 12000:
            alpha = 150.0
        else:
            alpha = 400.0

        obj_smooth, grad_z, a = _smooth_obj_and_grad(z, alpha, n)

        # true objective (max‑convolution) – cheap, already have `conv`
        conv = _conv_self_fft(a)
        true_obj = 2.0 * n * conv.max()

        if true_obj < best_score:
            best_score, best_seq = true_obj, a.copy()
            print(f"[{time.time()-start_time:.2f}s] Descent it={it:,} true_obj={true_obj:.6f} α={alpha:.1f}")

        # Adam update (step‑size decays as 1/√t)
        lr = 0.08 / np.sqrt(it)
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_opt)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine runs three phases (fast sweeps, Dirichlet sampling,
    smooth‑max Adam descent) and returns the best sequence found.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1    # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a larger `n` gives more freedom at modest cost.
    # --------------------------------------------------------------
    n = 8192
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time()-start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Use a globally supplied starter if present.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time()-start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Light parametric sweeps (≈120 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 600.0   # reserve ~10 min for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet exploration (≈200 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Smooth‑max Adam descent (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting smooth‑max Adam descent")
        best_seq, best_score = _smooth_descent(best_seq, deadline,
                                               best_seq, best_score)
        print(f"[{time.time()-start_time:.2f}s] End of descent – best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Final normalisation & return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time()-start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
