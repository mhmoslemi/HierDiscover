"""
Improved three‑phase optimisation for the L∞ self‑convolution problem.

Phase 1 – diverse seeds (uniform, Dirichlet, geometric, power‑law,
sinusoidal, Gaussian shapes).  Keep the K best seeds after a cheap
exact evaluation.

Phase 2 – continuation on the smooth‑max surrogate  fβ(a)=2·n·β⁻¹·log
∑exp(β·conv(a,a)).  We use a schedule β∈{1,2,4,…,1024} and run L‑BFGS‑B
from every seed, updating the seeds after each β.  A larger β and a
higher iteration limit than the reference code give a tighter
approximation of the true max.

Phase 3 – polishing.  First exponentiated‑gradient descent (multiplicative
updates) for many iterations, then a simulated‑annealing “mass‑transfer”
hill‑climb with an exponential cooling schedule.  All steps respect a
1000 s wall‑time budget; the best sequence found is returned normalized
to sum = 1.
"""

import time
import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (full length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Exact evaluation (fast version, used throughout the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_geo_pow(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑type shape, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian (centred) normalised to sum = 1."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_sinusoid(n: int, power: float) -> np.ndarray:
    """Half‑sinusoid raised to a power, normalised."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    shape = np.sin(np.pi * x) ** power
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex blend with the uniform distribution."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Smooth‑max surrogate (for L‑BFGS)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable that returns the smooth‑max surrogate value and its gradient.
    Variables are unrestricted logits x; the probability vector is a = softmax(x).
    The surrogate is
        fβ(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k(a)).
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        return exp_x / exp_x.sum()

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        conv = _conv_self_fft(a, self.L_fft)

        # smooth‑max weights (soft‑max on β·conv)
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Polishing phases
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            # occasional tiny random jitter
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _simulated_annealing(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mass‑transfer annealing – move tiny amounts of probability between random indices."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    T0 = 0.01  # initial temperature
    while time.time() < deadline:
        it += 1
        # decay temperature linearly with remaining time
        remaining = deadline - time.time()
        total = deadline - start_time
        T = max(T0 * remaining / total, 1e-6)

        # choose source and destination
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-14:
            continue

        # propose transfer of a tiny fraction of src mass
        delta = a[src] * rng.random() * 5e-4   # ≤0.05 % of src
        prop = a.copy()
        prop[src] -= delta
        prop[dst] += delta
        new_score = _eval_fast(prop, L_fft)

        # accept if improvement or with Metropolis probability
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / T):
            a = prop
            if new_score < best_score:
                best_score = new_score
                best_seq = prop.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} score={new_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
        2·n·max(conv(a,a)) / (sum(a)**2)

    Returns a plain Python list (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10  # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good speed/quality trade‑off.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) if defined.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Generate a large pool of diverse seeds.
    # --------------------------------------------------------------
    seeds = [best_seq]  # already contains uniform / starter

    # parametric families
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9995), 9)
    p_vals = np.linspace(-4.0, 3.0, 15)
    lam_vals = [0.0, 0.2, 0.5, 0.8]
    for r in r_vals:
        for p in p_vals:
            shape = _make_geo_pow(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    alpha_vals = [0.5, 1.0, 2.0, 3.0]
    beta_vals = [0.5, 1.0, 2.0, 3.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # additional sinusoidal and Gaussian families
    for power in [0.5, 1.0, 2.0, 4.0]:
        seeds.append(_make_sinusoid(n, power))
    for sigma in [n / 16.0, n / 8.0, n / 4.0]:
        seeds.append(_make_gaussian(n, sigma))

    # random Dirichlet draws with a variety of concentration parameters
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(8):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # --------------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # --------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 10
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Continuation with L‑BFGS on the smooth‑max surrogate.
    # --------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    max_iter_per_beta = 120  # more iterations than the reference

    for beta in beta_schedule:
        if time.time() >= deadline - 300:   # keep ~5 min for polishing
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for idx in range(len(pool)):
            if time.time() >= deadline - 300:
                break
            a0 = pool[idx]
            eps = 1e-12
            x0 = np.log(a0 + eps)          # initialise logits

            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # --------------------------------------------------------------
    # 3️⃣ Exponentiated‑gradient polishing (longer than reference).
    # --------------------------------------------------------------
    if time.time() < deadline - 180:
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              deadline - 180, start,
                                              best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 4️⃣ Simulated‑annealing fine‑grained mass‑transfer.
    # --------------------------------------------------------------
    if time.time() < deadline - 60:
        best_seq, best_score = _simulated_annealing(best_seq, L_fft,
                                                    deadline - 60, start,
                                                    best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 5️⃣ Final random hill‑climb (tiny moves) – safety net.
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 4e-3
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
