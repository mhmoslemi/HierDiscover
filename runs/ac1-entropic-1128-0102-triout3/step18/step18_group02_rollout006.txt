"""
Propose a non‑negative step function (vector a_i ≥ 0, Σa_i = 1) that
minimises 2·n·max(conv(a,a)) / (sum(a)^2).  The optimiser proceeds in three
stages:

1️⃣   Generate a large, diverse pool of candidate shapes (geometric×power,
     beta‑type, sine‑type and random Dirichlet draws) and keep the K best
     after a cheap exact evaluation.

2️⃣   Refine the pool with a continuation of smooth‑max L‑BFGS‑B optimisation.
     The smooth surrogate
         f_β(a)=2·n·β⁻¹·log Σ_k exp(β·conv_k(a))
     is differentiable; a soft‑max re‑parameterisation a=softmax(x) removes the
     simplex constraint.  A schedule of β∈{1,…,1024} with increasingly many
     L‑BFGS iterations drives the solution close to the true L∞ objective.

3️⃣   Long polishing with exponentiated‑gradient and projected sub‑gradient
     descent, followed by small‑mass random hill‑climbing and a few
     convex‑mixes with the uniform and the reversed distribution.

All steps respect the 1000 s wall‑time budget; the best sequence found
is returned as a plain Python list (already normalised to sum = 1).
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a probability vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: (i+1)^{α-1}·(n-i)^{β-1}, normalised.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_sine_shape(n: int, power: float) -> np.ndarray:
    """Sine‑based shape a_i ∝ sin(π·(i+0.5)/n)^power, normalised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = np.sin(np.pi * i / n) ** power
    shape /= shape.sum()
    return shape

# Smooth‑max surrogate wrapped for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------

    """

    Variables are logits `x`; the probability vector is a = softmax(x).
    The surrogate is

        fβ(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k(a))

    Differentiable for any β>0.
    """

        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable soft‑max conversion."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        """
        Compute (fβ(a), ∇_x fβ(a)) at the point x.

        Returns
        -------
        f_val : float
            Smooth‑max surrogate value.
        grad_x : np.ndarray, shape (n,)
            Gradient w.r.t. the logits x.
        """
        a = self._softmax_logits(x)

        # convolution a*a (full linear)
        conv = _conv_self_fft(a, self.L_fft)

        # smooth‑max weights (soft‑max over β·conv)
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a (unnormalised)
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule for logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Polishing phases (exponentiated‑gradient & sub‑gradient descent)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative exponentiated‑gradient updates on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 3000 == 0:
            # occasional jitter to escape plateaus
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move (simulated‑annealing style)
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.15  # safety margin

    # Problem size – 4096 is a good trade‑off between expressiveness and speed
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter exists, incorporate it.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Build a rich pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [best_seq]  # already contains uniform / starter

    # Geometric×power family
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9995), 12)
    p_vals = np.linspace(-5.0, 4.0, 20)
    lam_vals = [0.0, 0.2, 0.5, 0.8]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Discrete Beta‑type family
    alpha_vals = [0.3, 0.5, 0.8, 1.0, 2.0, 3.0, 4.0]
    beta_vals = [0.3, 0.5, 0.8, 1.0, 2.0, 3.0, 4.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # Sine‑type family (different powers)
    for power in [0.5, 1.0, 2.0, 3.0, 4.0]:
        seeds.append(_make_sine_shape(n, power))

    # Random Dirichlet draws with varied concentration
    for concentration in [0.1, 0.2, 0.5, 1.0]:
        for _ in range(6):
            seeds.append(rng.dirichlet(np.full(n, concentration)))

    # ------------------------------------------------------------------
    # Keep the K best after cheap evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 12
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_seq = pool[0].copy()
    best_score = pool_scores[0]
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Continuation with L‑BFGS on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    # iteration budget per β (higher β → more iterations)
    iter_schedule = {
        1: 80, 2: 80, 4: 120, 8: 160, 16: 200,
        32: 240, 64: 280, 128: 320, 256: 400,
        512: 500, 1024: 600,
    }

    for beta in beta_schedule:
        # keep a small reserve for polishing
        if time.time() >= deadline - 180:
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, float(beta))
        max_iter = iter_schedule.get(beta, 200)
        for idx in range(len(pool)):
            if time.time() >= deadline - 180:
                break
            a0 = pool[idx]
            eps = 1e-12
            x0 = np.log(a0 + eps)
            # L‑BFGS‑B optimisation
            res = opt.minimize(fun=obj.fun_and_grad,
                               x0=x0,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': max_iter, 'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()
            cand_score = _eval_fast(a_candidate, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # ------------------------------------------------------------------
    # Final fine‑tuning at a very high β (optional, if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        beta = 2048.0
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        max_iter = 800
        x0 = np.log(best_seq + 1e-12)
        res = opt.minimize(fun=obj.fun_and_grad,
                           x0=x0,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter, 'disp': False})
        best_seq = np.exp(res.x - np.max(res.x))
        best_seq /= best_seq.sum()
        best_score = _eval_fast(best_seq, L_fft)
        print(f"[{time.time() - start:.2f}s] high‑β L‑BFGS β={beta:.0f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing (longer than before).
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 80, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 30, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Small convex mixes with the uniform and the reversed distribution.
    # ------------------------------------------------------------------
    if time.time() < deadline - 20:
        for lam in np.linspace(0.0, 0.05, 11):
            mixed = lam * uniform + (1.0 - lam) * best_seq
            mixed /= mixed.sum()
            score = _eval_fast(mixed, L_fft)
            if score < best_score:
                best_score = score
                best_seq = mixed.copy()
                print(f"[{time.time() - start:.2f}s] uniform mix λ={lam:.3f} score={score:.6f}")

    if time.time() < deadline - 15:
        rev = best_seq[::-1]
        for lam in np.linspace(0.0, 0.05, 11):
            mixed = lam * rev + (1.0 - lam) * best_seq
            mixed /= mixed.sum()
            score = _eval_fast(mixed, L_fft)
            if score < best_score:
                best_score = score
                best_seq = mixed.copy()
                print(f"[{time.time() - start:.2f}s] reversed mix λ={lam:.3f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final random hill‑climb with tiny mass transfers.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # up to 0.4 % of the source mass
        eps = best_seq[src] * rng.random() * 4e-3
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        # re‑normalise (tiny numerical errors may accumulate)
        prop = np.maximum(prop, 0.0)
        prop /= prop.sum()
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
