def _smooth_eg(a, L_self, L_corr, deadline, start, rng):
    # initial parameters
    it = 0
    best = a.copy()
    best_score = _eval_fast(best, L_self)
    eta0 = 0.5
    beta0 = 0.1
    while time.time() < deadline:
        it += 1
        # compute conv self
        conv = _conv_self_fft(a, L_self)
        # softmax weights
        beta = beta0 * (1 + it * 0.001)  # gradually increase
        conv_max = conv.max()
        w = np.exp(beta * (conv - conv_max))
        w = w / w.sum()
        # compute gradient via cross-correlation
        # compute cross-correlation conv of w with reversed a
        a_rev = a[::-1]
        L = L_corr
        W_fft = np.fft.rfft(w, n=L)
        A_rev_fft = np.fft.rfft(a_rev, n=L)
        corr_full = np.fft.irfft(W_fft * A_rev_fft, n=L)
        # extract gradient segment
        # corr_full length >= 3n-2, slice from (n-1) to (n-1+n)
        g = 2.0 * corr_full[n-1: n-1+n]
        # step size
        eta = eta0 / np.sqrt(it)
        # update with exponentiated gradient
        mult = np.exp(-eta * g)
        a = a * mult
        # re-normalize onto simplex (project onto simplex)
        a = _project_simplex(a)
        # compute score
        sc = _eval_fast(a, L_self)
        if sc < best_score:
            best_score, best = sc, a.copy()
            print(f"[{time.time() - start:.2f}s] smoothEG it={it} score={sc:.6f}")
        # occasional jitter
        if it % 2000 == 0:
            eps = rng.random(a.shape) * a * 1e-4
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
        # maybe break after enough iterations
        if it >= 5000:
            break
    return best, best_score

"""
Search for a non‑negative step function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  An extended parametric sweep over a two‑parameter family
    a_i ∝ (i+1)^p·exp(-λ·i) (including strong positive / negative powers)
    and over a pure power‑law a_i ∝ (i+1)^{-p}.  The sweep is fast enough for
    the first ~2 minutes and provides a solid initial candidate.

2️⃣  A smooth exponentiated‑gradient descent on the exact objective.
    Instead of the sub‑gradient at the single maximal index we use a
    soft‑max weighting of *all* convolution entries.
    The gradient is obtained by a cross‑correlation (implemented via FFT)
    and the update is performed on the probability simplex with a
    decreasing step size.  The temperature β of the soft‑max grows
    slowly, turning the smooth objective into the original max‑form.

3️⃣  A final random hill‑climb that makes tiny mass‑transfer moves
    (occasionally larger jumps) and accepts only improvements.
    This fine‑tunes the solution found by the previous phase.

All phases respect the overall 1000 s budget.  The returned sequence is
normalised to sum = 1, consists only of non‑negative floats and can be
passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return a normalised a_i ∝ (i+1)^p·exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)          # stability
    a = np.exp(log_shape)
    a /= a.sum()
    return a


def _make_power_law(n: int, p: float) -> np.ndarray:
    """Return a normalised a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient descent (Phase 2)
# ----------------------------------------------------------------------


def _smooth_eg(a0: np.ndarray,
               L_self: int,
               L_corr: int,
               deadline: float,
               start: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the gradient of the *soft* max
    (soft‑max weighting of all convolution entries) is used.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_self)

    eta0 = 0.5          # initial step‑size
    beta = 10.0         # initial temperature for the soft‑max
    it = 0

    while time.time() < deadline:
        it += 1

        # self‑convolution
        conv = _conv_self_fft(a, L_self)

        # soft‑max weights w_k ∝ exp(β·conv_k)
        cmax = conv.max()
        w = np.exp(beta * (conv - cmax))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·cross_correlation(w, a)  (restricted to the first n entries)
        a_rev = a[::-1]
        W_fft = np.fft.rfft(w, n=L_corr)
        A_fft = np.fft.rfft(a_rev, n=L_corr)
        corr_full = np.fft.irfft(W_fft * A_fft, n=L_corr)
        g = 2.0 * corr_full[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # decreasing step size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update on the simplex
        mult = np.exp(-eta * g)
        a = a * mult
        a = _project_simplex(a)

        # evaluate
        sc = _eval_fast(a, L_self)
        if sc < best_score:
            best_score = sc
            best = a.copy()
            print(
                f"[{time.time() - start:.2f}s] smoothEG it={it:,} β={beta:.2f} score={sc:.6f}"
            )

        # anneal the temperature (slowly turn the smooth objective into the true max)
        beta = min(beta * 1.001, 5000.0)

        # occasional tiny jitter to avoid exact plateaus
        if it % 2000 == 0:
            eps = rng.random(a.shape) * a * 5e-5
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best, best_score

# Final random hill‑climb (Phase 3)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑transfer moves with a simple acceptance rule.
    The move size is a few‑percent of the source mass.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        # choose a source index that actually carries mass
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:

        sc = _eval_fast(cand, L_fft)

            best_score = sc
            best = cand.copy()
            a = cand

            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start:.2f}s] random jitter score={sc:.6f}")

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum → 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.05   # small safety margin

    # --------------------------------------------------------------
    # Problem size – 8192 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 8192
    L_self = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (if the competition provides one).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # bring it to the desired length
            if starter.size > n:
                starter = starter[:n]
            elif starter.size < n:
                starter = np.pad(starter, (0, n - starter.size), "constant")
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Extended parametric sweep (≈120 s)
    # --------------------------------------------------------------
    sweep_deadline = start + 120.0
    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)

    lam_grid = np.logspace(-6, -2.5, 80)          # 1e‑6 … 3e‑3
    p_grid = np.linspace(-12.0, 12.0, 49)        # strong positive & negative powers

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * i_arr
        for p in p_grid:
            log_shape = base + p * log_i
            log_shape -= np.max(log_shape)
            a = np.exp(log_shape)
            a /= a.sum()
            sc = _eval_fast(a, L_self)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(
                    f"[{time.time() - start:.2f}s] sweep λ={lam:.2e} p={p:+.1f} score={sc:.6f}"
                )

    # pure power‑law family
    p_grid2 = np.linspace(0.1, 20.0, 40)
    for p in p_grid2:
        if time.time() >= sweep_deadline:
            break
        a = (i_arr + 1.0) ** (-p)
        a /= a.sum()
        sc = _eval_fast(a, L_self)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start:.2f}s] power‑law p={p:.2f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Smooth exponentiated‑gradient descent (≈60 % of remaining budget)
    # --------------------------------------------------------------
    eg_deadline = start + 0.60 * budget_s
    if time.time() < eg_deadline:
        best_seq, best_score = _smooth_eg(
            best_seq, L_self, L_corr, eg_deadline, start, rng
        )

    # --------------------------------------------------------------
    # 3️⃣ Final hill‑climb (use whatever time is left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_self, deadline, start, rng)

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Report final bound using the provided evaluate_sequence (optional)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
