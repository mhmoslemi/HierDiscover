"""
Improved three‑phase search for a step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm builds on the previous approach but adds several
enhancements:

* Larger problem size (n = 8192) – gives a better asymptotic regime.
* Much richer seed pool (geometric‑power, beta‑shapes, exponential,
  many Dirichlet draws, and several mixtures with the uniform vector).
* Continuation with a smoother surrogate up to β = 1024 and more
  L‑BFGS iterations per β.
* Extra high‑β restarts from perturbed copies of the best solution.
* Longer polishing: exponentiated‑gradient, projected sub‑gradient,
  a guided local‑search that moves mass away from indices that most
  contribute to the current maximal convolution entry, and a final
  random hill‑climb.
* Careful time bookkeeping – everything stops well before the 1000 s
  budget expires.

The function returns a plain list of non‑negative floats whose sum is
normalised to 1.  The returned sequence is the best one found within
the time limit.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Parametric families for seed generation

def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type shape, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_exponential(n: int, r: float) -> np.ndarray:
    """Pure exponential decay a_i ∝ r**i, normalised."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(i * np.log(r))
    shape /= shape.sum()
    return shape

# Smooth‑max objective for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable object that evaluates the smooth‑max surrogate and its gradient.
    The variables are the unconstrained logits `x`; the probability vector is
    obtained by a soft‑max:  a = softmax(x).  The surrogate is

        fβ(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k(a)) .
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        conv = _conv_self_fft(a, self.L_fft)

        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a of the smooth‑max term
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates (exponentiated‑gradient) on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5                     # a bit more conservative than before
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 4000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.25
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


def _guided_local_search(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy guided moves that transfer a tiny amount of mass from indices that
    contribute most to the current maximal convolution entry to those that
    contribute least. Occasionally accepts uphill moves with a simple
    simulated‑annealing rule.
    """
    a = a0.copy()
    n = a.shape[0]
    while time.time() < deadline:
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        contrib = g * a
        src = int(np.argmax(contrib))
        dst = int(np.argmin(contrib))

        if src == dst or a[src] < 1e-15:
            # fall back to a random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst:
                continue

        # move a *relative* amount (up to ~0.4 % of source mass)
        eps = a[src] * rng.random() * 4e-3
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps

        # keep feasibility
        if prop[src] < 0.0:
            prop[src] = 0.0
        s = prop.sum()
        if s == 0.0:
            prop = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            prop /= s

        new_score = _eval_fast(prop, L_fft)

        if new_score < best_score:
            best_score = new_score
            best_seq = prop.copy()
            a = prop
            print(f"[{time.time() - start_time:.2f}s] guided move src={src} dst={dst} eps={eps:.2e} score={new_score:.6f}")
        else:
            # simple temperature schedule: linear decay
            remaining = deadline - time.time()
            total = deadline - start_time
            T = max(1e-8, remaining / total)
            if rng.random() < np.exp(-(new_score - best_score) / (0.1 * T + 1e-12)):
                a = prop

        # occasional jitter to shake out stagnation
        if rng.random() < 0.001:
            eps_vec = rng.random(n) * 1e-4 * a
            a = np.maximum(a + eps_vec, 0.0)
            a /= a.sum()
    return best_seq, best_score

        if time.time() >= deadline - 250:   # keep ~250 s for polishing
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for idx in range(len(pool)):
            if time.time() >= deadline - 250:
                break
            a0 = pool[idx]
            x0 = np.log(a0 + 1e-12)

            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # ------------------------------------------------------------------
    # 2b️⃣ High‑β restarts from perturbed copies of the best solution.
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        high_beta_obj = _SmoothMaxObj(n, L_fft, L_grad, 1024)
        for _ in range(4):
            if time.time() >= deadline - 150:
                break
            # small Gaussian noise in the log‑domain
            log_best = np.log(best_seq + 1e-12)
            perturbed = np.exp(log_best + rng.normal(scale=0.001, size=n))
            perturbed /= perturbed.sum()
            x0 = np.log(perturbed + 1e-12)
            res = scipy.optimize.minimize(fun=high_beta_obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': 80, 'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()
            cand_score = _eval_fast(a_candidate, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] high‑β restart score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Exponentiated‑gradient polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 100:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 100, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣ Projected sub‑gradient descent polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 60:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 60, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣ Guided local search (mass‑transfer based).
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        best_seq, best_score = _guided_local_search(
            best_seq, L_fft, deadline - 30, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 6️⃣ Final random hill‑climb (tiny mass transfers) – uses remaining time.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 5e-3   # up to 0.5 % of source mass
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
