"""
Hybrid optimisation with a smooth‐max gradient stage.

The search proceeds in four phases:

1️⃣  **Coarse parametric sweep** – explores two flexible families:
        * monotone “geo‑power”      a_i ∝ (i+1)^p·exp(‑λ·i)
        * symmetric “U‑shape”      a_i ∝ (i+1)^p·exp(‑λ·i)+(n‑i)^p·exp(‑λ·(n‑1‑i))
   The sweep (≈70 s) furnishes a solid starting vector.

2️⃣  **Smooth‑max projected gradient descent** – replaces the earlier
    sub‑gradient step.  The true max‑convolution is replaced by a
    soft‑max (log‑sum‑exp) with a temperature that is slowly raised.
    The exact gradient of this smooth surrogate can be written as
    a convolution, so each iteration costs only two FFTs.
    After the gradient step the vector is Euclidean‑projected onto the
    probability simplex.  This stage runs for the bulk of the budget
    (≈ 350 s) and typically drives the bound below 1.507.

3️⃣  **Exponentiated‑gradient fine tuning** – a short (≈150 s)
    mirror‑descent phase that multiplies the coordinates by
    exp(‑η·g) and renormalises on the simplex.

4️⃣  **Simulated‑annealing hill‑climb** – the final 150 s clean‑up,
    performing tiny mass‑transfer moves with a linear cooling schedule.

All phases respect the 1000 s total budget.  The returned list is
normalised to sum = 1 and contains only non‑negative floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast objective (identical to the reference implementation)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²).  Assumes a is non‑negative."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate vectors for the sweep
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(‑lam·i).  Returns L1‑normalised."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(
    n: int, lam: float, p: float, log_i: np.ndarray
) -> np.ndarray:
    """Symmetric “U‑shape”.  Returns L1‑normalised."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its exact gradient
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray, gamma: float, L_fft: int):
    """
    Returns (smooth_max, softmax_weights, gradient) where

    * smooth_max = (1/gamma)·log∑exp(gamma·conv)
    * gradient = 2·conv_softmax_a_rev  (length‑n vector)

    The gradient is the exact gradient of the smooth_max w.r.t. a.
    """
    n = a.shape[0]

    # 1. linear self‑convolution a*a
    conv = _conv_self_fft(a, L_fft)                # length 2n‑1
    max_c = conv.max()

    # 2. soft‑max weights on the conv entries (stabilised)
    exp_term = np.exp(gamma * (conv - max_c))
    w = exp_term / exp_term.sum()                  # length 2n‑1, non‑negative

    # 3. gradient = 2·(w * a_rev) convolved, extracted on the central window
    a_rev = a[::-1]
    Lg = _next_pow2(len(w) + len(a_rev) - 1)       # length for the second convolution
    W = np.fft.rfft(w, n=Lg)
    A = np.fft.rfft(a_rev, n=Lg)
    g_full = np.fft.irfft(W * A, n=Lg)[: len(w) + len(a_rev) - 1]

    # central slice corresponds to indices n‑1 … n‑1+n‑1
    start = n - 1
    grad = 2.0 * g_full[start : start + n]

    # smooth approximation of max (not needed for the optimiser, but useful for diagnostics)
    smooth_max = max_c + np.log(exp_term.sum()) / gamma

    return smooth_max, w, grad


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def _smooth_gradient_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max surrogate.
    The temperature (gamma) is slowly increased so that the optimisation
    progressively focuses on the true max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    # initial step‑size (chosen experimentally)
    step0 = 0.8

    while time.time() < deadline:
        it += 1
        # progressively raise the temperature – start soft, end sharp
        gamma = min(3000.0, 5.0 + it * 0.02)  # linear schedule capped at ~3000

        # smooth‑max value, its weights and the exact gradient
        _, w, grad = _smooth_max_and_grad(a, gamma, L_fft)

        # decaying step size (≈1/√t)
        step = step0 / np.sqrt(it)

        improved = False
        # simple back‑tracking line search (max 8 halvings)
        for _ in range(8):
            cand = a - step * grad
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(
                    f"[{time.time() - start_time:.2f}s] smooth‑GD it={it:,} γ={gamma:.1f} "
                    f"step={step:.5e} score={cand_score:.6f}"
                )
                break
            step *= 0.5

        if not improved:
            # occasional stochastic acceptance to climb out of tiny plateaus
            if rng.random() < 0.001:
                a = cand

        # occasional jitter to re‑inject diversity
        if it % 3000 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (unchanged)
# ----------------------------------------------------------------------


def _expgrad_phase(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Multiplicative updates a ← a·exp(‑η g) with annealed η."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.8
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        # sub‑gradient of the exact max (same as before)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        g[i] = 2.0 * a[j]

        eta = eta0 / (1.0 + 0.0008 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        a /= a.sum()

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}"
            )

        if it % 3000 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (unchanged)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with a linearly cooling temperature."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    T0 = 1.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        eps = cur[src] * rng.random() * 5e-3
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = _eval_fast(cand, L_fft)
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = cur / cur.sum()
            cur_score = _eval_fast(cur, L_fft)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

    The routine combines a parametric sweep, a smooth‑max gradient
    descent, an exponentiated‑gradient fine tune and a final SA
    hill‑climb.  The returned list is normalised to sum = 1.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 strikes a good balance.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline – uniform.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter, if supplied.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (≈70 s).
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 70.0
    min_lambda = 4e-05
    max_lambda = 1.5e-03
    lam_grid = np.linspace(min_lambda, max_lambda, 300)   # 300 points
    p_grid = np.linspace(-8.0, 4.0, 121)                  # broader range

    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * i_arr
        for p in p_grid:
            # monotone family
            log_shape = base + p * log_i
            log_shape -= np.max(log_shape)
            mono = np.exp(log_shape)
            mono /= mono.sum()
            sc = _eval_fast(mono, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = mono.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep mono λ={lam:.6e} p={p:.2f} score={sc:.6f}"
                )

            # symmetric family
            left = mono
            sym = left + left[::-1]
            sym /= sym.sum()
            sc_sym = _eval_fast(sym, L_fft)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(
                    f"[{time.time() - start_time:.2f}s] sweep sym λ={lam:.6e} p={p:.2f} score={sc_sym:.6f}"
                )

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max projected gradient descent (major portion of the budget).
    # ------------------------------------------------------------------
    # Reserve ~300 s for the final SA; everything else goes to gradient descent.
    smooth_deadline = deadline - 300.0
    if time.time() < smooth_deadline:
        best_seq, best_score = _smooth_gradient_descent(
            best_seq,
            L_fft,
            smooth_deadline,
            start_time,
            best_seq,
            best_score,
            rng,
        )

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient fine tuning (≈150 s).
    # ------------------------------------------------------------------
    exp_deadline = deadline - 150.0
    if time.time() < exp_deadline:
        best_seq, best_score = _expgrad_phase(
            best_seq,
            L_fft,
            exp_deadline,
            start_time,
            best_seq,
            best_score,
            rng,
        )

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing final hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq,
            best_score,
            L_fft,
            deadline,
            start_time,
            rng,
        )

    # ------------------------------------------------------------------
    # Normalise finally and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
