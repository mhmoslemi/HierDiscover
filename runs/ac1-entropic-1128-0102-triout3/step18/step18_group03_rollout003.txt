"""
Hybrid random‑seed + Adam on a smooth‑max surrogate + long sub‑gradient polishing.

The search works on a simplex vector a (non‑negative, sum = 1) of length n.
The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a)**2) = 2 * n * max(conv(a, a))

(because sum(a) is normalised to 1).

Algorithm outline
-----------------
1.  Build a diverse pool of seed vectors:
    * uniform, the optional global starter, geometric‑power families,
      sine‑shaped, Gaussian‑shaped and very sparse Dirichlet draws.
2.  Keep the K best seeds (cheap exact evaluation).
3.  For each seed run Adam on a smooth‑max surrogate
    fβ(a) = log‑sum‑exp(β·conv(a,a)) with a schedule of β growing up to 1024.
    Adam steps are projected onto the probability simplex.
4.  Perform a long mirror‑descent / sub‑gradient phase directly on the
    L∞ objective (max‑convolution).  The step size slowly decays;
    occasional random perturbations avoid stagnation.
5.  Finish with a tiny random hill‑climb that moves a small fraction of mass.
6.  Return the best vector found (as a plain Python list).

All heavy work uses FFT‑based convolutions, Euclidean projection onto the simplex,
and the exact objective for fast evaluation.
The total running time stays below the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2 * Σ_j a_j p_{i+j}.
    Implemented as a convolution of reversed a with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Exact objective (fast)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generators for seed vectors
# ----------------------------------------------------------------------


def _make_general_shape(n: int, r: float = 0.9, p: float = 0.0, q: float = 0.0) -> np.ndarray:
    """
    a_i ∝ r**i · (i+1)**p · (n-i+1)**q, normalised to sum = 1.
    r∈(0,1) is a geometric decay.
    p and q shape the left/right polynomial tails.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)               # prevent overflow
    shape = np.exp(log_shape)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _sine_shape(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^power, normalised."""
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * (i + 0.5) / n) ** power
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _gaussian_shape(n: int, sigma_frac: float = 0.1) -> np.ndarray:
    """Discrete Gaussian centred in the middle, width = sigma_frac·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    shape = np.exp(- ((i - centre) ** 2) / (2.0 * sigma ** 2))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (2·a_{k‑i} on the support)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start <= i_end:
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        g[i] = 2.0 * a[j]
    return g

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation function.
    Returns a plain Python list whose entries sum to 1.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10          # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 offers a good trade‑off.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (if provided by the environment).
    # --------------------------------------------------------------
    try:
        # height_sequence_1 may have a different length – resize if needed.
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # linear interpolation to the required length
                x_old = np.linspace(0.0, 1.0, starter.size)
                x_new = np.linspace(0.0, 1.0, n)
                starter = np.interp(x_new, x_old, starter)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
            else:
                starter = uniform.copy()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Generate a broad pool of seeds.
    # --------------------------------------------------------------
    seeds = [best_seq]   # start with the uniform (or starter) vector

    # Classic geometric × power family
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9999), 9)
    p_vals = np.linspace(-5.0, 5.0, 11)
    lam_vals = [0.0, 0.4, 0.8]
    for r in r_vals:
        for p in p_vals:
            shape = _make_general_shape(n, r=r, p=p, q=0.0)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Sine‑shaped families
    for power in [0.5, 1.0, 2.0, 3.0]:
        shape = _sine_shape(n, power=power)
        for lam in [0.0, 0.5]:
            seeds.append(_mix_with_uniform(shape, lam))

    # Gaussian‑shaped families
    for sigma_frac in [0.05, 0.1, 0.2]:
        shape = _gaussian_shape(n, sigma_frac=sigma_frac)
        for lam in [0.0, 0.5]:
            seeds.append(_mix_with_uniform(shape, lam))

    # Low‑α Dirichlet draws – generate very sparse vectors
    for alpha in [0.01, 0.02, 0.05]:
        for _ in range(6):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # --------------------------------------------------------------
    # Keep the K best seeds after cheap exact evaluation.
    # --------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 5
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_idx = int(np.argmin(pool_scores))
    best_seq = pool[best_idx].copy()
    best_score = pool_scores[best_idx]
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam on the smooth‑max surrogate (β grows gradually).
    # --------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    adam_lr = 0.55
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for beta in beta_schedule:
        # leave enough room for the polishing phase
        if time.time() >= deadline - 250:
            break
        max_iter_per_vec = 8000
        for idx in range(len(pool)):
            a = pool[idx]
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            cur_score = pool_scores[idx]
            t = 0
            while time.time() < deadline - 250:
                t += 1
                conv = _conv_self_fft(a, L_fft)

                # soft‑max over the convolution entries (stable)
                shift = conv.max()
                exps = np.exp(beta * (conv - shift))
                p_weights = exps / exps.sum()

                # gradient of the surrogate
                g = _grad_softmax_new(a, p_weights, L_grad)

                # Adam update (mirrored descent)
                m = beta1 * m + (1.0 - beta1) * g
                v = beta2 * v + (1.0 - beta2) * (g * g)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)

                a_candidate = a - step
                a_candidate = _project_simplex(a_candidate)

                cand_score = _eval_fast(a_candidate, L_fft)

                # accept improvement outright; occasional uphill moves help escape plateaus
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    if rng.random() < np.exp(-(cand_score - cur_score) * 3.0 / (np.sqrt(t) + 1.0)):
                        a = a_candidate
                        cur_score = cand_score

                # update global best
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start:.2f}s] Adam β={beta} it={t} score={cand_score:.6f}")

                # keep the pool entry up‑to‑date
                if cand_score < pool_scores[idx]:
                    pool_scores[idx] = cand_score
                    pool[idx] = a_candidate.copy()

                if t >= max_iter_per_vec:
                    break

    # --------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient (multiplicative) polishing – modest budget.
    # --------------------------------------------------------------
    if time.time() < deadline - 150:
        a = best_seq.copy()
        it = 0
        eta0 = 1.0
        while time.time() < deadline - 150:
            it += 1
            conv = _conv_self_fft(a, L_fft)
            k = int(conv.argmax())
            g = _subgrad_max_conv(a, k)
            eta = eta0 / (1.0 + 0.0008 * it)
            mult = np.exp(-eta * g)
            a = a * mult
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else uniform.copy()
            score = _eval_fast(a, L_fft)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start:.2f}s] exp‑grad it={it} score={score:.6f}")
            # occasional jitter to avoid exact stagnation
            if it % 4000 == 0:
                jitter = 1e-4 * rng.random(n) * a
                a = np.maximum(a + jitter, 0.0)
                a = a / a.sum()

    # --------------------------------------------------------------
    # 4️⃣  Long projected sub‑gradient descent on the true L∞ objective.
    # --------------------------------------------------------------
    if time.time() < deadline - 60:
        a = best_seq.copy()
        step0 = 0.35
        it = 0
        while time.time() < deadline - 60:
            it += 1
            conv = _conv_self_fft(a, L_fft)
            k = int(conv.argmax())
            g = _subgrad_max_conv(a, k)

            step = step0 / (it ** 0.58)          # slowly decaying step
            cand = a - step * g
            cand = _project_simplex(cand)

            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start:.2f}s] subgrad it={it} step={step:.5e} score={cand_score:.6f}")

            # accept the candidate if it improves, otherwise keep the old point
            if cand_score < _eval_fast(a, L_fft):
                a = cand

            if it % 3000 == 0:
                # small random perturbation to escape flat regions
                eps = 1e-4 * rng.random(n) * a
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

    # --------------------------------------------------------------
    # 5️⃣  Final random hill‑climb (tiny mass transfers).
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move ≤0.2 % of the source mass
        eps_move = best_seq[src] * rng.random() * 2e-3
        prop = best_seq.copy()
        prop[src] -= eps_move
        prop[dst] += eps_move
        prop = _project_simplex(prop)
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
