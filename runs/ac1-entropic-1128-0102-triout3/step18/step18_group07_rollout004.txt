"""
Mirror‑descent optimiser for the max‑convolution objective.

The code builds a set of simple seed windows (uniform, sine, Hann, triangular,
Gaussian) together with a few random Dirichlet draws and keeps the best one.
From that seed we run a large number of *mirror‑descent* (exponentiated‑gradient)
iterations on the sub‑gradient of the max entry of the self‑convolution –
the update shrinks coordinates that contribute heavily to the current maximum
and boosts the others while staying on the probability simplex.  Small random
jitter is injected periodically to escape plateaus.  The final few seconds are
spent polishing the result with a cheap random hill‑climb (mass‑transfer).

The whole routine respects the 1000 s budget, prints progress messages and
returns a plain list of non‑negative floats that can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used only for logging)
# ----------------------------------------------------------------------
start_time = None

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Exact objective for a probability vector: 2·n·max(conv(a,a))."""
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _deterministic_shapes(n: int):
    """Yield a handful of classic windows, already normalised to the simplex."""
    # sine window
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # Gaussian windows of various widths
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

# Main optimisation routine
# ----------------------------------------------------------------------
