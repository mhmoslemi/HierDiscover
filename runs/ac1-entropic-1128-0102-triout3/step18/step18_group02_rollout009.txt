"""
Hybrid search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

Approach
--------
1️⃣  Generate a diverse pool of candidate distributions:
    * uniform baseline
    * parametric families (geometric·power, beta, power‑law tails,
      Laplace‑type, cosine‑power)
    * random Dirichlet draws with several concentrations
    * optional global starter (height_sequence_1)

2️⃣  Keep the best K seeds (cheap exact evaluation).

3️⃣  Annealed smooth‑max optimisation:
    * surrogate  f_β(a)=2·n·β⁻¹·log Σ exp(β·conv(a,a))
    * increasing β (1 → … → 2048) and L‑BFGS‑B on the logits (soft‑max
      re‑parameterisation).  Later β receive more L‑BFGS iterations.

4️⃣  Polishing with the true L∞‑objective:
    * exponentiated‑gradient descent (many iterations)
    * projected sub‑gradient descent (back‑tracking line search)

5️⃣  Final stochastic hill‑climb with larger random mass transfers.

All phases respect the 1000 s time budget; the best vector is returned as
a plain Python list (normalised to sum = 1).  The code works with any
`evaluate_sequence` that follows the specification.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    p : np.ndarray, shape (2·n‑1,)
        Soft‑max weights over the convolution entries.
    L : int
        Length of the zero‑padded FFT (power of two).

    Returns
    -------
    g : np.ndarray, shape (n,)
        Gradient ∂/∂a_i  Σ_k p_k conv_k .
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: (i+1)^{α-1}·(n-i)^{β-1}, normalised.
    Allows symmetric or skewed U‑/∧‑shapes.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_powerlaw_tail_shape(n: int, s: float) -> np.ndarray:
    """Heavy tails at both ends: (i+1)^{-s} + (n-i)^{-s}."""
    i = np.arange(n, dtype=np.float64)
    shape = (1.0 / (i + 1.0) ** s) + (1.0 / (n - i) ** s)
    shape /= shape.sum()
    return shape


def _make_laplace_shape(n: int, scale: float) -> np.ndarray:
    """Symmetric Laplace‑type shape centred in the middle."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-np.abs(i - mid) / scale)
    shape /= shape.sum()
    return shape


def _make_cosine_power_shape(n: int, p: float) -> np.ndarray:
    """U‑shape using |cos|^p (peaks at the ends, zero in centre)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max surrogate wrapped for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable that evaluates the smooth‑max surrogate and its gradient.
    Variables are logits `x`; a = softmax(x).  The surrogate is

        f_β(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k)

    which is smooth for any β>0.
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable soft‑max → probability vector."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        """Return (surrogate value, gradient w.r.t. logits)."""
        a = self._softmax_logits(x)

        # full convolution a*a
        conv = _conv_self_fft(a, self.L_fft)

        # soft‑max weights over β·conv
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a of Σ p_k conv_k
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule to logits
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Polishing phases (same as reference, slightly enlarged step limits)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at convolution index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative exponentiated‑gradient on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.35  # a slightly larger start step
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move based on Metropolis criterion
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1 if present).
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Generate a rich pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [best_seq]  # contains uniform / starter already

    # ---- geometric·power family -------------------------------------------------
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9995), 9)
    p_vals = np.linspace(-4.0, 3.0, 15)
    lam_vals = [0.0, 0.2, 0.5, 0.8]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # ---- beta shapes ------------------------------------------------------------
    alpha_vals = [0.5, 1.0, 2.0, 3.0, 4.0]
    beta_vals = [0.5, 1.0, 2.0, 3.0, 4.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # ---- power‑law tails (U‑shape) ---------------------------------------------
    for s in [0.5, 1.0, 1.5, 2.0, 3.0]:
        shape = _make_powerlaw_tail_shape(n, s)
        for lam in lam_vals:
            seeds.append(_mix_with_uniform(shape, lam))

    # ---- Laplace‑type tails ------------------------------------------------------
    for scale in [200.0, 400.0, 800.0, 1600.0]:
        shape = _make_laplace_shape(n, scale)
        for lam in lam_vals:
            seeds.append(_mix_with_uniform(shape, lam))

    # ---- cosine‑power U‑shape ----------------------------------------------------
    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        shape = _make_cosine_power_shape(n, p)
        for lam in lam_vals:
            seeds.append(_mix_with_uniform(shape, lam))

    # ---- random Dirichlet draws --------------------------------------------------
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(6):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Keep the K best seeds after cheap exact evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 12
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Continuation with L‑BFGS on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]
    # progressively longer inner optimisation for larger β
    max_iter_schedule = [30, 30, 30, 30, 40, 40, 50, 50, 60, 80, 120, 200]

    for beta, max_iter in zip(beta_schedule, max_iter_schedule):
        # leave enough time for polishing at the end
        if time.time() >= deadline - 300:
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for idx in range(len(pool)):
            if time.time() >= deadline - 300:
                break
            a0 = pool[idx]
            eps = 1e-12
            x0 = np.log(a0 + eps)

            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter,
                                                   'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] β={beta} seed={idx} new best {cand_score:.6f}")

            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # ------------------------------------------------------------------
    # 3️⃣  Perturb the current best a few times and re‑optimise (helps escape local minima)
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        for _ in range(40):
            # small Gaussian perturbation + re‑normalisation
            pert = rng.normal(scale=0.03, size=n)
            candidate = best_seq + pert
            candidate = np.maximum(candidate, 0.0)
            if candidate.sum() == 0.0:
                candidate = uniform.copy()
            else:
                candidate /= candidate.sum()
            # run a short L‑BFGS on the perturbed point with moderate β
            beta = 128
            obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
            x0 = np.log(candidate + 1e-12)
            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': 60, 'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()
            cand_score = _eval_fast(a_candidate, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] local‑perturb improvement {cand_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient polishing (extended).
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 120, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Projected sub‑gradient descent polishing (extended).
    # ------------------------------------------------------------------
    if time.time() < deadline - 50:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 50, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 6️⃣  Final random hill‑climb with larger moves.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to ~0.8 % of source mass (larger than before)
        eps = best_seq[src] * rng.random() * rng.uniform(0.002, 0.008)
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        prop = np.maximum(prop, 0.0)
        prop /= prop.sum()
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
