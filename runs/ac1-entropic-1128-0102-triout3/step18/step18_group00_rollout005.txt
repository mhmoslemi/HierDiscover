def propose_candidate(seed=0,budget_s=1000,**kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    # choose n = 4096
    n = 4096
    L_fft = _next_pow2(2*n-1)

    # baseline uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[...] uniform baseline score={best_score}")

    # if height_sequence_1 exists and matches length, use it as candidate
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # adjust length: if different length, we can resample
            if hs.size != n:
                # resample to length n by interpolation
                idx_old = np.linspace(0, 1, hs.size)
                idx_new = np.linspace(0, 1, n)
                hs = np.interp(idx_new, idx_old, hs)
            hs = np.maximum(hs, 0.0)
            if hs.sum() > 0:
                hs = hs / hs.sum()
                sscore = _eval_fast(hs, L_fft)
                if sscore < best_score:
                    best_score = sscore
                    best_seq = hs.copy()
                    print(f"[...] global starter score={sscore}")
    except Exception:
        pass

    # Sweep: same as earlier but maybe limited time.
    sweep_deadline = start_time + min(60.0, budget_s*0.06)
    # define grid
    lam_grid = np.linspace(4e-05, 1.5e-03, 200) # reduce from 300 to 200 for speed
    p_grid = np.linspace(-10.0, 4.0, 141) # step ~0.1

    i_arr = np.arange(n, dtype=np.float64)
    log_i = np.log(i_arr + 1.0)

    for lam in lam_grid:
        if time.time() >= sweep_deadline:
            break
        base = -lam * i_arr
        for p in p_grid:
            # monotone
            log_shape = base + p*log_i
            log_shape -= np.max(log_shape)
            mono = np.exp(log_shape)
            mono_sum = mono.sum()
            if mono_sum == 0:
                continue
            mono = mono / mono_sum
            sc = _eval_fast(mono, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = mono.copy()
                print(f"[...] sweep mono lam={lam:.6e} p={p:.2f} score={sc:.6f}")

            # symmetric
            left = mono
            sym = left + left[::-1]
            sym = sym / sym.sum()
            sc_sym = _eval_fast(sym, L_fft)
            if sc_sym < best_score:
                best_score = sc_sym
                best_seq = sym.copy()
                print(f"[...] sweep sym lam={lam:.6e} p={p:.2f} score={sc_sym:.6f}")

    # Gradient smoothing
    grad_deadline = deadline - 150.0
    if time.time() < grad_deadline:
        best_seq, best_score = _smooth_gradient_descent(best_seq, L_fft, grad_deadline, start_time, best_seq, best_score, rng)

    # Final SA hill climb
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq, best_score, L_fft, deadline, start_time, rng)

    # final normalization and conversion to list
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform

    print(f"[...] Finished – best score ≈ {best_score}")
    return [float(v) for v in best_seq]

"""
Hybrid optimiser for the L∞‑convolution bound.

The search proceeds in three stages:

1️⃣  Parametric sweep over simple monotone / symmetric “geo‑power”
    families (a_i ∝ (i+1)^p·exp(‑λ·i)).  The sweep is limited to a
    modest grid but explores a wider p‑range than before.

2️⃣  Smooth‑gradient descent.
    – Minimise the smooth approximation of the max‑convolution
      M_α(a)= (1/α)·log∑_k exp(α·(a*a)_k)  (α is increased from 1 to 200).
    – The exact gradient ∇M_α is obtained by a second FFT‑based convolution
      (weights w_k = soft‑max).  After each step the vector is projected
      onto the probability simplex.
    – Step size decays roughly as 1/√t; occasional jitter helps to escape
      flat plateaus.

3️⃣  Final simulated‑annealing hill‑climb (tiny mass‑transfer moves)
    to polish the best solution found.

All phases respect the 1000 s budget and the routine returns the best
sequence (normalised to sum = 1) as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv.real


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return the exact objective 2·n·max(conv(a,a))/sum(a)^2."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families for the sweep
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape a_i ∝ (i+1)^p·exp(‑lam·i).  Returned vector is L1‑normalised."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Symmetric “U‑shape”: a_i ∝ (i+1)^p·exp(‑lam·i) + (n‑i)^p·exp(‑lam·(n‑1‑i))."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    sym = left + left[::-1]
    sym /= sym.sum()
    return sym


# ----------------------------------------------------------------------
# Smooth‑gradient descent (soft‑max on the convolution)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    global_start: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth approximation of the max‑convolution.
    The smoothing parameter α is increased linearly from 1 up to 200.
    """
    a = a0.copy()
    n = a.shape[0]
    a_rev = a[::-1]  # will be updated each iteration
    step0 = 0.07
    alpha0 = 1.0
    alpha_max = 200.0
    it = 0
    grad_start = time.time()

    while time.time() < deadline:
        it += 1

        # ----- exact self‑convolution (used for the true score) -----
        conv = _conv_self_fft(a, L_fft)

        # ----- schedule the smoothing parameter α -----
        phase_elapsed = time.time() - grad_start
        phase_total = deadline - grad_start
        frac = min(1.0, phase_elapsed / phase_total)
        alpha = alpha0 + frac * (alpha_max - alpha0)

        # ----- soft‑max weights w_k -----
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_vals = np.exp(scaled - max_scaled)          # stabilise
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp

        # ----- smooth max value (for diagnostics) -----
        smooth_max = (max_scaled + np.log(sum_exp)) / alpha

        # ----- gradient of the smooth max w.r.t. a -----
        #   ∇M = 2 * (w ⋆ a_rev) evaluated on indices n‑1 … 2n‑2
        a_rev = a[::-1]
        L2 = _next_pow2(len(w) + n - 1)
        Wf = np.fft.rfft(w, n=L2)
        Afr = np.fft.rfft(a_rev, n=L2)
        conv_wr = np.fft.irfft(Wf * Afr, n=L2)[: len(w) + n - 1]
        grad = 2.0 * conv_wr[n - 1 : n - 1 + n]  # length n

        # ----- step size -----
        step = step0 / np.sqrt(it)

        # ----- projected gradient update -----
        cand = a - step * grad
        cand = _project_simplex(cand)

        # ----- evaluate exact objective (sum(a)=1 ⇒ score = 2·n·max(conv)) -----
        conv_cand = _conv_self_fft(cand, L_fft)
        score_cand = 2.0 * n * conv_cand.max()  # sum(cand)=1 after projection

        if score_cand < best_score:
            best_score = score_cand
            best_seq = cand.copy()
            print(
                f"[{time.time() - global_start:.2f}s] smooth‑GD it={it:,} "
                f"α={alpha:.1f} score={score_cand:.6f}"
            )

        a = cand

        # occasional jitter to escape tiny plateaus
        if it % 4000 == 0:
            eps = 5e-5 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    global_start: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with a decreasing temperature."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score

    n = cur.shape[0]
    T0 = 2.0e-3
    Tf = 5.0e-7

    while time.time() < deadline:
        # linear cooling based on elapsed fraction of the whole SA phase
        elapsed = time.time() - (deadline - (deadline - global_start) * 0.2)  # approximate start
        frac = (elapsed) / (deadline - (deadline - global_start) * 0.2 + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 0.5 % of source mass (slightly larger than before)
        eps = cur[src] * rng.random() * 5e-3
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        cand_score = 2.0 * n * _conv_self_fft(cand, L_fft).max()
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - global_start:.2f}s] SA improved score={cand_score:.6f}"
                )

        # occasional random jitter to keep diversity
        if rng.random() < 0.001:
            jitter = 1e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = cur / cur.sum()
            cur_score = 2.0 * n * _conv_self_fft(cur, L_fft).max()
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - global_start:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score
