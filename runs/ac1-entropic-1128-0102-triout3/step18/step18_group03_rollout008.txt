"""
Propose a high‑quality non‑negative step‑function (probability vector) that
minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search works as follows

* A fairly large problem size (n = 8192) is chosen – the extra degrees of
  freedom allow the auto‑convolution to be flattened further than with n=4096.
* A very diverse seed pool is built:
    – the uniform vector,
    – the (optional) global starter up‑sampled to length n,
    – geometric‑power shapes, their symmetric counterparts,
    – mixtures with the uniform distribution,
    – low‑α Dirichlet draws,
    – smooth random vectors (Gaussian → absolute → moving‑average),
    – a few random “block” patterns.
* The K best seeds (K=10) are kept and each of them is refined by Adam on the
  smooth‑max surrogate  fβ(a)=log‑sum‑exp(β·conv(a,a))  with a β‑schedule that
  grows up to 4096.  The Adam updates are mirrored (projected onto the
  probability simplex) and occasional uphill moves are allowed.
* After the Adam phase two polishing stages are applied:
    – an exponentiated‑gradient (mirror‑descent with KL‑divergence) that uses
      the exact max‑convolution sub‑gradient,
    – a classic projected sub‑gradient descent on the true L∞ objective.
* Finally a random hill‑climb that moves a tiny fraction of mass between two
  positions is performed until the time budget expires.
* The best vector found is returned as a plain Python list (already normalised
  to sum = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2 * Σ_j a_j p_{i+j}.
    Implemented as a convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    return 2.0 * conv[start:start + n]


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact objective (used for ranking & polishing)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_general_shape(n: int, r: float = 0.9, p: float = 0.0, q: float = 0.0) -> np.ndarray:
    """
    Shape a_i ∝ r**i · (i+1)**p · (n-i+1)**q, normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)                # avoid overflow
    shape = np.exp(log_shape)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _smooth_random(rng: np.random.Generator, n: int, sigma: float) -> np.ndarray:
    """
    Produce a smooth random non‑negative vector:
        1. draw i.i.d. N(0,1) and take absolute value,
        2. convolve with a short Gaussian kernel of width sigma,
        3. normalise to sum = 1.
    """
    x = np.abs(rng.normal(size=n))
    # Gaussian kernel (odd length)
    half = int(3 * sigma)
    t = np.arange(-half, half + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (t / sigma) ** 2)
    kernel /= kernel.sum()
    # cheap convolution (kernel size ≪ n)
    y = np.convolve(x, kernel, mode='same')
    y = np.maximum(y, 0.0)
    y_sum = y.sum()
    if y_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return y / y_sum


def _block_pattern(rng: np.random.Generator, n: int, blocks: int = 20) -> np.ndarray:
    """
    Random piecewise‑constant pattern: divide the domain into `blocks`
    intervals of (approximately) equal size, assign each a random weight,
    and normalise.
    """
    if blocks <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    # generate block weights from a Dirichlet
    w = rng.dirichlet(np.ones(blocks))
    # assign each block a constant value
    out = np.empty(n, dtype=np.float64)
    idx = np.arange(n)
    block_ids = np.floor(idx * blocks / n).astype(int)
    out = w[block_ids]
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out

# Sub‑gradient helpers (used in polishing phases)

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative (exponential) updates on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 4000 == 0:
            # occasional jitter to escape flat regions
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the L∞ objective with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.40
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False

            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # small stochastic acceptance of uphill moves
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score
