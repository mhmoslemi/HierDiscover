def _hill_climb_importance(a0, score0, deadline):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))
        # compute contributions
        contributions = np.zeros(n)
        # for i in range(n):
        #    j = kmax - i
        #    if 0 <= j < n:
        #        contributions[i] = a[i] * a[j]
        # else zero
        # Use vectorized approach
        indices = np.arange(n)
        j = kmax - indices
        mask = (j >= 0) & (j < n)
        contributions[mask] = a[indices[mask]] * a[j[mask]]
        # select src weighted by contributions
        w_total = contributions.sum()
        if w_total > 0:
            src = rng.choice(n, p=contributions / w_total)
        else:
            src = rng.integers(n)
        # choose dst such that xi yields low contribution (e.g., near edges)
        dst = rng.integers(n)
        if src == dst:
            continue
        if a[src] == 0.0:
            continue
        # transfer up to a fraction of src mass, we can use up to 0.2
        delta = a[src] * rng.random() * 0.2
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] imp-hill new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score
    return best_a, best_score

"""
Improved optimiser for 2·n·max(conv(a,a))/(sum(a)²).

The optimiser builds on the previous Frank‑Wolfe style pipeline but adds
several new ideas:

* **Larger problem size** – we use n = 8192 (still fast with FFT).
* **Richer initial pool** – besides the Dirichlet draws we also try
  deterministic windows (linear ramps, exponentials, beta‑shaped PDFs)
  and several specialised random mixtures.
* **More aggressive ADAM phase** – a longer schedule with finer
  learning‑rate control and many restarts.
* **Importance‑driven hill‑climb** – the move that reduces the current
  maximal convolution entry is chosen with probability proportional to
  its contribution (a[i]·a[kmax−i]).  Larger transfer steps (up to 20 % of
  the source mass) are used and the move is accepted whenever it
  improves the objective (or occasionally for diversification).

All phases respect a global 1000 s budget.  The best vector found is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer for logging
# ----------------------------------------------------------------------
start_time = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (sum forced to 1 -> denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient – used by ADAM
# ----------------------------------------------------------------------
def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    Uses the identity grad = 2 * (a_rev ⋆ w) where w are the soft‑max
    weights of the convolution entries.
    """
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()
    rev_corr = _convolve_fft(a[::-1], w)            # length ≈ 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Deterministic “windows” – additional shapes beyond the original code
# ----------------------------------------------------------------------
def _deterministic_shapes(n: int):
    """Yield a handful of classic and custom windows, already normalised."""
    # sine
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

    # linear ramps (upward and downward)
    y = np.arange(1, n + 1, dtype=float)          # 1 … n
    yield _project_simplex(y / y.sum())
    y = y[::-1]                                    # n … 1
    yield _project_simplex(y / y.sum())

    # exponential decay (left‑to‑right and right‑to‑left)
    for rate in (0.001, 0.005, 0.02, 0.1):
        y = np.exp(-rate * np.arange(n))
        yield _project_simplex(y / y.sum())
        y = y[::-1]
        yield _project_simplex(y / y.sum())

    # beta‑shaped PDFs (flexible asymmetric shapes)
    for a, b in ((0.3, 2.0), (2.0, 0.3), (0.5, 0.5), (2.0, 2.0)):
        x = np.linspace(0, 1, n)
        y = (x ** (a - 1)) * ((1 - x) ** (b - 1))
        if np.all(y == 0):
            continue
        yield _project_simplex(y / y.sum())


# ----------------------------------------------------------------------
# Random Dirichlet / Gamma draws with broader concentration range
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻⁴, 10]  (wider than before)
        conc_log = rng.uniform(-4.0, 1.0)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4g} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Alternating projection (flatten Fourier magnitude)
# ----------------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, max_iter: int, deadline: float):
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a, n)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of zero‑padded a
        F = np.fft.rfft(a, n=L)

        # Desired flat magnitude – preserve total energy
        energy = np.dot(a, a)               # Σ a_i²
        flat_mag = np.sqrt(energy / L)

        # Replace magnitude, retain phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Inverse FFT and truncate
        a_full = np.fft.irfft(F_flat, n=L)
        a = a_full[:n]

        # Projection onto feasible set
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        score = _objective_score(a, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] alt‑proj iter {i:5d} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# ADAM optimiser on the smooth max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, score0: float,
                  beta_vals, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score

    # ADAM hyper‑parameters
    lr = 0.09
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            # ADAM update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand, a.shape[0])
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            # accept if improvement (or tiny stochastic acceptance)
            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            # early stop if step becomes negligible
            if np.linalg.norm(step) < 1e-14:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Importance‑driven hill climbing (focus on the current maximal entry)
# ----------------------------------------------------------------------
def _importance_hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # contributions to the maximal entry
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[j[mask]]
        total = contrib.sum()

        # pick a source index proportional to its contribution (or uniform if all zero)
        if total > 0:
            src = rng.choice(n, p=contrib / total)
        else:
            src = rng.integers(n)

        # pick a destination far from the “mirror’’ of src
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue

        # transfer up to 20 % of the source mass (larger steps than before)
        delta = a[src] * rng.random() * 0.20
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] imp‑hill new best {best_score:.6f}")
        elif cand_score < cur_score:
            # allow small downhill moves for exploration
            a = cand
            cur_score = cand_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a list of floats (probability vector).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressiveness and speed
    # ------------------------------------------------------------------
    n = 8192                    # 8 k points gives more flexibility while staying fast
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Incorporate a global starter if present (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n)
                if sc < best_score:
                    best_seq, best_score = starter, sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws + deterministic windows (≈15 % time)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 0.15 * budget_s)
    if time.time() < phase1_deadline:
        max_trials = 5000
        seq, sc = _random_dirichlet_trials(n, max_trials, phase1_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

        for shp in _deterministic_shapes(n):
            sc = _objective_score(shp, n)
            if sc < best_score:
                best_seq, best_score = shp.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] deterministic shape score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – alternating projection (≈30 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        proj_deadline = time.time() + 0.30 * remaining
        max_iter_per_run = 5000          # more iterations per run
        restarts = 12

        for r in range(restarts):
            if time.time() >= proj_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.015, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            a_opt, sc_opt = _alternating_projection(a0, max_iter=max_iter_per_run,
                                                    deadline=proj_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] alt‑proj restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – ADAM optimiser on smooth max (≈35 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.35 * remaining
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 8000.0]
        restarts = 8

        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.008, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)
            a0_score = _objective_score(a0, n)

            a_opt, sc_opt = _adam_descent(a0, a0_score,
                                          beta_vals=beta_schedule,
                                          deadline=adam_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] Adam restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – importance driven hill‑climb (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting importance hill‑climb")
        best_seq, best_score = _importance_hill_climb(best_seq,
                                                      best_score,
                                                      deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
