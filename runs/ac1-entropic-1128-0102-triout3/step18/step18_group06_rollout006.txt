def _subgradient_descent(a0, max_iters, deadline, best_seq, best_score):
    # mirror descent as earlier but break after max_iters or time
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_score:
            best_score,best_seq = obj,a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad_start it={it} obj={obj:.6f}")
        # compute subgrad
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        # same as earlier
        for i in active:
            lo = max(0, i - (n - 1))
            hi = min(n-1, i)
            j = np.arange(lo, hi+1)
            k = i - j
            grad[lo:hi+1] += a[k]
        grad = 2.0 * grad / len(active)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum
    return best_seq, best_score
```

"""
Propose a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm follows a multi‑stage strategy:

1️⃣  Fast parametric sweeps (including geometric, power, exponential,
    two‑block, triangle and Beta‑shaped families) to get strong seeds.

2️⃣  Random Dirichlet draws.  Promising draws are refined with a short
    sub‑gradient (mirror) descent, keeping the best result.

3️⃣  An extended mirror‑descent from the current best point for the
    remaining budget (exact sub‑gradient of the max‑convolution term).

4️⃣  A tiny random‑local‑refinement to push the value a little further.

All steps respect the 1000 s CPU budget and return the best sequence
found.  The search works on the probability simplex (the sum of the
coefficients is forced to 1), because the objective is homogeneous.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection, …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Objective for a sequence normalised to sum=1:
        2·n·max(conv(seq,seq))
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)               # distance to the nearer edge
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_triangle(n, best_seq, best_score, deadline):
    """Symmetric triangular (tent) shape."""
    if time.time() >= deadline:
        return best_seq, best_score
    mid = n // 2
    left = np.arange(1, mid + 1)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    tri = np.concatenate([left, right])
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    sc = _objective_score(tri)
    if sc < best_score:
        best_score, best_seq = sc, tri.copy()
        print(f"[{time.time() - start_time:.2f}s] Triangle score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta(n, best_seq, best_score, deadline):
    """Beta(α,α) family, α∈(0.1,5). Includes mixtures with the uniform vector."""
    if time.time() >= deadline:
        return best_seq, best_score
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)            # avoid exact 0/1
    for α in np.linspace(0.2, 5.0, 70):
        if time.time() >= deadline:
            break
        # density ∝ x^{α-1} (1-x)^{α-1}
        seq = (x ** (α - 1)) * ((1 - x) ** (α - 1))
        seq += eps                              # avoid exact zeros
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Beta α={α:.3f} score={sc:.6f}")
        # also try a convex mix with the uniform vector
        for lam in (0.2, 0.5, 0.8):
            mix = lam * np.full(n, 1.0 / n) + (1.0 - lam) * seq
            mix /= mix.sum()
            sc2 = _objective_score(mix)
            if sc2 < best_score:
                best_score, best_seq = sc2, mix.copy()
                print(f"[{time.time() - start_time:.2f}s] BetaMix α={α:.3f} lam={lam:.1f} score={sc2:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws (global exploration)
# ----------------------------------------------------------------------
def _random_dirichlet_draw(n, rng):
    """Draw a point on the simplex using Gamma’s (concentration = c)."""
    logc = rng.uniform(-2.0, 0.7)          # c ∈ [10^{-2}, 5]
    c = 10 ** logc
    a = rng.gamma(shape=c, scale=1.0, size=n)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# 3️⃣  Exact sub‑gradient (mirror) descent on the simplex
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         max_iters: int,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         tag: str = ""):
    """
    Mirror‑descent using the exact sub‑gradient of max(conv(a,a)).
    Averages the sub‑gradients over the active set.
    Returns the best sequence (and its score) seen during this run.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 0.5

    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val

        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] {tag} it={it:,} obj={obj:.6f}")

        # --- sub‑gradient of the max‑convolution term ---
        tol = max_val * 1e-12
        active = np.where(conv >= max_val - tol)[0]
        m = active.size
        if m == 0:
            break

        grad = np.zeros_like(a)
        for i in active:
            lo = max(0, i - (n - 1))
            hi = min(n - 1, i)
            # slice j = lo..hi ; corresponding k = i - j
            j = np.arange(lo, hi + 1)
            k = i - j
            grad[lo:hi + 1] += a[k]
        grad = 2.0 * grad / m   # average over the active set

        # --- mirror (multiplicative) update ---
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total == 0.0:                 # safeguard against under‑flow
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Tiny random local refinement (stochastic mass transfers)
# ----------------------------------------------------------------------
def _local_refinement(a: np.ndarray,
                      max_steps: int,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float):
    """Perform very cheap random mass‑transfer moves; keep any improvement."""
    rng = np.random.default_rng()
    cur = a.copy()
    n = cur.shape[0]
    for step in range(max_steps):
        if time.time() >= deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0:
            continue
        # move a tiny fraction (up to 0.1 % of the source mass)
        delta = cur[i] * rng.uniform(1e-7, 1e-3)
        cur[i] -= delta
        cur[j] += delta
        # sum stays 1, no need to renormalise
        sc = _objective_score(cur)
        if sc < best_score:
            best_score, best_seq = sc, cur.copy()
            print(f"[{time.time() - start_time:.2f}s] Local refine step={step} new best={sc:.6f}")
    return best_seq, best_score

    if time.time() < phase_deadline:

    if time.time() < phase_deadline:

        sc0 = _objective_score(a0)
        if sc0 < best_score:
            best_score, best_seq = sc0, a0.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet fresh best score={sc0:.6f}")

        # If the sample looks promising, launch a brief mirror‑descent.
        # “Promising” → within 5 % of the current best.
        if sc0 <= best_score * 1.05:
            a_refined, sc_refined = _subgradient_descent(
                a0,
                max_iters=2000,
                deadline=phase_deadline,
                best_seq=best_seq,
                best_score=best_score,
                tag="ShortMD"
            )
            if sc_refined < best_score:
                best_score, best_seq = sc_refined, a_refined.copy()

    # --------------------------------------------------------------
    # 3️⃣  Long exact sub‑gradient mirror descent from the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting long mirror‑descent")
        best_seq, best_score = _subgradient_descent(
            best_seq,
            max_iters=200_000,                 # generous iteration budget; will stop early on timeout
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
            tag="LongMD"
        )
        print(f"[{time.time() - start_time:.2f}s] End of long descent, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Fine‑grained stochastic local refinement (uses any leftover time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_steps = int(5e5 * (remaining / budget_s))   # scale steps with remaining budget
        best_seq, best_score = _local_refinement(
            best_seq,
            max_steps=max_steps,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After local refinement score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – final best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
