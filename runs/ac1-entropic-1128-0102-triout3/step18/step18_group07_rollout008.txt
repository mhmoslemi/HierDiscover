def _max_pair_sum(mask):
    n = mask.shape[0]
    L = 1 << ((2*n-2).bit_length()) # next power of two >=2n-1
    M = np.fft.rfft(mask, n=L)
    conv = np.fft.irfft(M * M, n=L)
    conv = np.rint(conv[:2*n-1]).astype(np.int64) # round
    max_count = int(conv.max())
    return max_count, conv

"""
Binary‑mask optimiser for the max‑convolution objective.

The objective  2·n·max(conv(a,a))/(sum(a)²)  is invariant to scaling, so we
search among probability vectors a with sum = 1.  By fixing the number of
non‑zero entries (k) we can write a_i = 1/k on a binary mask and the
objective reduces to

    score = 2·n·max_cnt / k²

where max_cnt is the largest entry of the integer self‑convolution of the
mask.  The problem is therefore to minimise the peak of the anti‑diagonal
sums of a binary vector.  Random dense masks already give scores around
1.1‑1.2 (the theoretical lower bound is 1).  We start from many random masks,
pick the best, and then perform a simulated‑annealing hill‑climb that swaps
one ↔ zero while keeping k constant.  The convolution is evaluated with an
FFT at every step – fast enough to perform hundreds of thousands of swaps
within the 1000 s budget.

The final sequence is the best mask found, normalised to a probability
vector (value = 1/k on the chosen positions, 0 elsewhere).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (for printing)
# ----------------------------------------------------------------------
start_time = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_count(mask: np.ndarray, L: int) -> int:
    """
    Return the maximal entry of the integer self‑convolution of a binary mask.
    The mask contains 0/1 values; the convolution is computed via FFT.
    """
    # real FFT of the mask (zero‑padded to length L)
    F = np.fft.rfft(mask, n=L)
    # element‑wise product gives convolution in frequency domain
    conv = np.fft.irfft(F * F, n=L)[: 2 * mask.size - 1]
    # round to nearest integer to avoid tiny floating errors
    max_cnt = int(np.rint(conv).max())
    return max_cnt


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a list of floats (probability vector) – the best found within the
    time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the same length as the reference solution.
    # ------------------------------------------------------------------
    n = 4096
    L = _next_pow2(2 * n - 1)   # FFT length for convolutions

    # ------------------------------------------------------------------
    # Phase 0 – random dense binary masks (different densities)
    # ------------------------------------------------------------------
    densities = [0.5, 0.6, 0.7, 0.8, 0.9]   # proportion of ones
    rng = np.random.default_rng(seed)

    best_score = float("inf")
    best_mask = None
    best_k = None

    print(f"[{time.time() - start_time:.2f}s] start random mask sampling")

    for d in densities:
        k = max(1, int(d * n))
        trials_per_density = 200   # cheap enough to stay well within the budget
        for t in range(trials_per_density):
            if time.time() >= deadline:
                break
            mask = np.zeros(n, dtype=np.float64)
            idx = rng.choice(n, size=k, replace=False)
            mask[idx] = 1.0

            max_cnt = _max_convolution_count(mask, L)
            score = 2.0 * n * max_cnt / (k * k)

            if score < best_score:
                best_score = score
                best_mask = mask.copy()
                best_k = k
                print(f"[{time.time() - start_time:.2f}s] "
                      f"density {d:.2f} trial {t:03d} → score {score:.6f}")

    # If for some reason nothing was generated, fall back to the uniform vector
    if best_mask is None:
        best_mask = np.ones(n, dtype=np.float64)
        best_k = n
        best_score = 2.0
        print(f"[{time.time() - start_time:.2f}s] fallback to uniform")

    # ------------------------------------------------------------------
    # Phase 1 – simulated‑annealing hill‑climb on the best mask
    # ------------------------------------------------------------------
    mask = best_mask.copy()
    k = best_k
    denom = k * k

    # maintain mutable index lists for O(1) random swaps
    ones = np.where(mask == 1)[0]
    zeros = np.where(mask == 0)[0]

    cur_max_cnt = _max_convolution_count(mask, L)
    cur_score = 2.0 * n * cur_max_cnt / denom

    # Temperature schedule (annealing)
    T = 0.10
    decay = 0.9995
    min_T = 1e-6

    iter_cnt = 0
    print(f"[{time.time() - start_time:.2f}s] start hill‑climb (k={k})")

    while time.time() < deadline:
        iter_cnt += 1

        # pick a random 1‑position and a random 0‑position
        src_pos = rng.integers(len(ones))
        dst_pos = rng.integers(len(zeros))
        src = ones[src_pos]
        dst = zeros[dst_pos]

        # perform the swap
        mask[src] = 0.0
        mask[dst] = 1.0

        # evaluate new objective
        max_cnt = _max_convolution_count(mask, L)
        new_score = 2.0 * n * max_cnt / denom
        delta = new_score - cur_score

        # acceptance rule
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0:
                if rng.random() < np.exp(-delta / T):
                    accept = True

        if accept:
            # keep the swap
            cur_score = new_score
            cur_max_cnt = max_cnt
            # update index lists
            ones[src_pos] = dst
            zeros[dst_pos] = src

            if new_score < best_score:
                best_score = new_score
                best_mask = mask.copy()
                print(f"[{time.time() - start_time:.2f}s] iter {iter_cnt:07d} "
                      f"new best {best_score:.6f}")
        else:
            # revert the swap
            mask[src] = 1.0
            mask[dst] = 0.0

        # temperature decay
        T = max(T * decay, min_T)

    # ------------------------------------------------------------------
    # Normalise the best mask to a probability vector and return it
    # ------------------------------------------------------------------
    best_seq = best_mask / best_k          # each non‑zero entry = 1/k
    # a tiny normalisation step to guard against rounding errors
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")

    return [float(x) for x in best_seq]
