```python
def _cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the cross-correlation (a ⋆ w) of vectors a (len n) and w (len 2n-1):
    (a ⋆ w)[k] = sum_{j=0}^{n-1} a[j] * w[j + k], for k = 0..n-1.
    Implemented via FFT: conj(FFT(a)) * FFT(w).
    """
    n = a.shape[0]
    L = _next_pow2(n + len(w) - 1)
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(np.conj(Fa) * Fw, n=L)
    # Take first n entries
    return corr[:n]
```

```python
def _smooth_gradient_descent(a0, deadline, best_seq, best_score):

        conv = _conv_self(a)
```

"""
Propose a non‑negative step‑function of length n=4096 that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

The search consists of three stages:

1️⃣  Fast parametric sweeps (geometric, power, exponential, generalised
    exponential with exponent γ) to obtain a good starting point.
2️⃣  Random Dirichlet draws – cheap global exploration of the simplex.
3️⃣  Smoothed exponential‑gradient descent on the true objective.
    The max‑convolution term is replaced by a log‑sum‑exp
    (temperature α) which yields a smooth gradient.
    An exponentiated‑gradient (mirror‑descent) update on the log‑probabilities
    together with a decaying step size η_k∝1/√k is used.
    α is increased slowly so that the algorithm first sees a smooth landscape
    and finally concentrates on the true maximum.

All heavy work uses FFT‑based convolutions and respects a 1000 s wall‑time
budget. The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, convolution, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (evaluate_sequence() is homogeneous, fixing sum=1 suffices.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[k] = Σ_j a[j]·w[j+k] for k=0..n‑1,
    where w has length 2·n‑1. Implemented via FFT.
    """
    n = a.shape[0]
    L = _next_pow2(n + len(w) - 1)
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    # conj(Fa) * Fw yields the linear (non‑circular) cross‑correlation
    corr = np.fft.irfft(np.conj(Fa) * Fw, n=L)
    return corr[:n]  # keep only the needed n entries


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_general_exponential(n, best_seq, best_score, deadline):
    """a_i ∝ exp(-λ·i^γ) with a few γ values."""
    idx = np.arange(n, dtype=np.float64)
    gammas = [0.5, 1.0, 1.5, 2.0]
    lam_grid = np.logspace(-4, -0.2, 60)
    for γ in gammas:
        for lam in lam_grid:
            if time.time() >= deadline:
                break
            seq = np.exp(-lam * (idx ** γ))
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] GenExp γ={γ:.1f} λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (cheap global search)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        logc = rng.uniform(-2.0, 0.7)      # concentration 10⁻² … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Smoothed exponential‑gradient descent (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             best_seq: np.ndarray, best_score: float):
    """
    Minimise 2·n·max(conv(a,a)) using a smooth approximation:
        f_α(a) = 2·n·(1/α)·log Σ_i exp(α·conv_i(a)).

    An exponentiated‑gradient (mirror descent) update on log‑probabilities
    is performed.  The temperature α is increased slowly so that the method
    first sees a smooth landscape and later focuses on the true maximum.
    """
    a = np.copy(a0)                     # ensure positivity
    n = a.shape[0]
    eps = 1e-30
    log_a = np.log(a + eps)            # work in log‑space (mirror descent)
    it = 0
    eta0 = 0.5                         # initial step‑size (tuned empirically)
    alpha = 1.0                        # start temperature
    alpha_max = 80.0
    alpha_inc = 0.015                  # multiplicative increase per iteration

    while time.time() < deadline:
        it += 1
        # reconstruct a from log_a and re‑normalise
        a = np.exp(log_a)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
            log_a = np.log(a)
            total = 1.0
        a /= total
        # current objective (exact max version)
        conv = _conv_self(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val
        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD it={it:,} obj={obj:.6f} α={alpha:.2f}")
        # ----- smooth max (log‑sum‑exp) gradient -----
        # softmax weights w_i ∝ exp(α·conv_i)
        conv_shift = conv - max_val                     # improves numerical stability
        exp_arg = alpha * conv_shift
        # clip to avoid overflow in extreme cases
        exp_arg = np.clip(exp_arg, -700, 0)             # np.exp(-700) ≈ 5e-305
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient of the smoothed objective:
        # ∇f = 4·n·(a ⋆ w)   (cross‑correlation defined above)
        grad = 4.0 * n * _cross_corr(a, w)   # shape (n,)
        # mirror‑descent (exponentiated‑gradient) step on log_a
        eta = eta0 / np.sqrt(it)
        log_a -= eta * grad
        # optional: keep log_a finite
        if np.any(np.isnan(log_a)) or np.any(np.isinf(log_a)):
            # fall back to a safe state
            log_a = np.log(np.maximum(a, eps))
        # increase temperature slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * (1.0 + alpha_inc))

    return best_seq, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises the
    evaluation function.  The routine runs a quick parametric sweep,
    a large random Dirichlet exploration and a smoothed gradient descent.
    The best sequence (normalised to sum = 1) is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweeps (~180 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0   # leave plenty of time for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_general_exponential(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈200 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Smoothed exponential‑gradient descent (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed gradient descent")
        best_seq, best_score = _smooth_gradient_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End of descent, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
