"""
Propose a low‑score non‑negative step function.

The search proceeds in several stages while respecting a 1000 s budget.

1️⃣  Quick Dirichlet draws + a few classic windows give a decent starting point.
2️⃣  Alternating projection flattens the Fourier magnitude (≈ constant spectrum)
    while staying on the probability simplex.
3️⃣  Projected Adam optimisation that uses the *exact* sub‑gradient of the
    max‑convolution (instead of a smooth approximation).  Adaptive step‑size
    and several restarts explore the landscape.
4️⃣  Simulated‑annealing style hill‑climbing transfers mass between entries,
    accepting occasional uphill moves according to a linear cooling schedule.
5️⃣  A tiny evolutionary phase mixes the best solutions and injects random
    mutations to escape any remaining local minima.

All vectors are projected onto the probability simplex (non‑negative, sum = 1),
so the evaluated objective equals
        2·n·max(conv(a,a))
which is exactly the value returned by `evaluate_sequence` for a unit‑sum vector.
The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used by helper functions for progress messages)
# ----------------------------------------------------------------------
start_time = None

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (sum forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Classic deterministic windows (already normalised to the simplex)
# ----------------------------------------------------------------------
def _deterministic_shapes(n: int):
    """Yield a few classic windows (sine, Hann, triangular, Gaussian)."""
    # sine
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))

# Phase 2 – alternating projection (flatten Fourier magnitude)
# ----------------------------------------------------------------------
def _alternating_projection(a0: np.ndarray, max_iter: int, deadline: float):
    """
    Repeatedly replace the Fourier magnitude of a by a constant while
    keeping the vector non‑negative and on the simplex.
    """
    n = a0.shape[0]
    L = _next_pow2(2 * n - 1)          # length for FFT (≥ 2·n‑1)
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    for i in range(max_iter):
        if time.time() >= deadline:
            break

        # FFT of zero‑padded a
        F = np.fft.rfft(a, n=L)

        # Desired flat magnitude (same total energy for all frequencies)
        energy = np.dot(a, a)                 # sum a_i²
        flat_mag = np.sqrt(energy / L)

        # Replace magnitude, keep original phase
        F_flat = flat_mag * np.exp(1j * np.angle(F))

        # Inverse FFT and truncate
        a_full = np.fft.irfft(F_flat, n=L)
        a = a_full[:n]

        # Projection onto feasible set
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        score = _objective_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] alt‑proj iter {i:5d} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Adam optimiser using the *exact* sub‑gradient of the max‑conv.
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray):
    """
    Return (grad, max_idx, max_val) where grad is the sub‑gradient of
    max_k conv_k(a) w.r.t. a, max_idx is the index attaining the maximum,
    and max_val = conv[max_idx].
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_idx = int(np.argmax(conv))
    # one‑hot vector in convolution domain
    w = np.zeros(2 * n - 1, dtype=float)
    w[max_idx] = 1.0
    rev_corr = _convolve_fft(a[::-1], w)        # length 2·n‑1
    grad = 2.0 * rev_corr[n - 1:n - 1 + n]
    return grad, max_idx, float(conv[max_idx])


def _adam_subgrad_descent(a0: np.ndarray, score0: float, deadline: float):
    """
    Projected Adam optimiser that follows the *sub‑gradient* of the max‑convolution.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = score0

    # Adam hyper‑parameters
    lr = 0.07
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        grad, _, _ = _subgradient(a)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam‑subgrad new best {best_score:.6f}")

        # Accept if improvement, otherwise keep current (no Metropolis step here)
        if cand_score < cur_score:
            a, cur_score = cand, cand_score

        if np.linalg.norm(step) < 1e-13:
            break

    return best_a, best_score

# Phase 4 – simulated‑annealing mass‑transfer hill climbing
# ----------------------------------------------------------------------
def _annealed_hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = score0
    n = a.shape[0]
    rng = np.random.default_rng()

    # linear cooling: start with temperature 0.1, finish at ~1e‑5
    T_start = 0.08
    while time.time() < deadline:
        # cooling proportion (0 → 1)
        elapsed = time.time() - start_time
        fraction = max(0.0, (deadline - time.time()) / (deadline - start_time))
        T = T_start * fraction

        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer up to 15 % of source mass
        delta = a[src] * rng.random() * 0.15
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)          # restores exact sum‑1

        cand_score = _objective_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            a, cur_score = cand, cand_score

        if cand_score < best_score:
            best_score, best_a = cand_score, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] annealed hill‑climb new best {best_score:.6f}")

    return best_a, best_score

# Phase 5 – tiny evolutionary mixing (convex combinations + mutation)
# ----------------------------------------------------------------------
def _population_evolution(best_seq: np.ndarray, best_score: float, deadline: float):
    """
    Keep a small population of good candidates.
    Generate offspring by convex combination of two parents + Gaussian mutation.
    """
    n = best_seq.shape[0]
    rng = np.random.default_rng()
    pop = [best_seq.copy()]
    pop_scores = [best_score]

    # initialise a few random individuals
    while len(pop) < 8 and time.time() < deadline:
        a = rng.dirichlet(alpha=np.ones(n))
        sc = _objective_score(a)
        pop.append(a)
        pop_scores.append(sc)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time() - start_time:.2f}s] pop init new best {best_score:.6f}")

    # evolve until deadline
    while time.time() < deadline:
        i, j = rng.choice(len(pop), size=2, replace=False)
        lam = rng.random()
        child = lam * pop[i] + (1 - lam) * pop[j]

        # small Gaussian mutation
        child += rng.normal(scale=0.008, size=n)
        child = np.maximum(child, 0.0)
        child = _project_simplex(child)

        sc = _objective_score(child)

        # replace worst if better
        worst_idx = np.argmax(pop_scores)
        if sc < pop_scores[worst_idx]:
            pop[worst_idx] = child
            pop_scores[worst_idx] = sc
            print(f"[{time.time() - start_time:.2f}s] population replace worst with score {sc:.6f}")

        if sc < best_score:
            best_seq, best_score = child, sc
            print(f"[{time.time() - start_time:.2f}s] population new global best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a plain list of floats (probability vector).
    """
    np.random.seed(seed)          # legacy seeding for reproducibility
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size (4096 works well for the given budget)
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # If a global starter exists (height_sequence_1) incorporate it
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random Dirichlet draws + deterministic windows
    # --------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 30.0)
    rng = np.random.default_rng(seed)
    while time.time() < phase1_deadline:
        # concentration log‑uniform on [10⁻³, 5]
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")

    # deterministic windows
    for shp in _deterministic_shapes(n):
        sc = _objective_score(shp)
        if sc < best_score:
            best_seq, best_score = shp.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] deterministic shape score {sc:.6f}")
        if time.time() >= deadline - 120:
            break

    # --------------------------------------------------------------
    # Phase 2 – alternating projection (≈ 30 % of remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        proj_deadline = time.time() + 0.30 * remaining
        restarts = 6
        for r in range(restarts):
            if time.time() >= proj_deadline:
                break
            jitter = rng.normal(scale=0.02, size=n)
            a0 = best_seq + jitter
            a0 = _project_simplex(np.maximum(a0, 0.0))
            a_opt, sc_opt = _alternating_projection(a0, max_iter=2500, deadline=proj_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] alt‑proj restart {r} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam optimiser with exact sub‑gradient (≈ 35 % of remaining)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.35 * remaining
        restarts = 6
        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            jitter = rng.normal(scale=0.008, size=n)
            a0 = best_seq + jitter
            a0 = _project_simplex(np.maximum(a0, 0.0))
            sc0 = _objective_score(a0)
            a_opt, sc_opt = _adam_subgrad_descent(a0, sc0, adam_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] Adam‑subgrad restart {r} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting annealed hill‑climb")
        best_seq, best_score = _annealed_hill_climb(best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # Phase 5 – tiny evolutionary mixing (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting population evolution")
        best_seq, best_score = _population_evolution(best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
