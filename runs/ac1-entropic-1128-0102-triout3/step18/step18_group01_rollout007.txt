"""
Hybrid optimisation for the non‑negative step‑function problem.

The search proceeds in three stages:

1️⃣  A fast parametric sweep over several 2‑parameter families
    (geometric×power, pure Gaussian, and flexible Beta‑shapes)
    to obtain a decent starting point.

2️⃣  Projected momentum sub‑gradient descent (a heavy‑ball variant).
    At each iteration we compute the self‑convolution via FFT,
    locate the index of the current maximum and move opposite to the
    sub‑gradient  g_i = 2·a_{k−i} (projected onto the probability simplex).
    Momentum (μ≈0.9) accelerates descent and helps escape shallow plateaus.

3️⃣  A short simulated‑annealing hill‑climb that performs tiny mass‑transfer
    moves, occasionally accepting a worsening move with a temperature that
    decays exponentially.  This phase cleans up the last remaining bumps.

All phases respect the global 1000 s budget.  The final sequence is normalised
to sum = 1 and returned as a plain list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_score(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at the active index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        i = np.arange(k + 1)
        j = k - i
    else:
        i = np.arange(k - n + 1, n)
        j = k - i
    np.add.at(g, i, 2.0 * a[j])
    return g


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta‑type shape:
        a_i ∝ (i+1)^{α‑1} · (n‑i)^{β‑1}.
    """
    i = np.arange(n, dtype=np.float64)
    log_i = np.log(i + 1.0)
    # n‑i is always ≥1, safe to take log
    log_ni = np.log(np.maximum(n - i, 1e-300))
    log_shape = (alpha - 1.0) * log_i + (beta - 1.0) * log_ni
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape_sum = shape.sum()
    if shape_sum == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape

# Main optimisation routine
# ----------------------------------------------------------------------
