"""
Hybrid sparse‑seed + smooth‑max optimisation.

1️⃣  Generate a very diverse pool of seeds:
    – uniform, a global starter (if defined),
    – geometric×power, beta‑shapes (as before),
    – highly‑sparse Dirichlet draws (α∈{0.01,0.05,0.1,0.2,0.5}),
    – random “hard‑sparse’’ vectors with a fixed support size,
    – exact Sidon‑set constructions (prime‑quadratic) of size ~√n.
    All seeds are projected onto the simplex.

2️⃣  Keep the K best (cheap exact evaluation).

3️⃣  Continuation on the smooth‑max surrogate
    fβ(a)=2·n·β⁻¹·log Σexp(β·conv(a,a)),
    with a soft‑max re‑parameterisation a=softmax(x).
    A schedule β∈{1,2,4,8,16,32,64,128,256,512} is used;
    for each β an L‑BFGS‑B run (≤60 iterations) is performed
    on every seed in the pool.

4️⃣  Polishing with exponentiated‑gradient and projected sub‑gradient
    descent on the true objective.

5️⃣  A final random hill‑climb with tiny mass transfers.

All steps respect the 1000 s budget; the best sequence (normalised to sum = 1)
is returned as a plain Python list of floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation (same as before)
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: (i+1)^{α‑1}·(n‑i)^{β‑1}, normalised.
    Allows symmetric or skewed U‑/∧‑shapes.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

# Sparse‑seed helpers

def _random_sparse_vector(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Pick m distinct indices uniformly, assign Dirichlet weights on them."""
    idx = rng.choice(n, size=m, replace=False)
    # Dirichlet(1,…,1) gives a random point on the simplex of dimension m
    weights = rng.dirichlet(np.ones(m, dtype=np.float64))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    return a


def _sidon_set(p: int) -> np.ndarray:
    """
    Classical quadratic‑residue Sidon set of size p:
        s_i = (i * p + i**2) % (p**2) for i = 0,…,p‑1.
    All values lie in [0, p**2‑1] and have distinct pairwise sums.
    """
    mod = p * p
    s = [(i * p + i * i) % mod for i in range(p)]
    return np.array(sorted(set(s)), dtype=np.int64)


def _sidon_seed(n: int, p: int) -> np.ndarray:
    """Create a probability vector supported on a Sidon set of size p."""
    indices = _sidon_set(p)
    # keep only those indices < n (the construction guarantees this if p**2 <= n)
    indices = indices[indices < n]
    m = len(indices)
    a = np.zeros(n, dtype=np.float64)
    if m == 0:
        return a
    a[indices] = 1.0 / m
    return a

# Smooth‑max objective wrapped for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable object that evaluates the smooth‑max surrogate and its gradient.
    Variables are the logits x; the probability vector a=softmax(x).
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable soft‑max conversion."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        conv = _conv_self_fft(a, self.L_fft)

        # soft‑max over β·conv
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t a
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule for logits
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates (exponentiated‑gradient) on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score

            a0 = pool[idx]
            # initialise logits from current probability vector
            eps = 1e-12
            x0 = np.log(a0 + eps)

            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            # obtain the new probability vector
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 120:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 120, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 70, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Final random hill‑climb (tiny mass transfers).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move at most 0.4 % of the source mass
        eps = best_seq[src] * rng.random() * 4e-3
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
