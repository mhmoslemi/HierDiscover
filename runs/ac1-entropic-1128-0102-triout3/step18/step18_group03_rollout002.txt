def propose_candidate(seed=0, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 8192  (or maybe 12288 if time?)

    L_fft = _next_pow2(2*n-1)
    # baseline uniform
    best_seq = uniform = np.full(n, 1/n)
    best_score = _eval_fast(best_seq, L_fft)

    # generate seeds

    seeds = [best_seq]

    # Add based on existing global starter variable if present

    # param families
    # 1. r,p,q variations (including combos)
    for _ in range(200):
        r = rng.uniform(0.7, 0.999999)
        p = rng.uniform(-10, 10)
        q = rng.uniform(-10, 10)
        shape = _make_general_shape(n, r=r, p=p, q=q)
        # mixture with uniform
        lam = rng.random()
        shape = _mix_with_uniform(shape, lam)
        seeds.append(shape)

    # Dirichlet draws
    for alpha in [0.001,0.005,0.01,0.02,0.05,0.1]:
        for _ in range(12):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # random mixing of two shapes
    for _ in range(100):
        # shape A
        r1 = rng.uniform(0.5,0.9999)
        p1 = rng.uniform(-12,12)
        q1 = rng.uniform(-12,12)
        shape1 = _make_general_shape(n, r=r1, p=p1, q=q1)
        # shape B
        r2 = rng.uniform(0.5,0.9999)
        p2 = rng.uniform(-12,12)
        q2 = rng.uniform(-12,12)
        shape2 = _make_general_shape(n, r=r2, p=p2, q=q2)
        w1 = rng.random()
        w2 = 1-w1
        mix = w1*shape1 + w2*shape2
        mix = mix / mix.sum()
        seeds.append(mix)

    # Also include low rank shape: triangular or sinusoidal
    # Triangular: a_i ∝ min(i+1, n-i)
    triangular = np.minimum(np.arange(n)+1, np.arange(n)[::-1]+1).astype(float)
    triangular /= triangular.sum()
    seeds.append(triangular)

    # sinusoidal: a_i ∝ sin(pi * (i+0.5)/n)
    sinshape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinshape = np.maximum(sinshape, 0)
    sinshape /= sinshape.sum()
    seeds.append(sinshape)

    # Evaluate all seeds quickly
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 12
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    # set best
    best_score = pool_scores[0]
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best = {best_score:.6f}")

    # Use smoother Adam with increasing beta schedule
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]
    adam_lr = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for beta in beta_schedule:
        if time.time() >= deadline - 250:
            break
        max_iter = 5000
        for idx in range(len(pool)):
            a = pool[idx]
            m = np.zeros_like(a)
            v_vec = np.zeros_like(a)
            cur_score = pool_scores[idx]
            t = 0
            while time.time() < deadline - 150:
                t += 1
                conv = _conv_self_fft(a, L_fft)  # length 2n-1
                p_weights = _softmax_weights(conv, beta)
                g = _grad_softmax_new(a, p_weights, L_fft*2) # use bigger L for gradient? maybe same L works
                # update
                m = beta1 * m + (1.0 - beta1) * g
                v_vec = beta2 * v_vec + (1.0 - beta2) * (g * g)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v_vec / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)
                a_candidate = a - step
                a_candidate = _project_simplex(a_candidate)
                cand_score = _eval_fast(a_candidate, L_fft)
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    # if not improvement accept with prob
                    prob = np.exp(-(cand_score - cur_score) * 2.0 / (np.sqrt(t) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score
                if cand_score < pool_scores[idx]:
                    pool_scores[idx] = cand_score
                    pool[idx] = a_candidate.copy()
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start:.2f}s] Adam β={beta} it={t} score={cand_score:.6f}")
                if t >= max_iter:
                    break

    # After Adam, perform aggressive simulated annealing hill climbs.
    # Multi-move annealing: each iteration randomly chooses a set of k indices (e.g., 2-5) and redistribute mass among them.

    def random_move(a):
        # choose source index i, destination j, amount epsilon
        i = rng.integers(n)
        if a[i] < 1e-12:
            return a, False
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        eps = a[i] * rng.random() * 0.05  # up to 5% move
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        return a_new, True

    # Simulated annealing schedule: start temperature T0 ~ 0.01 or 0.02 relative to objective difference.
    # Actually we shall accept moves with probability exp(-Δ / T) where Δ = cand_score - cur_score.
    # We'll schedule T decreasing.
    T0 = 0.05
    T_min = 1e-5

"""
High‑precision search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The approach builds on the previous implementation but adds several
independent improvements:

1.  A much larger candidate pool (≈ 800 seeds) generated from
    – geometric‑power families,
    – mixtures of two such families,
    – low‑α Dirichlet draws,
    – simple triangular / sinusoidal shapes.
2.  A longer Adam optimisation on the smooth‑max surrogate with an
    extended β‑schedule (up to 2 048) and a larger problem size
    (n = 8192) – more degrees of freedom give a lower achievable bound.
3.  A guided simulated‑annealing hill‑climb that moves mass from the
    active support of the current maximal convolution entry to a
    complementary index.  The temperature is decreased linearly to
    concentrate the search near the end of the budget.
4.  A final diversification step that occasionally injects a tiny
    random perturbation of a pool entry.

All heavy lifting uses FFT‑based convolutions (O(n log n)) and a cheap
Euclidean projection onto the probability simplex.  The routine respects
the 1000 s budget: it continuously checks the deadline and stops cleanly.
The returned vector is normalised to sum = 1 and consists only of
non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2 * Σ_j a_j p_{i+j}.
    Implemented as a convolution of a_rev with p (zero‑padded to length L).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries (numerically stable)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact objective (used for cheap ranking)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_general_shape(n: int, r: float = 0.9, p: float = 0.0, q: float = 0.0) -> np.ndarray:
    """
    Shape a_i ∝ r**i · (i+1)**p · (n-i+1)**q, normalised to sum = 1.
    r∈(0,1) gives a geometric decay,
    p controls a power at the left boundary,
    q controls a power at the right boundary.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)                # prevent overflow
    shape = np.exp(log_shape)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

    seeds.append(sinshape)

    # ------------------------------------------------------------------
    # Keep the K best seeds after a cheap exact evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 12
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = pool_scores[0]
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam on the smooth‑max surrogate with a growing β‑schedule.
    # ------------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]
    adam_lr = 0.70
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for beta in beta_schedule:
        if time.time() >= deadline - 250:
            break
        max_iter_vec = 2500
        for idx in range(len(pool)):
            a = pool[idx]
            m = np.zeros_like(a)
            v_vec = np.zeros_like(a)
            cur_score = pool_scores[idx]
            t = 0
            while time.time() < deadline - 150:
                t += 1
                conv = _conv_self_fft(a, L_fft)
                p_weights = _softmax_weights(conv, beta)
                g = _grad_softmax_new(a, p_weights, L_grad)

                # Adam update (mirrored descent)
                m = beta1 * m + (1.0 - beta1) * g
                v_vec = beta2 * v_vec + (1.0 - beta2) * (g * g)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v_vec / (1.0 - beta2 ** t)
                step = adam_lr * m_hat / (np.sqrt(v_hat) + eps)

                a_candidate = a - step
                a_candidate = _project_simplex(a_candidate)

                cand_score = _eval_fast(a_candidate, L_fft)

                # Accept improvements outright; occasional uphill moves help escape plateaus.
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                else:
                    prob = np.exp(-(cand_score - cur_score) * 3.0 / (np.sqrt(t) + 1.0))
                    if rng.random() < prob:
                        a = a_candidate
                        cur_score = cand_score

                # Update global best and pool entry.
                if cand_score < pool_scores[idx]:
                    pool_scores[idx] = cand_score
                    pool[idx] = a_candidate.copy()
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time() - start:.2f}s] Adam β={beta} it={t:,} score={cand_score:.6f}")

                if t >= max_iter_vec:
                    break

    # ------------------------------------------------------------------
    # 3️⃣  Guided simulated‑annealing hill‑climb.
    #    Moves are biased to shift mass away from the active support of the
    #    current maximal convolution entry.
    # ------------------------------------------------------------------
    if time.time() < deadline - 60:
        # temperature schedule (linear decay)
        T_start = 0.04
        T_end = 5e-5
        while time.time() < deadline - 60:
            elapsed = time.time() - start
            frac = elapsed / (deadline - start - 60)
            T = T_start * (1.0 - frac) + T_end * frac

            # compute current active support
            conv_cur = _conv_self_fft(best_seq, L_fft)
            k_max = int(np.argmax(conv_cur))
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)

            # choose source inside the support (weighted by current mass)
            support = np.arange(i_start, i_end + 1, dtype=np.int64)
            if support.size == 0:
                continue
            src_weights = best_seq[support]
            wsum = src_weights.sum()
            if wsum == 0.0:
                src = rng.choice(support)
            else:
                src = rng.choice(support, p=src_weights / wsum)

            # choose destination outside the support
            if i_start > 0:
                left = np.arange(0, i_start, dtype=np.int64)
            else:
                left = np.empty(0, dtype=np.int64)
            if i_end < n - 1:
                right = np.arange(i_end + 1, n, dtype=np.int64)
            else:
                right = np.empty(0, dtype=np.int64)
            outside = np.concatenate((left, right))
            if outside.size == 0:
                continue
            dst = rng.choice(outside)

            # amount to move (up to 5 % of source mass)
            eps_move = best_seq[src] * rng.random() * 0.05
            if eps_move <= 0.0:
                continue
            a_new = best_seq.copy()
            a_new[src] -= eps_move
            a_new[dst] += eps_move
            # numeric safety
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()

            new_score = _eval_fast(a_new, L_fft)
            delta = new_score - best_score
            if delta < 0 or rng.random() < np.exp(-delta / T):
                best_seq = a_new
                best_score = new_score
                print(f"[{time.time() - start:.2f}s] SA move score={new_score:.6f}")

            # occasional diversification from pool
            if rng.random() < 0.001:
                idx = rng.integers(len(pool))
                cand = pool[idx].copy()
                eps = rng.random(n) * cand * 1e-4
                cand = np.maximum(cand + eps, 0.0)
                cand /= cand.sum()
                cand_score = _eval_fast(cand, L_fft)
                if cand_score < best_score:
                    best_seq = cand
                    best_score = cand_score
                    print(f"[{time.time() - start:.2f}s] diversification score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
