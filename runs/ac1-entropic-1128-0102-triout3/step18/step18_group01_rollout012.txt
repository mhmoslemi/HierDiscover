"""
Propose a non‑negative step function (probability vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The search runs for at most the given time budget (default 1000 s).  It
consists of five stages:

1️⃣  Very fast parametric sweep over two‑parameter families
    (geometric·power and pure Gaussian) to obtain a decent starter.

2️⃣  Classical projected sub‑gradient descent on the exact L∞ objective
    (as in the reference implementation).

3️⃣  Exponentiated‑gradient (mirror descent) on the simplex.

4️⃣  Smoothed‑gradient descent: a temperature‑controlled “soft‑max”
    replaces the hard max in the objective, giving a smoother direction.
    The updates are multiplicative and stay on the probability simplex.

5️⃣  Simulated‑annealing hill‑climb with occasional large mass transfers
    and swaps, accepting worsening moves with a decaying temperature.
    This phase exploits the remaining time budget.

All phases keep track of the best sequence seen so far (according to the
exact objective).  The final vector is normalised to sum = 1 and returned
as a plain Python list of floats.  The implementation respects the
security checks of ``evaluate_sequence`` (non‑negative entries,
sum > 0.01 and values ≤ 1000).  The code is completely self‑contained
and uses only NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """Return a shape ∝ (i+1)^p·exp(-lam·i) (normalised to sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    log_idx = np.log(i + 1.0)
    log_shape = -lam * i + p * log_idx
    log_shape -= np.max(log_shape)           # for numerical stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """Return a shape ∝ exp(-c·i²) (normalised)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Phase‑2: projected sub‑gradient descent (hard max)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at the index k where the maximum is attained."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    np.add.at(g, i, 2.0 * a[j])
    return g


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact L∞ objective."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    step0 = 0.45
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        step = step0 / (it ** 0.55)
        cand = a - step * g
        cand = _project_simplex(cand)
        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} score={cand_score:.6f}")
        else:
            if it % 1200 == 0:
                eps = rng.random(a.shape) * a * 8e-5
                a = np.maximum(a + eps, 0.0)
                a = _project_simplex(a)

        if time.time() - last_improve > 30.0 and it % 2000 == 0:
            eps = rng.random(best.shape) * best * 2e-4
            a = np.maximum(best + eps, 0.0)
            a = _project_simplex(a)
            last_improve = time.time()

    return best, best_score

# Phase‑3: exponentiated‑gradient (mirror descent) on the exact max
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates a ← a·exp(−η·g) with annealed η."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    it = 0
    eta0 = 0.8

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        eta = eta0 / (1.0 + 0.001 * it)
        a = a * np.exp(-eta * g)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / total

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] expgrad it={it:,} score={score:.6f}")

        if it % 2500 == 0:
            eps = rng.random(a.shape) * a * 5e-5
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score

# Phase‑4: smoothed‑gradient descent (soft‑max of the convolution)
# ----------------------------------------------------------------------


def _smooth_grad(a: np.ndarray, L_fft: int, alpha: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (conv, grad) where conv = a*a (full) and grad is the gradient of the
    soft‑max approximation of max(conv) with temperature 1/alpha.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L_fft)

    # hard convolution (needed for the soft‑max weights as well)
    conv = np.fft.irfft(a_fft * a_fft, n=L_fft)[: 2 * n - 1]

    # soft‑max weights (normalised)
    max_c = conv.max()
    exps = np.exp(alpha * (conv - max_c))
    w = exps / exps.sum()                     # length 2n‑1, sums to 1

    # gradient = 2·∑_j a_j w_{i+j}
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=L_fft)
    grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L_fft)[: 2 * n - 1]
    grad = 2.0 * grad_full[2 * n - 2 - np.arange(n, dtype=int)]

    return conv, grad


def _smooth_gradient_descent(a0: np.ndarray,
                             L_fft: int,
                             deadline: float,
                             start_time: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mirror‑descent using a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    it = 0

    # schedule for the temperature (alpha)
    alpha_start = 2.0
    alpha_end = 5000.0
    eta0 = 0.6

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        alpha = alpha_start + frac * (alpha_end - alpha_start)

        conv, grad = _smooth_grad(a, L_fft, alpha)

        # multiplicative mirror update
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / s

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} score={score:.6f}")

        # occasional tiny jitter to avoid tiny plateaus
        if it % 2000 == 0:
            eps = rng.random(a.shape) * a * 1e-4
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best, best_score

def _simulated_annealing(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Annealed random walk that occasionally accepts worsening moves."""
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    T0 = 0.02          # initial temperature
    T_end = 1e-4       # final temperature

    while time.time() < deadline:
        # Choose a move
        move_type = rng.random()
        cand = a.copy()

        if move_type < 0.55:                           # modest transfer
            src = rng.integers(n)
            if cand[src] <= 0.0:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            eps = cand[src] * rng.random() * 5e-3       # up to 0.5 %
            cand[src] -= eps
            cand[dst] += eps

        elif move_type < 0.85:                         # larger transfer
            src = rng.integers(n)
            if cand[src] <= 0.0:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            eps = cand[src] * rng.random() * 2e-2       # up to 2 %
            cand[src] -= eps
            cand[dst] += eps

        else:                                          # swap two entries
            i, j = rng.integers(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # Normalise (tiny numerical drift)
        cand = cand / cand.sum()

        cand_score = _eval_fast(cand, L_fft)

        # Acceptance test
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] SA new best {cand_score:.6f}")
        else:
            # temperature schedule (linear decay)
            elapsed = time.time() - start_time
            total = deadline - start_time
            phi = min(1.0, elapsed / total)
            T = T0 * (1 - phi) + T_end * phi
            # Metropolis criterion
            if np.log(rng.random()) < -(cand_score - best_score) / T:
                a = cand

        # occasional tiny jitter directly on the best solution
        if rng.random() < 0.001:
            idx = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[idx] = max(cand[idx] + jitter, 0.0)
            cand = cand / cand.sum()
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                a = cand
                print(f"[{time.time() - start_time:.2f}s] SA jitter {cand_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain list of floats (probability vector).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # a little safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 provides a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – if present.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur_score = _eval_fast(starter, L_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Parametric sweep (≈60 s)
    # ------------------------------------------------------------------
    sweep_end = start_time + 60.0
    i_arr = np.arange(n, dtype=np.float64)
    log_idx = np.log(i_arr + 1.0)

    # geometric × power family
    lam_grid = np.linspace(1e-05, 3e-03, 400)    # finer grid
    p_grid = np.linspace(-9.0, 6.0, 151)         # step ≈0.1

    for lam in lam_grid:
        if time.time() >= sweep_end:
            break
        base = -lam * i_arr
        for p in p_grid:
            log_shape = base + p * log_idx
            log_shape -= np.max(log_shape)
            cand = np.exp(log_shape)
            cand /= cand.sum()
            sc = _eval_fast(cand, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep λ={lam:.6e} p={p:.2f} score={sc:.6f}")

    # pure Gaussian family
    c_grid = np.logspace(-8.5, -5.5, 55)
    for c in c_grid:
        if time.time() >= sweep_end:
            break
        log_shape = -c * (i_arr * i_arr)
        log_shape -= np.max(log_shape)
        cand = np.exp(log_shape)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] gaussian c={c:.2e} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Projected sub‑gradient descent (≈30 % of budget)
    # ------------------------------------------------------------------
    sub_end = start_time + 0.30 * budget_s
    if time.time() < sub_end:
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_end, start_time, rng)

    # ------------------------------------------------------------------
    # 3️⃣ Exponentiated‑gradient phase (≈20 % of budget)
    # ------------------------------------------------------------------
    exp_end = start_time + 0.50 * budget_s
    if time.time() < exp_end:
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_end, start_time, rng)

    # ------------------------------------------------------------------
    # 4️⃣ Smoothed‑gradient descent (≈15 % of budget)
    # ------------------------------------------------------------------
    smooth_end = start_time + 0.65 * budget_s
    if time.time() < smooth_end:
        best_seq, best_score = _smooth_gradient_descent(best_seq, L_fft,
                                                        smooth_end, start_time, rng)

    # ------------------------------------------------------------------
    # 5️⃣ Simulated‑annealing hill‑climb (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(best_seq, L_fft,
                                                    deadline, start_time, rng)

    # ------------------------------------------------------------------
    # Final normalisation and safety checks
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Report final bound using the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    try:
        # `evaluate_sequence` is assumed to be already imported in the runner.
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
