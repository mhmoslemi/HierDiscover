def _hill_climb_opt(a, score, deadline):
    a_curr = a.copy()
    cur_score = score
    best_seq = a.copy()
    best_score = score
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a_curr[i] <= 1e-12:
            continue
        delta = a_curr[i] * rng.random() * 0.2
        cand = a_curr.copy()
        cand[i] -= delta
        cand[j] += delta
        # project onto simplex: but after transfer sum remains same and positivity holds
        # though cand[i] may become negative; we enforce max(0)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue

"""
Hybrid optimisation for the step‑function attack.

* Represent the candidate a directly on the probability simplex.
* Use a smooth approximation of the max‑convolution (log‑sum‑exp)
  and minimise it with L‑BFGS‑B (continuous optimisation on the
  simplex via a soft‑max parametrisation).
* A continuation on the smoothness parameter β gradually focuses the
  optimiser on the true ℓ∞‑objective.
* Several random restarts and a final short hill‑climbing phase (mass
  transfers) polish the best solution found.
* All operations use FFT‑based convolutions (O(n log n)) so many
  optimiser steps fit easily into the 1000‑second budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global timing / parameters (filled in propose_candidate)
# ----------------------------------------------------------------------
start_time = None          # start of the whole optimisation
_n = None                  # problem size (length of the sequence)
_L = None                  # FFT length (next power of two ≥ 2·n‑1)
_beta = 1.0                # current temperature for the smooth‑max

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length _L."""
    A = np.fft.rfft(a, n=_L)
    # |A|² is the power spectrum; the inverse transform yields the
    # (linear) autocorrelation.
    conv = np.fft.irfft(np.abs(A) ** 2, n=_L)[: 2 * _n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    conv = _conv_self(a)
    return float(2.0 * _n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_weights(conv: np.ndarray, beta: float):
    """Return smooth‑max value and the associated softmax weights."""
    m = conv.max()
    # shift by the maximum for numerical stability
    diff = conv - m
    exps = np.exp(beta * diff)
    sum_exps = exps.sum()
    smooth = m + np.log(sum_exps) / beta
    w = exps / sum_exps
    return smooth, w


def _smooth_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max w.r.t. a.
    Implements  ∇_a  Σ_k w_k (a⋆a)_k = 2·(rev(a) ⋆ w) restricted to the centre.
    """
    rev_a = a[::-1]
    rev_fft = np.fft.rfft(rev_a, n=_L)
    w_fft = np.fft.rfft(w, n=_L)
    rev_conv = np.fft.irfft(rev_fft * w_fft, n=_L)[: 3 * _n - 2]
    grad = 2.0 * rev_conv[_n - 1 : _n - 1 + _n]
    return grad


# ----------------------------------------------------------------------
# Objective and gradient w.r.t. the soft‑max parameters v
# (a_i = exp(v_i) / Σ exp(v) ).
# ----------------------------------------------------------------------
def _objective_and_grad(v: np.ndarray):
    """
    Returns (smooth_max, grad_v) for the current temperature _beta.
    The simplex constraint Σ a_i = 1 is enforced by the soft‑max.
    """
    # stable soft‑max
    v_shift = v - np.max(v)
    a = np.exp(v_shift)
    a = a / a.sum()

    # convolution via FFT
    A = np.fft.rfft(a, n=_L)
    conv = np.fft.irfft(np.abs(A) ** 2, n=_L)[: 2 * _n - 1]

    # smooth max and the corresponding weights
    smooth, w = _smooth_max_and_weights(conv, _beta)

    # gradient w.r.t. a
    grad_a = _smooth_gradient(a, w)

    # chain rule through the soft‑max
    dot = np.dot(a, grad_a)               # ⟨a,∇_a f⟩
    grad_v = a * (grad_a - dot)           # ∂f/∂v_j = a_j (∂f/∂a_j - ⟨a,∇f⟩)

    return smooth, grad_v


# ----------------------------------------------------------------------
# Simple hill‑climbing (mass‑transfer) for final polishing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float, deadline: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = score0
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        # move up to 20 % of the source mass
        delta = a[src] * rng.random() * 0.20
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # re‑normalise to stay on the simplex (floating‑point errors)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        cand_score = _objective_score(cand)
        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a plain list of floats (probability vector).

    The optimiser combines:
    * several random Dirichlet starts,
    * a continuation L‑BFGS‑B on a smooth max‑approximation,
    * a final mass‑transfer hill‑climb.
    """
    global start_time, _n, _L, _beta

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    _n = 4096
    _L = _next_pow2(2 * _n - 1)

    # uniform simplex start (baseline)
    uniform = np.full(_n, 1.0 / _n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – quick Dirichlet sampling (helps seed the optimiser)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(800):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in [10⁻³, 10⁰]
        conc = 10 ** rng.uniform(-3.0, 0.0)
        a = rng.gamma(shape=conc, scale=1.0, size=_n)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet seed score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS‑B on smooth max (continuation on β)
    # ------------------------------------------------------------------
    from scipy.optimize import Bounds

    beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 8000.0]
    restarts = 12                     # more restarts → higher chance of escaping poor basins
    maxfun_per_call = 300            # moderate budget per L‑BFGS sub‑problem

    # initialise the first parameter vector (log‑softmax of the best_seq)
    v0 = np.log(np.maximum(best_seq, 1e-12))

    for r in range(restarts):
        if time.time() >= deadline:
            break

        # jitter the start point a little to encourage diversity
        jitter = rng.normal(scale=0.05, size=_n)
        v_start = v0 + jitter

        for beta in beta_schedule:
            if time.time() >= deadline:
                break
            _beta = beta

            # minimise the smooth max; the optimizer returns both value and gradient
            res = minimize(
                fun=_objective_and_grad,
                x0=v_start,
                method='L-BFGS-B',
                jac=True,
                bounds=Bounds(-np.inf, np.inf),
                options={'maxfun': maxfun_per_call, 'ftol': 1e-12, 'gtol': 1e-9, 'disp': False}
            )
            v_start = res.x

            # convert back to the simplex vector
            a = np.exp(v_start - np.max(v_start))   # shift for stability
            a /= a.sum()

            cur_score = _objective_score(a)
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] L‑BFGS restart {r} β={beta:.0f} new best {best_score:.6f}")

        # keep the final v from this restart as a seed for the next one
        v0 = v_start

    # ------------------------------------------------------------------
    # Phase 2 – final hill‑climb (few seconds left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise once more (just in case) and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
