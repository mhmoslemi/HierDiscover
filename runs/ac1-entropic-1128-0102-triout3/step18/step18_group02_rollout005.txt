"""
Improved three‑phase search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm differs from the reference implementation by

* using a richer set of parametric seeds (geometric·power, beta, sine,
  wider r/p ranges, more mixing with the uniform vector);
* a longer smooth‑max L‑BFGS continuation (β up to 1024, 200 iters per β);
* a dedicated projected gradient‑descent phase on the *true* max‑conv
  objective (instead of only exponentiated‑gradient);
* a perturb‑and‑re‑optimise loop that creates fresh seeds around the best
  solution and runs a short L‑BFGS on them;
* a final simulated‑annealing hill‑climb with an adaptive temperature.

All phases respect the 1000 s time budget and the best sequence found is
returned as a plain Python list (already normalised to sum = 1).
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) for a probability vector a."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑conv objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type shape, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_sine_shape(n: int) -> np.ndarray:
    """Sin² shape (positive, smooth, symmetric)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * (i + 0.5) / (2.0 * n)) ** 2
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable object that evaluates the smooth‑max surrogate

        fβ(a) = 2·n·β⁻¹·log Σ exp(β·conv_k(a))

    together with its gradient w.r.t. the logits x (a = softmax(x)).
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable soft‑max conversion."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        a = self._softmax_logits(x)

        conv = _conv_self_fft(a, self.L_fft)

        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps                       # soft‑max weights

        # gradient w.r.t. a of the smooth‑max term
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule to go from a‑gradient to logits‑gradient
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation
    function. Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10  # small safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good compromise between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Baseline – uniform distribution (objective = 2)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (named ``height_sequence_1`` if present)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Generate a rich pool of seed vectors.
    # --------------------------------------------------------------
    seeds = [best_seq]   # already contains uniform / starter

    # Geometric × power families – wider sweep
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9999), 9)
    p_vals = np.linspace(-6.0, 6.0, 21)
    lam_vals = [0.0, 0.2, 0.5, 0.8, 1.0]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Discrete Beta families – additional extremes
    alpha_vals = [0.3, 0.5, 0.8, 1.0, 2.0, 3.0]
    beta_vals = [0.3, 0.5, 0.8, 1.0, 2.0, 3.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # Sine² shape (smooth symmetric bump)
    sine_shape = _make_sine_shape(n)
    for lam in [0.0, 0.3, 0.6, 0.9]:
        seeds.append(_mix_with_uniform(sine_shape, lam))

    # Random Dirichlet draws at several concentrations
    for conc in [0.1, 0.2, 0.5, 1.0]:
        for _ in range(8):
            seeds.append(rng.dirichlet(np.full(n, conc)))

    # --------------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # --------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 12
    best_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in best_idx]
    pool_scores = [scores[i] for i in best_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Continuation with L‑BFGS on the smooth‑max surrogate.
    # --------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    max_iter_per_beta = 200  # more inner iterations than the reference

    for beta in beta_schedule:
        # Reserve ~250 s for the polishing phases that follow
        if time.time() >= deadline - 250:
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for idx in range(len(pool)):
            if time.time() >= deadline - 250:
                break
            a0 = pool[idx]
            x0 = np.log(a0 + 1e-12)          # logits initialisation
            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            a_new = np.exp(res.x - np.max(res.x))
            a_new /= a_new.sum()

            cand_score = _eval_fast(a_new, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            pool[idx] = a_new
            pool_scores[idx] = cand_score

    # --------------------------------------------------------------
    # 3️⃣  Projected gradient descent on the true max‑conv objective.
    # --------------------------------------------------------------
    def _gd_phase(a_start, cur_best_seq, cur_best_score, deadline_gd):
        """Simple projected GD with back‑tracking on the true L∞ objective."""
        a = a_start.copy()
        it = 0
        step0 = 0.7
        while time.time() < deadline_gd:
            it += 1
            conv = _conv_self_fft(a, L_fft)
            k = int(np.argmax(conv))
            g = _subgrad_max_conv(a, k)
            step = step0 / np.sqrt(it)

            improved = False
            # back‑track up to ten times
            for _ in range(10):
                a_cand = a - step * g
                a_cand = _project_simplex(a_cand)
                cand_score = _eval_fast(a_cand, L_fft)
                if cand_score < cur_best_score:
                    cur_best_score = cand_score
                    cur_best_seq = a_cand.copy()
                    a = a_cand
                    improved = True
                    print(f"[{time.time() - start:.2f}s] GD it={it:,} step={step:.5e} score={cand_score:.6f}")
                    break
                step *= 0.5
            if not improved:
                # If back‑tracking fails repeatedly we shrink the overall step.
                if step < 1e-12:
                    break
        return cur_best_seq, cur_best_score

    if time.time() < deadline - 120:
        best_seq, best_score = _gd_phase(best_seq, best_seq, best_score, deadline - 120)

    # --------------------------------------------------------------
    # 4️⃣  Perturb‑and‑re‑optimise (few extra L‑BFGS runs from neighbours).
    # --------------------------------------------------------------
    if time.time() < deadline - 80:
        for pert in range(5):
            if time.time() >= deadline - 80:
                break
            lam = rng.uniform(0.01, 0.15)
            noisy = rng.dirichlet(np.full(n, 0.5))
            a0 = (1 - lam) * best_seq + lam * noisy
            a0 /= a0.sum()
            x0 = np.log(a0 + 1e-12)
            # a short L‑BFGS run with a relatively high β
            obj = _SmoothMaxObj(n, L_fft, L_grad, beta=512)
            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': 120, 'disp': False})
            a_new = np.exp(res.x - np.max(res.x))
            a_new /= a_new.sum()
            cand_score = _eval_fast(a_new, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start:.2f}s] perturb‑L‑BFGS #{pert} score={cand_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing hill‑climb.
    # --------------------------------------------------------------
    def _annealing_phase(a_cur, cur_best_seq, cur_best_score, deadline_ann):
        """Tiny mass‑transfer moves with a temperature schedule."""
        cur = a_cur.copy()
        cur_score = cur_best_score
        while time.time() < deadline_ann:
            src = rng.integers(n)
            if cur[src] < 1e-14:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            # move a random fraction (up to 0.4 %) of the source mass
            eps = cur[src] * rng.random() * 4e-3
            prop = cur.copy()
            prop[src] -= eps
            prop[dst] += eps

            new_score = _eval_fast(prop, L_fft)
            # temperature decays linearly with remaining time
            remaining = deadline_ann - time.time()
            T = 1e-3 * max(remaining, 1.0) / (deadline_ann - start)
            accept = new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T)
            if accept:
                cur = prop
                cur_score = new_score
                if new_score < cur_best_score:
                    cur_best_score = new_score
                    cur_best_seq = prop.copy()
                    print(f"[{time.time() - start:.2f}s] anneal new best={new_score:.6f}")
        return cur_best_seq, cur_best_score

    if time.time() < deadline - 30:
        best_seq, best_score = _annealing_phase(best_seq, best_seq, best_score, deadline - 30)

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")

    # Optional sanity check using the provided evaluate_sequence (if available)
    try:
        final_score = evaluate_sequence([float(v) for v in best_seq])
        print(f"evaluate_sequence reported score: {final_score}")
    except Exception:
        pass

    return [float(v) for v in best_seq]
