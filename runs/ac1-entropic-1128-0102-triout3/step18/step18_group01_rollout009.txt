Implementation:

```
def _annealing(a0, L_fft, deadline, start_time, rng):
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        # propose:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0:
            continue
        eps = a[src] * rng.random() * 0.01  # up to 1% transfer
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_fft)
        # temperature schedule:
        t = max(1e-4, 0.1 * (deadline - time.time()) / (deadline - start_time))
        if cand_score < best_score or rng.random() < np.exp(-(cand_score - best_score) / t):
            # accept
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it} score={cand_score:.6f}")
    return best, best_score
```

assistantfinal**Solution Overview**

The evaluation function  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is invariant under scaling, therefore we may restrict the search to **probability vectors**
(`∑a_i = 1 , a_i ≥ 0`).  
For a fixed length `n` the problem is:

* minimise the maximal entry of the self‑convolution `a*a`.*

The algorithm below consists of four independent phases that together spend the whole
budget (≈ 1000 s).  

1. **Wide parametric sweep** – several families of analytically tractable shapes
   (exponential, power‑exponential, symmetric “double‑exponential”, Beta‑like,
   stretched‑exponential) are sampled on a dense grid.  The best candidate from
   this sweep becomes the seed for the subsequent optimisation.
2. **Adam optimiser on the exact `ℓ∞`‑objective** – we treat the worst convolution
   entry as a (non‑smooth) objective and apply Adam (first‑order adaptive moments)
   together with a projection onto the probability simplex.  Adam converges much
   faster than a pure sub‑gradient method, allowing many more iterations.
3. **Simulated‑annealing hill‑climb** – random “mass‑transfer’’ moves are accepted
   with a Metropolis rule that slowly cools, helping the search to escape shallow
   local minima that Adam can get stuck in.
4. **Fine‑grained deterministic hill‑climb** – tiny transfers of ≤ 0.02 % of the
   source mass are tried exhaustively; any improvement is kept.

All phases are time‑controlled, use only NumPy (no external files) and respect the
security checks of `evaluate_sequence`.  The final sequence is normalised to sum = 1
and returned as a plain Python list.

```python
"""Improved search for a step‑function minimising
2·n·max(conv(a,a))/(sum(a)**2).

The algorithm:
0️⃣ choose problem size (default n=8192, adapted to a global starter);
1️⃣ dense parametric sweep over several analytic families;
2️⃣ Adam optimiser on the exact ℓ∞‑objective with simplex projection;
3️⃣ simulated‑annealing mass‑transfer moves (temperature slowly decays);
4️⃣ final fine‑grained hill‑climb.
All phases respect the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (objective) – works for arbitrary sum
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p · exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * i + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i²)."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -c * (i * i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_double_exp(n: int, lam: float) -> np.ndarray:
    """Symmetric double‑exponential: exp(-lam·i) + exp(-lam·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    log_shape = np.logaddexp(-lam * i, -lam * (n - 1 - i))
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_beta_like(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^p · (n‑i)^q   (Beta‑type shape, symmetric when p=q)."""
    i = np.arange(n, dtype=np.float64)
    # avoid log(0) – the ends are never exactly zero for finite p,q
    log_shape = p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_stretched_exp(n: int, lam: float, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i**alpha).   0<alpha≤2."""
    i = np.arange(n, dtype=np.float64)
    log_shape = -lam * np.power(i, alpha)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Sub‑gradient of the worst convolution entry
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the index where the maximum is attained.
    For each i involved in the max sum (i+j=k) we get 2·a_j.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    np.add.at(g, i, 2.0 * a[j])
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the ℓ∞‑objective (phase 2)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact max‑convolution objective.
    Uses projection onto the probability simplex after each step.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(conv.argmax())
        g = _subgrad_max_conv(a, k)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (mild decay)
        lr = lr0 / (1.0 + 1e-4 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(cand)

        score = _eval_fast(cand, L_fft)
        if score < best_score:
            best_score = score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={score:.6f}")
        else:
            # occasional jitter to avoid stagnation
            if it % 800 == 0:
                jitter = rng.random(a.shape) * a * 4e-5
                a = np.maximum(a + jitter, 0.0)
                a = _project_simplex(a)

        # simple restart if we have been idle for a while
        if it % 5000 == 0 and time.time() < deadline - 10.0:
            jitter = rng.random(best.shape) * best * 2e-4
            a = np.maximum(best + jitter, 0.0)
            a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing mass transfer (phase 3)
# ----------------------------------------------------------------------


def _annealing(a0: np.ndarray,
               L_fft: int,
               deadline: float,
               start_time: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style annealing: random mass transfers are accepted
    either because they improve the score or with probability exp(-Δ/T).
    Temperature decays linearly with the remaining time.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]
    it = 0

    while time.time() < deadline:
        it += 1
        # propose a transfer
        src = rng.integers(n)
        if a[src] == 0.0:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 0.004   # up to 0.4 % transfer
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()

        cand_score = _eval_fast(cand, L_fft)

        # linear cooling: T in (1e‑4 … 0.15)
        remaining = deadline - time.time()
        total = deadline - start_time
        T = max(1e-4, 0.15 * remaining / total)

        accept = cand_score < best_score or rng.random() < math.exp(-(cand_score - best_score) / T)
        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal it={it:,} score={cand_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Final fine‑grained hill‑climb (phase 4)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny deterministic mass‑transfer moves (≤ 0.02 % of source mass)
    accepted only if they improve the score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-4   # ≤ 0.02 %
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cand_score:.6f}")

    return best, best_score

    anneal_end = start_time + 0.85 * budget_s
    if time.time() < anneal_end:
        best_seq, best_score = _annealing(best_seq, L_fft,
                                          anneal_end, start_time, rng)

    # --------------------------------------------------------------
    # 4️⃣ Final fine‑grained hill‑climb – use whatever time is left.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _hill_climb(best_seq, L_fft,
                                           deadline, start_time, rng)

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Optional final report using the provided evaluate_sequence
    try:
        from __main__ import evaluate_sequence  # may already be imported
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
```

* A dedicated simulated‑annealing phase lets the search cross narrow “energy barriers’’ that deterministic descent cannot.
* The final deterministic hill‑climb fine‑tunes the distribution to the last possible decimal places.
