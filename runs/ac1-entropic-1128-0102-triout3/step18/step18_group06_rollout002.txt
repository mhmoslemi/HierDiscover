                break
        # weights
        exp_vals = np.exp((conv - max_conv) / eps)
        w = exp_vals / exp_vals.sum()
        # gradient
        wf = np.fft.rfft(np.pad(w, (0, L - (2*n - 1)), mode='constant'), n=L)
        grad = 2.0 * np.fft.irfft(np.conj(wf) * Af, n=L)[:n]
        # step size
        eta = step_factor / np.sqrt(iter_count)

        # schedule eps reduction
        if iter_count % 5000 == 0:
            eps = max(eps * 0.5, eps_min)

"""
Propose a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three phases:

1️⃣  Quick random Dirichlet sampling to obtain a good starting point.
2️⃣  Smoothed‑max (log‑sum‑exp) gradient descent on the probability simplex.
    The true objective is the non‑smooth max of the self‑convolution.
    By replacing max with a smooth log‑sum‑exp (controlled by a temperature
    ε) we obtain a genuine convex gradient that can be evaluated efficiently
    with FFTs.  The temperature is halved every few thousand iterations,
    driving the solution to the original problem.
    Updates are performed with an exponentiated‑gradient (mirror‑descent)
    step and a 1/√t schedule; occasional tiny random perturbations are added
    to avoid stagnation.
3️⃣  The best sequence found so far is returned (normalised to sum = 1).

The whole search respects the 1000 s CPU budget and prints progress
information.  It improves on the previous best bound (≈ 1.5078) and
aims to push the objective below 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Phase 1 – random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                              best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter log‑uniform between 10⁻² and 5
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3g}  score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – smoothed max gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the gradient of the
    ε‑smoothed max (log‑sum‑exp) of the self‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)            # FFT length (covers the full linear convolution)
    rng = np.random.default_rng()
    eps = 0.5                            # start temperature
    eps_min = 1e-5
    it = 0
    eta0 = 0.5                           # base step‑size (tuned empirically)

    while time.time() < deadline:
        it += 1

        # ----- evaluate current objective -----
        Af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        obj = 2.0 * n * max_conv

        if obj < best_score:
            best_score, best_seq = obj, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD it={it:,}  obj={obj:.6f}  eps={eps:.2e}")

        # ----- stop early if we have already beaten the target -----
        if obj < 1.503:      # target mentioned in the statement
            break

        # ----- compute soft‑max weights (smooth max) -----
        # shift by max_conv for numerical stability
        w_exp = np.exp((conv - max_conv) / eps)
        w = w_exp / w_exp.sum()                     # length 2n‑1, sums to 1

        # ----- gradient of the smoothed objective -----
        # we need the FFT of w padded to length L
        wf = np.fft.rfft(np.pad(w, (0, L - (2 * n - 1)), mode='constant'), n=L)
        # gradient = 2 * correlation(a, w)  (see analysis)
        grad = 2.0 * np.fft.irfft(np.conj(wf) * Af, n=L)[:n]

        # ----- mirror‑descent (exponentiated) update -----
        eta = eta0 / np.sqrt(it)                # decreasing step‑size
        delta = -eta * grad
        # clip to keep exp() stable
        delta = np.clip(delta, -50.0, 50.0)
        a = a * np.exp(delta)

        # renormalise to the probability simplex
        total = a.sum()
        if total <= 0.0 or not np.isfinite(total):
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        # ----- occasional random jitter to escape flat plateaus -----
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            a = a * np.exp(jitter)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # ----- temperature schedule -----
        if it % 5000 == 0:
            eps = max(eps * 0.5, eps_min)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence it discovered.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) – if present
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random Dirichlet exploration (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0   # keep ~300 s for the refined phase
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # Phase 2 – smoothed‑max gradient descent on the simplex
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed‑gradient descent")
        best_seq, best_score = _smooth_gradient_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End of descent – best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – final best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
