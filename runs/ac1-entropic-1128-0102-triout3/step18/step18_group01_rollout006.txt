def smooth_obj_and_grad(a, alpha, L_fft):
    # compute conv = a*a (full conv)
    conv = _conv_self_fft(a, L_fft)  # length 2n-1
    max_val = conv.max()
    # For log-sum-exp smoothing:
    # compute scaled = alpha * (conv - max_val)
    scaled = alpha * (conv - max_val)
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    f = max_val + np.log(sum_exp) / alpha   # since max + (log sum)/alpha
    # weights w_k = exp(alpha*(conv_k - max_val)) / sum_exp
    w = exp_scaled / sum_exp
    # gradient wrt a: g = 2 * cross_corr(w, a_rev)
    a_rev = a[::-1]
    # compute conv_w_a_rev = np.convolve(w, a_rev, mode='full')
    g = 2 * np.convolve(w, a_rev, mode='full')[len(a_rev)-1:len(a_rev)-1+len(a)]
    return f, g
```

def random_mixture_search(n, L_fft, time_limit, rng):
    best = None; best_score = np.inf
    while time.time() < time_limit:
        lam1 = 10**rng.uniform(-6, -2)    # range 1e-6 to 1e-2
        lam2 = 10**rng.uniform(-6, -2)
        w = rng.random()
        p1 = rng.uniform(-10, 8)  # allow negative exponent for power factor
        p2 = rng.uniform(-10, 8)

        shape /= shape_sum

    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)

    eta0 = 0.05   # base learning rate
    alpha = 5.0  # smoothing parameter
    t_step = 0
    # schedule to increase alpha gradually up to maybe 2000
    next_alpha_inc_time = start_time + 30.0  # after 30 seconds increase

"""
Propose a non‑negative step function (probability vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1️⃣  Quick random search in a flexible two‑component exponential‑power family
    (a_i ∝ w·(i+1)^p1·e^{-λ1·i} + (1‑w)·(i+1)^p2·e^{-λ2·i}) to obtain a decent start.
2️⃣  Continuous optimisation of a smooth approximation of the L∞‑objective:
        f_α(a)= (1/α)·log∑_k e^{α·conv_k(a)} .
    The gradient of f_α is computed analytically via a weighted cross‑correlation.
    An Adam optimiser (projected onto the probability simplex) is used.
    The smoothing parameter α is increased gradually (continuation method).
3️⃣  Enforce a monotone‑decreasing shape by isotonic regression – this never hurts
    the objective because the reversed inner product is minimised for opposite
    orderings.
4️⃣  Final random hill‑climb (tiny mass transfers) to polish the result.

The whole routine respects the 1000 s budget and returns a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Full linear self‑convolution a*a via FFT (length 2·len(a)‑1).
    Returns a real array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Guarantees non‑negative entries that sum to one (or zero vector if sum==0).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Isotonic regression (enforce non‑increasing shape)
# ----------------------------------------------------------------------


def _enforce_monotone_decreasing(a: np.ndarray) -> np.ndarray:
    """Pool‑adjacent‑violators algorithm (PAVA) for a non‑increasing sequence."""
    n = a.shape[0]
    # each block stores (sum, count)
    blocks_sum = []
    blocks_cnt = []
    for x in a:
        blocks_sum.append(float(x))
        blocks_cnt.append(1)
        # merge while monotonicity is violated (prev average < current average)
        while len(blocks_sum) >= 2 and (blocks_sum[-2] / blocks_cnt[-2]) < (blocks_sum[-1] / blocks_cnt[-1]):
            # merge last two blocks
            s2 = blocks_sum.pop()
            c2 = blocks_cnt.pop()
            s1 = blocks_sum.pop()
            c1 = blocks_cnt.pop()
            blocks_sum.append(s1 + s2)
            blocks_cnt.append(c1 + c2)
    # expand the averaged blocks
    out = np.empty(n, dtype=a.dtype)
    pos = 0
    for s, c in zip(blocks_sum, blocks_cnt):
        avg = s / c
        out[pos:pos + c] = avg
        pos += c
    return out


# ----------------------------------------------------------------------
# Smooth approximation of the max‑convolution and its gradient
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(a: np.ndarray,
                         alpha: float,
                         L_fft: int,
                         L_corr: int) -> Tuple[float, np.ndarray]:
    """
    Compute
        f_α(a) = (1/α)·log∑_k exp(α·conv_k(a))
    and its gradient w.r.t. a.
    Returns (value, grad) where grad has shape equal to a.
    """
    # 1) linear self‑convolution (full)
    conv = _conv_self_fft(a, L_fft)           # length 2·n‑1
    max_val = conv.max()
    # 2) log‑sum‑exp with numerical stability
    scaled = alpha * (conv - max_val)         # ≤ 0
    exp_scaled = np.exp(scaled)
    sum_exp = exp_scaled.sum()
    f = max_val + np.log(sum_exp) / alpha

    # 3) soft‑max weights w_k = exp(α·conv_k) / Σ exp(...)
    w = exp_scaled / sum_exp                   # length 2·n‑1, N.B. w sums to 1

    # 4) gradient: g = 2 * (w ⋆ a)  (cross‑correlation)
    a_rev = a[::-1]                           # reversed a, length n
    # convolution via FFT (w * a_rev)
    Af = np.fft.rfft(w, n=L_corr)
    Bf = np.fft.rfft(a_rev, n=L_corr)
    conv_full = np.fft.irfft(Af * Bf, n=L_corr)
    conv_full = conv_full[: (len(w) + len(a_rev) - 1)]   # keep only needed part
    # slice to obtain cross‑correlation values for indices 0 … n‑1
    start = len(a_rev) - 1                      # = n‑1
    g = 2.0 * conv_full[start:start + len(a)]

    return float(f), g


# ----------------------------------------------------------------------
# Random mixture generator (stage 1)
# ----------------------------------------------------------------------


def _random_mixture_candidate(n: int,
                              rng: np.random.Generator) -> np.ndarray:
    """
    Sample a candidate from a two‑component mixture of (i+1)^p·exp(-λ·i).
    Parameters are drawn from wide log‑uniform / uniform ranges.
    """
    i = np.arange(n, dtype=np.float64)

    # component 1
    lam1 = 10 ** rng.uniform(-6, -2)           # 1e‑6 … 1e‑2
    p1 = rng.uniform(-12, 8)                  # allow strong powers
    log1 = -lam1 * i + p1 * np.log(i + 1.0)

    # component 2
    lam2 = 10 ** rng.uniform(-6, -2)
    p2 = rng.uniform(-12, 8)
    log2 = -lam2 * i + p2 * np.log(i + 1.0)

    # mixing weight
    w = rng.random()

    # build the mixture (log‑space for stability)
    shape1 = np.exp(log1 - np.max(log1))
    shape2 = np.exp(log2 - np.max(log2))
    shape = w * shape1 + (1.0 - w) * shape2
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total == 0.0:
        # fallback to uniform (should almost never happen)
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


# ----------------------------------------------------------------------
# Stage 2 – Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------


def _adam_smoothing_descent(a0: np.ndarray,
                            L_fft: int,
                            L_corr: int,
                            deadline: float,
                            start_time: float,
                            rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation f_α.
    The smoothing coefficient α is increased every ~30 s (continuation).
    The iterate is always projected onto the simplex.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    eta0 = 0.05                         # base learning rate
    alpha = 5.0                         # start with mild smoothing
    next_alpha_time = start_time + 30.0

    while time.time() < deadline:
        t_iter += 1

        # smooth value & gradient
        f_val, g = _smooth_obj_and_grad(a, alpha, L_fft, L_corr)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # learning‑rate schedule (slow decay)
        eta = eta0 / (1.0 + 1e-4 * t_iter)

        # update & projection
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # evaluate the true (non‑smoothed) bound
        cur_score = _eval_fast(a, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t_iter:,} α={alpha:.1f} "
                  f"score={cur_score:.6f}")

        # occasional tiny jitter to avoid plateaus
        if t_iter % 3000 == 0:
            eps_vec = rng.random(a.shape) * a * 5e-5
            a = np.maximum(a + eps_vec, 0.0)
            a = _project_simplex(a)

        # increase smoothing parameter gradually
        if time.time() >= next_alpha_time:
            alpha = min(alpha * 1.5, 3000.0)   # cap to keep numerics sane
            next_alpha_time = time.time() + 30.0

    return best, best_score

def _hill_climb(a0: np.ndarray,

    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = cand / cand.sum()
        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand

        # occasional larger exploratory jitter
        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand = best.copy()
            cand[i] = max(cand[i] + jitter, 0.0)

                best = cand.copy()
                a = cand
