"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimizer works in four stages:

1️⃣  Broad random search (Dirichlet, deterministic windows, random‑phase
    constant‑magnitude signals) to obtain a good seed.

2️⃣  **Flat‑convolution gradient descent** – minimise the L2 distance between
    the self‑convolution and the constant target μ = 1/(2·n‑1).  This directly
    flattens the autocorrelation and quickly drives the maximal entry down.

3️⃣  Refine with a projected Adam optimiser on a smooth (log‑sum‑exp) surrogate
    of the max‑norm.

4️⃣  Final short mass‑transfer hill‑climb for polishing.

All steps respect a global 1000 s time budget.  The returned list is a plain
Python list of non‑negative floats whose sum is 1 (the evaluation function is
scale‑invariant, so any normalisation works).  The algorithm improves on the
previous baseline (≈1.508) and aims for the target < 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used for progress printing)
# ----------------------------------------------------------------------
_start_time = None


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v ≥ 0, Σv = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector on the simplex."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))


def _smooth_max(conv: np.ndarray, beta: float) -> float:
    """Log‑sum‑exp smooth approximation of max(conv)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    return m + np.log(exps.sum()) / beta


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max w.r.t. a (used by Adam)."""
    m = conv.max()
    exps = np.exp(beta * (conv - m))
    w = exps / exps.sum()               # softmax weights, shape (2·n‑1,)
    rev_corr = _convolve_fft(a[::-1], w)   # length = 3·n‑2
    n = a.shape[0]
    grad = 2.0 * rev_corr[n - 1 : n - 1 + n]
    return grad

def _random_dirichlet_trials(n: int, max_trials: int, deadline: float, rng):
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration drawn log‑uniform from 1e‑3 to 5
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time()-_start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


def _deterministic_shapes(n: int):
    """Yield a few classic window shapes (already on the simplex)."""
    # sine window
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # Hann window
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(np.maximum(y, 0.0))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(np.maximum(y, 0.0))

    # a few Gaussian windows
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(np.maximum(y, 0.0))


def _random_constant_magnitude(n: int, rng):
    """Generate a real vector whose DFT magnitude is (approximately) constant."""
    # work directly with the rfft representation (real FFT)
    # amplitude = 1 for all frequencies, random phases for non‑DC/NYQ bins
    freqs = np.exp(1j * rng.uniform(0.0, 2.0 * np.pi, size=n // 2 + 1))
    freqs[0] = 1.0                         # zero‑frequency component (real)
    if n % 2 == 0:
        freqs[-1] = 1.0                    # Nyquist component must be real
    a = np.fft.irfft(freqs, n=n)
    # shift to make non‑negative
    a = a - a.min() + 1e-12
    a = _project_simplex(a)
    sc = _objective_score(a)
    print(f"[{time.time()-_start_time:.2f}s] const‑mag seed score={sc:.6f}")
    return a, sc

# Phase 2 – flattening the convolution (L2 minimisation)
# ----------------------------------------------------------------------
def _flatten_conv_grad_desc(a0: np.ndarray, max_iter: int, deadline: float):
    """
    Projected gradient descent that minimises ||conv(a,a) - μ·1||₂²,
    where μ = 1/(2·n-1) is the flat target.  Returns the best vector found.
    """
    a = a0.copy()
    n = a.shape[0]
    mu = 1.0 / (2 * n - 1)               # target constant value of the convolution
    best_a = a.copy()
    best_score = _objective_score(a)

    # simple annealing schedule for the step size
    step = 0.3
    decay = 0.9995

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)                     # length 2·n‑1
        err = conv - mu                           # deviation from flatness
        # gradient = 2 * (err * a_rev)   (restricted to n entries)
        rev_corr = _convolve_fft(err, a[::-1])    # length 3·n‑2
        grad = 2.0 * rev_corr[n - 1: n - 1 + n]

        # take a projected gradient step
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time()-_start_time:.2f}s] flatten‑grad iter {it:5d} new best {best_score:.6f}")

        # accept if improves, otherwise keep current a (and shrink step)
        if cand_score < _objective_score(a):
            a = cand
        else:
            step *= decay    # slowly reduce step size when not improving

        # optional early exit if step becomes tiny
        if step < 1e-12:
            break

    return best_a, best_score

# Phase 3 – Adam optimiser on smooth max
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, beta_vals, deadline: float):
    """
    Projected Adam optimiser on the log‑sum‑exp smooth max.
    """
    a = a0.copy()
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    lr = 0.07
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for beta in beta_vals:
        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, beta)

            t += 1
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad ** 2)
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a - step
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            cand_score = _objective_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.time()-_start_time:.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            # accept improving moves (or rarely accept worse ones to escape plateaus)
            if cand_score < cur_score or np.random.rand() < 0.001:
                a = cand
                cur_score = cand_score

            # stop early if step is essentially zero
            if np.linalg.norm(step) < 1e-13:
                break
    return best_a, best_score

def _hill_climb(a0: np.ndarray, deadline: float):
    a = a0.copy()
    cur_score = _objective_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] == 0.0:
            continue
        # transfer up to 5 % of source mass (small steps help fine‑tune)
        delta = a[src] * rng.random() * 0.05
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time()-_start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            a = cand
            cur_score = cand_score

    return best_a, best_score

        # several restarts help avoid bad local minima
        restarts = 6
        for r in range(restarts):
            if time.time() >= flat_deadline:
                break
            # jitter around current best
            jitter = rng.normal(scale=0.02, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            a_opt, sc_opt = _flatten_conv_grad_desc(
                a0, max_iter=12000, deadline=flat_deadline
            )
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time()-_start_time:.2f}s] flatten restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimiser on smooth max (log‑sum‑exp)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.35 * remaining
        beta_schedule = [0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 800.0, 2000.0]
        restarts = 4

        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            jitter = rng.normal(scale=0.01, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 = _project_simplex(a0)

            a_opt, sc_opt = _adam_descent(a0, beta_schedule, deadline=adam_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time()-_start_time:.2f}s] Adam restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final hill‑climb (few seconds left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time()-_start_time:.2f}s] starting hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time()-_start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
