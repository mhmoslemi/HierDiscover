def _greedy_refine(a, cur_score, deadline, rng):
    n = len(a)
    step_factor = 0.2
    best = a.copy()
    best_score = cur_score
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        conv = _conv_self(best)
        max_val = conv.max()
        t_max = int(conv.argmax())
        # compute contributions
        contributions = np.zeros_like(best)
        i_start = max(0, t_max - (n-1))
        i_end = min(n-1, t_max)
        idx = np.arange(i_start, i_end+1)
        contributions[idx] = best[t_max - idx]
        # source: max contributions where best > small
        mask_pos = best > 1e-15
        # artificially set contributions zero where mask: maybe contributions already zero if best=0; fine.
        src = int(np.argmax(contributions * mask_pos))
        # Ensure source has positive mass
        if best[src] <= 1e-15:
            # fallback random
            src = rng.integers(n)
            while best[src] <= 1e-15:
                src = rng.integers(n)
        dst = int(np.argmin(contributions))
        if src == dst:
            dst = (dst + 1) % n
        delta = best[src] * step_factor
        if delta < 1e-12:
            # reduce step factor
            step_factor *= 0.5
            continue
        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        # reproject to simplex (clip negative and renorm)
        cand = np.clip(cand, 0, None)
        cand_sum = cand.sum()
        if cand_sum > 0:

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    global global_start_time
    global_start_time = time.time()
    deadline = global_start_time + budget_s - 0.05

    rng = np.random.default_rng(seed)

    # load starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
            f = np.maximum(starter, 0.0)
            total = f.sum()
            if total > 0:
                f /= total
            else:
                n = 4096
                f = np.full(n, 1.0 / n)
        else:
            raise Exception()
    except Exception:
        n = 4096
        f = np.full(n, 1.0 / n)

    f = np.clip(f, 0, None)
    f /= f.sum()
    best_f = f.copy()
    best_score = _objective_score(best_f)
    print(f"[{time.time() - global_start_time:.2f}s] Initial score = {best_score:.6f}")

    # LP phase
    if _cvxpy_available:
        lp_deadline = min(deadline - 30, global_start_time + 300)  # limit LP to first ~300s or leave at least 30s for greedy
        if time.time() < lp_deadline:
            best_f, best_score = _lp_phase(best_f, lp_deadline, rng)
    else:
        print("cvxpy not available, skipping LP phase")

    # Greedy refinement
    if time.time() < deadline - 1.0:
        print(f"[{time.time() - global_start_time:.2f}s] Starting greedy refinement")
        best_f, best_score = _greedy_refine(best_f, best_score, deadline, rng)

    # final normalisation
    best_f = np.clip(best_f, 0, None)
    s = best_f.sum()
    if s > 0:
        best_f /= s
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    print(f"[{time.time() - global_start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_f.tolist()]

def _lp_improve(f, M, n):
    try:
        import cvxpy as cp
    except Exception:
        return None
    g = cp.Variable(n)
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f
    conv_fg = cp.conv(f_param, g)
    constraints = [conv_fg <= M, g >= 0]

"""
Propose a step‑function (non‑negative sequence) that minimises

        F(a) = 2·n·max(conv(a,a)) / (∑a)²

The algorithm consists of three stages

1. Initialisation – start from the supplied ``height_sequence_1`` (if
   present) otherwise from the uniform vector and normalise to unit sum.

2. Alternating linear‑program (LP) phase – repeatedly solve the LP
   ``max Σ g_j  s.t.  conv(f,g) ≤ M, g ≥ 0`` where ``M`` is the current
   peak of ``conv(f,f)``.  The LP therefore produces a vector with larger
   ℓ₁‑norm while keeping the same bound.  A fine line‑search over the
   convex combination ``(1‑t)f + t·g`` is performed by evaluating the
   full convolution analytically (no extra FFT per t).  This phase
   continues while time permits (up to a modest number of iterations).

3. Greedy “mass‑transfer’’ refinement – the remaining time is spent
   moving a small amount of mass from positions that contribute most to
   the current maximal autocorrelation towards positions that contribute
   least.  The move size is adaptively reduced and a simple back‑track
   line‑search on the transferred amount is used.  Only strict
   improvements are accepted, guaranteeing monotone progress.

Both stages respect the non‑negativity and simplex constraints; the best
vector seen is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global timing handle (set at the beginning of propose_candidate)
# ----------------------------------------------------------------------
global_start_time = None

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_cross(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b (both length n)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a)) assuming Σ a = 1.
    (The evaluation function also rescales to unit sum before calling us.)
    """
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))


# ----------------------------------------------------------------------
# Linear‑programming improvement step (cvxpy)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, M: float, n: int):
    """
    Solve
        maximise   Σ g_j
        subject to conv(f,g) ≤ M   (elementwise)
                   g ≥ 0
    Returns a scaled version of the optimal g with Σ g = Σ f,
    or ``None`` if the LP cannot increase the ℓ₁‑norm.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    g = cp.Variable(n)
    f_par = cp.Parameter(n, nonneg=True)
    f_par.value = f

    conv_fg = cp.conv(f_par, g)               # length 2·n‑1
    constraints = [conv_fg <= M, g >= 0]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=20000,
                   abstol=1e-9, reltol=1e-9, verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    g_opt = np.asarray(g.value).flatten()
    if g_opt is None:
        return None
    g_opt = np.maximum(g_opt, 0.0)
    sum_g = g_opt.sum()
    sum_f = f.sum()

    # No ℓ₁‑increase → we are at a stationary point
    if sum_g <= sum_f + 1e-12:
        return None

    # Rescale back to the original total mass (here Σf = 1)
    scale = sum_f / sum_g
    return g_opt * scale


# ----------------------------------------------------------------------
# LP driven descent (with cheap analytic line‑search)
# ----------------------------------------------------------------------
def _lp_phase(start_vec: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Run a few LP improvement steps while time permits.
    Returns the best vector seen and its objective value.
    """
    f = start_vec.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_score = _objective_score(best_f)
    max_lp_iter = 30
    lp_iter = 0

    while time.time() < deadline - 5.0 and lp_iter < max_lp_iter:
        lp_iter += 1
        M = _max_conv(f)

        g_scaled = _lp_improve(f, M, n)
        if g_scaled is None:
            print(f"[{time.time() - global_start_time:.2f}s] LP iter {lp_iter}: no ℓ₁‑increase")
            break

        # Pre‑compute the three convolutions needed for the line‑search
        conv_ff = _conv_self(f)
        conv_gg = _conv_self(g_scaled)
        conv_fg = _conv_cross(f, g_scaled)

        # Fine line‑search over t ∈ [0,1]
        ts = np.linspace(0.0, 1.0, 101)
        best_local_M = M
        best_local_t = 0.0
        improved = False
        for t in ts:
            # (1‑t)²·ff + t²·gg + 2t(1‑t)·fg  (point‑wise)
            combo = (1.0 - t) ** 2 * conv_ff + t ** 2 * conv_gg + 2.0 * t * (1.0 - t) * conv_fg
            cur_max = combo.max()
            if cur_max < best_local_M - 1e-12:
                best_local_M = cur_max
                best_local_t = t
                improved = True

        if not improved:
            print(f"[{time.time() - global_start_time:.2f}s] LP iter {lp_iter}: line‑search failed")
            break

        # Update f with the best convex combination
        f = (1.0 - best_local_t) * f + best_local_t * g_scaled
        f = np.clip(f, 0.0, None)
        total = f.sum()
        if total > 0.0:
            f /= total

        cur_score = 2.0 * n * best_local_M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time() - global_start_time:.2f}s] LP iter {lp_iter} t={best_local_t:.3f}  new best = {best_score:.6f}")

    return best_f, best_score


# ----------------------------------------------------------------------
# Greedy mass‑transfer refinement
# ----------------------------------------------------------------------
def _greedy_refine(a: np.ndarray, cur_score: float,
                   deadline: float, rng: np.random.Generator):
    """
    Repeatedly move a small amount of mass from a position that contributes
    most to the current maximal autocorrelation to a position that contributes
    least.  The transferred amount is reduced by a back‑track line‑search.
    Only strict improvements are kept.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = cur_score
    step_factor = 0.15               # start with ~15 % of the source mass
    min_delta = 1e-12

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # 1) current convolution and its maximal index
        conv = _conv_self(best)
        max_val = conv.max()
        t_max = int(conv.argmax())

        # 2) compute contribution of each coefficient to the peak at t_max
        contributions = np.zeros_like(best)
        i_start = max(0, t_max - (n - 1))
        i_end = min(n - 1, t_max)
        idx = np.arange(i_start, i_end + 1)
        contributions[idx] = best[t_max - idx]

        # source – highest contribution among indices that actually hold mass
        pos_src = np.where(best > 1e-15)[0]
        if pos_src.size == 0:
            break
        src = pos_src[np.argmax(contributions[pos_src])]

        # destination – lowest contribution (can be a zero entry)
        dst = int(np.argmin(contributions))
        if dst == src:
            # pick the second‑lowest if we happen to select the same place
            sorted_idx = np.argsort(contributions)
            for cand in sorted_idx:
                if cand != src:
                    dst = int(cand)
                    break

        # maximal feasible transfer from src
        max_transfer = best[src] * step_factor
        if max_transfer > best[src]:
            max_transfer = best[src]

        # back‑track line‑search on the transferred mass
        delta = max_transfer
        improved = False
        while delta > min_delta:
            cand = best.copy()
            cand[src] -= delta
            cand[dst] += delta
            # clipping is just a safety net (should stay non‑negative)
            cand = np.clip(cand, 0.0, None)
            # renormalise to counteract floating‑point drift
            s = cand.sum()
            if s > 0.0:
                cand /= s
            cand_score = _objective_score(cand)
            if cand_score < best_score - 1e-12:
                best = cand
                best_score = cand_score
                improved = True
                # after a success we keep the step size relatively large
                step_factor = min(step_factor * 1.05, 0.5)
                print(f"[{time.time() - global_start_time:.2f}s] greedy improved to {best_score:.6f}")
                break
            delta *= 0.5

        if not improved:
            # slowly shrink the step factor when we cannot improve
            step_factor = max(step_factor * 0.9, 0.001)

        # occasional random perturbation to escape tiny plateaus
        if iteration % 2000 == 0:
            d = rng.normal(size=n)
            d -= d.mean()
            norm = np.linalg.norm(d)
            if norm > 0:
                alpha = step_factor * 0.2
                cand = best + alpha * d
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0:
                    cand /= s
                cand_score = _objective_score(cand)
                if cand_score < best_score - 1e-12:
                    best = cand
                    best_score = cand_score
                    print(f"[{time.time() - global_start_time:.2f}s] random perturb improved to {best_score:.6f}")

    return best, best_score

    global_start_time = time.time()

    # --------------------------------------------------------------
    lp_deadline = min(deadline - 30.0, global_start_time + 250.0)  # keep room for refinement
    if time.time() < lp_deadline:
        best_f, best_score = _lp_phase(best_f, lp_deadline, rng)
    else:
        print(f"[{time.time() - global_start_time:.2f}s] Skipping LP phase (time budget)")

    # --------------------------------------------------------------
    # 2) Greedy mass‑transfer refinement (main work)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time() - global_start_time:.2f}s] Starting greedy refinement")
        best_f, best_score = _greedy_refine(best_f, best_score, deadline, rng)

    # --------------------------------------------------------------
    # 3) Final normalisation and return
    # --------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, None)
    total = best_f.sum()
    if total > 0.0:
        best_f = best_f / total
    else:
        best_f = np.full_like(best_f, 1.0 / best_f.size)

    print(f"[{time.time() - global_start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_f.tolist()]
