def _differential_evolution(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    popsize = 12
    F = 0.8
    CR = 0.5
    # Initialize population with Dirichlet draws and best_seq
    population = []
    scores = []
    # include best_seq as first member
    population.append(best_seq.copy())
    scores.append(best_score)
    # fill rest with random Dirichlet
    while len(population) < popsize:
        # random Dirichlet concentration uniform in log range
        c = 10 ** rng.uniform(-2, 0.7)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        population.append(a)
        scores.append(_objective_score(a))
    # Ensure best_seq maybe improved
    for i in range(popsize):
        if scores[i] < best_score:
            best_score = scores[i]
            best_seq = population[i].copy()
    print(f"[...] Initial DE population best score={best_score:.6f}")

    while time.time() < deadline:
        for i in range(popsize):
            # mutation: random distinct a,b,c not i
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            mutant = population[a_idx] + F * (population[b_idx] - population[c_idx])
            # clip to non-neg
            np.maximum(mutant, 0, out=mutant)
            # renormalize
            msum = mutant.sum()
            if msum == 0:
                mutant = np.full(n, 1.0/n)
            else:
                mutant /= msum
            # crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            # renormalize trial again (should already sum to 1)
            tsum = trial.sum()
            if tsum == 0:
                trial = np.full(n, 1.0/n)
            else:
                trial /= tsum
            # evaluate
            score = _objective_score(trial)
            if score < scores[i]:
                population[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_seq = trial.copy()
                    print(f"[{time.time() - start_time:.2f}s] DE new best score={best_score:.6f}")
        # Optional break if no improvement after many generations; but continue until time.
    return best_seq, best_score

"""
Propose a non‑negative step‑function that (almost) minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1️⃣  Fast parametric sweeps (geometric, power, exponential, two‑block,
    and a new Beta‑family sweep) produce good initial candidates.

2️⃣  A large cheap random‑Dirichlet search explores the simplex globally.

3️⃣  A lightweight Differential‑Evolution (DE) optimiser runs on the
    remaining time, using FFT‑based convolutions for cheap objective
    evaluation.  DE mixes the best sequences found so far with random
    mutants; the population size is moderate (≈20) so that many
    generations (tens of thousands) can be performed within the 1000 s
    budget.

The best sequence discovered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:

        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta(n, best_seq, best_score, deadline):
    """
    Sweep a discrete Beta family:
        a_i ∝ x_i^α (1-x_i)^β,   x_i = (i+0.5)/n .
    The shape is mixed with the uniform distribution (lam∈{0,0.5,1}).
    """
    uniform = np.full(n, 1.0 / n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x_clip = np.clip(x, eps, 1 - eps)

    alphas = np.linspace(-0.5, 2.5, 31)   # 31×31·3 ≈ 2883 evaluations
    betas = np.linspace(-0.5, 2.5, 31)

    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                break
            # base Beta shape
            log_seq = a * np.log(x_clip) + b * np.log(1.0 - x_clip)
            seq = np.exp(log_seq)
            # avoid pathological all‑zero (should not happen)
            if seq.sum() == 0:
                continue
            for lam in (0.0, 0.5, 1.0):
                merged = lam * uniform + (1.0 - lam) * seq
                merged /= merged.sum()
                sc = _objective_score(merged)
                if sc < best_score:
                    best_score, best_seq = sc, merged.copy()
                    print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (cheap global search)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter: log‑uniform between 10⁻⁵ and 10²
        logc = rng.uniform(-5.0, 2.0)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3e} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Differential Evolution (population‑based optimisation)
# ----------------------------------------------------------------------
def _differential_evolution(n, best_seq, best_score, deadline):
    """
    Simple DE/rand/1/bin with a modest population.
    Operates entirely on the probability simplex (vectors are re‑normalised
    after each mutation).  The population is seeded with the current best
    sequence plus several random Dirichlet draws.
    """
    rng = np.random.default_rng()
    popsize = 20
    F = 0.8            # differential weight
    CR = 0.5           # crossover probability

    # ----- initialise population -----
    population = []
    scores = []

    # keep current best as first member
    population.append(best_seq.copy())
    scores.append(best_score)

    while len(population) < popsize:
        # random Dirichlet start
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        population.append(a)
        scores.append(_objective_score(a))

    # ensure best is recorded
    for i, sc in enumerate(scores):
        if sc < best_score:
            best_score, best_seq = sc, population[i].copy()

    print(f"[{time.time() - start_time:.2f}s] DE initial best score={best_score:.6f}")

    # ----- main DE loop -----
    while time.time() < deadline:
        for i in range(popsize):
            # choose three distinct indices ≠ i
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)

            mutant = population[a_idx] + F * (population[b_idx] - population[c_idx])
            # enforce non‑negativity
            np.maximum(mutant, 0, out=mutant)
            msum = mutant.sum()
            if msum == 0:
                mutant = np.full(n, 1.0 / n)
            else:
                mutant /= msum

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])
            tsum = trial.sum()
            if tsum == 0:
                trial = np.full(n, 1.0 / n)
            else:
                trial /= tsum

            # evaluate
            trial_score = _objective_score(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_seq = trial.copy()
                    print(f"[{time.time() - start_time:.2f}s] DE new best score={best_score:.6f}")

        # occasional diversity injection (helps escaping plateaus)
        if rng.random() < 0.05 and time.time() < deadline:
            worst = int(np.argmax(scores))
            logc = rng.uniform(-2.0, 0.7)
            c = 10 ** logc
            a = rng.gamma(shape=c, scale=1.0, size=n)
            s = a.sum()
            if s == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s
            population[worst] = a
            scores[worst] = _objective_score(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine runs quick parametric sweeps,
    a large random Dirichlet exploration and a population based
    Differential‑Evolution optimiser.  The best sequence (normalised to sum = 1)
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (if a previous construction is available)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweeps (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 600.0   # leave ~600 s for the stochastic phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_beta(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈200 s)
    # --------------------------------------------------------------
    dir_deadline = deadline - 300.0
    if time.time() < dir_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, dir_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Differential Evolution (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Differential Evolution")
        best_seq, best_score = _differential_evolution(n, best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] DE finished, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
