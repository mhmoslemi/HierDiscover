def _local_search(a, cur_score, deadline, rng):

    best = a.copy()
    best_score = cur_score
    # hold cur values
    cur = a.copy()
    cur_score = cur_score

    while time.time() < deadline:
        # compute full conv, max and index
        conv = _conv_self(cur)
        max_val = float(conv.max())
        k_max = int(conv.argmax())
        # contributions for indices in range
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idxs = np.arange(lo, hi+1)
        contrib = cur[idxs] * cur[k_max - idxs]  # vector
        total_contrib = contrib.sum()

            # choose random src
            src = rng.integers(n)
        else:

            src = rng.choice(idxs, p=prob)

        if lo > 0:
            left = np.arange(0, lo)
        else:

            right = np.arange(hi+1, n)

            right = np.array([], dtype=int)

"""
Hybrid random‐annealing refinement for the step‑function bound.

The objective (after normalising Σa = 1) is
        F(a) = 2·n·max(conv(a,a)).
Uniform a gives F = 2, the best known construction reaches ≈1.5078.
This implementation starts from several simple seeds (uniform, the supplied
global starter, a few Dirichlet draws) and then performs a long simulated‑annealing
local search:

* In each iteration the current maximal convolution entry (k_max) is located.
* The contribution of each coefficient to that entry is a_i·a_{k_max‑i}.
* A source index is chosen with probability proportional to its contribution,
  a destination index outside the “active” range (where the contribution is
  zero) is chosen whenever possible, and a small random fraction of the source
  mass (up to 20 %) is transferred.
* The move is always accepted if it improves the objective; otherwise it can be
  accepted with a probability exp(−Δ/T) where the temperature T decays
  exponentially from 1e‑3 to 1e‑7 over the allocated time.
* The best vector seen so far (across all restarts) is kept and finally
  normalised to unit ℓ₁‑norm.

All heavy work (convolution, max‑search) is done via an FFT‑based
implementation (`_conv_self`).  The algorithm respects the 1000 s budget and
returns a non‑negative list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a)) assuming Σ a = 1.
    The vector a must be non‑negative and already normalised.
    """
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement
# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray,
                  cur_score: float,
                  start_time: float,
                  deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Heavy‑weight random search that moves mass between two positions.
    Accepts strictly improving moves and, according to a decreasing temperature,
    occasionally uphill moves.
    Returns the best vector and its objective value found before *deadline*.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = cur_score

    # temperature schedule parameters
    T_start = 1e-3
    T_end = 1e-7

    # minimal improvement to consider a move “strictly better”
    eps_improve = 1e-13

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) compute current max‑convolution and the shift where it occurs
        # ------------------------------------------------------------------
        conv_full = _conv_self(cur)
        max_val = float(conv_full.max())
        k_max = int(conv_full.argmax())

        # ------------------------------------------------------------------
        # 2) contributions of each coefficient to that maximal entry
        # ------------------------------------------------------------------
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idxs = np.arange(lo, hi + 1)                     # indices that actually contribute
        contrib = cur[idxs] * cur[k_max - idxs]          # a_i * a_{k_max-i}
        sum_contrib = contrib.sum()

        # ------------------------------------------------------------------
        # 3) pick a source index proportionally to its contribution
        # ------------------------------------------------------------------
        if sum_contrib > 0.0:
            prob_src = contrib / sum_contrib
            src = rng.choice(idxs, p=prob_src)
        else:
            src = rng.integers(n)

        # ------------------------------------------------------------------
        # 4) pick a destination index where the contribution to k_max is zero
        #    (i.e. outside the active interval).  If none exist, pick a low‑
        #    contribution index.
        # ------------------------------------------------------------------
        # indices with zero contribution:
        left = np.arange(0, lo) if lo > 0 else np.empty(0, dtype=int)
        right = np.arange(hi + 1, n) if hi < n - 1 else np.empty(0, dtype=int)
        zero_region = np.concatenate((left, right))

        if zero_region.size > 0:
            dst = rng.choice(zero_region)
        else:
            # all indices participate – choose the one with minimal contribution
            all_contrib = np.zeros(n)
            all_contrib[idxs] = contrib
            # invert to favour small values, but keep src out of the choice
            inv = all_contrib.max() - all_contrib + 1e-15
            inv[src] = 0.0
            inv_sum = inv.sum()
            if inv_sum == 0.0:
                dst = rng.integers(n)
                while dst == src:
                    dst = rng.integers(n)
            else:
                prob_dst = inv / inv_sum
                dst = rng.choice(np.arange(n), p=prob_dst)

        if src == dst:
            continue
        if cur[src] <= 1e-16:
            continue

        # ------------------------------------------------------------------
        # 5) transfer a random fraction (up to 20 %) of the source mass
        # ------------------------------------------------------------------
        delta = cur[src] * rng.random() * 0.20   # up to 20 % of source value
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta

        if cand[src] < -1e-12:          # safety check (numerical)
            continue

        cand_score = _objective_score(cand)

        # ------------------------------------------------------------------
        # 6) acceptance rule (simulated annealing)
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        if total <= 0.0:
            temp = 0.0
        else:
            frac = elapsed / total
            # exponential cooling
            temp = T_start * ((T_end / T_start) ** frac)

        accept = False
        if cand_score < cur_score - eps_improve:
            accept = True
        elif temp > 0.0:
            prob = np.exp(-(cand_score - cur_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score

            if cur_score < best_score - eps_improve:
                best = cur.copy()
                best_score = cur_score
                print(f"[{time.time() - start_time:.2f}s] new local best = {best_score:.6f}")

    return best, best_score

# Main entry point required by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the supplied
    ``evaluate_sequence`` objective.  The algorithm combines several simple
    initialisations with an aggressive simulated‑annealing refinement.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (use the provided global starter if sensible)
    # ------------------------------------------------------------------
    default_n = 4096
    starter = None
    if "height_sequence_1" in globals():
        try:
            h = np.asarray(globals()["height_sequence_1"], dtype=float)
            if h.ndim == 1 and h.size > 0:
                starter = h
                print(f"[{time.time() - start_time:.2f}s] Using global starter of length {h.size}")
        except Exception:
            starter = None

    n = starter.size if starter is not None else default_n

    # ------------------------------------------------------------------
    # Build a list of initial vectors (uniform, global starter, random Dirichlet)
    # ------------------------------------------------------------------
    init_vectors = []

    # 1) Global starter (if present)
    if starter is not None:
        a0 = np.maximum(starter, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        # If the starter length differs from the chosen n, reshape by truncating
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                # pad with zeros to reach n
                a0 = np.concatenate([a0, np.zeros(n - a0.size)])
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        init_vectors.append(a0)

    # 2) Uniform vector
    init_vectors.append(np.full(n, 1.0 / n, dtype=float))

    # 3) A few random Dirichlet draws (simple Gamma sampling)
    for _ in range(3):
        gamma = rng.random(n) + 1e-12   # Gamma(1,1) is exponential; uniformly random is fine
        a_rand = gamma / gamma.sum()
        init_vectors.append(a_rand)

    # ------------------------------------------------------------------
    # Search over the initial vectors
    # ------------------------------------------------------------------
    best_overall = None
    best_score = float("inf")
    iteration = 0

    for a0 in init_vectors:
        if time.time() >= deadline:
            break

        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)

        cur_score = _objective_score(a0)
        print(f"[{time.time() - start_time:.2f}s] init {iteration} – score = {cur_score:.6f}")

        # ------------------------------------------------------------------
        # Heavy local refinement (simulated annealing)
        # ------------------------------------------------------------------
        a_refined, refined_score = _local_refine(a0, cur_score, start_time, deadline, rng)

        if refined_score < best_score:
            best_score = refined_score
            best_overall = a_refined.copy()
            print(f"[{time.time() - start_time:.2f}s] *** new global best = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Fallback – if nothing was produced, return a simple uniform vector
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.full(n, 1.0 / n, dtype=float)
        best_score = _objective_score(best_overall)

    # ------------------------------------------------------------------
    # Normalise once more (protect against tiny numerical drift) and return
    # ------------------------------------------------------------------
    best_overall = np.maximum(best_overall, 0.0)
    total = best_overall.sum()
    if total > 0.0:
        best_overall = best_overall / total
    else:
        best_overall = np.full_like(best_overall, 1.0 / best_overall.size)

    print(f"[{time.time() - start_time:.2f}s] finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_overall.tolist()]
