        t+=1
        # compute a from u
        a = np.exp(u)
        sum_a = a.sum()

            a = np.full(n, 1.0/n)
            sum_a = 1.0
        a /= sum_a
        # conv and weights

        maxc = conv.max()

        w = exp_vals / exp_vals.sum()

        w_rev = w[::-1]

        idx = 2*n - 2 - np.arange(n)
        grad_a = 2.0 * conv_aw[idx]
        # gradient w.r.t u (softmax)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u * grad_u)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 / (np.sqrt(t))  # decreasing step
        u -= step * m_hat / (np.sqrt(v_hat) + 1e-8)
        # record improvement every few iters

            if obj < best_score:

                print(f"[{time.time() - start_time:.2f}s] SmoothGrad it={t} obj={obj:.6f} tau={tau:.5f}")
        # schedule temperature
        if t % 2000 == 0:
            tau = max(0.001, tau * 0.9)
    return best_seq, best_score

"""
Propose a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages:

1️⃣  Light‑weight parametric sweeps (geometric, power, exponential and an
    extended two‑block family) to obtain a good initial guess.

2️⃣  Cheap global exploration with many random Dirichlet samples.

3️⃣  An accelerated smooth‑max gradient descent on the simplex.
    The objective max(conv) is replaced by a temperature‑smoothed
    log‑sum‑exp; the gradient w.r.t. the unconstrained logits of a
    (soft‑max parametrisation) is computed analytically using FFT‑based
    convolutions.  Adam with a decaying step‑size drives the optimisation.
    The temperature τ is reduced progressively, yielding ever tighter
    approximations of the true non‑smooth objective.

The routine respects the 1000 s CPU budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    (The true evaluate_sequence multiplies by 1/(sum(seq)**2) but the
    expression is homogeneous, so fixing sum=1 is sufficient.)
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geometric eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-3.0, 3.0, 121):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exponential λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """
    Extended two‑block family: vary block length from 1 % up to 45 %
    of n and sweep the left‑block weight finely.
    """
    for L_frac in np.linspace(0.01, 0.45, 9):          # 1 % … 45 %
        L = int(round(L_frac * n))
        if L < 1 or L > n // 2:
            continue
        for w in np.linspace(0.1, 0.9, 17):           # left‑mass fraction
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Two‑blocks L={L} w={w:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws (cheap global search)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration parameter: log‑uniform between 10⁻² and 10⁰·⁷
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Smooth‑max gradient descent (Adam on the simplex)
# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0: np.ndarray, deadline: float,
                                 best_seq: np.ndarray, best_score: float):
    """
    Optimises the smoothed max‑convolution objective.
    a is represented via logits u (soft‑max) so that a_i = exp(u_i)/∑exp(u).
    Adam is used on u with a decaying step‑size.
    Temperature τ is repeatedly reduced (τ←0.9·τ) to approach the true
    non‑smooth max.
    """
    n = a0.shape[0]
    eps = 1e-12
    u = np.log(np.maximum(a0, eps))            # logits
    m = np.zeros_like(u)                       # Adam first moment
    v = np.zeros_like(u)                       # Adam second moment
    beta1, beta2 = 0.9, 0.999
    t = 0
    tau = 0.02                                 # initial smoothing temperature

    # Pre‑allocated FFT length (covers both 2n‑1 and 3n‑2 convolutions)
    L_fft = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        t += 1
        # ---- construct a from logits (soft‑max) ----
        a = np.exp(u)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
            a_sum = 1.0
        a /= a_sum

        # ---- full self‑convolution (via one FFT) ----
        Af = np.fft.rfft(a, n=L_fft)
        conv_full = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]

        # ---- smooth‑max weights ----
        maxc = conv_full.max()
        # subtract maxc for numerical stability
        exp_vals = np.exp((conv_full - maxc) / tau)
        w = exp_vals / exp_vals.sum()          # w_i ≈ softmax(conv_i / τ)

        # ---- gradient of smooth‑max w.r.t. a ----
        # g_a = 2 * Σ_i w_i ∂conv_i/∂a  = 2 * (a *⊛ w_rev)  (⊛ = convolution)
        w_rev = w[::-1]
        Wf = np.fft.rfft(w_rev, n=L_fft)
        conv_aw = np.fft.irfft(Af * Wf, n=L_fft)[: 3 * n - 2]
        # pick indices 2n‑2‑k, k=0..n‑1
        idx = 2 * n - 2 - np.arange(n)
        grad_a = 2.0 * conv_aw[idx]

        # ---- chain rule through the soft‑max parametrisation ----
        # ∂L/∂u = a * (grad_a - (a·grad_a))
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.1 / np.sqrt(t)                # decaying base step‑size
        u -= step * m_hat / (np.sqrt(v_hat) + 1e-12)

        # ---- occasional logging and true‑objective check ----
        if t % 200 == 0:
            obj = _objective_score(a)          # exact objective (τ=0)
            if obj < best_score:
                best_score, best_seq = obj, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SmoothGrad it={t:,} obj={obj:.6f} τ={tau:.5f}")

        # ---- temperature schedule (geometric cooling) ----
        if t % 1000 == 0:
            tau = max(0.001, tau * 0.90)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine runs a quick parametric sweep,
    a large random Dirichlet exploration and a deterministic smooth‑max
    gradient descent (Adam) on the simplex.  The best sequence (normalised
    to sum = 1) is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # leave a tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweep (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈200 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Smooth‑max gradient descent (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max gradient descent")
        best_seq, best_score = _smooth_max_gradient_descent(
            best_seq, deadline, best_seq, best_score
        )
        # If time permits, try a few random restarts as well
        while time.time() < deadline - 30.0:
            rng = np.random.default_rng()
            a0 = rng.dirichlet(np.ones(n))
            best_seq, best_score = _smooth_max_gradient_descent(
                a0, deadline, best_seq, best_score
            )
        print(f"[{time.time() - start_time:.2f}s] End of descent, best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
