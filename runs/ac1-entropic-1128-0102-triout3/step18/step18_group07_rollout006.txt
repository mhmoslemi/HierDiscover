"""
Propose a step‑function (non‑negative vector) that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

Improvements over the reference implementation
------------------------------------------------
* **Soft‑max parametrisation** – the optimisation variables are the
  unconstrained logits `z`.  The probability vector `a = softmax(z)`
  is always non‑negative and automatically normalised, thus we avoid the
  costly Euclidean projection onto the simplex after every step.
* **Adam on a smooth max** – we minimise a log‑sum‑exp (smooth) version
  of the peak of the convolution and anneal the temperature `β` from low
  to very high values.  Gradient of the smooth max is obtained analytically
  via a convolution with the soft‑max weights.
* **Longer, more aggressive ADAM phase** – the scheduler now performs many
  more iterations (limited only by the global deadline) and uses a larger
  learning‑rate schedule.
* **Simulated‑annealing polishing** – after gradient descent a classic
  SA routine performs mass‑transfer moves on the probability simplex.
  The temperature schedule allows occasional uphill moves early on and
  focuses on strict improvements later.
* **More random restarts & Dirichlet draws** – we increase the number of
  Dirichlet initialisations and jittered restarts, which improves the
  chance of escaping poor basins.

The whole search respects the 1000 s budget.  The best vector found is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used by helper functions for progress messages)
# ----------------------------------------------------------------------
_start_time = None


def _elapsed():
    """seconds elapsed since start of the optimiser."""
    return time.time() - _start_time


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a)


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    # a is assumed to be non‑negative and sum≈1 (tiny rounding errors tolerated)
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * np.max(conv))

# 1️⃣  Random Dirichlet initialisation (wide concentration sweep)

def _dirichlet_start(n: int, max_trials: int, deadline: float):
    rng = np.random.default_rng()
    best_a = None
    best_score = np.inf

    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration parameter c in [1e‑3, 10] (log‑uniform)
        c = 10 ** rng.uniform(-3.0, 1.0)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{_elapsed():.2f}s] Dirichlet c={c:.4f}  score={sc:.6f}")
    return best_a, best_score

# 2️⃣  Classic deterministic windows (already normalised)

def _deterministic_windows(n: int):
    """Yield a few classic windows, already projected onto the simplex."""
    # helper – Euclidean projection onto simplex
    def _project_simplex(v):
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

    # sine window
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    yield _project_simplex(y)

    # Hann window
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    yield _project_simplex(y)

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    yield _project_simplex(y)

    # a few Gaussians
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        yield _project_simplex(y)

# 3️⃣  ADAM on the smooth max (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_softmax(a0: np.ndarray, beta_schedule, deadline: float):
    """
    Adam optimiser on a smooth approximation of the max‑convolution.
    The variables are logits `z` such that a = softmax(z) (always feasible).
    """
    eps = 1e-12
    # initialise logits
    z = np.log(np.maximum(a0, eps))

    n = a0.shape[0]

    # Adam state
    lr = 0.10
    beta1 = 0.9
    beta2 = 0.999
    epsadam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0

    # track best solution
    cur_a = np.exp(z)
    cur_a /= cur_a.sum()
    cur_score = _objective_score(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    for beta in beta_schedule:
        # each temperature runs as long as we have time left
        while time.time() < deadline:
            # ----- probability vector from logits -----
            a = np.exp(z)
            a /= a.sum()                     # ensures sum=1

            # ----- convolution and smooth‑max -----
            conv = _conv_self(a)
            m_val = conv.max()
            exps = np.exp(beta * (conv - m_val))
            w = exps / exps.sum()            # soft‑max over convolution entries

            # ----- gradient w.r.t. a (smooth max) -----
            # conv gradient = 2 * (a_rev ⊛ w)   (where ⊛ is linear convolution)
            rev = _convolve_fft(a[::-1], w)   # length 3·n‑2
            grad_a = 2.0 * rev[n - 1 : n - 1 + n]

            # ----- chain‑rule to logits (softmax) -----
            avg = np.dot(grad_a, a)           # ⟨grad_a , a⟩
            grad_z = a * (grad_a - avg)       # dL/dz_i = a_i (∂L/∂a_i - ⟨∂L/∂a , a⟩)

            # ----- Adam update -----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + epsadam)

            z_new = z - step

            # evaluate new candidate
            a_new = np.exp(z_new)
            a_new /= a_new.sum()
            new_score = _objective_score(a_new)

            # keep the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{_elapsed():.2f}s] Adam β={beta:.1f} new best {best_score:.6f}")

            # accept if improvement (no metropolis here – ADAM already handles noise)
            if new_score < cur_score:
                z = z_new
                cur_a = a_new
                cur_score = new_score

            # early stop if steps become tiny
            if np.linalg.norm(step) < 1e-14:
                break

    return best_a, best_score

# 4️⃣  Simulated‑annealing polishing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, score0: float, deadline: float):
    """
    Classic SA on the simplex: pick two indices, move a random fraction of
    source mass to the destination.  Temperature is geometrically cooled.
    """
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = score0

    n = a.shape[0]
    rng = np.random.default_rng()

    # start relatively warm – allows occasional uphill steps
    T = 0.05
    Tmin = 1e-7
    cooling = 0.9995

    while time.time() < deadline and T > Tmin:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] < 1e-15:
            continue

        # move up to 10 % of the source mass (scaled by temperature)
        max_delta = a[src] * 0.10
        delta = max_delta * rng.random() * (T / 0.05)  # larger moves while hot
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # tiny renormalisation to combat numerical drift
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        cand_score = _objective_score(cand)
        dE = cand_score - cur_score

        # Metropolis acceptance
        if dE < 0 or rng.random() < np.exp(-dE / T):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{_elapsed():.2f}s] SA T={T:.2e} new best {best_score:.6f}")

        T *= cooling

    return best_a, best_score

    # --------------------------------------------------------------
    # Incorporate a possibly‑provided global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size < n:
                    starter = np.concatenate([starter, np.zeros(n - starter.size)])
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter, sc
                    print(f"[{_elapsed():.2f}s] global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – Dirichlet draws + deterministic windows (≈20 s)
    # --------------------------------------------------------------
    phase1_deadline = min(deadline, _start_time + 20.0)
    if time.time() < phase1_deadline:
        seq, sc = _dirichlet_start(n, max_trials=4000, deadline=phase1_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{_elapsed():.2f}s] after Dirichlet best = {best_score:.6f}")

        for shp in _deterministic_windows(n):
            sc = _objective_score(shp)
            if sc < best_score:
                best_seq, best_score = shp.copy(), sc
                print(f"[{_elapsed():.2f}s] deterministic shape score {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on smooth max (soft‑max parametrisation)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.65 * remaining   # give most of the budget to ADAM
        beta_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 30.0, 100.0, 300.0, 800.0,
                      2000.0, 5000.0]
        restarts = 4

        for r in range(restarts):
            if time.time() >= adam_deadline:
                break
            # jitter current best to escape a possible plateau
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.02, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            a_opt, sc_opt = _adam_softmax(a0, beta_sched, deadline=adam_deadline)
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{_elapsed():.2f}s] ADAM restart {r} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final Simulated‑Annealing polishing (last ≈10 % of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{_elapsed():.2f}s] starting SA polishing")
        best_seq, best_score = _simulated_annealing(best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # Normalise, guard against numerical drift and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{_elapsed():.2f}s] finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
