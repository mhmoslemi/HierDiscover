"""
Proposer – random + sub‑gradient + targeted hill‑climb.

The evaluation to minimise is
    2·n·max(conv(a,a)) / (sum(a)**2) .
If we normalise the vector to sum = 1 the denominator disappears and the
objective becomes 2·n·max(conv(a,a)).  The algorithm works in three
stages:

1️⃣  Random Dirichlet draws (many thousands) to obtain a decent starting
    point.  The concentration of the Dirichlet is sampled on a log‑scale
    so that both “almost uniform’’ and fairly sparse vectors are explored.

2️⃣  A cheap sub‑gradient descent that directly reduces the currently
    maximal convolution entry.  For the index *k* where the
    convolution attains its maximum we have
        ∂b_k/∂a_i = 2·a_{k‑i}   (if the index is inside the array)
    and we move a little in the opposite direction, then renormalise onto
    the probability simplex.

3️⃣  A final targeted hill‑climbing phase.  The source index is chosen
    among the 5 % largest coefficients, the destination among the 5 %
    smallest; a fraction of the source mass is transferred.  Improvements
    are kept and a very small uphill allowance is used to escape tiny
    plateaus.

All phases respect the global time budget (default 1000 s).  The best
sequence found is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full convolution via FFT (real inputs, length L padding)."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    return np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _convolve_fft(a, a, L)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v ≥ 0, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, L: int) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = _conv_self(a, L)
    return float(2.0 * n * np.max(conv))


def _deterministic_shapes(n: int) -> list[np.ndarray]:
    """Classic windows, already normalised onto the simplex."""
    out = []

    # sine window
    y = np.sin(np.pi * (np.arange(n) + 1) / (n + 1))
    out.append(_project_simplex(np.maximum(y, 0.0)))

    # Hann window (raised cosine)
    y = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    out.append(_project_simplex(np.maximum(y, 0.0)))

    # symmetric triangular
    y = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    out.append(_project_simplex(np.maximum(y, 0.0)))

    # a few Gaussian windows of varying width
    centre = (n - 1) / 2.0
    for sigma in np.linspace(n / 20.0, n / 2.0, 5):
        y = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        out.append(_project_simplex(np.maximum(y, 0.0)))

    return out


def _gradient_max_conv(a: np.ndarray, idx_max: int) -> np.ndarray:
    """
    Sub‑gradient of the *active* convolution entry.
    For the chosen index k = idx_max we have
        ∂b_k/∂a_i = 2·a_{k‑i}   when the opposite index lies inside the vector,
    otherwise the derivative is zero.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, idx_max - (n - 1))
    i_max = min(n - 1, idx_max)
    i_range = np.arange(i_min, i_max + 1)
    g[i_range] = 2.0 * a[idx_max - i_range]
    return g


# ----------------------------------------------------------------------
# Phase 1 – many random Dirichlet draws
# ----------------------------------------------------------------------
def _random_dirichlet_draws(n: int, max_trials: int,
                           deadline: float, rng: np.random.Generator,
                           L: int, start_time: float):
    best_seq = None
    best_score = float("inf")
    for t in range(max_trials):
        if time.time() >= deadline:
            break
        # draw a concentration parameter on a log‑uniform interval
        conc = 10 ** rng.uniform(-1.0, 0.7)          # roughly 0.1 … 5
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a, L)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet t={t} c={conc:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – sub‑gradient descent on the active entry
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, score0: float,
                         L: int, deadline: float, start_time: float):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = score0
    step = 0.5                     # initial step size (in probability‑mass units)
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)

    while time.time() < deadline:
        conv = _conv_self(a, L)
        idx_max = int(conv.argmax())
        grad = _gradient_max_conv(a, idx_max)

        cand = a - step * grad
        # clip & renorm (cheap simplex projection)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = uniform.copy()

        cand_score = _objective_score(cand, L)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad new best {best_score:.6f}")

        if cand_score < cur_score:
            # successful move – keep it and try a slightly larger step
            a = cand
            cur_score = cand_score
            step = min(step * 1.05, 1.0)
        else:
            # move was not improving → shrink step
            step *= 0.5

        if step < 1e-9:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – targeted hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, score0: float,
                L: int, deadline: float, start_time: float, rng: np.random.Generator):
    a = a0.copy()
    cur_score = score0
    best_a = a.copy()
    best_score = score0
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)

    top_frac = 0.05   # top 5 % for source, bottom 5 % for destination
    top_k = max(1, int(top_frac * n))
    bot_k = max(1, int(top_frac * n))

    while time.time() < deadline:
        # source: among the largest coefficients
        src_candidates = np.argpartition(-a, top_k - 1)[:top_k]
        src = int(rng.choice(src_candidates))
        # destination: among the smallest coefficients
        dst_candidates = np.argpartition(a, bot_k - 1)[:bot_k]
        dst = int(rng.choice(dst_candidates))

        if src == dst or a[src] == 0.0:
            continue

        # transfer up to 15 % of the source mass
        delta = a[src] * rng.random() * 0.15
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # normalise onto the simplex (cheap clip+renorm)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = uniform.copy()

        cand_score = _objective_score(cand, L)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand
            cur_score = cand_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb new best {best_score:.6f}")
        elif cand_score < cur_score:
            # occasional downhill move (tiny Metropolis acceptance)
            a = cand
            cur_score = cand_score

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a list of floats (probability vector).
    """
    # deterministic seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good compromise between expressiveness
    # and speed (FFT of size 8192 fits comfortably in memory).
    # ------------------------------------------------------------------
    n = 4096
    L = _next_pow2(2 * n - 1)                     # length for zero‑padded FFT

    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, L)
    print(f"[{time.time() - start_time:.2f}s] uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – deterministic window shapes (very cheap)
    # ------------------------------------------------------------------
    for shape in _deterministic_shapes(n):
        sc = _objective_score(shape, L)
        if sc < best_score:
            best_seq, best_score = shape.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] deterministic shape score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – massive random Dirichlet sampling (≈30 % of total budget)
    # ------------------------------------------------------------------
    phase1_deadline = min(deadline, start_time + 0.30 * budget_s)
    if time.time() < phase1_deadline:
        rng = np.random.default_rng(seed + 123)
        seq, sc = _random_dirichlet_draws(
            n=n,
            max_trials=50000,                # will stop early if time runs out
            deadline=phase1_deadline,
            rng=rng,
            L=L,
            start_time=start_time,
        )
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] after Dirichlet best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent (≈40 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + 0.40 * remaining
        restarts = 5
        rng = np.random.default_rng(seed + 456)

        for r in range(restarts):
            if time.time() >= subgrad_deadline:
                break
            # jitter the current best to escape possible flat regions
            jitter = rng.normal(scale=0.02, size=n)
            a0 = best_seq + jitter
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            if a0_sum > 0.0:
                a0 /= a0_sum
            else:
                a0 = uniform.copy()

            a_opt, sc_opt = _subgradient_descent(
                a0=a0,
                score0=_objective_score(a0, L),
                L=L,
                deadline=subgrad_deadline,
                start_time=start_time,
            )
            if sc_opt < best_score:
                best_seq, best_score = a_opt, sc_opt
                print(f"[{time.time() - start_time:.2f}s] sub‑grad restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climb (consume the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] starting hill‑climb")
        rng = np.random.default_rng(seed + 789)
        best_seq, best_score = _hill_climb(
            a0=best_seq,
            score0=best_score,
            L=L,
            deadline=deadline,
            start_time=start_time,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Normalise one last time (protect against numerical drift)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] finished – best objective ≈ {best_score:.6f}")
    # Return a plain Python list as required by the harness
    return [float(v) for v in best_seq]
