"""
Propose a step‑function that minimises
    F(a) = 2·n·max(conv(a,a)) / (∑a)²
using a fast greedy refinement.
The method:

1. Initialise a probability distribution (∑a = 1) – use the global
   ``height_sequence_1`` if it exists, otherwise a uniform vector.
2. Compute the full self‑convolution (FFT, O(n log n)).
3. Repeatedly:
   * Locate the index k where the convolution attains its maximum.
   * For that k compute the contribution a[i]·a[k‑i] of each entry.
   * Choose the index `src` with the largest contribution (the entry that
     participates most in the worst sum) and a destination `dst` that has a
     very small current weight.
   * Transfer a fraction of the mass from `src` to `dst`.  The convolution
     can be updated in O(n) by a simple slice operation, so a huge number of
     tiny moves can be tried quickly.
   * If the new maximum convolution is smaller, keep the move;
     otherwise reject it and shrink the step size.
   * Occasionally recompute the convolution from scratch to avoid numerical drift
     and occasionally try a random move (small probability) to escape plateaus.
4. After the time budget is exhausted, clip tiny negatives, renormalise and
   return the list of floats.

The algorithm works directly on the scale‑invariant representation
(∑a = 1), therefore the evaluation function is simply
`2·n·max(conv)`.  It improves on the previous random‑local‑refine
by always attacking the entries that actually form the current worst
convolution, and it uses O(n) updates instead of recomputing a full FFT
every step.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    return float(_conv_self(a).max())


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σ a = 1."""
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))


# ----------------------------------------------------------------------
# Helper: generate an arcsine‑like family (used only for optional warm‑start)
# ----------------------------------------------------------------------
def _shape_sequence(n: int, r: float) -> np.ndarray:
    """Symmetric non‑negative sequence with a_i ∝ ((i+0.5)*(n-0.5-i))^{-r},
    normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    d = (i + 0.5) * (n - 0.5 - i)
    a = d ** (-r)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step function that minimises the given evaluation."""
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1️⃣  Initialise the sequence (∑a = 1)

            else:

            print(f"[{time.time() - start_time:.2f}s] Loaded global starter, n={n}")
        else:
            raise Exception
    except Exception:
        # no good starter – fall back to a uniform vector of a reasonable size
        n = 4096
        a = np.full(n, 1.0 / n, dtype=float)
        print(f"[{time.time() - start_time:.2f}s] Using uniform start, n={n}")

    # --------------------------------------------------------------
    # 2️⃣  Initial evaluation
    # --------------------------------------------------------------
    conv = _conv_self(a)
    cur_max = float(conv.max())
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_score = cur_score
    print(f"[{time.time() - start_time:.2f}s] Initial objective = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Greedy refinement – push mass away from the worst sum
    # --------------------------------------------------------------
    step_factor = 0.05               # typical fraction of src‑mass to move
    iteration = 0
