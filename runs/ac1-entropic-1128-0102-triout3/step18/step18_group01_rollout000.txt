def _smooth_gradient(a: np.ndarray, tau: float, L_corr: int) -> np.ndarray:

    n = a.shape[0]
    # Compute FFT of a padded to L_corr
    A = np.fft.rfft(a, n=L_corr)
    # Compute full convolution a*a (length up to 2n-1)
    conv_full = np.fft.irfft(A * A, n=L_corr)[:2*n-1]
    max_c = conv_full.max()
    # Softmax
    # To avoid overflow: shift by max_c
    exp_vals = np.exp((conv_full - max_c) / tau)

"""
Improved search for a non‑negative step function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three phases:

1️⃣  Fast random sampling of many Dirichlet‑type simplices (different
    concentration parameters) to obtain a good starting point.

2️⃣  Projected gradient descent on a *smooth* approximation of the L∞
    objective.  The max is replaced by a temperature‑controlled soft‑max,
    giving a dense gradient that can be evaluated via FFT‐based
    correlation.  The temperature τ is annealed from 2·10⁻¹ down to 10⁻⁴,
    while the step size decays.  After each update we Euclidean‑project
    back onto the probability simplex.

3️⃣  A cheap random hill‑climb that moves tiny bits of mass between
    random indices, accepting only improvements.

All phases respect the global time budget (default 1000 s).  The best
vector found (normalised to sum = 1) is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2) (sum(a) is assumed >0)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max over the convolution values)
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, tau: float, L_corr: int) -> np.ndarray:
    """
    Gradient of the smoothed max objective:
        f_τ(a) = τ * log Σ_k exp(conv_k(a)/τ)

    Returns ∇f_τ(a) (shape = a.shape).  The gradient is
        2 * (a ⋆ softmax(conv)),
    where ⋆ denotes linear correlation.
    """
    n = a.shape[0]

    # FFT of a (zero‑padded to L_corr)
    A = np.fft.rfft(a, n=L_corr)

    # Full self‑convolution (length 2·n‑1)
    conv_full = np.fft.irfft(A * A, n=L_corr)[: 2 * n - 1]

    # Soft‑max over the convolution entries
    max_c = conv_full.max()
    # shift for numerical stability
    exp_vals = np.exp((conv_full - max_c) / tau)
    softmax = exp_vals / exp_vals.sum()

    # FFT of the weight vector (also zero‑padded to L_corr)
    W = np.fft.rfft(softmax, n=L_corr)

    # Linear correlation a ⋆ softmax = ifft( A * conj(W) )
    corr_full = np.fft.irfft(A * np.conj(W), n=L_corr)

    # The correlation length is 3·n‑2; the slice that corresponds to shift i≥0
    # starts at (len(weight)‑1) = 2·n‑2.
    start = 2 * n - 2
    grad = 2.0 * corr_full[start:start + n]          # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------


def _random_dirichlet_sampling(n: int,
                               L_fft: int,
                               rng: np.random.Generator,
                               deadline: float,
                               start_time: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """
    Generate a stream of Dirichlet‑type vectors with random concentration
    parameters.  Keep the best one seen before the deadline.
    """
    while time.time() < deadline:
        # draw a concentration α in a wide interval; α<1 favours sparsity
        alpha = rng.uniform(0.2, 2.0)
        a = rng.gamma(shape=alpha, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet α={alpha:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed‑max projected gradient descent
# ----------------------------------------------------------------------


def _smooth_descent(a0: np.ndarray,
                    L_fft: int,
                    L_corr: int,
                    deadline: float,
                    start_time: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the τ‑smoothed max objective.
    τ is exponentially annealed from τ₀≈2·10⁻¹ to τ_end≈10⁻⁴.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)

    # annealing schedules
    tau_start = 0.2
    tau_end = 1e-4
    step0 = 0.45

    it = 0
    # we keep a simple counter for “stagnation” restarts
    last_improve = time.time()

    while time.time() < deadline:
        it += 1

        # progress ratio (capped at 1) – controls τ
        progress = min(it / 25000.0, 1.0)
        tau = tau_start * (tau_end / tau_start) ** progress

        step = step0 / (1.0 + 1e-4 * it)   # slowly decreasing step

        g = _smooth_gradient(a, tau, L_corr)

        cand = a - step * g
        cand = _project_simplex(cand)

        cand_score = _eval_fast(cand, L_fft)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand                       # continue from the improved point
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] smooth it={it:,} τ={tau:.2e} "
                  f"step={step:.3e} score={cand_score:.6f}")
        else:
            # occasional tiny jitter to escape tiny plateaus
            if it % 1200 == 0:
                eps = rng.random(a.shape) * a * 4e-5
                a = _project_simplex(a + eps)

        # restart from the current best if we have been stuck for a while
        if time.time() - last_improve > 30.0 and it % 2500 == 0:
            eps = rng.random(best.shape) * best * 1e-4
            a = _project_simplex(best + eps)
            last_improve = time.time()

    return best, best_score


# ----------------------------------------------------------------------
# Final random hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def _hill_climb(a0: np.ndarray,
                L_fft: int,
                deadline: float,
                start_time: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb that moves a tiny fraction of mass from a source index
    to a destination index.  Only improvements are kept.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _eval_fast(best, L_fft)
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand /= cand.sum()

        cand_score = _eval_fast(cand, L_fft)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={cand_score:.6f}")

        # occasional larger exploratory jitter
        if rng.random() < 0.001:
            i = rng.integers(n)
            jitter = rng.random() * 1e-4
            cand2 = best.copy()
            cand2[i] = max(cand2[i] + jitter, 0.0)
            cand2 /= cand2.sum()
            cand2_score = _eval_fast(cand2, L_fft)
            if cand2_score < best_score:
                best_score = cand2_score
                best = cand2.copy()
                a = cand2
                print(f"[{time.time() - start_time:.2f}s] random jitter score={cand2_score:.6f}")

    return best, best_score

    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
