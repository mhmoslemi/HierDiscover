    L∞‑objective: step size is chosen adaptively so that
    η·max(g) ≈ target (≈0.8) and slowly annealed.

4️⃣  A final high‑β L‑BFGS refinement from several perturbations of the
    best point, followed by a short projected sub‑gradient descent and a
    tiny random hill‑climb.

All phases respect the 1000 s budget and the best sequence (normalised to
sum = 1) is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: (i+1)^{α-1}·(n-i)^{β-1}, normalised.
    Allows symmetric or skewed U‑/∧‑shapes.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_sine_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Sinusoidal shape a_i ∝ sin^p(π i/(n+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = np.sin(np.pi * i / (n + 1)) ** power
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _make_triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular shape (increasing then decreasing)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    shape = 1.0 - np.abs(i - mid) / (mid + 1.0)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max objective wrapped for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable object that evaluates the smooth‑max surrogate and its gradient.
    The variables are the unconstrained logits `x`; the probability vector is
    obtained by a soft‑max:  a = softmax(x).  The surrogate is

        fβ(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k(a)).

    This function is differentiable for any β>0.
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable soft‑max conversion."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        """Return (smooth‑max value, gradient w.r.t. logits)."""
        a = self._softmax_logits(x)

        # convolution a*a (full linear)
        conv = _conv_self_fft(a, self.L_fft)

        # smooth‑max weights (soft‑max over β·conv)
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a of the smooth‑max term (unnormalised)
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule: ∂f/∂x_i = a_i·(∂f/∂a_i - Σ_j a_j ∂f/∂a_j)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # smooth‑max surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Polishing phases (same as the reference implementation)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        cand_score = None
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10  # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 is a good compromise between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Baseline – uniform distribution.
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1 if present).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Generate a rich pool of seed vectors.
    # --------------------------------------------------------------
    seeds = [best_seq]  # includes uniform / starter

    # Geometric×power sweep (expanded)
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9999), 11)
    p_vals = np.linspace(-6.0, 6.0, 19)
    lam_vals = [0.0, 0.1, 0.3, 0.5, 0.7, 0.9]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Discrete Beta‑type shapes (expanded)
    alpha_vals = [0.2, 0.5, 1.0, 2.0, 3.0, 5.0]
    beta_vals = [0.2, 0.5, 1.0, 2.0, 3.0, 5.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # Sinusoidal family
    for power in [0.5, 1.0, 2.0, 3.0]:
        seeds.append(_make_sine_shape(n, power))

    # Triangular shape
    seeds.append(_make_triangular_shape(n))

    # Random Dirichlet draws with various concentrations
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # --------------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # --------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 8
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Continuation with L‑BFGS on the smooth‑max surrogate.
    # --------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    max_iter_per_beta = {b: 80 if b <= 256 else 120 for b in beta_schedule}

    for beta in beta_schedule:
        if time.time() >= deadline - 300:      # reserve time for polishing
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for idx in range(len(pool)):
            if time.time() >= deadline - 300:
                break
            a0 = pool[idx]
            eps = 1e-12
            x0 = np.log(a0 + eps)

            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta[beta],
                                                   'disp': False})
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # --------------------------------------------------------------
    # 3️⃣  Perturb the current best and run a high‑β refinement.
    # --------------------------------------------------------------
    if time.time() < deadline - 180:
        high_beta = 2048
        obj_high = _SmoothMaxObj(n, L_fft, L_grad, high_beta)
        log_best = np.log(best_seq + 1e-12)
        perturbed_seeds = []
        for _ in range(5):
            sigma = 0.1 * (0.5 + rng.random())   # moderate noise
            log_new = log_best + sigma * rng.standard_normal(n)
            a_new = np.exp(log_new - np.max(log_new))
            a_new /= a_new.sum()
            perturbed_seeds.append(a_new)

        for a0 in perturbed_seeds:
            if time.time() >= deadline - 180:
                break
            x0 = np.log(a0 + 1e-12)
            res = scipy.optimize.minimize(fun=obj_high.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': 200, 'disp': False})
            a_refine = np.exp(res.x - np.max(res.x))
            a_refine /= a_refine.sum()
            score = _eval_fast(a_refine, L_fft)
            if score < best_score:
                best_score = score
                best_seq = a_refine.copy()
                print(f"[{time.time() - start:.2f}s] high‑β refinement score={score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient (mirror descent) on the true L∞ objective.
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        a = best_seq.copy()
        n = a.shape[0]
        target_eta = 0.8          # we aim for η·max(g) ≈ target_eta
        decay = 0.9995           # exponential cooling of the target
        it = 0
        while time.time() < deadline - 30:
            it += 1
            conv = _conv_self_fft(a, L_fft)
            k = int(np.argmax(conv))
            g = _subgrad_max_conv(a, k)
            max_g = g.max()
            if max_g <= 0:
                break
            eta = target_eta / max_g
            mult = np.exp(-eta * g)
            a_new = a * mult
            s = a_new.sum()
            if s == 0.0:
                a_new = uniform.copy()
            else:
                a_new /= s
            score = _eval_fast(a_new, L_fft)
            if score < best_score - 1e-9:
                best_score = score
                best_seq = a_new.copy()
                a = a_new
                print(f"[{time.time() - start:.2f}s] exp‑grad it={it:,} η={eta:.3e} score={best_score:.6f}")
                # keep the target relatively high after improvement
                target_eta = target_eta * 0.995
            else:
                # no improvement: jitter and slowly cool
                eps = 1e-5 * rng.random(n) * a
                a = np.maximum(a + eps, 0.0)
                a /= a.sum()
                target_eta *= decay
            if it % 5000 == 0:
                # occasional larger random perturbation to escape plateaus
                log_a = np.log(a + 1e-12)
                log_a += 0.02 * rng.standard_normal(n)
                a = np.exp(log_a - np.max(log_a))
                a /= a.sum()

    # --------------------------------------------------------------
    # 5️⃣  Final projected sub‑gradient descent.
    # --------------------------------------------------------------
    if time.time() < deadline - 10:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 10, start,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 6️⃣  Tiny random hill‑climb for any remaining time.
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 5e-4  # ≤0.05 % of source mass
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
