assistantfinal**Solution Overview**

The program searches for a non‑negative step function `a = (a₀,…,aₙ₋₁)` that minimises  

\[
F(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2}
\tag{1}
\]

where `*` denotes the linear convolution.  
Because `F` is scale‑invariant we can restrict ourselves to probability vectors
(`∑a = 1`).  The optimisation is performed on the logarithmic parameters  
`wᵢ = log aᵢ` (soft‑max representation) which automatically guarantees non‑negativity and the simplex constraint.

The core of the search is an **annealed soft‑max smoothed gradient descent**:

* a temperature `τ` is used to replace the `max` in (1) by a smooth soft‑max
  `p_k ∝ exp(conv_k/τ)`.  
  For `τ → 0` the soft‑max becomes a hard max and the gradient coincides with the
  sub‑gradient used in the reference code.
* the exact gradient of the smoothed objective w.r.t. `a` is  

  \[
  ∇_a = 4\,n\; (p * a) ,
  \]

  i.e. a cross‑correlation of the probability vector `a` with the weight vector `p`.
* the gradient w.r.t. the log‑parameters `w` (soft‑max Jacobian) is  

  \[
  ∇_w = a\;\bigl(∇_a - (∇_a·a)\bigr) .
  \]

* Adam (adaptive moments) is employed on `w`.  The learning‑rate decays slowly,
  and `τ` is linearly cooled from `τ₀ = 1e‑2` down to `τ_min = 1e‑8`.
* Whenever a strictly better exact objective is found (`F(a)`) it is stored.
  If no improvement occurs for 30 seconds a **restart** is performed near the
  best‑known solution with added random noise – this helps escape plateaus.
* The algorithm works on a relatively large problem size (`n = 32768`) but
  uses FFT‑based convolutions, so each iteration is cheap (≈ 2 FFT’s).

The whole optimisation runs inside the 1000 s budget, after which the best
sequence found is normalised and returned as a plain Python list.  The internal
best value and, if available, the value computed by the provided
`evaluate_sequence` function are printed for logging.



```python
"""
Annealed soft‑max Adam optimiser for the step‑function problem.

The algorithm works on the log‑probability parameters w (a_i = softmax(w_i)).
A temperature τ slowly cools from 1e‑2 to 1e‑8, turning the L∞ objective
into a smooth soft‑max.  The exact gradient of the smoothed objective is
computed analytically (a cross‑correlation) and Adam updates are applied.
Restarts after stagnation and optional warm‑starts from the global variable
`height_sequence_1` are used to avoid local minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, all top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear convolution x*y via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    if L is None:
        L = _next_pow2(n + m - 1)
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)[: n + m - 1]
    return conv


def _cross_corr(p: np.ndarray, a: np.ndarray, L_corr: int) -> np.ndarray:
    """
    Compute g_i = Σ_j p_{i+j} a_j for all i (cross‑correlation).
    Returned array has length len(a).
    """
    a_rev = a[::-1]
    conv = _conv_fft(p, a_rev, L_corr)          # length = 3·n‑2
    start = a.shape[0] - 1                      # index of a_rev[0] in conv
    return conv[start:start + a.shape[0]]       # slice of length n


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = w.max()
    e = np.exp(w - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def _adam_optimize(w_init: np.ndarray,
                   n: int,
                   L_fft: int,
                   L_corr: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform annealed soft‑max Adam optimisation.

    Parameters
    ----------
    w_init : np.ndarray
        Log‑probability initialisation (size n).
    n : int
        Sequence length.
    L_fft, L_corr : int
        Pre‑computed FFT sizes for the two convolutions.
    deadline : float
        Absolute time (unix timestamp) at which the optimisation must stop.
    start_time : float
        Time at which the whole search started (for temperature schedule).
    best_seq, best_score : current best solution.
    rng : np.random.Generator
        Random generator for restarts.

    Returns
    -------
    best_seq, best_score
    """
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.5
    # temperature schedule
    tau_start = 1e-2
    tau_end = 1e-8

    w = w_init.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    it = 0
    last_improve = time.time()
    stagnation_limit = 30.0          # seconds without improvement → restart
    max_restarts = 5
    n_restarts = 0

    while time.time() < deadline:
        it += 1

        # ------- probability vector a -------
        a = _softmax(w)                       # shape (n,)

        # ------- convolution a*a -------
        conv = _conv_self_fft(a, L_fft)       # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv        # exact objective (sum(a)=1)

        # store improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{time.time() - start_time:.2f}s] iter={it:,} "
                  f"score={cur_score:.6f}")

        # ------- temperature -------
        elapsed = time.time() - start_time
        total = deadline - start_time
        tau = tau_start * (1.0 - elapsed / total) + tau_end * (elapsed / total)

        # ------- soft‑max over the convolution (weights p) -------
        if tau < 1e-9:
            # hard max → sub‑gradient
            p = np.zeros_like(conv)
            p[conv.argmax()] = 1.0
        else:
            shift = conv - max_conv                     # avoid overflow
            exp_shift = np.exp(shift / tau)
            p = exp_shift / exp_shift.sum()

        # ------- gradient w.r.t. a (exact formula) -------
        g_a = 4.0 * n * _cross_corr(p, a, L_corr)     # length n

        # ------- gradient w.r.t. w (soft‑max Jacobian) -------
        dot = np.dot(g_a, a)
        g_w = a * (g_a - dot)

        # ------- Adam parameter update -------
        m = beta1 * m + (1.0 - beta1) * g_w
        v = beta2 * v + (1.0 - beta2) * (g_w * g_w)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (1.0 + 0.0005 * it)           # mild decay
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- stagnation detection & restart -------
        if time.time() - last_improve > stagnation_limit and n_restarts < max_restarts:
            # restart close to the best known solution with added noise
            eps = 1e-12
            w = np.log(best_seq + eps) + rng.normal(scale=0.05, size=n)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve = time.time()
            n_restarts += 1
            print(f"[{time.time() - start_time:.2f}s] restart #{n_restarts}")

    return best_seq, best_score

    if total > 0.0:

    else:
        best_seq = uniform.copy()

    # --------------------------------------------------------------
    # Report final bound using the provided evaluate_sequence (optional).
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())

    except Exception:
        print(f"[{time.time() - start_time:.2f}s] Finished – best internal score ≈ {best_score:.6f}")

    # Return a plain Python list of floats.
    return [float(v) for v in best_seq]
```
