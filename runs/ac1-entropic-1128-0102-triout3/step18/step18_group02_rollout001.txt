"""
Improved search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm expands on the previous three‑phase scheme:

1️⃣  **Larger discretisation** – use n = 8192 (instead of 4096) which gives
    more degrees of freedom without changing the asymptotic cost too much.

2️⃣  **Richer seed pool** – geometric×power, discrete beta, their reverses,
    uniform mixes and many Dirichlet draws (several concentration parameters).
    The 12 best seeds are kept.

3️⃣  **Smooth‑max continuation** – a smooth surrogate
    fβ(a)=2·n·β⁻¹·log Σ exp(β·conv(a,a)) is optimised with L‑BFGS‑B.
    The β‑schedule is extended up to 2048 and each stage runs up to 200
    L‑BFGS iterations per seed.

4️⃣  **Intensive polishing** – exponentiated‑gradient, projected sub‑gradient
    descent and a long random‑hill‑climb are run on the exact objective,
    each using the remaining time budget.

All computations use FFTs for O(n log n) convolution cost.
The final sequence is normalised to sum = 1 and returned as a plain list.
"""

import time

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    p : np.ndarray, shape (2·n‑1,)
        Soft‑max weights over the convolution entries.
    L : int
        Length of the zero‑padded FFT (power of two).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape: (i+1)^{α-1}·(n-i)^{β-1}, normalised.
    Allows symmetric or skewed U‑/∧‑shapes.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

# Smooth‑max surrogate object for L‑BFGS
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable object that evaluates the smooth‑max surrogate and its gradient.
    The optimisation variables are the logits x; the probability vector is
    a = softmax(x).  The surrogate is

        fβ(a) = 2·n·β⁻¹·log Σ exp(β·conv_k(a))

    which is smooth for any β>0.
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable soft‑max conversion."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        """
        Compute (fβ(a), ∇_x fβ(a)) at the point x.

        Returns
        -------
        f_val : float
            Smooth‑max surrogate value.
        grad_x : np.ndarray, shape (n,)
            Gradient w.r.t. the logits x.
        """
        a = self._softmax_logits(x)

        # convolution a*a (full linear)
        conv = _conv_self_fft(a, self.L_fft)

        # smooth‑max weights (soft‑max over β·conv)
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a of the smooth‑max term
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule: ∂f/∂x = a * (∂f/∂a - (a·∂f/∂a)·1)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient utilities for the exact L∞ objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (linear self‑convolution)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            # occasional uphill move (simulated‑annealing style)
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


def _random_hill_climb(a_best: np.ndarray,
                       L_fft: int,
                       deadline: float,
                       start_time: float,
                       best_score: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers – accept only improvements."""
    a = a_best.copy()
    n = a.shape[0]
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 4e-3   # ≤0.4 % of source mass
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            a = prop
            best_score = new_score
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={new_score:.6f}")
    return a, best_score

                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter_per_beta, 'disp': False}
            )
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()
            cand_score = _eval_fast(a_candidate, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            # keep updated seed for next β‑step
            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing (use the rest of the budget).
    # ------------------------------------------------------------------
    if time.time() < deadline - 150:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft, deadline - 150,
            start, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft, deadline - 70,
            start, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Final random hill‑climb (tiny mass transfers).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _random_hill_climb(
            best_seq, L_fft, deadline, start, best_score, rng)

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
