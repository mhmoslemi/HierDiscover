"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm builds a very large and diverse pool of random seeds
(generalised geometric/power shapes, Dirichlet draws, exponential draws,
symmetrised and permuted variants).  The K best seeds are kept and
optimised with a multi‑stage procedure:

1.  Adam on a smooth‑max surrogate (β grows from 0.5 to 2048).  The
    surrogate gradient is obtained via FFT‑based convolution, which is
    orders of magnitude faster than a naïve implementation.

2.  A short evolutionary‑mixing step that repeatedly blends two of the
    current best vectors and re‑optimises the offspring.

3.  Exponentiated‑gradient (multiplicative) polishing of the global best.

4.  Projected sub‑gradient descent on the true L∞ objective.

5.  Final random hill‑climb that moves tiny amounts of mass between
    positions.

All operations respect the simplex constraint (non‑negative, sum = 1)
and all heavy work uses FFT‑based convolutions.  The routine runs for at
most `budget_s` seconds (default 1000 s) and returns the best vector
found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _grad_softmax_new(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns g_i = 2 * Σ_j a_j * p_{i+j}.
    Implemented as a convolution of a_rev with p.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact objective (used for speed inside the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_general_shape(
    n: int, r: float = 0.9, p: float = 0.0, q: float = 0.0
) -> np.ndarray:
    """
    Shape a_i ∝ r**i · (i+1)**p · (n-i+1)**q, normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)  # avoid overflow
    shape = np.exp(log_shape)
    s = shape.sum()
    if s == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= s
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient helpers (used in polishing phases)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Multiplicative (exponential) updates on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(
    a0: np.ndarray,
    L_fft: int,
    deadline: float,
    start_time: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the L∞ objective with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(
                    f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}"
                )
                break
            step *= 0.5
        if not improved:
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
