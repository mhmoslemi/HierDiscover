"""
Propose a non‑negative step function minimizing

        2 * n * max(conv(a, a)) / (sum(a)**2)

by a richer multi‑seed continuation + high‑β smooth‑max L‑BFGS,
followed by a deterministic gradient‑guided hill‑climb that moves
mass from the most “responsible’’ coordinates (largest sub‑gradient)
to the least‑responsible ones.  The method stays well within the
1000 s budget and usually reaches a score ≈ 1.503 for n≈4000.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

    """
    Gradient of the smooth‑max surrogate w.r.t. a.
    Returns 2·∑_j a_j p_{i+j} (i = 0,…,n‑1).

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    p : np.ndarray, shape (2·n‑1,)
        Soft‑max weights over the convolution entries.
    L : int
        Length of the zero‑padded FFT (power of two).

    Returns
    -------
    g : np.ndarray, shape (n,)
        Gradient ∂/∂a_i  Σ_k p_k conv_k .
    """
    n = a.shape[0]
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L)
    Pf = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Pf, n=L)
    start = n - 1
    g = conv[start:start + n]
    return 2.0 * g


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _make_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑type shape: (i+1)^{α-1}·(n-i)^{β-1}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    log_shape = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, normalised."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    shape /= shape.sum()
    return shape


def _make_sine_square_shape(n: int) -> np.ndarray:
    """Sin² shape on (0,π), normalised."""
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * (i + 0.5) / n) ** 2
    shape /= shape.sum()
    return shape


def _make_exponential_ends(n: int, r: float) -> np.ndarray:
    """Mass decaying exponentially away from the ends, symmetrised."""
    i = np.arange(n, dtype=np.float64)
    shape = r ** i + r ** (n - 1 - i)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max objective wrapped for L‑BFGS (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


class _SmoothMaxObj:
    """
    Callable object that evaluates the smooth‑max surrogate and its gradient.
    The variables are the unconstrained logits `x`; the probability vector is
    obtained by a soft‑max:  a = softmax(x).  The surrogate is

        fβ(a) = 2·n·β⁻¹·log Σ_k exp(β·conv_k(a))

    which is differentiable for any β>0.
    """

    def __init__(self, n: int, L_fft: int, L_grad: int, beta: float):
        self.n = n
        self.L_fft = L_fft
        self.L_grad = L_grad
        self.beta = beta

    def _softmax_logits(self, x: np.ndarray) -> np.ndarray:
        """Stable soft‑max conversion."""
        shift = np.max(x)
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()
        return a

    def fun_and_grad(self, x: np.ndarray) -> tuple[float, np.ndarray]:
        """
        Compute (fβ(a), ∇_x fβ(a)) at the point x.

        Returns
        -------
        f_val : float
            Smooth‑max surrogate value.
        grad_x : np.ndarray, shape (n,)
            Gradient w.r.t. the logits x.
        """
        a = self._softmax_logits(x)

        # convolution a*a (full linear)
        conv = _conv_self_fft(a, self.L_fft)

        # smooth‑max weights (soft‑max over β·conv)
        max_conv = conv.max()
        exps = np.exp(self.beta * (conv - max_conv))
        sum_exps = exps.sum()
        p = exps / sum_exps

        # gradient w.r.t. a of the smooth‑max term (unnormalised)
        grad_a = 2.0 * self.n * _grad_softmax_new(a, p, self.L_grad)

        # chain rule: ∂f/∂x_i = a_i·(∂f/∂a_i - Σ_j a_j ∂f/∂a_j)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # smooth‑max surrogate value
        f_val = 2.0 * self.n * (max_conv + np.log(sum_exps) / self.beta)
        return float(f_val), grad_x.astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient helper (true L∞ objective)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Polishing phases
# ----------------------------------------------------------------------


def _gradient_guided_hillclimb(a0: np.ndarray,
                               L_fft: int,
                               deadline: float,
                               start_time: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb guided by the sub‑gradient of the max‑convolution.
    At each step we move a small fraction of mass from the coordinate with
    the largest sub‑gradient (most responsible for the current peak) to the
    coordinate with the smallest sub‑gradient.  The move is accepted only
    if it improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    # initial step size as a fraction of the source mass
    base_frac = 0.02  # 2 % of source mass, later reduced
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k_max = int(np.argmax(conv))
        grad = _subgrad_max_conv(a, k_max)

        # source: index with largest sub‑gradient; destination: smallest
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            # fallback to random move
            src = rng.integers(n)
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

        # adapt the fraction of transferred mass (decrease slowly)
        frac = base_frac / np.sqrt(it + 1.0)
        eps = a[src] * frac
        if eps < 1e-12:
            # nothing to move – break out
            break

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # maintain non‑negativity exactly
        if a_new[src] < 0:
            continue

        # exact evaluation
        new_score = _eval_fast(a_new, L_fft)

        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new  # continue from the improved point
            print(f"[{time.time() - start_time:.2f}s] grad‑climb it={it:,} score={new_score:.6f}")

        # occasional random jitter to escape shallow plateaus
        if it % 2500 == 0:
            jitter = rng.random(n) * 1e-4 * a
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()  # keep on simplex
    return best_seq, best_score


def _final_random_hillclimb(a0: np.ndarray,
                            L_fft: int,
                            deadline: float,
                            start_time: float,
                            best_seq: np.ndarray,
                            best_score: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers; accept only improvements."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        src = rng.integers(n)
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 4e-3   # ≤0.4 % of source mass
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            a = prop
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb it={it:,} score={new_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
        2 * n * max(conv(a,a)) / (sum(a)**2)

    Returns a plain Python list (the distribution is normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.15   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter, if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Seed generation – a richer pool.
    # ------------------------------------------------------------------
    seeds = [best_seq]   # already contains uniform / starter

    # Geometric × power family
    r_vals = np.logspace(np.log10(0.85), np.log10(0.9995), 9)
    p_vals = np.linspace(-4.0, 3.0, 15)
    lam_vals = [0.0, 0.2, 0.5, 0.8]
    for r in r_vals:
        for p in p_vals:
            shape = _make_shape(n, r, p)
            for lam in lam_vals:
                seeds.append(_mix_with_uniform(shape, lam))

    # Beta‑type family (broader range)
    alpha_vals = [0.2, 0.5, 1.0, 2.0, 3.0, 5.0]
    beta_vals = [0.2, 0.5, 1.0, 2.0, 3.0, 5.0]
    for a in alpha_vals:
        for b in beta_vals:
            seeds.append(_make_beta_shape(n, a, b))

    # Gaussian bumps of varying width
    sigma_vals = [n / 20.0, n / 10.0, n / 5.0, n / 3.0]
    for sigma in sigma_vals:
        seeds.append(_make_gaussian_shape(n, sigma))

    # Sine‑square shape (single deterministic seed)
    seeds.append(_make_sine_square_shape(n))

    # Exponential decay from the ends
    r_end_vals = [0.998, 0.999, 0.9995]
    for r in r_end_vals:
        seeds.append(_make_exponential_ends(n, r))

    # Random Dirichlet draws with varied concentration
    for alpha in [0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # Keep the K best seeds after cheap evaluation.
    # ------------------------------------------------------------------
    scores = [_eval_fast(s, L_fft) for s in seeds]
    K = 12
    top_idx = np.argsort(scores)[:K]
    pool = [seeds[i].copy() for i in top_idx]
    pool_scores = [scores[i] for i in top_idx]

    best_score = float(pool_scores[0])
    best_seq = pool[0].copy()
    print(f"[{time.time() - start:.2f}s] initial pool best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Continuation with L‑BFGS on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    beta_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    max_iter_per_beta = 120   # a bit higher than the baseline

    for beta in beta_schedule:
        # reserve at least ~200 s for the polishing phases
        if time.time() >= deadline - 200:
            break
        obj = _SmoothMaxObj(n, L_fft, L_grad, beta)
        for idx in range(len(pool)):
            if time.time() >= deadline - 200:
                break
            a0 = pool[idx]
            eps = 1e-12
            x0 = np.log(a0 + eps)   # logits

            # L‑BFGS‑B (bounded irrelevant because of softmax)
            res = scipy.optimize.minimize(fun=obj.fun_and_grad,
                                          x0=x0,
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': max_iter_per_beta,
                                                   'disp': False})
            # probability vector from logits
            a_candidate = np.exp(res.x - np.max(res.x))
            a_candidate /= a_candidate.sum()

            cand_score = _eval_fast(a_candidate, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start:.2f}s] L‑BFGS β={beta} seed={idx} score={cand_score:.6f}")

            # update the pool entry for possible further β‑steps
            pool[idx] = a_candidate
            pool_scores[idx] = cand_score

    # ------------------------------------------------------------------
    # 3️⃣  Gradient‑guided hill‑climb (deterministic).
    # ------------------------------------------------------------------
    if time.time() < deadline - 80:
        best_seq, best_score = _gradient_guided_hillclimb(
            best_seq, L_fft, deadline - 80, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Final random hill‑climb (tiny moves) for polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 15:
        best_seq, best_score = _final_random_hillclimb(
            best_seq, L_fft, deadline - 15, start,
            best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
