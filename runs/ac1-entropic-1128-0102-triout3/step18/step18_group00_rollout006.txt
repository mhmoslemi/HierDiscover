def _smooth_max_gradient(a, beta, Lg):

    n = a.shape[0]
    conv = _conv_self_fft(a, L_fft)  # L_fft known globally; yields full conv length 2n-1
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/(len(w)))
    else:
        w /= w_sum

    # compute gradient via convolution of a_rev and w
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=Lg)
    Wf = np.fft.rfft(w, n=Lg)
    gr = np.fft.irfft(Af * Wf, n=Lg)[: (n + (2*n-1) - 1)]  # length = 3n-2
    grad = 2.0 * gr[n-1 : n-1 + n]
    return grad, conv.max()

"""
Hybrid optimisation with a smooth‑max gradient.
------------------------------------------------
* Phase 0 – initialise from the provided global starter (if any) or
  from a uniform distribution.
* Phase 1 – random “geo‑power’’ sweep (≈1 min) to get a decent start.
* Phase 2 – projected gradient descent on a *smooth* approximation of
  the max‑convolution.  The weights for the gradient are
  w_k ∝ exp(β·(conv_k – max(conv))) so that several large peaks are
  reduced simultaneously.  β is annealed from a small to a larger value,
  letting the algorithm first smooth the whole distribution and later
  focus on the remaining spikes.
* Phase 3 – simulated‑annealing hill‑climb with larger mass‑transfer
  moves to escape the last tiny plateaux.
The whole routine respects a 1000 s budget and returns a plain Python
list of non‑negative numbers (sum≈1) that minimises

        2·n·max(conv(a,a)) / (∑a)² .
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Families of candidate vectors (same as the baseline code)
# ----------------------------------------------------------------------


def _make_geo_power(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Monotone shape: a_i ∝ (i+1)^p·exp(‑lam·i)  (L1‑normalised)."""
    log_shape = -lam * np.arange(n, dtype=np.float64) + p * log_i
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_geo_power_symmetric(n: int, lam: float, p: float, log_i: np.ndarray) -> np.ndarray:
    """Symmetric “U‑shape’’ (left + right mirror, then normalised)."""
    base = -lam * np.arange(n, dtype=np.float64) + p * log_i
    base -= np.max(base)
    left = np.exp(base)
    sym = left + left[::-1]
    sym /= sym.sum()
    return sym


# ----------------------------------------------------------------------
# Smooth‑max gradient (weighted sub‑gradient)
# ----------------------------------------------------------------------


def _smooth_max_gradient(
    a: np.ndarray,
    beta: float,
    L_fft: int,
    Lg: int,
) -> tuple[np.ndarray, float]:
    """
    Returns (grad, max_conv) where grad is the gradient of the
    soft‑max surrogate of max(conv(a,a)).  The weights are
        w_k ∝ exp(beta·(conv_k – max_conv)).
    """
    n = a.shape[0]

    # 1) self‑convolution (full length 2n‑1)
    conv = _conv_self_fft(a, L_fft)
    max_c = conv.max()

    # 2) soft‑max weights on the convolution entries
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient = 2·(a_rev * w) slice
    a_rev = a[::-1]                     # reversed a
    Af = np.fft.rfft(a_rev, n=Lg)
    Wf = np.fft.rfft(w, n=Lg)
    prod = Af * Wf
    conv_rw = np.fft.irfft(prod, n=Lg)[: (3 * n - 2)]   # length 3n‑2
    grad = 2.0 * conv_rw[n - 1 : n - 1 + n]

    return grad, max_c


def _score_from_max_conv(max_conv: float, n: int) -> float:
    """Objective value for a normalised vector (∑a = 1)."""
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (tiny mass‑transfer moves)
# ----------------------------------------------------------------------


def _simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    L_fft: int,
    deadline: float,
    start_time: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Final SA phase – mass‑transfer moves with linear cooling."""
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    T0 = 2.0e-4
    Tf = 1.0e-8

    while time.time() < deadline:
        # linear cooling based on elapsed fraction
        frac = (time.time() - start_time) / (deadline - start_time + 1e-12)
        T = T0 * (1.0 - frac) + Tf * frac

        src = rng.integers(n)
        if cur[src] < 1e-18:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # move up to 1 % of the source mass (a bit larger than baseline)
        eps = cur[src] * rng.random() * 1e-2
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps

        conv = _conv_self_fft(cand, L_fft)
        cand_score = _score_from_max_conv(conv.max(), n)

        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA improved score={cand_score:.6f}"
                )

        # occasional jitter to keep diversity
        if rng.random() < 0.001:
            jitter = 5e-4 * rng.random(n) * cur
            cur = np.maximum(cur + jitter, 0.0)
            cur = _project_simplex(cur)
            conv = _conv_self_fft(cur, L_fft)
            cur_score = _score_from_max_conv(conv.max(), n)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(
                    f"[{time.time() - start_time:.2f}s] SA jitter improved score={cur_score:.6f}"
                )

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------

        conv = _conv_self_fft(mono, L_fft)
        sc = _score_from_max_conv(conv.max(), n)

            best_seq = mono.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sweep mono λ={lam:.6e} p={p:.2f} score={sc:.6f}"
            )

        # symmetric “U‑shape’’
        sym = _make_geo_power_symmetric(n, lam, p, log_i)
        conv = _conv_self_fft(sym, L_fft)
        sc = _score_from_max_conv(conv.max(), n)
        if sc < best_score:
            best_score = sc
            best_seq = sym.copy()
            print(
                f"[{time.time() - start_time:.2f}s] sweep sym λ={lam:.6e} p={p:.2f} score={sc:.6f}"
            )

    # 2️⃣  Smooth‑max projected gradient descent
    # --------------------------------------------------------------
    grad_deadline = deadline - 200.0                # leave ~200 s for SA

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
