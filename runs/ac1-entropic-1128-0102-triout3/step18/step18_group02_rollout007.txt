"""
LP‑refinement + exponential‑gradient polishing

1️⃣  Initialise a non‑negative vector (taken from the global
    ``height_sequence_1`` if present, otherwise uniform).

2️⃣  Repeatedly solve the linear programme

        maximise   ∑ g_i
        subject to  (a * g)_k ≤ M   ∀ k,
                    g_i ≥ 0,

    where ``a`` is the current distribution, ``M = max(a*a)``.
    The feasible set contains ``a`` itself, and the optimiser drives the
    sum of ``g`` up, i.e. makes ``g`` more “spread”.  We replace
    ``a ← (1‑t) a + t·(g/∑g)`` with a small mixing factor ``t`` and keep the
    best vector seen.

3️⃣  When the time budget is low we fall back to the smooth‑max L‑BFGS
    continuation and the exponentiated‑gradient / sub‑gradient polishing
    used by the reference implementation.  Finally a tiny random hill‑climb
    is performed.

All operations respect the 1000 s budget; the best sequence (normalised to
unit sum) is returned as a plain Python list.
"""

import time
import numpy as np
import scipy.optimize
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Fast exact evaluation (used during the search)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2) for a positive vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient helpers (used in polishing)
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates (exponentiated‑gradient) on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        eta = eta0 / (1.0 + 0.001 * it)
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={score:.6f}")
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with back‑tracking on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5
        if not improved:
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The method combines an LP‑based refinement (max‑convolution
    bound) with exponential‑gradient / sub‑gradient polishing.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.10  # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Initialise a distribution (use the global starter if present)
    # --------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        if init.size == 0 or init.sum() == 0.0:
            raise ValueError
        init = init / init.sum()
        n = init.shape[0]
    except Exception:
        n = 4096
        init = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = init.copy()
    L_fft = _next_pow2(2 * n - 1)
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] initial baseline (n={n}) score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Pre‑compute the sparse convolution matrix pattern (row/col indices)
    # --------------------------------------------------------------
    nnz = n * n
    row_inds = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n)
    col_inds = np.tile(np.arange(n, dtype=np.int32), n)
    A_pattern = csr_matrix((np.empty(nnz), (row_inds, col_inds)),
                           shape=(2 * n - 1, n))

    # --------------------------------------------------------------
    # 3️⃣  LP‑based refinement loop (aggressive early phase)
    # --------------------------------------------------------------
    # mixing factors to try at each LP step
    t_candidates = [0.05, 0.10, 0.20, 0.40, 0.60]
    max_lp_iters = 30                     # safety bound, usually far fewer succeed
    no_improve_cnt = 0

    for it in range(max_lp_iters):
        if time.time() >= deadline - 120:   # keep ~2 min for polishing
            break

        # current max‑convolution value
        conv_cur = _conv_self_fft(best_seq, L_fft)
        M = conv_cur.max()

        # build the actual matrix with the *current* coefficients
        A_pattern.data = np.repeat(best_seq, n)

        # linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M + 1e-12, dtype=np.float64)  # tiny slack for numeric stability
        bounds = [(0.0, None)] * n

        res = scipy.optimize.linprog(
            c,
            A_ub=A_pattern,
            b_ub=b_ub,
            bounds=bounds,
            method='highs',
            options={'presolve': True, 'time_limit': 5.0}
        )

        if not res.success:
            print(f"[{time.time() - start:.2f}s] LP #{it} failed (status={res.status})")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print(f"[{time.time() - start:.2f}s] LP #{it} returned zero‐mass vector")
            break
        g_norm = g / sum_g

        improved = False
        # try different mixing factors, keep the best improvement
        for t in t_candidates:
            cand = (1.0 - t) * best_seq + t * g_norm
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                improved = True
                print(f"[{time.time() - start:.2f}s] LP #{it} t={t:.2f} → score={cand_score:.6f}")
                break

        if not improved:
            no_improve_cnt += 1
            if no_improve_cnt >= 3:
                # three successive LP steps without improvement → stop early
                print(f"[{time.time() - start:.2f}s] LP stopped after {it+1} iterations (no improvement)")
                break
        else:
            no_improve_cnt = 0

    # --------------------------------------------------------------
    # 4️⃣  Polishing: exponential‑gradient + sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 70:
        best_seq, best_score = _expgrad_phase(
            best_seq, L_fft,
            deadline - 70, start,
            best_seq, best_score, rng)

    if time.time() < deadline - 30:
        best_seq, best_score = _subgradient_descent(
            best_seq, L_fft,
            deadline - 30, start,
            best_seq, best_score, rng)

    # --------------------------------------------------------------
    # 5️⃣  Final tiny random hill‑climb (mass transfers)
    # --------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-14:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 4e-3  # ≤0.4 % of the source mass
        prop = best_seq.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_score = _eval_fast(prop, L_fft)
        if new_score < best_score:
            best_seq = prop
            best_score = new_score
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (ensure a genuine probability vector) and return.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
