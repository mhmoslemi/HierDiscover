"""
Optimiser for the step‑function evaluation problem.

The algorithm consists of several lightweight phases:

1.  Initialise from a few analytically‑motivated shapes (uniform,
    triangular, inverse‑sqrt) and from a possible global starter.
2.  Random Dirichlet / geometric‑mixture draws to obtain a good
    starting point.
3.  Smoothed‑max gradient descent with Adam optimisation.
    The max‑convolution is approximated by a soft‑max with a
    temperature β that is increased during the run.  The gradient of
    the soft‑max is obtained by a single FFT‑based correlation,
    which makes each iteration O(n log n).
4.  A final tiny‑mass hill‑climb (the same idea as in the reference
    code) to polish the result.

All phases respect a global 1000 s wall‑clock budget.
The best sequence found is returned as a plain list of Python floats.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (using FFT)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Analytic seed generators
# ----------------------------------------------------------------------


def _triangular_seq(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the middle)."""
    idx = np.arange(n, dtype=float)
    half = (n - 1) / 2.0
    seq = np.minimum(idx, n - 1 - idx)
    seq = seq.astype(float)
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


def _inverse_sqrt_seq(n: int) -> np.ndarray:
    """Inverse‑sqrt shape ≈ arcsine law: 1/√(x(1‑x))."""
    i = np.arange(n, dtype=float) + 0.5
    xs = i / n
    seq = 1.0 / np.sqrt(xs * (1.0 - xs))
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    trials: int = 300,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)  # concentration from ~0.01 to ~5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    trials: int = 250,
) -> tuple[np.ndarray, float]:
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=float)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} w={w.tolist()} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed‑max gradient (Adam optimiser)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights that approximate the max‑operator."""
    m = conv.max()
    # subtract the max for numerical stability
    w = np.exp(beta * (conv - m))
    s = w.sum()
    if s == 0.0:
        # fallback to uniform weights
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max of the self‑convolution.
    Implements  g = 2 * correlation(a, w) for w of length 2n‑1.
    """
    n = a.shape[0]
    L = _next_pow2(n + w.size - 1)
    Af = rfft(a, n=L)
    # correlation a ⋆ w == convolution(a, reverse(w))
    w_rev = w[::-1]
    W = rfft(w_rev, n=L)
    corr_full = irfft(Af * W, n=L)[: n + w.size - 1]
    offset = w.size - 1
    grad = corr_full[offset : offset + n]
    return 2.0 * grad


def _adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    β (soft‑max temperature) is increased linearly during the run.
    """
    a = start_seq.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    lr = 0.07  # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={cur_score:.6f}")

        # schedule for β – start small, finish large
        beta = min(250.0, 2.0 + 0.12 * it)  # grows roughly linearly

        w = _softmax_weights(conv, beta)
        grad = _grad_softmax(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final tiny‑mass hill‑climb (polishing)
# ----------------------------------------------------------------------


def _hill_climb(
    start_a: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Very small mass transfers guided by the true max‑convolution."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        # exact sub‑gradient of the max‑convolution
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        # candidates for source (large positive gradient) and target (small gradient)
        src_cand = np.where(a > 1e-15)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.02 + 1e-6)  # up to 2 % of source mass
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # random jitter to escape a plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.01 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score

        * random Dirichlet / mixture draws
        * smoothed‑max Adam optimisation
        * tiny‑mass hill‑climb polishing
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    np.random.seed(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate n keeps each iteration cheap.
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # Initialise best candidate with a few deterministic shapes
    candidates = [
        uniform,
        _triangular_seq(n),
        _inverse_sqrt_seq(n),
    ]
    best_seq = candidates[0].copy()
    best_score = _objective_score(best_seq)

    for cand in candidates[1:]:
        sc = _objective_score(cand)
        if sc < best_score:
            best_seq, best_score = cand.copy(), sc
    print(f"[{time.time() - start_time:.2f}s] Baseline best → {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈15 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.15
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=400
        )

    # --------------------------------------------------------------
    # Phase 2 – random geometric / uniform mixtures (≈15 % of the budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.30
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=300
        )

    # --------------------------------------------------------------
    # Phase 3 – Smoothed‑max Adam optimisation (≈55 % of the budget)
    # --------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.85
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, phase3_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final tiny‑mass hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
