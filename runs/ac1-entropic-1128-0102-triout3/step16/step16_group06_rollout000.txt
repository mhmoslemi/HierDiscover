"""
Multi‑restart sub‑gradient search for a non‑negative step function.
The algorithm:
  1. Initialise with the uniform distribution and optional global starter.
  2. Do many random Dirichlet / geometric mixture draws to find a good seed.
  3. Repeatedly
        – perturb the current best sequence,
        – run a short projected sub‑gradient descent on 2·n·max(conv(a,a)),
        – keep any improvement.
     This multi‑restart scheme explores many basins of attraction.
  4. Finish with a tiny‑mass hill‑climb for fine polishing.
All work respects the supplied wall‑clock budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution (a * a)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assume sum(a)=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return the evaluation function value for a (sum(a)=1)."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (normalised to sum=1).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=2000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)      # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=1000):
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} w={w} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (limited iterations)
# ----------------------------------------------------------------------


def _subgradient_descent_limited(start_seq, deadline, max_iter=2000):
    """
    Run projected sub‑gradient descent from start_seq.
    Stops after max_iter steps or when the deadline is hit.
    Returns the best sequence seen during this run and its score.
    """
    a = start_seq.copy()
    n = a.shape[0]
    step = 0.2
    min_step = 1e-12
    best_local_seq = a.copy()
    best_local_score = _objective_score(a)
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
        # sub‑gradient of max‑convolution
        grad = _subgradient_maxconv(a, conv)
        gnorm = np.linalg.norm(grad) + 1e-12
        step_len = step / gnorm

        # back‑tracking line search
        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_score = float(2.0 * n * _conv_self(a_new).max())
            if new_score < cur_score or step_len <= min_step:
                break
            step_len *= 0.5

        if new_score < cur_score:
            a = a_new
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, 1e-12)

        # occasional jitter to escape plateaus
        if iteration % 2000 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.01 + 1e-6)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# Tiny‑mass hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # random jitter to break a possible plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the given
    evaluation function.  The method combines many random Dirichlet /
    mixture seeds with a multi‑restart projected sub‑gradient search
    and a final hill‑climb.  The whole procedure stops before the
    supplied wall‑clock budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate n keeps each iteration cheap.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    starter_obj = globals().get('height_sequence_1')
    if starter_obj is not None:
        try:
            starter = np.asarray(starter_obj, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                starter = np.maximum(starter, 0.0)
                s = starter.sum()
                if s > 0:
                    starter = starter / s
                    sc = _objective_score(starter)
                    if sc < best_score:
                        best_score, best_seq = sc, starter.copy()
                        print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈20% of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.20
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=2000
        )

    # ------------------------------------------------------------------
    # Phase 2 – random mixtures (≈10% of budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.30
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=1200
        )

    # ------------------------------------------------------------------
    # Phase 3 – multi‑restart sub‑gradient refinement (≈45% of budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.75
    rng = np.random.default_rng()
    while time.time() < phase3_deadline:
        # perturb current best – multiplicative log‑normal noise
        remaining = phase3_deadline - time.time()
        if remaining < 0.5:
            break
        # noise magnitude shrinks as time runs out
        eps = max(0.002, min(0.08, remaining / 2000.0))
        noisy = best_seq * np.exp(eps * rng.standard_normal(n))
        noisy = np.maximum(noisy, 0.0)
        noisy /= noisy.sum()

        # allocate a tiny slice of the remaining time for this run
        run_deadline = time.time() + min(0.5, remaining * 0.5)
        # run a short sub‑gradient descent
        cand_seq, cand_score = _subgradient_descent_limited(
            noisy, run_deadline, max_iter=1500
        )
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart improved → {best_score:.6f}")

    print(f"[{time.time() - start_time:.2f}s] After restarts → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final tiny‑mass hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
