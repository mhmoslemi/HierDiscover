def simplex_projection(v):
    # projects onto probability simplex (non-negative and sum=1)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function that (almost) minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm uses a smooth approximation of the max‑convolution
(Lipschitz‑smooth log‑sum‑exp) and performs exponentiated‑gradient descent
on the probability simplex.  The temperature (softmax sharpness) β is raised
gradually so that the optimisation first moves in a smooth landscape and
then hones in on the true max‑objective.  After each gradient step the
vector is projected back onto the simplex (simplex‑projection by Duchi et
al.).  The best vector seen so far (according to the exact objective) is
kept and returned.  Random jitter is injected when progress stalls to help
escape flat regions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # index (0‑based) of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The method consists of a smooth, exponentiated‑gradient descent on the
    soft‑max approximation of the max‑convolution, with a gradual increase of
    the soft‑max temperature β.  The best sequence (normalised to sum = 1) is
    finally returned as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    L = _next_pow2(3 * n - 2)               # length for all FFTs (covers 3n‑2)

    # --------------------------------------------------------------
    # Initial vector (try to reuse a previous construction if present)
    # --------------------------------------------------------------
    a = np.full(n, 1.0 / n)                  # uniform start
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                a = starter / s
    except Exception:
        pass

    # initial score
    Af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    best_score = float(2 * n * conv_full.max())
    best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Gradient‑descent parameters
    # --------------------------------------------------------------
    beta = 0.5               # soft‑max sharpness
    beta_max = 40.0
    step0 = 0.3              # base step‑size for exponentiated update
    it = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # ---- convolution and soft‑max weights ---------------------------------
        Af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()

        # soft‑max (log‑sum‑exp) – shift by max for numerical stability
        shifted = beta * (conv_full - max_conv)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerically improbable, fall back to uniform weights
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth objective ---------------------------------
        # grad_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]                     # reversed weight vector
        Wf = np.fft.rfft(w_rev, n=L)
        conv_rev = np.fft.irfft(Af * Wf, n=L)[: 3 * n - 2]   # length >= 3n‑2
        idx = 2 * n - 2 - np.arange(n)      # mapping from i to conv_rev index
        grad = 2.0 * conv_rev[idx]

        # ---- exponentiated‑gradient update (mirror descent on the simplex) ---
        eta = step0 / np.sqrt(it)          # diminishing step‑size
        # prevent overflow/underflow in the exponential
        grad_clip = np.clip(grad, -5.0, 5.0)
        factor = np.exp(-eta * grad_clip)
        a_new = a * factor
        a_new_sum = a_new.sum()
        if a_new_sum == 0.0 or np.isnan(a_new_sum):
            # fall back to uniform if everything collapsed
            a_new = np.full(n, 1.0 / n)
        else:
            a_new = a_new / a_new_sum

        # ---- evaluate the true objective --------------------------------------
        Af_new = np.fft.rfft(a_new, n=L)
        conv_new = np.fft.irfft(Af_new * Af_new, n=L)[: 2 * n - 1]
        score_new = float(2 * n * conv_new.max())

        # keep the best ever seen
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] it={it:,} beta={beta:.2f} "
                  f"score={best_score:.6f}")
        else:
            no_improve += 1

        # ---- schedule for β (soft‑max sharpness) ------------------------------
        if it % 200 == 0 and beta < beta_max:
            beta = min(beta * 1.12, beta_max)

        # ---- occasional random jitter to escape plateaus -----------------------
        if no_improve >= 8000:
            # inject a small zero‑mean perturbation and renormalise
            eps = rng.normal(scale=0.005, size=n)
            a_pert = a_new + eps
            a_pert = np.clip(a_pert, 0.0, None)
            s = a_pert.sum()
            if s > 0.0:
                a = a_pert / s
            else:
                a = np.full(n, 1.0 / n)
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] jitter injected (it={it})")
        else:
            a = a_new

    # --------------------------------------------------------------
    # Final normalisation and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
