"""
Propose a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

by a three‑phase optimisation:

1.  Random Dirichlet / mixture exploration for a good start.
2.  Smooth Adam descent on a *soft* (log‑sum‑exp) surrogate of the
    max‑convolution.  The distribution is parametrised by logits and
    kept on the probability simplex via a soft‑max.
    The smoothing parameter β is increased during the run, turning the
    surrogate gradually into the true max.
3.  A final tiny‑mass hill‑climb (as in the reference code) for last‑mile
    polishing.

The whole procedure respects a 1000 s wall‑clock budget and prints
intermediate scores.  The returned list contains non‑negative numbers
whose sum is 1 (any scaling leaves the evaluation unchanged).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Return the true objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (sum(a)=1).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _random_dirichlet_trials(
    n, best_seq, best_score, deadline, trials=300
):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)  # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(
    n, best_seq, best_score, deadline, trials=250
):
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * np.full(n, 1.0 / n)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} w={w} → {sc:.6f}")
    return best_seq, best_score


def _hill_climb(start_a, deadline):
    """Tiny‑mass hill‑climb (same as in the reference implementation)."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        # move a tiny amount from a “large‑gradient” index to a “small‑gradient” one
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = np.maximum(a, 0.0)
                a /= a.sum()
                no_improve = 0
    return best_a, best_score

                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈20 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.20
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=500
        )

    # --------------------------------------------------------------
    # Phase 2 – random mixtures (≈10 % of the budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.30
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=300
        )

    # --------------------------------------------------------------
    # Phase 3 – smooth Adam descent on the log‑sum‑exp surrogate
    #          (≈45 % of the budget)
    # --------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.75
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth Adam descent")
        # initialise logits from the best sequence so far
        eps = 1e-15
        z = np.log(best_seq + eps)

        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        lr = 0.03                      # base learning rate for logits
        it = 0

        # smoothing parameter: start very smooth and increase
        beta_smooth = 1.0
        beta_target = 80.0

        while time.time() < phase3_deadline:
            it += 1

            # soft‑max to obtain a probability vector (sum = 1)
            maxz = np.max(z)
            expz = np.exp(z - maxz)
            a = expz / expz.sum()

            # FFT size sufficient for both convolution and correlation
            L = _next_pow2(3 * n - 2)

            # FFT of a (shared for convolution and correlation)
            A = rfft(a, n=L)

            # exact self‑convolution (linear)
            conv = irfft(A * A, n=L)[: 2 * n - 1]

            # smooth max – log‑sum‑exp with current β
            mconv = conv.max()
            scaled = beta_smooth * (conv - mconv)          # stabilise exponentials
            # compute softmax of β·conv (used as weights)
            max_scaled = scaled.max()
            sumexp = np.exp(scaled - max_scaled).sum()
            p = np.exp(scaled - max_scaled) / sumexp        # = softmax(β·conv)

            # correlation (grad of smooth max w.r.t a)
            P = rfft(p, n=L)
            corr = irfft(A * np.conj(P), n=L)[:n].real
            grad_a = 4.0 * n * corr

            # gradient w.r.t logits (soft‑max Jacobian)
            avg = np.dot(grad_a, a)
            grad_z = a * (grad_a - avg)

            # Adam update on logits
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            z -= lr * m_hat / (np.sqrt(v_hat) + epsilon)

            # periodic evaluation of the true (non‑smoothed) objective
            if it % 80 == 0:
                cur_score = _objective_score(a)
                if cur_score < best_score:
                    best_score, best_seq = cur_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={cur_score:.6f}")

            # slowly tighten the smooth approximation
            if it % 200 == 0 and beta_smooth < beta_target:
                beta_smooth = min(beta_target, beta_smooth * 1.12)

            # occasional random jitter to escape plateaus
            if it % 1500 == 0:
                rng = np.random.default_rng()
                i, j = rng.choice(n, size=2, replace=False)
                noise = rng.normal(scale=0.001, size=n)
                z += noise * 0.3

        # after the loop, evaluate one final time
        final_a = np.exp(z - np.max(z))
        final_a /= final_a.sum()
        final_score = _objective_score(final_a)
        if final_score < best_score:
            best_score, best_seq = final_score, final_a.copy()
        print(f"[{time.time() - start_time:.2f}s] End of Adam → best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final tiny‑mass hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
