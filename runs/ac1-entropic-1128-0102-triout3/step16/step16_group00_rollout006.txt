"""
Smooth‑max gradient descent for the step‑function objective.

The objective (with the sum of the coefficients normalised to 1) is

    f(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full linear self‑convolution.
The max is replaced by a smooth approximation

    S_β(a) = (1/β)·log Σ_k exp(β·conv_k)

and the exact gradient of S_β can be written as a single convolution,
which we evaluate with FFTs.  Starting from a collection of diverse
initial vectors (uniform, Dirichlet draws, exponential, triangular, sparse)
the algorithm iteratively reduces the true objective, tightening the
smoothness parameter β and adapting the step size.  All iterates are
projected onto the probability simplex to keep the coefficients
non‑negative and summing to 1.

The routine respects a 1000 s budget and returns the best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) ; assumes Σa = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _grad_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max (β‑approximation) w.r.t. a.
    w has length 2·n‑1, a has length n.
    The formula is 2·(w * a_rev)_{n‑1 … n‑1+n-1},
    where * denotes linear convolution and a_rev = a[::-1].
    """
    n = a.shape[0]
    M = w.shape[0]           # = 2·n‑1
    L = _next_pow2(M + n - 1)  # size for the FFT
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a[::-1], n=L)
    full = np.fft.irfft(W * A_rev, n=L)[: M + n - 1]
    # slice n‑1 … n‑1+n‑1 (inclusive) gives n entries
    grad = 2.0 * full[n - 1: n - 1 + n]
    return grad

    if sc < best_score:
        best_score, best_seq = sc, inv_tri.copy()
        print(f"[{time.time() - start_time:.2f}s] Inverse‑triangular score={sc:.6f}")

    # Sparse random support
    for k in (5, 20, 50, 150):
        if time.time() >= deadline:
            break
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        a = np.zeros(n)
        a[idx] = vals
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sparse k={k}   score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣   Smooth‑max gradient descent with adaptive β and step size
    # ------------------------------------------------------------------
    a = best_seq.copy()
    lr = 0.5                        # tentative step length
    beta = 2.0                      # start with a mild smoothing
    beta_max = 2000.0
    beta_inc = 1.005                # slowly sharpen the max approximation

    iteration = 0
    while time.time() < deadline - 0.3:
        iteration += 1

        # current convolution and true objective
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # record best
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] New best (iter {iteration})  score={best_score:.6f}")

        # smooth weights: w_i ∝ exp(β·(conv_i – max))
        w = np.exp(beta * (conv - max_val))          # max entry gives weight 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient of the smoothed objective
        grad = _grad_from_weights(w_norm, a)

        # ---------- back‑tracking line search ----------
        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - lr * factor * grad
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            conv_cand = _conv_self(cand)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score:
                a = cand
                lr = min(lr * 1.05, 2.0)          # accelerate a bit after success
                accepted = True
                break

        if not accepted:
            lr = max(lr * 0.5, 1e-10)            # shrink step when stuck
            # tighten the approximation to the true max
            beta = min(beta * beta_inc, beta_max)
        else:
            # when we are making progress we can also let β grow a little
            beta = min(beta * 1.001, beta_max)

    # ------------------------------------------------------------------
    # 3️⃣   Return the best candidate found (normalised to sum 1)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
