"""
Propose a step‑function (non‑negative vector) that minimises

        evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)

The code expands on the previous pipeline:

1.  A richer parametric sweep (geometric, double‑exponential, extended beta,
    stretched‑exponential, logistic and sine‑power families).
2.  More aggressive Dirichlet sampling.
3.  Exponentiated‑gradient descent on the exact max‑convolution
    (mirrored‑gradient turns out to be very effective on the simplex).
4.  A refined projected sub‑gradient descent (larger initial step,
    simple back‑tracking line‑search, occasional jitter).
5.  A final hill‑climb / simulated‑annealing phase with larger moves.

All stages respect the 1000 s budget.  The routine returns a plain Python
list of floats (a probability vector, sum ≈ 1).  The implementation uses
only NumPy and the provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a * a using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts are active their gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]    # ordered‑pair contribution
    if active.size:
        grad = 2.0 * grad / active.size   # factor 2 because each unordered pair yields two ordered terms
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used during optimisation: 2·n·max(conv(seq,seq)) (sum(seq)==1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _sweep_families(
    n: int,

    deadline: float,

    idx = np.arange(n, dtype=np.float64)

    # ---- uniform (baseline) ------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:

        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- geometric ---------------------------------------------------------
    eps_grid = np.logspace(-6, -1, 300)        # finer and reaches deeper
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):

            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time()-start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- double‑exponential (symmetric) -----------------------------------
    # a_i = exp(-c*i) + exp(-c*(n-1-i))

        seq = left + right
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time()-start_time:.2f}s] DoubleExp c={c:.2f} score={sc:.6f}")

    # ---- power --------------------------------------------------------------
    p_grid = np.linspace(-3.0, 3.0, 151)
    for p in p_grid:

        pow_seq = np.power(idx + 1.0, p)
        pow_seq /= pow_seq.sum()
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time()-start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type (extended range) -----------------------------------------
    alphas = np.linspace(0.05, 10.0, 81)

    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time()-start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- stretched‑exponential ----------------------------------------------
    scales = np.linspace(0.15, 1.0, 15)
    p_vals = np.linspace(0.2, 6.0, 61)
    for scale in scales:
        if time.time() >= deadline:

        for p in p_vals:
            if time.time() >= deadline:

            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):

            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:

def _random_dirichlet_trials(
    n: int,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 6000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration parameter spans a wide range (very spiky to almost uniform)
        log_c = rng.uniform(np.log(0.02), np.log(30.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time()-start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _expgrad_descent(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using an exponential map.
    a_{t+1,i} ∝ a_{t,i} * exp(-η_t * ∇_i f(a_t)),
    where f(a)=2·n·max(conv(a,a)).
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0          # initial step size
    while time.time() < deadline and it < 25000:
        it += 1
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time()-start_time:.2f}s] EG it={it:,} score={cur_score:.6f}")

        # gradient of the exact objective
        subgrad = _subgradient_maxconv(a, conv)      # ∂max/∂a
        grad = 2.0 * n * subgrad                     # ∂obj/∂a

        # step size decays like 1/√t (standard for mirror descent)
        eta = eta0 / np.sqrt(it)

        # multiplicative update (clip to avoid overflow)
        # we cap the exponent to a reasonable range
        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -30.0, 30.0)
        a = a * np.exp(exp_arg)

        # projection onto simplex – simple renormalisation (positivity is preserved)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            # fall back to the best known point if numerics explode
            a = best_seq.copy()
        else:
            a = a / s

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Refined projected sub‑gradient descent (with back‑tracking line‑search)
# ----------------------------------------------------------------------


def _pgd_descent(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent with a simple back‑tracking line‑search.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.8                     # initial step size
    it = 0
    no_improve = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time()-start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad

        # ----- back‑tracking line‑search ----------
        step = eta
        for _ in range(10):
            cand = a - step * grad
            cand = _project_simplex(cand)
            new_score = _objective_score(cand)
            if new_score <= cur_score - 1e-14:
                break
            step *= 0.5
        else:
            # if we never found improvement, shrink aggressively
            step *= 0.5
            cand = a - step * grad
            cand = _project_simplex(cand)
            new_score = _objective_score(cand)

        if new_score < cur_score - 1e-14:
            a = cand
            cur_score = new_score
            no_improve = 0
            eta = max(eta, step)            # keep a healthy step size
        else:
            no_improve += 1
            eta *= 0.7                       # reduce step size on stall
            if no_improve % 400 == 0:
                # occasional jitter to disturb flat regions
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.0005, size=n)
                a = np.clip(a + jitter, 0.0, None)
                a = _project_simplex(a)

        if eta < 1e-9:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final hill‑climb / simulated‑annealing (with larger moves)
# ----------------------------------------------------------------------


def _final_hill_climb(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score

    T0 = 0.03
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # indices that still carry mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            # fallback to random move
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # move a relative amount of mass (up to a few percent)
        frac = rng.random() * 0.02 + 0.0005
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-14:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(
                    f"[{time.time()-start_time:.2f}s] Hill‑climb best score={best_score:.6f}"
                )

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a list of floats (a probability vector).  The algorithm runs a
    multi‑stage pipeline (parameter sweep → Dirichlet sampling → exponentiated‑
    gradient → refined projected GD → hill‑climb) while respecting the time
    budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # small safety margin

    # ------------------------------------------------------------------
    # Problem size (larger n gives finer discretisation, but 4096 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialisation (uniform or external starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time()-start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1) – if present in the environment
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time()-start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad parametric sweep (expanded families & grids)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 230.0  # keep about 230 s for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (more trials)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 180.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient optimisation on the exact max‑convolution
    # ------------------------------------------------------------------
    eg_deadline = deadline - 110.0
    if time.time() < eg_deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting exponentiated‑gradient descent")
        best_seq, best_score = _expgrad_descent(
            best_seq, start_time, eg_deadline, best_seq, best_score
        )
        print(f"[{time.time()-start_time:.2f}s] After EG best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Refined projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print(f"[{time.time()-start_time:.2f}s] Starting refined PGD descent")
        pgd_deadline = deadline - 60.0
        best_seq, best_score = _pgd_descent(
            best_seq, start_time, pgd_deadline, best_seq, best_score
        )
        print(f"[{time.time()-start_time:.2f}s] After PGD best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final hill‑climb / simulated‑annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time()-start_time:.2f}s] Final best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
