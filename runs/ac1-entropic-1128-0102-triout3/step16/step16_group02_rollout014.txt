"""
Propose a step‑function (non‑negative list) that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three phases:

1️⃣  A quick broad sweep of simple parametric families together with many
    random Dirichlet draws (different concentration parameters) – this
    gives a decent starting point.

2️⃣  A smooth‑max gradient descent.  Instead of the non‑smooth
    max (conv) we minimise the log‑sum‑exp of the convolution,
    which is a smooth upper‑bound that becomes tight as a schedule
    parameter β grows.  The gradient of this surrogate is obtained
    analytically (a short convolution).  An exponential‑update
    (a_i ← a_i·exp(−η·∇)) followed by renormalisation stays on the
    probability simplex and automatically respects the non‑negativity
    constraints.

3️⃣  A short random‑jitter / hill‑climb phase to escape possible
    plateaus.

All heavy work (convolutions) is performed with FFTs, the whole routine
stops well before the 1000 s limit and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b via FFT (returns full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = np.fft.rfft(a, n=L)
    Bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns (smooth_max, gradient) where gradient is wrt a.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    # soft‑max weights over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)   # shift for stability
    Z = exp_scaled.sum()
    w = exp_scaled / Z                         # weights, sum to 1

    # gradient:  d/d a_i  (∑_k w_k * conv_k) = 2 * Σ_j a_j * w_{i+j}
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)          # length 3n‑2
    start = len(w) - 1                      # = 2n‑2
    G = 2.0 * conv_rev[start - np.arange(n)]

    # smooth max value = (log Σ exp(beta·conv)) / beta
    smooth_max = (np.log(Z) + max_scaled) / beta
    return smooth_max, G


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponential (mirror‑descent) update: a_i ← a_i·exp(−η·grad_i), then renormalise.
    Guarantees a_i ≥ 0 and Σ a_i = 1.
    """
    # work in log‑space to avoid overflow/underflow
    log_factor = -eta * grad
    # shift so the largest exponent is zero
    log_factor -= np.max(log_factor)
    factor = np.exp(log_factor)
    a_new = a * factor
    total = a_new.sum()
    if total <= 0 or not np.isfinite(total):
        # fall back to a uniform distribution
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total

    eta = 0.5                                 # initial step size
    min_eta = 1e-12
    while time.time() < deadline:
        # linear schedule for β (more aggressive as we go)
        t = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + t * (beta_end - beta_start)

        smooth, grad = _smooth_max_and_grad(a, beta)

        # try exponential update with backtracking line search
        eta_try = eta
        improved = False
        while eta_try > min_eta:
            a_new = _exp_update(a, grad, eta_try)
            new_score = _objective_score(a_new)
            if new_score <= cur_score:
                improved = True
                break
            eta_try *= 0.5

        if improved:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Smooth β={beta:.2f} η={eta_try:.2e} score={new_score:.6f}")
            # slowly increase η when we are making progress
            eta = min(1.0, eta * 1.02)
        else:
            # could not improve – inject a mild random perturbation
            rng = np.random.default_rng()
            eps = 1e-4 * rng.random() * a
            a = a + eps
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            eta = max(min_eta, eta * 0.5)

        # early exit condition (optional)
        if best_score <= 1.503:
            break
    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
    `evaluate_sequence`.  The routine runs a quick parametric sweep,
    many Dirichlet samples, then a smooth‑max gradient descent that
    stays on the simplex via exponential updates.  The best vector
    (normalised to sum = 1) is returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size (the length of the step function)
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweep (a few cheap families)
    # --------------------------------------------------------------
    phase_deadline = deadline - 500.0   # reserve most time for optimisation
    if time.time() < phase_deadline:
        best_seq, best_score = _quick_sweeps(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, trials=3000
        )

    # --------------------------------------------------------------
    # 3️⃣ Smooth‑max gradient descent (core optimiser)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a clean list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
