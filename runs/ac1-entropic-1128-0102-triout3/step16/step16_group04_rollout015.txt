"""
Simulated‑annealing optimizer for the step‑function score

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The evaluator works directly on the raw list, therefore we keep the
vector on the probability simplex (∑a_i = 1).  The algorithm consists of
three phases:

1️⃣  Warm‑start – a quick sweep of Dirichlet draws (or the optional
    global starter) to obtain a decent initial point.

2️⃣  Sub‑gradient mirror descent – we repeatedly locate the index *k* that
    attains the current maximum of the convolution and take an
    exponentiated‑gradient step in the direction of the exact sub‑gradient
    of `max(conv)`.  This drives the mass away from the “hot spot”.

3️⃣  Simulated annealing – random pairwise transfers of probability mass
    are performed.  Worse moves are accepted with probability
    `exp(-(Δscore)/T)`, where the temperature `T` is linearly cooled.
    The best vector seen during the whole run is retained.

All heavy work (convolution, score) is performed with FFTs, which makes a
single evaluation cheap enough to allow hundreds of thousands of
iterations within the 1000 s budget.

The returned list contains non‑negative floats and sums to (approximately)
one, so the evaluator will use it directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length = 2·len(a)-1)."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (2·n·max(conv))."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣ Warm‑start
# ----------------------------------------------------------------------
def _warm_start(n: int, rng: np.random.Generator, deadline: float,
                starter: np.ndarray | None) -> tuple[np.ndarray, float]:
    """Return a Dirichlet‑draw (or the supplied starter) with the smallest score."""
    # start from the provided starter if it looks reasonable
    best_a = None
    best_score = math.inf

    if starter is not None:
        s = starter.sum()
        if s > 0:
            cand = starter / s
            sc = _score(cand, n, _next_pow2(2 * n - 1))
            best_a, best_score = cand, sc

    # cheap Dirichlet sweep – concentration is drawn log‑uniformly
    while time.time() < deadline:
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = math.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        cand = sample / sample.sum()
        sc = _score(cand, n, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_a, best_score = cand, sc
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Sub‑gradient mirror descent (exact max, not soft‑max)
# ----------------------------------------------------------------------
def _subgrad_descent(a: np.ndarray, n: int, L_fft: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using the exact sub‑gradient of the
    `max(conv)` functional.  The iterate stays on the simplex.
    """
    cur = a.copy()
    cur_score = _score(cur, n, L_fft)
    best = cur.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline:
        it += 1
        # compute full convolution once per iteration
        conv = _conv_self_fft(cur, L_fft)
        k = int(np.argmax(conv))                     # active index
        # sub‑gradient: g_i = 2 * a_{k-i} (or 0 outside range)
        grad = np.zeros_like(cur)
        # vectorised fill
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * cur[j[mask]]

        # step size schedule (classic 1/√t)
        eta = 0.8 / math.sqrt(it + 1.0)
        # exponentiated update, then re‑normalise
        exp_factor = np.exp(-eta * grad)
        # avoid extreme values
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        cur *= exp_factor
        cur /= cur.sum()

        cur_score = _score(cur, n, L_fft)
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Desc it={it:,}  score={cur_score:.6f}")

        # ordinary gradient descent sometimes stalls; add a tiny random jitter
        if np.random.rand() < 0.003:
            eps = np.random.normal(scale=1e-3, size=n)
            cur = np.maximum(cur + eps, 0.0)
            cur /= cur.sum()
            cur_score = _score(cur, n, L_fft)

    return best, best_score


# ----------------------------------------------------------------------
# 3️⃣ Simulated annealing + fine mixing
# ----------------------------------------------------------------------
def _anneal(a: np.ndarray, n: int, L_fft: int,
            deadline: float) -> tuple[np.ndarray, float]:
    """
    Random pairwise mass transfers accepted with a temperature schedule.
    The best vector ever seen is returned.
    """
    cur = a.copy()
    cur_score = _score(cur, n, L_fft)
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    # Parameters – they are tuned empirically
    T0 = 5e-3                     # initial temperature
    max_delta = 0.02              # maximal transferred mass in a move
    decay_factor = 0.9999         # slowly shrink max_delta
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # choose a transferable amount (cannot exceed cur[i])
        move_limit = min(cur[i], max_delta)
        if move_limit <= 0.0:
            continue
        delta = rng.random() * move_limit

        # apply move in‑place
        old_i, old_j = cur[i], cur[j]
        cur[i] = old_i - delta
        cur[j] = old_j + delta

        new_score = _score(cur, n, L_fft)

        # accept if better, otherwise Metropolis
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best, best_score = cur.copy(), new_score
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Anneal it={it:,}  new best {new_score:.6f}")
        else:
            # temperature linear in remaining time
            remaining = deadline - time.time()
            frac = remaining / (budget_s - 0.15)   # normalised fraction
            T = T0 * max(frac, 0.0)
            if T > 0 and rng.random() < math.exp(-(new_score - cur_score) / T):
                cur_score = new_score
            else:
                # revert move
                cur[i] = old_i
                cur[j] = old_j

        # slowly tighten the move size
        max_delta = max(max_delta * decay_factor, 5e-6)

    return best, best_score


def _final_mix(best: np.ndarray, n: int, L_fft: int,
               deadline: float) -> np.ndarray:
    """Scan a few convex mixtures with the uniform vector."""
    uniform = np.full(n, 1.0 / n)
    best_score = _score(best, n, L_fft)

    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * best + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _score(cand, n, L_fft)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → {sc:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    `evaluate_sequence`.  The routine respects the time budget and prints
    progress.  The returned list is normalized to (approximately) unit sum.
    """
    global start_time, budget_s
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a safety margin – keep a bit of time for the final return
    deadline = start_time + budget_s - 0.15
    budget_s = budget_s  # needed for temperature computation later

    # --------------------------------------------------------------
    # Problem size – keep it the same as the reference implementation
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    starter = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
    except Exception:
        starter = None

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start – a few Dirichlet draws (≈10 % of time)
    # --------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.10 * (deadline - start_time))
    best_a, best_score = _warm_start(n, rng, warm_deadline, starter)

    print(f"[{time.time() - start_time:.2f}s] Warm‑start best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Sub‑gradient mirror descent (≈30 % of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        descend_deadline = min(deadline,
                               start_time + 0.40 * (deadline - start_time))
        best_a, best_score = _subgrad_descent(best_a, n, L_fft, descend_deadline)
        print(f"[{time.time() - start_time:.2f}s] Sub‑gradient descent best {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Simulated annealing (remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        best_a, best_score = _anneal(best_a, n, L_fft, anneal_deadline)
        print(f"[{time.time() - start_time:.2f}s] Annealing best {best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Final cheap mixing with the uniform distribution
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a = _final_mix(best_a, n, L_fft, deadline)

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    final_score = _score(best_a, n, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(v) for v in best_a]
