"""
search_step_function.py
-----------------------

Search for a non‑negative step‑function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution (full length 2·n‑1).  The code runs for
up to the supplied budget (default 1000 s) and returns the best sequence it
found.

The strategy improves over the previous baseline by

* exploring a new “dual exponential” family (both halves decreasing from the
  centre, which anti‑aligns the sequence with its reversal);
* a dense random sweep of Gaussian‑mixture shapes;
* many more Dirichlet draws;
* a fast Adam–style projected gradient descent from the current best
  candidate.

All operations stay on the simplex (non‑negative, sum = 1) and use an FFT–
based convolution for O(n log n) evaluation.  The final sequence is returned
as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timer (used by many helpers for progress prints)
# ----------------------------------------------------------------------
start_time = None

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣ Custom dual‑exponential family
# ----------------------------------------------------------------------
def _sweep_dual_exponential(n, best_seq, best_score, deadline):
    """
    Both halves decrease when moving away from the centre:
        left  : a_i ∝ exp(-λ_L·i)          (i = 0 … n/2‑1)
        right : a_i ∝ exp(-λ_R·(i‑mid))    (i = mid … n‑1)

    The centre (index mid) is the highest point of the right half.
    """
    mid = n // 2
    left_idx = np.arange(mid, dtype=np.float64)
    right_idx = np.arange(mid, dtype=np.float64)          # offset from mid

    # sample λ on a log‑scale – a few hundred values give a fine grid
    lam_grid = np.logspace(-4, -0.2, 120)                 # 120 × 120 ≈ 14 k evals

    for lam_left in lam_grid:
        if time.time() >= deadline:
            break
        left = np.exp(-lam_left * left_idx)
        for lam_right in lam_grid:
            if time.time() >= deadline:
                break
            right = np.exp(-lam_right * right_idx)
            seq = np.empty(n, dtype=np.float64)
            seq[:mid] = left
            seq[mid:] = right
            s = seq.sum()
            if s == 0.0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] DualExp λL={lam_left:.1e}"
                      f" λR={lam_right:.1e} score={sc:.6f}")
    return best_seq, best_score

def _random_gaussian_mix_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 6000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # pick two Gaussian components
        w1 = rng.random()
        w2 = 1.0 - w1
        mu1 = rng.integers(0, n)
        mu2 = rng.integers(0, n)
        sigma1 = rng.uniform(0.01 * n, 0.2 * n)
        sigma2 = rng.uniform(0.01 * n, 0.2 * n)

        idx = np.arange(n, dtype=np.float64)
        g1 = np.exp(-0.5 * ((idx - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((idx - mu2) / sigma2) ** 2)
        seq = w1 * g1 + w2 * g2
        s = seq.sum()
        if s == 0.0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] GaussMix w1={w1:.3f}"
                  f" μ1={mu1} σ1={sigma1:.1f} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 12000                         # substantially more samples
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration in log‑space, allowing extremely sparse draws
        logc = rng.uniform(-3.5, 0.8)       # c ≈ 0.003 … 6.3
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f}"
                  f" score={sc:.6f}")
    return best_seq, best_score

# 4️⃣ Adam‑style projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _adam_projected_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam PGD iter={t:,}"
                  f" best={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # Adam update (descent direction = -grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.05 / (np.sqrt(v_hat) + eps)   # base learning‑rate

        a_new = a - step * m_hat
        # projection onto simplex guarantees feasibility
        a = _project_simplex(a_new)

        # early exit if we have already reached the target bound
        if best_score <= 1.503:
            break

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that (almost) minimises
    ``evaluate_sequence``.  The routine consists of:

    1.  A dense sweep of a new “dual‑exponential” family (both halves
        decreasing away from the centre) – this family aligns the
        sequence with its reverse to suppress the dominant centre peak.
    2.  Random Gaussian‑mixture draws (flexible two‑bump shapes).
    3.  A large set of random Dirichlet initialisers.
    4.  An Adam‑style projected gradient descent that refines the best
        candidate found so far.

    The best vector (normalised to sum = 1) is returned as a plain list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – the original challenge used n = 4096
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Dual‑exponential sweep (most time‑consuming but very promising)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 300.0   # reserve ~5 min for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_dual_exponential(
            n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 2️⃣ Random Gaussian‑mixture draws
    # ------------------------------------------------------------------
    if time.time() < phase_deadline:
        best_seq, best_score = _random_gaussian_mix_trials(
            n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 3️⃣ Random Dirichlet draws (much larger pool than the previous code)
    # ------------------------------------------------------------------
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 4️⃣ Adam‑style projected gradient descent (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam‑PGD")
        best_seq, best_score = _adam_projected_gradient_descent(
            best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End Adam‑PGD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise once more (numerical safety) and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
