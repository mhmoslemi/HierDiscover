    best_local = a.copy()
    best_score = _objective_score(a)
    it = 0

        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        score = 2 * n * max_conv

            best_local = a.copy()

"""
Improved search for a step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines a very fast parametric sweep with many
randomised restarts of an entropy‑mirrored gradient descent on the
simplex.  Each restart begins from a Dirichlet draw (different
concentration) and performs a multiplicative (exponential) update
that is guaranteed to stay on the simplex.  A back‑tracking line‑search
ensures monotone decrease of the objective.  The best sequence found
throughout the allotted 1000 s is returned.

The two main ingredients are:

* _subgradient_maxconv – exact sub‑gradient of max(self‑convolution).
* _entropy_descent    – mirror‑descent on the simplex using the KL‑
                       divergence (i.e. an exponential move in the
                       original variables).

Both are cheap (O(n log n) per iteration) and allow many thousands of
iterations within the time budget, giving a good chance to break past
the previous bound 1.5078 and reach the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1.
    This is the exact value that `evaluate_sequence` will compute
    (apart from the final normalisation of the sum).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    The returned vector has the same length as a.
    """
    n = a.shape[0]
    max_val = conv.max()
    # active shifts where the max is attained (within a tiny tolerance)
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]               # contribution from the other factor
    if active.size:
        grad = 2.0 * grad / active.size         # factor 2 from symmetry
    return grad


# ----------------------------------------------------------------------
# Entropy‑mirrored gradient descent (exponential update)
# ----------------------------------------------------------------------
def _entropy_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     n: int,
                     global_best_seq: np.ndarray,
                     global_best_score: float,
                     target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Perform mirror descent with KL‑divergence on the simplex.
    a0 must already be normalised (sum=1, non‑negative).
    Returns the best sequence (and its score) encountered during this run.
    """
    # initialise log‑variables (soft‑max representation)
    x = np.log(a0 + 1e-30)          # prevents log(0)
    a = a0.copy()
    best_local_seq = a.copy()
    best_local_score = _objective_score(a)

    it = 0
    # a small safety margin to keep the outer loop responsive
    while time.time() < deadline:
        it += 1
        # current convolution and objective
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())

        # keep the best seen locally (also handy for early exit)
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            # report improvement
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Descent it={it:,} score={cur_score:.6f}")

        # reached the target – can stop early
        if best_local_score <= target:
            break

        # sub‑gradient of the *max* convolution
        grad_max = _subgradient_maxconv(a, conv)            # ∂ max(conv)/∂a
        grad_obj = 2.0 * n * grad_max                       # ∂ objective/∂a

        # centred gradient for the exponential map (entropy mirror)
        avg = np.dot(a, grad_obj)                           # scalar = ⟨a,∇⟩
        gx = a * (grad_obj - avg)                           # ∇_x f(x)

        # start with a moderately aggressive step, shrink if needed
        eta = 0.8 / np.sqrt(it)     # simple diminishing schedule
        while eta > 1e-9:
            # exponential move in the original variables
            x_new = x - eta * gx
            # recompute a via soft‑max (numerically stable)
            x_shift = x_new - np.max(x_new)
            a_new = np.exp(x_shift)
            a_new_sum = a_new.sum()
            if a_new_sum == 0:
                # numerical disaster – fall back to uniform temporary
                a_new = np.full(n, 1.0 / n)
            else:
                a_new /= a_new_sum

            # evaluate the candidate
            conv_new = _conv_self(a_new)
            new_score = float(2.0 * n * conv_new.max())
            if new_score <= cur_score:
                # accept the move
                x = x_new
                a = a_new
                break          # exit the inner line‑search
            else:
                # reject – shorten the step
                eta *= 0.5

        # if we could not make any progress, inject a tiny random jitter
        if eta <= 1e-9:
            eps = 1e-7
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
            x = np.log(a + 1e-30)

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# A very fast parametric sweep (geometric + power, extended ranges)
# ----------------------------------------------------------------------
def _sweep_geometric_extended(n, best_seq, best_score,
                              deadline, start_time):
    """Scanning r^i with r very close to 1 (eps down to 1e‑6)."""
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-6, -1, 200)          # finer near zero
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0:
            continue
        geo /= geo_sum
        # blend with uniform – a dense grid of mixing coefficients
        for lam in np.linspace(0.0, 1.0, 21):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power_extended(n, best_seq, best_score,
                          deadline, start_time):
    """Power law a_i ∝ (i+1)^p, p in a broader interval."""
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-6.0, 2.0, 241):      # finer grid
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence.  The routine mixes a short parametric sweep with many
    entropy‑mirrored gradient‑descent restarts.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    n = 4096
    uniform = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # initialise the global best
    # ------------------------------------------------------------------
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Baseline uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (previous best if available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Brief parametric sweep (≈ 30 s)
    # ------------------------------------------------------------------
    phase_deadline = min(deadline, start_time + 30.0)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric_extended(
            n, best_seq, best_score, phase_deadline, start_time)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power_extended(
            n, best_seq, best_score, phase_deadline, start_time)

    # ------------------------------------------------------------------
    # 2️⃣ Randomised entropy‑descent restarts
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    # allocate the rest of the budget to the descent phase
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 15.0:        # need a few seconds per run
            break

        # draw a Dirichlet‑type start with a random concentration
        logc = rng.uniform(-2.0, 0.7)          # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a0 = rng.gamma(shape=c, scale=1.0, size=n)
        if a0.sum() == 0:
            continue
        a0 /= a0.sum()

        # give each run approx 12 s (or the remaining time, whichever smaller)
        run_deadline = time.time() + min(12.0, remaining)

        seq, sc = _entropy_descent(
            a0,
            run_deadline,
            start_time,
            n,
            best_seq,
            best_score,
            target=1.503)

        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best score={sc:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣ (Optional) final very‑light simulated‑annealing cleanup
    # ------------------------------------------------------------------
    if time.time() < deadline and best_score > 1.503:
        # a few thousand cheap moves with a generous temperature
        a = best_seq.copy()
        cur_score = best_score
        rng = np.random.default_rng()
        total = deadline - time.time()
        start_sa = time.time()
        it = 0
        while time.time() < deadline:
            it += 1
            # linear cooling – start at ~0.5 % of the current score
            T = (1.0 - (time.time() - start_sa) / total) * cur_score * 0.005
            # propose: move a random fraction from a source to a destination
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst:
                continue
            frac = rng.random() * 0.3               # up to 30 % transfer
            eps = a[src] * frac
            if eps < 1e-12:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            # evaluate
            conv = _conv_self(a_prop)
            new_score = float(2.0 * n * conv.max())
            delta = new_score - cur_score
            if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
                a = a_prop
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SA it={it:,} "
                          f"score={best_score:.6f}")
                if best_score <= 1.503:
                    break

    # ------------------------------------------------------------------
    # Normalise and return the final candidate
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
