"""
Propose a non‑negative step function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm differs from the previous one: it performs a sequence of
linear‑programming updates.  For the current candidate `f` we compute the
current worst convolution value `M = max(f*f)`.  We then solve the LP

    maximise    ∑ g_j
    subject to  g ≥ 0
                (f * g)[k] ≤ M    for a selected set of shifts k

which is linear in the unknown vector `g`.  The solution contains a lot of
mass on positions where `f` is small, thereby allowing a convex combination

    f ← (1‑t)·f + t·g

to strictly reduce the maximal self‑convolution.  After a number of such
LP‑updates we finish with a short hill‑climb (move a tiny amount of mass
from the “over‑represented’’ index to the “under‑represented’’ one).  The
whole search respects a 1000 s wall‑clock budget.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Linear‑programming update
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray,
             max_val: float,
             active_shifts: np.ndarray,
             t: float,
             rng: np.random.Generator) -> np.ndarray:
    """
    Solve the LP maximise ∑ g_j  subject to  g≥0  and (f*g)[k] ≤ max_val
    for every shift k in active_shifts.  Return the convex combination
    (1‑t)·f + t·g (renormalised to sum‑1).
    """
    n = f.shape[0]
    nr = len(active_shifts)

    # Build sparse constraint matrix A such that A·g ≤ max_val
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_shifts):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # vectorised construction
        j = np.arange(lo, hi + 1, dtype=int)
        i = k - j
        coeff = f[i]                # coefficients are the entries of f
        nz = coeff != 0.0
        rows.extend([row] * np.count_nonzero(nz))
        cols.extend(j[nz].tolist())
        data.extend(coeff[nz].tolist())

    A = sparse.csr_matrix((data, (rows, cols)), shape=(nr, n))
    b = np.full(nr, max_val, dtype=float)

    # Linear program: maximise sum(g)  → minimise -∑g
    c = -np.ones(n, dtype=float)

    # Solve with HiGHS (fast and handles sparse A)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=(0, None),
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        # If the LP fails (unlikely) just return the original f
        return f.copy()

    g = res.x
    # Normalise g to have the same L1‑mass as f (which is 1)
    sum_g = g.sum()
    if sum_g == 0:
        return f.copy()
    g = g / sum_g

    # Convex combination
    new_f = (1.0 - t) * f + t * g
    # Re‑project to the simplex to eliminate any tiny drift
    new_f = _project_simplex(new_f)
    return new_f


# ----------------------------------------------------------------------
# Hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb that repeatedly moves a tiny fraction of mass from the
    index with the largest sub‑gradient to the one with the smallest.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.01 + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising the evaluation function.
    The routine mixes a linear‑programming based descent with a final
    hill‑climb, respecting the 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin for the caller

    # --------------------------------------------------------------
    # Problem size – keep it moderate for the LP.
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initialise from a global starter if available, otherwise uniform.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur_f = starter.copy()
            else:
                cur_f = uniform.copy()
        else:
            cur_f = uniform.copy()
    except Exception:
        cur_f = uniform.copy()

    cur_f = _project_simplex(cur_f)
    best_seq = cur_f.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – LP‑based descent (major part of the budget).
    # --------------------------------------------------------------
    lp_deadline = deadline - 120.0   # keep ~2 min for the final hill‑climb
    t = 0.2                         # initial mixing factor
    iteration = 0

    while time.time() < lp_deadline:
        iteration += 1
        conv = _conv_self(cur_f)
        M = conv.max()

        # pick a modest active set (centre + the top few peaks)
        centre = n - 1                     # zero‑shift index
        # we always include the centre
        # then add a few largest other shifts
        top_k = 20
        # argsort descending
        ordered = np.argsort(-conv)
        active = [centre]
        added = 0
        for k in ordered:
            if k == centre:
                continue
            active.append(k)
            added += 1
            if added >= top_k:
                break
        active_shifts = np.array(active, dtype=int)

        # one LP step
        new_f = _lp_step(cur_f, M, active_shifts, t, rng)

        # adapt the step size – if the bound got smaller we can be bolder
        new_score = _objective_score(new_f)
        if new_score < best_score:
            best_seq = new_f.copy()
            best_score = new_score
            cur_f = new_f
            # slowly increase t (up to 0.5) to accelerate when we are improving
            t = min(0.5, t * 1.03)
            print(f"[{time.time() - start_time:.2f}s] LP it={iteration} score={best_score:.6f}")
        else:
            # no improvement – cut the step size and jitter a bit
            t = max(0.05, t * 0.7)
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = cur_f[i1] * (rng.random() * 1e-5 + 1e-9)
            cur_f[i1] -= eps
            cur_f[i2] += eps
            cur_f = _project_simplex(np.maximum(cur_f, 0.0))

    # --------------------------------------------------------------
    # Phase 2 – final hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, deadline, best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
