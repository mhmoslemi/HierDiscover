def _adam_subgrad(a0, steps, lr, beta1, beta2, L_fft, best_seq, best_score, start, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for it in range(1, steps+1):
        if time.time() >= deadline: break
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # optional print
        # compute subgradient of max conv:
        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad
        # adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        # optionally adapt lr schedule
    return best_seq, best_score

def _hill_climb(a0, start, deadline):
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2*n-1)
    cur_score = _objective_score(a) # 2*n*max(conv)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # Choose src with large a and dst with small a
        src = np.argmax(a)
        dst = np.argmin(a)
        # random fraction
        frac = rng.random() * 0.01 + 1e-5  # up to 1%
        eps = a[src] * frac
        if eps <= 0: continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # evaluate
        conv = _conv_self(a_prop, L_fft)
        new_score = 2.0 * n * conv.max()
        if new_score < cur_score - 1e-12:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a lightweight five‑stage pipeline that works within the
1000 s time limit:

1️⃣  Very coarse sweep over a few analytic families (geometric,
    power, beta, stretched‑exp, logistic, sine‑power).

2️⃣  Massive random Dirichlet sampling (as many draws as time permits).

3️⃣  Multiple restarts of Adam optimisation on the exact max‑convolution
    (sub‑gradient + Adam moments, projection onto the simplex).

4️⃣  Aggressive hill‑climb / simulated‑annealing that moves a small
    fraction of mass from a high‑gradient index to a low‑gradient one.

5️⃣  Normalisation and return as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _objective_score(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) given sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# 1️⃣  Very coarse parametric sweep
# ----------------------------------------------------------------------
def _coarse_sweep_families(n: int,
                           best_seq: np.ndarray,
                           best_score: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # Uniform baseline
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # Geometric (r = 1‑eps)
    eps_grid = np.logspace(-5, -1, 30)               # fewer points
    lam_grid = np.linspace(0.0, 1.0, 5)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in lam_grid:
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # Power family a_i ∝ (i+1)^p
    p_grid = np.linspace(-2.0, 2.0, 41)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # Beta family (α,β)
    alphas = np.linspace(0.05, 4.0, 25)
    betas = np.linspace(0.05, 4.0, 25)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # Stretched‑exponential
    scales = np.linspace(0.25, 1.0, 5)
    p_vals = np.linspace(0.2, 5.0, 20)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretch p={p:.2f} sc={scale:.2f} score={sc:.6f}")

    # Logistic
    mu_factors = np.linspace(0.2, 0.8, 7)
    s_factors = np.linspace(0.01, 0.2, 7)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # Sine‑power
    p_vals = np.linspace(0.1, 5.0, 20)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    count = 0
    # try up to 20000 draws or until time runs out
    while time.time() < deadline and count < 20000:
        # concentration parameter c drawn log‑uniformly
        log_c = rng.uniform(np.log(0.01), np.log(5.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
        count += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation on the exact max‑convolution (sub‑gradient + Adam)
# ----------------------------------------------------------------------
def _adam_subgrad(a0: np.ndarray,
                  n: int,
                  best_seq: np.ndarray,
                  best_score: float,
                  L_fft: int,
                  start_time: float,
                  deadline: float,
                  max_iters: int = 15000) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Adam on the true max‑convolution (sub‑gradient).
    Returns the (potentially improved) sequence, the global best sequence
    seen so far and its score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] AdamSubgrad it={it:,} score={cur_score:.6f}")

        # sub‑gradient of max(conv) at a
        subgrad = _subgradient_maxconv(a, conv)      # ∂max/∂a
        grad = 2.0 * n * subgrad                     # ∂objective/∂a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional learning‑rate decay
        if it % 4000 == 0:
            lr *= 0.7

    return a, best_seq, best_score


def _adam_subgrad_multiple_restarts(n: int,
                                    best_seq: np.ndarray,
                                    best_score: float,
                                    start_time: float,
                                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Run several independent Adam‑subgradient runs (including the current
    best as a restart).  The number of restarts adapts to the remaining
    time budget.
    """
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    restarts = 0
    while time.time() < deadline - 5.0 and restarts < 12:
        # pick a start point
        if restarts == 0:
            a0 = best_seq.copy()
        else:
            # fresh Dirichlet start
            log_c = rng.uniform(np.log(0.01), np.log(5.0))
            c = np.exp(log_c)
            a0 = rng.gamma(shape=c, scale=1.0, size=n)
            a0 /= a0.sum()
        # run Adam sub‑gradient optimisation
        a0, best_seq, best_score = _adam_subgrad(
            a0, n, best_seq, best_score, L_fft,
            start_time, deadline, max_iters=12000
        )
        restarts += 1
    return best_seq, best_score

# 4️⃣  Aggressive hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _aggressive_hill_climb(a0: np.ndarray,
                           n: int,
                           best_seq: np.ndarray,
                           best_score: float,
                           start_time: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    L_fft = _next_pow2(2 * n - 1)
    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_seq, best_score = a.copy(), cur_score

    rng = np.random.default_rng()
    T0 = 0.02
    t0 = time.time()

    while time.time() < deadline:
        # pick a source (must have some mass) and a destination
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[rng.integers(src_candidates.size)]
        dst = rng.integers(n)
        if src == dst:
            continue

        # move a small (but not too tiny) fraction of the source mass
        frac = rng.random() * 0.05 + 1e-6   # up to ~5 %
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * np.exp(- (time.time() - t0) / (deadline - t0))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            accept = rng.random() < np.exp(-Δ / max(T, 1e-12))

        if accept:
            a = a_prop

# Main entry point

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine follows a lightweight five‑stage
    pipeline and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3    # safety margin

    # --------------------------------------------------------------
    # Problem size (feel free to adjust if time permits)
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise with the uniform vector (and a possible external starter)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0    # reserve time for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _coarse_sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet sampling
    # --------------------------------------------------------------
    phase_deadline = deadline - 150.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation on the exact max‑convolution (multiple restarts)
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam‑subgradient restarts")
        best_seq, best_score = _adam_subgrad_multiple_restarts(
            n, best_seq, best_score, start_time, deadline - 80.0
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam restarts best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Aggressive hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting aggressive hill‑climb")
        best_seq, best_score = _aggressive_hill_climb(
            best_seq, n, best_seq, best_score, start_time, deadline
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
