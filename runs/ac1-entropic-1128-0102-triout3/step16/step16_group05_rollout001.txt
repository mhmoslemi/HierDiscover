"""
An annealed‑softmax projected Adam optimizer.

The objective (with sum(a)=1) is  J(a)=2·n·max(a*a) .
`max(a*a)` is non‑smooth, so we replace the max by a softmax with a
temperature τ that is slowly annealed to 0.  The resulting gradient is a
smooth weighted average of the true sub‑gradients of all convolution
entries.  We update the probability vector with Adam and project onto the
simplex after each step.  After the bulk of the budget a short
hill‑climb moves tiny amounts of mass from “over‑represented’’ indices
to “under‑represented’’ ones.

The routine respects a 1000 s wall‑clock limit, prints intermediate
scores and returns the best sequence found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution (real) of two vectors."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Softmax over conv entries with temperature tau."""
    # shift for numerical stability
    shifted = conv - conv.max()
    # avoid underflow when tau is extremely small
    if tau <= 0:
        # hard max – one‑hot at the maximum location
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    ex = np.exp(shifted / tau)
    s = ex.sum()
    if s == 0:
        # degenerate case – fall back to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s


# ----------------------------------------------------------------------
# Optimisation stages
# ----------------------------------------------------------------------
def _adam_step(a: np.ndarray,
               grad: np.ndarray,
               step_scale: float,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One Adam update on the simplex (no projection performed here)."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = -step_scale * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a + step
    # keep feasibility before projection
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_simplex(a_new)
    return a_new, m, v

    Search for a non‑negative step function minimising
        2 * n * max(conv(a, a)) / (sum(a)**2)

    short hill‑climb.  The whole run respects the supplied time budget.
    """

        cur_score = 2.0 * n * max_conv

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0

        else:

        # active‑set sub‑gradient (exact max)

        tol = 1e-12

        grad = np.zeros_like(a)
        for k in active:

            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)

        # source – index with large gradient that still has mass
        src_candidates = np.where(a > 1e-15)[0]

            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination – index with smallest gradient

        if eps <= 0:

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        new_score = 2.0 * n * _conv_self(a_prop).max()
        if new_score < cur_score:
            a = a_prop
            no_improve = 0
        else:
            # after a while without improvement inject a tiny random jitter

                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
