    Full‑mass transfers are attempted first; if they do not improve the
    objective, the transferred fraction is halved repeatedly.

All phases abort automatically when the global deadline is reached.
The best sequence found is normalised to sum = 1 and returned as a plain
Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using an FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used during the search: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweep (original + new families)
# ----------------------------------------------------------------------
def _sweep_families_extended(
    n: int, best_seq: np.ndarray, best_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # ---- uniform baseline -------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- geometric --------------------------------------------------------
    eps_grid = np.logspace(-5, -1, 150)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = r ** idx
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 7):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power -------------------------------------------------------------
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        pow_seq = (idx + 1.0) ** p
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ---------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 41)
    betas = np.linspace(0.05, 4.0, 41)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- stretched‑exponential ---------------------------------------------
    for scale in np.linspace(0.25, 1.0, 9):
        if time.time() >= deadline:
            break
        for p in np.linspace(0.2, 5.0, 31):
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} sc={scale:.2f} score={sc:.6f}")

    # ---- logistic ---------------------------------------------------------
    for mu_f in np.linspace(0.2, 0.8, 11):
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in np.linspace(0.01, 0.2, 9):
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # ---- sine‑power --------------------------------------------------------
    for p in np.linspace(0.1, 5.0, 31):
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    # ==== NEW FAMILY 1: inverse distance from centre =======================
    # a_i ∝ 1 / (|i - c| + eps)^p   (large mass at the ends)
    centre = (n - 1) / 2.0
    for eps in np.logspace(-4, -1, 30):
        if time.time() >= deadline:
            break
        for p in np.linspace(0.5, 8.0, 26):
            if time.time() >= deadline:
                break
            dist = np.abs(idx - centre) + eps
            seq = 1.0 / (dist ** p)
            seq = np.clip(seq, 0, None)
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] InvDist eps={eps:.1e} p={p:.2f} score={sc:.6f}")

    # ==== NEW FAMILY 2: double‑ended exponentials ===========================
    # a_i ∝ exp( - (i/(α n))^p ) + exp( - ((n-1-i)/(α n))^p )
    for scale in np.linspace(0.01, 0.2, 10):
        if time.time() >= deadline:
            break
        for p in np.linspace(0.5, 5.0, 21):
            if time.time() >= deadline:
                break
            left = np.exp(- (idx / (scale * n)) ** p)
            right = np.exp(- ((n - 1 - idx) / (scale * n)) ** p)
            seq = left + right
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] EndExp scale={scale:.3f} p={p:.2f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random sampling (Dirichlet, log‑normal, sparse subsets)
# ----------------------------------------------------------------------
def _random_sampling_all(
    n: int, best_seq: np.ndarray, best_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # --- Dirichlet with scalar concentration ---
    trials = 0
    while time.time() < deadline and trials < 12000:
        trials += 1
        log_c = rng.uniform(np.log(0.01), np.log(50.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")

    # --- Log‑normal (Gaussian in log‑domain) ---
    while time.time() < deadline:
        sigma = rng.uniform(0.1, 2.5)
        mu = 0.0
        log_vals = rng.normal(loc=mu, scale=sigma, size=n)
        a = np.exp(log_vals)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] LogNormal σ={sigma:.2f} score={sc:.6f}")
        # stop early if very short time left
        if time.time() > deadline - 2.0:
            break

    # --- Sparse random subsets (uniform weights on a random support) ---
    while time.time() < deadline:
        k = rng.integers(80, 500)            # support size
        support = rng.choice(n, size=k, replace=False)
        weights = rng.gamma(shape=1.0, scale=1.0, size=k)
        a = np.zeros(n, dtype=np.float64)
        a[support] = weights
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sparse k={k} score={sc:.6f}")
        # safety break
        if time.time() > deadline - 1.0:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adaptive projected gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_adaptive(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with a simple back‑tracking line search.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    it = 0
    # initial step size – empirically a few units works well
    eta0 = 1.0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)          # ∂max/∂a
        grad = 2.0 * n * subgrad                         # ∂obj/∂a

        # back‑track line search (max 6 reductions)
        eta = eta0 / (1.0 + it * 1e-4)   # modest decay
        for _ in range(6):
            cand = a - eta * grad
            cand = _project_simplex(cand)
            cand_score = _objective_score(cand)
            if cand_score < cur_score - 1e-14:
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score, best_seq = cand_score, cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD LS it={it:,} score={cand_score:.6f}")
                break
            else:
                eta *= 0.5

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            jitter = np.random.default_rng().normal(scale=0.002, size=n)
            a = _project_simplex(np.maximum(a + jitter, 0.0))
            cur_score = _objective_score(a)

        if eta0 < 1e-8:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Aggressive hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _aggressive_hill_climb(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Move large fractions of mass from the index with the highest sub‑gradient to
    the one with the lowest. Full‑mass transfers are attempted first; if they
    do not improve the objective the transferred fraction is halved repeatedly.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        subgrad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(subgrad))
        dst = int(np.argmin(subgrad))

        # ensure src actually has something to give
        if a[src] < 1e-12:
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero))

        max_transfer = a[src]

        # try a cascade of fractions, starting with the whole amount
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01]:
            eps = max_transfer * frac
            if eps < 1e-12:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            if a_prop[src] < -1e-12:
                continue
            new_score = _objective_score(a_prop)
            if new_score < cur_score - 1e-12:
                a = a_prop
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] AggHill it={it:,} score={new_score:.6f}")
                break  # accept and move to next iteration

        # occasional random perturbation to shake out of flat regions
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.0015, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
            cur_score = _objective_score(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # safety margin

    # problem size
    n = 4096

    # initialise with uniform distribution
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep (including new families)
    # --------------------------------------------------------------
    phase_deadline = deadline - 600.0   # leave plenty for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families_extended(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Massive random sampling
    # --------------------------------------------------------------
    phase_deadline = deadline - 350.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_sampling_all(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Adaptive projected gradient descent on exact max‑conv
    # --------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time() - start_time:.2f}s] Starting adaptive PGD")
        best_seq, best_score = _pgd_adaptive(
            best_seq, start_time, deadline - 200.0, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After PGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Aggressive hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting aggressive hill‑climb")
        best_seq, best_score = _aggressive_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
