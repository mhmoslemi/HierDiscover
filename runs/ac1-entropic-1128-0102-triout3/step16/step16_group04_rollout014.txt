"""
A fast local‑search optimiser for the “max‑convolution” score.

The objective (with a normalised sum Σa_i = 1) is

    score = 2·n·max_k (a * a)[k]               (1)

where “*” is ordinary linear convolution.  The algorithm works in three
stages:

0️⃣  Initialise – use a uniform vector, a global starter (if present) and a
    large set of random Dirichlet / sparse draws to obtain a decent start.

1️⃣  Local descent – repeatedly locate the index k where the convolution is
    maximal, identify the pairs (i , j) with i+j=k that contribute most to
    that peak and move a fraction of the larger partner to a random other
    coordinate.  The move is accepted only if it strictly reduces the
    score.  When no improvement is found for a while a small random jitter
    is performed to escape plateaus.

2️⃣  Refinement – a short convex‐mixing sweep with the uniform distribution
    followed by a cheap random‑pair hill‑climb.

All work is performed on the simplex (the vector is kept non‑negative and
renormalised to sum = 1 after each move).  The routine is time‑aware and
stops just before the allotted budget (default 1000 s).  The returned list
contains plain Python floats and satisfies the constraints of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)−1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator (sum(a) is assumed to be 1)."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())

                a = sample / sample.sum()
                iters_without_improve = 0
                print(f"[{time.time()-start_time:.2f}s] Random restart (Dirichlet)")

    # --------------------------------------------------------------
    # 2️⃣  Final refinement – mixing with uniform + cheap hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        uniform = np.full(n, 1.0 / n)

        # convex mixing sweep
        for lam in np.linspace(0.0, 1.0, 31):
            if time.time() >= deadline:
                break
            cand = lam * best_a + (1.0 - lam) * uniform
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _score(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time()-start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

        # light random‑pair hill‑climb
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = min(best_a[i], 0.005) * rng.random()
            if delta <= 0.0:
                continue
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = _score(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time()-start_time:.2f}s] Hill‑climb i={i} j={j} δ={delta:.6f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # Normalise & return a plain Python list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
