def _proj_simplex(v):
    """Euclidean projection of v onto simplex {x>=0, sum x = 1}."""
    # algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimizer for the step‑function problem (improved).

Main ideas
----------
1️⃣  Initialise with a variety of seeds (uniform, Dirichlet draws, several
    Beta‑shaped densities, and the optional global starter).

2️⃣  Euclidean projected‑gradient descent (mirror descent on the simplex
    with an L2‑projection).  The sub‑gradient of the max‑convolution is
    computed analytically, the step‑size decays as 1/√t and occasional
    random perturbations are added when the progress stalls.

3️⃣  Simulated‑annealing hill‑climbing on the simplex.  Moves transfer a
    (small or moderate) amount of mass from a donor index to a receiver
    index; the move is accepted if it improves the objective or with a
    temperature‑scaled probability.  Occasionally a “swap‑shuffle’’ of a
    few top entries is performed to explore different spatial orderings.

4️⃣  Fine mixing with the uniform distribution – a golden‑section search
    over λ∈[0,1] for the convex combination λ·a+(1−λ)·uniform.

The routine respects the total time budget (default 1000 s) and
returns the best sequence (as a plain list of floats) found within the
budget.  All numbers are non‑negative and the sequence is normalised to
sum 1 before returning.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat everything within a tiny tolerance as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution a_{k‑i}
    if active.size:
        grad = 2.0 * grad / active.size   # factor 2 because each term appears twice
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Return the evaluation score (without the denominator, sum is 1)."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n          # mid‑point of each bin
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x≥0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Optional global starter
# ----------------------------------------------------------------------
def _maybe_use_global_starter(n: int) -> tuple[np.ndarray, float]:
    """If a global variable `height_sequence_1` exists, try to use it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            total = starter.sum()
            if total > 0.0:
                starter = starter / total
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


# ----------------------------------------------------------------------
# 1️⃣  Broad random initialisation
# ----------------------------------------------------------------------
def _initial_population(n: int,
                        best_seq: np.ndarray,
                        best_score: float,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """Generate many cheap candidates (Dirichlet + a few Beta shapes)."""
    rng = np.random.default_rng()
    beta_params = [(0.2, 0.2), (0.3, 0.3), (0.5, 0.5),
                   (0.2, 0.8), (0.8, 0.2), (0.3, 0.7), (0.7, 0.3)]
    beta_seeds = [_beta_shape(a, b, n) for a, b in beta_params
                  if _beta_shape(a, b, n) is not None]

    while time.time() < deadline:
        # Dirichlet‑like draw (Gamma iid, normalised)
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = math.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        if sample.sum() == 0.0:
            continue
        a = sample / sample.sum()
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

        # occasional Beta seed
        if beta_seeds and rng.random() < 0.005:
            v = beta_seeds[rng.integers(len(beta_seeds))]
            sc = _objective_score(v, n)
            if sc < best_score:
                best_score, best_seq = sc, v.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta seed → score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Euclidean projected gradient descent
# ----------------------------------------------------------------------
def _proj_grad_descent(a0: np.ndarray,
                       n: int,
                       L_fft: int,
                       start_time: float,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex.
    Step size η = η0 / sqrt(t).  When stagnation is detected a tiny random
    perturbation is added.
    """
    a = a0.copy()
    rng = np.random.default_rng()
    it = 0
    last_improve = 0
    eta0 = 0.8          # initial step‑size factor

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score, best_seq = cur_obj, a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] GD it={it:,} score={cur_obj:.6f}")

        # sub‑gradient of the max‑convolution (for max(conv)); scale by 2n
        grad = _subgradient_maxconv(a, conv)
        grad = grad * (2.0 * n)

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        a_new = a - eta * grad
        a_new = _proj_simplex(a_new)

        # evaluate new point
        conv_new = _conv_self(a_new, L_fft)
        new_obj = 2.0 * n * conv_new.max()

        # accept if improvement, otherwise keep current (no acceptance of uphill)
        if new_obj < cur_obj:
            a = a_new
        else:
            # simple restart mechanism after long stall
            if it - last_improve > 200_000:
                perturb = rng.normal(scale=0.05, size=n) * a
                a = np.maximum(a + perturb, 0.0)
                a = _proj_simplex(a)
                last_improve = it

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------
def _annealing_hill_climb(a0: np.ndarray,
                          n: int,
                          L_fft: int,
                          start_time: float,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float) -> tuple[np.ndarray, float]:
    """
    Small random mass‑transfer moves plus occasional shuffle of the top‑k
    entries.  A temperature that slowly decays is used to accept uphill
    steps.
    """
    a = a0.copy()
    cur_obj = 2.0 * n * _conv_self(a, L_fft).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ---- mass transfer move -------------------------------------------------
        i = rng.integers(n)                     # donor
        j = rng.integers(n - 1)                 # receiver (different from i)
        if j >= i:
            j += 1
        if a[i] <= 1e-14:
            continue

        # transfer a fraction (up to 20 % of donor) – allows larger jumps
        eps = min(0.2 * a[i], rng.random() * a[i])
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # evaluate
        new_obj = 2.0 * n * _conv_self(a_new, L_fft).max()
        delta = new_obj - cur_obj

        # temperature schedule (linear decay)
        remain = max(0.0, deadline - time.time())
        total = max(0.1, deadline - start_time)
        T = 0.01 * (remain / total)       # starts ~0.01, ends ~0

        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            a = a_new
            cur_obj = new_obj
            if new_obj < best_score:
                best_score, best_seq = new_obj, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improved score={new_obj:.6f}")

        # ---- occasional shuffle of the largest entries -------------------------
        if rng.random() < 0.001:
            # select a modest set of the biggest values
            k = max(2, n // 200)            # ≈0.5 % of entries
            idx = np.argpartition(-a, k)[:k]   # indices of the k largest entries
            perm = rng.permutation(k)
            a_shuf = a.copy()
            a_shuf[idx] = a[idx[perm]]
            # keep the shuffled version only if it does not worsen the score
            shuf_obj = 2.0 * n * _conv_self(a_shuf, L_fft).max()
            if shuf_obj < cur_obj:
                a = a_shuf
                cur_obj = shuf_obj
                if shuf_obj < best_score:
                    best_score, best_seq = shuf_obj, a_shuf.copy()
                    print(f"[{time.time() - start_time:.2f}s] Shuffle improved score={shuf_obj:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Optimise λ∈[0,1] for seq = λ·a + (1−λ)·uniform using a few
    golden‑section evaluations.
    """
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0

    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered within the time
    budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.25          # small safety margin

    # Problem size – a power of two works nicely with the FFT.
    # Using a larger n than before often gives a little more freedom.
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣ Initialise (uniform baseline + optional global starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Broad random search (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_dead = start + 0.15 * (deadline - start)
    if time.time() < phase1_dead:
        best_seq, best_score = _initial_population(
            n, best_seq, best_score, phase1_dead, start)

    # ------------------------------------------------------------------
    # 2️⃣ Euclidean projected gradient descent (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase2_dead = start + 0.70 * (deadline - start)
    if time.time() < phase2_dead:
        print(f"[{time.time() - start:.2f}s] Starting projected‑gradient descent")
        best_seq, best_score = _proj_grad_descent(
            best_seq, n, L_fft, start, phase2_dead, best_seq, best_score)

    # ------------------------------------------------------------------
    # 3️⃣ Simulated‑annealing hill‑climbing (≈20 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start:.2f}s] Starting hill‑climb (annealing)")
        best_seq, best_score = _annealing_hill_climb(
            best_seq, n, L_fft, start, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # 4️⃣ Fine mixing with the uniform distribution (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start, best_seq, best_score)

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]
