def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0
    tau_factor = 1.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max over convolution values
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max (same as in the original script)
        a_rev = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(w, n=_next_pow2(3 * n - 2)) *
                              np.fft.rfft(a_rev, n=_next_pow2(3 * n - 2)),
                              n=_next_pow2(3 * n - 2))[:n]
        grad = 2.0 * conv_w

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)
    return best_seq, best_score

# ----------------------------------------------------------------------
def _pgd_descent(a0: np.ndarray,
                 start_time: float,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """
    Classic projected (onto the probability simplex) gradient descent
    using the exact sub‑gradient of max(conv(a,a)).  The step size
    decays slowly and occasional random perturbations help escape
    plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.5          # initial step size (will be decayed)
    it = 0
    no_improve = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        subgrad = _subgradient_maxconv(a, conv)          # ∂max/∂a
        grad = 2.0 * n * subgrad                         # ∂obj/∂a

        # decaying step size
        eta_t = eta / (1.0 + it * 1e-4)

        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-12:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            # slowly shrink the step if we are not improving
            eta *= 0.995
            # occasional random shake to get out of flat regions
            if no_improve % 500 == 0:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.001, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                a = a_jit
                cur_score = _objective_score(a)

        if eta < 1e-8:
            # step size too tiny – stop early
            break

    return best_seq, best_score

# 5️⃣  Final hill‑climb / simulated‑annealing (unchanged)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score
