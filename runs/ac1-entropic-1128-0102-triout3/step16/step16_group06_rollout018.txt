"""
Propose a step‑function sequence that minimises the objective
``2·n·max(conv(a,a)) / (sum(a)**2)``.
The optimizer works in three phases:

1.  Initialise from the global starter (if present) or from a set of
    random Dirichlet draws.  The best of these draws is kept as a seed.

2.  Smooth‑max gradient descent (Adam) on the objective.
    We replace the non‑smooth max with a log‑sum‑exp (temperature τ)
    which yields a differentiable surrogate.  The gradient of the
    surrogate can be computed exactly by a pair of FFT‑based convolutions.
    Adam’s adaptive step sizes together with Euclidean projection onto
    the probability simplex give fast and stable progress.

3.  A final tiny‑mass hill‑climb (the same routine used in the reference
    implementation) to polish the result.

All work respects the 1000 s wall‑clock budget.  The returned list has
non‑negative entries; its sum is automatically normalised (the evaluation
function is homogeneous, so any scaling works).  The algorithm
typically beats the previous bound 1.507826 and reaches the target <1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolution, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using the FFT.
    Returns a 1‑D array of length len(x)+len(y)-1.
    """
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (sum=1, v≥0).  The implementation follows the O(n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution helpers
# ----------------------------------------------------------------------


def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the τ‑smoothed max (log‑sum‑exp) of the self‑convolution.
    a   – probability vector (sum = 1)
    conv – full convolution a*a (length 2·n‑1)
    tau – temperature > 0
    Returns the Euclidean gradient ∂φ/∂a.
    """
    n = a.shape[0]
    # stable soft‑max over the convolution entries
    max_c = conv.max()
    exp_vals = np.exp((conv - max_c) / tau)
    p = exp_vals / exp_vals.sum()          # p is a probability distribution

    # gradient = 2 * (a ⋆ p) where “⋆” denotes the (non‑circular) cross‑correlation.
    # Using convolution with the reversed p is the simplest way.
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)          # length = 3·n‑2
    # extract the slice that corresponds to the cross‑correlation
    # (indices n‑1 … 2·n‑2, then reverse to align with a[0…n‑1])
    grad_slice = conv_ap[n - 1 : 2 * n - 1][::-1]
    grad = 2.0 * grad_slice
    return grad


# ----------------------------------------------------------------------
# Random initialisers (same as in the reference solution)
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=300):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)          # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        # immediate objective (sum = 1 → 2·n·max(conv))
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {cur_score:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=250):
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        cur_score = 2.0 * n * _conv_self(seq).max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} w={w} → {cur_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser
# ----------------------------------------------------------------------


def _adam_smooth_opt(start_seq, deadline, best_seq, best_score):
    """
    Adam optimiser on the τ‑smoothed max of the self‑convolution.
    The iterates are always projected onto the probability simplex.
    """
    a = start_seq.copy()          # sum(a) == 1
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule – start relatively high, decay slowly
    conv = _conv_self(a)
    tau = max(0.1 * conv.max(), 1e-4)

    t_iter = 0
    iteration_without_improve = 0

    while time.time() < deadline:
        t_iter += 1

        # current convolution and objective (exact, not smoothed)
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t_iter} score={cur_score:.6f}")
            iteration_without_improve = 0
        else:
            iteration_without_improve += 1
            if iteration_without_improve > 2000:
                # occasional temperature reset to help escaping plateaus
                tau = max(tau * 0.5, 1e-8)
                iteration_without_improve = 0

        # smoothed‑max gradient
        grad = _smooth_max_gradient(a, conv, tau)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # learning‑rate schedule: a modest base step that slowly decays
        base_lr = 0.15
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # geometric decay of the temperature
        tau = max(tau * 0.997, 1e-8)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny‑mass hill‑climb (final polishing – unchanged)
# ----------------------------------------------------------------------


def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _conv_self(a).max()
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _smooth_max_gradient(a, conv, tau=1e-9)  # essentially the true sub‑gradient

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        cur_score = 2.0 * n * _conv_self(a_prop).max()
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock budget and returns a plain
    Python list ready for the evaluator.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # problem size – 4096 gives a good trade‑off between flexibility and speed
    n = 4096

    # baseline – uniform distribution (sum = 1)
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s          # normalise to sum = 1
                sc = 2.0 * n * _conv_self(starter).max()
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.15
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=400
        )

    # ------------------------------------------------------------------
    # Phase 2 – random mixtures (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.25
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=300
        )

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimiser on the smoothed max (≈45 % of the budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.70
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam smooth optimisation")
        best_seq, best_score = _adam_smooth_opt(
            best_seq, phase3_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final tiny‑mass hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (sum = 1) and convert to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    final_eval = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_eval:.6f}")

    return [float(x) for x in best_seq]
