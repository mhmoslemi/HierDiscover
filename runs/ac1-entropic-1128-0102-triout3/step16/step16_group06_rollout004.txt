"""
Fast gradient‑based optimisation for the step‑function “max‑convolution”
problem.

The algorithm works in three phases:

1.  Random Dirichlet draws (few seconds) → a good initial simplex point.
2.  Smooth “log‑sum‑exp” approximation of the max‑convolution, minimised
    by Adam on the soft‑max (log‑probability) parametrisation of the simplex.
    The smoothing parameter β is increased continuously (continuation method)
    so that the optimiser first sees a very smooth landscape and later
    concentrates on the true ℓ∞‑objective.
3.  A short tiny‑mass hill‑climb for final polishing.

All phases respect a global 1000 s wall‑clock budget and only use NumPy
(FFT based convolutions) – no external solvers are required.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length L.
    Returns a real array of length 2*len(a)-1.
    """
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv.real


def _softmax(w: np.ndarray) -> np.ndarray:
    """Stable soft‑max → a probability vector (sum = 1, a≥0)."""
    w_max = np.max(w)
    e = np.exp(w - w_max)
    e_sum = e.sum()
    return e / e_sum


def _objective_score(a: np.ndarray) -> float:
    """Exact score 2·n·max(conv(a,a)) for a on the simplex."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    conv = _conv_self_fft(a, L)
    return float(2.0 * n * conv.max())


def _smooth_gradient(a: np.ndarray, beta: float, L: int) -> np.ndarray:
    """
    Gradient of the smooth objective
        fβ(a) = (1/β)·log Σ_k exp(β·conv_k)
    w.r.t. a (a lies on the simplex).  Implements the formula

        g = 2 * h,   h_i = Σ_j a_j·p_{i+j}

    where p is the normalised soft‑max of β·conv.
    """
    n = a.shape[0]

    # 1) compute the (full) self‑convolution
    conv = _conv_self_fft(a, L)                     # length 2n‑1

    # 2) soft‑max weights over the convolution entries
    exp_beta = np.exp(beta * conv - np.max(beta * conv))   # stability
    p = exp_beta / exp_beta.sum()                    # length 2n‑1, sums to 1

    # 3) compute h = Σ_j a_j·p_{i+j} for all i
    #    This is a linear convolution of a_rev and p, then sliced.
    a_rev = a[::-1]                                 # length n
    L2 = _next_pow2(len(a_rev) + len(p) - 1)        # ≥ 3n‑2
    A_rev_f = rfft(a_rev, n=L2)
    P_f = rfft(p, n=L2)
    conv_rev = irfft(A_rev_f * P_f, n=L2)[: len(a_rev) + len(p) - 1]  # length 3n‑2
    offset = n - 1
    h = conv_rev[offset: offset + n]               # length n
    grad_a = 2.0 * h
    return grad_a


def _hill_climb(start_a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Final local optimisation: move a tiny amount of mass from a
    relatively “bad” coordinate to a “good” one, keeping the vector
    on the simplex.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        # evaluate gradient of the true max‑convolution (sub‑gradient)
        L = _next_pow2(2 * n - 1)
        conv = _conv_self_fft(a, L)
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient (average over active shifts)
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        # pick a source (large gradient) and a destination (small gradient)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        eps = a[src] * (rng.random() * 0.004 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _softmax(np.log(np.maximum(a, 1e-30)))
                no_improve = 0

    return best_a

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the global time budget (≈1000 s) and returns a
    plain Python list ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate dimension gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – quick Dirichlet exploration (≈5 % of budget)
    # ------------------------------------------------------------------
    phase0_end = start_time + budget_s * 0.05
    while time.time() < phase0_end:
        conc_log = rng.uniform(-2.0, 0.7)          # concentration ≈ 0.01 … 5
        conc = 10 ** conc_log
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={conc:.3f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth Adam optimisation with continuation on β.
    # ------------------------------------------------------------------
    phase1_end = start_time + budget_s * 0.80   # most of the budget
    print(f"[{time.time() - start_time:.2f}s] Starting smooth Adam optimisation")

    # initialise Adam on the log‑probability parametrisation
    w = np.log(np.maximum(best_seq, 1e-30))
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.12                         # base learning rate
    it = 0
    beta_start, beta_end = 5.0, 2000.0

    # pre‑compute FFT lengths once
    L_conv = _next_pow2(2 * n - 1)      # for self‑convolution
    L_grad = _next_pow2(3 * n - 2)      # for gradient convolution (a_rev * p)

    while time.time() < phase1_end:
        it += 1
        # ---- current simplex point ----
        a = _softmax(w)                # sum = 1, a_i≥0

        # ---- exact objective (for book‑keeping) ----
        conv = _conv_self_fft(a, L_conv)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={cur_score:.6f}")

        # ---- continuation schedule for β ----
        elapsed = time.time() - start_time
        progress = (elapsed) / (phase1_end - start_time)
        beta = beta_start + (beta_end - beta_start) * progress

        # ---- smooth gradient w.r.t. a ----
        grad_a = _smooth_gradient(a, beta, L_grad)   # size n

        # ---- convert to gradient w.r.t. w (soft‑max parametrisation) ----
        avg = np.dot(a, grad_a)
        grad_w = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        w -= step

        # occasional tiny random jitter to avoid stagnation
        if it % 4000 == 0:
            i1, i2 = rng.choice(n, size=2, replace=False)
            jitter = a[i1] * (rng.random() * 0.005 + 1e-6)
            w[i1] -= jitter
            w[i2] += jitter

    print(f"[{time.time() - start_time:.2f}s] End of Adam → best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final tiny‑mass hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq = _hill_climb(best_seq, deadline)
        best_score = _objective_score(best_seq)
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
