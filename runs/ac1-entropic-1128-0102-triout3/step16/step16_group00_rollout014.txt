"""
Optimisation of the objective

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the full linear self‑convolution and all entries of ``a`` must be
non‑negative.  The optimum is known to be slightly below the naïve uniform
distribution (which gives a value of 2).  The algorithm below attempts to beat the
previous best (≈1.508) by combining several ideas:

1️⃣  **Diversified initialisation**
    – a uniform start,
    – many random Dirichlet draws with a wide concentration range,
    – a sparse random start (few non‑zero entries).

2️⃣  **Extended simulated annealing**
    – two kinds of moves: a gradient‑guided transfer of mass from the most to the
      least contributing index, and a small proportion of completely random
      transfers (helps escaping flat regions).
    – a slower temperature decay and a slowly shrinking maximal transfer
      fraction, allowing finer and finer adjustments.
    – the annealing phase is split into several independent runs (each
      re‑initialised from the current best) to increase the chance of finding
      a better basin.

3️⃣  **Projected gradient descent**
    – after annealing a standard back‑tracking line‑search is applied using the
      sub‑gradient of the active (maximal) convolution entries.
    – the step size is adapted dynamically.

The whole search respects the 1000 s wall‑time limit; if it runs out early the
best vector found so far is returned.  The final vector is normalised to have
unit ℓ₁‑norm (the scale does not matter for the objective because the denominator
cancels the scaling).  The function ``propose_candidate`` is the entry point
used by the judge.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Global timing
# ----------------------------------------------------------------------
start_time = None  # will be set in propose_candidate


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (v≥0, Σv=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (the denominator is 1 because we keep the vector on the simplex)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at ``a`` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Initialisation helpers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float, rng: np.random.Generator):
    """Return the best vector obtained from a batch of Dirichlet draws."""
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # concentration log‑uniform on [10⁻³, 10⁰·⁷] (covers very spiky to near‑uniform)
        conc_log = rng.uniform(-3.0, 0.7)
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


def _sparse_initial(n: int, sparsity_ratio: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random sparse vector with roughly ``sparsity_ratio`` non‑zero entries.
    The non‑zero values are drawn from an exponential distribution and then
    normalised to sum to one.
    """
    k = max(2, int(sparsity_ratio * n))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 2️⃣  Simulated annealing (extended version, several independent runs)
# ----------------------------------------------------------------------
def _simulated_annealing_extended(start_vec: np.ndarray,
                                  start_score: float,
                                  deadline: float,
                                  rng: np.random.Generator,
                                  run_id: int = 0) -> (np.ndarray, float):
    """
    Run a simulated‑annealing optimisation starting from ``start_vec``.
    The routine mixes a gradient‑guided move (transfer mass from the index that
    contributes most to the current maximal convolution to the least contributing
    one) with occasional completely random transfers.  The temperature decays
    slowly and the maximal allowed transfer fraction shrinks linearly with the
    remaining time.
    """
    a = start_vec.copy()
    n = a.shape[0]
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    # temperature scheme – start with a small fraction of the current value
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.99995

    max_frac_start = 0.30   # early transfers may move up to 30 % of source mass
    max_frac_end = 0.005    # final moves become very fine‑grained

    while time.time() < deadline:
        # recompute convolution and sub‑gradient
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # choose the move type
        if rng.random() < 0.15:               # 15 % random move
            src = rng.integers(n)
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)
        else:                                 # gradient‑guided move
            src = int(np.argmax(grad))
            dst = int(np.argmin(grad))

        # linear schedule for the maximal transfer fraction
        t_rem = max(deadline - time.time(), 0.0)
        total_rem = max(deadline - start_time, 1e-12)
        frac = max_frac_end + (max_frac_start - max_frac_end) * max(t_rem / total_rem, 0.0)

        delta = a[src] * rng.random() * frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        if cand[src] < 0.0:                     # safety (should not happen)
            continue

        cand_score = _objective_score(cand)
        diff = cand_score - cur_score

        # Metropolis acceptance
        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Run{run_id} SA new best {best_score:.6f}")

        # temperature cooling
        T = max(T * decay, 1e-12)

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Projected gradient descent (back‑tracking line‑search)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray,
                                cur_score: float,
                                deadline: float) -> (np.ndarray, float):
    """
    Fine‑tune ``a0`` by moving opposite to the sub‑gradient of the active
    convolution entries and projecting back onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    step = 0.5

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        # try a handful of shrinking step factors
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)   # shrink step if no progress
        else:
            step = min(step * 1.05, 1.0)    # slightly increase step when moving downhill

    return best_a, best_score


# ----------------------------------------------------------------------
# 4️⃣  Small random “local‑search” refinement after PGD
# ----------------------------------------------------------------------
def _random_local_refinement(a0: np.ndarray,
                             cur_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> (np.ndarray, float):
    """
    Very cheap random pairwise transfers (no temperature) – only accept strict
    improvements.  Acts as a final polishing step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst or a[src] <= 1e-15:
            continue
        # move a tiny random fraction of the source mass
        delta = a[src] * rng.random() * 0.005
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand_score = _objective_score(cand)
        if cand_score < best_score:
            a = cand
            best_score = cand_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Local refine new best {best_score:.6f}")

    return best_a, best_score
