def propose_candidate(seed=0, budget_s=1000):

    start = time.time()
    deadline = start+budget_s-0.2
    n=4096
    L_fft = next_pow2(2*n-1)
    best_seq = np.full(n, 1/n) # uniform
    best_score = objective_score(best_seq,n)
    # optionally read global starter.

    # We'll start with a Dirichlet sample for initial a maybe from global or random
    a = best_seq.copy()

    # main loop:
    it=0
    last_improve=0
    alpha=10.0
    while time.time() < deadline:
        it+=1
        # conv a*a
        conv = _conv_self(a, L_fft) # length 2n-1
        # compute objective
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve=it
            print(f"[{time.time()-start:.2f}s] iter {it} new best {cur_score:.6f} alpha={alpha:.2f}")

        # compute smooth gradient
        max_conv = conv.max()
        # compute weights (softmax-like)
        w = np.exp(alpha * (conv - max_conv))  # max exponent = 0; w <=1
        # compute gradient
        w_rev = w[::-1]

        # convolution via FFT for speed
        Wa = np.fft.rfft(w_rev, n=Lg)
        Aa = np.fft.rfft(a, n=Lg)
        conv2 = np.fft.irfft(Wa * Aa, n=Lg)[:len(w_rev) + n - 1]

        eta = 0.5 / np.sqrt(it)  # maybe bigger than 0.8 earlier

        exp_factor = np.exp(-eta * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a_new = a * exp_factor
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:
            a_new = np.full(n, 1/n)
        a = a_new

        # adapt alpha: if no improvement for many iters, increase alpha to sharpen
        if it - last_improve > 5000 and alpha < 500:
            alpha *= 1.5
            last_improve = it # reset

            a = sample / sample.sum()
            print(f"[{time.time()-start:.2f}s] restart with Dirichlet conc {conc:.3f}")

    # after loop, return best_seq as list.

"""
Hybrid smooth‑max gradient descent for the step‑function problem.

The algorithm works on a probability vector a (non‑negative, sum = 1)
and minimises the objective 2·n·max(conv(a,a)).
Instead of the nondifferentiable max, we use a smooth approximation
log‑sum‑exp with a temperature α that is increased during the run.
The update is performed in the exponential (mirror‑descent) domain,
followed by a projection onto the simplex.
To avoid stagnation the method injects occasional Gaussian noise,
increases α when the progress stalls and performs cheap random restarts.
The best sequence found within the allotted time‑budget is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) via FFT.
    a is assumed to be a 1‑D numpy array of floats.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """
    The quantity that evaluate_sequence returns after normalising a to sum 1:
    2·n·max(conv(a,a)).
    """
    L = _next_pow2(2 * n - 1)
    conv = _conv_self(a, L)
    return float(2.0 * n * conv.max())


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray, float]:
    """
    If a global variable `height_sequence_1` exists, normalise it and return it.
    """
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max function
        F(a)= (1/α)·log∑_k exp(α·conv_k)
    w.r.t. a.  Returns an (n,) array.
    """
    n = a.shape[0]
    max_val = conv.max()
    # weights proportional to exp(α·(conv-max))
    w = np.exp(alpha * (conv - max_val))          # max weight = 1, others ≤1
    w_rev = w[::-1]                               # reverse for convolution
    Lg = _next_pow2(len(w_rev) + n - 1)
    Wa = np.fft.rfft(w_rev, n=Lg)
    Aa = np.fft.rfft(a, n=Lg)
    conv2 = np.fft.irfft(Wa * Aa, n=Lg)[: len(w_rev) + n - 1]
    # gradient_i = 2 * Σ_j a_j·w_{i+j}
    grad = 2.0 * conv2[(len(w_rev) - 1) - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the runner
    deadline = start_time + budget_s - 0.25
    total_budget = deadline - start_time

    # problem size – keep the same size as the reference solution
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ---------- initialise ----------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter, starter_score = _maybe_use_global_starter(n)
    if starter is not None and starter_score < best_score:
        best_seq, best_score = starter, starter_score
        print(f"[0.00s] Global starter improved score={starter_score:.6f}")

    # start from the current best
    a = best_seq.copy()
    rng = np.random.default_rng()

    # algorithmic parameters
    it = 0
    last_improve_it = 0
    alpha = 8.0                     # starts fairly smooth
    max_alpha = 2000.0
    alpha_inc_iter = 5000          # increase α every this many iterations if stuck
    perturb_iter = 50000           # inject noise after this many idle iterations
    restart_iter = 200000          # cheap random restart interval
    noise_scale = 0.02              # relative amplitude of Gaussian noise

    while time.time() < deadline:
        it += 1
        # ----- evaluate current point -----
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] it={it:,} new best {cur_score:.6f} α={alpha:.2f}")

            # early exit if we have already crossed the target
            if best_score <= 1.503:
                print("[INFO] Target reached, exiting early.")
                break

        # ----- compute smoothed gradient -----
        grad = _smooth_gradient(a, conv, alpha)

        # ----- exponential‑gradient update (mirror descent) -----
        eta = 0.6 / math.sqrt(it)               # step‑size schedule
        exp_factor = np.exp(-eta * grad)
        # avoid extreme under/overflow
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a_new = a * exp_factor
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:
            a_new = uniform.copy()

        # ----- accept / reject (simulated‑annealing style) -----
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            remaining = max(0.0, deadline - time.time())
            T = 5e-5 * (remaining / total_budget)   # tiny temperature decaying to zero
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True
        if accept:
            a = a_new
        # else keep the old a (no change)

        # ----- adapt smoothing parameter α -----
        if (it - last_improve_it) > alpha_inc_iter and alpha < max_alpha:
            alpha = min(max_alpha, alpha * 1.5)
            last_improve_it = it           # reset stall counter after α change
            print(f"[{time.time() - start_time:.2f}s] increasing α to {alpha:.2f}")

        # ----- occasional random perturbation to escape flat regions -----
        if (it - last_improve_it) > perturb_iter:
            perturb = rng.normal(scale=noise_scale, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = uniform.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] injected Gaussian noise")

        # ----- cheap random restart when we have been idle too long -----
        if it % restart_iter == 0:
            # draw a fresh Dirichlet‑like point with a random concentration
            log_c = rng.uniform(np.log(0.02), np.log(20.0))
            c = np.exp(log_c)
            sample = rng.gamma(shape=c, scale=1.0, size=n)
            if sample.sum() > 0:
                a = sample / sample.sum()
                print(f"[{time.time() - start_time:.2f}s] restart at it={it:,} (c={c:.3f})")

    # ------------------------------------------------------------------
    # finalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    final_score = _objective_score(best_seq, n)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}")

    # Convert to plain python list of floats for the runner
    return [float(v) for v in best_seq]
