"""candidate_proposer.py
Optimisation of the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine proceeds in three stages:

1️⃣  An expanded parametric sweep (geometric, power‑shifted, cosine / Hann,
    log‑inverse families and a refined two‑block family).  All candidates are
    normalised to sum = 1 and the best score is retained.

2️⃣  A massive random‑Dirichlet search.  The concentration parameter c is drawn
    uniformly in log‑space from 10⁻⁴ … 10², far wider than before; 2000 draws
    are attempted unless the time budget expires.

3️⃣  Local optimisation from the current best point using an Adam‑style
    optimiser on the exact (sub‑)gradient of the max‑convolution, together with
    occasional random “shake‑ups”.  The optimiser runs until the deadline is
    reached, constantly projecting onto the probability simplex.

4️⃣  A short simulated‑annealing (SA) hill‑climb to escape any remaining
    local minima.

The final sequence is clipped to the domain required by ``evaluate_sequence``,
renormalised and returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max(conv(a,a))` at `a` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fall back to the uniform simplex point
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# 1️⃣ Expanded parametric families
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -0.5, 250)
    uniform = np.full(n, 1.0 / n)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq = _project_simplex(seq)
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.1f}s] Geom eps={eps:.1e} lam={lam:.2f} → {sc:.6f}")
    return best_seq, best_score


def _sweep_power_shifted(n, best_seq, best_score, deadline):
    """a_i ∝ (i+δ)^p  –  adds a small shift δ to avoid a zero at i=0."""
    idx = np.arange(n, dtype=np.float64)
    deltas = [0.0, 0.5, 1.0]
    for δ in deltas:
        for p in np.linspace(-2.5, 2.5, 141):
            if time.time() >= deadline:
                break
            seq = np.power(idx + 1.0 + δ, p)
            seq = _project_simplex(seq)
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.1f}s] Power p={p:.2f} δ={δ:.1f} → {sc:.6f}")
    return best_seq, best_score


def _sweep_cosine(n, best_seq, best_score, deadline):
    """Cosine window a_i ∝ 1 - cos(π·i/(n‑1))."""
    idx = np.arange(n, dtype=np.float64)
    window = 1.0 - np.cos(np.pi * idx / (n - 1))
    seq = _project_simplex(window)
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.1f}s] Cosine window → {sc:.6f}")
    return best_seq, best_score


def _sweep_hann(n, best_seq, best_score, deadline):
    """Hann (raised‑cosine) window a_i ∝ 0.5·(1 - cos(2π·i/(n‑1)))."""
    idx = np.arange(n, dtype=np.float64)
    window = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    seq = _project_simplex(window)
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.1f}s] Hann window → {sc:.6f}")
    return best_seq, best_score


def _sweep_log_inverse(n, best_seq, best_score, deadline):
    """Inverse‑log: a_i ∝ 1 / (log(i+2))   (adds +2 to avoid log(0))."""
    idx = np.arange(n, dtype=np.float64)
    for base in [np.e, 2.0, 10.0]:
        seq = 1.0 / np.log(idx + 2.0) ** 1.0
        seq = _project_simplex(seq)
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.1f}s] Log⁻¹ base={base:.1f} → {sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks_refined(n, best_seq, best_score, deadline):
    """Two uniform blocks, but allow a gap of zeros between them."""
    for L_frac in np.linspace(0.01, 0.3, 30):
        L = int(L_frac * n)
        if L < 1:
            continue
        for gap_frac in np.linspace(0.0, 0.5, 6):
            gap = int(gap_frac * n)
            if 2 * L + gap > n:
                continue
            for w in np.linspace(0.1, 0.9, 9):
                if time.time() >= deadline:
                    break
                seq = np.zeros(n)
                seq[:L] = w / L
                seq[-L:] = (1.0 - w) / L
                # a zero gap sits automatically between the two blocks
                seq = _project_simplex(seq)
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.1f}s] TwoBlocks L={L} gap={gap} w={w:.2f} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Massive Dirichlet search
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 2000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration drawn log‑uniformly over 10⁻⁴ … 10²
        logc = rng.uniform(-4.0, 2.0)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        seq = _project_simplex(a)
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.1f}s] Dirichlet c={c:.3g} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Adam‑style gradient descent on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0, deadline, best_seq, best_score):
    """Adam optimiser using the exact sub‑gradient of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_sc = float(2.0 * n * conv.max())
        if cur_sc < best_score:
            best_score, best_seq = cur_sc, a.copy()
            print(f"[{time.time() - start_time:.1f}s] Adam it={t:,} score={cur_sc:.6f}")
            if cur_sc <= 1.503:
                break

        grad = _subgradient_maxconv(a, conv)
        # zero‑mean step (keeps the ℓ₁‑norm = 1)
        d = -(grad - grad.mean())

        t += 1
        m = beta1 * m + (1 - beta1) * d
        v = beta2 * v + (1 - beta2) * (d ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = m_hat / (np.sqrt(v_hat) + eps)

        # line‑search scaling – start with a moderate factor
        scale = 1.0
        while scale > 1e-12:
            a_new = a + scale * step
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            sc_new = float(2.0 * n * conv_new.max())
            if sc_new <= cur_sc:
                a = a_new
                break
            scale *= 0.5

        if scale <= 1e-12:
            # failed to improve – random jitter to escape
            jitter = 1e-6 * np.random.randn(n)
            a = _project_simplex(a + jitter)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣ Light simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = best_score
    start = time.time()
    total = max(deadline - start, 1e-6)

    # Pre‑compute a rough gradient for source selection
    grad = _subgradient_maxconv(a, _conv_self(a))

    it = 0
    no_improve = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total
        T = (1.0 - frac) * cur_score * 0.002  # start temperature ≈0.2 % of score

        # Update gradient occasionally (cheap & still useful)
        if it % 150 == 0:
            grad = _subgradient_maxconv(a, _conv_self(a))

        # Choose a source index (prefer large positive gradient)
        prob = np.maximum(grad, 0)
        prob_sum = prob.sum()
        src = rng.choice(n, p=prob / prob_sum) if prob_sum > 0 else rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue

        # Transfer a fraction of the mass (up to 30 %)
        max_f = 0.3
        f = rng.random() * max_f
        eps = a[src] * f
        if eps < 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(a_prop)

        conv_new = _conv_self(a_prop)
        sc_new = float(2.0 * n * conv_new.max())
        delta = sc_new - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_prop
            cur_score = sc_new
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                no_improve = 0
                print(f"[{time.time() - start_time:.1f}s] SA it={it:,} score={best_score:.6f}")
                if best_score <= 1.503:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 4000:
            # random shake‑up to re‑explore
            i, j = rng.choice(n, size=2, replace=False)
            eps2 = a[i] * (rng.random() * max_f)
            a[i] -= eps2
            a[j] += eps2
            a = _project_simplex(a)
            cur_score = float(2.0 * n * _conv_self(a).max())
            no_improve = 0

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list[float]) that minimises
    ``evaluate_sequence``.  The method combines a broad parametric sweep,
    a large Dirichlet random search, Adam‑based projected gradient descent and a
    final simulated‑annealing hill‑climb.  The best vector (normalised to sum = 1)
    is returned.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # --------------------------------------------------------------
    # Problem size (more than enough resolution)
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.1f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣ Expanded parametric sweep
    # --------------------------------------------------------------
    sweep_deadline = deadline - 350.0
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_power_shifted(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_cosine(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_hann(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_log_inverse(n, best_seq, best_score, sweep_deadline)
    if time.time() < sweep_deadline:
        best_seq, best_score = _sweep_two_blocks_refined(n, best_seq, best_score, sweep_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Massive Dirichlet draws
    # --------------------------------------------------------------
    dirichlet_deadline = deadline - 250.0
    if time.time() < dirichlet_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, dirichlet_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Adam‑style projected gradient descent
    # --------------------------------------------------------------
    opt_deadline = deadline - 150.0
    if time.time() < opt_deadline:
        print(f"[{time.time() - start_time:.1f}s] Starting Adam optimiser")
        best_seq, best_score = _adam_optimize(best_seq, opt_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.1f}s] After Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.1f}s] Starting final SA refinement")
        best_seq, best_score = _simulated_annealing(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.1f}s] Final SA score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise once more and return a plain Python list
    # --------------------------------------------------------------
    best_seq = _project_simplex(best_seq)
    print(f"[{time.time() - start_time:.1f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
