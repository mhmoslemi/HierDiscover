def step_reduce_max(a):
    n = len(a)
    conv = _conv_self(a)
    max_val = conv.max()
    # tolerance for active set
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    m = len(active)
    # Build A_ub
    rows = []
    cols = []
    data = []
    # For each active shift
    for idx, k in enumerate(active):
        # row index idx
        # variable indices for δ_i
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # For i in [i_start,i_end]
        # coefficient is 2 * a_{k-i}
        # Write to arrays
        for i in range(i_start, i_end+1):
            j = k - i
            coeff = 2 * a[j]  # because conv derivative w.r.t a_i is sum over j a_j for shift; but we multiply by 2? Let's verify.
            rows.append(idx)
            cols.append(i)  # δ_i column
            data.append(coeff)
        # coefficient for λ variable (index n)
        rows.append(idx)
        cols.append(n)
        data.append(-1.0)
    # Build sparse matrix

"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in two phases:

1. **Initialisation** – start from the global `height_sequence_1` if it exists,
   otherwise from a random Dirichlet vector (all entries ≥0, sum = 1).  The
   length is fixed to n = 4096 (the value used by the reference solution).

2. **Coupled LP / local search** – repeatedly
   * compute the self‑convolution,
   * pick the K largest convolution entries (these are the “active” constraints),
   * solve a tiny linear programme that finds a direction `δ` moving mass between
     coordinates while *simultaneously* decreasing all active convolution
     values,
   * apply the full step (or a back‑tracked fraction) if it really reduces the
     true maximum of the convolution,
   * fall back to a small random jitter when the LP gives no improvement.

The LP has the form

    minimise   λ
    subject to 2·a⋆δ ≤ λ – conv[a,a]   (for each active shift)
                Σ δ_i = 0
                -a_i ≤ δ_i ≤ a_i   (keep the new vector non‑negative)

where `a⋆δ` is the linear convolution of the current sequence with the
direction.  The bounds on `δ` guarantee feasibility of the simplex after the
step.  The routine runs until the wall‑clock budget (≈ 1000 s) is exhausted,
printing any improvement it finds.  The best sequence seen is returned as a
plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Objective value 2·n·max(conv) for a normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# LP step
# ----------------------------------------------------------------------
def _lp_improve(
    a: np.ndarray,
    K: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, bool]:
    """
    Solve a tiny LP that tries to decrease the largest K convolution entries.
    Returns a new candidate sequence (already simplex‑projected) and a flag
    indicating whether an improvement was made.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    cur_max = conv.max()

    # select K shifts with largest convolution values (unsorted, cheaper)
    if K >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(-conv, K - 1)[:K]   # indices of K largest values

    m = len(idx)

    # Build the dense A_ub matrix (shape m × (n+1))
    A_ub = np.zeros((m, n + 1), dtype=np.float64)
    b_ub = -conv[idx]  # right‑hand side = -current value

    for row, k in enumerate(idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vectorised assignment for the δ‑coefficients of this row
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        A_ub[row, i_range] = 2.0 * a[j_range]
        A_ub[row, n] = -1.0                     # coefficient of λ

    # Equality: Σ δ_i = 0 (preserve total mass)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0], dtype=np.float64)

    # Variable bounds: -a_i ≤ δ_i ≤ a_i  (allow moving at most the whole mass)
    bounds = [(-a[i], a[i]) for i in range(n)] + [(0.0, None)]   # λ ≥ 0

    # Objective: minimise λ
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    # Solve the LP (HiGHS is the default for recent SciPy versions)
    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        return a, False

    delta = res.x[:n]
    lam_opt = res.x[-1]

    # If the LP suggests a strictly smaller λ, attempt to apply the direction.
    if lam_opt < cur_max - 1e-12:
        # Try the full step first
        cand = a + delta
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        new_max = _conv_self(cand).max()
        if new_max < cur_max - 1e-12:
            return cand, True

        # If the full step fails, back‑track geometrically
        for alpha in (0.5, 0.25, 0.125, 0.0625):
            cand = a + alpha * delta
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            new_max = _conv_self(cand).max()
            if new_max < cur_max - 1e-12:
                return cand, True

    # No improvement found via the LP direction
    return a, False


# ----------------------------------------------------------------------
# Simple random jitter (fallback)
# ----------------------------------------------------------------------
def _random_jitter(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a tiny random amount of mass from one index to another."""
    n = a.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    eps = a[i] * (rng.random() * 5e-4 + 1e-9)   # ≤ 0.05 % of source mass
    if eps <= 0.0:
        return a
    a[i] -= eps
    a[j] += eps
    return _project_simplex(a)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function.  The routine starts from the global
    `height_sequence_1` if present, otherwise from a random Dirichlet vector,
    and then repeatedly performs a coordinated LP‑based improvement step
    followed by occasional random jitter.  The wall‑clock limit is respected.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise from the global starter if possible
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size == 0:
            raise ValueError
        starter = np.maximum(starter, 0.0)
        starter = starter / starter.sum()
        if starter.size != n:
            # pad or truncate to the required length
            if starter.size < n:
                starter = np.concatenate([starter, np.zeros(n - starter.size)])
            else:
                starter = starter[:n]
            starter = starter / starter.sum()
        a = starter.astype(np.float64)
    except Exception:
        # random Dirichlet (guaranteed on the simplex)
        a = rng.dirichlet(np.ones(n)).astype(np.float64)

    # Ensure exact normalisation
    a = _project_simplex(a)
    best_seq = a.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – LP based fine‑tuning
    # --------------------------------------------------------------
    # Number of active shifts used in the LP (tunable)
    K_active = 80
    iter_no = 0
    no_improve = 0

    while time.time() < deadline:
        iter_no += 1
        a, improved = _lp_improve(a, K_active, rng)

        cur_score = _objective_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] LP iter {iter_no} score = {best_score:.6f}")
        else:
            no_improve += 1
            # occasional jitter to shake out of a plateau
            if no_improve > 300:
                a = _random_jitter(a, rng)
                no_improve = 0

        # after a while, increase K to tighten the LP (helps in late stages)
        if iter_no % 5000 == 0 and K_active < 200:
            K_active = min(200, K_active + 20)

    # --------------------------------------------------------------
    # Final tidy‑up – ensure a valid probability vector
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    final_score = _objective_score(best_seq)
    print(
        f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}"
    )
    # return as plain Python list of floats
    return [float(x) for x in best_seq]
