def _fft_convolve(x, y):

    return conv[:len(x)+len(y)-1]

def _smooth_exponential_descent(a0, deadline, start_score, best_seq):
    a = a0.copy()
    best = best_seq.copy()
    best_score = start_score
    # schedule parameters
    tau = 0.5
    eta = 0.1
    iteration = 0
    while time.time() < deadline:
        # compute conv self
        conv = _conv_self(a)
        # compute smoothed gradient
        maxc = conv.max()
        # compute softmax weights
        exp_arg = tau * (conv - maxc)
        # to avoid overflow, cap
        exp_arg = np.clip(exp_arg, -700, 0)  # np.exp overflow at ~709
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            # all underflow, fallback to delta at max index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum
        # gradient
        w_rev = w[::-1]
        conv_rev = _fft_convolve(a, w_rev)
        n = a.shape[0]
        grad = 2.0 * conv_rev[n-1:2*n-1][::-1]
        # EG update
        step = eta * grad
        # clamp step maybe
        # we want a_i_new = a_i * exp(-step_i)
        exp_factor = np.exp(-np.clip(step, -50, 50))
        a = a * exp_factor
        # renorm to simplex
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0/a.shape[0])
        else:
            a = a / sum_a
        # evaluate score
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth EG iter={iteration} tau={tau:.3e} eta={eta:.3e} score={cur_score:.6f}")
        # schedule update
        iteration += 1
        # increase tau gradually
        tau *= 1.001  # maybe small factor
        # adapt eta maybe
        if iteration % 100 == 0:
            eta *= 0.99
    return best, best_score

def _project_simplex(v):
    """Projection of vector v onto probability simplex."""
    # ensure np array
    v = np.asarray(v)

    # sort descending
    u = np.sort(v)[::-1]

def _smooth_pgd(a0: np.ndarray, deadline: float,
                best_seq: np.ndarray, best_score: float):
    """
    Projected gradient descent on a smooth log‑sum‑exp surrogate of max(conv).
    Starts from a0 (already on the simplex).  Returns the best sequence
    encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_local_seq = best_seq.copy()
    best_local_score = best_score
    # Parameters
    step = 0.1          # initial step size for projection descent
    tau = 0.1           # initial smoothing temperature
    it = 0
    while time.time() < deadline:
        it += 1
        # current exact score
        cur_score = _objective_score(a)
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothPGD it={it} score={cur_score:.6f}")
            if cur_score <= 1.503:
                break

        # self‑convolution
        conv = _conv_self(a)
        # smooth weights (softmax over conv)
        maxc = conv.max()
        exp_arg = tau * (conv - maxc)
        # protect against overflow
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # degenerate case – fall back to delta at maximum
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0

"""
Search for a non‑negative step‑function that (almost) minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1️⃣  Multiple short runs of a *smooth* projected‑gradient descent.
    The non‑smooth “max” is approximated by a log‑sum‑exp
    (temperature τ).  Starting from random Dirichlet vectors the
    method performs a gradient step on the smoothed surrogate,
    projects back onto the probability simplex and slowly raises τ.
    The best vector over all runs is kept.

2️⃣  A standard projected‑gradient descent on the exact objective
    (sub‑gradient of the max‑convolution) for fine‑tuning.

3️⃣  A short simulated‑annealing hill‑climb to escape possible
    shallow local minima.

All phases respect the 1000 s budget and print progress
information.  The final sequence is normalised to sum = 1 and
returned as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _fft_convolve(a, a)


def _objective_score(seq: np.ndarray) -> float:
    """
    Exact objective for a sequence normalised to sum = 1:
        2·n·max(conv(seq,seq))
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1} (Duchi et al. 2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0

    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]

    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣ Smooth projected gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_pgd(a0: np.ndarray, deadline: float,
                best_seq: np.ndarray, best_score: float):
    """
    Projected gradient descent on a smooth surrogate of the max‑convolution.
    The surrogate is  (1/τ)·log Σ exp(τ·conv), τ grows slowly so that the
    algorithm gradually focuses on the true max.
    Returns the best vector seen during the run and its exact objective
    (not the surrogate).
    """
    a = a0.copy()
    n = a.shape[0]

    best_local_seq = best_seq.copy()
    best_local_score = best_score

    step = 0.1        # initial step size for the projected gradient
    tau = 0.1         # initial smoothing temperature
    it = 0

    while time.time() < deadline:
        it += 1
        cur_score = _objective_score(a)

        # record improvement
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothPGD it={it} score={cur_score:.6f}")
            if cur_score <= 1.503:      # target reached → stop early
                break

        # ---- compute surrogate gradient ----
        conv = _conv_self(a)
        maxc = conv.max()
        # soft‑max weights (stable formulation)
        exp_arg = tau * (conv - maxc)
        exp_arg = np.clip(exp_arg, -700, 0)   # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:        # degenerate (should not happen)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient = 2 * (a * w_rev) restricted to the original indices
        w_rev = w[::-1]
        conv_rev = _fft_convolve(a, w_rev)          # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : 2 * n - 1][::-1]   # shape (n,)

        # ---- back‑tracking line search on the *exact* objective ----
        step_cur = step
        while step_cur > 1e-12:
            a_tmp = a - step_cur * grad
            a_tmp = _project_simplex(a_tmp)
            new_score = _objective_score(a_tmp)
            if new_score <= cur_score:
                a = a_tmp
                step = min(step_cur * 1.1, 1.0)   # gently increase step
                break
            step_cur *= 0.5
        else:
            # no improvement this iteration → shrink step
            step *= 0.5

        # make the surrogate sharper
        tau *= 1.005

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
# 2️⃣ Projected gradient descent on the exact objective (sub‑gradient)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")
            if cur_score <= 1.503:
                break

        grad = _subgradient_maxconv(a, conv)
        d = -(grad - grad.mean())               # keep sum zero

        # largest admissible step that preserves non‑negativity
        idx_neg = np.where(d < 0)[0]
        if idx_neg.size:
            max_step = np.min(-a[idx_neg] / d[idx_neg])
        else:
            max_step = 1.0
        step = max_step * 0.9

        # simple back‑tracking line search
        while step > 1e-14:
            a_new = a + step * d
            if np.any(a_new < 0):
                step *= 0.5
                continue
            conv_new = _conv_self(a_new)
            score_new = float(2.0 * n * conv_new.max())
            if score_new <= cur_score:
                a = a_new
                break
            step *= 0.5

        if step <= 1e-14:        # stagnation → add tiny jitter
            eps = 1e-7
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Simulated‑annealing hill‑climb (short final polish)
# ----------------------------------------------------------------------
def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = best_score
    cur_a = a.copy()

    total = max(deadline - time.time(), 1e-6)
    start = time.time()
    it = 0
    no_improve = 0

    grad = _subgradient_maxconv(a, _conv_self(a))

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total
        T = (1.0 - frac) * cur_score * 0.001

        if it % 100 == 0:
            grad = _subgradient_maxconv(a, _conv_self(a))
            pos = grad.max()
            neg = grad.min()

        prob = np.maximum(grad, 0)
        prob_sum = prob.sum()
        if prob_sum > 0:
            src = rng.choice(n, p=prob / prob_sum)
        else:
            src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue

        max_frac = 0.25
        frac_transfer = rng.random() * max_frac
        eps = a[src] * frac_transfer
        if eps <= 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        conv_new = _conv_self(a_prop)
        score_new = float(2.0 * n * conv_new.max())
        delta = score_new - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_prop
            cur_score = score_new
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} score={best_score:.6f}")
                if best_score <= 1.503:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 5000:
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps2 = a[i1] * (rng.random() * max_frac)
            a[i1] -= eps2
            a[i2] += eps2
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = float(2.0 * n * _conv_self(a).max())
            no_improve = 0

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine uses a smooth surrogate,
    a refined sub‑gradient descent and a short simulated annealing
    phase, staying within the supplied time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Multiple short runs of the smooth PGD (random restarts)
    # --------------------------------------------------------------
    # Reserve some time for the exact PGD and the final SA.
    reserve_time = 250.0          # seconds kept for later phases
    while time.time() < deadline - reserve_time:
        # draw a random Dirichlet‑like start
        rng = np.random.default_rng()
        logc = rng.uniform(-2.0, 0.8)       # concentration ≈ 0.01 … 6.3
        c = 10 ** logc
        a0 = rng.gamma(shape=c, scale=1.0, size=n)
        a0 /= a0.sum()
        # give each restart about 30 s (or whatever is left)
        sub_deadline = min(time.time() + 30.0, deadline - reserve_time)
        a_opt, sc_opt = _smooth_pgd(a0, sub_deadline, best_seq, best_score)
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] New best after smooth run: {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Projected gradient descent on the exact objective
    # --------------------------------------------------------------
    pgd_deadline = deadline - 100.0
    if time.time() < pgd_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting exact projected gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq, pgd_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After exact PGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Short simulated‑annealing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated‑annealing polish")
        best_seq, best_score = _simulated_annealing(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final SA score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
