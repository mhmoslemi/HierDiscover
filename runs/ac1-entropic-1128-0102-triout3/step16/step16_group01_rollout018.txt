"""
Propose a non‑negative step function minimizing

        2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation improves on the previous one by

* a richer parametric sweep (including reversed shapes and mixtures),
* several independent Adam‑style smoothed‑max gradient runs from random starts,
* a tightened exponentiated‑gradient phase,
* a classic sub‑gradient descent with back‑tracking,
* and a final random hill‑climb.

All phases respect the 1000 s budget, work on the full simplex,
and keep the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast, assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2).  a is already normalised to sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Soft‑max utilities (smooth approximation of max)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """p_k = exp(beta*conv_k) / Σexp(...).  Numerically stable."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k·conv_k  (soft‑max objective):
        g_i = 2· Σ_j a_j·p_{i+j}
    Computed efficiently as a cross‑correlation via FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    corr = np.fft.irfft(Af * Pf, n=L_grad)
    # We need the central n entries (indices n‑1 … 2n‑2)
    start = n - 1
    g = corr[start:start + n][::-1]          # reverse to get correct ordering
    return 2.0 * g


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]          # diagonal term counted twice (as required)
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (multiplicative updates)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    eta0 = 1.0
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # annealed step size
        eta = eta0 / (1.0 + 0.0008 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} score={sc:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 2000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam‑style smoothed‑max gradient descent
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray,
                L_fft: int,
                L_grad: int,
                deadline: float,
                start_time: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth‑max objective.
    Uses a schedule for β that slowly drives the soft‑max towards the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        # ----- temperature schedule -----
        beta = 1.0 + min(5000.0, it * 0.02)   # grows slowly, caps around 5000
        conv = _conv_self_fft(a, L_fft)
        p = _softmax_weights(conv, beta)

        # ----- gradient of the smooth‑max -----
        g = _grad_softmax(a, p, L_grad)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / (np.sqrt(v_hat) + eps)   # base learning rate = 0.5
        a = a - step * m_hat
        a = _project_simplex(a)

        sc = _eval_fast(a, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} β={beta:.1f} score={sc:.6f}")

        # occasional tiny jitter to avoid stagnation
        if it % 5000 == 0:
            eps_j = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps_j, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

# Projected sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on max(conv) with a simple back‑tracking line‑search."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        step = step0 / (it ** 0.55)
        improved = False
        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_sc = _eval_fast(cand, L_fft)
            if cand_sc < best_score:
                best_score = cand_sc
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cand_sc:.6f}")
                break
            step *= 0.5

        if not improved:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cand_sc - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns the best sequence as a plain Python list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 provides a good balance between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)   # needed for soft‑max gradients

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a global starter (height_sequence_1) exists, try it.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Enriched parametric sweep (≈45 s)
    #   – geometric × power shapes,
    #   – their reversed copies,
    #   – mixtures with the uniform distribution,
    #   – convex mixtures of a shape and its mirror.
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 45.0
    r_grid = np.logspace(np.log10(0.85), np.log10(0.9999), 30)
    p_grid = np.linspace(-3.5, 3.5, 35)
    lam_grid = np.linspace(0.0, 1.0, 13)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            shape = _make_shape(n, r, p)
            shape_rev = shape[::-1]

            # plain shape
            sc = _eval_fast(shape, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = shape.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.3f} plain score={sc:.6f}")

            # reversed shape
            sc = _eval_fast(shape_rev, L_fft)
            if sc < best_score:
                best_score = sc
                best_seq = shape_rev.copy()
                print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.3f} rev   score={sc:.6f}")

            for lam in lam_grid:
                # mix with uniform (both directions)
                cand = _mix_with_uniform(shape, lam)
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.3f} λU={lam:.2f} score={sc:.6f}")

                cand = _mix_with_uniform(shape_rev, lam)
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep rev r={r:.5f} p={p:.3f} λU={lam:.2f} score={sc:.6f}")

                # mixture of shape and its mirror
                cand = lam * shape + (1.0 - lam) * shape_rev
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep mix r={r:.5f} p={p:.3f} λM={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random mixture of a few parametric bases (≈30 s)
    # ------------------------------------------------------------------
    mix_deadline = start_time + 75.0
    for _ in range(600):
        if time.time() >= mix_deadline:
            break
        # pick three random geometric‑power bases and blend them
        weights = rng.random(3)
        weights /= weights.sum()
        base = np.zeros(n, dtype=np.float64)
        for i in range(3):
            r = rng.uniform(0.85, 0.9999)
            p = rng.uniform(-3.5, 3.5)
            shape = _make_shape(n, r, p)
            base += weights[i] * shape
        # optionally blend with uniform
        lam = rng.random()
        cand = lam * np.full(n, 1.0 / n) + (1.0 - lam) * base
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] random mix score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam‑style smoothed‑max gradient descent (≈300 s total)
    # ------------------------------------------------------------------
    # we run several independent starts (the current best and a few random seeds)
    adam_end = deadline - 350.0
    for start_type in ("best", "rand", "rand"):
        if time.time() >= adam_end:
            break
        if start_type == "best":
            a0 = best_seq.copy()
        else:
            a0 = rng.random(n)
            a0 = a0 / a0.sum()

        best_seq, best_score = _adam_phase(a0, L_fft, L_grad,
                                           adam_end, start_time,
                                           best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Exponentiated‑gradient phase on the exact L∞‑objective.
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:
        exp_deadline = deadline - 250.0
        best_seq, best_score = _expgrad_phase(best_seq, L_fft,
                                              exp_deadline, start_time,
                                              best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 5️⃣  Projected sub‑gradient descent with back‑tracking line‑search.
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        sub_deadline = deadline - 150.0
        best_seq, best_score = _subgradient_descent(best_seq, L_fft,
                                                    sub_deadline, start_time,
                                                    best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 6️⃣  Final random hill‑climb – tiny mass transfers.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 4e-3   # up to 0.4 % of source mass
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return the plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    final_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}")
    return [float(v) for v in best_seq]
