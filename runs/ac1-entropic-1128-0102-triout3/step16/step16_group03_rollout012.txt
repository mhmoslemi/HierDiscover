def evaluate_sequence(sequence):
    sequence validated, negative numbers clipped to zero, also clamped to max 1000.
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)

def simplex_projection(v):
    """Projection of vector v onto the probability simplex (nonnegative components that sum to 1)."""
    # Standard algorithm: sort descending, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # v is not positive enough, return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

The goal is to minimise  

\[
F(\mathbf a)=\frac{2\,n\;\max\big((\mathbf a*\mathbf a)\big)}{\big(\sum_i a_i\big)^2}\;,
\qquad a_i\ge 0 .
\]

Because the objective is homogeneous, we may normalise any candidate to a probability
distribution (`∑ a_i = 1`).  
With this normalisation \(F(\mathbf a)=2n\;\max(\mathbf a*\mathbf a)\).

The previous optimiser relied on a sub‑gradient descent on the *hard* max.
Here we replace the non‑smooth max by a **smooth log‑sum‑exp approximation**  

\[
\tilde F_\varepsilon(\mathbf a)=2n\,\varepsilon\,
\log\Bigl(\sum_k\exp\bigl((\mathbf a*\mathbf a)_k/\varepsilon\bigr)\Bigr),
\]

whose gradient can be written in closed form.  
A simple Adam optimiser on the simplex (projection after every step) is used.
The smoothing parameter \(\varepsilon\) is decreased in stages, allowing a coarse
global search followed by fine local refinement.

The algorithm proceeds in three phases:

1. **Broad random search** – a quick Dirichlet sampling yields a few promising
   seeds (kept in a size‑\(K\) pool).
2. **Smoothed Adam descent** – each seed is optimised by the smooth objective.
   The optimiser runs on the simplex, uses Adam‑style adaptive steps and a
   schedule of decreasing \(\varepsilon\).
3. **Fine mixing** – a golden‑section search mixes the best candidate with the
   uniform distribution, which is cheap and often reduces the max‑convolution
   further.

All phases respect the global time budget (≈ 1000 s).  The final sequence is
projected onto the simplex, guaranteeing non‑negativity and normalisation; it
can be safely passed to the provided `evaluate_sequence` function.

"""
Hybrid optimiser using a smoothed log‑sum‑exp objective and Adam on the simplex.

Phases
------
1️⃣  Broad Dirichlet sampling – keep the K best seeds.
2️⃣  Smoothed Adam descent – minimise
        2·n·ε·log(∑exp(conv/ε))
    with decreasing ε, projecting onto the probability simplex after every step.
3️⃣  Fine mixing with the uniform distribution (golden‑section search).

All phases respect a 1000 s time budget and finally return the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)−1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised (∑a=1) vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (∑x=1, x≥0). """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_smooth(a: np.ndarray, eps: float):
    """
    Gradient of the smoothed objective
        2·n·eps·log(∑exp(conv/eps))
    with respect to a (already normalised).  Returns (grad, max(conv)).
    """
    n = a.shape[0]

    # 1. self‑convolution
    conv = _conv_self(a)                     # shape: 2·n−1

    # 2. softmax weights for the log‑sum‑exp
    max_val = conv.max()
    # exponentials are safe because conv-max_val ≤ 0
    w = np.exp((conv - max_val) / eps)
    w /= w.sum()                              # ∑w = 1

    # 3. correlation g_j = Σ_i w_{i+j}·a_i   (j = 0..n−1)
    #    can be obtained by a linear convolution with the reversed weight vector
    w_rev = w[::-1]
    conv_ar = np.convolve(a, w_rev)          # length 3·n−2
    # slice the needed part and reverse to obtain the correct ordering
    g = conv_ar[n - 1 : 2 * n - 1][::-1]      # shape: n

    # gradient of the smoothed objective = 4·n·g  (see derivation in analysis)
    grad = 4.0 * n * g
    return grad, conv.max()


def _smooth_descent(a0: np.ndarray,
                    eps_schedule,
                    lr: float,
                    max_iters_per_eps: int,
                    start_time: float,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float):
    """
    Adam optimiser on the simplex for the smoothed objective.
    Returns the best sequence (and its score) found during the run.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    t = 0

    # keep a local best (may be useful for early exit)
    local_best_seq = a.copy()
    local_best_score = best_score

    for eps in eps_schedule:
        # optional tighter learning rate for very small eps
        cur_lr = lr * max(eps, 1e-4)

        for _ in range(max_iters_per_eps):
            if time.time() >= deadline:
                return best_seq, best_score

            # gradient of the smoothed surrogate
            grad, _ = _gradient_smooth(a, eps)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = cur_lr * m_hat / (np.sqrt(v_hat) + 1e-12)

            a = a - step
            a = _simplex_projection(a)

            # exact objective (no smoothing) – cheap because we already have conv
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()

            if cur_score < local_best_score:
                local_best_score = cur_score
                local_best_seq = a.copy()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth descent eps={eps:.6f} "
                          f"score={cur_score:.6f}")

        # end of inner loop for a particular eps
    return best_seq, best_score


def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """Golden‑section search on λ∈[0,1] for λ·a+(1‑λ)·uniform."""
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – the runner will kill the process after budget_s seconds
    deadline = start_time + budget_s - 0.25

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # If a global starter is available, reuse its length; otherwise fall back to 4096.
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        n = starter.shape[0]
    except Exception:
        n = 4096

    # ------------------------------------------------------------------
    # Baseline (uniform)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter
    # ------------------------------------------------------------------
    try:
        gs = np.asarray(height_sequence_1, dtype=float)
        gs = np.maximum(gs, 0.0)
        if gs.sum() > 0:
            gs = gs / gs.sum()
            sc = _objective_score(gs)
            if sc < best_score:
                best_seq, best_score = gs, sc
                print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – Broad random search (≈10 % of the budget)
    # ------------------------------------------------------------------
    K = 5                         # keep the K best random seeds
    top_candidates = [(best_score, best_seq)]  # list of (score, seq)
    rng = np.random.default_rng()
    rand_deadline = start_time + 0.10 * (deadline - start_time)

    while time.time() < rand_deadline:
        # Dirichlet draw with log‑uniform concentration
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a)

        # keep the K best candidates seen so far
        if len(top_candidates) < K or sc < max(s for s, _ in top_candidates):
            top_candidates.append((sc, a.copy()))
            top_candidates = sorted(top_candidates, key=lambda x: x[0])[:K]

    print(f"[{time.time() - start_time:.2f}s] Random search kept {len(top_candidates)} seeds")

    # ------------------------------------------------------------------
    # Phase 2 – Smoothed Adam descent on each seed
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    # reserve 20 % of what remains for the final mixing stage
    descent_budget = remaining * 0.80
    mix_budget = remaining - descent_budget
    per_seed_budget = descent_budget / max(1, len(top_candidates))

    eps_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005]
    lr = 0.12               # base learning rate for Adam
    max_iters = 3000       # per‑ε iterations (the inner loop may stop earlier)

    for sc, seed_seq in top_candidates:
        cand_deadline = time.time() + per_seed_budget
        if time.time() >= cand_deadline:
            break
        best_seq, best_score = _smooth_descent(
            seed_seq,
            eps_schedule,
            lr,
            max_iters,
            start_time,
            cand_deadline,
            best_seq,
            best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] After smooth descent best≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Fine mixing with the uniform distribution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mix_deadline = time.time() + mix_budget
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, mix_deadline, start_time, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Mixing finished, score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    # returning a plain Python list as required by the evaluator
    return [float(v) for v in best_seq]
