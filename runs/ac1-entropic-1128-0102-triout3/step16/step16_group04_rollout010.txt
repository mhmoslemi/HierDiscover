"""
Local “max‑flattening” hill‑climb.

The score 2·n·max(conv(a,a))/(sum(a)²) is scale‑invariant, so we keep the
vector on the probability simplex (∑a = 1).  For a given vector a the
convolution g = a∗a is a non‑negative sequence of length 2·n‑1 and the
objective is simply proportional to max(g).  The algorithm repeatedly
chooses a pair of indices (i, j) and moves a small amount δ from i to j.
The convolution can be updated analytically in O(n) time:
    g′ = g – 2·δ·shift(i,a) + 2·δ·shift(j,a) + δ²·e₂ᵢ + δ²·e₂ⱼ – 2·δ²·e_{i+j},
where ``shift(p,a)`` places a on the interval [p, p+n) of the full
convolution array and e_k is a unit vector at position k.
Only moves that do **not increase** the current maximum of the convolution
are accepted; this monotone descent quickly flattens the peak.
We focus the move on the current maximal convolution index k*:
    • find the pair (i) with the largest contribution a_i·a_{k*‑i},
    • find another index (j) with the smallest such contribution,
    • move a small fraction of a_i to a_j.
Occasional random moves are allowed to escape shallow plateaus.
The whole search respects a 1000 s budget and returns the best vector
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Compute the full linear convolution a*a via FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score from a pre‑computed convolution (sum(a) is assumed 1)."""
    return float(2.0 * n * conv.max())


def _product_for_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Return the vector p where p[i] = a[i] * a[k‑i] if the index is in range,
    otherwise 0.  Used to locate the largest contribution to conv[k].
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    p = np.zeros_like(a)
    p[mask] = a[mask] * a[j[mask]]
    return p


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine uses a greedy hill‑climb that
    directly reduces the maximal entry of the self‑convolution while keeping
    the vector on the simplex.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin before the evaluator kills the process
    deadline = start_time + budget_s - 0.15

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            # keep its length (should be the same as the previous run)

        else:
            raise ValueError
    except Exception:
        # default length used by the previous code (4096)
        n = 4096
        a = np.full(n, 1.0 / n)

    # Normalise to the probability simplex (sum = 1)
    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.shape[0]
    conv = _full_convolution(a)
    cur_max = conv.max()
    cur_score = _score_from_conv(conv, n)

    best_a = a.copy()
    best_score = cur_score
    best_conv = conv.copy()
    best_max = cur_max

    print(f"[0.00s] Initial length={n}, score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Hill‑climb loop
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # 1) locate the index of the current maximal convolution entry
        k_max = int(conv.argmax())

        # 2) contributions a[i] * a[k_max-i]
        prod = _product_for_k(a, k_max)

        # indices with a positive mass (candidates for receiving mass)
        positive_mask = a > 1e-12
        if not positive_mask.any():
            break

        # 3) source index: the largest contribution (largest product)
        i_src = int(prod.argmax())

        # 4) destination index: the smallest contribution among indices that already
        #    carry some mass (so we can add to them).  Use a large value for
        #    zero‑mass entries to avoid picking them as destination.
        prod_tmp = prod.copy()
        prod_tmp[~positive_mask] = np.inf
        i_dst = int(prod_tmp.argmin())

        # If source and destination coincide (unlikely) pick a random destination
        if i_src == i_dst:
            choices = np.where(positive_mask)[0]
            i_dst = int(rng.choice(choices))

        # 5) size of the transfer – a fraction of the source mass,
        #    limited to a modest absolute step.
        max_transfer = min(a[i_src], 0.02)          # at most 2 % of total mass per step
        if max_transfer <= 0.0:
            break
        delta = max_transfer * rng.uniform(0.4, 0.6)   # randomise a little

        # --------------------------------------------------------------
        # Compute the candidate new convolution (O(n) update)
        # --------------------------------------------------------------
        # copy current state – cheap, length ≈ 8 k
        conv_cand = conv.copy()
        a_old = a.copy()          # needed for the shift updates

        # shift updates, using slicing; each slice has length n
        conv_cand[i_src:i_src + n] -= 2.0 * delta * a_old
        conv_cand[i_dst:i_dst + n] += 2.0 * delta * a_old

        # quadratic self‑interaction terms
        conv_cand[2 * i_src] += delta ** 2
        conv_cand[2 * i_dst] += delta ** 2
        if i_src != i_dst:
            conv_cand[i_src + i_dst] -= 2.0 * delta ** 2

        # --------------------------------------------------------------
        # Acceptance test – only accept non‑increasing max‑value
        # --------------------------------------------------------------
        cand_max = conv_cand.max()
        if cand_max <= cur_max + 1e-14:          # allow tiny numerical wiggle
            # accept the move
            a[i_src] -= delta
            a[i_dst] += delta
            conv = conv_cand
            cur_max = cand_max
            cur_score = _score_from_conv(conv, n)

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_conv = conv.copy()
                best_max = cur_max
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Iter={iteration:,}  score={cur_score:.6f}")

        else:
            # occasional random “uphill” move (simulated annealing style)
            # temperature decays roughly as 1/sqrt(iter)
            T = 0.001 / np.sqrt(iteration)
            if rng.random() < np.exp(-(cand_max - cur_max) / T):
                a[i_src] -= delta
                a[i_dst] += delta
                conv = conv_cand
                cur_max = cand_max
                cur_score = _score_from_conv(conv, n)

        # Occasionally perform a tiny random shuffle to help escape
        if iteration % 2500 == 0:
            i, j = rng.choice(n, 2, replace=False)
            eps = rng.normal(scale=0.001)
            if a[i] + eps > 0 and a[j] - eps > 0:
                a[i] += eps
                a[j] -= eps
                # recompute full convolution (costly but very rare)
                conv = _full_convolution(a)
                cur_max = conv.max()
                cur_score = _score_from_conv(conv, n)

    # ------------------------------------------------------------------
    # Final normalisation – ensure non‑negative and sum = 1
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full_like(best_a, 1.0 / best_a.size)

    final_score = _score_from_conv(_full_convolution(best_a), n)
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {final_score:.6f}")

    # return plain Python list
    return [float(v) for v in best_a]
