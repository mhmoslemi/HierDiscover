def propose_candidate(seed=0, budget_s=1000, **kwargs):
    # set rng
    rng = np.random.default_rng(seed)
    # start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    n = 8192  # or maybe 4096 if need faster
    # Because we anticipate maybe 8192 is better.

    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score {best_score:.6f}")

    # Phase 0: random Dirichlet draws
    phase0_deadline = start_time + budget_s * 0.15
    trials = 300  # for each can sample
    for i in range(trials):
        if time.time() >= phase0_deadline: break
        logc = rng.uniform(-2.0, 0.7) # concentration exponent
        c = 10**logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0: continue
        a /= a_sum
        sc = 2.0 * n * _conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.2f} → score {sc:.6f}")
    # Also mixture random
    phase1_deadline = start_time + budget_s * 0.25
    if time.time() < phase1_deadline:
        for i in range(200):
            if time.time() >= phase1_deadline: break
            r = rng.uniform(0.85, 0.99)
            idx = np.arange(n, dtype=np.float64)
            left = r ** idx
            right = r ** (n-1 - idx)
            left /= left.sum()
            right /= right.sum()
            w = rng.random(3)
            w /= w.sum()
            seq = w[0] * left + w[1] * right + w[2] * uniform
            seq /= seq.sum()
            sc = 2.0 * n * _conv_self(seq).max()
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

    # Phase 2: smooth gradient descent with continuation

        beta = 1.0
        beta_max = 200.0
        # step size initial

        while time.time() < phase2_deadline and beta <= beta_max:
            # Run limited number of steps for current beta
            steps_for_beta = 400  # maybe vary
            for iter_in_beta in range(steps_for_beta):
                if time.time() >= phase2_deadline:
                    break
                # compute conv_self and smooth gradient
                conv = _conv_self(a)  # length 2n-1
                # compute smooth weight
                maxc = conv.max()
                # compute exponent with shift to avoid overflow
                exp_beta = np.exp(beta * (conv - maxc))
                sum_exp = exp_beta.sum()
                w = exp_beta / sum_exp
                # gradient
                a_rev = a[::-1]
                conv_w_rev = _conv_fft(w, a_rev)  # length 3n-2
                grad = 2.0 * conv_w_rev[(n-1):(n-1+n)]

                # line search backtracking
                cur_score = 2.0 * n * conv.max()
                gnorm = np.linalg.norm(grad) + 1e-12
                step_len = step / gnorm
                while True:
                    a_new = a - step_len * grad
                    # enforce nonnegative and simplex projection
                    a_new = np.maximum(a_new, 0.0)
                    a_new = _project_simplex(a_new)
                    conv_new = _conv_self(a_new)
                    new_score = 2.0 * n * conv_new.max()

                        break
                    step_len *= 0.5
                if new_score < cur_score:
                    a = a_new
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = a.copy()
                        print(f"[{time.time() - start_time:.2f}s] beta={beta:.1f} iter={iter_in_beta} score={new_score:.6f}")

                    a[i2] += eps

                    a = _project_simplex(np.maximum(a, 0.0))
                    no_improve = 0
    # final normalisation (already sum=1)

    total = best_seq.sum()

The program searches for a non‑negative step‑function (a vector `a`) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max\_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `n = len(a)` and `a*a` is the linear self‑convolution.  
The search consists of four phases:

1. **Random initialisation** – many Dirichlet draws and a mixture of geometric
   sequences give a set of promising starting points.
2. **Continuation gradient descent** – we minimise a *smooth* surrogate of the
   max‑convolution  

   \[
   f_\beta(a)=\frac{1}{\beta}\log\!\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr)
   \]

   (soft‑max).  
   For a fixed temperature `β` the gradient can be computed analytically with
   FFT‑based convolutions; we perform projected gradient steps on the probability
   simplex.  Starting with a small `β` we slowly increase it (continuation), so
   the optimisation first sees a smooth landscape and finally approaches the
   original non‑smooth objective.
3. **Fine‑grained hill‑climb** – a tiny‑mass move (same idea as the reference
   implementation) removes the last remaining plateaus.
4. **Normalisation & return** – the best vector found is normalised to sum = 1
   and returned as a plain Python list.

All operations respect a global 1000 s wall‑clock budget; the algorithm prints
progress messages so the evolution of the bound can be followed.

```python
"""
Step‑function optimiser.

* Random Dirichlet / geometric mixtures for diverse starts.
* Continuation gradient descent on a soft‑max approximation of max(conv).
  The gradient is evaluated with two FFT‑based convolutions and the iterate is
  projected onto the probability simplex.
* Final tiny‑mass hill‑climb for polishing.
* Runs for at most 1000 s and returns the best sequence (list of floats).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (components ≥0, sum = 1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (a lies on the simplex).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Gradient of the smooth (soft‑max) surrogate
# ----------------------------------------------------------------------


def _smooth_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, score) where
      * grad is the gradient of 2·n·softmax(conv(a,a))
      * score is the true hard‑max objective 2·n·max(conv).
    All vectors are assumed to sum to 1.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    hard_score = 2.0 * n * conv.max()

    # soft‑max weights
    maxc = conv.max()
    exp_beta = np.exp(beta * (conv - maxc))   # shift to avoid overflow
    w = exp_beta / exp_beta.sum()            # w_k ≥0, sum w = 1

    # gradient = 2 * (w ⋆ a_rev)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    conv_wr = _conv_fft(w, a_rev)            # length 3n‑2
    grad = 2.0 * conv_wr[(n - 1):(n - 1 + n)]

    return grad, hard_score


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with minimal evaluate_sequence score.
    The algorithm follows the four phases described in the module docstring.
    Returns a plain list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a tighter asymptotic bound,
    # but we keep it modest enough for many FFT‑based steps.
    # ------------------------------------------------------------------
    n = 16384                     # feel free to lower if time becomes tight
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – interpolate if needed.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                if starter.size != n:
                    # linear interpolation to required length
                    old_x = np.linspace(0, 1, starter.size)
                    new_x = np.linspace(0, 1, n)
                    starter = np.interp(new_x, old_x, starter)
                    starter = np.maximum(starter, 0.0)
                    starter /= starter.sum()
                sc = 2.0 * n * _conv_self(starter).max()
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – random Dirichlet draws (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + budget_s * 0.15
    if time.time() < phase0_deadline:
        for _ in range(400):
            if time.time() >= phase0_deadline:
                break
            logc = rng.uniform(-2.0, 0.7)          # concentration 0.01 … 5
            c = 10 ** logc
            a = rng.gamma(shape=c, scale=1.0, size=n)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            sc = 2.0 * n * _conv_self(a).max()
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random mixtures of geometric / uniform components (≈10 %)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.25
    if time.time() < phase1_deadline:
        idx = np.arange(n, dtype=np.float64)
        for _ in range(250):
            if time.time() >= phase1_deadline:
                break
            r = rng.uniform(0.85, 0.995)
            left = np.power(r, idx)
            right = np.power(r, n - 1 - idx)
            left /= left.sum()
            right /= right.sum()
            w = rng.random(3)
            w /= w.sum()
            seq = w[0] * left + w[1] * right + w[2] * uniform
            seq /= seq.sum()
            sc = 2.0 * n * _conv_self(seq).max()
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – continuation gradient descent on smooth surrogate
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.85
    if time.time() < phase2_deadline:
        a = best_seq.copy()
        beta = 1.0
        beta_max = 200.0
        step = 0.2                         # nominal step size in the tangent space

        while time.time() < phase2_deadline and beta <= beta_max:
            # number of inner steps for the current temperature
            inner_steps = 500
            for it in range(inner_steps):
                if time.time() >= phase2_deadline:
                    break

                grad, cur_score = _smooth_gradient(a, beta)

                # back‑tracking line search (using the *hard* score)
                gnorm = np.linalg.norm(grad) + 1e-12
                step_len = step / gnorm

                while True:
                    a_new = a - step_len * grad
                    a_new = np.maximum(a_new, 0.0)
                    a_new = _project_simplex(a_new)

                    new_score = 2.0 * n * _conv_self(a_new).max()
                    if new_score < cur_score or step_len < 1e-12:
                        break
                    step_len *= 0.5

                # accept / adapt step size
                if new_score < cur_score:
                    a = a_new
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = a.copy()
                        print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} it={it} → {new_score:.6f}")
                    step = min(step * 1.05, 2.0)
                else:
                    step = max(step * 0.5, 1e-12)

                # occasional tiny jitter to leave plateaus
                if it % 2000 == 0:
                    i1, i2 = rng.choice(n, size=2, replace=False)
                    eps = a[i1] * (rng.random() * 0.001 + 1e-6)
                    a[i1] -= eps
                    a[i2] += eps
                    a = _project_simplex(np.maximum(a, 0.0))

            # increase temperature
            beta = min(beta * 1.5, beta_max)
            step = 0.2                     # reset nominal step size for the new β

    # ------------------------------------------------------------------
    # Phase 3 – final tiny‑mass hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        a = best_seq.copy()
        n = a.shape[0]
        best_a = a.copy()
        best_hill_score = best_score
        rng_h = np.random.default_rng(seed + 999)
        stagnation = 0

        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _subgradient_maxconv(a, conv)

            # choose a source index with large gradient and a sink with small gradient
            src_candidates = np.where(a > 1e-15)[0]
            if src_candidates.size == 0:
                break
            src = src_candidates[np.argmax(grad[src_candidates])]
            dst = np.argmin(grad)

            if src == dst:
                src, dst = rng_h.choice(n, size=2, replace=False)
                if a[src] < 1e-15:
                    continue

            eps = a[src] * (rng_h.random() * 0.004 + 1e-6)
            if eps <= 0:
                continue

            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps

            cand_score = 2.0 * n * _conv_self(a_prop).max()
            if cand_score < best_hill_score:
                a = a_prop
                best_a = a.copy()
                best_hill_score = cand_score
                best_seq = a.copy()
                best_score = cand_score
                stagnation = 0
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb → {cand_score:.6f}")
            else:
                stagnation += 1
                if stagnation > 3000:
                    # random jitter to break a possible deadlock
                    i1, i2 = rng_h.choice(n, size=2, replace=False)
                    jitter = a[i1] * (rng_h.random() * 0.004 + 1e-6)
                    a[i1] -= jitter
                    a[i2] += jitter
                    a = _project_simplex(np.maximum(a, 0.0))
                    stagnation = 0

    # ------------------------------------------------------------------
    # Normalise the final answer and return it as a Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
