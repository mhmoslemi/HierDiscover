"""
Propose a step‑function sequence with a very low evaluation score.

The optimiser works in several stages, each respecting the global
1000 s wall‑clock budget:

1.  Quick random Dirichlet / mixture trials – to obtain a decent
    starting point.
2.  Classical projected sub‑gradient descent on the true L∞‑objective
    (the maximum of the self‑convolution).  This finds a good local
    basin.
3.  Mirror‑descent (exponentiated‑gradient) on the simplex – a
    geometry‑aware step that can escape the plateau where the ordinary
    sub‑gradient gets stuck.
4.  Soft‑max (smooth‑max) gradient descent – a smoothed version of the
    objective that provides a well‑behaved gradient and enables a
    fine‑grained reduction of the peak.
5.  A final tiny‑mass hill‑climb for last‑minute polishing.

All phases keep track of the best sequence seen so far and return
the best one found before the deadline.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog  # only imported for possible future extensions

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return the evaluation metric 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (normalised to sum=1).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# Random initialisers

def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=300):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)          # concentration roughly 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=250):
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} w={w} → {sc:.6f}")
    return best_seq, best_score

def _subgradient_descent(start_seq, deadline, best_seq, best_score):
    """Standard projected sub‑gradient descent on the max‑convolution."""
    a = start_seq.copy()
    n = a.shape[0]
    step = 0.2
    min_step = 1e-12
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={iteration} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        gnorm = np.linalg.norm(grad) + 1e-12
        step_len = step / gnorm

        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_score = float(2.0 * n * _conv_self(a_new).max())
            if new_score < cur_score or step_len <= min_step:
                break
            step_len *= 0.5

        if new_score < cur_score:
            a = a_new
            step = min(step * 1.05, 1.0)
        else:
            step = max(step * 0.5, 1e-12)

        if iteration % 5000 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.01 + 1e-6)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score

def _mirror_descent(start_seq, deadline, best_seq, best_score):
    """Exponentiated‑gradient (mirror) descent on the simplex."""
    a = start_seq.copy()
    n = a.shape[0]
    eta = 0.5                # initial learning rate
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Mirror it={iteration} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # multiplicative update
        a_new = a * np.exp(-eta * grad)
        total = a_new.sum()
        if total == 0:
            # fall back to simple projection
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / total

        new_score = float(2.0 * n * _conv_self(a_new).max())
        if new_score < cur_score:
            a = a_new
            eta = min(eta * 1.03, 2.0)   # slowly increase step if we improve
        else:
            eta = max(eta * 0.7, 1e-4)   # otherwise shrink it

        if iteration % 500 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.01 + 1e-6)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score

def _softmax_gradient_descent(start_seq, deadline, best_seq, best_score):
    """
    Gradient descent on a smooth approximation of the max‑convolution
    using a temperature‑controlled soft‑max weighting.
    """
    a = start_seq.copy()
    n = a.shape[0]
    step = 0.3               # nominal step size (in the Euclidean space)
    min_step = 1e-12
    iteration = 0
    tau_factor = 0.985        # geometric cooling of the temperature

    # initialise temperature loosely from the current spread of the convolution
    conv = _conv_self(a)
    tau = (conv.max() - conv.min()) * 0.5 + 1e-9
    if tau <= 0:
        tau = conv.max() * 0.1 + 1e-9

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SoftGrad it={iteration} score={cur_score:.6f}")

        # smooth weighting of the convolution entries
        max_c = conv.max()
        w = np.exp((conv - max_c) / tau)           # stable: max exponent = 0
        w_sum = w.sum()
        if w_sum == 0:
            # fall back to ordinary sub‑gradient
            grad = _subgradient_maxconv(a, conv)
        else:
            w_norm = w / w_sum
            a_rev = a[::-1]
            grad_full = _conv_fft(w_norm, a_rev)   # length 3n‑2, we need the middle n entries
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # line‑search (projected back‑tracking)
        gnorm = np.linalg.norm(grad) + 1e-12
        step_len = step / gnorm

        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_score = float(2.0 * n * _conv_self(a_new).max())
            if new_score < cur_score or step_len <= min_step:
                break
            step_len *= 0.5

        if new_score < cur_score:
            a = a_new
            step = min(step * 1.04, 1.0)   # keep the step aggressive
        else:
            step = max(step * 0.6, 1e-12)

        # cool the temperature – it never goes below a tiny floor
        tau = max(tau * tau_factor, 1e-12)

        if iteration % 800 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 0.01 + 1e-6)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score

def _hill_climb(start_a, deadline):
    """Very small deterministic moves that transfer mass from high‑gradient
    to low‑gradient indices."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score
