"""
Propose a step‑function (non‑negative vector with unit sum) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm follows three stages:

1️⃣  Random Dirichlet initialisations – a quick sweep of diverse concentration
    parameters to obtain a good starting point.

2️⃣  Aggressive simulated‑annealing on the simplex.  Temperature starts
    relatively high (≈30 % of the current score) and decays very slowly.
    Each move transfers a configurable fraction of mass from the index that
    contributes most to the current maximal convolution (sub‑gradient) to the
    index that contributes least.  Every few thousand steps a global “jump”
    (convex combination with a fresh Dirichlet draw) is tried to escape
    plateaus.

3️⃣  Smoothed projected gradient descent.  The non‑smooth max is approximated
    by a log‑sum‑exp (temperature λ).  Starting with a small λ the method
    performs a standard back‑tracking line‑search on the smooth surrogate,
    projecting each iterate onto the probability simplex.  λ is increased
    gradually so the algorithm ends up minimising the true max‑convolution.

All heavy work (convolutions, gradients) uses FFTs; simplex projection uses
the O(n) algorithm of Duchi et al.  The routine respects a 1000 s budget and
returns the best vector found so far.
"""

import time
import numpy as np
from scipy.optimize import linprog   # optional, not used in the final scheme

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, so denominator equals 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):

            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4g} score={sc:.6f}")
    return best_seq, best_score

def _simulated_annealing_improved(a0: np.ndarray, cur_score: float, deadline: float):
    """Simulated annealing with a higher starting temperature and occasional global jumps."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # higher initial temperature – 30 % of the current objective (instead of 2 %)
    T = max(cur_score * 0.30, 1e-8)
    decay = 0.99995          # very slow cooling
    max_frac_start = 0.30   # keep the large early moves
    max_frac_end = 0.01

    # statistics for occasional global jumps
    steps_since_improve = 0
    GLOBAL_JUMP_EVERY = 4000   # try a global jump after this many steps

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # compute current sub‑gradient to locate source / destination indices
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # fraction of source mass that may be moved this step
        t_rem = deadline - time.time()
        total_rem = deadline - start_time
        frac = max_frac_end + (max_frac_start - max_frac_end) * max(t_rem / total_rem, 0.0)

        delta = a[src] * rng.random() * frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # safeguard against rounding errors
        if cand[src] < 0.0:
            continue

        cand_score = _objective_score(cand)
        diff = cand_score - cur_score

        # Metropolis acceptance
        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            steps_since_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best {best_score:.6f}")
        else:
            steps_since_improve += 1

        # occasional global jump to shake the system
        if iter_cnt % GLOBAL_JUMP_EVERY == 0:
            # blend with a new Dirichlet sample
            alpha = 0.07                # how much new randomness we inject
            rand = rng.gamma(shape=0.3, scale=1.0, size=n)   # heavy‑tailed random
            rand /= rand.sum()
            jump_cand = (1 - alpha) * a + alpha * rand
            jump_cand = _project_simplex(jump_cand)
            jump_score = _objective_score(jump_cand)
            if jump_score < cur_score:
                a = jump_cand
                cur_score = jump_score
                steps_since_improve = 0
                if jump_score < best_score:

                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SA global jump improved to {best_score:.6f}")

def _smooth_gradient_descent(a0: np.ndarray, cur_score: float, deadline: float):
    """
    Gradient descent on the λ‑smoothed approximation of max(conv(a,a)).
    The smoothing parameter λ is increased slowly, turning the surrogate
    into the true max as the optimisation proceeds.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score

    # initial smoothing temperature
    lam = 5.0                # start with a fairly smooth landscape
    max_lam = 250.0
    lam_factor = 1.25        # how fast we increase λ
    lam_inc_every = 25.0    # seconds between λ increases

    # step‑size handling
    step = 1.0               # initial step length in the gradient direction
    min_step = 1e-10
    max_step = 5.0

    # timing helpers for λ schedule
    next_lam_inc = time.time() + lam_inc_every

    while time.time() < deadline:
        # ----- compute smoothed gradient -----
        conv = _conv_self(a)               # length 2n‑1
        max_c = conv.max()
        # avoid overflow – shift by max_c
        w = np.exp(lam * (conv - max_c))
        p = w / w.sum()                     # weights over shifts, sum to 1

        # gradient of the smoothed objective:
        #   grad = 2 * Σ_k p_k * shift_k(a)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        Af = np.fft.rfft(a_rev, n=L)
        Bf = np.fft.rfft(p, n=L)
        conv_rev = np.fft.irfft(Af * Bf, n=L)[: 3 * n - 2]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

        # ----- back‑tracking line search -----
        improved = False
        cur_obj = cur_score
        for factor in (1.0, 0.5, 0.25, 0.125):
            step_try = step * factor
            cand = a - step_try * grad
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_score = _objective_score(cand)
            if cand_score < cur_obj - 1e-12:
                a = cand
                cur_score = cand_score
                improved = True
                # adapt step size
                step = min(step_try * 1.2, max_step)
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Smooth GD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, min_step)

        # ----- increase λ slowly -----
        if time.time() >= next_lam_inc and lam < max_lam:
            lam = min(lam * lam_factor, max_lam)
            next_lam_inc = time.time() + lam_inc_every

    return best_a, best_score

    (default 1000 s) and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Choose a working dimension.  4096 works well with FFT‑based tools.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) if it exists
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisers (≈20 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 20.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=400, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] After Dirichlet best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Aggressive simulated annealing
    # ------------------------------------------------------------------
    anneal_deadline = min(deadline, start_time + 620.0)   # majority of the budget
    if time.time() < anneal_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting aggressive simulated annealing")
        best_seq, best_score = _simulated_annealing_improved(best_seq, best_score, anneal_deadline)
        print(f"[{time.time() - start_time:.2f}s] Annealing finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed projected gradient descent (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed gradient descent")
        best_seq, best_score = _smooth_gradient_descent(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Gradient descent finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```

This implementation expands on the earlier approach by

* widening the Dirichlet‑search space,
* using a higher initial temperature and very slow cooling in the annealer,
* injecting occasional global jumps to avoid stagnation,
* and finally employing a smoothed log‑sum‑exp surrogate for the maximum, with a
  back‑tracking line‑search and an adaptive smoothing parameter λ.

All three phases respect the 1000 s budget and return the best vector discovered.
