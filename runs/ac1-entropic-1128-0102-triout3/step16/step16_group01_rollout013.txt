def _adam_optimize(a_init, L_fft, L_grad, deadline, start_time, best_seq, best_score, rng):
    n = a_init.shape[0]
    u = np.log(a_init + 1e-12) # log-prob
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    it = 0
    beta = 5.0
    beta_max = 256.0
    beta_inc = 0.001  # incremental per iteration
    while time.time() < deadline:
        it += 1
        # compute probabilities
        u_max = u.max()
        exp_u = np.exp(u - u_max)
        a = exp_u / exp_u.sum()
        # compute conv
        conv = _conv_self_fft(a, L_fft)
        # compute true objective
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={score:.6f}")
        # smoothing beta schedule
        beta = min(beta_max, beta * (1 + beta_inc))
        # compute softmax weights
        p = _softmax_weights(conv, beta)
        # gradient wrt a
        grad_a = _grad_softmax(a, p, L_grad)
        # gradient wrt u
        dot = np.dot(grad_a, a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)
        # jitter occasional
        if it % 2000 == 0:
            u += 1e-6 * rng.standard_normal(n)
    return best_seq, best_score

def _subgrad_polish(a_init, L_fft, deadline, start_time, best_seq, best_score, rng):

def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of  Σ_k p_k conv_k (soft‑max objective):
        g_i = 2 * Σ_j a_j * p_{i+j}
    Computed as a cross‑correlation via FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)

Optimizes a step function on the simplex to minimise 2·n·max(conv(a,a))/(sum(a)**2).
The method combines a dense parametric sweep, Adam optimisation on a
soft‑max smoothed objective and a short projected sub‑gradient polish.
"""

import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v):
    # Euclidean projection onto the probability simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a, L_fft):
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

def _make_shape(n, r, p):
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

def _mix_with_uniform(shape, lam):
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

def _softmax_weights(conv, beta):
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()

def _grad_softmax(a, p, L_grad):
    # returns gradient of Σ_k p_k conv_k w.r.t a
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    corr = np.fft.irfft(Af * Pf, n=L_grad)
    # keep only the part that contributes to indices 0..n-1
    # length of needed segment = (2n-1)+(n-1) = 3n-2
    corr = corr[: (2 * n - 1 + n - 1)]
    start = n - 1
    g = corr[start:start + n][::-1]
    return 2.0 * g

def _subgrad_max_conv(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

def _logits_to_probs(u):
    u_max = u.max()
    exp_u = np.exp(u - u_max)
    return exp_u / exp_u.sum()

def _adam_optimize(a_start, L_fft, L_grad, deadline, start_time,
                   best_seq, best_score, rng):
    n = a_start.shape[0]
    # initialise logits (avoid log(0))
    u = np.log(np.maximum(a_start, 1e-15))
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    it = 0
    beta = 5.0
    beta_max = 256.0
    beta_inc = 0.001  # multiplicative increment each iteration
    while time.time() < deadline:
        it += 1
        # probabilities
        a = _logits_to_probs(u)
        # full conv and current objective
        conv = _conv_self_fft(a, L_fft)
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={score:.6f}")
        # increase smoothing temperature gradually
        beta = min(beta_max, beta * (1.0 + beta_inc))
        # smooth‑max weights
        p = _softmax_weights(conv, beta)
        # gradient w.r.t a
        grad_a = _grad_softmax(a, p, L_grad)
        # gradient w.r.t logits u
        dot = np.dot(grad_a, a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)
        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            u += 1e-6 * rng.standard_normal(n)
    return best_seq, best_score

def _subgrad_polish(a_start, L_fft, deadline, start_time,
                    best_seq, best_score, rng):
    a = a_start.copy()
    n = a.shape[0]
    it = 0
    step = 0.3
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)
        # simple back‑tracking line‑search
        step_try = step
        improved = False
        for _ in range(8):
            cand = a - step_try * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                step = step_try * 1.1  # slightly increase step after success
                improved = True
                print(f"[{time.time() - start_time:.2f}s] sub‑polish it={it} step={step_try:.5e} score={cand_score:.6f}")
                break
            step_try *= 0.5
        if not improved:
            step *= 0.7
        # jitter every few thousand iterations
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
    return best_seq, best_score

def _final_hill_climb(a_start, L_fft, deadline, start_time,
                      best_seq, best_score, rng):
    a = a_start.copy()
    n = a.shape[0]
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 5e-4  # up to 0.05% of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_score:.6f}")
    return best_seq, best_score

"""
Optimises a non‑negative step‑function (a probability vector) to minimise

        2·n·max(conv(a, a)) / (sum(a)**2)

The optimiser works in four stages:

1️⃣  **Dense parametric sweep** – explores geometric × power shapes and mixtures
    with the uniform distribution; top‑k candidates are retained.

2️⃣  **Adam optimisation on a smooth‑max objective** – a continuation on the
    soft‑max temperature β, with automatic learning‑rate adaptation and Euclidean
    projection onto the simplex hidden inside the soft‑max parametrisation.

3️⃣  **Projected sub‑gradient polish** – a classic back‑tracking line‑search on
    the true L∞‑objective to tighten the final bound.

4️⃣  **Final random hill‑climb** – tiny mass‑transfer moves with simulated‑annealing
    acceptance for the last seconds.

All phases respect a 1000 s wall‑time budget, keep the vector on the simplex,
and return the best sequence found as a plain Python list of floats.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    if cond.any():
        rho = np.nonzero(cond)[0][-1]

        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a: np.ndarray, L_fft: int) -> float:

    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)               # stabilise numerics

def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """p_k = exp(beta·conv_k) / Σ exp(...) – numerically stable."""
    shift = conv.max()

def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k·conv_k with respect to a.
    g_i = 2·Σ_j a_j·p_{i+j}.
    Computed via FFT‑based cross‑correlation.

    The probabilities are represented as logits u (soft‑max → a) which
    automatically enforces positivity and unit sum.

def _final_hill_climb(a_start: np.ndarray,
                      L_fft: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very small stochastic mass‑transfer moves until the budget expires."""
    a = a_start.copy()
    n = a.shape[0]

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 5e-4   # ≤0.05 % of source mass
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_score = _eval_fast(a_prop, L_fft)
        if new_score < best_score:
            best_score = new_score
            best_seq = a_prop
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    Returns a plain list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 provides a finer discretisation without excess cost.
    # ------------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (height_sequence_1) if available.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size < n:
                starter = np.concatenate([starter, np.zeros(n - starter.size)])
            elif starter.size > n:
                starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Dense parametric sweep (≈45 s)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 45.0
    max_cand = 12
    candidates = []       # list of (score, vector)

    r_grid = np.logspace(np.log10(0.90), np.log10(0.9999), 32)
    p_grid = np.linspace(-5.0, 5.0, 41)
    lam_grid = np.linspace(0.0, 1.0, 13)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            shape = _make_shape(n, r, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(shape, lam)
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} p={p:.3f} λ={lam:.2f} score={sc:.6f}")
                # keep top candidates
                if len(candidates) < max_cand:
                    candidates.append((sc, cand.copy()))
                else:
                    worst_idx = max(range(len(candidates)),
                                   key=lambda i: candidates[i][0])
                    if sc < candidates[worst_idx][0]:
                        candidates[worst_idx] = (sc, cand.copy())

    # a few random extra shapes for robustness
    for _ in range(2000):
        if time.time() >= sweep_deadline:
            break
        r = rng.uniform(0.90, 0.9999)
        p = rng.uniform(-5.0, 5.0)
        lam = rng.uniform(0.0, 1.0)
        shape = _make_shape(n, r, p)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] rnd shape score={sc:.6f}")
        if len(candidates) < max_cand:
            candidates.append((sc, cand.copy()))
        else:
            worst_idx = max(range(len(candidates)),
                           key=lambda i: candidates[i][0])
            if sc < candidates[worst_idx][0]:
                candidates[worst_idx] = (sc, cand.copy())

    # sort and keep the best few
    candidates.sort(key=lambda x: x[0])
    top_seqs = [seq for _, seq in candidates[:max_cand]]

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimisation (smooth‑max continuation)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 120.0:
        adam_budget = remaining * 0.6
        adam_deadline_global = time.time() + adam_budget

        # run Adam on each top sequence while we have budget
        for seq in top_seqs:
            now = time.time()
            if now >= adam_deadline_global:
                break
            per_deadline = min(now + 30.0, adam_deadline_global)
            best_seq, best_score = _adam_optimize(seq, L_fft, L_grad,
                                                  per_deadline, start_time,
                                                  best_seq, best_score, rng)

        # a few random restarts starting from the current best
        while time.time() < adam_deadline_global - 5.0:
            a0 = best_seq * (1.0 + 0.02 * rng.standard_normal(n))
            a0 = np.maximum(a0, 0.0)
            tot = a0.sum()
            if tot > 0.0:
                a0 /= tot
            else:
                a0 = uniform.copy()
            best_seq, best_score = _adam_optimize(a0, L_fft, L_grad,
                                                  adam_deadline_global,
                                                  start_time,
                                                  best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient polish
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + min(20.0, remaining * 0.5)
        best_seq, best_score = _subgrad_polish(best_seq, L_fft,
                                               sub_deadline, start_time,
                                               best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Final random hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.0:
        best_seq, best_score = _final_hill_climb(best_seq, L_fft,
                                                 deadline, start_time,
                                                 best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
