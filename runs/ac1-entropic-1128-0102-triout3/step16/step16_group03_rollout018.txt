def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1
    n = 4096
    L_fft = 1 << ((2*n-1).bit_length())
    # uniform baseline
    uniform = np.full(n, 1.0/n)
    best_seq = uniform
    best_score = _objective_score(uniform, n)
    print(f"[{time.time()-start:.2f}s] uniform baseline score {best_score:.6f}")

    # parameter sweep for power-law decreasing
    p_vals = np.linspace(0.2, 0.8, 61) # step 0.01? maybe 0.01 step for more refine
    for idx, p in enumerate(p_vals):
        if time.time() >= deadline: break
        i = np.arange(1, n+1, dtype=float)
        a = i ** (-p)
        a = a / a.sum()
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time()-start:.2f}s] power p={p:.4f} → score {sc:.6f}")

    # same for finer around best p region
    # maybe use golden-section or local gradient descent on p
    # can use simple refinement:
    best_p = None # we can track best p if needed
    # but we already tracked best_seq.

    # exponentials
    beta_vals = np.linspace(0.0001, 0.01, 100) # range
    for beta in beta_vals:
        if time.time() >= deadline: break
        a = np.exp(-beta * np.arange(n, dtype=float))
        a = a / a.sum()
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time()-start:.2f}s] exp beta={beta:.6f} → score {sc:.6f}")

    # local hill climbing: move epsilon mass from max to min
    a = best_seq.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    while time.time() < deadline:
        i = np.argmax(a)
        j = np.argmin(a)
        if a[i] <= 1e-12: break
        eps = min(0.01 * a[i], 1e-4)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        sc = _objective_score(a_new, n)
        if sc < cur_score:
            a, cur_score = a_new, sc
            best_seq, best_score = a_new.copy(), sc
            print(f"[{time.time()-start:.2f}s] hill-climb improved to {sc:.6f}")
    # final normalization (though already normalized)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq/total

    print(f"[{time.time()-start:.2f}s] finished best score {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Hybrid search for a low‑objective step function.

The optimizer works in three very cheap stages:

1️⃣  **Parametric sweep** – test simple analytically‑motivated families:
    * Power‑law decreasing a[i] ∝ (i+1)^{-p}
    * Exponential a[i] ∝ exp(-β·i)
    The best candidate (lowest 2·n·max(conv)) is kept.

2️⃣  **Fine local refinement** – a tiny deterministic mass transfer
    (move a fraction from the current maximum entry to the current
    minimum entry).  Accept only improving moves; this quickly reduces
    the remaining tiny excess of the maximal convolution.

3️⃣  **Mixture with the uniform distribution** – a short golden‑section
    search over λ∈[0,1] for seq = λ·best + (1‑λ)·uniform, keeping the
    best mixture.  This smoothes the shape and can shave a few 10⁻⁴
    off the final score.

All stages respect a global time budget (default 1000 s).  The routine
returns the best sequence found (normalised to sum 1) as a plain Python
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (same as in the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score 2·n·max(a*a) for a already normalised to sum 1."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric sweep (power‑law & exponential families)
# ----------------------------------------------------------------------
def _search_powerlaw(n: int, deadline: float, start: float,
                     best_seq: np.ndarray, best_score: float):
    """Sweep a_i ∝ (i+1)^{-p} for p in a sensible interval."""
    # coarse grid first
    for p in np.linspace(0.20, 0.80, 61):          # step ≈0.01
        if time.time() >= deadline:
            break
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p)
        a /= a.sum()
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start:.2f}s] power p={p:.4f} → score {sc:.6f}")

    # fine refinement around the best p found so far
    # (use a simple golden‑section style local search)
    # first locate best p among the evaluated ones
    p_vals = np.linspace(0.20, 0.80, 61)
    idx_best = np.argmin([_objective_score((np.arange(1, n + 1) ** (-p)) /
                                            ((np.arange(1, n + 1) ** (-p)).sum()), n)
                          for p in p_vals])
    p_center = p_vals[idx_best]

    lo, hi = max(0.01, p_center - 0.05), min(0.99, p_center + 0.05)
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    while time.time() < deadline and hi - lo > 1e-4:
        p1 = hi - (hi - lo) / phi
        p2 = lo + (hi - lo) / phi
        a1 = (np.arange(1, n + 1) ** (-p1))
        a1 /= a1.sum()
        sc1 = _objective_score(a1, n)
        a2 = (np.arange(1, n + 1) ** (-p2))
        a2 /= a2.sum()
        sc2 = _objective_score(a2, n)

        if sc1 < sc2:
            hi = p2
            if sc1 < best_score:
                best_score, best_seq = sc1, a1.copy()
                print(f"[{time.time() - start:.2f}s] power fine p={p1:.5f} → score {sc1:.6f}")
        else:
            lo = p1
            if sc2 < best_score:
                best_score, best_seq = sc2, a2.copy()
                print(f"[{time.time() - start:.2f}s] power fine p={p2:.5f} → score {sc2:.6f}")

    return best_seq, best_score


def _search_exponential(n: int, deadline: float, start: float,
                       best_seq: np.ndarray, best_score: float):
    """Sweep a_i ∝ exp(-β·i) for a small set of β."""
    # a sensible β‑range (β·n is of order 1)
    for beta in np.linspace(0.0002, 0.0200, 81):
        if time.time() >= deadline:
            break
        a = np.exp(-beta * np.arange(n, dtype=float))
        a /= a.sum()
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start:.2f}s] exp β={beta:.6f} → score {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Tiny deterministic hill‑climbing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, n: int, L_fft: int,
                start: float, deadline: float,
                best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Move a small fraction from the largest entry to the smallest.
    Accept only improvements; repeat until time runs out."""
    a = a0.copy()
    cur_score = best_score
    while time.time() < deadline:
        i = int(np.argmax(a))
        j = int(np.argmin(a))
        if a[i] <= 1e-14:
            break
        eps = min(0.02 * a[i], 1e-5)          # a tiny chunk of mass
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # re‑normalise (the total mass is unchanged up to round‑off)
        sc = _objective_score(a_new, n)
        if sc < cur_score:
            a, cur_score = a_new, sc
            if sc < best_score:
                best_score, best_seq = sc, a_new.copy()
                print(f"[{time.time() - start:.2f}s] hill‑climb improved to {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Mixing with uniform (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Optimise λ∈[0,1] for λ·a + (1‑λ)·uniform."""
    uniform = np.full(n, 1.0 / n)

    def score_lambda(lam):
        seq = lam * a + (1.0 - lam) * uniform
        seq /= seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    while time.time() < deadline and hi - lo > 1e-5:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = score_lambda(lam1)
        sc2 = score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = (lam1 * a + (1.0 - lam1) * uniform)
                best_seq /= best_seq.sum()
                print(f"[{time.time() - start:.2f}s] mix λ={lam1:.5f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = (lam2 * a + (1.0 - lam2) * uniform)
                best_seq /= best_seq.sum()
                print(f"[{time.time() - start:.2f}s] mix λ={lam2:.5f} → score={sc2:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence().  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25          # safety margin

    # problem size – 4096 works well and matches the previous best
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ---- baseline (uniform) -------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score = {best_score:.6f}")

    # ---- 1️⃣  Parametric sweeps ---------------------------------------------
    phase1_deadline = start_time + 0.30 * (deadline - start_time)
    if time.time() < phase1_deadline:
        best_seq, best_score = _search_powerlaw(
            n, phase1_deadline, start_time, best_seq, best_score
        )
    if time.time() < phase1_deadline:
        best_seq, best_score = _search_exponential(
            n, phase1_deadline, start_time, best_seq, best_score
        )
    # ---- 2️⃣  Tiny hill‑climb ------------------------------------------------
    phase2_deadline = start_time + 0.80 * (deadline - start_time)
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting deterministic hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, n, L_fft, start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score ≈ {best_score:.6f}")

    # ---- 3️⃣  Fine mixing with the uniform distribution ------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform(
            best_seq, n, start_time, deadline, best_seq, best_score
        )

    # ---- final normalisation ------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # return plain Python list (evaluate_sequence expects a list)
    return [float(v) for v in best_seq]
