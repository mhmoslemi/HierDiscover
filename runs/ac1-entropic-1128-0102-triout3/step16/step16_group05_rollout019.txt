def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    n = 8192

    # arcsine shape
    idx = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(idx * (n - idx + 0.0))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    a = a / a_sum  # normalize sum=1

    best_seq = a.copy()
    best_score = _objective_score(best_seq)

    # Simulated annealing parameters
    T0 = 1.0
    T_end = 1e-4
    # schedule based on remaining fraction time
    while time.time() < deadline:
        # progress fraction
        t = (time.time() - start_time) / budget_s
        T = max(T_end, T0 * (1 - t))

        # propose move
        src, dst = rng.choice(n, size=2, replace=False)
        if best_seq[src] == 0:
            continue
        eps = best_seq[src] * (rng.random() * 0.01 + 1e-8)
        if eps <= 0:
            continue
        # create candidate
        cand = best_seq.copy()
        cand[src] -= eps
        cand[dst] += eps
        # ensure nonnegative and projection onto simplex (maybe not needed)
        # compute score
        cur_score = _objective_score(cand)
        delta = cur_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            best_seq = cand
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] New best {best_score:.6f}")

    # final projection onto simplex
    best_seq = _project_simplex(best_seq)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]

"""
search for a non‑negative step function that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm:
1. Initialise a distribution with an arcsine‑type shape (large mass near the
   ends, small mass in the centre).  This shape tends to flatten the
   self‑convolution.
2. Run a simulated‑annealing / Metropolis‑Hastings walk on the simplex:
   * pick two indices i→j,
   * move a tiny fraction of the mass from i to j,
   * accept the move if the objective improves or with probability
     exp(‑Δ / T) where T is a linearly cooling temperature.
   The walk keeps the sum exactly 1 and never creates negative entries.
3. Keep the best sequence seen so far and finally project it onto the
   probability simplex (numerical drift).
4. Return the best sequence as a plain Python list.

The routine respects a 1000 s wall‑clock budget, prints intermediate
scores and never uses closures or lambda expressions.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising the evaluation function.
    The routine uses a simulated‑annealing walk starting from an arcsine‑type
    distribution and respects the 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more freedom to flatten the convolution.
    # ------------------------------------------------------------------
    n = 8192                               # feel free to change; 8192 works in <1 s per eval
    # arcsine‑type initialisation (large values near the ends, small in the centre)
    idx = np.arange(1, n + 1, dtype=float) - 0.5
    a = 1.0 / np.sqrt(idx * (n + 1 - idx))
    a = np.maximum(a, 0.0)
    a = a / a.sum()                        # normalise to sum=1

    try:

        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.ndim == 1 and hs.size == n:

            hs = hs / hs.sum()
            a = 0.5 * a + 0.5 * hs
            a = a / a.sum()
    except Exception:
        pass

    cur = a.copy()
    cur_score = _objective_score(cur)

    best = cur.copy()
    best_score = cur_score
    print(f"[0.00s] Initial score = {cur_score:.6f}")

    # ------------------------------------------------------------------

    T0 = 0.12          # initial temperature
    T_min = 1e-7       # final temperature
    no_improve = 0

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # linear cooling

        T = max(T_min, T0 * (1.0 - elapsed))

    else:

    return [float(x) for x in best]
