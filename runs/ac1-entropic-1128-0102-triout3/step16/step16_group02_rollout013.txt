def _subgradient_maxconv(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

            a += np.random.uniform(-1e-6, 1e-6, size=n)
            a = np.clip(a, 0, None)
            a /= a.sum()
            no_improve += 1
        else:
            no_improve = 0
        if no_improve > 2000:
            break
    return best_seq, best_score

"""
search for a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The strategy is a combination of:

1.  Very broad parametric sweeps (geometric, power‑law, linear,
    log‑decay, inverse‑power, asymmetric Beta, two‑block).
2.  Hundreds of random Dirichlet draws with a wide range of
    concentration parameters.
3.  A fast greedy “pairwise mass transfer” local optimiser that
    repeatedly moves mass from a high‑gradient index to a low‑gradient
    index, trying a few step sizes and accepting the best improvement.
4.  A final projected‑gradient descent (PGD) polish.

The whole pipeline respects a 1000 s budget and prints intermediate
scores so you can see the progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-6, -0.5, 400)          # finer & wider
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 13):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-4.0, 4.0, 301):          # wider exponent range
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:

        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score

def _sweep_inv_power(n, best_seq, best_score, deadline):

    for p in np.linspace(0.5, 6.0, 150):

def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.1, 5.0, 31)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends, variable block length and mass split."""
    for L_frac in np.linspace(0.01, 0.5, 50):          # allow up to half the vector
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.05, 0.95, 19):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score

def _projected_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        # descent direction that keeps the sum zero
        d = -(grad - grad.mean())

        # largest feasible step without breaking non‑negativity
        idx_neg = np.where(d < 0)[0]
        if idx_neg.size:
            max_step = np.min(-a[idx_neg] / d[idx_neg])
        else:
            max_step = 1.0
        step = max_step * 0.9

        # back‑tracking line search
        while step > 1e-14:
            a_new = a + step * d
            if np.any(a_new < 0):
                step *= 0.5
                continue
            conv_new = _conv_self(a_new)
            if float(2.0 * n * conv_new.max()) <= cur_score:
                a = a_new
                break
            step *= 0.5

        # if we are stuck, inject a negligible random jitter
        if step <= 1e-14:
            eps = 1e-7
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣ Greedy pairwise mass‑transfer optimiser
# ----------------------------------------------------------------------
def _pairwise_greedy(a0, deadline, max_iters=5000):
    """Repeatedly move mass from a high‑gradient index to a low‑gradient one."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())

        # keep global best up‑to‑date
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Greedy it={it:,} new best {best_score:.6f}")
            no_improve = 0

        # sub‑gradient gives a good proxy for “how much this index hurts the max”
        grad = _subgradient_maxconv(a, conv)

        # candidate sources / destinations (few top / bottom entries)
        src_candidates = np.argsort(-grad)[:3]
        dst_candidates = np.argsort(grad)[:3]

        improved = False
        for src in src_candidates:
            if a[src] <= 1e-12:
                continue
            for dst in dst_candidates:
                if src == dst:
                    continue
                # try a handful of fractions of the available mass at src
                delta_max = a[src]
                best_local_score = cur_score
                best_delta = 0.0
                for frac in (0.5, 0.8, 0.2, 1.0):
                    delta = delta_max * frac
                    if delta <= 0:
                        continue
                    a_tmp = a.copy()
                    a_tmp[src] -= delta
                    a_tmp[dst] += delta
                    sc = _objective_score(a_tmp)
                    if sc < best_local_score:
                        best_local_score = sc
                        best_delta = delta
                if best_delta > 0:
                    a[src] -= best_delta
                    a[dst] += best_delta
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # random jitter to escape plateaux
            a += np.random.uniform(-1e-6, 1e-6, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            no_improve += 1
        else:
            no_improve = 0

        if no_improve > 2000:          # give up if we’re stuck
            break

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm combines broad parameter sweeps, many Dirichlet draws,
    a greedy pairwise optimiser and a final PGD polish.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1     # safety margin

    # problem size (the original run used 4096)
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (if a previous construction is available)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣ Broad parametric sweep (reserve ~400 s for later phases)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 400.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_linear(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_log_decay(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_inv_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 2️⃣ Random Dirichlet initialisers
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 3️⃣ Greedy pairwise optimiser (the main improvement engine)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting greedy pairwise optimiser")
        best_seq, best_score = _pairwise_greedy(best_seq, deadline, max_iters=8000)
        print(f"[{time.time() - start_time:.2f}s] Greedy finished best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final PGD polish (if a small slice of time remains)
    # ------------------------------------------------------------------
    pgd_deadline = deadline - 10.0
    if time.time() < pgd_deadline:
        print(f"[{time.time() - start_time:.2f}s] Final projected gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq, pgd_deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
