def build_A(f):

def _optimal_mix_lambda(conv_f, conv_g, conv_fg, n, max_iter=25, tol=1e-8):
    phi = (1 + math.sqrt(5)) / 2
    lo, hi = 0.0, 1.0
    # Evaluate at endpoints for maybe best
    def score(t):
        conv_t = (1 - t) ** 2 * conv_f + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_g
        return 2.0 * n * np.max(conv_t)
    best_t = 0.0
    best_sc = score(0.0)
    # also evaluate at 1
    sc1 = score(1.0)
    if sc1 < best_sc:
        best_sc, best_t = sc1, 1.0
    for _ in range(max_iter):
        t1 = hi - (hi - lo) / phi
        t2 = lo + (hi - lo) / phi
        s1 = score(t1)
        s2 = score(t2)
        if s1 < s2:
            hi = t2
            if s1 < best_sc:
                best_sc, best_t = s1, t1
        else:
            lo = t1
            if s2 < best_sc:
                best_sc, best_t = s2, t2
        if hi - lo < tol:
            break
    return best_t, best_sc
```

def _hill_climb_refine(a, n, L_fft, deadline, start_time, best_seq, best_score):
    # similar to earlier code but uses simple hill climb with simulated annealing.
    a_cur = a.copy()
    cur_score = 2.0 * n * _conv_self(a_cur, L_fft).max()
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick donor and receiver indices.
        i = np.argmax(a_cur) if rng.random() < 0.7 else rng.integers(n)
        j = np.argmin(a_cur) if rng.random() < 0.7 else rng.integers(n)
        if i == j or a_cur[i] <= 1e-14:
            continue
        eps = min(0.02 * a_cur[i], 1e-4)  # ensure small mass move
        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps
        new_score = 2.0 * n * _conv_self(a_new, L_fft).max()
        if new_score < cur_score or np.random.rand() < 0.001:
            a_cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
    return best_seq, best_score

"""
LP‑based alternating optimisation for the step‑function evaluation problem.

The algorithm iteratively:
1. Computes M = max(f*f) for the current normalised sequence f.
2. Solves the linear programme
       max   Σ g_j
       s.t.  (f * g)_k ≤ M   for all k,
            g_j ≥ 0.
   The constraints are linear in g because the convolution (f*g) is a
   Toeplitz matrix acting on g.  The LP is solved with HiGHS (SciPy).

3. Normalises the obtained g to have unit ℓ¹‑norm and finds the optimal
   mixing coefficient λ∈[0,1] which minimises the objective for the convex
   combination (1‑λ)·f + λ·g by a short golden‑section search.

4. Updates f ← (1‑λ)·f + λ·g and repeats until the time budget is exhausted.

Finally a cheap hill‑climbing refinement (mass‑transfer moves with simulated‑
annealing acceptance) is performed for any remaining time.

All sequences are kept non‑negative and normalised to sum 1, thus the
objective value equals `2·n·max(conv(f,f))`, i.e. the same quantity used by
`evaluate_sequence`.  The routine respects the 1000 s budget and returns the
best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# 0️⃣ Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _conv_two(a: np.ndarray, b: np.ndarray, L_fft: int) -> np.ndarray:
    """Linear convolution a*b via FFT (length 2·len(a)‑1, a and b same length)."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_fft)
    Bf = np.fft.rfft(b, n=L_fft)
    conv = np.fft.irfft(Af * Bf, n=L_fft)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score 2·n·max(conv(a,a)) for a normalised (sum=1) sequence."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣ Build the Toeplitz constraint matrix for the LP  (f * g ≤ M)
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Returns a sparse matrix A (shape (2n‑1, n)) such that
    (A @ g)[k] = (f * g)[k] for all k.
    The matrix consists of n constant diagonals with values f[i].
    """
    n = f.shape[0]
    K = 2 * n - 1
    # each diagonal i contains the constant f[i] repeated n times
    data = np.tile(f, (n, 1))                     # shape (n, n)
    offsets = np.arange(n, dtype=np.int32)        # i = row – col
    A = sp.dia_matrix((data, offsets), shape=(K, n), dtype=np.float64)
    return A.tocsr()


def _solve_lp_max_sum_g(f: np.ndarray, M: float) -> linprog.Result:
    """
    Solve  max Σ g_j  s.t.  (f * g)_k ≤ M  for all k,  g ≥ 0.
    Returns the SciPy optimisation result.
    """
    n = f.shape[0]
    A = _build_constraint_matrix(f)
    K = A.shape[0]
    # right‑hand side; a tiny epsilon avoids numeric infeasibility
    b = np.full(K, M + 1e-12, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximise sum ⇔ minimise –sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    return res


# ----------------------------------------------------------------------
# 2️⃣  One‑dimensional line search for optimal mixing λ
# ----------------------------------------------------------------------
def _optimal_mix_lambda(conv_f: np.ndarray,
                        conv_g: np.ndarray,
                        conv_fg: np.ndarray,
                        n: int,
                        max_iter: int = 25,
                        tol: float = 1e-8) -> tuple[float, float]:
    """
    Golden‑section search for λ∈[0,1] minimising
        2·n·max( (1‑λ)²·conv_f + 2λ(1‑λ)·conv_fg + λ²·conv_g )
    Returns (λ_opt, best_score).
    """
    phi = (1.0 + math.sqrt(5.0)) / 2.0

    def _score(lam: float) -> float:
        conv = ((1.0 - lam) ** 2) * conv_f + (2.0 * lam * (1.0 - lam)) * conv_fg + (lam ** 2) * conv_g
        return 2.0 * n * np.max(conv)

    lo, hi = 0.0, 1.0
    best_lam = lo
    best_sc = _score(lo)

    sc_hi = _score(hi)
    if sc_hi < best_sc:
        best_sc, best_lam = sc_hi, hi

    for _ in range(max_iter):
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score(lam1)
        sc2 = _score(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_sc:
                best_sc, best_lam = sc1, lam1
        else:
            lo = lam1
            if sc2 < best_sc:
                best_sc, best_lam = sc2, lam2

        if hi - lo < tol:
            break
    return best_lam, best_sc


# ----------------------------------------------------------------------
# 3️⃣  Cheap hill‑climbing refinement (mass transfer moves)
# ----------------------------------------------------------------------
def _hill_climb_refine(a0: np.ndarray,
                       n: int,
                       L_fft: int,
                       deadline: float,
                       start_time: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """
    Small stochastic hill‑climbing: move a tiny amount of mass from a large
    entry to a small entry, accepting improvements or occasional uphill
    moves (simulated‑annealing style).  Runs until the time budget expires.
    """
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a, L_fft).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # choose donor (large mass) and receiver (small mass)
        i = np.argmax(a) if rng.random() < 0.7 else rng.integers(n)
        j = np.argmin(a) if rng.random() < 0.7 else rng.integers(n)

        if i == j or a[i] <= 1e-14:
            continue

        # move a tiny fraction of the donor's mass
        eps = min(0.02 * a[i], 1e-4)  # at most 2 % or absolute cap
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        new_score = 2.0 * n * _conv_self(a_new, L_fft).max()
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
        else:
            # annealing acceptance
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25  # safety margin

    # problem size – the previous best used 4096, which is a good trade‑off
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise with a global starter if available, otherwise uniform
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                f = starter / s
            else:
                f = np.full(n, 1.0 / n)
        else:
            f = np.full(n, 1.0 / n)
    except Exception:
        f = np.full(n, 1.0 / n)

    best_f = f.copy()
    best_score = _objective_score(f, n)
    print(f"[0.00s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main alternating optimisation loop (LP + mixing)
    # ------------------------------------------------------------------
    stagnation_counter = 0
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # 1) current maximum self‑convolution
        conv_f = _conv_self(f, L_fft)
        M = conv_f.max()

        # 2) solve LP for g maximizing its ℓ¹‑norm under the bound M
        lp_res = _solve_lp_max_sum_g(f, M)
        if not lp_res.success:
            print(f"[{time.time() - start_time:.2f}s] LP failed (iter {iteration}): {lp_res.message}")
            break

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print(f"[{time.time() - start_time:.2f}s] LP returned zero mass, terminating.")
            break

        # 3) normalise g to unit ℓ¹ so that mixing preserves normalisation
        g = g0 / sum_g0

        # 4) pre‑compute convolutions needed for the 1‑D line search
        conv_g = _conv_self(g, L_fft)
        conv_fg = _conv_two(f, g, L_fft)

        # 5) optimal mixing coefficient λ
        lam_opt, mixed_score = _optimal_mix_lambda(conv_f, conv_g, conv_fg, n)

        # 6) convex update
        f = (1.0 - lam_opt) * f + lam_opt * g
        f = np.maximum(f, 0.0)                     # safety against tiny negatives
        f /= f.sum()                               # re‑normalise (should stay ≈1)

        cur_score = _objective_score(f, n)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            stagnation_counter = 0
            print(f"[{time.time() - start_time:.2f}s] iter {iteration:3d} λ={lam_opt:.5f} → score={cur_score:.6f}")
        else:
            stagnation_counter += 1

        # --------------------------------------------------------------
        # Simple stagnation handling – random perturbation if stuck too long
        # --------------------------------------------------------------
        if stagnation_counter >= 12:
            if time.time() < deadline - 5.0:
                rng = np.random.default_rng()
                perturb = rng.normal(scale=0.001, size=n) * f
                f = np.maximum(f + perturb, 0.0)
                f /= f.sum()
                print(f"[{time.time() - start_time:.2f}s] Random perturbation after stagnation")
                stagnation_counter = 0
            else:
                break

        # safety break to avoid endless looping when time is almost over
        if iteration >= 50 and time.time() > deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # Final refinement with cheap hill‑climbing if we still have time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb refinement")
        best_f, best_score = _hill_climb_refine(
            best_f, n, L_fft, deadline, start_time, best_f, best_score
        )

    # ------------------------------------------------------------------
    # Normalise final answer and return as plain Python list
    # ------------------------------------------------------------------
    best_f = np.maximum(best_f, 0.0)
    total = best_f.sum()
    if total > 0.0:
        best_f = best_f / total
    else:
        best_f = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_f]
```

A final stochastic hill‑climb cleans up any remaining imperfections.
