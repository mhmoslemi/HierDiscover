"""
Improved optimiser for the step‑function evaluation problem.

The algorithm combines several complementary ideas:

  1.  A quick global sampling phase (Dirichlet / geometric mixtures) to
      obtain a decent starting point.
  2.  A *smooth* gradient‑descent phase that optimises a
      soft‑max approximation of the L∞‑norm of the self‑convolution.
      The gradient is obtained by a single FFT‑based correlation,
      and each iterate is Euclidean‑projected onto the probability simplex.
  3.  A final tiny‑mass hill‑climb (exact sub‑gradient) for fine‑grained
      polishing.

All phases respect a wall‑clock budget of 1000 s.  The routine returns
the best sequence found as a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes ∑a = 1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv(a,a)) at a (∑a = 1)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=600):
    """Draw Dirichlet vectors with a wide range of concentrations."""
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.5, 0.8)          # concentration ≈ 0.003 … 6.3
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=350):
    """
    Convex mixtures of a left‑geometric, a right‑geometric and a uniform component.
    """
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.80, 0.998)                 # slower decay ⇒ heavier tails
        left = np.power(r, idx)                      # decreasing
        right = np.power(r, n - 1 - idx)             # increasing
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.4f} w={w} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth max‑convolution optimiser
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the probability vector p_k = exp(alpha * conv[k]) / Σexp.
    The subtraction of the maximum prevents overflow.
    """
    max_c = conv.max()
    exps = np.exp(alpha * (conv - max_c))
    return exps / exps.sum()


def _smooth_descent(start_seq,
                    deadline,
                    best_seq,
                    best_score,
                    alpha_start: float = 1.0,
                    alpha_max: float = 40.0):
    """
    Gradient descent on a smooth approximation of max(conv(a,a)).
    Each iteration:
        * evaluate the soft‑max weights,
        * compute the correlation w * a (via FFT) – the gradient,
        * take a back‑tracking line‑search step,
        * project onto the simplex.
    Alpha is increased geometrically so that the approximation becomes tighter.
    """
    a = start_seq.copy()
    n = a.shape[0]

    # nominal step size (in the projected space)
    step = 0.2
    min_step = 1e-12
    iteration = 0
    alpha = alpha_start

    while time.time() < deadline:
        iteration += 1

        # current convolution and objective
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth it={iteration} α={alpha:.2f} → {cur_score:.6f}")

        # occasionally tighten the soft‑max
        if iteration % 250 == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.5, alpha_max)

        # ----- gradient of the smooth max -----
        w = _softmax_weights(conv, alpha)               # length 2n‑1
        a_rev = a[::-1]                                 # reversed a
        grad = _conv_fft(w, a_rev)                      # length 3n‑2
        grad = grad[n - 1: n - 1 + n]                   # keep the n relevant entries

        # The true objective also contains the factor 2·n, we absorb it into the step size.
        gnorm = np.linalg.norm(grad) + 1e-12
        step_len = step / gnorm

        # back‑tracking line search (projected) -------------------------------------------------
        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_score = float(2.0 * n * _conv_self(a_new).max())
            if new_score < cur_score or step_len <= min_step:
                break
            step_len *= 0.5

        # accept / adapt step -------------------------------------------------
        if new_score < cur_score:
            a = a_new
            step = min(step * 1.07, 5.0)        # gently grow step
        else:
            step = max(step * 0.5, min_step)    # shrink step

    return best_seq, best_score

# ----------------------------------------------------------------------


def _hill_climb(start_a, deadline):
    """Exact sub‑gradient hill‑climb used in the reference implementation."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)   # up‑to 0.4 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The search proceeds through four phases
    described in the module docstring.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # ------------------------------------------------------------------
    # Problem size: a modest but sufficiently fine discretisation.
    # Larger n allows a finer approximation of the continuous optimum.
    # ------------------------------------------------------------------
    n = 8192                                          # double the previous 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – may have a different length.
    # If necessary we simply resample it by linear interpolation.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # resample to the current length if needed
            if starter.size != n:
                # simple linear interpolation on the interval [0,1]
                xs = np.linspace(0, 1, starter.size)
                xt = np.linspace(0, 1, n)
                starter = np.interp(xt, xs, starter)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter /= s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.20
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=800
        )

    # ------------------------------------------------------------------
    # Phase 2 – random geometric / uniform mixtures (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.35
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=400
        )

    # ------------------------------------------------------------------
    # Phase 3 – smooth‑max gradient descent (≈40 % of the budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.75
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_descent(
            best_seq, phase3_deadline, best_seq, best_score,
            alpha_start=1.0, alpha_max=40.0
        )
        print(f"[{time.time() - start_time:.2f}s] After smooth descent → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final tiny‑mass hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
