"""
improved_step_function_search.py

Search for a non‑negative step function (probability vector) that minimises
the bound 2·n·max(conv(a,a)) (equivalently evaluate_sequence when the vector
is normalised to sum = 1).

The algorithm consists of four stages, each respecting the overall 1000 s
time budget:

1️⃣  Broad parametric sweep over many analytic families, including new
    “inverse‑power” U‑shaped families and finer beta‑family grids.
2️⃣  Massive random Dirichlet sampling (more samples, wider concentration range).
3️⃣  Multiplicative projected‑gradient descent on the exact max‑convolution
    (mirror‑descent on the simplex).  This converges much faster than the
    additive PGD used previously.
4️⃣  A final simulated‑annealing phase with large multiplicative perturbations
    that can escape narrow local minima.

All stages keep track of the best vector seen and constantly print progress.
The final normalised vector is returned as a plain Python list.
"""

import time
import numpy as np

# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _objective_score(seq: np.ndarray) -> float:
    """
    Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1).
    """
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    """
    Exhaustive (but bounded) search over many analytic families.
    New families:
      * inverse‑power U‑shape   a_i ∝ ((i+1)*(n-i))^p   (p < 0)
      * finer beta grid (α,β ∈ [1e‑4,10] log‑spaced)
    Mix every candidate with the uniform distribution to explore convex
    combinations.
    """
    idx = np.arange(n, dtype=np.float64)

    # baseline uniform (already known, but keep for completeness)
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:

        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ------------------------------------------------------------------
    # Helper to try a raw (already normalised) sequence, optionally mixed
    # with the uniform distribution.
    # ------------------------------------------------------------------

            seq = lam * uniform + (1.0 - lam) * raw
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] {name} lam={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    p_grid = np.linspace(-2.5, 2.5, 101)

        pow_seq = np.power(idx + 1.0, p)

    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:

        for b in betas:

            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            _try_seq(seq, f"Beta α={a:.4g} β={b:.4g}")

    # ------------------------------------------------------------------
    # Stretched‑exponential
    # ------------------------------------------------------------------
    scales = np.linspace(0.2, 1.0, 9)
    p_vals = np.linspace(0.2, 5.0, 31)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            _try_seq(seq, f"Stretched p={p:.2f} sc={scale:.2f}")

    # ------------------------------------------------------------------
    # Logistic family
    # ------------------------------------------------------------------
    mu_factors = np.linspace(0.2, 0.8, 7)
    s_factors = np.linspace(0.01, 0.2, 7)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-9:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            _try_seq(seq, f"Logistic mu={mu_f:.2f} s={s_f:.2f}")

    # ------------------------------------------------------------------
    # Sine‑power (already in original)
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.1, 5.0, 31)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        _try_seq(seq, f"Sine‑power p={p:.2f}")

    # ------------------------------------------------------------------
    # NEW: inverse‑power U‑shape  a_i ∝ ((i+1)*(n-i))^p  with p<0

    p_vals = np.linspace(-4.0, -0.1, 40)
    for p in p_vals:
        if time.time() >= deadline:
            break

        # use log‑space to avoid overflow when p is large negative
        log_seq = p * np.log(prod)
        seq = np.exp(log_seq - np.max(log_seq))
        _try_seq(seq, f"Inv‑Power p={p:.2f}")

    return best_seq, best_score

# 3️⃣  Multiplicative Projected Gradient Descent (mirror‑descent)
# ----------------------------------------------------------------------
def _multiplicative_pgdescent(a0: np.ndarray,
                              start_time: float,
                              deadline: float,
                              best_seq: np.ndarray,
                              best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using the exact sub‑gradient of
    max(conv(a,a)).  The update is a_i ← a_i·exp(‑η·g_i) followed by a
    projection onto the probability simplex (which is just renormalisation).
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 5.0          # initial step size (in log‑space)
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] MPGD it={it:,} best={best_score:.6f}")

        # exact sub‑gradient of the max‑convolution
        subgrad = _subgradient_maxconv(a, conv)

        # decaying step size (log‑space)
        eta = eta0 / (1.0 + it * 1e-4)

        # multiplicative mirror update
        upd = np.exp(-eta * subgrad)
        a = a * upd
        # renormalise (projection onto the simplex)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional random jitter to leave plateaus
        if it % 500 == 0:
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        if eta < 1e-9:
            break
    return best_seq, best_score

# 4️⃣  Simulated‑Annealing with large multiplicative perturbations
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Large‑scale SA that perturbs the whole vector multiplicatively.
    Temperature decays linearly; the proposal is a_i ← a_i·exp(σ·N(0,1)).
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    T0 = 0.12          # initial temperature
    rng = np.random.default_rng()
    while time.time() < deadline:
        # linear temperature schedule
        t_elapsed = time.time() - start_time
        total = deadline - start_time
        frac = t_elapsed / total
        T = T0 * (1.0 - frac)          # decreases to 0

        # noise magnitude shrinks with temperature
        sigma = 0.12 * (1.0 - frac) + 0.002

        # multiplicative proposal
        prop = a * np.exp(rng.normal(scale=sigma, size=n))
        prop = np.maximum(prop, 0.0)
        s = prop.sum()
        if s == 0.0 or not np.isfinite(s):
            continue
        prop /= s

        cand_score = _objective_score(prop)
        diff = cand_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = prop
            cur_score = cand_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA new best={best_score:.6f}")

    return best_seq, best_score

        )

    # --------------------------------------------------------------
    # 3️⃣  Multiplicative projected gradient descent (≈250 s)
    # --------------------------------------------------------------
    phase_deadline = min(deadline - 150.0, time.time() + 250.0)
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting MPGD")
        best_seq, best_score = _multiplicative_pgdescent(
            best_seq, start_time, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After MPGD best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final large‑move Simulated Annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final SA")
        best_seq, best_score = _simulated_annealing(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
