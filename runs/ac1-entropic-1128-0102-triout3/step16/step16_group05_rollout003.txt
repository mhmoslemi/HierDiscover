def _softmax_subgradient(a, conv, alpha=10):
    max_val = conv.max()
    exp_weights = np.exp(alpha * (conv - max_val))
    total_w = exp_weights.sum()
    # Weighted sum of gradients for each shift
    n = a.shape[0]
    grad = np.zeros_like(a)
    # For each shift k, weighted contribution:
    for k in range(2*n-1):
        w = exp_weights[k]
        if w == 0:
            continue
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        idx = np.arange(lo, hi+1)
        grad[idx] += w * a[k - idx]
    grad = 2.0 * grad / total_w
    return grad
```

def _adam_descent(a0, deadline, best_seq, best_score, rng):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lr = 1e-2
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * a.shape[0] * conv.max()
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t} best={best_score:.6f}")
        # compute subgradient
        grad = _subgradient_maxconv(a, conv)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # update a
        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        # optional: decay lr slowly
        lr = max(lr * 0.9995, 1e-6)
    return best_seq, best_score
```

def _simulated_annealing(a0, deadline, best_seq, best_score, rng):
    cur = a0.copy()
    cur_score = _objective_score(cur)
    n = cur.shape[0]
    T0 = best_score * 0.02  # small temperature
    iter_num = 0
    while time.time() < deadline:
        iter_num += 1
        # choose source index weighted by current mass
        src = rng.choice(n, p=cur)  # probability proportional to mass
        dst = rng.integers(n)
        if src == dst:
            continue
        max_eps = cur[src] * (0.5 if iter_num % 10 == 0 else 0.1)
        if max_eps <= 0:

        eps = rng.random() * max_eps
        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        delta = cand_score - cur_score
        # exponential cooling schedule:
        T = T0 * (0.9999 ** iter_num)
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start_time:.2f}s] SA iter={iter_num} new best={best_score:.6f}")

            cur[i1] -= jitter
            cur[i2] += jitter
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _objective_score(cur)

"""
Propose a non‑negative step function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm – three complementary stochastic phases:

1. **Projected Adam descent** on the sub‑gradient of the max‑convolution.
   The iterate stays on the probability simplex (sum = 1, a≥0) by Euclidean
   projection.  Adam’s adaptive step‑size gives fast, stable progress.

2. **Simulated‑annealing local search**.
   Randomly move a small amount of mass between two indices.
   Moves that raise the objective are occasionally accepted according to a
   geometric temperature schedule, helping to escape shallow plateaus.

3. **Deterministic hill‑climb** that repeatedly shifts a tiny fraction of
   mass from the index with the largest sub‑gradient to the one with the
   smallest gradient.  A final brief smoothing step is applied if it improves
   the score.

The routine respects a 1000 s wall‑clock budget, prints intermediate scores
and returns the best sequence found (normalised to sum = 1).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Projected Adam descent
# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 8e-3                # initial learning rate
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        cur_score = 2.0 * a.shape[0] * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam iter={t} best={best_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # very slow decay of the global learning rate
        lr = max(lr * 0.9995, 1e-5)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 2 – Simulated annealing
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    cur_score = _objective_score(cur)
    n = cur.shape[0]

    T0 = best_score * 0.02          # initial temperature (≈2 % of the score)
    iter_num = 0

    while time.time() < deadline:
        iter_num += 1

        # source index, weighted by current mass (more mass → more likely to donate)
        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(n)
        if src == dst:
            continue

        # occasionally allow a relatively large transfer (to jump out of basins)
        max_transfer = cur[src] * (0.5 if (iter_num % 10 == 0) else 0.08)
        if max_transfer <= 0.0:
            continue
        eps = rng.random() * max_transfer

        cand = cur.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = _project_simplex(cand)

        cand_score = _objective_score(cand)
        delta = cand_score - cur_score

        # geometric cooling
        T = T0 * (0.9999 ** iter_num)

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={iter_num} best={best_score:.6f}")

        # occasional jitter to keep the chain moving
        if iter_num % 12000 == 0:
            i1, i2 = rng.choice(n, size=2, replace=False)
            jitter = cur[i1] * (rng.random() * 1e-4 + 1e-6)
            cur[i1] -= jitter
            cur[i2] += jitter
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _objective_score(cur)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 3 – Deterministic hill climb + optional smoothing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # source: index with largest gradient among those that still own mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        # move a modest fraction of the source mass (up to 2 %)
        eps = a[src] * (rng.random() * 0.02 + 1e-10)
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        a_new = _project_simplex(a_new)

        new_score = _objective_score(a_new)
        if new_score < best_score:
            best_seq = a_new.copy()
            best_score = new_score
            a = a_new
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step new best={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 4000:
                # small random jitter to escape a plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-8)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    # One optional smoothing pass (kernel [0.25,0.5,0.25])
    while time.time() < deadline:
        kernel = np.array([0.25, 0.5, 0.25])
        a_smooth = np.convolve(best_seq, kernel, mode='same')
        a_smooth = np.maximum(a_smooth, 0.0)
        a_smooth = a_smooth / a_smooth.sum()
        smooth_score = _objective_score(a_smooth)
        if smooth_score < best_score - 1e-9:
            best_seq = a_smooth
            best_score = smooth_score
            print(f"[{time.time() - start_time:.2f}s] Smoothing improved to {best_score:.6f}")
        else:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising the evaluation function.
    The routine combines Adam descent, simulated annealing and a final hill‑climb,
    respecting the 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)

    global start_time
    start_time = time.time()
    deadline_total = start_time + budget_s - 0.25   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initialise a starting point
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                starter = starter / starter.sum()
                n = starter.shape[0]
                best_seq = starter.copy()
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        # fallback to a uniform distribution of a modest size
        n = 4096
        best_seq = np.full(n, 1.0 / n)

    best_seq = _project_simplex(np.maximum(best_seq, 0.0))
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam descent (≈30 % of the budget)
    # ------------------------------------------------------------------
    deadline_phase1 = start_time + 0.30 * budget_s
    if time.time() < deadline_phase1:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam descent")
        best_seq, best_score = _adam_descent(best_seq,
                                             deadline_phase1,
                                             best_seq,
                                             best_score,
                                             rng)

    # ------------------------------------------------------------------
    # Phase 2 – Simulated annealing (≈30 % of the budget)
    # ------------------------------------------------------------------
    deadline_phase2 = start_time + 0.60 * budget_s
    if time.time() < deadline_phase2:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq,
                                                    deadline_phase2,
                                                    best_seq,
                                                    best_score,
                                                    rng)

    # ------------------------------------------------------------------
    # Phase 3 – Hill climb & smoothing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline_total:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill climb")
        best_seq, best_score = _hill_climb(best_seq,
                                           deadline_total,
                                           best_seq,
                                           best_score,
                                           rng)

    # ------------------------------------------------------------------
    # Normalise one last time and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
