"""
A three‑phase optimizer for the step‑function objective.

Phase 1 – many short random Dirichlet draws (fast diversification).
Phase 2 – an *alternating linear‑program* scheme inspired by the
          state‑of‑the‑art description: for the current vector f
          (normalised to sum = 1) we maximise the total mass of a new
          vector g under the linear constraints  f ∗ g ≤ M,
          where M = max(f∗f).  The LP is solved with SciPy’s HiGHS
          interior‑point method on a sparse Toeplitz matrix.
          After solving, g is normalised and mixed with f by a
          back‑tracking line search to strictly decrease M.
Phase 3 – a few projected‑gradient steps (smooth fine‑tuning).

The whole routine respects the 1000 s budget, prints a few progress
lines and finally returns the best normalised sequence discovered.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT of length power‑of‑2."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑program utilities
# ----------------------------------------------------------------------


def _precompute_conv_matrix_indices(n: int):
    """
    Returns the row‑ and column‑index arrays that describe the
    (2·n‑1)×n convolution matrix A such that A @ g = f ∗ g.
    The values are supplied later (they are just repetitions of f).
    """
    # column index repeats each column number n times
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    # row index: for column j the rows are j … j+n‑1
    row_parts = [np.arange(j, j + n, dtype=np.int32) for j in range(n)]
    row_idx = np.concatenate(row_parts)
    return row_idx, col_idx


def _solve_max_sum_lp(f: np.ndarray, M: float, row_idx, col_idx):
    """
    Maximises sum(g) subject to   f ∗ g ≤ M   and   g ≥ 0.
    Returns the optimal g (may be empty if the solver fails).
    """
    n = f.size
    # data = tiled copy of f (one copy per column)
    data = np.tile(f, n)                     # length n·n
    A = sparse.csr_matrix((data, (row_idx, col_idx)),
                          shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  →  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:   # pragma: no cover
        print(f"  LP failed with exception: {e}")
        return None

    if not res.success:
        # occasional numerical failures – treat as no progress
        return None
    return res.x


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The method consists of:

    1️⃣ Random Dirichlet warm‑starts.
    2️⃣ Alternating LP improvement (the novel part).
    3️⃣ A short projected‑gradient polishing phase.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Choose the working dimension (2048 is a sweet spot for speed/quality)
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # Initialise from the global starter if available, otherwise uniform
    # ------------------------------------------------------------------
    init = None
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size >= n:
            init = starter[:n].copy()
            print(f"[{time.time() - start_time:.2f}s] Using global starter (first {n} entries).")
    except Exception:
        pass

    if init is None:
        init = np.full(n, 1.0 / n, dtype=float)
        print(f"[{time.time() - start_time:.2f}s] No global starter – using uniform.")

    # Normalise to sum = 1 (objective is scale‑invariant)
    init = np.maximum(init, 0.0)
    if init.sum() == 0:
        init[:] = 1.0 / n
    else:
        init /= init.sum()

    best_seq = init.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet warm‑starts (≈15 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 15.0)
    rng = np.random.default_rng(seed + 12345)
    while time.time() < dirichlet_deadline:
        conc = 10 ** rng.uniform(-3.0, 0.5)          # spread of the Dirichlet
        cand = rng.gamma(shape=conc, scale=1.0, size=n)
        if cand.sum() == 0:
            continue
        cand /= cand.sum()
        sc = _objective_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute the sparse convolution matrix index pattern (independent of f)
    # ------------------------------------------------------------------
    row_idx, col_idx = _precompute_conv_matrix_indices(n)

    # ------------------------------------------------------------------
    # 2️⃣  Alternating LP improvement (main work)
    # ------------------------------------------------------------------
    f = best_seq.copy()
    iteration = 0
    while time.time() < deadline - 2.0:   # leave a couple seconds for the polishing phase
        iteration += 1
        conv_f = _conv_self(f)
        M = conv_f.max()
        # --------------------------------------------------------------
        # solve the LP: maximise sum(g) subject to f∗g ≤ M
        # --------------------------------------------------------------
        g_raw = _solve_max_sum_lp(f, M, row_idx, col_idx)
        if g_raw is None:
            print(f"[{time.time() - start_time:.2f}s] LP failed – terminating LP loop.")
            break
        sum_g = g_raw.sum()
        if sum_g <= 1e-12:
            print(f"[{time.time() - start_time:.2f}s] LP returned near‑zero vector – stopping.")
            break

        g = g_raw / sum_g                     # enforce sum(g)=1

        # --------------------------------------------------------------
        # try to mix f and g; back‑track on step size t until M decreases
        # --------------------------------------------------------------
        improved = False
        t = 0.5
        while t >= 1e-4:
            h = (1.0 - t) * f + t * g
            M_h = _conv_self(h).max()
            if M_h < M - 1e-12:               # strict improvement
                f = h
                best_score = 2.0 * n * M_h
                best_seq = f.copy()
                print(f"[{time.time() - start_time:.2f}s] Iter {iteration:02d} "
                      f"t={t:.4f} improved to {best_score:.6f}")
                improved = True
                break
            t *= 0.5

        if not improved:
            # No reduction found – stop the alternating process
            print(f"[{time.time() - start_time:.2f}s] Iter {iteration:02d} – no t gave improvement.")
            break

        # safety: stop after a reasonable number of outer iterations
        if iteration >= 30:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final projected‑gradient polishing (≈10 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        step = 0.8
        while time.time() < deadline:
            conv = _conv_self(a)
            max_val = conv.max()
            # sub‑gradient of the max‑convolution (averaged over all active shifts)
            tol = 1e-12
            active = np.where(conv >= max_val - tol)[0]
            grad = np.zeros_like(a)
            for k in active:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1, dtype=int)
                grad[idx] += a[k - idx]
            if active.size:
                grad = 2.0 * grad / active.size
            # move opposite to the gradient and re‑project
            cand = a - step * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < best_score - 1e-9:
                a = cand
                best_seq = a.copy()
                best_score = cand_score
                print(f"[{time.time() - start_time:.2f}s] PG polish score {best_score:.6f}")
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.5
                if step < 1e-6:
                    break

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0.0:
        best_seq = best_seq / s
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
