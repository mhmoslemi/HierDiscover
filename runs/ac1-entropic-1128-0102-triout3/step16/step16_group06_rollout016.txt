"""
Optimiser for the step‑function evaluation problem using a
continuation of smooth max‑convolution minimisation.
Key ideas:
* Random Dirichlet / geometric mixture initialisers.
* A soft‑max (log‑sum‑exp) approximation of the max‑convolution is
  minimised with exponentiated‑gradient (mirror descent on the simplex).
  The smoothing parameter λ is increased progressively (continuation
  method) so that the optimisation first sees a smooth landscape
  and later focuses on the true sup‑norm.
* A final tiny‑mass hill‑climb (same as the reference code) polishes the
  result.
All steps respect a 1000 s wall‑clock budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient (log‑sum‑exp)
# ----------------------------------------------------------------------


def _smoothed_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_λ(a) = (1/λ)·log∑_k exp(λ·(a*a)_k)
    with respect to a (∑a_i = 1).  Returns a vector of length n.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable)
    exp_args = lam * (conv - max_c)
    np.clip(exp_args, -700, 0, out=exp_args)  # prevent underflow/overflow
    w = np.exp(exp_args)
    w_sum = w.sum()
    if w_sum == 0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum                          # normalised soft‑max

    # correlation a ⋆ p  (g_k = Σ_j a_j p_{j+k})
    p_rev = p[::-1]
    L = _next_pow2(len(a) + len(p_rev) - 1)
    Af = rfft(a, n=L)
    Pr = rfft(p_rev, n=L)
    corr_full = irfft(Af * Pr, n=L)[: len(a) + len(p_rev) - 1]
    # slice to obtain the n‑dimensional gradient
    start = 2 * n - 2
    stop = n - 2
    grad = corr_full[start:stop:-1]    # length n
    grad = 2.0 * grad                  # factor 2 from ∂(a*a)/∂a
    return grad


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated gradient) update on the simplex:
        a ← a ⊙ exp(-η·grad)  ;   ⊙ = element‑wise product,
    followed by renormalisation.
    """
    # work in log‑space for numerical stability
    log_a = np.log(a + 1e-30)
    log_new = log_a - eta * grad
    log_new -= np.max(log_new)            # shift to avoid overflow
    a_new = np.exp(log_new)
    a_new_sum = a_new.sum()
    if a_new_sum == 0:
        #should not happen, fall back to uniform
        n = a.shape[0]
        a_new = np.full(n, 1.0 / n)
    else:
        a_new = a_new / a_new_sum
    return a_new

# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=500):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.5, 1.0)   # concentration ≈ 0.003 … 10
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=400):
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        # geometric ratio in (0.7, 0.9995)
        r = rng.uniform(0.70, 0.9995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        # random weights for the three components
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * (1.0 / n)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.5f} w={w} → {sc:.6f}")
    return best_seq, best_score


def _random_geometric_trials(n, best_seq, best_score, deadline, trials=200):
    """Pure left‑geometric families a_i ∝ r^i (later normalised)."""
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.5, 0.9999)
        a = np.power(r, np.arange(n, dtype=np.float64))
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Geom r={r:.5f} → {sc:.6f}")
    return best_seq, best_score

# Tiny‑mass hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (normalised to sum=1).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)   # ≤0.4 % of source mass
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the provided `evaluate_sequence`
    objective.  The search proceeds through:

    1. Optional global starter.
    2. A large number of cheap random Dirichlet / mixture / geometric draws.
    3. Continuation optimisation: minimise a smoothed log‑sum‑exp version of
       the max‑convolution using exponentiated‑gradient updates while the
       smoothing parameter λ grows (5 → 160).
    4. A final tiny‑mass hill‑climb for fine‑grained polishing.

    The total wall‑clock time never exceeds `budget_s` seconds.
    """

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.15
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=800
        )

    # ------------------------------------------------------------------
    # Phase 2 – random mixtures (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.30
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=600
        )

    # ------------------------------------------------------------------
    # Phase 3 – pure geometric families (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.40
    if time.time() < phase3_deadline:
        best_seq, best_score = _random_geometric_trials(
            n, best_seq, best_score, phase3_deadline, trials=400
        )

    # ------------------------------------------------------------------
    # Phase 4 – smoothed continuation optimiser (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase4_deadline = start_time + budget_s * 0.70
    if time.time() < phase4_deadline:
        # Build a small pool of promising starting points.
        init_pool = [best_seq, uniform]
        rev = best_seq[::-1]
        if not np.allclose(best_seq, rev):
            init_pool.append(rev)

        lambda_schedule = [5, 10, 20, 40, 80, 160]
        for idx, a0 in enumerate(init_pool):
            a = a0.copy()
            # enforce strict positivity to avoid log(0) later
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            for lam in lambda_schedule:
                # step size slowly decreases with λ
                eta = 0.5 / np.sqrt(lam)
                # allocate a fixed budget per (init,λ) pair, but respect global deadline
                max_iter = 4000
                for it in range(max_iter):
                    if time.time() >= deadline:
                        break
                    grad = _smoothed_gradient(a, lam)
                    a = _exp_update(a, grad, eta)
                    cur_score = _objective_score(a)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(
                            f"[{time.time() - start_time:.2f}s] Init{idx} λ={lam} it={it} score={cur_score:.6f}"
                        )
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # Phase 5 – final tiny‑mass hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]

"""
Optimiser for the step‑function evaluation problem using a
continuation of smooth max‑convolution minimisation.

Key ideas
---------
* Random Dirichlet / geometric‑mixture initialisers to obtain diverse seeds.
* A soft‑max (log‑sum‑exp) approximation of the max‑convolution is minimised
  with exponentiated‑gradient (mirror descent on the probability simplex).
  The smoothing parameter λ is increased progressively (continuation) so that
  the optimisation first sees a smooth landscape and later focuses on the
  true L∞‑norm.
* A final tiny‑mass hill‑climb (identical to the reference code) polishes the
  result.

All phases respect a 1000 s wall‑clock budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient (log‑sum‑exp)
# ----------------------------------------------------------------------


def _smoothed_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_λ(a) = (1/λ)·log∑_k exp(λ·(a*a)_k)
    with respect to a (∑a_i = 1).  Returns a vector of length n.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    exp_args = lam * (conv - max_c)
    np.clip(exp_args, -700, 0, out=exp_args)   # avoid overflow / underflow
    w = np.exp(exp_args)
    w_sum = w.sum()
    if w_sum == 0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum

    # correlation a ⋆ p  (g_k = Σ_j a_j p_{j+k})
    p_rev = p[::-1]
    L = _next_pow2(len(a) + len(p_rev) - 1)
    Af = rfft(a, n=L)
    Pr = rfft(p_rev, n=L)
    corr_full = irfft(Af * Pr, n=L)[: len(a) + len(p_rev) - 1]

    # slice to obtain the n‑dimensional gradient
    start = 2 * n - 2
    stop = n - 2
    grad = corr_full[start:stop:-1]      # length n
    grad = 2.0 * grad                    # factor 2 from ∂(a*a)/∂a
    return grad


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) update on the simplex:
        a ← a ⊙ exp(-η·grad)  ;   ⊙ = element‑wise product,
    followed by renormalisation.

    log_new = log_a - eta * grad
    log_new -= np.max(log_new)               # shift for numerical safety
    a_new = np.exp(log_new)

    if s == 0:
        n = a.shape[0]

    else:
        a_new = a_new / s
    return a_new

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.15
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=800
        )

    # ------------------------------------------------------------------
    # Phase 2 – random mixtures (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.30
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=600
        )

    # ------------------------------------------------------------------
    # Phase 3 – pure geometric families (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.40
    if time.time() < phase3_deadline:
        best_seq, best_score = _random_geometric_trials(
            n, best_seq, best_score, phase3_deadline, trials=400
        )

    # ------------------------------------------------------------------
    # Phase 4 – smoothed continuation optimiser (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase4_deadline = start_time + budget_s * 0.70
    if time.time() < phase4_deadline:
        # Build a small pool of promising starting points.
        init_pool = [best_seq, uniform]
        rev = best_seq[::-1]
        if not np.allclose(best_seq, rev):
            init_pool.append(rev)

        lambda_schedule = [5, 10, 20, 40, 80, 160]
        for idx, a0 in enumerate(init_pool):
            a = a0.copy()
            a = np.maximum(a, 1e-12)   # keep strict positivity
            a /= a.sum()
            for lam in lambda_schedule:
                eta = 0.5 / np.sqrt(lam)   # step size shrinks with λ
                max_iter = 4000
                for it in range(max_iter):
                    if time.time() >= deadline:
                        break
                    grad = _smoothed_gradient(a, lam)
                    a = _exp_update(a, grad, eta)
                    cur_score = _objective_score(a)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(
                            f"[{time.time() - start_time:.2f}s] Init{idx} λ={lam} it={it} score={cur_score:.6f}"
                        )
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # Phase 5 – final tiny‑mass hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
