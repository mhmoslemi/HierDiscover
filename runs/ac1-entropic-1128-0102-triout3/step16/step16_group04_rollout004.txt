def next_pow2(x):
    return 1 << ((x - 1).bit_length())

def compute_conv_fft(a, L_fft):
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)

    max_val = conv.max()

    i_range = np.arange(i_start, i_end+1)
    j = k_max - i_range
    g[i_range] = 2.0 * a[j]

"""
Hybrid sub‑gradient mirror descent + random refinement.

* Starts from the global starter (height_sequence_1) or uniform.
* Repeatedly:
    – compute the full self‑convolution via FFT,
    – locate the shift with maximal value,
    – obtain the exact sub‑gradient of the max‑convolution (2·a shifted),
    – perform an exponentiated‑gradient (mirror) update on the simplex.
* Occasionally adds a tiny Gaussian jitter to escape plateaus.
* After the main phase mixes the best vector with the uniform distribution
  (coarse λ‑grid) and finally performs a cheap random‑pair hill‑climb.
* Returns a plain Python list of non‑negative floats (≈ unit‑sum).

The algorithm respects the 1000 s time budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)−1) via FFT.
    Returns a real‐valued array of length 2·len(a)−1.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    return conv[: 2 * a.shape[0] - 1]


def _score_from_conv(conv: np.ndarray, a_sum: float, n: int) -> float:
    """
    Exact score used by the evaluator:
        2·n·max(conv) / (sum(a)²)
    """
    return float(2.0 * n * conv.max() / (a_sum * a_sum))


def _max_shift_gradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at the active shift k_max.
    For each i, grad_i = 2·a_{k_max−i} if 0≤k_max−i<n, else 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    # valid i satisfy 0 ≤ j = k_max−i < n  ⇔  k_max−(n−1) ≤ i ≤ k_max
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)               # inclusive
    if i_start <= i_end:
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k_max - i_idx
        grad[i_idx] = 2.0 * a[j_idx]
    return grad


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores.  The returned list contains non‑negative entries
    (approximately unit sum – the evaluator will clamp and renormalise
    automatically).
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so we never overrun the limit
    deadline = start_time + budget_s - 0.15

    # ------------------------------------------------------------------
    # Problem size – keep it fixed for reproducibility
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)          # length for FFT‑based convolution

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – use the globally supplied starter if present
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1:
            raise ValueError
        # Pad or truncate to the required length
        if starter.size != n:
            if starter.size > n:
                starter = starter[:n]
            else:
                starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
        a = np.maximum(starter, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s                     # normalise to unit sum
        else:
            a = np.full(n, 1.0 / n)
    except Exception:
        a = np.full(n, 1.0 / n)            # fallback: uniform

    # initial score
    conv = _conv_full_fft(a, L_fft)
    best_score = _score_from_conv(conv, a.sum(), n)
    best_a = a.copy()
    print(f"[0.00s] Initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Sub‑gradient mirror descent on the exact max‑convolution
    # ------------------------------------------------------------------
    # Allocate ~70 % of the budget to this phase
    sub_deadline = min(deadline, start_time + 0.70 * (deadline - start_time))
    eta0 = 2.0                               # initial learning‑rate scale
    it = 0
    rng = np.random.default_rng()

    while time.time() < sub_deadline:
        it += 1

        # Full convolution and active shift
        conv = _conv_full_fft(a, L_fft)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # Sub‑gradient from the active shift
        grad = _max_shift_gradient(a, k_max)

        # Step‑size schedule (classical 1/√t)
        eta = eta0 / np.sqrt(it + 1)

        # Exponentiated‑gradient update (mirror descent on the simplex)
        exp_factor = np.exp(-eta * grad)
        # avoid under‑/overflow
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a = a * exp_factor
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # Small random jitter every few iterations helps to escape flat plateaus
        if rng.random() < 0.01:   # 1 % chance
            eps = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

        # Evaluate current score (no need for a second FFT)
        cur_score = _score_from_conv(conv, a.sum(), n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] Iter={it:,}  score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Mixing with the uniform distribution (coarse λ‑grid)
    # ------------------------------------------------------------------
    # Reserve a few seconds for the final hill‑climb
    mix_deadline = deadline - 5.0
    uniform = np.full(n, 1.0 / n)

    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() > mix_deadline:
            break
        cand = lam * best_a + (1.0 - lam) * uniform
        cand = cand / cand.sum()
        conv = _conv_full_fft(cand, L_fft)
        sc = _score_from_conv(conv, cand.sum(), n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Light random‑pair hill‑climb (the remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # Move a modest amount δ from i to j (never exceeds a[i])
        delta_max = min(best_a[i], 0.01)     # cap the step size
        if delta_max <= 0.0:
            continue
        delta = rng.uniform(0.0, delta_max)
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalise – sum stays unchanged, but guard against round‑off
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            cand = uniform.copy()
        else:
            cand = cand / cand_sum
        conv = _conv_full_fft(cand, L_fft)
        sc = _score_from_conv(conv, cand.sum(), n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb i={i} j={j} δ={delta:.6f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Convert to plain Python list as required by the evaluator
    return [float(v) for v in best_a]
