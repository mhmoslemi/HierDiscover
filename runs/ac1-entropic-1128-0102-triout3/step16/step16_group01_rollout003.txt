"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm adds a new family of *edge‑peaked* shapes (mass concentrated
near both ends) to the usual geometric/power sweeps, then refines the best
candidate with a sequence of continuous‑optimisation phases:

1️⃣  Fast parametric sweep (small n) – explores monotone and edge‑peaked
    families, also mixing them with a uniform component.

2️⃣  Upscale the best parameters to the target length (4096) and initialise
    the optimisation.

3️⃣  Smoothed‑max (soft‑max) gradient descent with temperature continuation.

4️⃣  Exponentiated‑gradient descent on the true L∞‑objective.

5️⃣  Projected sub‑gradient descent with simple back‑tracking.

6️⃣  Final random hill‑climb (tiny mass transfers) to polish the result.

All phases respect the simplex constraints (non‑negative, sum = 1) and
run within the 1000 s budget.  The best sequence found is returned as a
plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_mono_shape(n: int, r: float, p: float) -> np.ndarray:
    """Monotone geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_edge_shape(n: int, r: float, p: float) -> np.ndarray:
    """Edge‑peaked shape: a_i ∝ r^{dist_to_edge}·(dist+1)^p, normalised."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)           # distance to the nearest end
    log_shape = d * np.log(r)
    if p != 0.0:
        log_shape += p * np.log(d + 1.0)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Smooth‑max utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights p_k = exp(beta*conv_k) / Σexp(...)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k conv_k (soft‑max objective):
        g = 2 * a  ⊛  p_rev       (cross‑correlation)
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    corr = np.fft.irfft(Af * Pf, n=L_grad)
    # `corr` length = L_grad ; we need entries corresponding to indices
    # i+j = shift where i∈[0,n-1], j∈[0,2n-2]
    # The relevant part is the slice [n-1 : n-1+n]
    start = n - 1
    g = corr[start:start + n][::-1]
    return 2.0 * g


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]               # includes diagonal correctly (2*a_i)
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (exact L∞‑objective)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) updates:
        a ← a * exp(- η * g)   ,   η annealed.
    After each update we renormalise to stay on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # Annealed step size (log‑schedule)
        eta = eta0 / (1.0 + 0.001 * it)

        # Multiplicative update – protect from overflow/underflow
        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        score = _eval_fast(a, L_fft)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} "
                  f"score={score:.6f}")

        # Occasional jitter to escape plateaux
        if it % 5000 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with back‑tracking
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on max(conv) with simple back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    step0 = 0.30

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # Simple diminishing step size
        step = step0 / (it ** 0.55)
        improved = False

        for _ in range(12):
            cand = a - step * g
            cand = _project_simplex(cand)
            cand_score = _eval_fast(cand, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                      f"step={step:.5e} score={cand_score:.6f}")
                break
            step *= 0.5

        if not improved:
            # occasional uphill move (simulated‑annealing flavour)
            prob = np.exp(-(cand_score - best_score) * 8.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = cand

        # jitter every few thousand iterations
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

            # occasional jitter to avoid stagnation
            if it % 4000 == 0:
                eps = 1e-4 * rng.random(n_target) * a
                a = _project_simplex(a + eps)

    # ------------------------------------------------------------------
    # Phase 4 – exponentiated‑gradient on the true L∞‑objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:
        exp_deadline = deadline - 250.0
        best_seq, best_score = _expgrad_phase(best_seq, L_fft_target,
                                              exp_deadline, start_time,
                                              best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Phase 5 – projected sub‑gradient descent with back‑tracking
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        sub_deadline = deadline - 150.0
        best_seq, best_score = _subgradient_descent(best_seq, L_fft_target,
                                                    sub_deadline, start_time,
                                                    best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Phase 6 – final random hill‑climb (tiny mass transfers)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n_target)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n_target)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 3e-3   # up to 0.3 % of source mass
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft_target)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n_target, 1.0 / n_target, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
