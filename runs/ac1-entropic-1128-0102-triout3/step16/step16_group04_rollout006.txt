"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The search consists of three phases:

1️⃣  Warm‑start – a handful of Dirichlet samples are tried (if time permits).

2️⃣  Targeted hill‑climbing – repeatedly pick the index that contributes most
    to the current convolution peak and move a small amount of probability
    mass from that index to a “quiet’’ index.  The convolution is updated
    incrementally (O(n) slice operations) so millions of proposals fit easily
    into the 1000 s budget.  Every few thousand accepted moves we recompute the
    convolution exactly via FFT to wipe out the O(δ²) approximation error.

3️⃣  Final polishing – a fine grid mixing with the uniform distribution and a
    short random‑pair hill‑climb (tiny transfers) are performed.

All arithmetic stays on the probability simplex (∑aᵢ=1) and the algorithm
keeps the best vector seen so far.  The routine is time‑aware and prints
intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)−1) via FFT."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score used by the evaluator (sum(a)=1)."""
    return float(2.0 * n * conv.max())


def _exact_score(a: np.ndarray, n: int, L_fft: int) -> float:
    conv = _conv_self_fft(a, L_fft)
    return _score_from_conv(conv, n), conv


# ----------------------------------------------------------------------
# 1️⃣ Warm‑start with a few Dirichlet draws
# ----------------------------------------------------------------------
def _warm_start(a_best: np.ndarray,
                best_score: float,
                n: int,
                L_fft: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Try a few random Dirichlet vectors; keep any improvement."""
    while time.time() < deadline:
        # randomly choose concentration in a wide range
        log_c = rng.uniform(np.log(0.05), np.log(10.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc, _ = _exact_score(a, n, L_fft)
        if sc < best_score:
            best_score, a_best = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Warm‑start improved to {sc:.6f}")
    return a_best, best_score


# ----------------------------------------------------------------------
# 2️⃣ Targeted hill‑climbing on the simplex
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray,
                best_score: float,
                n: int,
                L_fft: int,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from the index that contributes
    most to the current convolution peak to a low‑contributing index.
    """
    a = a_start.copy()
    conv = _conv_self_fft(a, L_fft)            # current convolution
    cur_score = _score_from_conv(conv, n)
    best_a = a.copy()
    best_sc = best_score

    it = 0
    accepted = 0
    recompute_every = 5000  # exact recompute interval

    # temperature schedule for occasional uphill jumps
    T0 = (best_sc - 1.4) * 0.5   # crude scaling, will decay to ~0
    T_end = 1e-6
    total_iters_est = 200000   # rough estimate for schedule
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Choose move parameters
        # --------------------------------------------------------------
        # 1. index of the global convolution peak
        k = int(np.argmax(conv))

        # 2. contribution of each position to that peak
        idx = np.arange(n)
        rev = k - idx
        mask = (rev >= 0) & (rev < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[rev[mask]]

        # source index: where contribution is largest
        i = int(np.argmax(contrib))
        if a[i] <= 1e-12:
            # nothing to move; pick a random i instead
            i = rng.integers(0, n)

        # destination index: pick among the 20 % least contributing positions
        low_thresh = np.quantile(contrib, 0.2)
        low_candidates = np.where(contrib <= low_thresh)[0]
        if low_candidates.size == 0:
            j = rng.integers(0, n)
        else:
            j = int(rng.choice(low_candidates))
        if j == i:
            j = (i + 1) % n

        # step size – a fraction of the source mass, bounded
        delta = min(a[i] * 0.15, 0.008)      # hard ceiling prevents too large jumps
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # Compute candidate convolution by cheap O(n) slice updates
        # --------------------------------------------------------------
        conv_cand = conv.copy()
        # subtraction of mass from i
        conv_cand[i:i + n] -= 2.0 * delta * a
        # addition of mass to j
        conv_cand[j:j + n] += 2.0 * delta * a

        cand_max = conv_cand.max()
        cand_score = 2.0 * n * cand_max

        # --------------------------------------------------------------
        # Acceptance test (hill‑climb with tiny uphill probability)
        # --------------------------------------------------------------
        improve = cand_score < best_sc - 1e-12
        # temperature falls linearly with iteration count
        T = max(T0 * (1 - it / total_iters_est), T_end)
        prob_up = np.exp((best_sc - cand_score) / max(T, 1e-12))

        if improve or rng.random() < prob_up:
            # accept move
            a[i] -= delta
            a[j] += delta
            conv = conv_cand
            accepted += 1
            if cand_score < best_sc - 1e-12:
                best_sc, best_a = cand_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Iter={it:,}  Accepted improvement → score={best_sc:.6f}")

        # --------------------------------------------------------------
        # Periodically recompute convolution exactly to discard O(δ²) error
        # --------------------------------------------------------------
        if accepted % recompute_every == 0 and accepted > 0:
            conv = _conv_self_fft(a, L_fft)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_sc - 1e-12:
                best_sc, best_a = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] "
                      f"Re‑eval after {accepted} accepts → score={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3️⃣ Final refinement: mixing with uniform + light random hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(best_a: np.ndarray,
                      best_score: float,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mix with the uniform distribution and perform a short random‑pair sweep."""
    uniform = np.full(n, 1.0 / n)
    a = best_a.copy()
    cur_score = best_score

    # ----- coarse mixing -------------------------------------------------
    for lam in np.linspace(0.0, 1.0, 61):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        sc, _ = _exact_score(cand, n, L_fft)
        if sc < cur_score:
            cur_score, a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # ----- tiny random pair hill‑climb ----------------------------------
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 1e-12:
            continue
        delta = rng.uniform(0.0, min(a[i] * 0.02, 0.001))
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc, _ = _exact_score(cand, n, L_fft)
        if sc < cur_score:
            cur_score, a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Tiny move i={i} j={j} δ={delta:.6f} → score={sc:.6f}")

    return a, cur_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  Returns a plain Python list.  The routine is
    time‑aware and keeps the best sequence found before the deadline.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # leave a tiny safety margin
    deadline = start_time + budget_s - 0.15
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – use the size of the provided starter if possible
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim != 1 or starter.size < 2:
            raise ValueError
        n = starter.size
    except Exception:
        n = 4096
        starter = None

    L_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣ Initialise – uniform or supplied starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    a_cur = uniform.copy()
    best_sc, _ = _exact_score(a_cur, n, L_fft)

    if starter is not None:
        s = starter.copy()
        s = np.maximum(s, 0.0)
        if s.sum() > 0.0:
            s = s / s.sum()
            sc, _ = _exact_score(s, n, L_fft)
            if sc < best_sc:
                best_sc, a_cur = sc, s.copy()
                print(f"[0.00s] Starter improved score={sc:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start via a few Dirichlet samples (≈5 % of time)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.05 * (deadline - start_time))
    if time.time() < warm_deadline:
        a_cur, best_sc = _warm_start(a_cur, best_sc, n, L_fft, warm_deadline, rng)

    # ------------------------------------------------------------------
    # 2️⃣ Targeted hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        a_cur, best_sc = _hill_climb(a_cur, best_sc, n, L_fft, deadline, rng)
        print(f"[{time.time() - start_time:.2f}s] Hill‑climb best ≈ {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement")
        a_cur, best_sc = _final_refinement(a_cur, best_sc, n, L_fft, deadline, rng)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    a_cur = np.maximum(a_cur, 0.0)
    total = a_cur.sum()
    if total > 0.0:
        a_cur = a_cur / total
    else:
        a_cur = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_sc:.6f}")
    return [float(v) for v in a_cur]
