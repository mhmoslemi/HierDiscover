"""
Improved search for a non‑negative step function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm blends:

1.  An expanded parametric sweep (U‑shaped, edge‑exponential,
    inverse‑polynomial, beta, power, geometric families).
2.  Heavy random Dirichlet / log‑normal sampling (both symmetric and
    highly asymmetric).
3.  Multiple short projected sub‑gradient descents from several
    random starts (aggressive step‑size schedule, occasional jitter).
4.  A final simulated‑annealing hill‑climb with larger mass moves
    and a slow temperature decay.

The total wall‑time is limited to *budget_s* (default 1000 s).  The best
sequence found is returned as a plain Python list of floats.  All
steps respect the simplex constraint (non‑negative, unit sum) and
periodically print progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())

# 1️⃣  Expanded parametric families
# ----------------------------------------------------------------------
def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Explore a large collection of analytic families."""
    idx = np.arange(n, dtype=np.float64)

    # ---- uniform baseline (already present) -------------------------
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- geometric ----------------------------------------------------
    eps_grid = np.logspace(-6, -1, 250)
    for eps in eps_grid:
        if time.time() >= deadline:
            return best_seq, best_score
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-2.5, 3.0, 121)
    for p in p_grid:
        if time.time() >= deadline:
            return best_seq, best_score
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.concatenate([np.linspace(0.05, 0.5, 10),
                             np.linspace(0.5, 2.0, 16),
                             np.linspace(2.0, 4.0, 10)])
    betas = alphas.copy()
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            return best_seq, best_score
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                return best_seq, best_score
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- stretched‑exponential -----------------------------------------
    scales = np.linspace(0.15, 1.0, 13)
    p_vals = np.linspace(0.1, 6.0, 51)
    for scale in scales:
        if time.time() >= deadline:
            return best_seq, best_score
        for p in p_vals:
            if time.time() >= deadline:
                return best_seq, best_score
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} sc={scale:.2f} score={sc:.6f}")

    # ---- logistic -------------------------------------------------------
    mu_factors = np.linspace(0.15, 0.85, 15)
    s_factors = np.linspace(0.005, 0.2, 12)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            return best_seq, best_score
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                return best_seq, best_score
            s_val = s_f * n
            if s_val < 1e-7:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.3f} score={sc:.6f}")

    # ---- sine‑power -----------------------------------------------------
    for p in np.linspace(0.05, 6.0, 61):
        if time.time() >= deadline:
            return best_seq, best_score
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    # ---- edge‑exponential (U‑shape) ------------------------------------
    for c in np.linspace(0.001, 6.0, 80):
        if time.time() >= deadline:
            return best_seq, best_score
        dist = np.minimum(idx, n - 1 - idx)
        seq = np.exp(c * dist)          # larger near edges
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Edge‑exp c={c:.3f} score={sc:.6f}")

    # ---- inverse‑polynomial (U‑shape) ----------------------------------
    for p in np.linspace(0.2, 2.5, 60):
        if time.time() >= deadline:
            return best_seq, best_score
        eps = 0.5
        seq = 1.0 / np.sqrt((idx + eps) * (n - 1 - idx + eps)) ** p
        seq = np.clip(seq, 0, None)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Inv‑poly p={p:.2f} score={sc:.6f}")

    # ---- mixture with uniform (for every family above) ---------------
    # (the sweep already mixes geometric with uniform; we add a generic mix)
    for mix_factor in np.linspace(0.1, 0.9, 9):
        if time.time() >= deadline:
            return best_seq, best_score
        seq = mix_factor * uniform + (1.0 - mix_factor) * best_seq
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix {mix_factor:.2f} score={sc:.6f}")

    return best_seq, best_score

# 2️⃣  Heavy random sampling (Dirichlet & log‑normal)
# ----------------------------------------------------------------------
def _random_sampling(n: int,
                     best_seq: np.ndarray,
                     best_score: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # Symmetric Dirichlet (already in baseline) – increase count
    for _ in range(6000):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.05), np.log(30.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")

    # Asymmetric Dirichlet – draw per‑coordinate concentration
    for _ in range(4000):
        if time.time() >= deadline:
            break
        # concentration parameters drawn log‑uniformly
        alphas = np.exp(rng.uniform(-4.0, 4.0, size=n))
        a = rng.gamma(shape=alphas, scale=1.0)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Asym‑Dirichlet score={sc:.6f}")

    # Log‑normal sampling (heavy‑tailed)
    for _ in range(3000):
        if time.time() >= deadline:
            break
        mu, sigma = rng.uniform(-2.0, 1.0), rng.uniform(0.2, 2.5)
        log_vals = rng.normal(loc=mu, scale=sigma, size=n)
        a = np.exp(log_vals)
        a = np.clip(a, 0, None)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] LogNormal μ={mu:.2f}σ={sigma:.2f} score={sc:.6f}")

    return best_seq, best_score

# 3️⃣  Projected sub‑gradient descent (multiple restarts)
# ----------------------------------------------------------------------
def _pgd_descent(a0: np.ndarray,
                 start_time: float,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on max(conv).
    Uses an aggressive step schedule and occasional jitter.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    it = 0
    eta0 = 1.0            # initial step size
    eta = eta0
    no_improve = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        subgrad = _subgradient_maxconv(a, conv)   # ∂max/∂a
        grad = 2.0 * n * subgrad                 # ∂obj/∂a

        # decaying step size – slower decay than the previous script
        eta = eta0 / (1.0 + it * 1e-5)

        a_candidate = a - eta * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-15:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            # when stuck, inject a tiny random jitter
            if no_improve % 400 == 0:
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.0005, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                a = a_jit
                cur_score = _objective_score(a)

        if eta < 1e-9:
            break

    return best_seq, best_score


def _pgd_multi_restart(best_seq: np.ndarray,
                       best_score: float,
                       start_time: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform several independent PGD runs from diverse starts.
    Each run receives a time slice (≈ 30 s) and the overall best
    result is kept.
    """
    n = best_seq.shape[0]
    rng = np.random.default_rng()
    # give each run a fair share of remaining time
    remaining = deadline - time.time()
    run_time = max(15.0, remaining / 8.0)   # at least 15 s per run, 8 runs max

    for run in range(8):
        if time.time() >= deadline:
            break
        # initialise either from the best so far or a fresh random point
        if run == 0:
            a0 = best_seq.copy()
        else:
            # random start: Dirichlet with moderate concentration + jitter
            c = rng.uniform(0.5, 5.0)
            a0 = rng.gamma(shape=c, scale=1.0, size=n)
            a0 /= a0.sum()
        sub_deadline = min(deadline, time.time() + run_time)
        best_seq, best_score = _pgd_descent(a0, start_time, sub_deadline,
                                            best_seq, best_score)
    return best_seq, best_score

# 4️⃣  Simulated‑annealing hill‑climb (larger moves)
# ----------------------------------------------------------------------
def _final_annealing(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = _objective_score(a)

    # Temperature schedule – start relatively hot, finish cool
    T0 = 0.03
    total_time = max(0.0, deadline - time.time())
    decay = (1e-6 / T0) ** (1.0 / total_time) if total_time > 0 else 1.0

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # source: among the top‑10 % heaviest coordinates
        # destination: among the bottom‑10 % (including zeros)
        heavy = np.where(a > np.quantile(a, 0.9))[0]
        light = np.where(a < np.quantile(a, 0.1))[0]
        if heavy.size == 0 or light.size == 0:
            src = rng.integers(n)
            dst = rng.integers(n)
        else:
            src = rng.choice(heavy)
            dst = rng.choice(light)

        # move a random fraction of the source mass
        frac = rng.random() * 0.5 + 0.001   # between 0.1 % and 50 %
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # re‑project only the two changed entries (simple projection)
        a_prop = _project_simplex(a_prop)

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-15:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal best score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine runs a five‑stage pipeline
    (expanded families → massive random sampling → multi‑restart PGD
    → simulated‑annealing) within the time budget and returns the best
    sequence as a list of Python floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    # --------------------------------------------------------------
    # Baseline – uniform distribution
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional external starter (height_sequence_1 if present)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Expanded parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Heavy random sampling
    # --------------------------------------------------------------
    phase_deadline = deadline - 180.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_sampling(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Multi‑restart Projected Sub‑Gradient Descent
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time() - start_time:.2f}s] Starting multi‑restart PGD")
        pgd_deadline = deadline - 120.0
        best_seq, best_score = _pgd_multi_restart(best_seq, best_score,
                                                  start_time, pgd_deadline)
        print(f"[{time.time() - start_time:.2f}s] After PGD best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final annealing")
        best_seq, best_score = _final_annealing(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
