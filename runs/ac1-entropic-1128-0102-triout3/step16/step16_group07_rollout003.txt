"""
propose_candidate.py – 1000 s search for a step‑function with a low
evaluate_sequence() bound.

Strategy
---------

0.  Choose a modest problem size (n = 2048).  Larger n gives a larger
    constant factor 2·n in the objective, while the “shape” of the
    optimal distribution does not change dramatically.  The reduced size
    makes the heavy linear‑programming (LP) stage fast enough to run
    many iterations.

1.  Quick random Dirichlet sampling to obtain a few promising starting
    points (including the uniform vector).  The best 10 candidates are
    kept.

2.  **LP improvement stage** – for each candidate we run a few
    iterations of the Frank‑Wolfe‑style scheme described in the
    statement:

        * fix f, compute M = max(conv(f,f))
        * solve  max Σ g_i   s.t.  conv(f,g) ≤ M   , g ≥ 0   (pure LP)
        * scale the solution to a probability vector and take a convex
          combination  f ← (1‑t)·f + t·g  (t = 0.3).

    The LP has 2·n‑1 linear constraints and n variables; with n = 2048 it
    is solved in a fraction of a second by SciPy’s “highs” solver.

3.  **Projected sub‑gradient descent** on the exact max‑convolution.
    This fine‑tunes the LP output using the sub‑gradient routine already
    present in the reference code.

4.  **Hill‑climb / simulated‑annealing** as a final polish.

All stages constantly check the global time budget and stop early if
necessary.  The best sequence found (normalised to sum = 1) is returned
as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a via FFT (real‐valued)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(seq: np.ndarray) -> float:
    """Score used during optimisation: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    The gradient is averaged over all active shifts.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _random_dirichlet_trials(n: int,
                             trials: int,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration parameter drawn on a log‑scale
        c = np.exp(rng.uniform(np.log(0.02), np.log(30.0)))
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Linear‑programming improvement (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _lp_improvement(a0: np.ndarray,
                    max_iter: int,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float,
                    t: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve   max Σ g_i   s.t.   conv(a,g) ≤ M , g ≥ 0
    where M = max(conv(a,a)).  The solution g is normalised to a
    probability vector and mixed with a (convex combination with weight t).
    """
    n = a0.shape[0]
    a = a0.copy()
    L_fft = _next_pow2(2 * n - 1)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, L_fft)
        M = conv.max()

        # Build constraint matrix A (shape (2n‑1, n))
        # Row k contains a[i] for i such that 0 ≤ k‑i < n.
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = a[i]

        # Linear programme: minimise –Σ g_i  (i.e. maximise Σ g_i)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.time() - start_time:.2f}s] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        sumg = g.sum()
        if sumg <= 1e-12:
            break

        # Normalise g to a probability vector
        g = g / sumg

        # Convex combination (both a and g already sum to 1)
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a)   # numerical safety

        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP it={it} score={sc:.6f}")

        # optional early stop if improvement stalls
        if it > 2 and abs(best_score - sc) < 1e-9:
            break

    return best_seq, best_score

# 3️⃣  Projected sub‑gradient descent (fine‑tune)
# ----------------------------------------------------------------------
def _pgd_descent(a0: np.ndarray,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """
    Classical projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    eta0 = 1.0          # initial step size (will decay)
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad

        # decaying step size
        eta_t = eta0 / (1.0 + it * 1e-4)

        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)

        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-12:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            eta0 *= 0.995
            if no_improve % 400 == 0:
                # occasional jitter to escape plateaus
                jitter = np.random.default_rng().normal(scale=0.001, size=n)
                a_jit = a + jitter
                a_jit = _project_simplex(a_jit)
                a = a_jit

        if eta0 < 1e-8:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb / simulated‑annealing
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = _objective_score(a)

    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score

    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))

        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that
    minimises `evaluate_sequence`.  The algorithm combines random
    Dirichlet sampling, a short LP‑based Frank‑Wolfe loop, projected
    sub‑gradient descent and a final annealed hill‑climb.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size (choose a size that balances quality and speed)
    # --------------------------------------------------------------
    n = 2048
    print(f"[{time.time() - start_time:.2f}s] Using n = {n}")

    # --------------------------------------------------------------
    # Baseline (uniform) and optional external starter
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # external starter if provided by the environment
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            starter = starter[:n]               # truncate / pad if needed
            if starter.shape[0] < n:
                # pad with zeros and re‑normalise
                starter = np.concatenate([starter,
                                          np.zeros(n - starter.shape[0])])
            starter_sum = starter.sum()
            if starter_sum > 0:
                starter /= starter_sum
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Random Dirichlet sampling (quick broad search)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0           # keep ~5 min for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n=n,
            trials=2500,
            deadline=phase_deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # --------------------------------------------------------------
    # 2️⃣ LP‑based improvement (core of new method)
    # --------------------------------------------------------------
    lp_phase_deadline = deadline - 200.0
    if time.time() < lp_phase_deadline:
        # take the current best plus a few additional random seeds
        candidates = [best_seq]
        # generate 9 extra random Dirichlet vectors (lightweight)
        extra = 9
        rng = np.random.default_rng()
        for _ in range(extra):
            if time.time() >= lp_phase_deadline:
                break
            c = np.exp(rng.uniform(np.log(0.02), np.log(30.0)))
            a = rng.gamma(shape=c, scale=1.0, size=n)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            candidates.append(a)

        for idx, cand in enumerate(candidates):
            if time.time() >= lp_phase_deadline:
                break
            print(f"[{time.time() - start_time:.2f}s] LP start on candidate {idx+1}/{len(candidates)}")
            best_seq, best_score = _lp_improvement(
                a0=cand,
                max_iter=12,                     # a dozen LP steps are cheap
                deadline=lp_phase_deadline,
                best_seq=best_seq,
                best_score=best_score,
                t=0.3,
            )

    # --------------------------------------------------------------
    # 3️⃣ Fine‑tune with projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting PGD fine‑tune")
        best_seq, best_score = _pgd_descent(
            a0=best_seq,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] After PGD best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Final hill‑climb / simulated‑annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            a0=best_seq,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
