def project_simplex(v):
    # v a numpy array
    # ensure non-negative after projection onto simplex
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    the method converge to a true max‑subgradient while escaping plateaus
    that trapped the previous sub‑gradient descent.

The best sequence found (normalised to sum = 1) is returned as a plain
Python list.  All utilities are top‑level; no closures or file I/O are
used.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a fast FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣ Parametric families (light‑weight sweeps)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    eps_grid = np.logspace(-5, -1, 120)          # reduced grid for speed
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        # mix with uniform – only a few mixing ratios
        for lam in (0.0, 0.5, 1.0):
            seq = lam * np.full(n, 1.0 / n) + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay from the centre."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.3, 80):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends with a tunable split."""
    for L_frac in (0.02, 0.05, 0.1, 0.2):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in (0.2, 0.5, 0.8):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣ Random Dirichlet draws
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 1200
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration between 0.01 and 5 (log‑uniform)
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣ Smooth‑max gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0, deadline, best_seq, best_score):
    """
    Gradient descent on the surrogate   τ·log∑exp(conv/τ) .
    τ is multiplied by a slowly decreasing factor.
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0

    # initial temperature – a fraction of the current max‑conv
    cur_conv = _conv_self(a)
    tau = max(cur_conv.max() * 0.4, 1e-6)   # start fairly smooth
    min_tau = 1e-9
    decay = 0.998                         # geometric cooling per iteration

    while time.time() < deadline and tau > min_tau:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        # ---- compute smoothed weights (softmax) ----
        shift = max_conv                       # for numerical stability
        log_w = (conv - shift) / tau
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0:
            # numerical disaster – reset temperature and continue
            tau *= decay
            continue
        p = w / w_sum                         # soft‑max probabilities

        # ---- gradient of the smooth max ----
        # grad = 2 * (p  * a_rev)  (convolution with reversed a)
        a_rev = a[::-1]
        L = _next_pow2(len(p) + len(a_rev) - 1)
        Pf = np.fft.rfft(p, n=L)
        Ar = np.fft.rfft(a_rev, n=L)
        conv_pa = np.fft.irfft(Pf * Ar, n=L)[: len(p) + len(a_rev) - 1]
        # pick the central window
        grad = 2.0 * conv_pa[(n - 1):(n - 1 + n)]

        # ---- line‑search on the true (non‑smoothed) objective ----
        cur_obj = 2.0 * n * max_conv
        step = 1.0
        improved = False
        while step > 1e-14:
            a_new = a - step * grad
            a_new = _project_simplex(a_new)     # keep on the simplex
            conv_new = _conv_self(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < cur_obj - 1e-12:       # strict improvement
                a = a_new
                cur_obj = obj_new
                improved = True
                if obj_new < best_score:
                    best_score, best_seq = obj_new, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] SmoothGD it={it:,} obj={obj_new:.6f} τ={tau:.2e}")
                break
            step *= 0.5

        if not improved:
            # small random jitter to escape flat regions
            eps = 1e-8
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)

        tau *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function.  The routine performs quick parametric sweeps,
    random Dirichlet initialisation and a smooth‑max gradient descent.
    The best sequence (normalised to sum = 1) is returned as a list of
    Python floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Quick parametric sweep (≈150 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 250.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (≈120 s)
    # --------------------------------------------------------------
    phase_deadline = deadline - 120.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Smooth‑max gradient descent (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max gradient descent")
        best_seq, best_score = _smooth_max_gradient_descent(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End of smooth‑max descent, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
