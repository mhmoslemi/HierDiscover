"""
Propose a step‑function sequence that minimises

    2 * n * max(conv(a, a)) / sum(a)^2

(the sum constraint is irrelevant after normalisation).  

The optimiser works in three stages:

1.  Random Dirichlet / geometric mixtures give a diverse pool of starting
    points.
2.  A smooth (log‑sum‑exp) approximation of the max‑convolution is minimised
    with Adam on the probability simplex.  The smoothing parameter β is
    increased gradually (continuation) so the iterates first see a soft
    objective and finally approach the true ℓ∞‑norm.
3.  A tiny‑mass hill‑climb (the “final polish” from the previous baseline)
    removes any remaining tiny peaks.

All steps respect the global 1000 s wall‑clock budget and the best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    (sum(v)=1, v_i ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum(a) = 1 is assumed)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _smooth_objective_and_grad(
    a: np.ndarray, beta: float
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv) using log‑sum‑exp with parameter beta.
    Returns (approx_max, grad) where grad is ∂/∂a of the smooth max
    (the scaling by 2·n is applied later).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    max_val = conv.max()
    shifted = conv - max_val                 # ≤ 0
    exp_beta = np.exp(beta * shifted)        # safe: max is 1, min underflows to 0
    Z = exp_beta.sum()
    softmax = exp_beta / Z
    approx_max = max_val + np.log(Z) / beta

    # gradient of the smooth max (no factor 2·n yet):
    #   ∂≈max/∂a_j = 2 * Σ_i a_i * softmax_{i+j}
    s = softmax
    rev_s = s[::-1]
    # convolution a * rev_s gives Σ_i a_i * s_{i+j} at index (len(s)-1‑j)
    conv2 = _conv_fft(a, rev_s)                      # length 3n‑2
    M = len(s)                                       # =2n‑1
    grad = 2.0 * conv2[M - 1 - np.arange(n)]         # shape (n,)
    return approx_max, grad


# ----------------------------------------------------------------------
# Local optimiser (smooth objective + Adam on the simplex)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    beta_start: float,
    beta_end: float,
    lr: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of 2·n·max(conv) via a smooth surrogate.
    Returns the best sequence encountered and its exact objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = _objective_score(a)

    t = 0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- continuation on the smoothing parameter -----------------
        beta = beta_start + (beta_end - beta_start) * min(it / max_iters, 1.0)

        # smooth objective + gradient (gradient is of the smooth max)
        approx_max, grad_approx = _smooth_objective_and_grad(a, beta)
        # gradient of the true objective = 2·n * grad_approx
        grad = 2.0 * n * grad_approx

        # --- Adam update --------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # exact objective for monitoring (cheap enough)
        cur_score = _objective_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it} β={beta:.1f} → {cur_score:.6f}"
            )

        a = a_new

    return best_a, best_score

def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using a Dirichlet distribution."""
    # concentration drawn log‑uniformly from 10⁻⁴ … 5
    logc = rng.uniform(-4.0, 0.7)
    c = 10 ** logc
    a = rng.gamma(shape=c, scale=1.0, size=n)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _random_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of left / right geometric tails and a uniform component."""
    idx = np.arange(n, dtype=np.float64)
    r = rng.uniform(0.80, 0.995)
    left = np.power(r, idx)
    right = np.power(r, n - 1 - idx)
    left /= left.sum()
    right /= right.sum()
    uniform = np.full(n, 1.0 / n)
    w = rng.random(3)
    w /= w.sum()
    a = w[0] * left + w[1] * right + w[2] * uniform
    a /= a.sum()
    return a

# Tiny‑mass hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _hill_climb(start_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Very small mass moves (≈0.4 % of a source entry) to fine‑tune the max."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)  # ≤0.4 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (sum(a)=1).  Used only by the hill‑climb.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate n gives a good trade‑off between quality
    # and speed.  Larger n often yields smaller bounds but slows each
    # iteration; 4096 works well for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Phase 1 – many cheap random starts + a few Adam steps each
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.2
    while time.time() < phase1_deadline:
        # choose a random type of starter
        if rng.random() < 0.6:
            a0 = _random_dirichlet(n, rng)
        else:
            a0 = _random_mixture(n, rng)

        sc0 = _objective_score(a0)
        if sc0 < best_score:
            best_score, best_seq = sc0, a0.copy()
            print(f"[{time.time() - start_time:.2f}s] Random start → {sc0:.6f}")

        # a tiny Adam run (few iterations – just to climb out of obvious bad spots)
        max_iters = 300
        a_opt, sc_opt = _adam_optimize(
            a0,
            max_iters=max_iters,
            beta_start=1.0,
            beta_end=30.0,
            lr=0.12,
            deadline=phase1_deadline,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Phase‑1 Adam → {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation starting from the current best
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.75
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting long Adam run")
        a_opt, sc_opt = _adam_optimize(
            best_seq,
            max_iters=5000,
            beta_start=10.0,
            beta_end=300.0,
            lr=0.08,
            deadline=phase2_deadline,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time() - start_time:.2f}s] Phase‑2 Adam → {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final tiny‑mass hill‑climb (uses the sub‑gradient routine)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (just in case) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
