def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using the FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _weighted_sum(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute   h_i = Σ_{j=0}^{n-1} w_{i+j} * a_j   (i = 0…n‑1).

    This is a single linear convolution using FFT:
        h = (w * a_rev)[n‑1 : n‑1+n]   with a_rev = a[::-1].
    """
    n = a.shape[0]
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, n=L)
    AR = np.fft.rfft(a[::-1], n=L)
    full = np.fft.irfft(W * AR, n=L)[: len(w) + n - 1]
    start = n - 1
    return full[start:start + n]


# ----------------------------------------------------------------------
# Simplex projection & objective helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Return the raw (un‑scaled) objective 2·n·max(conv(a,a))."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=20000):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)          # concentration ~0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=15000):
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix r={r:.3f} w={w} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent
# ----------------------------------------------------------------------


def _smoothed_descent(start_seq, deadline, best_seq, best_score):
    """
    Gradient descent on the smoothed maximum of the self‑convolution.
    The temperature ε is annealed from a relatively large value to a tiny
    one, turning the smooth objective into the true max‑objective.
    """
    a = start_seq.copy()
    n = a.shape[0]

    # annealing schedule
    eps = 0.05           # initial temperature
    eps_min = 1e-6
    eps_decay = 0.995   # multiplicative decay per 20 iterations

    # step‑size control
    step_nom = 0.5
    min_step = 1e-12

    iteration = 0

    while time.time() < deadline and eps > eps_min:
        iteration += 1

        # current objective (exact max, not the smoothed one)
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smoothed it={iteration} ε={eps:.2e} → {cur_score:.6f}")

        # ----- smoothed gradient -----
        cmax = cur_max
        # robust softmax (shift to avoid overflow)
        exp_arg = (conv - cmax) / eps
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w /= w_sum                     # w is a probability distribution over shifts

        # weighted convolution term   h_i = Σ_j w_{i+j} a_j
        h = _weighted_sum(a, w)
        grad = 2.0 * h                 # exact gradient of ε·log‑sum‑exp

        # back‑tracking line search in the projected space
        gnorm = np.linalg.norm(grad) + 1e-12
        step_len = step_nom / gnorm

        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)

            new_max = _conv_self(a_new).max()
            new_score = 2.0 * n * new_max
            if new_score < cur_score or step_len <= min_step:
                break
            step_len *= 0.5

        if new_score < cur_score:
            a = a_new
            step_nom = min(step_nom * 1.05, 2.0)   # gently increase nominal step
        else:
            step_nom = max(step_nom * 0.5, min_step)

        # anneal temperature every ~20 iterations
        if iteration % 20 == 0:
            eps = max(eps * eps_decay, eps_min)

        # occasional tiny random jitter to escape plateaus
        if iteration % 300 == 0:
            rng = np.random.default_rng()
            i1, i2 = rng.choice(n, size=2, replace=False)
            delta = a[i1] * (rng.random() * 0.005 + 1e-8)
            a[i1] -= delta
            a[i2] += delta
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (tiny‑mass moves)
# ----------------------------------------------------------------------


def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        # pick source (largest gradient) and destination (smallest)
        conv = _conv_self(a)
        # gradient of max‑objective (same as in the reference code)
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        src_cand = np.where(a > 1e-15)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        eps = a[src] * (rng.random() * 0.004 + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        prop_score = float(2.0 * n * _conv_self(a_prop).max())
        if prop_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = prop_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-8)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 provides a good trade‑off between flexibility
    # and speed (the constant in the bound is essentially independent of n).
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – Dirichlet random draws (≈20 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.20
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=20000
        )

    # ------------------------------------------------------------------
    # Phase 2 – geometric / uniform mixtures (≈10 % of budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.30
    if time.time() < phase2_deadline:
        best_seq, best_score = _random_mixture_trials(
            n, best_seq, best_score, phase2_deadline, trials=15000
        )

    # ------------------------------------------------------------------
    # Phase 3 – smoothed‑max gradient descent (≈45 % of budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + budget_s * 0.75
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed‑max descent")
        best_seq, best_score = _smoothed_descent(
            best_seq, phase3_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After descent → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final tiny‑mass hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
