def _conv_fft(a,b):
    n = a.shape[0]; m = b.shape[0]
    L = _next_pow2(n+m-1)
    Af = rfft(a, n=L)

"""
Smoothed‑max Adam optimizer + short hill‑climb.

We minimise   2·n·max(conv(a,a))  with a≥0, sum(a)=1.
The objective is scale‑invariant, so we keep the vector on the probability simplex.
A smooth approximation of the max (log‑sum‑exp) provides a useful gradient.
Adam adapts the step‑size, and a final greedy hill‑climb cleans up the last
tiny improvements.

The routine respects a 1000 s wall‑clock budget and works for any length n
(4096 ≤ n ≤ 16384).  If a global starter `height_sequence_1` is present it is used,
otherwise an arcsine‑shaped distribution (large weight at the ends) is the start.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _full_conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using FFT (real → real)."""
    L = _next_pow2(len(x) + len(y) - 1)
    Xf = rfft(x, n=L)
    Yf = rfft(y, n=L)
    conv = irfft(Xf * Yf, n=L)[: len(x) + len(y) - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {v≥0, sum(v)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (used for the final hill‑climb).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _grad_smoothed_max(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) of conv(a,a).
    Uses the identity  ∂conv_k/∂a = a_{k‑i}+a_{i‑k}.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    max_val = conv.max()
    exp_vals = np.exp(alpha * (conv - max_val))
    w = exp_vals / exp_vals.sum()            # soft‑max weights, sum to 1

    # term2 = (w * a)[0:n]
    term2 = _full_conv_fft(w, a)[:n]

    # term1 = (w ⋆ a)[0:n]  (cross‑correlation) → conv(w, a[::-1]) slice
    a_rev = a[::-1]
    term1_full = _full_conv_fft(w, a_rev)
    term1 = term1_full[n - 1 : n - 1 + n]

    grad = 2.0 * (term1 + term2)
    return grad


# ----------------------------------------------------------------------
# Optimisation stages
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed‑max objective."""
    global start_time
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                     # base learning‑rate
    it = 0

    # smoothing schedule
    alpha_start, alpha_max = 2.0, 200.0
    while time.time() < deadline:
        it += 1
        # linear increase of the smoothing until a plateau
        if it < 20000:
            alpha = alpha_start + (alpha_max - alpha_start) * (it / 20000.0)
        else:
            alpha = alpha_max

        # gradient of the smooth max
        grad = _grad_smoothed_max(a, alpha)

        # Adam update (descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it} score={cur_score:.6f}")

        # occasional tiny jitter to escape plateaus
        if it % 5000 == 0:
            i1, i2 = rng.choice(len(a), size=2, replace=False)
            epsj = a[i1] * (rng.random() * 1e-4 + 1e-8)
            a[i1] -= epsj
            a[i2] += epsj
            a = _project_simplex(np.maximum(a, 0.0))

    return best_a, best_score


def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small deterministic hill‑climb that transfers a tiny fraction of mass
    from the “over‑represented’’ index to the “under‑represented’’ one.
    """
    global start_time
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_cand = np.where(a > 1e-15)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # move up to about 2 % of the source mass
        eps = a[src] * (rng.random() * 0.02 + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_seq = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 5000:
                # random jitter to get out of a flat region
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising
        2·n·max(conv(a,a)) / (sum(a)^2)

    The routine:
      1. Chooses a problem size (4096–16384) depending on the time budget.
      2. Starts from an arcsine‑shaped distribution (or the global starter if present).
      3. Runs a smoothed‑max Adam optimiser for most of the time.
      4. Finishes with a short greedy hill‑climb.
      5. Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin for the caller

    # --------------------------------------------------------------
    # Choose problem size – larger n gives a better asymptotic bound.
    # --------------------------------------------------------------
    if budget_s >= 800:
        n = 16384
    elif budget_s >= 400:
        n = 8192
    else:
        n = 4096

    # --------------------------------------------------------------
    # Initialise – use a global starter if it exists, otherwise an arcsine shape.
    # --------------------------------------------------------------
    a0 = None
    if 'height_sequence_1' in globals():
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                # truncate or pad to length n
                if starter.size >= n:
                    a0 = starter[:n].copy()
                else:
                    a0 = np.pad(starter, (0, n - starter.size), constant_values=0.0)
        except Exception:
            a0 = None

    if a0 is None:
        # arcsine‑like weights: w_i ∝ 1/√((i+1)*(n‑i))
        idx = np.arange(n)
        w = 1.0 / np.sqrt((idx + 1) * (n - idx))
        a0 = w / w.sum()

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)      # ensure exact normalisation
    best_seq = a0.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial n={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – Adam optimiser (most of the budget)
    # --------------------------------------------------------------
    adam_deadline = deadline - 60.0      # keep ~1 min for the final hill‑climb
    if time.time() < adam_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimiser")
        best_seq, best_score = _adam_optimize(best_seq, adam_deadline, rng)

    # --------------------------------------------------------------
    # Stage 2 – final hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline,
                                           best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
