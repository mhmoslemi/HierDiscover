assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where `a*a` is the linear self‑convolution.  
Because the objective is homogeneous, we may restrict ourselves to the
probability simplex (`∑a_i = 1`).  Then  

\[
F(a)=2\,n\;\max_k (a*a)_k .
\]

The optimisation is non‑convex, therefore a *multi‑phase* strategy is used:

1. **Broad random exploration** – many Dirichlet draws and a small
   parametric sweep (left/right geometric shapes).  
2. **Short restarts of a first‑order optimiser** – each restart starts
   from a different random point and performs a few Adam‑style
   sub‑gradient steps (projected onto the simplex).  The best result of
   all restarts is kept.
3. **Long Adam descent** – starting from the best candidate of step 2,
   Adam with a back‑tracking line‑search is run for the bulk of the
   time‑budget.
4. **Simulated annealing fine‑tuning** – random mass transfers between
   two coordinates are accepted with a Metropolis rule (temperature is
   slowly reduced).  This helps escaping the tiny plateaus that the
   deterministic sub‑gradient method cannot leave.

All phases respect the global 1000 s wall‑clock budget.
The code follows the same low‑level utilities as the reference
implementation (FFT based convolution, Euclidean projection onto the
simplex) but adds the extra phases and a more aggressive stochastic
search.

```python
"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four phases:
  1) Wide random exploration (Dirichlet draws and simple geometric families).
  2) Many short Adam‑style sub‑gradient restarts.
  3) One long Adam descent from the best restart.
  4) Simulated‑annealing fine‑tuning.

All phases stop automatically when the overall 1000 s budget is reached.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities (same as in the reference code)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (non‑circular) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assume sum(a)=1)
# ----------------------------------------------------------------------


def _objective_score(a: np.ndarray) -> float:
    """2 * n * max(conv(a,a)) for a on the simplex."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (a lies on the simplex).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _explore_random(n, deadline, best_seq, best_score):
    """Dirichlet draws + simple geometric families."""
    rng = np.random.default_rng()
    # Dirichlet draws (different concentration levels)
    while time.time() < deadline:
        logc = rng.uniform(-2.0, 0.7)          # concentration ~ 0.01 … 5
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
        # stop after a few thousand samples to keep time budget for later phases
        if time.time() > deadline:
            break

    # Simple geometric families (left, right, symmetric)
    idx = np.arange(n, dtype=np.float64)
    while time.time() < deadline:
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        sym = np.power(r, np.abs(idx - (n - 1) / 2.0))
        for base in (left, right, sym):
            base = base / base.sum()
            sc = _objective_score(base)
            if sc < best_score:
                best_score, best_seq = sc, base.copy()
                print(f"[{time.time() - start_time:.2f}s] Geometric r={r:.3f} → {sc:.6f}")
    return best_seq, best_score

def _adam_restart(start_a, max_iters, deadline, best_seq, best_score):
    """Very short Adam descent (max_iters iterations or time‑out)."""
    a = start_a.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.08                     # nominal step size
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_self(a)
        cur = float(2.0 * n * conv.max())
        if cur < best_score:
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Restart it={t} score={cur:.6f}")

        grad = _subgradient_maxconv(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        nxt = float(2.0 * n * _conv_self(a_new).max())
        if nxt < cur:
            a = a_new                       # accept improvement
        else:
            lr *= 0.7                       # shrink step if we are stuck
    return best_seq, best_score


def _many_short_restarts(n, deadline, best_seq, best_score):
    """Generate many random starting points and run a few Adam steps each."""
    rng = np.random.default_rng()
    restart_deadline = deadline
    while time.time() < restart_deadline:
        # a fresh random start – Dirichlet with one of few concentrations
        logc = rng.uniform(-1.5, 0.5)
        c = 10.0 ** logc
        a0 = rng.gamma(shape=c, scale=1.0, size=n)
        if a0.sum() == 0:
            continue
        a0 /= a0.sum()
        # a very short run (≈ 80 inner iterations or the remaining time)
        best_seq, best_score = _adam_restart(
            a0, max_iters=80, deadline=restart_deadline,
            best_seq=best_seq, best_score=best_score
        )
    return best_seq, best_score

            iter_no_improve += 1

        grad = _subgradient_maxconv(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        nxt = float(2.0 * n * _conv_self(a_new).max())

        if nxt < cur:
            a = a_new                   # keep improvement
            lr = min(lr * 1.03, 0.6)    # gently increase step size
        else:
            lr = max(lr * 0.6, 1e-4)    # shrink step size

        # Every few hundred iterations add a tiny random jitter to escape flats
        if t % 500 == 0:
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps_j = a[i1] * (rng.random() * 0.002 + 1e-6)
            a[i1] -= eps_j
            a[i2] += eps_j
            a = _project_simplex(np.maximum(a, 0.0))

        # If we have been stagnant for a long time, re‑randomise a small part
        if iter_no_improve > 3000:
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps_j = a[i1] * (rng.random() * 0.005 + 1e-6)
            a[i1] -= eps_j
            a[i2] += eps_j
            a = _project_simplex(np.maximum(a, 0.0))
            iter_no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated annealing fine‑tuning
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq, deadline, best_seq, best_score):
    """Metropolis‑type annealing with mass transfers between two coordinates."""
    a = start_seq.copy()
    n = a.shape[0]
    cur_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    # Annealing schedule
    T = 0.015                # initial temperature
    Tmin = 1e-6
    cooling = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1
        # choose a source (must have some mass) and a destination
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(0, n)
        if src == dst:
            continue

        # transfer a small (but not infinitesimal) fraction of src
        max_eps = a[src] * 0.12            # up to 12 % of the source mass
        eps = rng.random() * max_eps

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        new_score = float(2.0 * n * _conv_self(a_prop).max())
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it} → {best_score:.6f}")

        # cool down
        T = max(T * cooling, Tmin)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with minimal evaluation score.

    The search consists of:
      * random Dirichlet / geometric exploration,
      * many short Adam restarts,
      * a long Adam descent,
      * final simulated‑annealing polishing.

    The total wall‑clock time never exceeds ``budget_s`` seconds.
    The function returns a plain Python list ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 entries give a good trade‑off between quality
    # and speed.  The algorithm works with any n ≥ 2.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) if it exists.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            ss = starter.sum()
            if ss > 0:
                starter = starter / ss
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – broad random exploration (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + budget_s * 0.10
    if time.time() < phase0_deadline:
        best_seq, best_score = _explore_random(
            n, phase0_deadline, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Phase 1 – many short Adam restarts (≈25 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.35
    if time.time() < phase1_deadline:
        best_seq, best_score = _many_short_restarts(
            n, phase1_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After short restarts → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam descent (≈45 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.80
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting long Adam descent")
        best_seq, best_score = _adam_long(
            best_seq, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After long Adam → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After annealing → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (just in case) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
