Now subgradient:

def _subgradient_maxconv(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]   # a[k-i] where idx = i
    if active.size:
        grad *= 2.0 / active.size
    return grad

def _projected_gradient_descent(a0, n, L_fft, start_time, deadline, best_seq, best_score):
    a = a0.copy()
    momentum = np.zeros_like(a)
    it = 0
    last_improve_it = 0
    restart_interval = 200_000  # may adjust
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] PGD it={it} best={cur_score:.6f}")

        # subgradient of max conv (increase direction)
        grad = _subgradient_maxconv(a, conv)

        # step size schedule
        eta = 1.0 / math.sqrt(it)  # can adjust scaling factor later

        # momentum update (Nesterov like)
        momentum = 0.9 * momentum - eta * grad
        a_candidate = a + momentum

        # project onto simplex
        a_candidate = _project_simplex(a_candidate)

        # evaluate candidate
        conv_cand = _conv_self(a_candidate, L_fft)
        cand_score = 2.0 * n * conv_cand.max()

        # accept if improvement
        if cand_score < cur_score:
            a = a_candidate
            # optionally keep momentum
        else:
            # with small probability accept uphill
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 1e-5 * (remaining / total)  # small temperature
            if np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = a_candidate
            else:
                # optionally discard momentum
                momentum = np.zeros_like(a)

        # restart if stagnation
        if it - last_improve_it > restart_interval:
            # random perturbation
            perturb = np.random.default_rng().normal(scale=0.02, size=n) * a
            a = _project_simplex(np.maximum(a + perturb, 0.0))
            momentum = np.zeros_like(a)
            last_improve_it = it
    return best_seq, best_score

"""
Projected‑gradient descent on the probability simplex for the
self‑convolution L∞‑minimisation problem.

The routine works in three stages

1️⃣  Random β‑shaped seeds (β‑distribution PDFs on the discrete grid).
    The best seed is kept as a warm‑start.

2️⃣  Additive projected‑gradient descent with Nesterov momentum.
    The sub‑gradient of the max‑convolution is used;
    after each step the iterate is projected onto the simplex
    (non‑negative entries summing to 1).  The step‑size decays as 1/√t
    and a tiny simulated‑annealing acceptance rule helps to escape plateaus.
    Random perturbations are injected if no improvement is seen for a long
    stretch.

3️⃣  A short golden‑section search mixes the best vector with the uniform
    distribution to exploit the convexity of the feasible set.

The whole search respects a 1000 s budget (a small safety margin is kept).
The function returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
from typing import Tuple, Optional

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat values within a tiny tolerance as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution a_{k‑i}
    if active.size:
        grad = 2.0 * grad / active.size   # factor 2 because each term appears twice
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used during optimisation: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be projected to zero – fall back to uniform
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(alpha: float, beta: float, n: int) -> Optional[np.ndarray]:
    """Return a normalised Beta(α,β)‑shaped discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n          # mid‑point of each bin
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _maybe_use_global_starter(n: int) -> Tuple[Optional[np.ndarray], float]:
    """If a global variable `height_sequence_1` exists, try to use it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


# ----------------------------------------------------------------------
# 1️⃣  Random β‑shape search
# ----------------------------------------------------------------------
def _beta_random_search(n: int,
                        best_seq: np.ndarray,
                        best_score: float,
                        deadline: float,
                        start_time: float) -> Tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # log‑uniform sampling of the two shape parameters
    log_a_min, log_a_max = math.log(0.01), math.log(10.0)
    while time.time() < deadline:
        alpha = math.exp(rng.uniform(log_a_min, log_a_max))
        beta = math.exp(rng.uniform(log_a_min, log_a_max))
        v = _beta_shape(alpha, beta, n)
        if v is None:
            continue
        sc = _objective_score(v, n)
        if sc < best_score:
            best_score, best_seq = sc, v.copy()
            print(f"[{time.time() - start_time:.2f}s] β‑seed α={alpha:.3f},β={beta:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Projected gradient descent with momentum
# ----------------------------------------------------------------------
def _pgd_momentum(a0: np.ndarray,
                  n: int,
                  L_fft: int,
                  start_time: float,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float) -> Tuple[np.ndarray, float]:
    """
    Additive projected‑gradient descent on the simplex with Nesterov‑style
    momentum.  The sub‑gradient of the max‑convolution is used.
    """
    a = a0.copy()
    momentum = np.zeros_like(a)
    it = 0
    last_improve_it = 0
    stagnation_limit = 250_000          # steps without improvement → perturb

    rng = np.random.default_rng()
    mu = 0.9                             # momentum coefficient

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()

        # record any improvement
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} best={cur_score:.6f}")

        # sub‑gradient of the current max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # step‑size schedule (1/√t) with a modest scaling factor
        eta = 0.8 / math.sqrt(it)

        # Nesterov‑style momentum update (additive)
        momentum = mu * momentum - eta * grad
        a_candidate = a + momentum
        a_candidate = _project_simplex(a_candidate)

        # evaluate candidate (cheap)
        conv_cand = _conv_self(a_candidate, L_fft)
        cand_score = 2.0 * n * conv_cand.max()

        # accept if improvement – otherwise occasional uphill move
        if cand_score < cur_score:
            a = a_candidate
        else:
            # very small temperature that decays to zero near the deadline
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                a = a_candidate
            else:
                # reject – discard the accumulated momentum to avoid drifting into bad region
                momentum = np.zeros_like(a)

        # stagnation handling
        if it - last_improve_it > stagnation_limit:
            # inject a modest random perturbation
            perturb = rng.normal(scale=0.02, size=n) * a
            a = _project_simplex(np.maximum(a + perturb, 0.0))
            momentum = np.zeros_like(a)
            last_improve_it = it

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> Tuple[np.ndarray, float]:
    """
    Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform using a few golden‑section
    evaluations.  The best mixture is kept.
    """
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    # golden‑section loop
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a tiny safety margin for the runner
    deadline = start_time + budget_s - 0.25

    # Problem size – 4096 is the size used in the reference solution
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ----- 0️⃣  Initialise with uniform & possible global starter -----
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ----- 1️⃣  Random β‑shape search (≈5 % of the budget) -----
    phase0_deadline = start_time + 0.05 * (deadline - start_time)
    if time.time() < phase0_deadline:
        best_seq, best_score = _beta_random_search(
            n, best_seq, best_score, phase0_deadline, start_time
        )
        print(f"[{time.time() - start_time:.2f}s] After β‑search best score≈{best_score:.6f}")

    # ----- 2️⃣  Projected‑gradient descent with momentum (≈80 % of the budget) -----
    phase1_deadline = deadline - 5.0            # reserve a few seconds for final mixing
    if time.time() < phase1_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting PGD with momentum")
        best_seq, best_score = _pgd_momentum(
            best_seq, n, L_fft, start_time, phase1_deadline,
            best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After PGD best score≈{best_score:.6f}")

    # ----- 3️⃣  Fine mixing with the uniform distribution (remaining time) -----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ----- 4️⃣  Final normalisation and return -----
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Convert to plain Python list of floats for the grader
    return [float(v) for v in best_seq]
