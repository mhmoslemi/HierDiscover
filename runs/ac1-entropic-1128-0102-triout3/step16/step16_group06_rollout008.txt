def _smooth_obj(x, eps):
    # x is vector of length n
    # Compute a = softmax(x)
    xm = np.max(x)
    ex = np.exp(x - xm)
    a = ex / ex.sum()
    # conv via FFT
    conv = _conv_self(a)
    # smooth max
    cmax = np.max(conv)
    # Compute log-sum-exp: logsumexp = cmax + eps * np.log(np.sum(np.exp((conv - cmax) / eps)))
    # Actually we need eps*logsumexp(conv/eps)
    # So:
    shifted = (conv - cmax) / eps
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth_max = cmax + eps * np.log(sum_exp)
    f = 2.0 * n * smooth_max
    # compute w = exp((conv-cmax)/eps) / sum_exp
    w = exp_shifted / sum_exp
    # gradient wrt a
    # grad_a = 2 * np.correlate(w, a, mode='valid')
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    # gradient wrt x via chain rule:
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return f, grad_x
```

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The method combines three ideas:

1.  A small‑scale L‑BFGS optimisation of a *smoothed* max‑convolution
    objective.  The vector is parametrised as a soft‑max (log‑weights)
    so that the simplex constraints are automatically satisfied.
    The smoothing parameter ε is decreased in stages (0.1 → 5e‑4).

2.  A fast up‑sampling of the best low‑dimensional solution to the
    required length (default n = 4096).

3.  A targeted hill‑climb that repeatedly reduces the dominant
    convolution peak by moving a tiny amount of mass from the pair of
    indices that contribute most to the current maximal shift
    towards a pair that contributes least.

All phases respect the global wall‑clock budget (default 1000 s) and
print intermediate scores.  The final sequence is normalised to sum to
1, which leaves the evaluation function invariant.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global state for convenience (set inside propose_candidate)
# ----------------------------------------------------------------------
_N = None          # length of the vector (set at runtime)
_start_time = None  # start time for logging


# ----------------------------------------------------------------------
# Low‑level FFT convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (for monitoring)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj(x: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv(a,a))
        grad = ∂f/∂x    (x are log‑weights, a = softmax(x)).
    eps – smoothing parameter (positive).
    """
    n = _N
    # soft‑max to obtain a simplex‑constrained vector
    xmax = np.max(x)
    e_x = np.exp(x - xmax)
    a = e_x / e_x.sum()

    # self‑convolution
    conv = _conv_self(a)

    # smooth approximation of max via log‑sum‑exp
    cmax = np.max(conv)
    # shift for stability
    shifted = (conv - cmax) / eps
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth_max = cmax + eps * np.log(sum_exp)      # = eps·logsumexp(conv/eps)

    f = 2.0 * n * smooth_max

    # soft‑max weights for the smoothed max (probability distribution)
    w = exp_shifted / sum_exp

    # gradient w.r.t a : grad_a = 2 * (w ⋆ a)   (cross‑correlation)
    # np.correlate implements the required sum_{k} w_k * a_{k-i}
    grad_a = 2.0 * np.correlate(w, a, mode='valid')

    # chain rule via soft‑max Jacobian:  ∂f/∂x = a * (grad_a – (a·grad_a))
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    return f, grad_x


# ----------------------------------------------------------------------
# Targeted hill‑climb that attacks the dominant convolution shift
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a tiny amount of mass from the index pair that
    contributes most to the current maximal convolution entry to the
    index pair that contributes least.  Stops when the time budget expires.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(best_a).max())
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        k_max = int(np.argmax(conv))

        # indices that actually participate in this shift
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution of each participating index
        contrib = a[idx] * a[k_max - idx]

        i_src = idx[int(np.argmax(contrib))]
        i_dst = idx[int(np.argmin(contrib))]

        # if both coincide pick a random alternative
        if i_src == i_dst:
            i_src, i_dst = rng.choice(n, size=2, replace=False)

        # move a small fraction of the source mass
        eps = a[i_src] * (rng.random() * 0.004 + 1e-6)   # up to ~0.4 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[i_src] -= eps
        a_prop[i_dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a_prop.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - _start_time:.2f}s] Hill‑climb step → {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # random jitter to escape possible plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                if jitter <= a[i1]:
                    a[i1] -= jitter
                    a[i2] += jitter
                no_improve = 0
    return best_a, best_score

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - _start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1, if supplied by the platform)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - _start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet draws (≈10 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = _start_time + budget_s * 0.10
    dirichlet_trials = 80
    for _ in range(dirichlet_trials):
        if time.time() >= phase1_deadline:
            break
        conc = rng.uniform(0.05, 5.0)          # concentration parameter
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - _start_time:.2f}s] Dirichlet trial → {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth L‑BFGS optimisation on the soft‑max parametrisation
    # ------------------------------------------------------------------
    # start point for the log‑weights (soft‑max inverse)
    x0 = np.log(np.maximum(best_seq, 1e-12))

    # schedule of ε values (coarse → fine)
    eps_schedule = [0.1, 0.03, 0.01, 0.003, 0.001, 5e-4]
    # 200 iterations per stage is generous yet cheap for n=4096
    maxiter_stage = 200

    for eps in eps_schedule:
        if time.time() >= deadline:
            break
        # minimise the smoothed objective; fun returns (f, grad)
        res = minimize(
            _smooth_obj,
            x0,
            args=(eps,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter_stage, 'disp': False}
        )
        x0 = res.x                       # warm‑start next stage
        # evaluate the exact (non‑smoothed) score
        a_opt = np.exp(x0 - np.max(x0))
        a_opt /= a_opt.sum()
        sc = _objective_score(a_opt)
        if sc < best_score:
            best_score, best_seq = sc, a_opt.copy()
            print(f"[{time.time() - _start_time:.2f}s] After ε={eps:.5f} → exact score {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted hill‑climb for final polishing (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - _start_time:.2f}s] Starting targeted hill‑climb")
        best_seq, best_score = _targeted_hill_climb(best_seq, deadline)
        print(f"[{time.time() - _start_time:.2f}s] Hill‑climb finished → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - _start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
