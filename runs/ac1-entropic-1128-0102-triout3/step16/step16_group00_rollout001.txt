"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search runs completely inside the supplied time budget (default 1000 s).
It proceeds in three stages:

1️⃣  **Population initialisation** – a small population of vectors on the
    probability simplex is created from diverse Dirichlet draws and from
    sparse random supports (including the uniform vector and, when available,
    a global starter `height_sequence_1`).

2️⃣  **Differential Evolution (DE)** – the population is evolved with the classic
    DE/rand/1/bin scheme.  After mutation and binomial crossover a candidate is
    projected back onto the simplex (Duchi et al. O(n log n) projection).  The
    best individual is kept as the current incumbent.

    DE is very cheap per evaluation (a single FFT‑based self‑convolution) and
    therefore can be run for the majority of the time budget, exploring the
    search space far more globally than the simulated‑annealing code used
    previously.

3️⃣  **Projected gradient descent with momentum** – the best vector from DE is
    fine‑tuned by moving opposite to a sub‑gradient of the max‑convolution.
    Momentum (exponential moving average of the gradients) accelerates the
    descent, and a simple back‑tracking line‑search adapts the step size.  The
    iterate is projected onto the simplex after each step.

All heavy operations (FFT, projection, gradient) are implemented with NumPy
only; no external solvers are required, keeping the memory footprint modest.
If the time limit is reached at any point the best vector seen so far is
returned (as a plain Python list).  The returned sequence is normalised to
unit sum – the objective is scale invariant, so any normalisation is valid.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a power‑of‑2 FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (v_i ≥ 0, Σv_i = 1).  Implements the O(n log n) algorithm of
    Duchi, Shalev‑Shwartz, Singer & Chandra.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (the vectors are kept normalised to sum=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) under the assumption Σa_i = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _dirichlet_vector(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random vector from a Dirichlet distribution with random concentration."""
    conc_log = rng.uniform(-3.0, 0.7)          # log10 of concentration ≈ [1e‑3, 5]
    c = 10 ** conc_log
    x = rng.gamma(shape=c, scale=1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else np.full(n, 1.0 / n)


def _sparse_vector(rng: np.random.Generator, n: int) -> np.ndarray:
    """Create a random sparse probability vector (few non‑zero entries)."""
    max_support = max(2, n // 20)           # at most 5 % non‑zero
    support = rng.integers(1, max_support + 1)
    idx = rng.choice(n, size=support, replace=False)
    vals = rng.random(support)
    vec = np.zeros(n, dtype=float)
    vec[idx] = vals
    s = vec.sum()
    if s == 0:
        vec = np.full(n, 1.0 / n)
    else:
        vec /= s
    return vec


# ----------------------------------------------------------------------
# Differential Evolution
# ----------------------------------------------------------------------
def _differential_evolution(pop: np.ndarray,
                            scores: np.ndarray,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    DE/rand/1/bin on the simplex.
    pop   – (pop_size, n) array of candidates (already feasible)
    scores – current objective values for the population
    Returns the best individual and its score.
    """
    pop_size, n = pop.shape
    rng = np.random.default_rng()
    F = 0.5               # differential weight
    CR = 0.9              # crossover probability

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    generation = 0
    while time.time() < deadline:
        for i in range(pop_size):
            # distinct indices a,b,c ≠ i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            donor = a_vec + F * (b_vec - c_vec)
            donor = np.maximum(donor, 0.0)

            # binomial crossover
            cross = rng.random(n) < CR
            trial = np.where(cross, donor, pop[i])

            # project back onto the simplex
            trial = _project_simplex(trial)

            trial_score = _objective_score(trial)

            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(f"[{time.time() - start_time:.2f}s] DE new best {best_score:.6f} (gen {generation})")
        generation += 1
        # adapt F/CR very slowly (optional – omitted for speed)
    return best_vec, best_score


# ----------------------------------------------------------------------
# Projected gradient descent with momentum
# ----------------------------------------------------------------------
def _proj_gradient_momentum(a0: np.ndarray,
                           cur_score: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Fine‑tune a vector by moving opposite to the sub‑gradient, using a
    momentum term and back‑tracking line‑search.  All iterates stay on the
    probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    step = 0.5
    mom = np.zeros_like(a)
    beta = 0.9

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # momentum update
        mom = beta * mom + (1.0 - beta) * grad

        cand = a - step * mom
        cand = _project_simplex(np.maximum(cand, 0.0))

        cand_score = _objective_score(cand)

        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            step = min(step * 1.05, 1.0)          # accelerate when successful
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
        else:
            step = max(step * 0.5, 1e-12)           # shrink step on failure
    return best_a, best_score
