"""
minimise_step_function.py

Search for a non‑negative step function (probability vector) that
minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The optimiser consists of five lightweight phases:

1️⃣  A fast parametric sweep over several analytic families.
2️⃣  Massive random Dirichlet sampling.
3️⃣  Adam optimisation of a smooth‑max surrogate.
4️⃣  Projected sub‑gradient descent with a back‑tracking line‑search
    (large, adaptive steps on the exact max‑convolution).
5️⃣  Aggressive “mass‑transfer” hill‑climb that moves a sizable
    fraction of probability from the highest‑gradient index to the
    lowest‑gradient one, with an inner line‑search for the optimal
    transfer amount.

All phases respect a global 1000 s budget; the best sequence found
is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used by the optimiser: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (geometric, power, beta, stretched‑exp,
#                     logistic, sine‑power)
# ----------------------------------------------------------------------
def _sweep_families(n: int,
                    best_seq: np.ndarray,
                    best_score: float,
                    deadline: float,
                    start_time: float) -> tuple[np.ndarray, float]:
    idx = np.arange(n, dtype=np.float64)

    # uniform baseline
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- geometric ----------------------------------------------------
    eps_grid = np.logspace(-5, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 9):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-2.0, 2.0, 101)
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta‑type ------------------------------------------------------
    alphas = np.linspace(0.05, 4.0, 61)
    betas = np.linspace(0.05, 4.0, 61)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # ---- stretched‑exponential -----------------------------------------
    scales = np.linspace(0.25, 1.0, 11)
    p_vals = np.linspace(0.2, 5.0, 41)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} scale={scale:.2f} score={sc:.6f}")

    # ---- logistic -------------------------------------------------------
    mu_factors = np.linspace(0.2, 0.8, 13)
    s_factors = np.linspace(0.01, 0.2, 11)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-6:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    # ---- sine‑power -----------------------------------------------------
    p_vals = np.linspace(0.1, 5.0, 41)
    for p in p_vals:
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    return best_seq, best_score

# 2️⃣  Random Dirichlet trials (more intensive)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 8000                     # increased number of trials
    for _ in range(trials):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.01), np.log(20.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   start_time: float,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04
    tau = 20.0
    tau_factor = 1.2
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max over convolution values
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max
        a_rev = a[::-1]
        conv_w = np.fft.irfft(np.fft.rfft(w, n=_next_pow2(3 * n - 2)) *
                              np.fft.rfft(a_rev, n=_next_pow2(3 * n - 2)),
                              n=_next_pow2(3 * n - 2))[:n]
        grad = 2.0 * conv_w

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        if it % 2500 == 0:
            tau = min(tau * tau_factor, 1e9)
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Projected sub‑gradient descent with line‑search (large steps)
# ----------------------------------------------------------------------
def _pgd_line_search(a0: np.ndarray,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(conv(a,a)).  At each iteration a back‑tracking line‑search
    is performed to obtain a robust step size (often much larger than
    the decaying step used in the original implementation).
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD‑LS it={it:,} best={best_score:.6f}")

        # sub‑gradient of max‑convolution (for the max value)
        subgrad = _subgradient_maxconv(a, conv)                 # ∂max/∂a
        grad = 2.0 * n * subgrad                               # ∂obj/∂a

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # initial step size – roughly 0.5 / ||grad|| (scale‑free)
        t = 0.5 / (grad_norm + 1e-12)

        # back‑tracking line search
        improved = False
        for _ in range(20):
            a_candidate = a - t * grad
            a_candidate = _project_simplex(a_candidate)
            new_score = _objective_score(a_candidate)
            if new_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = new_score
                improved = True
                break
            t *= 0.5

        if not improved:
            # occasional random shake to escape flat plateaux
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
            cur_score = _objective_score(a)

        # stop if step becomes vanishingly small
        if t * grad_norm < 1e-13:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Aggressive mass‑transfer hill‑climb (large ε, inner line‑search)
# ----------------------------------------------------------------------
def _large_move_refinement(a0: np.ndarray,
                           start_time: float,
                           deadline: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly transfers a sizable chunk of probability from the index
    with the highest sub‑gradient to the one with the lowest.
    For each pair an inner line‑search over the transfer fraction is
    performed, allowing much larger jumps than the original 0.4 % moves.
    """
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Large‑move it={it:,} best={best_score:.6f}")

        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # try a hierarchy of fractions (up to 10 % of src mass)
        fractions = [0.10, 0.05, 0.025, 0.0125, 0.00625]
        moved = False
        for frac in fractions:
            eps = a[src] * frac
            if eps <= 0:
                continue
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            new_score = 2.0 * n * _conv_self(a_prop, L_fft).max()
            if new_score < cur_score - 1e-12:
                a = a_prop
                cur_score = new_score
                moved = True
                if new_score < best_score:
                    best_score, best_seq = new_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] Move src={src}→dst={dst} frac={frac:.4f} score={new_score:.6f}")
                break

        if not moved:
            # random jitter when no improvement found
            jitter = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats (the best sequence found).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialisation (uniform or external starter)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad parametric sweep
    # --------------------------------------------------------------
    phase_deadline = min(deadline, start_time + 150.0)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline, start_time
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (more trials)
    # --------------------------------------------------------------
    phase_deadline = min(deadline, start_time + 300.0)
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline, start_time
        )

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation of smooth‑max surrogate
    # --------------------------------------------------------------
    adam_end = min(deadline, start_time + 450.0)
    if time.time() < adam_end:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, start_time, adam_end, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent with line‑search
    # --------------------------------------------------------------
    if time.time() < deadline:
        pgd_end = min(deadline, start_time + 750.0)
        print(f"[{time.time() - start_time:.2f}s] Starting PGD‑LS")
        best_seq, best_score = _pgd_line_search(
            best_seq, start_time, pgd_end, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After PGD‑LS best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Aggressive large‑move hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting large‑move refinement")
        best_seq, best_score = _large_move_refinement(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
