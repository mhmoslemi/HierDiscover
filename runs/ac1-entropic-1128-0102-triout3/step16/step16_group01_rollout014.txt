import numpy as np
import time

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _softmax_weights(conv, beta):
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()

def _grad_softmax(a, p, L=None):
    # compute gradient: 2 * a convolve p_rev
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1 + n - 1)  # enough length for correlation
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L)
    Pf = np.fft.rfft(p_rev, n=L)
    corr = np.fft.irfft(Af * Pf, n=L)
    # we need central n entries of correlation: indices (n-1) .. (2n-2)
    start = n - 1
    g = corr[start:start + n][::-1]  # reverse to align
    return 2.0 * g

def _subgrad_max_conv(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g

def _eval_fast(a, L_fft):
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

def _make_shape(n, r, p):
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

def _mix_with_uniform(shape, lam):
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

def _random_dirichlet(n, alpha, rng):
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return x / s

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation function.
    The search combines:
      * a fast parametric sweep over geometric×power families,
      * many random restarts each followed by an exponentiated‑gradient descent
        on the true max‑convolution objective (adaptive step size, momentum),
      * a final short random hill‑climb that moves tiny mass amounts.
    Total runtime respects the supplied budget (default 1000 s).
    Returns a plain Python list of floats (sum normalised to 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    # problem size
    n = 4096  # sweet spot between expressivity and speed
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(2 * n - 1 + n - 1)   # length for soft‑max gradients

    # ------------------------------------------------------------------
    # 0. baseline (uniform)
    # ------------------------------------------------------------------
    a_uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = a_uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0b. global starter if supplied
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (≈30 s)
    # ------------------------------------------------------------------
    sweep_deadline = start + 30.0
    r_grid = np.logspace(np.log10(0.905), np.log10(0.9999), 30)
    p_grid = np.linspace(-4.0, 3.0, 35)
    lam_grid = np.linspace(0.0, 1.0, 11)

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            shape = _make_shape(n, r, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(shape, lam)
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start:.2f}s] sweep r={r:.5f} p={p:.3f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random‑restart exponentiated‑gradient descent
    #    (≈600 s total)
    # ------------------------------------------------------------------
    # main hyper‑parameters
    eta0 = 1.0          # initial step size for multiplicative update
    decay = 0.9995      # exponential decay per iteration
    momentum = 0.9      # exponential moving average of gradients
    max_iter_total = 200000   # safety cap
    it = 0
    # store moving average of gradients to obtain a momentum‑like direction
    grad_ma = np.zeros(n, dtype=np.float64)

    while time.time() < deadline - 200.0 and it < max_iter_total:
        # possibly restart from a fresh random point
        if it % 5000 == 0:
            # take a fresh start that is a noisy copy of the current best
            a = best_seq.copy()
            eps = 0.02 * rng.random(n) * a   # up‑to 2 % relative noise
            a = a + eps
            a = np.maximum(a, 0.0)
            a /= a.sum()
        # one gradient step (true sub‑gradient of max‑conv)
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)   # non‑negative vector
        # momentum update of gradient
        grad_ma = momentum * grad_ma + (1 - momentum) * g
        # multiplicative update with decayed step size
        eta = eta0 * (decay ** it)
        mult = np.exp(-eta * grad_ma)
        a = a * mult
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        cur_score = _eval_fast(a, L_fft)
        it += 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start:.2f}s] exp‑grad it={it:,} score={cur_score:.6f}")

        # very occasional large jitter to escape deep plateaus
        if it % 20000 == 0:
            jitter = 0.05 * rng.random(n) * a
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            grad_ma.fill(0.0)   # reset momentum after big move

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam descent (≈150 s)
    # ------------------------------------------------------------------
    # Using a soft‑max approximation of the max with an increasing temperature β.
    a = best_seq.copy()
    cur_score = best_score
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_adam = 0
    betas = [2, 8, 32, 128, 512, 2048]   # increasingly sharp soft‑max
    for beta in betas:
        if time.time() >= deadline - 100.0:
            break
        stage_deadline = min(deadline - 100.0, time.time() + 25.0)
        while time.time() < stage_deadline:
            t_adam += 1
            conv = _conv_self_fft(a, L_fft)
            p = _softmax_weights(conv, beta)
            g = _grad_softmax(a, p, L_grad)      # smooth gradient
            # Adam update on the log‑domain (mirrored descent)
            m_adam = beta1 * m_adam + (1 - beta1) * g
            v_adam = beta2 * v_adam + (1 - beta2) * (g * g)
            m_hat = m_adam / (1 - beta1 ** t_adam)
            v_hat = v_adam / (1 - beta2 ** t_adam)
            step = 0.05 / (np.sqrt(v_hat) + eps_adam)   # element‑wise step
            log_a = np.log(a + 1e-30) - step * m_hat     # a stays positive
            a = np.exp(log_a)
            a /= a.sum()
            cand_score = _eval_fast(a, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                print(f"[{time.time() - start:.2f}s] Adam β={beta} it={t_adam:,} score={cand_score:.6f}")
            # tiny random nudges to avoid stagnation
            if t_adam % 5000 == 0:
                jitter = 0.005 * rng.random(n) * a
                a = np.maximum(a + jitter, 0.0)
                a /= a.sum()

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb (remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 5e-3   # up to 0.5 % shift
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_sc:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Search for a non‑negative step‑function (vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds in four stages:

1️⃣  Fast parametric sweep over geometric × power families (including a
    uniform mixture).  This quickly yields a decent starting shape.

2️⃣  Random‑restart exponentiated‑gradient descent on the *exact*
    max‑convolution objective.  The update is multiplicative,
    keeps the vector on the probability simplex and uses an exponential
    moving‑average of sub‑gradients to provide momentum.  Restarts and
    occasional jitter avoid stagnation.

3️⃣  Continuation with a smooth‑max (soft‑max) surrogate and an Adam‑style
    mirror‑descent on the log‑coordinates.  The temperature β is increased
    in stages, driving the surrogate closer to the true max.

4️⃣  A final short hill‑climb that moves tiny masses between random
    positions (simulated‑annealing style) to polish the last bits.

All phases respect the 1000 s budget, keep every coefficient ≥ 0,
and return the best vector found (as a plain Python list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """p_k = exp(beta*conv_k) / Σexp(...).  Numerically stable."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective
        Σ_k p_k·conv_k   with  conv = a*a .
    Using the identity   g_i = 2·Σ_j a_j·p_{i+j}.
    Implemented via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L)
    Pf = np.fft.rfft(p_rev, n=L)
    corr = np.fft.irfft(Af * Pf, n=L)
    # keep the middle n entries (indices n‑1 … 2n‑2) and reverse them
    start = n - 1
    g = corr[start:start + n][::-1]
    return 2.0 * g


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(conv) at index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]                 # includes the diagonal term correctly
    return g


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)                 # for numerical stability
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration α (scalar)."""
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return x / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises the given evaluation function.  Returns a plain Python list
    of floats (the vector is normalised to sum = 1).
    """

    start = time.time()
    deadline = start + budget_s - 0.1                 # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)                     # length for soft‑max grads

    # ------------------------------------------------------------------
    # 0️⃣  Baseline (uniform) and possible global starter
    # ------------------------------------------------------------------
    a_uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = a_uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"[{time.time() - start:.2f}s] uniform baseline score={best_score:.6f}")

    try:
        starter = np.asarray(height_sequence_1, dtype=float)      # may be undefined
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                cur = _eval_fast(starter, L_fft)
                if cur < best_score:
                    best_score = cur
                    best_seq = starter.copy()
                    print(f"[{time.time() - start:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Fast parametric sweep (≈30 s)
    # ------------------------------------------------------------------
    sweep_deadline = start + 30.0
    r_grid = np.logspace(np.log10(0.905), np.log10(0.9999), 30)
    p_grid = np.linspace(-4.0, 3.0, 35)

                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time() - start:.2f}s] sweep r={r:.5f} p={p:.3f} λ={lam:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random‑restart exponentiated‑gradient descent on the *true* max‑conv.
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current working vector
    grad_ma = np.zeros(n, dtype=np.float64) # exponential moving‑average of sub‑gradients
    it = 0
    eta0 = 1.0
    decay = 0.9995
    momentum = 0.9
    max_iters = 250_000                     # safety cap

    while time.time() < deadline - 200.0 and it < max_iters:
        # every few thousand steps we inject a small random perturbation
        if it % 5000 == 0:
            a = best_seq.copy()
            eps = 0.02 * rng.random(n) * a            # up to 2 % relative noise
            a = np.maximum(a + eps, 0.0)
            a /= a.sum()
            grad_ma.fill(0.0)

        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)          # exact sub‑gradient of the max

        # momentum on the sub‑gradient
        grad_ma = momentum * grad_ma + (1.0 - momentum) * g

        # exponentially decaying step size
        eta = eta0 * (decay ** it)

        # multiplicative (exponentiated) update and renormalisation
        mult = np.exp(-eta * grad_ma)
        a = a * mult
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= a_sum

        cur_score = _eval_fast(a, L_fft)
        it += 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start:.2f}s] exp‑grad it={it:,} score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max (soft‑max) Adam descent (≈150 s)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_adam = 0
    betas = [2, 8, 32, 128, 512, 2048]     # increasing temperature

    for beta in betas:
        if time.time() >= deadline - 100.0:
            break
        stage_deadline = min(deadline - 100.0, time.time() + 25.0)
        while time.time() < stage_deadline:
            t_adam += 1
            conv = _conv_self_fft(a, L_fft)
            p = _softmax_weights(conv, beta)
            g = _grad_softmax(a, p, L_grad)   # smooth gradient

            # Adam on the *log* of a (mirrored descent)
            m_adam = beta1 * m_adam + (1.0 - beta1) * g
            v_adam = beta2 * v_adam + (1.0 - beta2) * (g * g)
            m_hat = m_adam / (1.0 - beta1 ** t_adam)
            v_hat = v_adam / (1.0 - beta2 ** t_adam)

            step = 0.05 / (np.sqrt(v_hat) + eps_adam)   # element‑wise step
            log_a = np.log(a + 1e-30) - step * m_hat
            a = np.exp(log_a)
            a /= a.sum()

            cand_score = _eval_fast(a, L_fft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                print(f"[{time.time() - start:.2f}s] Adam β={beta} it={t_adam:,} score={cand_score:.6f}")

            # occasional tiny jitter
            if t_adam % 5000 == 0:
                jitter = 0.005 * rng.random(n) * a
                a = np.maximum(a + jitter, 0.0)
                a /= a.sum()

    # ------------------------------------------------------------------
    # 4️⃣  Final random hill‑climb (remaining seconds)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        src = rng.integers(n)
        if best_seq[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = best_seq[src] * rng.random() * 5e-3   # move up to 0.5 % of source mass
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start:.2f}s] final hill‑climb score={new_sc:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
