def project_simplex(v):
    # from https://gist.github.com/...
    v = np.maximum(v, 0)
    if v.sum() == 0:
        return np.full_like(v, 1.0/v.size)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def annealing_optimize(a, n, deadline):
    best_seq = a.copy()
    best_score = _objective_score(a, n)
    start = time.time()
    L_corr = _next_pow2(3 * n)
    beta = 0.5
    beta_max = 5000.0
    # Step size schedule corresponds to 0.2 / (beta+1)
    step_factor = 0.3
    iteration = 0
    while time.time() < deadline and beta < beta_max:
        # Compute transforms
        A_fft = np.fft.rfft(a, n=L_corr)  # shape L_corr//2+1
        conv = np.fft.irfft(A_fft * A_fft, n=L_corr)[:2 * n - 1]
        max_c = conv.max()
        # compute softmax weights p
        exp_vals = np.exp(beta * (conv - max_c))
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp
        # correlation
        P_fft = np.fft.rfft(p, n=L_corr)
        corr_full = np.fft.irfft(A_fft * np.conj(P_fft), n=L_corr)[:3 * n - 2]
        center = len(p) - 1  # 2n - 2
        g1 = corr_full[center:center + n]
        g2 = corr_full[center - n + 1:center + 1][::-1]
        grad = 4.0 * n * (g1 + g2)  # gradient of smooth objective
        # Update with exponentiated gradient
        eta = step_factor / (beta + 1.0)  # decaying
        a_new = a * np.exp(-eta * grad)
        a_new_sum = a_new.sum()
        if a_new_sum > 0:
            a_new /= a_new_sum
        else:
            a_new = np.full_like(a, 1.0 / n)
        a = a_new
        iteration += 1
        # Periodically compute exact max objective
        if iteration % 5 == 0:
            cur_score = _objective_score(a, n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] anneal β={beta:.2f} score={cur_score:.6f}")
        # Update beta gradually
        beta = beta * 1.03  # increase by 3%
    return best_seq, best_score

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _correlation_fft(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear correlation (a ⋆ p) of lengths n and 2n‑1.
    Result length = 3·n‑2, centred at index len(p)‑1.
    """
    Af = np.fft.rfft(a, n=L)
    Pf = np.fft.rfft(p, n=L)
    corr = np.fft.irfft(Af * np.conj(Pf), n=L)[: 3 * a.shape[0] - 2]
    return corr


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a sum‑normalised vector."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.maximum(v, 0.0)
    if v.sum() == 0:
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_sequence(n: int) -> np.ndarray:

        old = np.maximum(old, 0.0)
        if old.ndim == 1 and old.size > 0 and old.sum() > 0:

            # linear interpolation onto the new grid
            x_old = (np.arange(old.size, dtype=float) + 0.5) / old.size
            x_new = (np.arange(n, dtype=float) + 0.5) / n
            up = np.interp(x_new, x_old, old)
            up = np.maximum(up, 0.0)
            up /= up.sum()
            return up
    except Exception:

                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the max‑convolution using
    exponentiated‑gradient updates.  The smoothing parameter beta is
    increased geometrically so the problem converges to the original one.
    """
    a = a_start.copy()
    best_seq = a.copy()
    rng = np.random.default_rng()
    beta = 0.5
    beta_max = 5000.0
    beta_factor = 1.07          # geometric increase per iteration
    step_factor = 0.30          # controls η = step_factor/(β+1)
    it = 0
    last_improve = 0

    while time.time() < deadline and beta < beta_max:
        it += 1

        # ---- FFT of the current sequence ---------------------------------
        A = np.fft.rfft(a, n=L_corr)

        # ---- self‑convolution (needed for the soft‑max) -------------------
        conv = np.fft.irfft(A * A, n=L_corr)[: 2 * n - 1]
        max_c = conv.max()

        # ---- soft‑max weights over the 2n‑1 convolution entries ------------
        exp_vals = np.exp(beta * (conv - max_c))        # safe, max entry → 1
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp                         # length 2n‑1, sums to 1

        # ---- correlation a ⋆ p (gives the smoothed gradient) -------------
        P = np.fft.rfft(p, n=L_corr)
        corr = np.fft.irfft(A * np.conj(P), n=L_corr)[: 3 * n - 2]

        center = len(p) - 1                # = 2·n‑2
        g1 = corr[center:center + n]       # Σ_j a_j·p_{j+k}
        g2 = corr[center - n + 1:center + 1][::-1]   # Σ_j a_j·p_{j‑k}
        grad = 4.0 * n * (g1 + g2)        # gradient of 2·n·F_β

        # ---- exponentiated‑gradient update (keeps us on the simplex) -----
        eta = step_factor / (beta + 1.0)
        arg = -eta * grad
        arg = np.clip(arg, -50.0, 50.0)   # avoid overflow
        a = a * np.exp(arg)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # ---- occasional exact evaluation ----------------------------------
        if it % 5 == 0:
            conv_exact = np.fft.irfft(np.fft.rfft(a, n=L_fft) ** 2,
                                      n=L_fft)[: 2 * n - 1]
            cur_score = float(2.0 * n * conv_exact.max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = it
                print(f"[{time.time() - start_time:.2f}s] anneal β={beta:.2f} score={cur_score:.6f}")

        # ---- geometric increase of beta -----------------------------------
        beta = min(beta_max, beta * beta_factor)

        # ---- stochastic perturbation if stuck for a long while ----------
        if it - last_improve > 5000:
            perturb = rng.normal(scale=0.006, size=n) * a
            a = a + perturb
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = it

    return best_seq, best_score


# ------------------------------------------------------------
# 3️⃣  Small targeted hill‑climbing (simulated‑annealing style)
# ------------------------------------------------------------
def _hill_climb_opt(a_start: np.ndarray,
                    n: int,
                    L_fft: int,
                    deadline: float,
                    start_time: float,
                    best_score: float) -> tuple[np.ndarray, float]:
    """Transfer a tiny fraction of mass from the biggest entry to the smallest."""
    a = a_start.copy()
    best_seq = a.copy()
    cur_score = best_score
    rng = np.random.default_rng()
    while time.time() < deadline:
        i = np.argmax(a)                      # donor
        j = np.argmin(a)                      # receiver
        if a[i] <= 1e-14:
            i = rng.integers(n)               # fallback random donor
        if i == j:
            continue
        eps = min(0.02 * a[i], 1e-5)           # tiny transfer
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < -1e-12:
            continue
        # exact evaluation
        conv = np.fft.irfft(np.fft.rfft(a_new, n=L_fft) ** 2,
                            n=L_fft)[: 2 * n - 1]
        new_score = float(2.0 * n * conv.max())
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb → {new_score:.6f}")
        else:
            # annealed uphill move
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score
    return best_seq, best_score


# ------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section)
# ------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq /= seq.sum()
        conv = np.fft.irfft(np.fft.rfft(seq, n=_next_pow2(2 * n - 1)) ** 2,
                            n=_next_pow2(2 * n - 1))[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq /= best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] mix λ={lam1:.4f} → {sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq /= best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] mix λ={lam2:.4f} → {sc2:.6f}")
        if hi - lo < 1e-5:
            break
    return best_seq, best_score


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and finally
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the runner
    deadline = start_time + budget_s - 0.25

    # --------------------------------------------------------
    # problem size – a larger n gives a finer discretisation
    # --------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)          # for exact max‑convolution
    L_corr = _next_pow2(3 * n)             # for correlation (gradient)

    # --------------------------------------------------------
    # 0️⃣ initialise
    # --------------------------------------------------------
    a = _initial_sequence(n)
    best_seq = a.copy()
    best_score = _objective_score(a, n, L_fft)
    print(f"[0.00s] Initial baseline score={best_score:.6f}")

    # --------------------------------------------------------
    # 1️⃣ short random search (≈5 % of the budget)
    # --------------------------------------------------------
    phase1_deadline = start_time + 0.05 * (deadline - start_time)
    if time.time() < phase1_deadline:
        a, best_score = _broad_random_search(a, best_score, n, L_fft,
                                             phase1_deadline, start_time)
        best_seq = a.copy()

    # --------------------------------------------------------
    # 2️⃣ deterministic annealing (≈75 % of the budget)
    # --------------------------------------------------------
    phase2_deadline = start_time + 0.80 * (deadline - start_time)
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing")
        best_seq, best_score = _annealing_optimize(best_seq, n, L_fft,
                                                   L_corr, phase2_deadline,
                                                   start_time, best_score)
        print(f"[{time.time() - start_time:.2f}s] Annealing finished – best ≈ {best_score:.6f}")

    # --------------------------------------------------------
    # 3️⃣ hill‑climbing fine‑tuning (remaining time)
    # --------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb_opt(best_seq, n, L_fft,
                                               deadline, start_time, best_score)
        print(f"[{time.time() - start_time:.2f}s] Hill‑climb finished – best ≈ {best_score:.6f}")

    # --------------------------------------------------------
    # 4️⃣ mixing with the uniform distribution (tiny remainder)
    # --------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(best_seq, n,
                                                      deadline, start_time,
                                                      best_seq, best_score)

    # --------------------------------------------------------
    # final normalisation and return
    # --------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
