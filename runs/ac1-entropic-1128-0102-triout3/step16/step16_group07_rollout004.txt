"""
Propose a non‑negative step function (probability vector) that
minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The search proceeds in two resolution levels:

1. **Low‑resolution exploration (n_small ≈ 256)**
   - Sweep several analytic families (geometric, power, beta, sine‑power,
     Hann / Hamming / Blackman windows, Kaiser, stretched‑exponential,
     logistic).
   - Massive random Dirichlet sampling.
   - Adam optimisation of a smooth‑max surrogate (many more iterations
     than the reference code).
   - Projected sub‑gradient descent on the exact max‑convolution.

   The cheap size permits very many trials, allowing us to locate a
   promising shape that the previous 4096‑dimensional search missed.

2. **Upsampling & high‑resolution refinement (n_big = 4096)**
   - Linear interpolation of the best small‑scale vector to the target
     dimension (preserving the shape).
   - A short random‑Dirichlet warm‑start on the full size.
   - Projected sub‑gradient descent + a final hill‑climb / simulated‑annealing
     phase.

Both stages respect the overall 1000 s budget and keep a safety margin.
All operations use FFT for fast convolution; the objective used during
search is `score = 2·n·max(conv)` (identical to `evaluate_sequence`
when the vector is normalised to sum = 1).
"""

import time
import numpy as np
from math import log, exp

# ----------------------------------------------------------------------
# Low‑level utilities (global, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT (real inputs)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) for a normalised sequence (sum=1)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# 1️⃣  Low‑resolution families (n_small)
# ----------------------------------------------------------------------
def _sweep_families_small(n: int,
                          best_seq: np.ndarray,
                          best_score: float,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """Exhaustive sweep over a broad set of analytic families (n is small)."""
    idx = np.arange(n, dtype=np.float64)

    # Uniform baseline (already present)
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # Geometric
    eps_grid = np.logspace(-5, -1, 150)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 7):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")

    # Power law
    for p in np.linspace(-2.0, 2.0, 101):
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        if pow_seq.sum() == 0:
            continue
        pow_seq /= pow_seq.sum()
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # Beta‑type (a,b in (0.05,4])
    alphas = np.linspace(0.05, 4.0, 41)
    betas = np.linspace(0.05, 4.0, 41)
    log_idx = np.log(idx + 1.0)
    log_rev = np.log(np.maximum(n - idx, 1.0))
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * log_idx
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * log_rev
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if not np.isfinite(s) or s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.2f} β={b:.2f} score={sc:.6f}")

    # Sine‑power
    for p in np.linspace(0.1, 6.0, 61):
        if time.time() >= deadline:
            break
        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        seq = np.clip(seq, 0, None)
        if seq.sum() == 0:
            continue
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sine‑power p={p:.2f} score={sc:.6f}")

    # Hann, Hamming, Blackman windows
    theta = 2 * np.pi * idx / (n - 1)
    hann = 0.5 * (1 - np.cos(theta))
    for name, w in (("Hann", hann),
                    ("Hamming", 0.54 - 0.46 * np.cos(theta)),
                    ("Blackman", 0.42 - 0.5 * np.cos(theta) + 0.08 * np.cos(2 * theta))):
        seq = np.clip(w, 0, None)
        if seq.sum() == 0:
            continue
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] {name} score={sc:.6f}")

    # Kaiser window (different betas)
    for beta in [0.0, 2.0, 5.0, 8.0, 12.0]:
        if time.time() >= deadline:
            break
        seq = np.kaiser(n, beta)
        seq = np.clip(seq, 0, None)
        if seq.sum() == 0:
            continue
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Kaiser β={beta:.1f} score={sc:.6f}")

    # Stretched exponential
    scales = np.linspace(0.25, 1.0, 9)
    p_vals = np.linspace(0.2, 5.0, 31)
    for scale in scales:
        if time.time() >= deadline:
            break
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            if seq.sum() == 0:
                continue
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} scale={scale:.2f} score={sc:.6f}")

    # Logistic family
    mu_factors = np.linspace(0.2, 0.8, 13)
    s_factors = np.linspace(0.01, 0.12, 9)
    for mu_f in mu_factors:
        if time.time() >= deadline:
            break
        mu = mu_f * n
        for s_f in s_factors:
            if time.time() >= deadline:
                break
            s_val = s_f * n
            if s_val < 1e-9:
                continue
            seq = 1.0 / (1.0 + np.exp((idx - mu) / s_val))
            seq = np.clip(seq, 1e-12, None)
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Logistic mu={mu_f:.2f} s={s_f:.2f} score={sc:.6f}")

    return best_seq, best_score

# 2️⃣  Random Dirichlet sampling (small n)
# ----------------------------------------------------------------------
def _random_dirichlet_small(n: int,
                           best_seq: np.ndarray,
                           best_score: float,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 8000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # draw concentration from a log‑uniform interval
        log_c = rng.uniform(-3.0, 3.0)   # c in ~[0.05, 20]
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam optimisation (smooth‑max surrogate) – small n
# ----------------------------------------------------------------------
def _adam_optimize_small(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate for a small‑n vector."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07          # a bit larger than before (small n is cheap)
    tau = 5.0          # start with a gentle soft‑max
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # soft‑max (smooth‑max) over convolution values
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max
        a_rev = a[::-1]
        # convolution of w with reversed a (size sufficient for linear conv)
        L_tmp = _next_pow2(3 * n - 2)
        conv_w = np.fft.irfft(np.fft.rfft(w, n=L_tmp) *
                              np.fft.rfft(a_rev, n=L_tmp),
                              n=L_tmp)[:n]
        grad = 2.0 * conv_w

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        # increase the smoothness slowly
        if it % 800 == 0:
            tau = min(tau * 1.3, 300.0)

    return best_seq, best_score

# 4️⃣  Projected sub‑gradient descent (exact max) – small n
# ----------------------------------------------------------------------
def _pgd_descent_small(a0: np.ndarray,
                       start_time: float,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution (small n)."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.7
    it = 0
    no_improve = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD small it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad

        eta_t = eta / (1.0 + it * 1e-4)

        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)
        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-13:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 400 == 0:
                eta *= 0.992
                # Jitter to escape flat regions
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.0007, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                a = a_jit
                cur_score = _objective_score(a)

        if eta < 1e-9:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Upsampling helper
# ----------------------------------------------------------------------
def _upsample(seq_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from small to big size, followed by simplex projection."""
    xs = np.linspace(0.0, 1.0, seq_small.size)
    xb = np.linspace(0.0, 1.0, n_big)
    seq_big = np.interp(xb, xs, seq_small)
    seq_big = np.maximum(seq_big, 0.0)
    seq_big = _project_simplex(seq_big)
    return seq_big


# ----------------------------------------------------------------------
# 6️⃣  Large‑scale PGD + hill‑climb (full n)
# ----------------------------------------------------------------------
def _pgd_descent_large(a0: np.ndarray,
                       start_time: float,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent for the full‑size vector."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.6
    it = 0
    no_improve = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD large it={it:,} score={cur_score:.6f}")

        subgrad = _subgradient_maxconv(a, conv)
        grad = 2.0 * n * subgrad

        eta_t = eta / (1.0 + it * 5e-5)

        a_candidate = a - eta_t * grad
        a_candidate = _project_simplex(a_candidate)
        new_score = _objective_score(a_candidate)

        if new_score < cur_score - 1e-13:
            a = a_candidate
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 600 == 0:
                eta *= 0.995
                # occasional random jitter
                rng = np.random.default_rng()
                jitter = rng.normal(scale=0.0003, size=n)
                a_jit = a + jitter
                a_jit = np.maximum(a_jit, 0.0)
                a_jit = _project_simplex(a_jit)
                a = a_jit
                cur_score = _objective_score(a)

        if eta < 5e-9:
            break

    return best_seq, best_score


def _hill_climb_large(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Final hill‑climb / simulated‑annealing on the full size."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = _objective_score(a)
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_time = max(0.0, deadline - time.time())
    if total_time <= 0:
        return best_seq, best_score

    T0 = 0.015
    decay = (1e-6 / T0) ** (1.0 / total_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.003 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-13:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine runs a cheap low‑resolution search,
    up‑samples the best candidate, and then refines it at full resolution.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3    # small safety margin

    # --------------------------------------------------------------
    # Problem sizes
    # --------------------------------------------------------------
    n_big = 4096               # target size
    n_small = 256              # cheap resolution for exploration

    # --------------------------------------------------------------
    # Initial uniform baselines
    # --------------------------------------------------------------
    uniform_big = np.full(n_big, 1.0 / n_big)
    best_seq_big = uniform_big.copy()
    best_score_big = _objective_score(best_seq_big)
    print(f"[{time.time() - start_time:.2f}s] Baseline (n={n_big}) score={best_score_big:.6f}")

    uniform_small = np.full(n_small, 1.0 / n_small)
    best_seq_small = uniform_small.copy()
    best_score_small = _objective_score(best_seq_small)
    print(f"[{time.time() - start_time:.2f}s] Baseline (n={n_small}) score={best_score_small:.6f}")

    # --------------------------------------------------------------
    # Reserve ~300 s for the low‑resolution stage, rest for high‑res.
    # --------------------------------------------------------------
    small_stage_deadline = min(deadline - 350.0, start_time + 300.0)
    if time.time() < small_stage_deadline:
        # 1️⃣  Parametric families (small n)
        best_seq_small, best_score_small = _sweep_families_small(
            n_small, best_seq_small, best_score_small,
            small_stage_deadline, start_time)

        # 2️⃣  Random Dirichlet trials (small n)
        if time.time() < small_stage_deadline:
            best_seq_small, best_score_small = _random_dirichlet_small(
                n_small, best_seq_small, best_score_small,
                small_stage_deadline, start_time)

        # 3️⃣  Adam optimisation (small n)
        if time.time() < small_stage_deadline - 30.0:
            print(f"[{time.time() - start_time:.2f}s] Starting Adam (small n)")
            best_seq_small, best_score_small = _adam_optimize_small(
                best_seq_small, start_time,
                small_stage_deadline - 30.0,
                best_seq_small, best_score_small)

        # 4️⃣  PGD descent (small n)
        if time.time() < small_stage_deadline - 10.0:
            print(f"[{time.time() - start_time:.2f}s] Starting PGD (small n)")
            best_seq_small, best_score_small = _pgd_descent_small(
                best_seq_small, start_time,
                small_stage_deadline - 10.0,
                best_seq_small, best_score_small)

    # --------------------------------------------------------------
    # Upsample the small‑scale candidate to full size
    # --------------------------------------------------------------
    best_seq_big = _upsample(best_seq_small, n_big)
    best_score_big = _objective_score(best_seq_big)
    print(f"[{time.time() - start_time:.2f}s] After up‑sampling score≈{best_score_big:.6f}")

    # --------------------------------------------------------------
    # Optional: a few large‑scale random Dirichlet warm‑starts
    # --------------------------------------------------------------
    if time.time() < deadline - 200.0:
        rng = np.random.default_rng()
        for _ in range(1200):
            if time.time() >= deadline - 200.0:
                break
            c = np.exp(rng.uniform(-2.0, 2.0))
            cand = rng.gamma(shape=c, scale=1.0, size=n_big)
            if cand.sum() == 0:
                continue
            cand /= cand.sum()
            sc = _objective_score(cand)
            if sc < best_score_big:
                best_score_big, best_seq_big = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Large Dirichlet c={c:.3f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Projected sub‑gradient descent on the full‑size vector
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print(f"[{time.time() - start_time:.2f}s] Starting PGD (full size)")
        best_seq_big, best_score_big = _pgd_descent_large(
            best_seq_big, start_time,
            deadline - 80.0,
            best_seq_big, best_score_big)

    # --------------------------------------------------------------
    # 6️⃣  Final hill‑climb / simulated‑annealing (full size)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq_big, best_score_big = _hill_climb_large(
            best_seq_big, start_time,
            deadline,
            best_seq_big, best_score_big)

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq_big = np.maximum(best_seq_big, 0.0)
    total = best_seq_big.sum()
    if total > 0:
        best_seq_big = best_seq_big / total
    else:
        best_seq_big = np.full(n_big, 1.0 / n_big)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score_big:.6f}")
    return [float(x) for x in best_seq_big]
