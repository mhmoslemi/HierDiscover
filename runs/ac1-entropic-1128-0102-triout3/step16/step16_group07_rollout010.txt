"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm combines several ideas that were not used in the previous
submission:

* a cheap random Dirichlet warm‑start,
* an exponentiated‑gradient optimiser that works on a smooth‑max
  surrogate of the max‑convolution (temperature is annealed),
* a cutting‑plane linear‑program that finds, for the current sequence f,
  a companion sequence g maximising ‖g‖₁ while keeping f∗g ≤ max(f∗f);
  the two sequences are mixed and the process is iterated,
* a fast simulated‑annealing hill‑climb for the final polishing.

All stages respect the global 1000 s budget and continuously keep the
best‐found sequence.  The functions are written with only top‑level
definitions, no closures or lambdas, and use only ``numpy`` and
``scipy.optimize.linprog`` (available in the default environment).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _conv_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    L = a.shape[0] + b.shape[0] - 1
    L_fft = _next_pow2(L)
    Af = np.fft.rfft(a, n=L_fft)
    Bf = np.fft.rfft(b, n=L_fft)
    conv = np.fft.irfft(Af * Bf, n=L_fft)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Score 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Exponentiated‑gradient optimisation of a smooth‑max surrogate
# ----------------------------------------------------------------------
def _expgrad_optimize(a0: np.ndarray, deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """EG on the soft‑max of the autocorrelation; temperature anneals."""
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    # hyper‑parameters (tuned empirically)
    tau = 5.0          # start temperature of the soft‑max
    tau_max = 800.0
    tau_factor = 1.12
    eta = 0.05         # step size for the exponential update
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGrad it={it:,} score={cur_score:.6f}")

        # ----- smooth max weights -----
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))          # values ≤ 1
        w = w_raw / w_raw.sum()                       # probability vector, length 2n‑1

        # ----- gradient of the smooth max -----
        a_rev = a[::-1]
        grad_tmp = _conv_general(w, a_rev)[:n]        # length n
        grad = 2.0 * grad_tmp

        # ----- exponential‑gradient update -----
        a = a * np.exp(-eta * grad)
        a = _project_simplex(a)

        # anneal temperature slowly
        if it % 800 == 0:
            tau = min(tau * tau_factor, tau_max)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Cutting‑plane LP refinement (f‑g alternating scheme)
# ----------------------------------------------------------------------
def _lp_alternating_refinement(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """
    For the current distribution f (named ``a``) we solve
        max   1ᵀ g
        s.t.  f * g ≤ M      (M = max(f*f))
              g ≥ 0
    by a cutting‑plane method (starting from the active max indices).
    The feasible g is turned into a probability vector and mixed
    with f using a small convex weight; the process is repeated.
    """
    a = a0.copy()
    n = a.shape[0]

    # maximal number of outer mixing steps
    for outer_it in range(12):
        if time.time() >= deadline:
            break

        conv_ff = _conv_self(a)
        M = conv_ff.max()

        # ---- initialise active constraint set (indices attaining the max) ----
        active = np.where(conv_ff >= M - 1e-12)[0]
        S = set(active.tolist())

        # ---- cutting‑plane loop (solve LP, add violated constraints) ----
        lp_success = False
        for cp_it in range(20):
            # build dense matrix A (|S| × n)
            rows = []
            for k in S:
                lo = max(0, k - (n - 1))
                hi = min(k, n - 1)
                row = np.zeros(n, dtype=np.float64)
                idx = np.arange(lo, hi + 1, dtype=int)
                row[lo:hi + 1] = a[k - idx]          # coefficient f_{k‑j}
                rows.append(row)
            if not rows:
                break
            A = np.vstack(rows)
            b_vec = np.full(A.shape[0], M, dtype=np.float64)

            # solve LP: maximise 1ᵀg  ⇔  minimise –1ᵀg
            res = linprog(-np.ones(n), A_ub=A, b_ub=b_vec,
                          bounds=[(0, None)] * n,
                          method='highs', options={'presolve': True})

            if not res.success:
                print(f"[{time.time() - start_time:.2f}s] LP failed (it={outer_it} cp={cp_it}): {res.message}")
                break
            g0 = res.x
            lp_success = True

            # check full constraint set
            conv_fg = _conv_general(a, g0)
            viol = np.where(conv_fg > M + 1e-12)[0]
            if viol.size == 0:
                # feasible solution found
                break
            # add a few most violating indices to the active set
            for k in viol[:5]:
                S.add(k)

        if not lp_success:
            # fall back to uniform if LP failed
            g = np.full_like(a, 1.0 / n)
        else:
            # normalise g to a probability vector
            g_sum = g0.sum()
            if g_sum > 0:
                g = g0 / g_sum
            else:
                g = np.full_like(a, 1.0 / n)

        # ---- Mix f and the freshly computed g ----
        t_mix = 0.18                # a modest step towards g
        a = (1.0 - t_mix) * a + t_mix * g
        a = _project_simplex(a)

        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP‑mix it={outer_it} score={cur_score:.6f}")

        # early exit if we have already crossed the target
        if cur_score < 1.502:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Aggressive simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Simple SA that repeatedly transfers a random fraction of mass."""
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)
    rng = np.random.default_rng()

    # temperature schedule (fast cooling)
    T0 = 0.02
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a random fraction (up to ~1 % of src mass)
        frac = rng.random() * 0.009 + 0.001
        delta = a[src] * frac

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        a_prop = _project_simplex(a_prop)   # keep it a probability vector
        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        # Metropolis acceptance
        T = T0 * (0.5 ** ((time.time() - start_time) / (deadline - start_time)))
        if Δ < -1e-12 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA best score={best_score:.6f}")

    return best_seq, best_score
