def smoothed_gradient(a, alpha, N):
    n = a.shape[0]
    A = np.fft.rfft(a, n=N)
    conv_full = np.fft.irfft(A * A, n=N)[:2*n-1]

def _smoothed_gradient(a: np.ndarray, alpha: float, N: int) -> np.ndarray:
    n = a.shape[0]
    A = np.fft.rfft(a, n=N)
    # self-convolution for smoothing
    conv_full = np.fft.irfft(A * A, n=N)[:2*n-1]
    conv_max = conv_full.max()
    # compute softmax weights
    # stable: subtract conv_max for exponent
    exp_term = np.exp(alpha * (conv_full - conv_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # reversed weights
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=N)
    # weighted M*a via convolution in freq domain
    conv_weighted_full = np.fft.irfft(A * W_rev, n=N)
    # extract grad_raw = 2 * M a for indices 0..n-1
    idx = (2 * n - 2) - np.arange(n)
    grad_raw = 2.0 * conv_weighted_full[idx]
    return grad_raw
```

"""
Propose a step–function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a projected gradient descent on a *smooth* approximation of
the max‑convolution (log‑sum‑exp).  The smoothing parameter α is increased
gradually so the optimisation first sees a smooth landscape and later
behaves like a sub‑gradient method for the true max.  After each gradient step
the iterate is projected onto the probability simplex (∑a_i = 1, a_i≥0).
Occasional random jitter and a short final hill‑climb are used to escape
plateaus.  The whole search respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _smoothed_gradient(a: np.ndarray, alpha: float, N: int) -> np.ndarray:
    """
    Gradient of the *smoothed* objective 2·n·phi,
    where phi = (1/α)·log∑exp(α·conv).
    Returns 2·M·a (M_ij = w_{i+j}) – i.e. the raw gradient before the
    outer factor 2·n.
    """
    n = a.shape[0]

    # FFT of the current sequence (length N ≥ 3n‑2)
    A = np.fft.rfft(a, n=N)

    # ----- self‑convolution (full) ------------------------------------
    conv_full = np.fft.irfft(A * A, n=N)[: 2 * n - 1]

    # ----- soft‑max weights -------------------------------------------
    conv_max = conv_full.max()
    # stable exponentiation
    exp_term = np.exp(alpha * (conv_full - conv_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum                     # w_k ≥ 0, ∑ w_k = 1

    # ----- gradient = 2·M·a -------------------------------------------
    # we need the Toeplitz matrix M_{ij}=w_{i+j}.  This is obtained by a
    # linear convolution of a with the reversed weight vector.
    w_rev = w[::-1]
    W_rev = np.fft.rfft(w_rev, n=N)

    # linear convolution a * w_rev   (length 3n‑2, we keep the whole vector)
    conv_weighted_full = np.fft.irfft(A * W_rev, n=N)

    # the required entries are conv_weighted[i] = (M a)[i] = Σ_j w_{i+j} a_j
    idx = (2 * n - 2) - np.arange(n)        # mapping described in the analysis
    grad_raw = 2.0 * conv_weighted_full[idx]  # 2·M·a
    return grad_raw


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising the evaluation function.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    N = _next_pow2(3 * n - 2)   # length sufficient for all FFT‑based convolutions

    # --------------------------------------------------------------
    # Initialise – use the global starter if present, otherwise uniform.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                a = starter / starter.sum()
            else:
                a = np.full(n, 1.0 / n)
        else:
            a = np.full(n, 1.0 / n)
    except Exception:
        a = np.full(n, 1.0 / n)

    # If the starter has the wrong length, resize by truncating / padding.
    if a.size != n:
        if a.size > n:
            a = a[:n]
        else:
            a = np.concatenate([a, np.zeros(n - a.size)])
        a = a / a.sum()

    a = _project_simplex(a)                     # guarantee exact simplex
    best_seq = a.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    step = 0.05                     # initial L2 step length
    min_step = 1e-9
    max_step = 2.0
    alpha_start = 0.5
    alpha_end = 300.0               # final amount of smoothing (≈ hard max)
    no_improve = 0
    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop – projected gradient descent on the smooth
    # approximation, with a line‑search on the *true* objective.
    # --------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linearly raise the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / (budget_s - 0.25))

        # gradient of the smoothed objective (2·M·a)
        grad_raw = _smoothed_gradient(a, alpha, N)
        # full gradient of 2·n·phi
        grad = 2.0 * n * grad_raw
        norm = np.linalg.norm(grad) + 1e-16
        direction = grad / norm

        cur_score = _objective_score(a)

        improved = False
        # simple back‑tracking line‑search on the *exact* objective
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step_sz = step * factor
            a_candidate = a - step_sz * direction
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)
            cand_score = _objective_score(a_candidate)

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = a_candidate.copy()
                a = a_candidate
                step = min(step * 1.02, max_step)
                improved = True
                no_improve = 0
                print(f"[{elapsed:.2f}s] it={iteration} score={cand_score:.6f} alpha={alpha:.1f}")
                break
            elif cand_score < cur_score:
                # accept a non‑best improvement to keep moving
                a = a_candidate
                step = min(step * 1.01, max_step)
                improved = True
                break

        if not improved:
            # no progress – shrink step and inject a tiny jitter
            step = max(step * 0.5, min_step)
            no_improve += 1
            if no_improve % 3000 == 0:
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps = a[i1] * (rng.random() * 1e-5 + 1e-9)
                if eps > 0:
                    a[i1] -= eps
                    a[i2] += eps
                    a = _project_simplex(np.maximum(a, 0.0))
                print(f"[{elapsed:.2f}s] jitter after {no_improve} stagnant steps")

        iteration += 1

    # --------------------------------------------------------------
    # Final quick hill‑climb (tiny mass transfers) for the last few seconds.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:   # run only if we have a noticeable chunk left
        hill_deadline = time.time() + time_left
        a_hc = best_seq.copy()
        while time.time() < hill_deadline:
            i, j = rng.choice(n, size=2, replace=False)
            if a_hc[i] < 1e-15:
                continue
            eps = a_hc[i] * (rng.random() * 0.01 + 1e-9)   # up to ≈1 % of source mass
            a_prop = a_hc.copy()
            a_prop[i] -= eps
            a_prop[j] += eps
            a_prop = _project_simplex(a_prop)
            prop_score = _objective_score(a_prop)
            if prop_score < best_score:
                best_score = prop_score
                best_seq = a_prop.copy()
                a_hc = a_prop
                print(f"[{time.time() - start_time:.2f}s] hill‑climb improved to {prop_score:.6f}")

    # --------------------------------------------------------------
    # Return a plain Python list (ensure non‑negativity and normalisation)
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
