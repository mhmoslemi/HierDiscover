def propose_candidate(seed=0,budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1
    n = 4096 # maybe 4096; can be adjust if needed
    # precompute FFT size
    L_fft = 1 << ((2*n-2).bit_length())
    # Helper compute score
    def compute_score(a):
        A = np.fft.rfft(a, n=L_fft)
        conv = np.fft.irfft(A * A, n=L_fft)[:2*n-1]
        max_b = conv.max()
        sum_a = a.sum()
        if sum_a < 0.01:
            return np.inf
        return 2.0 * n * max_b / (sum_a * sum_a)
    # Warm start
    best_a = np.full(n, 1.0 / n)
    best_score = compute_score(best_a)
    print(f"[{time.time()-start:.2f}s] Uniform score {best_score}")
    # try global starter if exists:
    try:

        s = np.asarray(height_sequence_1, dtype=float)
        if s.ndim==1 and s.size>0:

            if s.sum()>0:
                s = s / s.sum()
                sc = compute_score(s)
                if sc < best_score:

                    best_a = s.copy()

"""
Optimise a non‑negative step‑function (a probability vector) for the
objective

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The evaluator is scale‑invariant, so we keep the vector on the simplex.
The search consists of three timed phases:

1️⃣ Warm‑start – a few Dirichlet draws (log‑uniform concentration) and the
   optional global starter ``height_sequence_1`` give a decent initial point.

2️⃣ Targeted hill‑climb – repeatedly identify the index *k* where the
   convolution attains its maximum.  The contribution of each coefficient
   to this peak is a_i·a_{k‑i}; we move a small amount of mass from the
   largest contributor towards an index that does **not** participate in the
   peak (typically a zero‑contribution index).  This directly attacks the
   L∞‑norm of the convolution and quickly lowers the score.

3️⃣ Simulated‑annealing fine‑tuning – after the peak‑focused phase we run a
   classic SA on random pairwise transfers (a_i → a_j) with an exponentially
   decaying temperature.  Occasionally the current vector is mixed with the
   uniform distribution to escape shallow basins.

All convolutions are evaluated with an FFT (O(n log n)) for speed; the final
vector is renormalised to sum = 1 before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas – top level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_max_fft(a: np.ndarray, L_fft: int) -> tuple[float, np.ndarray]:
    """
    Return (max(conv(a,a)), full convolution) using real FFT.
    The convolution length is ``2·len(a)-1``; values outside are discarded.
    """
    A = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(A * A, n=L_fft)[: 2 * a.shape[0] - 1]
    return float(conv_full.max()), conv_full


def _score_from_conv(max_conv: float, sum_a: float, n: int) -> float:
    """Exact score used by the evaluator."""
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def _compute_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Fast score evaluation (identical to evaluate_sequence)."""
    max_c, _ = _conv_max_fft(a, L_fft)
    return _score_from_conv(max_c, a.sum(), n)


# ----------------------------------------------------------------------
# Phase 1 – warm‑start with Dirichlet draws (and optional global starter)
# ----------------------------------------------------------------------
def _warm_start(a: np.ndarray,
                best_score: float,
                n: int,
                L_fft: int,
                deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        # concentration drawn log‑uniformly in [0.1, 20]
        logc = rng.uniform(np.log(0.1), np.log(20.0))
        c = np.exp(logc)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        cand = sample / sample.sum()
        sc = _compute_score(cand, n, L_fft)
        if sc < best_score:
            best_score, a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return a, best_score


# ----------------------------------------------------------------------
# Phase 2 – targeted peak reduction
# ----------------------------------------------------------------------
def _targeted_hillclimb(a: np.ndarray,
                        best_a: np.ndarray,
                        best_score: float,
                        n: int,
                        L_fft: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly attacks the index where the convolution is maximal.
    Mass is transferred from the largest local contributor to a zero‑contributor.
    """
    rng = np.random.default_rng()
    step_frac = 0.10               # initial fraction of a_i that may be moved
    while time.time() < deadline:
        max_c, conv = _conv_max_fft(a, L_fft)
        cur_score = _score_from_conv(max_c, a.sum(), n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] New best (peak‑climb) {cur_score:.6f}")

        # index of the current maximum of the convolution
        k = int(np.argmax(conv))

        # contributions of each coefficient to this peak:
        # w[i] = a[i] * a[k‑i]   (zero if k‑i out of range)
        i_idx = np.arange(n)
        j_idx = k - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        w = np.zeros(n)
        w[mask] = a[i_idx[mask]] * a[j_idx[mask]]

        # pick the index with the largest contribution
        i_max = int(np.argmax(w))

        # pick a destination index that does *not* contribute to the peak
        zero_mask = (w == 0.0)
        if zero_mask.any():
            j_dest = int(rng.choice(np.flatnonzero(zero_mask)))
        else:
            # fallback – any index different from i_max
            while True:
                j_dest = int(rng.integers(0, n))
                if j_dest != i_max:
                    break

        # move a small amount of probability mass
        delta = rng.uniform(0.0, min(a[i_max] * step_frac, 0.005))
        if delta <= 0.0:
            continue
        a[i_max] -= delta
        a[j_dest] += delta

        # slowly shrink the step size so that we finish with a fine adjustment
        step_frac = max(0.001, step_frac * 0.9999)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         best_a: np.ndarray,
                         best_score: float,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Classic SA on random pairwise transfers.  Temperature decays exponentially,
    and the maximal transferable fraction of a[i] shrinks together with T.
    """
    rng = np.random.default_rng()
    cur_a = a.copy()
    cur_score = _compute_score(cur_a, n, L_fft)

    # temperature schedule – start roughly 1 % of the current score
    T0 = max(best_score * 0.01, 1e-4)
    T_final = 1e-8

    total = deadline - time.time()
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        T = T0 * ((T_final / T0) ** frac)

        # transferable fraction shrinks with the temperature
        max_frac = max(0.01, 0.10 * (1.0 - frac))

        # random pairwise transfer
        i, j = rng.choice(n, size=2, replace=False)
        if cur_a[i] <= 0.0:
            continue
        delta = rng.uniform(0.0, min(cur_a[i] * max_frac, 0.005))
        if delta == 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _compute_score(cand, n, L_fft)
        dE = cand_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score, best_a = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] SA improved to {cand_score:.6f}")

        # occasional mixing with the uniform distribution helps escape plateaus
        if rng.random() < 0.002:
            lam = rng.random()
            mixed = lam * cur_a + (1.0 - lam) * (np.full(n, 1.0 / n))
            mixed /= mixed.sum()
            mixed_score = _compute_score(mixed, n, L_fft)
            if mixed_score < best_score:
                best_score, best_a = mixed_score, mixed.copy()
                cur_a, cur_score = mixed.copy(), mixed_score
                print(f"[{time.time() - start_time:.2f}s] Mixed with uniform → {mixed_score:.6f}")

    return best_a, best_score

    best_a = a.copy()
    best_score = _compute_score(a, n, L_fft)
    print(f"[0.00s] Uniform baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) if the host provides it
    # ------------------------------------------------------------------
    try:
        global height_sequence_1
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _compute_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    a = starter.copy()
                    print(f"[0.00s] Global starter improves score to {sc:.6f}")
    except Exception:
        # missing global variable – simply ignore
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start via Dirichlet draws (≈30 s)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 30.0)
    if time.time() < warm_deadline:
        a, best_score = _warm_start(a, best_score, n, L_fft, warm_deadline)
        best_a = a.copy()

    # ------------------------------------------------------------------
    # 2️⃣  Targeted peak‑reduction hill‑climb (≈½ of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_deadline = min(deadline, time.time() + 0.5 * (deadline - time.time()))
        best_a, best_score = _targeted_hillclimb(best_a, best_a, best_score,
                                                 n, L_fft, phase2_deadline)

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing fine‑tuning (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = _simulated_annealing(best_a, best_a, best_score,
                                                  n, L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # convert to plain Python list as required by the harness
    return [float(v) for v in best_a]
