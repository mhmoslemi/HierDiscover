def _smooth_obj_and_grad(p, beta):
    conv = _conv_self(p) # length 2n-1
    max_conv = conv.max()
    # for numerical stability, subtract max
    exp_term = np.exp(beta * (conv - max_conv))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp  # length 2n-1
    # compute gradient: grad = 2 * n * correlation(w, p)
    p_rev = p[::-1]
    L = _next_pow2(len(w) + n - 1)
    # compute convolution via FFT
    W = np.fft.rfft(w, n=L)
    P_rev = np.fft.rfft(p_rev, n=L)
    conv_full = np.fft.irfft(W * P_rev, n=L)[:len(w) + n - 1]
    grad = 2.0 * n * conv_full[n-1 : n-1 + n]
    # compute smooth objective (maybe return but not needed)
    smooth_obj = (2.0 * n / beta) * (np.log(sum_exp) + beta * max_conv)  # because we shifted by max_conv
    return smooth_obj, grad

"""
Propose a candidate step‑function that (almost) minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The new search replaces the sub‑gradient PGD of the previous
implementation by a *smooth* (log‑sum‑exp) surrogate of the max‑norm and
uses mirror (exponentiated) descent on the simplex.  The surrogate
parameter β is annealed from a small value (smooth landscape) to a large
value (close to the true max) while the learning‑rate is adapted
automatically.  Random Dirichlet starts are used to diversify the
search, and the best vector found is always recorded using the exact
objective (``_objective_score``).  The whole routine respects the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (same as the previous version)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Exact objective assuming sum(seq)=1 → 2·n·max(conv)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(p: np.ndarray, beta: float):
    """
    Returns (smooth_obj, grad) where

        smooth_obj = (2·n/β)·log Σ_k exp(β·conv_k)

    and   grad = ∂ smooth_obj / ∂ p .
    """
    n = p.shape[0]

    # ---- convolution (exact) ----
    conv = _conv_self(p)                     # length 2·n‑1
    maxc = conv.max()                        # for numerical stability

    # ---- soft‑max weights w_k = exp(β·conv_k)/Σ exp ----
    # use shift by maxc to avoid overflow
    shifted = beta * (conv - maxc)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp                 # length 2·n‑1, sums to 1

    # ---- smooth objective (not really needed for the optimiser) ----
    # smooth = (2·n/β)*(np.log(sum_exp) + beta*maxc)
    # we keep it only for potential logging
    smooth = (2.0 * n / beta) * (np.log(sum_exp) + beta * maxc)

    # ---- gradient.
    #  ∂/∂p_j  (2·n Σ_k w_k conv_k) = 2·n Σ_k w_k (p_{k‑j}+p_{j‑k})
    #  = 2·n·corr(w, p)   (linear correlation, lag ≥0)
    # correlation can be obtained as a convolution of w with the reversed p.
    p_rev = p[::-1]                          # length n
    L = _next_pow2(len(w) + len(p_rev) - 1)
    W = np.fft.rfft(w, n=L)
    P = np.fft.rfft(p_rev, n=L)
    full = np.fft.irfft(W * P, n=L)[: len(w) + len(p_rev) - 1]
    # for lag ℓ (0 ≤ ℓ < n) the correlation is full[n‑1+ℓ]
    grad = 2.0 * n * full[n - 1 : n - 1 + n]   # shape (n,)

    return smooth, grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_exponential_descent(p0: np.ndarray,
                                deadline: float,
                                best_seq: np.ndarray,
                                best_score: float):
    """
    Mirror descent on the simplex using the smooth surrogate.
    β is linearly increased from β_start to β_end.
    """
    p = p0.copy()
    n = p.shape[0]

    # schedule parameters -------------------------------------------------
    beta_start = 5.0            # very smooth → almost quadratic
    beta_end   = 250.0          # close to true max‑norm
    total_time = max(deadline - time.time(), 1e-6)
    start = time.time()

    # learning‑rate (in log‑space) ---------------------------------------
    eta = 0.5                   # initial step size for log‑variables
    min_eta = 1e-6
    max_eta = 1.5

    it = 0
    while time.time() < deadline:
        it += 1
        # ----- β‑annealing ------------------------------------------------
        elapsed = time.time() - start
        frac = elapsed / total_time
        beta = beta_start + (beta_end - beta_start) * frac
        if beta > beta_end:
            beta = beta_end

        # ----- smooth objective & gradient --------------------------------
        smooth, grad = _smooth_obj_and_grad(p, beta)

        # ----- record exact score -----------------------------------------
        cur_score = _objective_score(p)
        if cur_score < best_score:
            best_score, best_seq = cur_score, p.copy()
            print(f"[{time.time() - start_time:.2f}s] EG it={it:,} β={beta:.1f} "
                  f"score={cur_score:.6f}")

        # ----- EG‑update (mirror descent with KL‑prox) --------------------
        # work in log‑domain; add a tiny epsilon to avoid log(0)
        logp = np.log(p + 1e-30)

        # normalise gradient to keep updates stable
        grad_norm = np.max(np.abs(grad)) + 1e-30
        step = eta * grad / grad_norm

        logp_new = logp - step
        p_new = np.exp(logp_new)
        p_new_sum = p_new.sum()
        if p_new_sum == 0 or not np.isfinite(p_new_sum):
            # fall back to a tiny random jitter
            eps = 1e-8
            p_new = np.clip(p + np.random.uniform(-eps, eps, size=n), 0, None)
            p_new /= p_new.sum()
            eta = max(min_eta, eta * 0.5)
            continue
        p_new /= p_new_sum

        new_score = _objective_score(p_new)

        # ----- accept/reject ------------------------------------------------
        if new_score < cur_score:
            p = p_new
            # if we improve we can be a bit more aggressive
            eta = min(max_eta, eta * 1.02)
        else:
            # reject – shrink learning rate
            eta = max(min_eta, eta * 0.5)
            # occasional random shake‑up to escape flat regions
            if eta < 5e-5:
                eps = 1e-6
                p = np.clip(p + np.random.uniform(-eps, eps, size=n), 0, None)
                p /= p.sum()
                eta = 0.5

        # safety: stop early if we already beat the target
        if best_score <= 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Random Dirichlet starter (used both for diversity and as a quick
#  sanity check before the main optimisation).
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 250          # modest number – the heavy optimiser will take
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration in [0.01, 5] (log‑uniform)
        logc = rng.uniform(-2.0, 0.7)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine consists of three phases:

    1. A few random Dirichlet seeds (to diversify the start).
    2. A smooth exponential‑gradient descent (log‑sum‑exp surrogate) with
       a β‑annealing schedule and adaptive learning‑rate.
    3. Return the best vector (normalised to sum = 1) as a plain list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same scale as the previous runner
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1) – may be supplied by the
    # evaluation harness.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – a handful of Dirichlet draws
    # ------------------------------------------------------------------
    phase_deadline = deadline - 250.0          # reserve time for EG phase
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – smooth exponential‑gradient descent (the core optimiser)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth EG descent")
        best_seq, best_score = _smooth_exponential_descent(
            best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] End of EG descent – best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – final best score ≈ {best_score:.6f}")
    # Convert to plain Python list of floats
    return [float(x) for x in best_seq]
