"""
Simulated‑annealing / targeted‑move local search.

The algorithm keeps a probability distribution `a` (non‑negative, sum = 1) and
maintains its self‑convolution `conv = a*a` incrementally.  At each step we
identify a shift where the convolution attains its maximum, pick a pair of
indices that contributes most to that shift and move a small fraction of mass
from the larger index to a random destination.  The move is accepted if the
objective `2·n·max(conv)` does not increase; otherwise it is accepted with a
temperature‑dependent probability (simulated annealing).  Periodically the
distribution is renormalised and the full convolution recomputed to avoid drift.
If no improvement is seen for many steps a random restart is performed.
The procedure runs for the allotted wall‑clock time (≈ 1000 s) and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (incremental updates)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Exact self‑convolution a*a (linear, not circular)."""
    return np.convolve(a, a)


def apply_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Move `eps` mass from `src` to `dst` (preserving the sum).
    The convolution `conv` (length 2·n‑1) is updated in‑place using the analytical
    formula:
        conv ← conv + 2·Δ_src·shift(a) + Δ_src²·δ_{2·src}
               + 2·Δ_dst·shift(a) + Δ_dst²·δ_{2·dst}
               + 2·Δ_src·Δ_dst·δ_{src+dst}
    where Δ_src = –eps and Δ_dst = +eps.
    The `shift(a)` term is realised by adding a slice of `a` to the appropriate
    segment of `conv`.  The update uses the values of `a` *before* they are
    modified.
    """
    n = a.shape[0]
    d_src = -eps
    d_dst = eps

    # contribution of the source index
    conv[src:src + n] += 2.0 * d_src * a
    # contribution of the destination index
    conv[dst:dst + n] += 2.0 * d_dst * a

    # square terms (impulses)
    conv[2 * src] += d_src * d_src
    conv[2 * dst] += d_dst * d_dst

    # cross term
    conv[src + dst] += 2.0 * d_src * d_dst

    # finally move the probability mass
    a[src] += d_src
    a[dst] += d_dst


def revert_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int, eps: float) -> None:
    """
    Revert a move performed by `apply_move`.  Swapping the roles of `src` and
    `dst` with the same `eps` exactly undoes the previous update.
    """
    apply_move(a, conv, dst, src, eps)


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step function minimising
        2·n·max(conv(a,a)) / (sum(a)²)

    The sum is fixed to 1, therefore the objective simplifies to
        2·n·max(conv(a,a)).

    The optimiser works on a dense vector of length `n` (default 4096).  It
    combines a targeted move heuristic with simulated‑annealing acceptance,
    periodic full recomputation of the convolution and random restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial distribution
    # ------------------------------------------------------------------
    n = 4096                     # length of the step function
    uniform = np.full(n, 1.0 / n)

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0:
                a = starter / starter.sum()
                if a.shape[0] != n:
                    # adapt the size: truncate or pad with zeros
                    if a.shape[0] > n:
                        a = a[:n]
                    else:
                        a = np.pad(a, (0, n - a.shape[0]), constant_values=0.0)
                    a /= a.sum()
            else:
                a = uniform.copy()
        else:
            a = uniform.copy()
    except Exception:
        a = uniform.copy()

    a = np.maximum(a, 0.0)
    a /= a.sum()                # enforce exact normalisation

    # initial full convolution
    conv = full_convolution(a)

    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    print(f"[0.00s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Annealing schedule & bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # small safety margin
    total_time = deadline - start_time

    # temperature parameters (scale of objective is ~1.5)
    T_start = 0.02
    T_end = 0.0001

    # move‑size parameters
    max_frac_initial = 0.25      # at most 25 % of a source entry can be moved
    max_frac_final = 0.001       # shrink to a very small fraction near the end

    it = 0
    no_improve = 0
    RENORMALISE_EVERY = 2000     # full recompute interval
    RESTART_AFTER = 5000        # steps without improvement -> random restart

    while time.time() < deadline:
        it += 1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # Track global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] New best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 1. Determine a shift where the convolution is maximal
        # ------------------------------------------------------------------
        # tolerance to collect a few equally maximal shifts
        tol = 1e-14
        active_shifts = np.where(conv >= cur_max - tol)[0]
        if active_shifts.size == 0:
            shift = rng.integers(2 * n - 1)
        else:
            shift = int(rng.choice(active_shifts))

        # ------------------------------------------------------------------
        # 2. Choose a contributing pair (i , j) with i+j = shift,
        #    biased by the magnitude of the product a_i·a_j.
        # ------------------------------------------------------------------
        lo = max(0, shift - (n - 1))
        hi = min(n - 1, shift)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = shift - idx
        prod = a[idx] * a[partner]
        if prod.sum() > 0:
            probs = prod / prod.sum()
            i = int(rng.choice(idx, p=probs))
        else:
            i = int(rng.choice(idx))
        j = shift - i

        # ------------------------------------------------------------------
        # 3. Pick the source (the larger of the two) and a random destination.
        # ------------------------------------------------------------------
        src = i if a[i] >= a[j] else j
        if a[src] <= 1e-15:          # fall back to any non‑zero entry
            nz = np.where(a > 1e-15)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz))

        dst = int(rng.integers(n))
        while dst == src:
            dst = int(rng.integers(n))

        # ------------------------------------------------------------------
        # 4. Determine the amount of mass to move.
        #    The allowed fraction decays linearly from max_frac_initial to
        #    max_frac_final.
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = max_frac_initial - (max_frac_initial - max_frac_final) * (elapsed / total_time)
        frac = max(frac, max_frac_final)

        eps = a[src] * rng.uniform(frac * 0.2, frac)   # keep it a bit smaller than the bound
        if eps <= 0.0 or eps > a[src]:
            continue

        # ------------------------------------------------------------------
        # 5. Apply the move, evaluate, and decide acceptance.
        # ------------------------------------------------------------------
        apply_move(a, conv, src, dst, eps)
        new_score = 2.0 * n * conv.max()

        # temperature decays linearly as well
        T = T_start - (T_start - T_end) * (elapsed / total_time)
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            # simulated‑annealing uphill acceptance
            if rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Improved (annealed) = {best_score:.6f}")
        else:
            # revert the move
            revert_move(a, conv, src, dst, eps)
            no_improve += 1

        # ------------------------------------------------------------------
        # 6. Periodic housekeeping: renormalise to avoid drift and full recompute.
        # ------------------------------------------------------------------
        if it % RENORMALISE_EVERY == 0:
            total = a.sum()
            if abs(total - 1.0) > 1e-12:
                a /= total
                conv = full_convolution(a)   # rebuild from scratch

        # ------------------------------------------------------------------
        # 7. Random restart if we have been stuck for a while.
        # ------------------------------------------------------------------
        if no_improve >= RESTART_AFTER:
            a = rng.exponential(scale=1.0, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            conv = full_convolution(a)
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Random restart (iteration {it})")

    # ------------------------------------------------------------------
    # Return the best sequence found (normalised, non‑negative)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
