"""
Enhanced search for a non‑negative step function that minimises
evaluate_sequence().  The algorithm consists of three phases:

1️⃣  Short random Dirichlet initialisation to obtain a decent starting
    point (the best of the trials is kept).

2️⃣  A smooth “log‑sum‑exp” optimisation.  The original objective
    2·n·max(conv(a,a)) is approximated by
        L_β(a) = 2·n·max(conv) + (2·n/β)·log Σ exp(β·(conv−max(conv))).
    For a sequence of increasing β this becomes a tighter approximation.
    In each β‑stage a multiplicative (exponentiated‑gradient) update
    a←a·exp(−η·∇L_β) is performed, the vector is re‑normalised onto the
    probability simplex and the best legal vector seen so far is tracked.

3️⃣  A final projected‑gradient descent on the exact non‑smooth objective
    (identical to the original PGD) to fine‑tune the candidate.

The whole search respects the 1000 s budget; if time runs out the best
vector found so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, so denominator equals 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Gradient of the smoothed log‑sum‑exp objective
# ----------------------------------------------------------------------
def _grad_from_lambda(a: np.ndarray, lam: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a L given λ = ∂L/∂conv (array size 2·n‑1).
    The formula is grad_i = 2·Σ_k λ_k·a_{k‑i}.
    Implemented via FFT (convolution with reversed λ).
    """
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(lam[::-1], n=L)      # λ reversed
    conv = np.fft.irfft(A * B, n=L)[: 3 * n - 2]
    part = conv[n - 1 : 2 * n - 1]       # length n
    grad = 2.0 * part[::-1]
    return grad


def _smooth_objective_and_grad(a: np.ndarray, beta: float):
    """
    Returns a tuple (smooth_obj, grad, conv, max_conv) where
    smooth_obj = 2·n·max(conv) + (2·n/β)·log Σ exp(β·(conv‑max(conv))).
    grad is the exact Euclidean gradient w.r.t. a.
    """
    n = a.shape[0]
    conv = _conv_self(a)                 # full self‑convolution
    maxc = conv.max()
    # shift to improve numerical stability
    shifted = conv - maxc                 # max becomes 0
    exp_vals = np.exp(beta * shifted)    # exp(0)=1 is the largest term
    sum_exp = exp_vals.sum()
    # soft‑max weights (probabilities)
    w = exp_vals / sum_exp
    # λ_k = ∂L/∂conv_k = 2·n·w_k
    lam = 2.0 * n * w
    # gradient of the smooth objective
    grad = _grad_from_lambda(a, lam)
    # smooth objective value
    smooth = 2.0 * n * maxc + (2.0 * n / beta) * np.log(sum_exp)
    return smooth, grad, conv, maxc


# ----------------------------------------------------------------------
# Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        conc_log = rng.uniform(-3.0, 0.7)          # concentration between 1e‑3 and 5
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Final projected gradient descent (as in the reference implementation)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray, cur_score: float, deadline: float):
    """Fine‑tune the vector by moving opposite to the sub‑gradient and projecting."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    step = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break

        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)
    return best_a, best_score

    # Problem size – keep the same as the reference solution
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # start from a uniform vector
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1) if present in the environment
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisation (≈15 s)
    # --------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 15.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=2000, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] After Dirichlet best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smoothed log‑sum‑exp optimisation (major part of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate ≈80 % of the remaining time to this phase
        smooth_deadline = time.time() + 0.80 * remaining
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
                         80.0, 160.0, 320.0, 640.0]   # increasingly tighter
        a = best_seq.copy()
        cur_obj = _objective_score(a)   # exact objective for tracking only
        for idx, beta in enumerate(beta_schedule):
            if time.time() >= smooth_deadline:
                break
            print(f"[{time.time() - start_time:.2f}s] Starting β={beta:.1f}")
            # give each β a slice of remaining time (roughly equal)
            phase_deadline = min(smooth_deadline,
                                 time.time() + 0.9 * (smooth_deadline - time.time()) / max(1, len(beta_schedule) - idx))
            step = 0.5   # initial multiplicative step size
            while time.time() < phase_deadline:
                smooth_val, grad, conv, maxc = _smooth_objective_and_grad(a, beta)
                # multiplicative update (exponentiated gradient)
                cand = a * np.exp(-step * grad)
                cand_sum = cand.sum()
                if cand_sum == 0.0:
                    # numerical catastrophe – fall back to a tiny uniform perturbation
                    cand = np.full_like(a, 1e-12)
                    cand_sum = cand.sum()
                cand = cand / cand_sum

                # evaluate the smooth objective for the candidate
                smooth_cand, _, _, _ = _smooth_objective_and_grad(cand, beta)

                if smooth_cand < smooth_val:
                    a = cand
                    smooth_val = smooth_cand
                    # evaluate the true (non‑smoothed) objective occasionally
                    true_score = _objective_score(a)
                    if true_score < best_score:
                        best_score = true_score
                        best_seq = a.copy()
                        print(f"[{time.time() - start_time:.2f}s] NEW BEST (β={beta:.1f}) score={best_score:.6f}")
                    # adapt step size upwards modestly
                    step = min(step * 1.10, 5.0)
                else:
                    # reject – shrink step
                    step *= 0.50
                # occasional random jitter to escape plateaus
                if np.random.rand() < 0.001:
                    jitter = np.random.normal(scale=1e-4, size=n)
                    a = np.maximum(a + jitter, 0.0)
                    a /= a.sum()
                # stop early if step becomes vanishingly small
                if step < 1e-12:
                    break
        # finish this phase with the best vector found
        best_seq = a.copy()
        best_score = _objective_score(best_seq)
        print(f"[{time.time() - start_time:.2f}s] End of smoothed optimisation – best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final projected‑gradient refinement (remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final projected‑gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Gradient descent finished, best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
