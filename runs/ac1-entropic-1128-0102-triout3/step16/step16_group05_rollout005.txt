"""
Propose a non‑negative step function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of several independent restarts.
Each restart runs a *smoothed* gradient descent in which the
max‑convolution is approximated by a log‑sum‑exp (soft‑max).
The gradient of this smooth surrogate is computed analytically via
FFT‑based convolutions.  A simple back‑tracking line search with
projection onto the probability simplex is used.  The smoothing
parameter λ is exponentially cooled, allowing the optimiser to
first escape plateaus and later focus on the true max‑objective.

When the allotted time for a restart is exhausted the algorithm
switches to the original projected sub‑gradient descent (the code
that gave the previous best result) for a short fine‑tuning phase,
followed by a tiny “mass‑transfer” hill‑climb that moves up to 1 %
of the source mass between indices.

Multiple restarts (uniform initialisation, random Dirichlet draws
and the previously found best sequence) are performed within the
global 1000 s budget; the best sequence ever seen is returned.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a (len n) and b (len m)."""
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    Af = rfft(a, n=L)
    Bf = rfft(b, n=L)
    conv = irfft(Af * Bf, n=L)[: n + m - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Smoothed gradient (soft‑max) of the max‑convolution
# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of λ·log‑sum‑exp(conv/λ) w.r.t. a.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length 2n‑1
    # stable soft‑max
    max_c = conv.max()
    exp_term = np.exp((conv - max_c) / lam)
    w = exp_term / exp_term.sum()            # weights, sum to 1

    # gradient = 2 * cross‑correlation of a with w
    # cross‑correlation can be obtained as convolution with the reversed w
    w_rev = w[::-1]
    full = _conv_full(a, w_rev)              # length 3n‑2
    L = conv.shape[0]                        # = 2n‑1
    grad = 2.0 * full[L - n : L][::-1]       # slice and reverse to length n
    return grad


# ----------------------------------------------------------------------
# Smoothed gradient descent (annealing λ)
# ----------------------------------------------------------------------
def _smoothed_descent(a0: np.ndarray,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth surrogate of the max‑convolution.
    λ is exponentially cooled; projection onto the simplex keeps the
    iterate feasible.  The best sequence seen (according to the true
    objective) is returned.
    """
    a = a0.copy()
    n = a.shape[0]

    # initial step size (in L2 norm) and schedule
    step_scale = 5e-3
    min_step = 1e-9

    # initialise λ after first convolution
    conv0 = _conv_self(a)
    lam_init = max(conv0.max() - conv0.min(), 1e-6)
    lam_min = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        # true objective (for acceptance)
        cur_score = _objective_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smoothed it={it} score={cur_score:.6f}")

        # cooling schedule for λ
        lam = max(lam_init * np.exp(-it / 2000.0), lam_min)

        # gradient of the smoothed objective
        grad = _grad_softmax(a, lam)
        grad_norm = np.linalg.norm(grad) + 1e-16
        direction = grad / grad_norm

        # back‑tracking line search on the *true* objective
        step = step_scale
        improved = False
        for _ in range(7):
            a_candidate = a - step * direction
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

            cand_score = _objective_score(a_candidate)
            if cand_score < cur_score - 1e-14:
                a = a_candidate
                cur_score = cand_score
                step_scale = min(step_scale * 1.02, 5e-2)
                improved = True
                break
            step *= 0.5

        if not improved:
            # no progress – shrink step and inject a tiny jitter
            step_scale = max(step_scale * 0.5, min_step)
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-5 + 1e-9)
            a[i1] -= eps
            a[i2] += eps
            a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Original sub‑gradient descent (kept for fine‑tuning)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (averaged over active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    step_scale = 5e-3
    min_step = 1e-10
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        grad_norm = np.linalg.norm(grad) + 1e-16

        step = step_scale
        improved = False
        for _ in range(6):
            direction = grad / grad_norm
            a_candidate = a - step * direction
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

            new_score = _objective_score(a_candidate)
            if new_score < cur_score:
                a = a_candidate
                cur_score = new_score
                step_scale = min(step_scale * 1.02, 5e-2)
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            step_scale = max(step_scale * 0.5, min_step)
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-5 + 1e-9)
            if eps > 0:
                a[i1] -= eps
                a[i2] += eps
                a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Directed hill‑climb moving up to 1 % of source mass."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.01 + 1e-8)   # up to 1 %
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimisation wrapper for one restart
# ----------------------------------------------------------------------
def _optimize_from(a0: np.ndarray,
                   run_deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run one optimisation run:
      * smoothed gradient descent (≈70 % of the allotted time)
      * short sub‑gradient descent (≈20 %)
      * final hill‑climb (remainder)
    Returns the best sequence and its score.
    """
    now = time.time()
    # allocate 70 % to smoothed phase
    smoothed_deadline = now + 0.68 * (run_deadline - now)
    best_seq, best_score = _smoothed_descent(
        a0, smoothed_deadline, a0, _objective_score(a0), rng)

    # sub‑gradient phase (if time permits)
    if time.time() < run_deadline - 5.0:
        sub_deadline = run_deadline - 5.0
        best_seq, best_score = _subgradient_descent(
            best_seq, sub_deadline, best_seq, best_score, rng)

    # final hill‑climb (use whatever time is left)
    if time.time() < run_deadline:
        best_seq, best_score = _hill_climb(
            best_seq, run_deadline, best_seq, best_score, rng)

    return best_seq, best_score

# ----------------------------------------------------------------------

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
