def weighted_grad(a, conv, tau):
    max_conv = conv.max()
    # compute weights using softmax

"""
Advanced search for a low‑peak self‑convolution step function.

The algorithm consists of three stages:

1️⃣  Fast random Dirichlet draws (≈30 s) to obtain a good seed.
2️⃣  Deterministic “analytic move’’ descent:
    – the most/least contributing indices are taken from a sub‑gradient,
    – for each source‑destination pair the exact optimal mass transfer
      that minimises the new maximal convolution is computed analytically
      (piece‑wise linear minimax).  The move is applied only if the true
      objective improves.  Several top candidates are examined each
      iteration.
3️⃣  Fine‑tuning with a smoothed‑max Adam optimiser.
    The max‑convolution is approximated by a log‑sum‑exp (temperature τ);
    its gradient is obtained by a weighted correlation (FFT).  After each
    step the vector is projected onto the probability simplex.

All phases respect the overall 1000 s budget and the best vector encountered
is returned as a plain list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Basic linear‑algebra utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ------------------------------------------------------------
# Objective helpers (the sum is forced to 1 → denominator = 1)
# ------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) under the condition sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at `a` (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance to capture all active maxima
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution a[k-idx] appears twice (i↔j)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ------------------------------------------------------------
# Smoothed max gradient (log‑sum‑exp) – used in the Adam stage
# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of  τ·log∑exp(conv/τ) w.r.t. `a`.
    Returns the vector dS/da (without the outer factor 2·n).
    """
    n = a.shape[0]
    max_c = conv.max()
    # soft‑max – stable
    w = np.exp((conv - max_c) / tau)
    w /= w.sum()

    # weighted correlation a ⋆ w  (size 2n‑1 ↔ n)
    L = _next_pow2(n + len(w) - 1)
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(Fa * np.conj(Fw), n=L)
    start = len(w) - 1                # index where correlation aligns with a[0]
    grad = 2.0 * corr[start:start + n]   # factor 2 from d(conv)/da
    return grad


# ------------------------------------------------------------
# Analytic move for a given (src,dst) pair
# ------------------------------------------------------------
def _analytic_move(a: np.ndarray, conv: np.ndarray, src: int, dst: int):
    """
    Compute the optimal mass transfer from `src` to `dst` that minimises
    the maximal self‑convolution (ignoring second‑order terms).
    Returns (new_a, delta, new_max) where `new_max` is the peak after the move.
    If no beneficial move exists, delta = 0 and new_a = a.
    """
    n = a.shape[0]
    # Build the two shifted slices of `a` that appear in the linearised change.
    idx = np.arange(2 * n - 1)
    t_src = idx - src
    t_dst = idx - dst

    term_src = np.zeros_like(conv)
    term_dst = np.zeros_like(conv)

    src_mask = (t_src >= 0) & (t_src < n)
    dst_mask = (t_dst >= 0) & (t_dst < n)

    term_src[src_mask] = a[t_src[src_mask]]
    term_dst[dst_mask] = a[t_dst[dst_mask]]

    diff = term_dst - term_src               # diff[k] = a_{k‑dst} – a_{k‑src}

    dec_mask = diff < 0   # shifts where the move lowers the convolution
    inc_mask = diff > 0   # shifts where it raises the convolution

    if not dec_mask.any():
        # moving mass would not lower any entry ⇒ no improvement possible.
        return a, 0.0, conv.max()

    # decreasing shift with the largest current value
    dec_idx = np.argmax(np.where(dec_mask, conv, -np.inf))
    conv_dec = conv[dec_idx]
    diff_dec = -diff[dec_idx]            # positive

    if inc_mask.any():
        # increasing shift with the largest current value
        inc_idx = np.argmax(np.where(inc_mask, conv, -np.inf))
        conv_inc = conv[inc_idx]
        diff_inc = diff[inc_idx]        # positive

        numer = conv_dec - conv_inc
        denom = 2.0 * (diff_dec + diff_inc)
        delta_opt = max(0.0, numer / denom) if denom > 0 else 0.0
    else:
        # No increase – we can transfer the whole source mass.
        delta_opt = a[src]

    delta_opt = min(delta_opt, a[src])
    if delta_opt <= 0.0:
        return a, 0.0, conv.max()

    # Apply the move and recompute the true peak (exact, not linearised)
    a_new = a.copy()
    a_new[src] -= delta_opt
    a_new[dst] += delta_opt
    conv_new = _conv_self(a_new)
    new_max = conv_new.max()
    return a_new, delta_opt, new_max

def _random_dirichlet_trials(n: int, max_time: float, deadline: float):
    """Return the best vector among many Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    start = time.time()
    while time.time() < deadline and (time.time() - start) < max_time:
        conc_log = rng.uniform(-3.0, 0.7)          # log10 concentration ∈ [‑3, 5]
        c = 10 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4g} score={sc:.6f}")
    return best_seq, best_score

    """

    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score

            for dst in dst_cand:

        if improvement:
            a = best_candidate
            cur_score = best_candidate_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Analytic descent new best {best_score:.6f}")
            continue

        # fallback: a tiny random transfer (helps to escape plateaus)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if a[src] <= 0:
            continue
        delta = a[src] * rng.random() * 0.001   # ≤0.1 % of source mass
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Random tweak new best {best_score:.6f}")
        else:
            # no progress – break after a few idle iterations?
            # Here we simply continue trying.
            pass

    return best_a, best_score


# ------------------------------------------------------------
# Phase 3 – smoothed‑max Adam optimisation
# ------------------------------------------------------------
def _adam_optim(a0: np.ndarray, cur_score: float, deadline: float):
    """
    Adam optimiser on the smoothed objective (log‑sum‑exp of the convolution).
    After each update the vector is projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05

    # temperature schedule for the smooth max
    conv_initial = _conv_self(a)
    tau = max(0.01, conv_initial.max() * 5.0)   # start relatively smooth
    tau_min = 1e-7
    tau_decay = 0.9995

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)

        # update temperature (slow cooling)
        tau = max(tau * tau_decay, tau_min)

        # gradient of the smoothed max (without the outer factor 2·n)
        grad_s = _smooth_gradient(a, conv, tau)
        grad = 2.0 * n * grad_s          # full gradient of the true objective

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(np.maximum(a, 0.0))

        # periodic evaluation (every 10 steps)
        if t % 10 == 0:
            sc = _objective_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam iter {t} new best {best_score:.6f}")

    return best_a, best_score


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list) found within the time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 works well for the time budget
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Global starter (if supplied by the surrounding environment)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet initialisation (≈30 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 30.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_time=30.0, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] After Dirichlet best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic analytic descent (up to ~10 min)
    # ------------------------------------------------------------------
    analytic_deadline = min(deadline, start_time + 600.0)   # use roughly first 10 min for this phase
    if time.time() < analytic_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting deterministic analytic descent")
        best_seq, best_score = _deterministic_descent(best_seq, best_score, analytic_deadline)
        print(f"[{time.time() - start_time:.2f}s] Analytic descent finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – smoothed‑max Adam fine‑tuning (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optim(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Adam optimisation finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
