def _smooth_gradient_descent(start_seq, deadline, best_seq, best_score):
    a = start_seq.copy()
    n = a.shape[0]
    step_nom = 0.5  # initial step magnitude (not scaled)
    tau_factor = 0.2  # start tau as fraction of current max
    # schedule tau decreasing multiplicatively
    tau = max(1e-12, tau_factor * _conv_self(a).max())
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        # update best
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD it={iteration} score={cur_score:.6f}")

        # compute weights for smooth max
        max_c = cur_max
        # for numerical stability, subtract max before exponentiating
        exp_vals = np.exp((conv - max_c) / max(tau, 1e-16))
        w_sum = exp_vals.sum()
        if w_sum == 0:

"""
Optimiser for the step‑function evaluation problem.

The algorithm:
    1.  Initialise from a global starter (if present) or from a few
        random Dirichlet draws.
    2.  Smooth (log‑sum‑exp) gradient descent on 2·n·max(conv(a,a)).
        The gradient is obtained from the smoothed max‑convolution,
        the step size is chosen by back‑tracking line search and the
        iterate is projected onto the probability simplex.
        The smoothing parameter τ is gradually reduced (continuation).
    3.  Aggressive hill‑climb: for the active constraints (those
        attaining the current maximum) compute the contribution of each
        coordinate, then move as much mass as safely possible from a
        “bad’’ coordinate to a “good’’ one (using a linear bound on the
        change of the peak of the convolution).  This step is monotone
        and often pushes the bound below the previous best.
    4.  Return the best sequence found (normalised to sum = 1).

The whole routine respects a 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Random initialisers
# ----------------------------------------------------------------------


def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=200):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)      # concentration ∈ [0.01,5]
        c = 10.0 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        cur_score = 2.0 * n * _conv_self(a).max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {cur_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp continuation)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(start_seq, deadline, best_seq, best_score):
    """
    Minimise 2·n·max(conv(a,a)) using a smooth approximation
    Mτ = τ·log∑exp(conv/τ).  The gradient of Mτ is obtained via a
    convolution of the weight vector with the reversed current sequence.
    The iterate is projected onto the simplex after each tentative step.
    """
    a = start_seq.copy()
    n = a.shape[0]
    step_nom = 0.5                     # nominal step size (pre‑normalisation)
    tau = max(1e-12, 0.2 * _conv_self(a).max())   # initial smoothing parameter
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD it={it} score={cur_score:.6f}")

        # ---------- smooth max ----------
        max_c = cur_max
        # stable softmax weights (w ≥ 0, ∑w = 1)
        w = np.exp((conv - max_c) / max(tau, 1e-16))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # ---------- gradient of the smoothed objective ----------
        # grad = 4·n·(w * a) where * denotes the correlation (w convolved with a_rev)
        a_rev = a[::-1]
        full = _conv_fft(w, a_rev)               # length = 3·n‑2
        start = n - 1
        grad_slice = full[start:start + n]       # length n
        grad = 4.0 * n * grad_slice

        # ---------- back‑tracking line search ----------
        gnorm = np.linalg.norm(grad) + 1e-16
        step_len = step_nom / gnorm
        while True:
            a_new = a - step_len * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            new_conv = _conv_self(a_new)
            new_max = new_conv.max()
            if new_max < cur_max or step_len <= 1e-12:
                break
            step_len *= 0.5

        if new_max < cur_max:
            a = a_new
            step_nom = min(step_nom * 1.05, 5.0)
        else:
            step_nom = max(step_nom * 0.5, 1e-12)

        # reduce τ slowly (continuation)
        if it % 100 == 0:
            tau = max(tau * 0.9, 1e-12)

    return best_seq, best_score

def _contributions_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """For each index i compute Σ_{k∈active} a_{k‑i}."""
    n = a.shape[0]
    L = 2 * n - 1
    ind = np.zeros(L, dtype=np.float64)
    ind[active] = 1.0
    # convolution of indicator with reversed a, then extract the middle n entries
    full = _conv_fft(ind, a[::-1])
    start = n - 1
    return full[start:start + n]


def _aggressive_hill_climb(start_seq, deadline, best_seq, best_score):
    """
    Deterministic hill‑climb:
        * compute the set of active constraints (positions attaining the current maximum);
        * evaluate each coordinate's contribution to those constraints;
        * move as much mass as possible from the worst (largest contribution) to the best (smallest contribution)
          while guaranteeing that the peak of the convolution does not increase.
    """
    a = start_seq.copy()
    n = a.shape[0]
    L = 2 * n - 1
    rng = np.random.default_rng()
    stagnation = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            stagnation = 0
            print(f"[{time.time() - start_time:.2f}s] AggHill score={cur_score:.6f}")
        else:
            stagnation += 1

        # ---------- active set ----------
        tol = 1e-12
        active = np.where(conv >= cur_max - tol)[0]
        if active.size == 0:
            break

        # ---------- contributions ----------
        contrib = _contributions_from_active(a, active)

        # choose source (large contribution, non‑zero mass) and destination (small contribution)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(contrib[src_candidates])]
        dst = int(np.argmin(contrib))

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        # ---------- maximal safe ε ----------
        # build shifted versions of a (zero‑padded)
        pad_src = np.concatenate([np.zeros(src), a, np.zeros(L - src - n)])
        pad_dst = np.concatenate([np.zeros(dst), a, np.zeros(L - dst - n)])
        diff = pad_dst - pad_src                # diff[k] = a_{k‑dst} – a_{k‑src}
        pos = diff > 1e-15
        if np.any(pos):
            numerator = cur_max - conv           # ≥0
            eps_allowed = numerator[pos] / (2.0 * diff[pos])
            eps_max = eps_allowed.min()
        else:
            eps_max = 0.0

        eps_max = min(eps_max, a[src])          # cannot take more than we have

        if eps_max <= 0.0:
            # no safe move – perform a tiny random jitter occasionally
            if stagnation > 500:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.01 + 1e-6)
                a[i1] = max(a[i1] - jitter, 0.0)
                a[i2] += jitter
                a = _project_simplex(a)
                stagnation = 0
            continue

        eps = 0.9 * eps_max                     # keep a safety margin
        a[src] -= eps
        a[dst] += eps
        a = np.maximum(a, 0.0)                  # numerical safety
        # renormalise (sum should be unchanged, but rounding may drift)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / n)

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The algorithm respects the wall‑clock budget
    (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05    # tiny safety margin

    # problem size – 4096 gives a good trade‑off between flexibility and speed
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = 2.0 * n * _conv_self(best_seq).max()
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (e.g. from a previous run)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = 2.0 * n * _conv_self(starter).max()
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter → {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – a few Dirichlet draws (≈10 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.10
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase1_deadline, trials=300
        )

    # ------------------------------------------------------------------
    # Phase 2 – smooth gradient descent with continuation (≈55 %)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.65
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth gradient descent")
        best_seq, best_score = _smooth_gradient_descent(
            best_seq, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After smooth GD → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting aggressive hill‑climb")
        best_seq, best_score = _aggressive_hill_climb(
            best_seq, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
