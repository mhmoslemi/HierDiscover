    iteration = 0
    # Simulated annealing temperature schedule
    T0 = 1.0
    # temperature decays with iterations: T = T0 * (1 - progress)
    # step_max decays

    L_fft = _next_pow2(2*n - 1)

    # Initialize with uniform
    a = np.full(n, 1.0/n)
    cur_score, _ = _objective_score(a, n, L_fft)
    best_a = a.copy()
    best_score = cur_score

    # Precompute weights
    idx = np.arange(n)
    centre_dist = np.abs(idx - (n-1)/2) / ((n-1)/2)   # 0 at centre, 1 at edges? Actually we want donor center high: we could set donor_weight = 1 - centre_dist (max at centre)
    donor_weight = 1 - centre_dist
    donor_weight /= donor_weight.sum()
    receiver_weight = centre_dist  # edges high, centre low
    receiver_weight /= receiver_weight.sum()

    iter_count = 0
    no_improve = 0
    while time.time() < deadline:
        iter_count += 1
        # choose donor, receiver
        i = rng.choice(n, p=donor_weight)
        j = rng.choice(n, p=receiver_weight)
        if i == j:
            continue
        if a[i] <= 1e-12:  # insufficient mass
            continue
        # step size decays
        max_delta = min(a[i], 0.02)  # allow up to 2% absolute.
        delta = rng.uniform(0, max_delta)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non-negativity
        if a_new[i] < 0:
            continue
        # recompute score
        new_score, _ = _objective_score(a_new, n, L_fft)
        # accept if improve
        if new_score < cur_score:

            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start:.2f}s] iter={iter_count} new best {new_score:.6f}")
            no_improve = 0
        else:

            temperature = 0.01 * (1 - (time.time() - start) / (budget_s - 0.1))
            if temperature > 0 and rng.random() < np.exp(-(new_score - cur_score) / (temperature + 1e-12)):

                cur_score = new_score

            no_improve = 0
    # final best normalization
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()

"""
Simulated‑annealing / hill‑climbing search for a low‑score step function.

The score to minimise is
    score = 2·n·max(conv(a,a)) / (sum(a)²)

Because the objective is scale‑invariant we keep the vector on the
probability simplex (∑a_i = 1).  Uniform weights give score = 2, but the
maximum of the convolution can be reduced by moving mass from the
middle of the vector to the ends – this flattens the distribution of
sums of two independent draws.

The algorithm works in three stages:

1. **Initialisation** – start from the supplied global starter (if any)
   or from the uniform distribution; apply a short random perturbation
   that breaks the symmetry.

2. **Annealed random transfers** – repeatedly pick a “donor’’ index near
   the centre and a “receiver’’ index near the edges, move a random
   amount of probability mass δ from donor to receiver, recompute the
   exact score (FFT‑based convolution) and accept the move if it improves
   the current score.  Occasionally uphill moves are accepted according
   to an exponential schedule (simulated annealing) to escape local
   minima.  The step size and temperature decay with elapsed time.

3. **Final polishing** – blend the best vector with the uniform
   distribution and perform a light random‑pair hill‑climb to squeeze out
   any remaining improvement.

The routine respects a user‑supplied time budget (default 1000 s) and
returns a plain Python list of non‑negative floats.  All helper functions
are defined at module level – no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)−1) via FFT.
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact score used by the evaluator:
        2·n·max(conv(a,a))
    (the denominator sum(a)² is 1 because we keep a on the simplex).
    """
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max())

# Core optimisation routine
# ----------------------------------------------------------------------

    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  The returned list sums to (approximately) one – the external
    evaluator will rescale it automatically.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final return
    deadline = start_time + budget_s - 0.15

    # --------------------------------------------------------------
    # Problem size – 4096 works well and is used by the reference code
    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0️⃣  Initialise – uniform, or a supplied global starter
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # try to use a globally supplied starter (height_sequence_1)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Break the symmetry with a short random perturbation
    # --------------------------------------------------------------
    # a few random mass transfers to move away from the exact uniform point
    for _ in range(2000):
        i, j = rng.choice(n, size=2, replace=False)
        if best_a[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, min(best_a[i], 0.01))
        best_a[i] -= delta
        best_a[j] += delta
    # re‑normalise (tiny numerical drift may have occurred)
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    cur_a = best_a.copy()
    cur_score = _objective_score(cur_a, n, L_fft)

    # --------------------------------------------------------------
    # 2️⃣  Main annealed random‑transfer search
    # --------------------------------------------------------------
    # Pre‑compute donor / receiver bias (centre → edges)
    idx = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    dist = np.abs(idx - centre) / centre          # 0 at centre, 1 at edges
    donor_bias = 1.0 - dist                        # high in the middle
    donor_bias = np.clip(donor_bias, 0.0, None)
    donor_bias /= donor_bias.sum()
    receiver_bias = dist                           # high at the edges
    receiver_bias = np.clip(receiver_bias, 0.0, None)
    receiver_bias /= receiver_bias.sum()

    it = 0
    no_improve = 0
    last_print = start_time

    while time.time() < deadline:
        it += 1
        # ----------------------------------------------------------
        # Choose a donor (centre) and a receiver (edge)
        # ----------------------------------------------------------
        i = rng.choice(n, p=donor_bias)
        j = rng.choice(n, p=receiver_bias)
        if i == j or cur_a[i] <= 1e-14:
            continue

        # Step size slowly shrinks from ~0.02 to ~0.001
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)
        max_delta = max(0.001, 0.02 * (1.0 - progress))
        delta = rng.uniform(0.0, min(cur_a[i], max_delta))
        if delta <= 0.0:
            continue

        # ----------------------------------------------------------
        # Propose the transfer and evaluate
        # ----------------------------------------------------------
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = _objective_score(cur_a, n, L_fft)

        # Accept if improvement, otherwise maybe accept uphill
        accepted = False
        if new_score < cur_score:
            accepted = True
        else:
            # Simulated‑annealing acceptance
            temperature = max(1e-6, 0.01 * (1.0 - progress))
            if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                accepted = True

        if accepted:
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time() - start_time:.2f}s] it={it:,} new best score={new_score:.6f}")
        else:
            # revert the move
            cur_a[i] += delta
            cur_a[j] -= delta
            no_improve += 1

        # occasional status print
        if time.time() - last_print > 30.0:
            print(f"[{time.time() - start_time:.2f}s] it={it:,} cur score={cur_score:.6f} best={best_score:.6f}")
            last_print = time.time()

        # if we have been stuck for a long time, inject a larger random jitter
        if no_improve > 200_000:
            eps = rng.normal(scale=0.005, size=n)
            cur_a = np.maximum(best_a + eps, 0.0)
            cur_a /= cur_a.sum()
            cur_score = _objective_score(cur_a, n, L_fft)
            no_improve = 0

    # --------------------------------------------------------------
    # 3️⃣  Final polishing: mix with uniform + tiny hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        uniform = np.full(n, 1.0 / n)
        # coarse mixing
        for lam in np.linspace(0.0, 1.0, 21):
            if time.time() >= deadline:
                break
            cand = lam * best_a + (1.0 - lam) * uniform
            cand /= cand.sum()
            sc = _objective_score(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] mix λ={lam:.3f} → score={sc:.6f}")

        # very light pairwise hill‑climb
        while time.time() < deadline:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(0.0, min(best_a[i], 0.003))
            if delta == 0.0:
                continue
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = _objective_score(cand, n, L_fft)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Normalise to the probability simplex and return a plain list
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a /= total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
