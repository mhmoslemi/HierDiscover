def _subgradient_maxconv(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)

        idx = np.arange(lo, hi + 1)
        grad[idx] += a[k - idx]
    if active.size:

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    n = 8192
    # initialize a as uniform
    a = np.full(n, 1.0/n)
    # alternative: start from height_sequence_1 if appropriate
    # For using Adam, we need x = log(a)
    x = np.full(n, 0.0) # uniform
    # Adam stats
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.05
    best_a = a.copy()
    conv = _conv_self(a)
    best_score = 2.0 * n * conv.max()
    best_iter_no_improve = 0
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        # compute a via softmax
        # shift x to avoid overflow
        x_shifted = x - np.max(x)
        exp_x = np.exp(x_shifted)
        a = exp_x / np.sum(exp_x)

        # compute conv and score
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_iter_no_improve = 0
            print(f"[{time.time() - start:.2f}s] iter {iter_count} new best score {best_score:.6f}")
        else:
            best_iter_no_improve += 1

        # gradient
        grad_a = _subgradient_maxconv(a, conv)
        # full objective gradient factor:
        grad_a_full = (2.0 * n) * grad_a  # optional multiply for scaling

        mu = np.dot(a, grad_a_full)
        grad_x = a * (grad_a_full - mu)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**iter_count)
        v_hat = v / (1 - beta2**iter_count)

        best_a, best_score = _hill_climb(best_a, hill_deadline, best_a, best_score, rng)

    # final normalization

    total = best_a.sum()
    if total > 0:

    else:

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]

"""
Propose a non‑negative step function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

* **Search space** – we use a probability vector `a` (sum = 1, a_i ≥ 0).
  The objective does not depend on the overall scale, thus this
  normalisation is harmless.

* **Smooth approximation** – the non‑smooth `max` is replaced by a
  temperature‑controlled soft‑max:
        f_β(a) = (1/β)·log Σ_k exp(β·conv_k(a)).
  As β→∞ this converges to `max(conv)`.  The gradient of `f_β` can be
  computed in O(n log n) by a single FFT‑based cross‑correlation.

* **Projected Adam** – we run an Adam optimiser on the smooth objective,
  projecting the iterate onto the simplex after each step
  (`_project_simplex`).  The temperature β is increased geometrically
  (annealing) so the algorithm first sees a smooth landscape and then
  focuses on the true max‑objective.

* **Multiple restarts** – the whole annealing‐Adam is started from several
  random Dirichlet initialisations (including the optional global starter
  `height_sequence_1`).  The best candidate seen so far is kept.

* **Final hill‑climb** – a deterministic greedy mass‑transfer routine
  (`_hill_climb`) refines the best vector for the last part of the
  budget.

The routine respects the 1000 s wall‑clock limit, prints intermediate
scores and finally returns a plain Python list suitable for
`evaluate_sequence`.  Larger `n` (default 8192) gives more degrees of
freedom while still being fast enough on a single CPU.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:                                   # numerical safety
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_β(a) = (1/β)·log Σ_k exp(β·conv_k)
    (i.e. the gradient of max(conv) smoothed by a soft‑max).
    Returns ∇_a max(conv) (without the outer factor 2·n).
    """
    n = a.shape[0]
    # soft‑max weights on the convolution entries
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w /= w.sum()

    # cross‑correlation w ⋆ a :  grad_i = 2· Σ_j w_{i+j}·a_j
    w_rev = w[::-1]
    L = _next_pow2(3 * n - 2)
    W = rfft(w_rev, n=L)
    A = rfft(a, n=L)
    tmp = irfft(W * A, n=L)[: 3 * n - 2]          # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)            # pick the needed slice
    grad = 2.0 * tmp[idx]                       # ∇ max(conv)
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _smooth_adam(a0: np.ndarray,
                 beta_start: float,
                 beta_max: float,
                 beta_growth: float,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth objective with geometric temperature increase.
    a is kept on the simplex via Euclidean projection.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.08            # base learning rate (tuned empirically)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = beta_start

    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] β={beta:.1f} it={it} score={cur_score:.6f}")

        # compute smooth gradient of max(conv)
        grad = _smooth_gradient(a, conv, beta)      # ∇ max(conv)
        grad_obj = 2.0 * n * grad                    # ∇ (2·n·max(conv))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_obj
        v = beta2 * v + (1.0 - beta2) * (grad_obj * grad_obj)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # increase temperature slowly
        if beta < beta_max:
            beta = min(beta_max, beta * beta_growth)

    return best_seq, best_score


def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb that moves a tiny fraction of mass from the index
    with the largest sub‑gradient to the one with the smallest.
    Occasionally adds random jitter to escape plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.01 + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score
