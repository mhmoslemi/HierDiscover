Projection onto simplex:

The algorithm: Given y (any real vector), compute sorted y descending: u = sorted(y)[::-1]; compute cumsum = np.cumsum(u) - 1; find rho = max { i | u_i > c_i / (i+1) }. Then theta = c_i_f / (i+1). Then x = max(y - theta, 0). Implementation using numpy.

Pseudo-code:

def simplex_projection(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimiser for the step-function evaluation problem.
1️⃣  Initialise from uniform, a possible external starter or a few random seeds.
2️⃣  Smoothed gradient descent on log‑sum‑exp of the self‑convolution,
    using FFT‑based gradients and multiplicative updates (mirror descent
    in the KL geometry).  The smoothing parameter γ is slowly increased,
    allowing the optimisation to first explore and then focus on the peak.
3️⃣  Local hill‑climbing (mass‑transfer moves) for fine tuning.
4️⃣  Optional mixing with uniform via a short golden‑section search.

The routine respects the given time‑budget (default 1000 s) and returns
the best non‑negative sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution a*a using FFT of length L."""
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * a.shape[0] - 1]
    return conv

def _objective_score(a: np.ndarray, n: int, L: int) -> float:
    """Score used for internal optimisation: 2·n·max(conv(a,a))."""
    conv = _conv_self_fft(a, L)
    return float(2.0 * n * conv.max())

def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β) discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The search combines random seeding, a smoothed
    gradient descent with FFT‑based gradients and a final local hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.25   # safety margin

    # --- problem size -------------------------------------------------
    # 8192 offers a slightly tighter theoretical bound than 4096 while still
    # fitting comfortably into the time budget.
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # --- baseline -----------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # --- optional global starter ---------------------------------------
    try:
        # the runner may provide a global variable `height_sequence_1`
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # adapt length if needed
            if starter.size > n:
                starter = starter[:n]
            elif starter.size < n:
                starter = np.pad(starter, (0, n - starter.size), 'constant')
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_seq, best_score = starter, sc
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # --- Phase 1: quick random seeding (≈10 % of budget) ---------------
    phase1_deadline = start + 0.10 * (deadline - start)
    while time.time() < phase1_deadline:
        # Dirichlet‑like draw (Gamma‑variables)
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time() - start:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

        # occasional beta‑shaped deterministic seed
        if rng.random() < 0.005:
            beta_params = [(0.2, 0.2), (0.5, 0.5), (0.3, 0.7), (0.7, 0.3),
                           (0.2, 0.8), (0.8, 0.2)]
            a_par, b_par = beta_params[rng.integers(len(beta_params))]
            v = _beta_shape(a_par, b_par, n)
            if v is not None:
                sc = _objective_score(v, n, L_fft)
                if sc < best_score:
                    best_seq, best_score = v, sc
                    print(f"[{time.time() - start:.2f}s] Beta seed ({a_par},{b_par}) → score={sc:.6f}")

    # --- Phase 2: smoothed gradient descent (≈70 % of budget) ----------
    phase2_deadline = start + 0.80 * (deadline - start)
    # parameters for the smoothed objective
    gamma0 = 5.0          # initial smoothing factor
    gamma_max = 200.0     # final smoothing factor
    gamma_inc_per_it = 0.0005   # additive increase per iteration
    lr0 = 0.5                     # initial step size for multiplicative update
    stagnation_patience = 200_000   # iterations without improvement before a perturbation

    a = best_seq.copy()
    it = 0
    last_improve = 0

    while time.time() < phase2_deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)                # length 2n‑1
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"[{time.time() - start:.2f}s] GD it={it:,} → score={cur_score:.6f}")

        # ----- smooth max via soft‑max -----
        gamma = gamma0 + it * gamma_inc_per_it
        if gamma > gamma_max:
            gamma = gamma_max
        # stable soft‑max
        maxc = conv.max()
        exp_vals = np.exp(gamma * (conv - maxc))
        w = exp_vals / exp_vals.sum()

        # ----- gradient of the smoothed objective (FFT‑based) -----
        W = np.fft.rfft(w, n=L_fft)
        A = np.fft.rfft(a, n=L_fft)
        grad = 2.0 * np.fft.irfft(np.conj(W) * A, n=L_fft)[:n]

        # ----- multiplicative (exponentiated) update -----
        eta = lr0 / math.sqrt(it)
        factor = np.exp(-eta * grad)
        # clip to avoid overflow / underflow
        factor = np.clip(factor, 1e-12, 1e12)
        a = a * factor
        a_sum = a.sum()
        if a_sum <= 0:
            a = uniform.copy()
        else:
            a = a / a_sum

        # ----- occasional random perturbation if we are stuck -----
        if it - last_improve > stagnation_patience:
            perturb = rng.normal(scale=0.01, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            a = a / a.sum()
            last_improve = it

    # --- Phase 3: local hill‑climbing (remaining time) -----------------
    remaining_deadline = deadline
    a = best_seq.copy()
    cur_score = 2.0 * n * _conv_self_fft(a, L_fft).max()
    while time.time() < remaining_deadline:
        # choose donor (large entry) and receiver (small entry)
        if rng.random() < 0.7:
            i = int(np.argmax(a))
        else:
            i = rng.integers(n)
        if a[i] <= 1e-14:
            continue
        if rng.random() < 0.7:
            j = int(np.argmin(a))
        else:
            j = rng.integers(n)
        if i == j:
            continue

        # move a tiny fraction of mass
        eps = min(0.02 * a[i], 1e-4)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        new_score = 2.0 * n * _conv_self_fft(a_new, L_fft).max()
        # accept if better; otherwise accept with a tiny temperature‑scaled probability
        if new_score < cur_score or rng.random() < 0.001:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start:.2f}s] Hill‑climb improved to {new_score:.6f}")

    # --- final sanity check -------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")

    # return as plain Python list
    return [float(v) for v in best_seq]

                (0.7, 0.3),
                (0.2, 0.8),
                (0.8, 0.2),
            ]
            a_par, b_par = beta_params[rng.integers(len(beta_params))]
            v = _beta_shape(a_par, b_par, n)
            if v is not None:
                sc = _objective_score(v, n, L_fft)
                if sc < best_score:
                    best_seq, best_score = v, sc
                    print(
                        f"[{time.time() - start:.2f}s] Beta seed ({a_par},{b_par}) → score={sc:.6f}"
                    )

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent (≈70 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start + 0.80 * (deadline - start)

    # smoothing (γ) parameters
    gamma0 = 5.0
    gamma_max = 200.0
    gamma_inc_per_it = 5e-4  # additive increase each iteration

    # step‑size for the multiplicative update
    lr0 = 0.5

    # stagnation handling
    stagnation_patience = 200_000

    a = best_seq.copy()
    it = 0
    last_improve = 0

    while time.time() < phase2_deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)               # length 2n‑1
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"[{time.time() - start:.2f}s] GD it={it:,} → score={cur_score:.6f}")

        # ----- soft‑max approximation of max(conv) -----
        gamma = gamma0 + it * gamma_inc_per_it
        if gamma > gamma_max:
            gamma = gamma_max
        maxc = conv.max()
        exp_vals = np.exp(gamma * (conv - maxc))       # stable exponentials
        w = exp_vals / exp_vals.sum()                  # soft‑max weights

        # ----- FFT‑based gradient of the smoothed objective -----
        W = np.fft.rfft(w, n=L_fft)
        A = np.fft.rfft(a, n=L_fft)
        grad = 2.0 * np.fft.irfft(np.conj(W) * A, n=L_fft)[:n]

        # ----- multiplicative (exponentiated) update -----
        eta = lr0 / math.sqrt(it)
        factor = np.exp(-eta * grad)
        factor = np.clip(factor, 1e-12, 1e12)          # protect against overflow
        a = a * factor
        a_sum = a.sum()
        if a_sum <= 0:
            a = uniform.copy()
        else:
            a = a / a_sum

        # ----- occasional random perturbation if stuck -----
        if it - last_improve > stagnation_patience:
            perturb = rng.normal(scale=0.01, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            a = a / a.sum()
            last_improve = it

    # ------------------------------------------------------------------
    # 3️⃣  Local hill‑climbing for the remaining time
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = 2.0 * n * _conv_self_fft(a, L_fft).max()
    while time.time() < deadline:
        # pick donor (large entry) and receiver (small entry)
        if rng.random() < 0.7:
            i = int(np.argmax(a))
        else:
            i = rng.integers(n)
        if a[i] <= 1e-14:
            continue
        if rng.random() < 0.7:
            j = int(np.argmin(a))
        else:
            j = rng.integers(n)
        if i == j:
            continue

        # move a tiny amount of mass
        eps = min(0.02 * a[i], 1e-4)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        new_score = 2.0 * n * _conv_self_fft(a_new, L_fft).max()
        # accept improvements; accept occasional uphill moves
        if new_score < cur_score or rng.random() < 0.001:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time() - start:.2f}s] Hill‑climb improved to {new_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity check & return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
