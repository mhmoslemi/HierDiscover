def _make_shape2(n, r, p, q):
    i = np.arange(n)
    idx = i + 1
    tail_idx = n - i
    log_shape = i*np.log(r) + p * np.log(idx) + q * np.log(tail_idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape
```

"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm follows several complementary phases:

1️⃣ **Extended parametric sweep** – dense grid over a three‑parameter family
   a_i ∝ r^i·(i+1)^p·(n−i)^q, plus a convex mixture with the uniform vector.
   This family contains geometric, power‑law, and tail‑controlled shapes.

2️⃣ **Adam optimisation on a smooth‑max surrogate** – we minimise
   f_β(a) = (1/β)·log Σ_k exp(β·conv_k(a,a)) using the Adam optimiser.
   The temperature β is increased gradually (continuation) so that the
   surrogate approaches the true max. After each step the iterate is
   Euclidean‑projected onto the probability simplex.

3️⃣ **Exponentiated‑gradient (mirror‑descent) on the exact L∞ objective** –
   multiplicative updates using the sub‑gradient at the worst convolution
   index, with an annealed step size.

4️⃣ **Final random hill‑climb** – tiny mass‑transfer moves with a simple
   acceptance rule to polish the solution until the time budget expires.

All phases respect the 1000 s budget, keep the vector on the simplex,
and retain the globally best sequence found.  The returned list is a plain
Python list of floats (sum ≈ 1).  Prints give a trace of the progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Exact objective (scale invariant – assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families (geometric × power × tail)
# ----------------------------------------------------------------------


def _make_shape3(n: int, r: float, p: float, q: float) -> np.ndarray:
    """Shape proportional to r^i·(i+1)^p·(n−i)^q, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    tail = n - i
    # Work in log‑space for numerical stability
    log_shape = i * np.log(r) + p * np.log(idx) + q * np.log(tail + 1e-12)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= shape_sum
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed_sum = mixed.sum()
    if mixed_sum == 0.0:
        mixed = uniform.copy()
    else:
        mixed /= mixed_sum
    return mixed


# ----------------------------------------------------------------------
# Smooth‑max utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max probabilities over the convolution entries."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k·conv_k(a,a) w.r.t. a.
    Implements  g_i = 2· Σ_j a_j·p_{i+j}   via FFT correlation.
    """
    n = a.shape[0]
    p_rev = p[::-1]                     # reversed soft‑max weights
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    corr = np.fft.irfft(Af * Pf, n=L_grad)
    # The correlation length is (n) + (2n‑1) – 1 = 3n‑2.
    # Extract the part that corresponds to indices i+j.
    start = n - 1
    g = corr[start:start + n][::-1]    # reverse to get correct ordering
    return 2.0 * g


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgrad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient at the worst index k (vector of length n)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    # Each pair (i,j) contributes 2·a_j to g_i (including diagonal)
    g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_phase(a0: np.ndarray,
                L_fft: int,
                L_grad: int,
                deadline: float,
                start_time: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Adam descent on f_β(a) = (1/β)·log Σ exp(β·conv) with continuation."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)               # first‑moment estimate
    v = np.zeros_like(a)               # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0
    max_beta = 1024.0                   # final temperature
    while time.time() < deadline:
        it += 1

        # Continuation schedule for β (smoothly rises to max_beta)
        beta = max_beta * (1.0 - np.exp(-it / 25000.0))

        # Convolution and soft‑max weights
        conv = _conv_self_fft(a, L_fft)
        p = _softmax_weights(conv, beta)

        # Gradient of the smooth surrogate
        g = _grad_softmax(a, p, L_grad)

        # Step‑size (decays like 1/√t)
        step = 0.06 / (np.sqrt(it) + 1.0)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        bias_c1 = 1.0 - beta1 ** it
        bias_c2 = 1.0 - beta2 ** it
        m_hat = m / bias_c1
        v_hat = v / bias_c2

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # Projection onto the simplex (keeps a ≥0 and Σa=1)
        a = _project_simplex(a)

        # Exact objective (for bookkeeping)
        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} "
                  f"β={beta:.0f} score={score:.6f}")

        # Small random jitter every few thousand steps to avoid stagnation
        if it % 2000 == 0:
            eps_move = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps_move, 0.0)
            a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase on the exact L∞ objective
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   L_fft: int,
                   deadline: float,
                   start_time: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Multiplicative updates a ← a·exp(−η·g) with annealed η."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k = int(np.argmax(conv))
        g = _subgrad_max_conv(a, k)

        # Annealed step size (log‑schedule)
        eta = eta0 / (1.0 + 0.001 * it)

        mult = np.exp(-eta * g)
        a = a * mult
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        score = _eval_fast(a, L_fft)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] exp‑grad it={it:,} "
                  f"score={score:.6f}")

        if it % 5000 == 0:
            eps_move = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps_move, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Final random hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------


def _final_hillclimb(a: np.ndarray,
                     L_fft: int,
                     deadline: float,
                     start_time: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very local moves – transfer ≤0.3 % of source mass to a random destination."""
    n = a.shape[0]
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 3e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        score = _eval_fast(a_prop, L_fft)
        if score < best_score:
            best_seq = a_prop
            best_score = score
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] hill‑climb score={score:.6f}")
    return best_seq, best_score
