def _beta_distribution(alpha, beta, n):
    """Return discrete Beta(alpha, beta) density over n points."""
    # x values centre of each bin.
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Use log to avoid overflow.
    log_a = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log1p(-x)
    # Clip extreme values to avoid overflow in exp.
    # np.log(0) = -inf; for x close to 0 or 1, log_a goes to -inf for alpha>1, but exp gives 0 fine.
    # For alpha<1, log_a may be large positive; clip.
    log_a = np.clip(log_a, -np.inf, 700.0)
    a = np.exp(log_a)
    # If any nan or inf occurs, fallback to uniform.
    if np.any(np.isnan(a)) or np.any(np.isinf(a)):
        a = np.maximum(a, 0)
    total = a.sum()
    if total <= 0:
        return np.full(n, 1.0/n)
    return a / total

def _conv_self_fft(a, L_fft):

"""
Step‑function optimiser.

We look for a non‑negative vector a (∑a = 1) that minimises

        score = 2·n·max(conv(a,a))

where conv(a,a) is the full linear convolution of a with itself.
The objective is scale‑invariant, so we stay on the probability simplex.

The search consists of three cheap, complementary phases:

1️⃣  Random “Beta‑shapes’’ – a_i ∝ x^{α−1}(1−x)^{β−1} on the grid
    x = (i+½)/n.  By varying (α,β) (including values < 1, i.e. U‑shapes)
    we scan a rich family of candidates.  Each candidate may additionally
    be blended with the uniform distribution.

2️⃣  Dirichlet draws – a standard way to obtain random points on the
    simplex, with a concentration parameter drawn log‑uniformly.

3️⃣  Sub‑gradient descent on the exact max‑convolution.  For the current
    vector we locate the index k⋆ where the convolution is maximal,
    compute the sub‑gradient ∇_i = 2·a_{k⋆−i} (zero when the index is out of
    range) and take a projected gradient step on the simplex.  The step
    size follows a 1/√t schedule.  This phase performs a precise local
    optimisation and is usually able to shave the last few thousandths
    off the score.

All phases respect the overall time budget (default 1000 s).  The routine
keeps the best vector seen so far and finally returns it as a plain list
of Python floats.

The code is deliberately self‑contained and uses only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)−1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact score used by the evaluator: 2·n·max(conv(a,a)).
    a is assumed to be normalised (sum = 1).
    """
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_distribution(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discrete Beta(α,β) density over n points (mid‑point rule).
    Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n

    # work in log‑space to avoid overflow for α,β<1
    log_a = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log1p(-x)
    # clip extreme positive values (exp(>700) would overflow)
    log_a = np.clip(log_a, -np.inf, 700.0)
    a = np.exp(log_a)
    # keep numbers in a sane range
    a = np.clip(a, 0.0, 1e100)

    total = a.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return a / total


# ----------------------------------------------------------------------
# 1️⃣  Random Beta‑shape search
# ----------------------------------------------------------------------
def _search_beta_shapes(a_best: np.ndarray,
                        score_best: float,
                        n: int,
                        L_fft: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Sample random Beta(α,β) densities (optionally mixed with the uniform
    distribution) and keep the best.
    """
    uniform = np.full(n, 1.0 / n)

    while time.time() < deadline:
        # α,β drawn log‑uniformly in a reasonable interval
        log_alpha = rng.uniform(np.log(0.2), np.log(5.0))
        log_beta = rng.uniform(np.log(0.2), np.log(5.0))
        alpha = np.exp(log_alpha)
        beta = np.exp(log_beta)

        a = _beta_distribution(alpha, beta, n)

        # blend with uniform – λ=0 (only uniform) never improves score,
        # but mixing is cheap and sometimes helps escape plateaus
        lam = rng.random()
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()

        sc = _objective_score(cand, n, L_fft)
        if sc < score_best:
            score_best, a_best = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Beta α={alpha:.3f} β={beta:.3f} λ={lam:.2f} → score={sc:.6f}")

    return a_best, score_best


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws
# ----------------------------------------------------------------------
def _search_dirichlet(a_best: np.ndarray,
                      score_best: float,
                      n: int,
                      L_fft: int,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Sample points on the simplex via a Dirichlet distribution with a
    log‑uniform concentration parameter.  Keep the best candidate.
    """
    while time.time() < deadline:
        # concentration c ∈ [0.02, 20] (log‑uniform)
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)

        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()

        sc = _objective_score(a, n, L_fft)
        if sc < score_best:
            score_best, a_best = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

    return a_best, score_best


# ----------------------------------------------------------------------
# 3️⃣  Sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a_start: np.ndarray,
                         score_start: float,
                         n: int,
                         L_fft: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly locate the index k⋆ where conv(a,a) is maximal,
    compute the sub‑gradient  ∇_i = 2·a_{k⋆−i} (zero outside the
    support) and take a projected gradient step.
    """
    a = a_start.copy()
    best_score = score_start
    best_a = a.copy()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        k_max = int(np.argmax(conv))

        # build the sub‑gradient
        grad = np.zeros_like(a)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        i_idx = np.arange(low, high + 1, dtype=int)
        j_idx = k_max - i_idx
        grad[i_idx] = 2.0 * a[j_idx]

        # step size η_t = 0.5 / sqrt(t)
        eta = 0.5 / np.sqrt(it + 1.0)

        cand = a - eta * grad
        cand = _project_simplex(cand)

        sc = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            a = cand
            print(f"[{time.time() - start_time:.2f}s] Subgrad it={it:,} → score={sc:.6f}")
            continue

        # even if not improving, move on (the projection may still change a)
        a = cand

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget, prints
    intermediate scores and finally returns the best vector found.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.05

    # ------------------------------------------------------------------
    # Problem size – we keep a moderate size (4096) for comparability.
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)          # for a*a

    # ------------------------------------------------------------------
    # 0️⃣  Baseline – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0.5️⃣  If a global starter exists, try it
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size))
            starter = np.maximum(starter, 0.0)
            if starter.sum() > 0.0:
                starter = starter / starter.sum()
                sc = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Random generator
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Random Beta‑shape phase (≈30 % of total budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * (deadline - start_time)
    if time.time() < phase1_deadline:
        best_a, best_score = _search_beta_shapes(
            best_a, best_score, n, L_fft, phase1_deadline, rng
        )

    # ------------------------------------------------------------------
    # 2️⃣  Dirichlet draws (≈30 % of total budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.60 * (deadline - start_time)
    if time.time() < phase2_deadline:
        best_a, best_score = _search_dirichlet(
            best_a, best_score, n, L_fft, phase2_deadline, rng
        )

    # ------------------------------------------------------------------
    # 3️⃣  Precise sub‑gradient refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient refinement")
        best_a, best_score = _subgradient_descent(
            best_a, best_score, n, L_fft, deadline
        )
        print(f"[{time.time() - start_time:.2f}s] Sub‑gradient phase finished")

    # ------------------------------------------------------------------
    # Final normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # plain Python list as required by the evaluator
    return [float(v) for v in best_a]
