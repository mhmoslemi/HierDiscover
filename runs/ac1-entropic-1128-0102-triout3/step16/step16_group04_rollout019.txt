"""
Probabilistic step‑function optimizer.

The objective is               score = 2·n·max(conv(a,a)) / (∑a)² .
Since the objective is scale‑invariant we keep the vector on the simplex
(∑a = 1) and minimise the peak of the self‑convolution.

Approach
--------
1. **Warm‑start** – many Dirichlet draws (different concentrations) give a
   diverse set of feasible vectors; the best one is kept.
2. **Hard‑max descent** – repeatedly locate the index `k` attaining the current
   maximum of the convolution.  The gradient of the *exact* max is
   `∂max/∂a_i = 2·a_{k‑i}` (zero outside the domain).  An exponentiated‑gradient
   (mirror descent on the simplex) step reduces the two entries that most
   contribute to the peak.  The stepsize follows `η_t = η₀/√t`.  Small random
   jitter is added occasionally to escape plateaus.
3. **Final polishing** – a short convex mixture with the uniform distribution
   followed by a random‑pair hill‑climb (larger transfer steps) squeezes the
   last bits of improvement.

All helper functions are top‑level; the routine respects a 1000 s budget
(the caller supplies a smaller value if desired).  The returned list contains
non‑negative floats that sum to one (the evaluator will re‑normalise them
anyway).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT (full length 2·len(a)‑1).
    """
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Exact score used by the evaluator (sum is 1 → 2·n·max(conv)).
    Returns the scalar score.
    """
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max()), conv


def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Mix with uniform and perform a slightly more aggressive hill‑climb.
    """
    best_a = a.copy()
    best_score, _ = _objective_score(best_a, n, L_fft)
    uniform = np.full(n, 1.0 / n)

    # convex mixture with uniform (coarse grid)
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc, _ = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → {sc:.6f}")

    # random pair hill‑climb – larger transfers
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, min(best_a[i], 0.02))   # up to 2 % of a_i
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc, _ = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb i={i} j={j} δ={delta:.5f} → {sc:.6f}")

    return best_a, best_score
