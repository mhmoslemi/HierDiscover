"""
Hybrid optimizer with a linear‑programming refinement.

The routine works in four stages (all respecting the total time budget).

0️⃣  Initialise – use the global starter ``height_sequence_1`` if present,
    otherwise fall back to the uniform distribution.

1️⃣  Linear‑programming refinement – given the current vector f,
    solve the LP

        maximise   Σ b_i
        subject to  (f * b)_k ≤ M  for all k          (M = max(f*f)),
                    b_i ≥ 0

    The LP pushes mass into indices where the weighted convolution with f
    is small.  After solving we obtain a vector g = b/Σb and replace f
    by a convex combination (τ=0.5) of f and g.  A few iterations are
    performed (at most six or until the time runs out).

2️⃣  Exponentiated‑gradient (mirror‑descent) on the simplex – the same
    sub‑gradient routine as in the reference implementation, but now
    started from the LP‑improved point.

3️⃣  Small hill‑climbing moves with simulated‑annealing acceptance,
    to fine‑tune the distribution.

4️⃣  Final convex mixing with the uniform distribution using a golden‑
    section search over λ∈[0,1].

All phases print intermediate scores; the best sequence found is returned
as a plain Python list of floats.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _maybe_use_global_starter(n: int) -> tuple[np.ndarray, float]:
    """If a global variable ``height_sequence_1`` exists, try to use it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


# ----------------------------------------------------------------------
# 2️⃣  Exponentiated‑gradient descent (mirror descent)
# ----------------------------------------------------------------------
def _expgrad_descent(a0: np.ndarray,
                     n: int,
                     L_fft: int,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    rng = np.random.default_rng()
    it = 0
    last_improve_it = 0
    restart_interval = 300_000

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] Exp‑grad it={it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        eta = 0.8 / math.sqrt(it)
        exp_factor = np.exp(-eta * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a_new = a * exp_factor
        s = a_new.sum()
        if s <= 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s

        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_new
        else:
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 1e-4 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new

        if it - last_improve_it > restart_interval:
            perturb = rng.normal(scale=0.01, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            last_improve_it = it
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climbing
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_fft: int,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a, L_fft).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.7:
            i = np.argmax(a)
        else:
            i = rng.integers(n)
        if rng.random() < 0.7:
            j = np.argmin(a)
        else:
            j = rng.integers(n)

        if i == j or a[i] <= 1e-14:
            continue

        eps = min(0.02 * a[i], 1e-4)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        new_score = 2.0 * n * _conv_self(a_new, L_fft).max()
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
        else:
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with uniform (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.05  # small safety margin

    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # 0️⃣ initialise
    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None:
        best_seq = starter_seq
        best_score = starter_sc
        print(f"[0.00s] Global starter score={best_score:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective_score(best_seq, n)
        print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 1️⃣ Linear‑programming refinement
    # -----------------------------------------------------------------
    max_lp_iters = 6
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        # current max‑convolution bound M
        conv = _conv_self(best_seq, L_fft)
        M = conv.max()
        eps = 1e-15
        nz = np.where(best_seq > eps)[0]
        if nz.size == 0:
            break

        # Build sparse A where (A @ b)_k = (best_seq * b)_k
        data = [np.full(n, best_seq[i]) for i in nz]
        offsets = list(nz)
        A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        # Solve maximise Σ b_i  subject to A b ≤ M, b ≥ 0.
        c = -np.ones(n)                        # minimise –Σb
        b_ub = np.full(2 * n - 1, M)

        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print(f"[{time.time() - start:.2f}s] LP #{lp_iter} failed (status {res.status})")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            print(f"[{time.time() - start:.2f}s] LP #{lp_iter} gave sum_b={sum_b:.6f} ≤ 1 → stop LP")
            break

        g = b / sum_b  # normalise to sum = 1
        # aggressive mixing
        tau = 0.5
        new_seq = (1.0 - tau) * best_seq + tau * g
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()
        new_score = _objective_score(new_seq, n)
        print(f"[{time.time() - start:.2f}s] LP #{lp_iter}: sum_b={sum_b:.6f}, score={new_score:.6f}")

        if new_score < best_score:
            best_seq, best_score = new_seq, new_score
            print(f"[{time.time() - start:.2f}s] LP improvement → {best_score:.6f}")
        else:
            # try a smaller step if the aggressive one failed
            tau = 0.2
            new_seq = (1.0 - tau) * best_seq + tau * g
            new_seq /= new_seq.sum()
            new_score = _objective_score(new_seq, n)
            if new_score < best_score:
                best_seq, best_score = new_seq, new_score
                print(f"[{time.time() - start:.2f}s] LP mild improvement → {best_score:.6f}")
            else:
                print(f"[{time.time() - start:.2f}s] No improvement from LP #{lp_iter}")
                break
        lp_iter += 1

    # -----------------------------------------------------------------
    # 2️⃣ Exponentiated‑gradient descent on the simplex
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        phase2_deadline = time.time() + remaining * 0.6
        print(f"[{time.time() - start:.2f}s] Starting exponentiated‑gradient descent")
        best_seq, best_score = _expgrad_descent(
            best_seq, n, L_fft, start, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start:.2f}s] After exp‑grad best ≈ {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3️⃣ Hill‑climbing fine tuning
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        phase3_deadline = time.time() + remaining * 0.7
        print(f"[{time.time() - start:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, n, L_fft, start, phase3_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start:.2f}s] After hill‑climb best ≈ {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4️⃣ Final mixing with the uniform distribution
    # -----------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start, best_seq, best_score
        )

    # -----------------------------------------------------------------
    # Normalise and return
    # -----------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
