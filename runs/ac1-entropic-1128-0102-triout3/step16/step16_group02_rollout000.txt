The submitted `propose_candidate` searches for a non‑negative step function that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` denotes the (zero‑padded) discrete convolution of `a` with itself.
The algorithm runs completely inside the 1000 s time budget and consists of four
phases:

1. **Fast parametric families** – geometric, power‑law, symmetric/skewed
   exponentials and a two‑block construction.  All parameters are swept on a
   coarse grid; any improvement is recorded.

2. **Massive random Dirichlet sampling** – draws a large number of random
   points on the simplex (different concentration parameters) and keeps the
   best one.

3. **Repeated projected sub‑gradient descent** – the current best vector is
   refined by a simple sub‑gradient method with a diminishing step size.
   The descent is run many times from freshly re‑initialised points (random
   jitter around the best) which lets the search escape local minima.

4. **Simulated‑annealing hill‑climb** – a classic Metropolis walk on the simplex
   that repeatedly transfers a random fraction of mass from a “high‑gradient”
   index to a “low‑gradient” index.  The temperature drops linearly; uphill
   moves are occasionally accepted, allowing the method to overcome small
   barriers.

The whole search works with **\(n=1024\)** (instead of 4096).  
For the optimal shape the quantity `max(a*a)` behaves like *c / n*,
so the term `2·n` in the objective cancels and the constant *c* – the true
quality measure – is independent of `n`.  
Using a smaller `n` makes each convolution cheap, so many more candidates can
be evaluated within the fixed time budget, dramatically increasing the chance
to beat the previous record (1.507818 → target 1.503).

All helper functions are top‑level; no closures or file I/O are used.
The final vector is normalised to have sum = 1 (the evaluate function enforces a
minimum sum of 0.01) and returned as a plain Python list.

"""
search for a non‑negative step sequence that minimises
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works on n=1024 (fast FFT) and combines
* coarse parametric sweeps,
* massive Dirichlet sampling,
* many short projected sub‑gradient descents (random restarts),
* a final simulated‑annealing hill‑climb.

It stays well under the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, objective, sub‑gradient)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    It averages the contributions of all active shifts (within 1e‑12 of the max).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    # For each active shift compute the contribution a_{k-i}
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / float(active.size)
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 150)          # coarser than before
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo_sum = geo.sum()
        if geo_sum == 0.0:
            continue
        geo /= geo_sum
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-3.0, 3.0, 181):   # extended range
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0.0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_skewed_exponential_break(n, best_seq, best_score, deadline):
    """
    Piecewise exponential with a tunable break point.
    left side (indices < b) uses λ₁, right side (≥ b) uses λ₂.
    """
    breaks = np.linspace(0.2, 0.8, 13)               # fraction of n
    lam_vals = np.logspace(-4, -0.2, 20)            # λ grid
    for bf in breaks:
        b = int(round(bf * n))
        if b <= 0 or b >= n:
            continue
        left_idx = np.arange(b, dtype=np.float64)          # 0 … b‑1
        right_idx = np.arange(b, n, dtype=np.float64)      # b … n‑1
        for lam_left in lam_vals:
            if time.time() >= deadline:
                break
            left_part = np.exp(-lam_left * (b - 1 - left_idx))
            for lam_right in lam_vals:
                seq = np.empty(n, dtype=np.float64)
                seq[:b] = left_part
                seq[b:] = np.exp(-lam_right * (right_idx - b))
                s = seq.sum()
                if s == 0.0:
                    continue
                seq /= s
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] SkewExpBreak b={b} λL={lam_left:.1e} λR={lam_right:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends, with independent block length."""
    for L_frac in np.linspace(0.01, 0.3, 30):
        L = int(round(L_frac * n))
        if L < 1:
            continue
        for w in np.linspace(0.0, 1.0, 21):
            if time.time() >= deadline:
                break
            seq = np.zeros(n, dtype=np.float64)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score

# 3️⃣ Projected sub‑gradient descent (many short restarts)
# ----------------------------------------------------------------------
def _proj_grad_step(a: np.ndarray) -> np.ndarray:
    """One projected sub‑gradient step with a diminishing step size."""
    n = a.shape[0]
    conv = _conv_self(a)
    grad = _subgradient_maxconv(a, conv)
    # direction that respects the simplex (zero‑sum) constraint
    d = -(grad - grad.mean())
    # maximal feasible step size respecting non‑negativity
    idx_neg = np.where(d < 0)[0]
    if idx_neg.size:
        max_step = np.min(-a[idx_neg] / d[idx_neg])
    else:
        max_step = 1.0
    # step length slowly decays with iteration count (here 0.5 of the max)
    step = 0.5 * max_step
    a_new = a + step * d
    # projection onto non‑negative orthant (clip) and renormalise
    a_new = np.clip(a_new, 0.0, None)
    a_new_sum = a_new.sum()
    if a_new_sum == 0.0:
        # fall back to uniform – should never happen
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= a_new_sum
    return a_new


def _local_descent(start_seq, deadline, best_seq, best_score, max_iters=200):
    """
    Run a short projected sub‑gradient descent from `start_seq`.
    Stops when the deadline is reached.
    """
    a = start_seq.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        a = _proj_grad_step(a)
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] GD it={it} score={cur_score:.6f}")
            # early exit if we already beat the target
            if best_score <= 1.503:
                return best_seq, best_score
    return best_seq, best_score


def _run_multiple_descent(n, best_seq, best_score, deadline):
    """
    Perform many short gradient runs, each starting from a jittered copy of
    the current best point.  The time budget is split evenly among the runs.
    """
    # we aim for roughly 30‑40 restarts (or as many as time permits)
    min_per_restart = 0.5   # seconds
    while time.time() < deadline - min_per_restart:
        # jitter the best point a little
        rng = np.random.default_rng()
        jitter = rng.normal(scale=0.02, size=n)
        cand = best_seq + jitter
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand = np.full(n, 1.0 / n)
        else:
            cand /= s
        # run a very short descent (≈ 30 iterations or time‑limited)
        sub_deadline = time.time() + min_per_restart
        best_seq, best_score = _local_descent(cand, sub_deadline,
                                              best_seq, best_score,
                                              max_iters=40)
        # break if we have already reached the target
        if best_score <= 1.503:
            break
    return best_seq, best_score

def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = _objective_score(a)
    cur_a = a.copy()
    total = max(deadline - time.time(), 1e-6)
    start = time.time()
    it = 0
    no_improve = 0
    # pre‑compute an initial gradient to guide source selection
    grad = _subgradient_maxconv(a, _conv_self(a))

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total
        # temperature falls linearly from 0.1·cur_score to 0
        T = (1.0 - frac) * cur_score * 0.1

        # refresh gradient occasionally (every 200 steps)
        if it % 200 == 0:
            grad = _subgradient_maxconv(a, _conv_self(a))

        # source is chosen proportional to positive gradient magnitude
        prob = np.maximum(grad, 0.0)
        prob_sum = prob.sum()
        if prob_sum > 0 and rng.random() < 0.7:
            src = rng.choice(n, p=prob / prob_sum)
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a random fraction of the source mass
        max_frac = 0.3
        frac_transfer = rng.random() * max_frac
        eps = a[src] * frac_transfer
        if eps < 1e-15:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # evaluate
        new_score = _objective_score(a_prop)
        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} score={best_score:.6f}")
                if best_score <= 1.503:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # if we have been stuck for a long time, make a bigger random shake‑up
        if no_improve > 5000:
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps2 = a[i1] * (rng.random() * max_frac)
            a[i1] -= eps2
            a[i2] += eps2
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            no_improve = 0

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes the evaluation score.

    The routine works on n=1024 (fast FFT) and performs
    * parametric sweeps,
    * massive Dirichlet sampling,
    * many short projected gradient descents (random restarts),
    * a final simulated‑annealing polishing step.

    The best vector found (normalised to sum = 1) is returned as a flat Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 1024                       # smaller n → many more evaluations
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optional global starter (if the notebook defines it)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep (≈ 200 s)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_skewed_exponential_break(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – massive Dirichlet draws (≈ 200 s)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 180.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # Phase 3 – many short projected gradient descents (≈ 250 s)
    # ------------------------------------------------------------------
    grad_deadline = deadline - 80.0
    if time.time() < grad_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting many short gradient descents")
        best_seq, best_score = _run_multiple_descent(n, best_seq, best_score, grad_deadline)
        print(f"[{time.time() - start_time:.2f}s] After GD best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final simulated annealing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final SA score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
