def _directed_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    step_factor = 0.5
    attempts_without_improve = 0
    while time.time() < deadline:
        # compute gradient / subgradient
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)
        src = grad.argmax()
        dst = grad.argmin()
        if a[src] < 1e-12:
            # cannot move from src, try random src
            src = np.random.choice(n)
            continue
        eps = a[src] * step_factor
        if eps < 1e-12:
            step_factor *= 0.5
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        sc = _objective_score(a_prop)
        if sc < best_score:
            best_score, best_seq = sc, a_prop.copy()
            a = a_prop
            attempts_without_improve = 0
            # Possibly increase step_factor back up a bit
            step_factor = min(step_factor*1.2, 0.9)
            print(f"[{time.time() - start_time:.2f}s] DirGrad src={src} dst={dst} eps={eps:.3e} score={sc:.6f}")
        else:
            attempts_without_improve += 1
            step_factor *= 0.9
            if attempts_without_improve > 200:
                # random restart: small jitter
                jitter = 1e-5
                a = a + np.random.uniform(-jitter, jitter, size=n)
                a = np.clip(a, 0, None)
                a /= a.sum()
                attempts_without_improve = 0
    return best_seq, best_score

"""
A lightweight multi‑phase optimizer for the “step‑function” objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

The routine works with any non‑negative list a (the sum is irrelevant as the
objective is scale‑invariant).  The search proceeds in four time‑controlled
stages:

1️⃣  A broad parametric sweep over several families (geometric, power,
    (a)symmetric Beta, exponential, two‑block, skewed‑exponential, a new
    arcsine‑U shape and a sine‑power shape).

2️⃣  Random Dirichlet draws, including sparse support variants.

3️⃣  A deterministic “gradient‑directed” mass‑transfer loop that repeatedly
    moves a fraction of the mass from the index with the largest sub‑gradient
    (i.e. contributing most to the current maximum convolution) to the index
    with the smallest sub‑gradient.  Simple back‑tracking line‑search guarantees
    feasibility and monotone improvement.

4️⃣  A short simulated‑annealing hill‑climb (multiple restarts) for final polishing.

All phases respect a global deadline (default 1000 s).  The best candidate found
is normalised to sum = 1 and returned as a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    The gradient entry i is proportional to the sum of a‑values that pair
    with a_i in the currently active (max‑attaining) convolutions.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat values within a tiny tolerance as active – a slightly larger
    # tolerance helps the search when the maximum plateau is flat.
    tol = 1e-10
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

# 1️⃣ Parametric families (including new arcsine‑U and sine‑power)

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_seq = (a - 1.0) * np.log(idx + 1.0) + (a - 1.0) * np.log(np.maximum(n - idx, 1))
        seq = np.exp(log_seq - np.max(log_seq))
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymBeta α={a:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.1, 5.0, 21)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay: a_i ∝ exp(-λ·min(i, n‑1‑i))."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 120):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends with a tunable mass split."""
    for L_frac in np.linspace(0.01, 0.4, 40):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_skewed_exponential(n, best_seq, best_score, deadline):
    """Piecewise exponential: left side uses λ₁, right side uses λ₂."""
    mid = n // 2
    left_idx = np.arange(mid, dtype=np.float64)
    right_idx = np.arange(mid, n, dtype=np.float64)
    for lam_left in np.logspace(-4, -0.2, 30):
        for lam_right in np.logspace(-4, -0.2, 30):
            if time.time() >= deadline:
                break
            seq = np.empty(n, dtype=np.float64)
            seq[:mid] = np.exp(-lam_left * left_idx)
            seq[mid:] = np.exp(-lam_right * (n - 1 - right_idx))
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SkewExp λL={lam_left:.1e} λR={lam_right:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_arcsine_u(n, best_seq, best_score, deadline):
    """
    U‑shaped “arcsine” family:
        a_i ∝ 1 / sqrt( (i+δ)·(n‑1‑i+δ) )^p
    The exponent p controls how sharply mass piles at the ends.
    """
    idx = np.arange(n, dtype=np.float64)
    eps = 1e-12
    for p in np.linspace(0.2, 3.0, 80):
        if time.time() >= deadline:
            break
        seq = 1.0 / np.sqrt((idx + eps) * (n - 1 - idx + eps)) ** p
        s = seq.sum()
        if s == 0 or not np.isfinite(s):

        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] ArcsineU p={p:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_sine_power(n, best_seq, best_score, deadline):
    """
    Sine‑power family (bell‑shaped).  Although it concentrates mass
    in the centre, for completeness we sweep it – occasionally it
    beats other families when combined later.
    """
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(0.5, 10.0, 100):
        if time.time() >= deadline:

        seq = np.sin(np.pi * (idx + 0.5) / n) ** p
        s = seq.sum()
        if s == 0:

        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:

            print(f"[{time.time() - start_time:.2f}s] SinePower p={p:.2f} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 1500
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # concentration spanning many orders of magnitude
        logc = rng.uniform(-4.0, 1.0)      # 1e‑4 … 10
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3e} score={sc:.6f}")

    # sparse versions – pick a small support and put Dirichlet mass on it
    while time.time() < deadline:
        support = rng.integers(2, min(200, n // 2))
        idx = rng.choice(n, size=support, replace=False)
        logc = rng.uniform(-5.0, 0.5)          # very spiky concentrations
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=support)
        a /= a.sum()
        seq = np.zeros(n)
        seq[idx] = a
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Sparse Dirichlet sup={support} c={c:.3e} score={sc:.6f}")

    return best_seq, best_score

# 3️⃣ Gradient‑directed mass transfer
# ----------------------------------------------------------------------
def _directed_gradient_descent(a0, deadline, best_seq, best_score):
    """
    Repeatedly move a fraction of the mass from the index with the largest
    sub‑gradient to the one with the smallest sub‑gradient.
    A simple back‑tracking line‑search on the transferred amount guarantees
    monotone improvement.
    """
    a = a0.copy()
    n = a.shape[0]
    step_factor = 0.5          # start with aggressive transfers
    fails = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            break

        # maximum transferable amount from src whilst staying non‑negative
        max_transfer = a[src]
        if max_transfer <= 1e-14:
            # source exhausted – pick a random new source
            src = np.random.randint(n)
            max_transfer = a[src]
            if max_transfer <= 1e-14:
                break

        eps = max_transfer * step_factor
        # back‑track until improvement is observed
        improved = False
        while eps > 1e-16:
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            sc = _objective_score(a_prop)
            if sc < best_score:
                best_score, best_seq = sc, a_prop.copy()
                a = a_prop
                improved = True
                fails = 0
                # after a success we may slightly loosen the step factor
                step_factor = min(step_factor * 1.1, 0.9)
                print(f"[{time.time() - start_time:.2f}s] GradMove src={src} dst={dst} eps={eps:.2e} score={sc:.6f}")
                break
            else:
                eps *= 0.5

        if not improved:
            fails += 1
            step_factor *= 0.7
            if fails > 300:
                # random jitter to escape a flat region
                jitter = 1e-6
                a = a + np.random.uniform(-jitter, jitter, size=n)
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                fails = 0
                step_factor = 0.5

    return best_seq, best_score

def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = best_score

    total = max(deadline - time.time(), 1e-6)
    start = time.time()
    it = 0
    no_improve = 0

    # pre‑compute a gradient to bias source selection
    grad = _subgradient_maxconv(a, _conv_self(a))

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / total
        # temperature decays linearly; initial temperature ≈ 0.2 % of the score
        T = (1.0 - frac) * cur_score * 0.002

        # refresh gradient occasionally
        if it % 200 == 0:
            grad = _subgradient_maxconv(a, _conv_self(a))

        prob = np.maximum(grad, 0)
        prob_sum = prob.sum()
        if prob_sum > 0:
            src = rng.choice(n, p=prob / prob_sum)
        else:
            src = rng.integers(n)

        dst = rng.integers(n)
        if src == dst:
            continue

        # fraction of source mass to move – schedule shrinks over time
        max_frac = 0.3 * (1.0 - frac) + 0.02
        eps = a[src] * rng.random() * max_frac
        if eps <= 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # sum unchanged, non‑negativity guaranteed
        score_new = _objective_score(a_prop)
        delta = score_new - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_prop
            cur_score = score_new
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} score={best_score:.6f}")
                if best_score <= 1.503:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 6000:
            # occasional shake‑up to escape stagnation
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps2 = a[i1] * rng.random() * max_frac
            a[i1] -= eps2
            a[i2] += eps2
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective_score(a)
            no_improve = 0

    return best_seq, best_score

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet (allocate ~20 % of budget)

    phase_deadline = deadline - max(0.20 * budget_s, 100.0)
    if time.time() < phase_deadline:

    # --------------------------------------------------------------
    # 3️⃣ Gradient‑directed mass transfer (allocate ~20 % of budget)
    # --------------------------------------------------------------
    phase_deadline = deadline - max(0.20 * budget_s, 80.0)
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting gradient‑directed descent")
        best_seq, best_score = _directed_gradient_descent(best_seq, phase_deadline,

    # --------------------------------------------------------------
    # 4️⃣ Simulated‑annealing hill‑climb (use all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:

        best_seq, best_score = _simulated_annealing(best_seq, deadline,

    best_seq = np.maximum(best_seq, 0.0)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
