"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

1️⃣  **Initial baselines** – uniform distribution and, if available,
    the global variable ``height_sequence_1``.

2️⃣  **Parametric sweep** – dense grid of geometric × power shapes,
    mixed with a uniform component.  The best parameters are stored.

3️⃣  **Refined sweep** – a finer grid centred around the best parameters
    found in step 2.

4️⃣  **Smooth‑max gradient descent** – continuation on an increasing
    temperature β.  The gradient of the smooth‑max surrogate is obtained
    via FFT and the iterate is projected onto the probability simplex.

5️⃣  **Simulated‑annealing hill‑climb** – the remaining budget is spent
    on a classic SA that repeatedly transfers a random fraction of mass
    from one index to another.  Moves are accepted with the Metropolis
    rule; the temperature and the transferred fraction are linearly
    cooled.  The best ever sequence is memorised.

6️⃣  **Final polish** – a very short run of the smooth‑max descent to
    tighten the last few digits.

All phases respect the 1000 s budget, keep the vector on the simplex,
and finally return the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # Compute in log‑space for stability
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution (lam∈[0,1])."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Smooth‑max utilities (used in the gradient phase)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max probabilities p_k = exp(beta*conv_k) / Σexp(...)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k * conv_k w.r.t. a.
    Returns g_i = 2 * Σ_j a_j * p_{i+j}.
    Computed by FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    corr = np.fft.irfft(Af * Pf, n=L_grad)
    # Extract the central n entries (see derivation in the comment above)
    start = n - 1
    g = corr[start:start + n][::-1]
    return 2.0 * g


# ----------------------------------------------------------------------
# Simulated‑annealing hill climb
# ----------------------------------------------------------------------


def _simulated_annealing(a_start: np.ndarray,
                         L_fft: int,
                         deadline: float,
                         start_time: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer SA that respects the simplex.
    Moves a small fraction of mass from a source index to a destination.
    Acceptance follows the Metropolis rule with a linearly cooling temperature.
    """
    a = a_start.copy()
    cur_score = _eval_fast(a, L_fft)
    n = a.shape[0]

    # SA schedule parameters
    total = max(deadline - time.time(), 1e-6)
    T0 = max(0.1 * best_score, 1e-4)          # initial temperature
    frac_start = 0.20                         # max fraction transferred initially
    frac_end = 0.0005                         # min fraction at the end
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        # linear cooling for both temperature and move size
        t_frac = (deadline - time.time()) / (deadline - start_time)
        T = max(T0 * t_frac, 1e-12)
        max_frac = frac_end + (frac_start - frac_end) * t_frac

        # pick a source proportionally to its mass (more likely to move larger amounts)
        src = rng.choice(n, p=a)
        if a[src] <= 1e-18:
            continue
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # amount to move
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _eval_fast(a_prop, L_fft)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_prop
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_seq = a_prop.copy()
                print(f"[{elapsed:.2f}s] SA it={it:,} score={new_score:.6f}")

        # occasional restart from the global best to avoid drifting too far
        if it % 200000 == 0:
            a = best_seq.copy()
            cur_score = best_score

    return best_seq, best_score

            # step size proportional to 1/‖g‖ (simple but effective)
            g_norm = np.linalg.norm(g) + 1e-12
            step = 0.40 / g_norm

            a_cand = a - step * g
            a_cand = _project_simplex(a_cand)

            cand_score = _eval_fast(a_cand, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_cand.copy()
                print(f"[{time.time() - start_time:.2f}s] β={beta} it={it} score={cand_score:.6f}")

            # accept if it improves the *current* iterate
            if cand_score < cur_score:
                a = a_cand
                cur_score = cand_score
            else:
                # occasional uphill move (temperature‑like acceptance)
                prob = np.exp(-(cand_score - cur_score) * 4.0 / (np.sqrt(it) + 1.0))
                if rng.random() < prob:
                    a = a_cand
                    cur_score = cand_score

            # tiny jitter every few thousand steps
            if it % 3000 == 0:
                eps = 1e-4 * rng.random(n) * a
                a = _project_simplex(a + eps)

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing hill‑climb (use up the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:  # keep ~20 s for a final polish
        best_seq, best_score = _simulated_annealing(
            best_seq, L_fft, deadline - 20.0, start_time,
            best_seq, best_score, rng
        )

    # ------------------------------------------------------------------
    # 5️⃣  Final short smooth‑max polish (≈15 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = best_score
        beta = 1024  # high temperature → close to true max
        end = deadline
        it = 0
        while time.time() < end:
            it += 1
            conv = _conv_self_fft(a, L_fft)
            p = _softmax_weights(conv, float(beta))
            g = _grad_softmax(a, p, L_grad)
            g_norm = np.linalg.norm(g) + 1e-12
            step = 0.25 / g_norm
            a_cand = a - step * g
            a_cand = _project_simplex(a_cand)
            cand_score = _eval_fast(a_cand, L_fft)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_cand.copy()
                print(f"[{time.time() - start_time:.2f}s] final polish it={it} score={cand_score:.6f}")

            if cand_score < cur_score:
                a = a_cand
                cur_score = cand_score
            else:
                # very low probability of accepting a worse move at this stage
                prob = np.exp(-(cand_score - cur_score) * 8.0 / (np.sqrt(it) + 1.0))
                if rng.random() < prob:
                    a = a_cand
                    cur_score = cand_score

            if it % 4000 == 0:
                eps = 1e-4 * rng.random(n) * a
                a = _project_simplex(a + eps)

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return as a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
