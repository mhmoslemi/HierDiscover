def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)−1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_k = exp(α·conv_k) / Σ exp(α·conv)
    in a numerically stable way.

    # shift by max for stability

    m = shifted.max()
    exp_shifted = np.exp(shifted - m)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, L_corr: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. a:
        ∇ = 2 * corr(w, a)   (valid correlation, length = len(a))

    The correlation is performed via FFT for speed.

    return grad
