"""
Propose a step function (non‑negative vector on the simplex) that
minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

using a multi‑phase search:

1️⃣  A large batch of random Dirichlet draws (wide range of concentrations)
    to obtain a good starting point.

2️⃣  Hybrid simulated annealing:
    * deterministic move that transfers mass from the index with the
      largest sub‑gradient to the one with the smallest,
    * occasional random moves for extra exploration,
    * temperature and transfer‑fraction schedules that decay slowly
      (more aggressive than the reference implementation).

3️⃣  Smooth‑approximation gradient descent.
    A log‑sum‑exp (soft‑max) approximation of the L∞‑norm makes the
    objective differentiable.  For a series of increasing smoothing
    parameters α we perform projected gradient steps using the exact
    gradient of the smoothed objective.

4️⃣  Final fine‑tuning with a projected‑gradient descent on the true
    max‑conv objective (identical to the reference “PGD” phase).

The algorithm respects the 1000 s budget; intermediate best results are
printed.  The returned sequence is a plain Python list of non‑negative
floats that sum to 1 (the scaling is irrelevant for the evaluation
function because the objective is homogeneous).

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (average over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _random_dirichlet_trials(n: int, max_trials: int, deadline: float):
    """Return the best vector seen among a number of Dirichlet draws."""
    rng = np.random.default_rng()
    best_seq = None
    best_score = float("inf")
    for _ in range(max_trials):
        if time.time() >= deadline:
            break
        # Sample concentration hyper‑parameter on a log‑scale
        conc_log = rng.uniform(-3.0, 0.7)   # 10^{-3} … 10^{0.7}
        c = 10.0 ** conc_log
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4e} score={sc:.6f}")
    return best_seq, best_score

def _simulated_annealing_hybrid(a0: np.ndarray, cur_score: float, deadline: float):
    """Anneal by moving mass between indices; mixes deterministic & random moves."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # start with a relatively high temperature and decay slowly
    T = max(cur_score * 0.05, 1e-8)
    decay = 0.99995
    prob_random = 0.25          # probability of a completely random move
    max_frac_start = 0.45       # early moves may shift up to 45 % of source mass
    max_frac_end = 0.008        # late moves become very fine‑grained

    while time.time() < deadline:
        # sub‑gradient used only for the deterministic move
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        if rng.random() < prob_random:
            src = int(rng.integers(n))
            dst = int(rng.integers(n))
        else:
            src = int(np.argmax(grad))
            dst = int(np.argmin(grad))

        # linearly decay the allowed transfer fraction
        t_rem = deadline - time.time()
        total_rem = deadline - start_time
        frac = max_frac_end + (max_frac_start - max_frac_end) * max(t_rem / total_rem, 0.0)

        delta = a[src] * rng.random() * frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        if cand[src] < 0.0:
            continue

        cand_score = _objective_score(cand)
        diff = cand_score - cur_score
        # Metropolis acceptance
        if diff <= 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA hybrid new best {best_score:.6f}")

        T = max(T * decay, 1e-12)

    return best_a, best_score

def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp (soft‑max) approximation of max(conv)
    w.r.t. a.  Returns a vector of length n.
    """
    n = a.shape[0]
    # stable soft‑max on the convolution values
    max_c = conv.max()
    exp_alpha = np.exp(alpha * (conv - max_c))
    Z = np.sum(exp_alpha)
    w = exp_alpha / Z                     # length 2·n‑1

    # correlation a ⋆ w  (grad[t] = 2 * Σ_i a_i * w[t+i])
    rev_w = w[::-1]                       # reverse for correlation via convolution
    L = _next_pow2(len(a) + len(rev_w) - 1)
    A = np.fft.rfft(a, n=L)
    R = np.fft.rfft(rev_w, n=L)
    conv_all = np.fft.irfft(A * R, n=L)[: len(a) + len(rev_w) - 1]

    idx = (len(rev_w) - 1) - np.arange(n)  # extract the required slice
    grad = 2.0 * conv_all[idx]
    return grad


def _smooth_gradient_descent(a0: np.ndarray, cur_score: float, deadline: float):
    """
    Perform projected gradient descent on the smoothed objective.
    A schedule of increasing α drives the approximation towards the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    # α values – start smooth, finish sharp
    alphas = [10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
    step = 0.8               # initial step length in the gradient direction

    for alpha in alphas:
        if time.time() >= deadline:
            break
        print(f"[{time.time() - start_time:.2f}s] Smooth‑GD start α={alpha}")

        while time.time() < deadline:
            conv = _conv_self(a)
            grad = _smooth_gradient(a, conv, alpha)

            improved = False
            # back‑tracking line search (few candidate step scalings)
            for factor in (1.0, 0.5, 0.25, 0.125):
                cand = a - step * factor * grad
                cand = _project_simplex(np.maximum(cand, 0.0))
                cand_score = _objective_score(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    improved = True
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = a.copy()
                        print(f"[{time.time() - start_time:.2f}s] Smooth‑GD α={alpha} new best {best_score:.6f}")
                    break

            if not improved:
                step = max(step * 0.5, 1e-12)
            else:
                step = min(step * 1.07, 2.0)   # let the step grow a little after progress

    return best_a, best_score

                    print(f"[{time.time() - start_time:.2f}s] External starter score = {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random Dirichlet initialisers (≈50 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 50.0)
    if time.time() < dirichlet_deadline:
        seq, sc = _random_dirichlet_trials(n, max_trials=5000, deadline=dirichlet_deadline)
        if seq is not None and sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] After Dirichlet best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Hybrid Simulated Annealing (≈600 s)
    # ------------------------------------------------------------------
    anneal_deadline = min(deadline, start_time + 650.0)
    if time.time() < anneal_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hybrid simulated annealing")
        best_seq, best_score = _simulated_annealing_hybrid(best_seq, best_score, anneal_deadline)
        print(f"[{time.time() - start_time:.2f}s] Annealing finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑approximation gradient descent (≈250 s)
    # ------------------------------------------------------------------
    smooth_deadline = min(deadline, start_time + 900.0)
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑gradient descent")
        best_seq, best_score = _smooth_gradient_descent(best_seq, best_score, smooth_deadline)
        print(f"[{time.time() - start_time:.2f}s] Smooth‑GD finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final projected‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final projected‑gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final PGD finished, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return (protect against tiny rounding errors)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
