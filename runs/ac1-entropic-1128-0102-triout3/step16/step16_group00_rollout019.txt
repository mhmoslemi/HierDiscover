"""
Search for a non‑negative step function (probability vector) that minimises
``evaluate_sequence``.  The routine combines several cheap ideas that were not
used before:

* a fast sweep over a two‑parameter power‑law family a_i ∝ (i+δ)^‑p
  (the classic √‑law gives a ≈ 1.57, but a tiny shift δ and a perturbed exponent
  can lower the bound);
* many more random Dirichlet draws than the original code;
* a deterministic Frank–Wolfe style refinement that moves mass from the
  index with the largest sub‑gradient to the one with the smallest,
  exploiting an exact line‑search over a few transfer fractions;
* a single linear‑programming improvement step (max‑conv ≤ M) that finds a
  direction g obeying the current worst‑case convolution bound and then mixes
  the current vector towards g;
* a final projected‑gradient descent (the same routine that the reference
  solution used) as a safety net.

All heavy work (convolution, sub‑gradient, simplex projection) is performed
with FFTs and O(n) operations, and the whole search respects the 1000 s
budget.  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1, so denominator equals 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Parameterised power‑law family
# ----------------------------------------------------------------------
def _generate_powerlaw(n: int, p: float, delta: float) -> np.ndarray:
    """
    a_i ∝ (i + delta + 1)^{-p}, normalised so that sum = 1.
    The +1 avoids a zero denominator when delta = 0.
    """
    idx = np.arange(n, dtype=np.float64)
    a = (idx + delta + 1.0) ** (-p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Deterministic Frank–Wolfe style refinement
# ----------------------------------------------------------------------
def _local_refine(a0: np.ndarray, deadline: float):
    """
    Repeatedly move a small amount of mass from the index with the largest
    sub‑gradient to the index with the smallest, trying a handful of transfer
    fractions and keeping any improvement.  Stops when the time budget expires
    or no improvement can be made.
    """
    a = a0.copy()
    n = a.shape[0]
    best_score = _objective_score(a)
    print(f"[{time.time() - start_time:.2f}s] Local‑refine start score {best_score:.6f}")

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        max_delta = a[src]
        improved = False
        # try a few transfer fractions (larger first)
        for frac in (0.5, 0.25, 0.125, 0.0625, 0.03125):
            delta = max_delta * frac
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            # sanity check – numerical noise may produce tiny negatives
            if np.min(cand) < -1e-12:
                continue
            sc = _objective_score(cand)
            if sc < best_score - 1e-12:
                a = cand
                best_score = sc
                improved = True
                print(f"[{time.time() - start_time:.2f}s] Refine improved to {best_score:.6f}")
                break
        if not improved:
            # no improvement with any fraction – give up
            break
    return a, best_score


# ----------------------------------------------------------------------
# One LP‑based improvement step
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, cur_score: float, deadline: float):
    """
    Solve a tiny linear programme:
        maximise   sum(g_i)
        s.t.       (a * g)[k] ≤ M   for the most restrictive shifts k
                   g_i ≥ 0
    where M = max(conv(a,a)).  Then mix a towards the normalised solution g.
    """
    n = a.shape[0]
    M = cur_score / (2.0 * n)           # because cur_score = 2·n·max(conv)
    conv = _conv_self(a)

    # keep only the shifts with the largest convolution values
    top_k = np.argsort(-conv)[:200]     # 200 most critical constraints
    rows, cols, data = [], [], []
    for row_id, k in enumerate(top_k):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            rows.append(row_id)
            cols.append(j)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(len(top_k), n))
    b = np.full(len(top_k), M)
    c = -np.ones(n)                     # maximise sum(g)  → minimise −sum(g)
    bounds = [(0, None)] * n

    # solve the LP (highs is fast and accepts sparse matrices)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return a, cur_score

    g = res.x
    if g.sum() == 0.0:
        return a, cur_score
    g = g / g.sum()                     # normalise to a probability vector

    # mix a little towards g
    mix = 0.2
    new_a = (1.0 - mix) * a + mix * g
    new_score = _objective_score(new_a)
    if new_score < cur_score:
        print(f"[{time.time() - start_time:.2f}s] LP step improved to {new_score:.6f}")
        return new_a, new_score
    return a, cur_score


# ----------------------------------------------------------------------
# Final projected‑gradient descent (identical to the reference version)
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0: np.ndarray, cur_score: float, deadline: float):
    """Fine‑tune the vector by moving opposite to the sub‑gradient and re‑projecting."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = cur_score
    step = 0.5
    while time.time() < deadline:
        conv = _conv_self(a)
        cur_score = _objective_score(a)
        grad = _subgradient_maxconv(a, conv)

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_simplex(np.maximum(cand, 0.0))
            cand_score = _objective_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] PGD new best {best_score:.6f}")
                break
        if not improved:
            step = max(step * 0.5, 1e-10)
        else:
            step = min(step * 1.05, 1.0)
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Locate a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list) found within the time budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    n = 4096
    best_seq = np.full(n, 1.0 / n)
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (e.g. from a previous run)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter improved to {best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 0️⃣  Fast sweep over the power‑law family (small n, then scale up)
    # --------------------------------------------------------------
    n_small = 1024
    powerlaw_deadline = min(deadline, start_time + 30.0)   # 30 s budget
    if time.time() < powerlaw_deadline:
        rng = np.random.default_rng()
        best_small_score = float("inf")
        for _ in range(2000):
            if time.time() >= powerlaw_deadline:
                break
            p = rng.uniform(0.35, 0.65)          # exponent near ½
            delta = rng.uniform(0.0, 200.0)      # shift to regularise the singularity
            a_small = _generate_powerlaw(n_small, p, delta)
            sc_small = _objective_score(a_small)
            if sc_small < best_small_score:
                best_small_score = sc_small
                # evaluate the same parameters at full resolution
                a_full = _generate_powerlaw(n, p, delta)
                sc_full = _objective_score(a_full)
                if sc_full < best_score:
                    best_seq, best_score = a_full.copy(), sc_full
                    print(f"[{time.time() - start_time:.2f}s] Power‑law p={p:.4f} δ={delta:.2f} score={sc_full:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Heavier Dirichlet random search (more trials than the reference)
    # --------------------------------------------------------------
    dirichlet_deadline = min(deadline, start_time + 120.0)   # another two minutes
    if time.time() < dirichlet_deadline:
        rng = np.random.default_rng()
        while time.time() < dirichlet_deadline:
            conc_log = rng.uniform(-4.0, 1.0)      # broader concentration range
            c = 10 ** conc_log
            a = rng.gamma(shape=c, scale=1.0, size=n)
            if a.sum() == 0.0:
                continue
            a /= a.sum()
            sc = _objective_score(a)
            if sc < best_score:
                best_seq, best_score = a.copy(), sc
                print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4f} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Deterministic Frank–Wolfe style refinement
    # --------------------------------------------------------------
    refine_deadline = min(deadline, start_time + 600.0)   # give it up to ~10 min
    if time.time() < refine_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting deterministic refinement")
        best_seq, best_score = _local_refine(best_seq, refine_deadline)
        print(f"[{time.time() - start_time:.2f}s] Refinement finished, best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  One LP‑based improvement step (adds a new feasible direction)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _lp_step(best_seq, best_score, deadline)

    # --------------------------------------------------------------
    # 4️⃣  Final projected‑gradient descent (safety net)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final projected‑gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq, best_score, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final PGD finished, best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0.0:
        best_seq = best_seq / tot
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
