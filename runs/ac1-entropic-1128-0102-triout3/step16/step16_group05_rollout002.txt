def lp_update(f, M, t):
    n = len(f)
    # Build sparse matrix A (conv matrix)
    diags_data = []
    offsets = []
    for d in range(n):
        # diag length = n - d
        diags_data.append(np.full(n - d, f[d]))
        offsets.append(d)
    A = scipy.sparse.diags(diags_data, offsets, shape=(2 * n - 1, n), format='csc')
    # Define LP
    # maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    b = np.full(2 * n - 1, M)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            # fallback uniform
            g = np.full(n, 1.0/n)
            sum_g = 1.0
        # Normalize to sum 1
        g_norm = g / sum_g
        # mixing
        return (1 - t) * f + t * g_norm
    else:
        # fallback to something else
        return f

"""
A mixed optimisation for the step‑function bound.

The routine works on a moderate size (n=1024) to keep each iteration cheap.
Three phases are used:

1. **Exponentiated‑gradient descent** on the max‑self‑convolution.
   The sub‑gradient is computed analytically, a multiplicative update
   (mirror‑descent w.r.t. the KL divergence) is performed and the vector
   is renormalised onto the simplex.  A simple back‑tracking line search
   controls the step size and tiny random jitter helps to escape plateaus.

2. **Linear‑programming refinement**.
   With the current shape `f` the bound `M = max(f*f)` is known.
   We solve the LP  

        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k,
                    g_i ≥ 0

   (the convolution with fixed `f` is linear in `g`).  The solution `g`
   is normalised to sum = 1 and mixed back into `f` using a relaxation
   factor `t`.  The LP is built as a sparse Toeplitz matrix, so it is
   fast for n≈1024 and only a few iterations are needed.

3. **Final hill‑climb** that moves a small fraction of mass from the
   index with the largest sub‑gradient to the one with the smallest.
   This polishes the solution to a local optimum.

All steps respect the 1000 s wall‑clock budget and the best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level utilities (independent of the optimisation loop)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The function follows the three‑phase scheme described in the module
    docstring.  The best sequence found before the wall‑clock deadline is
    returned (normalised to sum = 1).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a modest dimension makes many cheap iterations possible.
    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # Initialise from the global starter if available, otherwise use uniform.
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != n:
                # down‑sample / up‑sample by block averaging / zero‑padding
                if starter.size > n:
                    factor = starter.size // n
                    starter = starter[: factor * n].reshape(n, factor).mean(axis=1)
                else:                                   # pad with zeros
                    starter = np.pad(starter,
                                     (0, n - starter.size),
                                     mode='constant')
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                a = starter / s
            else:
                a = np.full(n, 1.0 / n)
        else:
            a = np.full(n, 1.0 / n)
    except Exception:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)               # ensure exact normalisation
    best_a = a.copy()
    best_score = _objective_score(a)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent (the bulk of the budget)
    # ------------------------------------------------------------------
    eta = 0.5                 # initial step size for the multiplicative update
    min_eta = 1e-5
    max_eta = 2.0
    it = 0
    # keep ~200 s for LP + hill‑climb
    phase1_deadline = deadline - 200.0
    while time.time() < phase1_deadline:
        it += 1
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # back‑tracking line search on the multiplicative update
        step = eta
        improved = False
        cur_score = _objective_score(a)
        for _ in range(8):
            a_candidate = a * np.exp(-step * grad)
            s = a_candidate.sum()
            if s == 0:
                a_candidate = a.copy()
            else:
                a_candidate /= s
            cand_score = _objective_score(a_candidate)
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                # enlarge step a little after success
                eta = min(step * 1.1, max_eta)
                break
            step *= 0.5

        if not improved:
            eta = max(step * 0.5, min_eta)
            # tiny random jitter to shake out of flat regions
            i, j = rng.choice(n, size=2, replace=False)
            eps = a[i] * (rng.random() * 1e-5 + 1e-9)
            if eps > 0:
                a[i] -= eps
                a[j] += eps
                a = _project_simplex(a)

        # track best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] EG it={it} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based refinement (a few iterations are cheap at n=1024)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        conv = _conv_self(a)
        M = conv.max()

        # Build the sparse Toeplitz matrix that represents (a * g)_k
        diag_data = []
        offsets = []
        for d in range(n):
            length = n - d
            diag_data.append(np.full(length, a[d]))
            offsets.append(d)
        A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')

        # Linear programme: maximise Σ g_i  <=> minimise -Σ g_i
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        try:
            res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                              bounds=bounds,
                              method='highs',
                              options={'presolve': True, 'time_limit': 15})
        except Exception as e:
            print(f"[{time.time() - start_time:.2f}s] LP error: {e}")
            break

        if not res.success:
            # If the LP fails (e.g. unbounded due to numerical zeros) break.
            print(f"[{time.time() - start_time:.2f}s] LP not successful")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g

        # Mix the new direction into the current iterate
        t = 0.5
        a = (1.0 - t) * a + t * g_norm
        a = _project_simplex(a)          # keep on the simplex

        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP mix score={best_score:.6f}")

        # If the LP does not improve the bound we stop the refinements.
        if cur_score >= best_score - 1e-8:
            break

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climb (small, focused moves)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        # Transfer up to 2 % of the source mass
        eps = a[src] * (rng.random() * 0.02 + 1e-9)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        prop_score = _objective_score(a_prop)
        if prop_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = prop_score
            print(f"[{time.time() - start_time:.2f}s] Hill climb improvement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (normalised to sum = 1)
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_a]
