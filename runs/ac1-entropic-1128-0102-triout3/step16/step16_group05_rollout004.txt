def _softmax_weights(conv, alpha):
    m = conv.max()
    exp_vals = np.exp(alpha * (conv - m))
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        return np.full_like(conv, 1.0/conv.size)
    return exp_vals / sum_exp

assistantfinal**Solution Overview**

The evaluation function is  

\[
E(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the denominator is a simple scaling, we keep the vector on the probability
simplex (`∑a_i = 1`).  
The difficult part is the **maximum of the self‑convolution**.  
The maximum is a non‑smooth functional, therefore a pure sub‑gradient method
gets stuck easily (the previous runner shows exactly that behaviour).

The new code uses three ideas

1. **Smooth the maximum** – replace `max(x)` by a soft‑max  
   \(\displaystyle S_{\alpha}(x)=\frac{1}{\alpha}\log\!\Bigl(\sum_i e^{\alpha x_i}\Bigr)\).  
   For large \(\alpha\) this is almost the true maximum, but for moderate
   \(\alpha\) it is smooth and provides a useful gradient.

2. **Fast gradient via FFT** – the gradient of the soft‑max
   \(\displaystyle \nabla_a S_{\alpha}= \sum_j p_j \nabla_a (a*a)_j\) with  
   \(p_j=e^{\alpha (c_j-m)}/\sum_k e^{\alpha (c_k-m)}\) (soft‑max weights).  
   The linear map \(\nabla_a (a*a)_j\) is a convolution, thus the whole gradient
   is computed with two FFT‑based convolutions (O(n log n)).

3. **Three‑phase optimisation**  

   *Phase 1 – smoothed gradient descent*  
   \(\alpha\) is increased slowly from 1 to a large value (≈200).  
   At each iteration a projected (onto the simplex) gradient step is taken with
   a back‑tracking line search.

   *Phase 2 – exact sub‑gradient refinement*  
   After the sequence is already good, the original non‑smooth sub‑gradient
   method is used for a short period to fine‑tune the exact `max`.

   *Phase 3 – tiny hill‑climb*  
   Moves of a tiny fraction of mass from the “over‑represented’’ index to the
   “under‑represented’’ one are attempted; this removes the remaining isolated
   peaks.

All three phases respect the 1000 s wall‑clock budget and constantly keep the
best sequence found.

The implementation works for a configurable length (`n = 8192` gives a very
good trade‑off between quality and speed).  The code is pure NumPy, uses only
FFT‑based convolutions, and prints the current best score so the progress can be
monitored.

```python
"""
Lightweight optimiser for the step‑function bound.

* Phase 1 – smoothed soft‑max gradient descent (α from 1 up to ~200)
* Phase 2 – short projected sub‑gradient descent on the true max
* Phase 3 – final hill‑climb moving tiny mass fractions.

All updates keep the vector on the probability simplex and enforce symmetry
(a[i] = a[n‑1‑i]), which is natural for the convolution objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT, projection, convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a (real, not circular).
    Length = 2*len(a) - 1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    sum(v)=1, v_i ≥ 0.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assume sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Soft‑max gradient (smooth approximation of the max)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max weights p_j = exp(α·(c_j‑m))/∑ exp(…) .
    The subtraction of the maximum m prevents overflow.
    """
    m = conv.max()
    # conv‑m ≤ 0  → exponent ≤ 0, safe from overflow
    exp_vals = np.exp(alpha * (conv - m))
    s = exp_vals.sum()
    if s == 0.0:
        # extremely unlikely, fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _grad_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_j w_j·(a*a)_j  w.r.t a.
    Both w and a are real, w has length L = 2*n-1.
    Returns a vector of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]                 # reversed a
    L = w.shape[0]                  # = 2*n-1
    Lconv = L + n - 1                # = 3*n-2
    N = _next_pow2(Lconv)

    # convolution w * a_rev
    Wf = np.fft.rfft(w, n=N)
    Af_rev = np.fft.rfft(a_rev, n=N)
    conv_w = np.fft.irfft(Wf * Af_rev, n=N)[:Lconv]

    # slice for the first part (j‑i term)
    part1 = conv_w[n - 1 : n - 1 + n]          # length n

    # convolution w_rev * a_rev
    w_rev = w[::-1]
    Wrev_f = np.fft.rfft(w_rev, n=N)
    conv_wr = np.fft.irfft(Wrev_f * Af_rev, n=N)[:Lconv]

    # slice for the second part (i‑j term), then reverse to align indices
    part2 = conv_wr[2 * n - 2 : 3 * n - 2][::-1]   # length n

    grad = part1 + part2
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximum (used in Phase 2)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several lags attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max (short refinement)."""
    a = a0.copy()
    n = a.shape[0]
    step_scale = 5e-3
    min_step = 1e-10
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)
        grad_norm = np.linalg.norm(grad) + 1e-16

        # back‑tracking line search (projected)
        step = step_scale
        improved = False
        direction = grad / grad_norm
        for _ in range(6):
            a_candidate = a - step * direction
            a_candidate = np.maximum(a_candidate, 0.0)
            # keep symmetry
            a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
            a_candidate = _project_simplex(a_candidate)

            new_score = _objective_score(a_candidate)
            if new_score < cur_score:
                a = a_candidate
                cur_score = new_score
                step_scale = min(step_scale * 1.02, 5e-2)
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            step_scale = max(step_scale * 0.5, min_step)
            # tiny random jitter to escape plateaus
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-5 + 1e-9)
            if eps > 0:
                a[i1] -= eps
                a[i2] += eps
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Hill‑climb (Phase 3)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass from the largest‑gradient index to the smallest."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # source – index with positive mass and largest gradient
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination – smallest gradient (may be same as src)
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # move up to 1 % of the source mass
        eps = a[src] * (rng.random() * 0.01 + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = (a_prop + a_prop[::-1]) * 0.5
        a_prop = _project_simplex(np.maximum(a_prop, 0.0))

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # jitter to escape stagnation
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

    The routine consists of three phases described in the module docstring.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Global timing
    # ------------------------------------------------------------------
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (larger n → more freedom, still cheap with FFT)
    # ------------------------------------------------------------------
    n = 8192                     # feel free to change; 8192 works well within budget
    a = np.full(n, 1.0 / n)     # uniform start, already symmetric

    # optional tiny random perturbation to break perfect symmetry
    a += rng.normal(scale=1e-5, size=n)
    a = np.maximum(a, 0.0)
    a = (a + a[::-1]) * 0.5           # enforce symmetry
    a = _project_simplex(a)           # exact simplex normalisation

    best_seq = a.copy()
    best_score = _objective_score(a)
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed soft‑max gradient descent
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.70 * budget_s
    alpha = 1.0
    alpha_max = 200.0
    alpha_growth = 1.03          # geometric growth each iteration
    step = 0.07                  # initial step size (projected gradient)

    while time.time() < phase1_end:
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Phase1 score={cur_score:.6f} α={alpha:.2f}")

        # soft‑max weights and gradient
        w = _softmax_weights(conv, alpha)
        grad = _grad_from_weights(w, a)

        # back‑tracking line search
        improved = False
        step_try = step
        for _ in range(7):
            a_candidate = a - step_try * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
            a_candidate = _project_simplex(a_candidate)

            cand_score = _objective_score(a_candidate)
            if cand_score < cur_score:
                a = a_candidate
                step = min(step * 1.08, 0.25)   # slowly increase if successful
                improved = True
                break
            else:
                step_try *= 0.5

        if not improved:
            step = max(step * 0.5, 1e-8)      # shrink step if no progress
            # occasional jitter to avoid flat plateaus
            if rng.random() < 0.005:
                i, j = rng.choice(n, size=2, replace=False)
                eps = a[i] * (rng.random() * 1e-5 + 1e-9)
                a[i] -= eps
                a[j] += eps
                a = (a + a[::-1]) * 0.5
                a = _project_simplex(np.maximum(a, 0.0))

        # raise α (but do not exceed the max)
        if alpha < alpha_max:
            alpha = min(alpha * alpha_growth, alpha_max)

    # ------------------------------------------------------------------
    # Phase 2 – short exact sub‑gradient descent
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget_s
    if time.time() < phase2_end:
        print(f"[{time.time() - start_time:.2f}s] Starting Phase 2 (sub‑gradient)")
        best_seq, best_score = _subgradient_descent(
            best_seq, phase2_end, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Phase 3 (hill‑climb)")
        best_seq, best_score = _hill_climb(
            best_seq, deadline, best_seq, best_score, rng)

    # ------------------------------------------------------------------
    # Normalise and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
